22 class LeafletMapExporter(out: ByteCounterStream) extends FeatureExporter with LazyLogging {
23 
24   private val coordMap = scala.collection.mutable.Map.empty[Coordinate, Int].withDefaultValue(0)
25 
26   private var writer: OutputStreamWriter = _
27   private var jsonWriter: JsonWriter = _
28   private var jsonSerializer: GeoJsonSerializer = _
29 
30   private var featureInfo = ""
31 
32   override def start(sft: SimpleFeatureType): Unit = {
33     featureInfo = LeafletMapExporter.getFeatureInfo(sft)
34     writer = new OutputStreamWriter(out, StandardCharsets.UTF_8)
35     jsonWriter = GeoJsonSerializer.writer(writer)
36     writer.write(LeafletMapExporter.IndexHead)
37     writer.write("var points = ")
38     writer.flush()
39     jsonSerializer = new GeoJsonSerializer(sft)
40     jsonSerializer.startFeatureCollection(jsonWriter)
41   }
42 
43   override def export(features: Iterator[SimpleFeature]): Option[Long] = {
44     var count = 0L
45     while (features.hasNext) {
46       val feature = features.next
47       jsonSerializer.write(jsonWriter, feature)
48       val geom = feature.getDefaultGeometry.asInstanceOf[Geometry]
49       if (geom != null) {
50         geom.getCoordinates.foreach(c => coordMap(c) += 1)
51       }
52       count += 1L
53     }
54     jsonWriter.flush()
55     Some(count)
56   }
57 
58   override def bytes: Long = out.bytes
59 
60   override def close(): Unit  = {
61     // finish writing GeoJson
62     if (writer != null) {
63       try {
64         jsonSerializer.endFeatureCollection(jsonWriter)
65         jsonWriter.flush()
66 
67         writer.write(";\n\n")
68         writer.write(featureInfo)
69         // Write Heatmap Data
70         writer.write("var heat = L.heatLayer([\n")
71         if (coordMap.isEmpty) {
72           logger.warn("No features were exported - the map will not render correctly")
73         } else {
74           val max = coordMap.maxBy(_._2)._2
75           val iter = coordMap.iterator
76           iter.take(1).foreach { case (coord, weight) =>
77             writer.write(s"        [${coord.y}, ${coord.x}, ${weight / max}]")
78           }
79           iter.foreach { case (coord, weight) =>
80             writer.write(s",\n        [${coord.y}, ${coord.x}, ${weight / max}]")
81           }
82         }
83         writer.write("\n    ], { radius: 25 });\n\n")
84         writer.write(LeafletMapExporter.IndexTail)
85       } finally {
86         jsonWriter.close() // also closes writer and output stream
87       }
88     }
89   }
90 }

109   private def getFeatureInfo(sft: SimpleFeatureType): String = {
110     val str = new StringBuilder()
111     str.append("    function onEachFeature(feature, layer) {\n")
112     if (sft == null) {
113       str.append("\n    };\n\n").toString
114     } else {
115       val attributes = sft.getAttributeDescriptors.asScala.map(_.getLocalName). collect {
116         case name if name != "geom" => s""""$name: " + feature.properties.$name"""
117       }
118       str.append("""        layer.bindPopup("ID: " + feature.id + "<br>" + """)
119       str.append(""""GEOM: " + feature.geometry.type + "[" + feature.geometry.coordinates + "]<br>" + """)
120       str.append(attributes.mkString(""" + "<br>" + """))

122     str.append(");\n    }\n\n")
