23   /**
24    * Create a filter
25    *
26    * @param in input
27    * @param root root node of the document being evaluated (for absolute path references)
28    * @param op the filter op
29    * @return
30    */
31   def apply(in: Input, root: ValuePointer, op: PathFilter.FilterOp): KryoJsonPathFilter =
32     new FilterBuilder(in, root)(op)

34   /**
35    * An expression in a filter - either a path, or a literal
36    */
37   private type Expression = ValuePointer => Any

39   /**
40    * Builder
41    *
42    * @param in input
43    * @param root root document path (for absolute path expressions)
44    */
45   private class FilterBuilder(in: Input, root: ValuePointer) {
46 
47     def apply(op: PathFilter.FilterOp): KryoJsonPathFilter = op match {
48       case EqualsOp(left, right)              => BinaryFilter.Equals(expression(left), expression(right))
49       case NotEqualsOp(left, right)           => BinaryFilter.NotEquals(expression(left), expression(right))
50       case LessThanOp(left, right)            => ComparisonFilter.LessThan(expression(left), expression(right))
51       case LessThanOrEqualsOp(left, right)    => ComparisonFilter.LessThanOrEquals(expression(left), expression(right))
52       case GreaterThanOp(left, right)         => ComparisonFilter.GreaterThan(expression(left), expression(right))
53       case GreaterThanOrEqualsOp(left, right) => ComparisonFilter.GreaterThanOrEquals(expression(left), expression(right))
54       case RegexOp(path, regex)               => RegexFilter(expression(path), regex)
55       case InOp(left, right)                  => ScalarArrayFilter.In(expression(left), expression(right))
56       case NotInOp(left, right)               => ScalarArrayFilter.NotIn(expression(left), expression(right))
57       case IsSubsetOp(left, right)            => ArrayFilter.Subset(expression(left), expression(right))
58       case AnyOfOp(left, right)               => ArrayFilter.AnyOf(expression(left), expression(right))
59       case NoneOfOp(left, right)              => ArrayFilter.NoneOf(expression(left), expression(right))
60       case EqualSizeOp(left, right)           => SizeFilter(expression(left), expression(right))
61       case IsEmptyOp(value, empty)            => IsEmptyFilter(expression(value), expression(empty))
62       case ExistsOp(value)                    => ExistsFilter(expression(value))
63       case AndFilterOp(filters)               => AndFilter(filters.map(this.apply))
64       case OrFilterOp(filters)                => OrFilter(filters.map(this.apply))
65       case NotFilterOp(filter)                => NotFilter(this.apply(filter))
66     }
67 
68     private def expression(exp: FilterExpression): Expression = exp match {
69       case p: PathExpression =>
70         if (p.path.elements.isEmpty) {
71           Constant(null)
72         } else if (p.absolute) {
73           Constant(new KryoJsonPath(in, root).deserialize(p.path))
74         } else {
75           Path(in, p.path)
76         }
77       case lit: LiteralExpression =>
78         Constant(lit.value)
79     }
80   }

101   /**
102    * Filter op between two arbitrary expressions
103    *
104    * @param left left expression
105    * @param right right expression
106    * @param op comparison op
107    */
108   private case class BinaryFilter(left: Expression, right: Expression, op: BinaryFilter.BinaryPredicate)
109       extends KryoJsonPathFilter {
110     override def apply(pointer: ValuePointer): Boolean = op(left(pointer), right(pointer))
111   }

113   private object BinaryFilter {
114     trait BinaryPredicate extends ((Any, Any) => Boolean) {
115       // constructor
116       def apply(left: Expression, right: Expression): BinaryFilter = BinaryFilter(left, right, this)
117     }
118 
119     object Equals extends BinaryPredicate {
120       override def apply(left: Any, right: Any): Boolean = left == right
121     }
122     object NotEquals extends BinaryPredicate {
123       override def apply(left: Any, right: Any): Boolean = left != right
124     }
125   }

127   /**
128    * Filter op between two 'comparable' expressions - expressions must both evaluate to either strings or numbers
129    *
130    * @param left left expression
131    * @param right right expression
132    * @param op comparison op
133    */
134   private case class ComparisonFilter(left: Expression, right: Expression, op: ComparisonFilter.ComparisonPredicate)
135     extends KryoJsonPathFilter {
136     override def apply(pointer: ValuePointer): Boolean = {
137       left(pointer) match {
138         case leftValue: Number =>
139           right(pointer) match {
140             case rightValue: Number => op(leftValue.doubleValue(), rightValue.doubleValue())
141             case _ => false
142           }
143         case leftValue: String =>
144           right(pointer) match {
145             case rightValue: String => op(leftValue, rightValue)
146             case _ => false
147           }
148         case _ => false
149       }
150     }
151   }
152 
153   private object ComparisonFilter {
154 
155     trait ComparisonPredicate {
156       // constructor
157       def apply(left: Expression, right: Expression): ComparisonFilter = ComparisonFilter(left, right, this)
158       def apply(left: Double, right: Double): Boolean
159       def apply(left: String, right: String): Boolean
160     }
161 
162     object LessThan extends ComparisonPredicate {
163       override def apply(left: Double, right: Double): Boolean = left < right
164       override def apply(left: String, right: String): Boolean = left < right
165     }
166     object LessThanOrEquals extends ComparisonPredicate {
167       override def apply(left: Double, right: Double): Boolean = left <= right
168       override def apply(left: String, right: String): Boolean = left <= right
169     }
170     object GreaterThan extends ComparisonPredicate {
171       override def apply(left: Double, right: Double): Boolean = left > right
172       override def apply(left: String, right: String): Boolean = left > right
173     }
174     object GreaterThanOrEquals extends ComparisonPredicate {
175       override def apply(left: Double, right: Double): Boolean = left >= right
176       override def apply(left: String, right: String): Boolean = left >= right
177     }
178   }

180   /**
181    * Regex filter op
182    *
183    * `=~` - left matches regular expression `[?(@.name =~ /foo.*?/i)]`
184    *
185    * @param left path expression, must evaluate to a string
186    * @param right regex to match
187    */
188   private case class RegexFilter(left: Expression, right: Pattern) extends KryoJsonPathFilter {
189     override def apply(pointer: ValuePointer): Boolean = {
190       left(pointer) match {
191         case s: String => right.matcher(s).matches()
192         case _ => false
193       }
194     }
195   }

197   private trait ArrayOps {
198     import scala.collection.JavaConverters._
199 
200     def toSeq(expression: Any): Option[Seq[Any]] = expression match {
201       case s: Seq[_] => Some(s)
202       case s: java.util.List[_] => Some(s.asScala.toSeq)
203       case _ => None
204     }
205 
206     def toString(expression: Any): Option[Seq[Any]] = expression match {
207       case s: String => Some(s)
208       case _ => None
209     }
210   }

212   /**
213    * Filter op between a scalar (number, string) and an array
214    *
215    * @param left scalar expression
216    * @param right array expression
217    * @param op comparison op
218    */
219   private case class ScalarArrayFilter(left: Expression, right: Expression, op: ScalarArrayFilter.ScalarArrayPredicate)
220     extends KryoJsonPathFilter with ArrayOps {
221     override def apply(pointer: ValuePointer): Boolean = toSeq(right(pointer)).exists(op(left(pointer), _))
222   }
223 
224   private object ScalarArrayFilter {
225 
226     trait ScalarArrayPredicate extends ((Any, Seq[Any]) => Boolean) {
227       // constructor
228       def apply(left: Expression, right: Expression): ScalarArrayFilter = ScalarArrayFilter(left, right, this)
229     }
230 
231     // in - left exists in right [?(@.size in ['S', 'M'])]
232     object In extends ScalarArrayPredicate {
233       override def apply(left: Any, right: Seq[Any]): Boolean = right.contains(left)
234     }
235 
236     // nin - left does not exists in right
237     object NotIn extends ScalarArrayPredicate {
238       override def apply(left: Any, right: Seq[Any]): Boolean = !right.contains(left)
239     }
240   }

242   /**
243    * Filter op between two arrays
244    *
245    * @param left left array
246    * @param right right array
247    * @param op comparison op
248    */
249   private case class ArrayFilter(left: Expression, right: Expression, op: ArrayFilter.ArrayPredicate)
250       extends KryoJsonPathFilter with ArrayOps {
251     override def apply(pointer: ValuePointer): Boolean = {
252       val seqs = for { s1 <- toSeq(left(pointer)); s2 <- toSeq(right(pointer)) } yield { op(s1, s2) }
253       seqs.getOrElse(false)
254     }
255   }
256 
257   private object ArrayFilter {
258 
259     trait ArrayPredicate extends ((Seq[Any], Seq[Any]) => Boolean) {
260       // constructor
261       def apply(left: Expression, right: Expression): ArrayFilter = ArrayFilter(left, right, this)
262     }
263 
264     // subsetof - left is a subset of right [?(@.sizes subsetof ['S', 'M', 'L'])]
265     object Subset extends ArrayPredicate {
266       override def apply(left: Seq[Any], right: Seq[Any]): Boolean = left.forall(right.contains)
267     }
268 
269     // anyof - left has an intersection with right [?(@.sizes anyof ['M', 'L'])]
270     object AnyOf extends ArrayPredicate {
271       override def apply(left: Seq[Any], right: Seq[Any]): Boolean = left.exists(right.contains)
272     }
273 
274     //noneof - left has no intersection with right [?(@.sizes noneof ['M', 'L'])]
275     object NoneOf extends ArrayPredicate {
276       override def apply(left: Seq[Any], right: Seq[Any]): Boolean = !left.exists(right.contains)
277     }
278   }

280   /**
281    * Size filter op
282    *
283    * `size` - size of left (array or string) should match right
284    *
285    * @param left left expression, array or string
286    * @param right right expression, array or string
287    */
288   private case class SizeFilter(left: Expression, right: Expression) extends KryoJsonPathFilter with ArrayOps {
289     override def apply(pointer: ValuePointer): Boolean = {
290       val leftValue = left(pointer)
291       val rightValue = right(pointer)
292       val arraySize = for { s1 <- toSeq(leftValue); s2 <- toSeq(rightValue) } yield { s1.size == s2.size }
293       lazy val stringSize = for { s1 <- toString(leftValue); s2 <- toString(rightValue) } yield { s1.size == s2.size }
294       arraySize.orElse(stringSize).getOrElse(false)
295     }
296   }
297 
298   /**
299    * Empty check filter op
300    *
301    * `empty` - left (array or string) should be empty
302    *
303    * @param value array or string
304    * @param empty empty or not empty
305    */
306   private case class IsEmptyFilter(value: Expression, empty: Expression) extends KryoJsonPathFilter {
307     override def apply(pointer: ValuePointer): Boolean = {
308       empty(pointer) match {
309         case b: Boolean =>
310           value(pointer) match {
311             case s: Seq[_] => s.isEmpty == b
312             case s: java.util.List[_] => s.isEmpty == b
313             case s: String => s.isEmpty == b
314             case null => b
315             case _ => false
316           }
317         case _ => false
318       }
319     }
320   }
321 
322   /**
323    * Filter op for whether expression exists (i.e. is not null)
324    *
325    * @param value any expression
326    */
327   private case class ExistsFilter(value: Expression) extends KryoJsonPathFilter {
328     override def apply(pointer: ValuePointer): Boolean = value(pointer) != null
329   }
330 
331   private case class AndFilter(filters: Seq[KryoJsonPathFilter]) extends KryoJsonPathFilter {
332     override def apply(pointer: ValuePointer): Boolean = filters.forall(_.apply(pointer))
333   }
334   private case class OrFilter(filters: Seq[KryoJsonPathFilter]) extends KryoJsonPathFilter {
335     override def apply(pointer: ValuePointer): Boolean = filters.exists(_.apply(pointer))
336   }
337   private case class NotFilter(filter: KryoJsonPathFilter) extends KryoJsonPathFilter {
338     override def apply(pointer: ValuePointer): Boolean = !filter.apply(pointer)
339   }
