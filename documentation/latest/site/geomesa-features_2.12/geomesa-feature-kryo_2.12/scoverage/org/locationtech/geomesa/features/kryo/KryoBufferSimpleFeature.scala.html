335   private class KryoBufferV3(serializer: KryoFeatureDeserialization, input: Input) extends KryoBufferDelegate {
336 
337     private var metadata: Metadata = _
338     private var transformer: Transformer = _
339 
340     override def reset(): Unit = metadata = Metadata(input) // reads count, size, nulls, etc
341 
342     override def setTransforms(schema: SimpleFeatureType, transforms: Array[Transform]): Unit = {
343       val indices = transforms.map {
344         case t: PropertyTransform => t.i
345         case t: RenameTransform => t.i
346         case _ => -1
347       }
348       if (indices.contains(-1)) {
349         transformer = new ReserializeTransformer(schema, transforms, serializer.options)
350       } else {
351         transformer = new BinaryTransformer(indices)
352       }
353     }
354 
355     override def id(): String = {
356       metadata.setIdPosition()
357       input.readString()
358     }
359 
360     override def getAttribute(index: Int): AnyRef = {
361       if (index >= metadata.count || metadata.nulls.contains(index)) { null } else {
362         metadata.setPosition(index)
363         serializer.readers(index).apply(input)
364       }
365     }
366 
367     override def getUserData: java.util.Map[AnyRef, AnyRef] = {
368       metadata.setUserDataPosition()
369       KryoUserDataSerialization.deserialize(input)
370     }
371 
372     override def getDateAsLong(index: Int): Long = {
373       if (index >= metadata.count || metadata.nulls.contains(index)) { 0L } else {
374         metadata.setPosition(index)
375         KryoLongReader.apply(input)
376       }
377     }
378 
379     override def getInput(index: Int): Option[Input] = {
380       if (index >= metadata.count || metadata.nulls.contains(index)) { None } else {
381         metadata.setPosition(index)
382         Some(input)
383       }
384     }
385 
386     override def transform(original: SimpleFeature): Array[Byte] = transformer.transform(original)
387 
388     // if we are just returning a subset of attributes, we can copy the bytes directly
389     private class BinaryTransformer(indices: Array[Int]) extends Transformer {
390 
391       private val positionsAndLengths = Array.ofDim[(Int, Int)](indices.length)
392 
393       override def transform(original: SimpleFeature): Array[Byte] = {
394         // +1 for version, +2 for count, +1 for size
395         var length = (metadata.size * (indices.length + 1)) + (IntBitSet.size(indices.length) * 4) + 4
396         val id = if (serializer.withoutId) { null } else {
397           val pos = metadata.setIdPosition() + metadata.offset
398           val id = input.readString()
399           length += input.position() - pos
400           id
401         }
402 
403         val nulls = IntBitSet(indices.length)
404 
405         // track the write position for copying attributes in the transformed array
406         var resultCursor = length
407 
408         var i = 0
409         while (i < indices.length) {
410           val index = indices(i) // the index of the attribute in the original feature
411           if (index >= metadata.count || metadata.nulls.contains(index)) { nulls.add(i) } else {
412             // read the offset and the subsequent offset to get the length
413             val pos = metadata.setPosition(index)
414             val len = metadata.setPosition(index + 1) - pos
415             length += len
416             positionsAndLengths(i) = (metadata.offset + pos, len)
417           }
418           i += 1
419         }
420 
421         // note: the input buffer is the raw buffer. we need to ensure that we use the
422         // offset into the raw buffer rather than the raw buffer directly
423         val buf = input.getBuffer
424 
425         val result = Array.ofDim[Byte](length)
426         val output = new Output(result, length)
427         output.writeByte(KryoFeatureSerializer.Version3)
428         output.writeShort(indices.length) // track the number of attributes
429         output.write(metadata.size)
430         i = 0
431         while (i < positionsAndLengths.length) {
432           // note: offset is always 4 here since we're using the full result array
433           if (metadata.size == 2) {
434             output.writeShort(resultCursor - 4)
435           } else {
436             output.writeInt(resultCursor - 4)
437           }
438           if (!nulls.contains(i)) {
439             val (pos, len) = positionsAndLengths(i)
440             System.arraycopy(buf, pos, result, resultCursor, len)
441             resultCursor += len
442           }
443           i += 1
444         }
445         // user data offset
446         if (metadata.size == 2) {
447           output.writeShort(resultCursor)
448         } else {
449           output.writeInt(resultCursor)
450         }
451 
452         // TODO user data?
453 
454         // write out nulls
455         nulls.serialize(output)
456 
457         // we're already positioned to write out the feature id
458         if (id != null) {
459           output.writeString(id)
460         }
461 
462         result
463       }
464     }
465   }

468     * Serialiation version 2 delegate
469     *
470     * @param serializer serializer
471     * @param input input
472     */
473   private class KryoBufferV2(serializer: KryoFeatureDeserialization, input: Input) extends KryoBufferDelegate {
474 
475     private val offsets = Array.ofDim[Int](serializer.out.getAttributeCount)
476     private var offset: Int = -1
477     private var startOfOffsets: Int = -1
478     private var missingAttributes: Boolean = false
479     private var userDataOffset: Int = -1
480 
481     private var reserializeTransform: Transformer = _
482     private var binaryTransform: Transformer = _
483 
484     override def reset(): Unit = {
485       // reset our offsets
486       offset = input.position() - 1 // we've already read the version byte
487       startOfOffsets = offset + input.readInt()
488       input.setPosition(startOfOffsets) // set to offsets start
489       var i = 0
490       while (i < offsets.length && input.position < input.limit) {
491         offsets(i) = offset + input.readInt(true)
492         i += 1
493       }
494       if (i < offsets.length) {
495         // attributes have been added to the sft since this feature was serialized
496         missingAttributes = true
497         while ({{ offsets(i) = -1; i += 1 }; i < offsets.length })()
498       } else {
499         missingAttributes = false
500       }
501       userDataOffset = input.position()
502     }
503 
504     override def setTransforms(schema: SimpleFeatureType, transforms: Array[Transform]): Unit = {
505       val indices = transforms.map {
506         case t: PropertyTransform => t.i
507         case t: RenameTransform => t.i
508         case _ => -1
509       }
510 
511       // transforms by evaluating the transform expressions and then serializing the resulting feature
512       // we use this for transform expressions and for data that was written using an old schema
513       reserializeTransform = new ReserializeTransformer(schema, transforms, serializer.options)
514       // if we are just returning a subset of attributes, we can copy the bytes directly
515       // and avoid creating new objects, reserializing, etc
516       binaryTransform =
517           if (indices.contains(-1)) { reserializeTransform } else { new BinaryTransformerV2(input, indices, offsets) }
518     }
519 
520     override def id(): String = {
521       input.setPosition(5)
522       input.readString()
523     }
524 
525     override def getAttribute(index: Int): AnyRef = {
526       val offset = offsets(index)
527       if (offset == -1) { null } else {
528         input.setPosition(offset)
529         serializer.readersV2(index)(input)
530       }
531     }
532 
533     override def getUserData: java.util.Map[AnyRef, AnyRef] = {
534       input.setPosition(userDataOffset)
535       KryoUserDataSerialization.deserialize(input)
536     }
537 
538     override def getDateAsLong(index: Int): Long = {
539       val offset = offsets(index)
540       if (offset == -1) { 0L } else {
541         input.setPosition(offset)
542         KryoFeatureDeserializationV2.LongReader.apply(input)
543       }
544     }
545 
546     override def getInput(index: Int): Option[Input] = {
547       val offset = offsets(index)
548       if (offset == -1) { None } else {
549         input.setPosition(offset)
550         Some(input)
551       }
552     }
553 
554     override def transform(original: SimpleFeature): Array[Byte] = {
555       // if attributes have been added to the sft, we have to reserialize to get the null serialized values
556       val transformer = if (missingAttributes) { reserializeTransform } else { binaryTransform }
557       transformer.transform(original)
558     }

560     /**
561       * Serialization version 2 binary transformer. Copies serialized attribute bytes directly without
562       * deserializing them
563       *
564       * @param input input
565       * @param indices transform indices
566       * @param offsets attribute offsets
567       */
568     private class BinaryTransformerV2(input: Input, indices: Array[Int], offsets: Array[Int]) extends Transformer {
569 
570       private val mutableOffsetsAndLength = Array.ofDim[(Int,Int)](indices.length)
571 
572       override def transform(original: SimpleFeature): Array[Byte] = {
573         // NOTE: the input buffer is the raw buffer. we need to ensure that we use the
574         // offset into the raw buffer rather than the raw buffer directly
575         val buf = input.getBuffer
576         var length = offsets(0) - offset // space for version, offset block and ID
577         var idx = 0
578         while (idx < mutableOffsetsAndLength.length) {
579           val i = indices(idx)
580           val el = (if (i < offsets.length - 1) { offsets(i + 1) } else { startOfOffsets }) - offsets(i)
581           length += el
582           mutableOffsetsAndLength(idx) = (offsets(i), el)
583           idx += 1
584         }
585 
586         val dst = Array.ofDim[Byte](length)
587         // copy the version, offset block and id
588         var dstPos = offsets(0) - offset
589         System.arraycopy(buf, offset, dst, 0, dstPos)
590         mutableOffsetsAndLength.foreach { case (o, l) =>
591           System.arraycopy(buf, o, dst, dstPos, l)
592           dstPos += l
593         }
594         // note that the offset block is incorrect - we couldn't use this in another lazy feature
595         // but the normal serializer doesn't care
596         dst
597       }
598     }
599   }

622         i += 1
623       }
624       serializer.serialize(sf)
625     }
