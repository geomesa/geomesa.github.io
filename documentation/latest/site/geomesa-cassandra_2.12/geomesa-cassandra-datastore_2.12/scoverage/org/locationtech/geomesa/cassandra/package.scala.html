17   object CassandraSystemProperties {
18     // schema changes are applied one at a time and have to be agreed on by the cluster (and dropping a table also
19     // snapshots it by default), so they can take much longer than the driver's default request timeout - a timeout
20     // part way through e.g. removing a schema leaves it partially deleted, so err on the side of waiting
21     val SchemaStatementTimeout: SystemProperty = SystemProperty("geomesa.cassandra.schema.timeout", "2 minutes")
22   }
23 
24   /**
25    * Creates a statement for a schema operation (create, drop or truncate table), which uses
26    * `geomesa.cassandra.schema.timeout` instead of the driver's default request timeout
27    *
28    * @param cql schema statement
29    * @return
30    */
31   def schemaStatement(cql: String): SimpleStatement = {
32     val timeout = CassandraSystemProperties.SchemaStatementTimeout.toDuration.get
33     SimpleStatement.newInstance(cql).setTimeout(java.time.Duration.ofMillis(timeout.toMillis))
34   }
