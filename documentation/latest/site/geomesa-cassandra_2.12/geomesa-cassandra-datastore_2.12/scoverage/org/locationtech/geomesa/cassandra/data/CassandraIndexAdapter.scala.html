88   override def createQueryPlan(strategy: QueryStrategy): CassandraQueryPlan = {
89     import org.locationtech.geomesa.index.conf.QueryHints.RichHints
90 
91     val processor = LocalProcessor(strategy.index.sft, strategy.hints, Some(ds.config.authProvider))
92 
93     val hints = strategy.hints
94 
95     if (strategy.keyRanges.isEmpty) { EmptyPlan(strategy, processor.reducer) } else {
96       val mapper = CassandraColumnMapper(strategy.index)
97       val selects = strategy.keyRanges.flatMap(mapper.select(_, strategy.tieredKeyRanges))
98       // an unfiltered full table scan covers every other range, so only run it to avoid returning duplicate results
99       val ranges = selects.find(_.clauses.isEmpty).map(Seq(_)).getOrElse(selects)
100       // some ranges can't be selected in cassandra without 'allow filtering' (e.g. open-ended time periods), which
101       // requires a full table scan - mark the strategy as a full table scan so that the query guards
102       // (i.e. blocking full table scans) are applied to it
103       val planStrategy = if (ranges.exists(r => r.clauses.isEmpty || r.allowFiltering)) {
104         strategy.copy(keyRanges = CassandraIndexAdapter.FullTableKeyRange)
105       } else {
106         strategy
107       }
108       val tables = strategy.index.getTablesForQuery(strategy.filter.filter)
109       val ks = Option(ds.session.getKeyspace.orElse(null)).map(_.asInternal())
110       val statements = tables.flatMap { table =>
111         ranges.map { r =>
112           val statement = CassandraIndexAdapter.statement(ks, table, r.clauses, r.allowFiltering)
113           // if not configured, the driver default consistency will be used
114           ds.config.queries.consistency.fold(statement)(statement.setConsistencyLevel)
115         }
116       }
117       val threads = ds.config.queries.threads
118       val project = hints.getProjection
119       StatementPlan(ds, planStrategy, tables, statements, threads, strategy.ecql, processor, project)
120     }

150   val MaxTimeToLive: Int = 20 * 365 * 24 * 60 * 60

152   /**
153    * Gets the table-level time-to-live for a feature type, based on its feature expiration. Only ingest-time
154    * expiration is supported, which maps to the Cassandra `default_time_to_live` table option.
155    *
156    * @param sft simple feature type
157    * @throws IllegalArgumentException if the feature expiration isn't supported
158    * @return time-to-live in seconds, or 0 if no expiration is configured
159    */
160   @throws(classOf[IllegalArgumentException])
161   def timeToLive(sft: SimpleFeatureType): Int = {
162     sft.getFeatureExpiration match {
163       case None => 0
164       case Some(IngestTimeExpiration(ttl)) =>
165         // round up so that features never expire before the configured time
166         val seconds = math.ceil(ttl.toMillis / 1000d).toLong
167         if (seconds > MaxTimeToLive) {
168           throw new IllegalArgumentException(
169             s"Invalid feature expiration '$ttl': Cassandra supports a time-to-live of up to 20 years " +
170                 s"($MaxTimeToLive seconds)")
171         }
172         seconds.toInt
173       case Some(expiration) =>
174         throw new IllegalArgumentException(
175           "Feature expiration based on an attribute is not supported by the Cassandra data store: " +
176               FeatureExpiration.unapply(expiration).getOrElse(expiration.toString))
177     }
178   }

198   def statement(
199       keyspace: Option[String],
200       table: String,
201       criteria: Seq[ColumnSelect],
202       allowFiltering: Boolean = false): SimpleStatement = {
203     val select = QueryBuilder.selectFrom(keyspace.orNull, table).all().where(
204       criteria.flatMap { c =>
205         val column = Relation.column(c.column.name)
206         if (c.start == null) {
207           if (c.end != null) {
208             if (c.endInclusive) {
209               Seq(column.isLessThanOrEqualTo(QueryBuilder.literal(c.end)))
210             } else {
211               Seq(column.isLessThan(QueryBuilder.literal(c.end)))
212             }
213           } else Seq.empty
214         } else if (c.end == null) {
215           if (c.startInclusive) {
216             Seq(column.isGreaterThanOrEqualTo(QueryBuilder.literal(c.start)))
217           } else {
218             Seq(column.isGreaterThan(QueryBuilder.literal(c.start)))
219           }
220         } else if (c.start == c.end) {
221           Seq(column.isEqualTo(QueryBuilder.literal(c.start)))
222         } else {
223           Seq(
224             if (c.startInclusive) {
225               column.isGreaterThanOrEqualTo(QueryBuilder.literal(c.start))
226             } else {
227               column.isGreaterThan(QueryBuilder.literal(c.start))
228             },
229             if (c.endInclusive) {
230               column.isLessThanOrEqualTo(QueryBuilder.literal(c.end))
231             } else {
232               column.isLessThan(QueryBuilder.literal(c.end))
233             }
234           )
235         }
236       }: _*
237     )
238     if (allowFiltering) { select.allowFiltering().build() } else { select.build() }
239   }

283     override protected def append(feature: WritableFeature, values: Array[RowKeyValue[_]]): Unit = {
284       i = 0
285       while (i < values.length) {
286         val (mapper, statement, _) = mappers(i)
287         values(i) match {
288           case kv: SingleRowKeyValue[_] =>
289             val bindings = mapper.bind(kv)
290             logger.trace(s"${statement.getQuery} : ${debug(bindings)}")
291             ds.session.execute(statement.bind(bindings: _*))

294             mkv.split.foreach { kv =>
295               val bindings = mapper.bind(kv)
296               logger.trace(s"${statement.getQuery} : ${debug(bindings)}")
297               ds.session.execute(statement.bind(bindings: _*))

299         }

301       }
302     }
