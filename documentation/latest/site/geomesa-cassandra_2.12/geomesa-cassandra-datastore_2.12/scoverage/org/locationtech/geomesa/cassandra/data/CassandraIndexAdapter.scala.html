126   def statement(keyspace: Option[String], table: String, criteria: Seq[ColumnSelect]): SimpleStatement = {
127     QueryBuilder.selectFrom(keyspace.orNull, table).all().where(
128       criteria.flatMap { c =>
129         val column = Relation.column(c.column.name)
130         if (c.start == null) {
131           if (c.end != null) {
132             if (c.endInclusive) {
133               Seq(column.isLessThanOrEqualTo(QueryBuilder.literal(c.end)))
134             } else {
135               Seq(column.isLessThan(QueryBuilder.literal(c.end)))
136             }
137           } else Seq.empty
138         } else if (c.end == null) {
139           if (c.startInclusive) {
140             Seq(column.isGreaterThanOrEqualTo(QueryBuilder.literal(c.start)))
141           } else {
142             Seq(column.isGreaterThan(QueryBuilder.literal(c.start)))
143           }
144         } else if (c.start == c.end) {
145           Seq(column.isEqualTo(QueryBuilder.literal(c.start)))
146         } else {
147           Seq(
148             if (c.startInclusive) {
149               column.isGreaterThanOrEqualTo(QueryBuilder.literal(c.start))
150             } else {
151               column.isGreaterThan(QueryBuilder.literal(c.start))
152             },
153             if (c.endInclusive) {
154               column.isLessThanOrEqualTo(QueryBuilder.literal(c.end))
155             } else {
156               column.isLessThan(QueryBuilder.literal(c.end))
157             }
158           )
159         }
160       }: _*
161     ).build()
162   }

188   class CassandraIndexWriter(
189       ds: CassandraDataStore,
190       indices: Seq[GeoMesaFeatureIndex[_, _]],
191       wrapper: FeatureWrapper[WritableFeature],
192       partition: Option[String]
193     ) extends BaseIndexWriter(indices, wrapper) with StrictLogging {
194 
195     private val mappers = indices.toArray.map { index =>
196       val mapper = CassandraColumnMapper(index)
197       // should always be writing to a single table here
198       val table = index.getTableName(partition)
199       val insert = mapper.insert(ds.session, table)
200       val delete = mapper.delete(ds.session, table)
201       (mapper, insert, delete)
202     }
203 
204     private var i = 0
205 
206     override protected def append(feature: WritableFeature, values: Array[RowKeyValue[_]]): Unit = {
207       i = 0
208       while (i < values.length) {
209         val (mapper, statement, _) = mappers(i)
210         values(i) match {
211           case kv: SingleRowKeyValue[_] =>
212             val bindings = mapper.bind(kv)
213             logger.trace(s"${statement.getQuery} : ${debug(bindings)}")
214             ds.session.execute(statement.bind(bindings: _*))
215 
216           case mkv: MultiRowKeyValue[_] =>
217             mkv.split.foreach { kv =>
218               val bindings = mapper.bind(kv)
219               logger.trace(s"${statement.getQuery} : ${debug(bindings)}")
220               ds.session.execute(statement.bind(bindings: _*))
221             }
222         }
223         i += 1
224       }
225     }
226 
227     override protected def update(
228         feature: WritableFeature,
229         values: Array[RowKeyValue[_]],
230         previous: WritableFeature,
231         previousValues: Array[RowKeyValue[_]]): Unit = {
232       delete(previous, previousValues)
233       append(feature, values)
234     }

236     override protected def delete(feature: WritableFeature, values: Array[RowKeyValue[_]]): Unit = {

255     }
