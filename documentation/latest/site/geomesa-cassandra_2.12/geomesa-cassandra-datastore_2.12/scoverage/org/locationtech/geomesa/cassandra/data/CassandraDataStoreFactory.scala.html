35   override def createDataStore(params: util.Map[String, _]): DataStore = {
36     val catalog = CatalogParam.lookup(params)
37     require(!catalog.contains("."), s"Invalid parameter '${CatalogParam.key}': catalog must not contain a dot. " +
38         s"Specify the keyspace separately using '${KeySpaceParam.key}' (-k) and the catalog using '${CatalogParam.key}' (-c)")
39 
40     val (cp, portString) = ContactPointParam.lookup(params).split(":") match {
41       case Array(one, two) => (one, two)
42       case parts => throw new IllegalArgumentException(s"Invalid parameter '${ContactPointParam.key}', " +
43           s"expected '<host>:<port>' but got '${parts.mkString(":")}'")
44     }
45     val port = try { portString.toInt } catch {
46       case NonFatal(_) => throw new IllegalArgumentException(s"Invalid parameter '${ContactPointParam.key}', " +
47           s"expected '<host>:<port>' but port is not a number: '$cp:$portString'")
48     }
49     val localDatacenter = LocalDatacenterParam.lookup(params)
50     val ks = KeySpaceParam.lookup(params)
51     val generateStats = GenerateStatsParam.lookup(params)
52     val audit = if (AuditQueriesParam.lookup(params)) {
53       Some(new AuditLogger("cassandra", AuditProvider.Loader.loadOrNone(params)))
54     } else {
55       None
56     }
57     val metrics = MetricsRegistryParam.lookupRegistry(params)
58 
59     val sessionBuilder = CqlSession.builder()
60       .addContactPoint(new InetSocketAddress(cp, port))
61       .withLocalDatacenter(localDatacenter)
62       .withKeyspace(ks)
63 
64     val user = UserNameParam.lookup(params)
65     val password = PasswordParam.lookup(params)
66     if (user != null && password != null) {
67       sessionBuilder.withAuthCredentials(user, password)
68     }
69 
70     val session = sessionBuilder.build()
71     val queries = CassandraQueryConfig(
72       threads = QueryThreadsParam.lookup(params),
73       timeout = QueryTimeoutParam.lookupOpt(params).map(_.toMillis),
74       looseBBox = LooseBBoxParam.lookup(params),
75       parallelPartitionScans = PartitionParallelScansParam.lookup(params)
76     )
77 
78     val authProvider = new DefaultAuthorizationsProvider(Seq.empty)
79 
80     val ns = Option(NamespaceParam.lookUp(params).asInstanceOf[String])
81 
82     val cfg = CassandraDataStoreConfig(catalog, generateStats, authProvider, audit, metrics, queries, ns)
83 
84     new CassandraDataStore(session, cfg)
85   }
