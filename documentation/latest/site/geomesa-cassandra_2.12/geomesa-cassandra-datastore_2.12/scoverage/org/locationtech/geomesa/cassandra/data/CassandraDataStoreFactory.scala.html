35   override def createDataStore(params: util.Map[String, _]): DataStore = {
36     val catalog = CatalogParam.lookup(params)
37     require(!catalog.contains("."), s"Invalid parameter '${CatalogParam.key}': catalog must not contain a dot. " +
38         s"Specify the keyspace separately using '${KeySpaceParam.key}' (-k) and the catalog using '${CatalogParam.key}' (-c)")
39 
40     val (cp, portString) = ContactPointParam.lookup(params).split(":") match {
41       case Array(one, two) => (one, two)
42       case parts => throw new IllegalArgumentException(s"Invalid parameter '${ContactPointParam.key}', " +
43           s"expected '<host>:<port>' but got '${parts.mkString(":")}'")
44     }
45     val port = try { portString.toInt } catch {
46       case NonFatal(_) => throw new IllegalArgumentException(s"Invalid parameter '${ContactPointParam.key}', " +
47           s"expected '<host>:<port>' but port is not a number: '$cp:$portString'")
48     }
49     val localDatacenter = LocalDatacenterParam.lookup(params)
50     val ks = KeySpaceParam.lookup(params)
51     // validated against the param enumerations
52     val consistency: Option[ConsistencyLevel] = QueryConsistencyParam.lookupOpt(params).map(DefaultConsistencyLevel.valueOf)
53     val generateStats = GenerateStatsParam.lookup(params)
54     val audit = if (AuditQueriesParam.lookup(params)) {
55       Some(new AuditLogger("cassandra", AuditProvider.Loader.loadOrNone(params)))
56     } else {
57       None
58     }
59     val metrics = MetricsRegistryParam.lookupRegistry(params)
60 
61     val sessionBuilder = CqlSession.builder()
62       .addContactPoint(new InetSocketAddress(cp, port))
63       .withLocalDatacenter(localDatacenter)
64       .withKeyspace(ks)
65 
66     val user = UserNameParam.lookup(params)
67     val password = PasswordParam.lookup(params)
68     if (user != null && password != null) {
69       sessionBuilder.withAuthCredentials(user, password)
70     }
71 
72     val session = sessionBuilder.build()
73     val queries = CassandraQueryConfig(
74       threads = QueryThreadsParam.lookup(params),
75       timeout = QueryTimeoutParam.lookupOpt(params).map(_.toMillis),
76       looseBBox = LooseBBoxParam.lookup(params),
77       parallelPartitionScans = PartitionParallelScansParam.lookup(params),
78       consistency = consistency
79     )
80 
81     val authProvider = new DefaultAuthorizationsProvider(Seq.empty)
82 
83     val ns = Option(NamespaceParam.lookUp(params).asInstanceOf[String])
84 
85     val cfg = CassandraDataStoreConfig(catalog, generateStats, authProvider, audit, metrics, queries, ns)
86 
87     new CassandraDataStore(session, cfg)
88   }
