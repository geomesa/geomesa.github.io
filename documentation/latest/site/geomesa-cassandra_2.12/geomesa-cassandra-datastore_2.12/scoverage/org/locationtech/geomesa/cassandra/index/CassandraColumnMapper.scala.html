23 object CassandraColumnMapper {
24 
25   import org.locationtech.geomesa.utils.geotools.RichSimpleFeatureType.RichSimpleFeatureType
26 
27   val FeatureIdColumnName = "fid"
28   val SimpleFeatureColumnName = "sf"
29 
30   val ShardColumn = NamedColumn("shard", 0, "tinyint", classOf[Byte], partition = true)
31 
32   def binColumn(i: Int): NamedColumn = NamedColumn("period", i, "smallint", classOf[Short], partition = true)
33   def zColumn(i: Int): NamedColumn = NamedColumn("z", i, "bigint", classOf[Long])
34 
35   def featureIdColumn(i: Int): NamedColumn = NamedColumn(FeatureIdColumnName, i, "text", classOf[String])
36   def featureColumn(i: Int): NamedColumn = NamedColumn(SimpleFeatureColumnName, i, "blob", classOf[ByteBuffer])
37 
38   def apply(index: GeoMesaFeatureIndex[_, _]): CassandraColumnMapper = {
39     index.name match {
40       case IdIndex.name                             => IdColumnMapper(index.sft)
41       case Z3Index.name | XZ3Index.name             => Z3ColumnMapper(index.sft.getZ3Shards)
42       case Z2Index.name | XZ2Index.name             => Z2ColumnMapper(index.sft.getZ2Shards)
43       case AttributeIndex.name if index.version > 7 => AttributeColumnMapper(index.sft.getAttributeShards)
44       case AttributeIndex.name                      => SharedAttributeColumnMapper
45       case _ => throw new IllegalArgumentException(s"Unexpected index: ${index.identifier}")
46     }
47   }
48 }

60   def insert(session: CqlSession, table: String): PreparedStatement = {
61     val cql = s"INSERT INTO $table (${columns.map(_.name).mkString(", ")}) " +
62         s"values (${Seq.fill(columns.length)("?").mkString(", ")})"
63     session.prepare(cql)
64   }
65 
66   def delete(session: CqlSession, table: String): PreparedStatement = {
67     val cols = columns.collect { case c if c.name != SimpleFeatureColumnName => c.name }
68     val cql = s"DELETE FROM $table WHERE ${cols.mkString("", " = ? and ", " = ?")}"
69     session.prepare(cql)
70   }
