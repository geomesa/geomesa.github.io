23 /**
24  * Cassandra-backed metadata
25  *
26  * @param session session
27  * @param catalog catalog table
28  * @param serializer serializer
29  * @param consistency consistency level for reads - if not specified, the driver default will be used
30  * @tparam T type param
31  */
32 class CassandraBackedMetadata[T](
33     val session: CqlSession,
34     val catalog: String,
35     val serializer: MetadataSerializer[T],
36     consistency: Option[ConsistencyLevel] = None
37   ) extends TableBasedMetadata[T] {
38 
39   // note: session gets closed by datastore dispose
40 
41   override protected def checkIfTableExists: Boolean = {
42     session.getKeyspace
43       .flatMap(session.getMetadata.getKeyspace)
44       .flatMap(_.getTable(catalog))
45       .isPresent
46   }
47 
48   override protected def createTable(): Unit =
49     session.execute(schemaStatement(s"CREATE TABLE IF NOT EXISTS $catalog (sft text, key text, value text, PRIMARY KEY ((sft), key))"))
50 
51   override protected def createEmptyBackup(timestamp: String): CassandraBackedMetadata[T] = {
52     val table = {
53       val full = s"${catalog}_${timestamp}_bak"
54       if (full.lengthCompare(CassandraIndexAdapter.TableNameLimit) <= 0) { full } else {
55         IndexAdapter.truncateTableName(full, CassandraIndexAdapter.TableNameLimit)
56       }
57     }
58     new CassandraBackedMetadata(session, table, serializer, consistency)
59   }
60 
61   override protected def write(typeName: String, rows: Seq[(String, Array[Byte])]): Unit = {
62     rows.foreach { case (key, value) =>
63       session.execute(s"INSERT INTO $catalog (sft, key, value) VALUES (?, ?, ?)",
64         typeName, key, new String(value, StandardCharsets.UTF_8))
65     }
66   }
67 
68   override protected def delete(typeName: String, keys: Seq[String]): Unit = {
69     keys.foreach { key =>
70       val query = QueryBuilder.deleteFrom(catalog)
71         .whereColumn("sft").isEqualTo(QueryBuilder.literal(typeName))
72         .whereColumn("key").isEqualTo(QueryBuilder.literal(key))
73         .build()
74       session.execute(query)
75     }
76   }
77 
78   override protected def scanValue(typeName: String, key: String): Option[Array[Byte]] = {
79     val query = QueryBuilder.selectFrom(catalog).column("value")
80       .whereColumn("sft").isEqualTo(QueryBuilder.literal(typeName))
81       .whereColumn("key").isEqualTo(QueryBuilder.literal(key))
82       .build()
83     val rows = session.execute(read(query)).all().asScala
84     if (rows.length < 1) { None } else {
85       Some(rows.head.getString("value").getBytes(StandardCharsets.UTF_8))
86     }
87   }
88 
89   override protected def scanValues(typeName: String, prefix: String): CloseableIterator[(String, Array[Byte])] = {
90     val select = QueryBuilder.selectFrom(catalog).columns("key", "value")
91       .whereColumn("sft").isEqualTo(QueryBuilder.literal(typeName))
92       .build()
93     val iter = session.execute(read(select)).all().iterator.asScala.map { row =>
94       (row.getString("key"), row.getString("value").getBytes(StandardCharsets.UTF_8))
95     }
96     if (prefix == null || prefix.isEmpty) {
97       CloseableIterator(iter)
98     } else {
99       CloseableIterator(iter.filter { case (k, _) => k.startsWith(prefix) })
100     }
101   }
102 
103   override protected def scanKeys(): CloseableIterator[(String, String)] = {
104     val select = QueryBuilder.selectFrom(catalog).columns("sft", "key").build()
105     val values = session.execute(read(select)).all().iterator.asScala.map(row => (row.getString("sft"), row.getString("key")))
106     CloseableIterator(values)
107   }
108 
109   // applies the configured consistency level to a read statement
110   private def read(statement: SimpleStatement): SimpleStatement = consistency.fold(statement)(statement.setConsistencyLevel)
111 }
