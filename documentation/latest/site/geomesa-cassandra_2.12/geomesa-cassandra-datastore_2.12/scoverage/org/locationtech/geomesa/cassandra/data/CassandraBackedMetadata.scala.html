21 class CassandraBackedMetadata[T](val session: CqlSession, val catalog: String, val serializer: MetadataSerializer[T])
22     extends TableBasedMetadata[T] {
23 
24   // note: session gets closed by datastore dispose
25 
26   override protected def checkIfTableExists: Boolean = {
27     session.getKeyspace
28       .flatMap(session.getMetadata.getKeyspace)
29       .flatMap(_.getTable(catalog))
30       .isPresent
31   }
32 
33   override protected def createTable(): Unit =
34     session.execute(s"CREATE TABLE IF NOT EXISTS $catalog (sft text, key text, value text, PRIMARY KEY ((sft), key))")
35 
36   override protected def createEmptyBackup(timestamp: String): CassandraBackedMetadata[T] = {
37     val table = {
38       val full = s"${catalog}_${timestamp}_bak"
39       if (full.lengthCompare(CassandraIndexAdapter.TableNameLimit) <= 0) { full } else {
40         IndexAdapter.truncateTableName(full, CassandraIndexAdapter.TableNameLimit)
41       }
42     }
43     new CassandraBackedMetadata(session, table, serializer)
44   }
45 
46   override protected def write(typeName: String, rows: Seq[(String, Array[Byte])]): Unit = {
47     rows.foreach { case (key, value) =>
48       session.execute(s"INSERT INTO $catalog (sft, key, value) VALUES (?, ?, ?)",
49         typeName, key, new String(value, StandardCharsets.UTF_8))
50     }
51   }
52 
53   override protected def delete(typeName: String, keys: Seq[String]): Unit = {
54     keys.foreach { key =>
55       val query = QueryBuilder.deleteFrom(catalog)
56         .whereColumn("sft").isEqualTo(QueryBuilder.literal(typeName))
57         .whereColumn("key").isEqualTo(QueryBuilder.literal(key))
58         .build()
59       session.execute(query)
60     }
61   }
62 
63   override protected def scanValue(typeName: String, key: String): Option[Array[Byte]] = {
64     val query = QueryBuilder.selectFrom(catalog).column("value")
65       .whereColumn("sft").isEqualTo(QueryBuilder.literal(typeName))
66       .whereColumn("key").isEqualTo(QueryBuilder.literal(key))
67       .build()
68     val rows = session.execute(query).all().asScala
69     if (rows.length < 1) { None } else {
70       Some(rows.head.getString("value").getBytes(StandardCharsets.UTF_8))
71     }
72   }
73 
74   override protected def scanValues(typeName: String, prefix: String): CloseableIterator[(String, Array[Byte])] = {
75     val select = QueryBuilder.selectFrom(catalog).columns("key", "value")
76       .whereColumn("sft").isEqualTo(QueryBuilder.literal(typeName))
77       .build()
78     val iter = session.execute(select).all().iterator.asScala.map { row =>
79       (row.getString("key"), row.getString("value").getBytes(StandardCharsets.UTF_8))
80     }
81     if (prefix == null || prefix.isEmpty) {
82       CloseableIterator(iter)
83     } else {
84       CloseableIterator(iter.filter { case (k, _) => k.startsWith(prefix) })
85     }
86   }
87 
88   override protected def scanKeys(): CloseableIterator[(String, String)] = {
89     val select = QueryBuilder.selectFrom(catalog).columns("sft", "key").build()
90     val values = session.execute(select).all().iterator.asScala.map(row => (row.getString("sft"), row.getString("key")))
91     CloseableIterator(values)
92   }
93 }
