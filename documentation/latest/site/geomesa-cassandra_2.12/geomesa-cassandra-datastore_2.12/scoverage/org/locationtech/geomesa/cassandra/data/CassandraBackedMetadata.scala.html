22 class CassandraBackedMetadata[T](val session: CqlSession, val catalog: String, val serializer: MetadataSerializer[T])
23     extends TableBasedMetadata[T] {
24 
25   // note: session gets closed by datastore dispose
26 
27   override protected def checkIfTableExists: Boolean = {
28     session.getKeyspace
29       .flatMap(session.getMetadata.getKeyspace)
30       .flatMap(_.getTable(catalog))
31       .isPresent
32   }
33 
34   override protected def createTable(): Unit =
35     session.execute(schemaStatement(s"CREATE TABLE IF NOT EXISTS $catalog (sft text, key text, value text, PRIMARY KEY ((sft), key))"))
36 
37   override protected def createEmptyBackup(timestamp: String): CassandraBackedMetadata[T] = {
38     val table = {
39       val full = s"${catalog}_${timestamp}_bak"
40       if (full.lengthCompare(CassandraIndexAdapter.TableNameLimit) <= 0) { full } else {
41         IndexAdapter.truncateTableName(full, CassandraIndexAdapter.TableNameLimit)
42       }
43     }
44     new CassandraBackedMetadata(session, table, serializer)
45   }
46 
47   override protected def write(typeName: String, rows: Seq[(String, Array[Byte])]): Unit = {
48     rows.foreach { case (key, value) =>
49       session.execute(s"INSERT INTO $catalog (sft, key, value) VALUES (?, ?, ?)",
50         typeName, key, new String(value, StandardCharsets.UTF_8))
51     }
52   }
53 
54   override protected def delete(typeName: String, keys: Seq[String]): Unit = {
55     keys.foreach { key =>
56       val query = QueryBuilder.deleteFrom(catalog)
57         .whereColumn("sft").isEqualTo(QueryBuilder.literal(typeName))
58         .whereColumn("key").isEqualTo(QueryBuilder.literal(key))
59         .build()
60       session.execute(query)
61     }
62   }
63 
64   override protected def scanValue(typeName: String, key: String): Option[Array[Byte]] = {
65     val query = QueryBuilder.selectFrom(catalog).column("value")
66       .whereColumn("sft").isEqualTo(QueryBuilder.literal(typeName))
67       .whereColumn("key").isEqualTo(QueryBuilder.literal(key))
68       .build()
69     val rows = session.execute(query).all().asScala
70     if (rows.length < 1) { None } else {
71       Some(rows.head.getString("value").getBytes(StandardCharsets.UTF_8))
72     }
73   }
74 
75   override protected def scanValues(typeName: String, prefix: String): CloseableIterator[(String, Array[Byte])] = {
76     val select = QueryBuilder.selectFrom(catalog).columns("key", "value")
77       .whereColumn("sft").isEqualTo(QueryBuilder.literal(typeName))
78       .build()
79     val iter = session.execute(select).all().iterator.asScala.map { row =>
80       (row.getString("key"), row.getString("value").getBytes(StandardCharsets.UTF_8))
81     }
82     if (prefix == null || prefix.isEmpty) {
83       CloseableIterator(iter)
84     } else {
85       CloseableIterator(iter.filter { case (k, _) => k.startsWith(prefix) })
86     }
87   }
88 
89   override protected def scanKeys(): CloseableIterator[(String, String)] = {
90     val select = QueryBuilder.selectFrom(catalog).columns("sft", "key").build()
91     val values = session.execute(select).all().iterator.asScala.map(row => (row.getString("sft"), row.getString("key")))
92     CloseableIterator(values)
93   }
94 }
