60   override protected def preSchemaUpdate(sft: SimpleFeatureType, previous: SimpleFeatureType): Unit = {
61     val rename = sft.getUserData.remove(SimpleFeatureTypes.Configs.UpdateRenameTables)
62     if (FastConverter.convert(rename, classOf[java.lang.Boolean]) == java.lang.Boolean.TRUE) {
63       logger.warn("Cassandra does not support renaming tables - disabling rename option")
64     }
65     sft.getUserData.put(SimpleFeatureTypes.Configs.UpdateRenameTables, java.lang.Boolean.FALSE)
66     super.preSchemaUpdate(sft, previous)
67     // validate any feature expiration - only if changed, so that existing schemas with an unsupported expiration
68     // can still be updated
69     if (sft.getFeatureExpiration != previous.getFeatureExpiration) {
70       // noinspection ScalaDeprecation
71       if (CassandraIndexAdapter.timeToLive(sft) > 0 && sft.isTableSharing) {
72         throw new IllegalArgumentException("Feature expiration is not supported for schemas with shared tables")
73       }
74     }
75   }

77   override protected def onSchemaUpdated(sft: SimpleFeatureType, previous: SimpleFeatureType): Unit = {
78     super.onSchemaUpdated(sft, previous)
79     // update the time-to-live of existing tables - note that this only affects rows written afterwards
80     if (sft.getFeatureExpiration != previous.getFeatureExpiration) {
81       val ttl = CassandraIndexAdapter.timeToLive(sft)
82       val keyspace = session.getKeyspace.flatMap(session.getMetadata.getKeyspace)
83       manager.indices(sft).flatMap(_.getTableNames()).distinct.foreach { table =>
84         if (keyspace.flatMap(_.getTable(table)).isPresent) {
85           val alter = s"ALTER TABLE $table WITH default_time_to_live = $ttl"
86           logger.debug(alter)
87           session.execute(schemaStatement(alter))
88         }
89       }
90     }
91   }

93   override protected def transitionIndices(sft: SimpleFeatureType): Unit = {
94     val dtg = sft.getDtgField.toSeq
95     val geom = Option(sft.getGeomField).toSeq
96 
97     val indices = Seq.newBuilder[IndexId]
98     val tableNameKeys = Seq.newBuilder[(String, String)]
99 
100     sft.getIndices.foreach {
101       case id if id.name == IdIndex.name =>
102         require(id.version == 1, s"Expected index version of 1 but got: $id")
103         indices += id.copy(version = 3)
104         tableNameKeys += ((s"table.${IdIndex.name}.v1", s"table.${IdIndex.name}.v3"))
105 
106       case id if id.name == Z3Index.name =>
107         require(id.version <= 2, s"Expected index version of 1 or 2 but got: $id")
108         indices += id.copy(attributes = geom ++ dtg, version = id.version + 3)
109         tableNameKeys += ((s"table.${Z3Index.name}.v${id.version}", s"table.${Z3Index.name}.v${id.version + 3}"))
110 
111       case id if id.name == XZ3Index.name =>
112         require(id.version == 1, s"Expected index version of 1 but got: $id")
113         indices += id.copy(attributes = geom ++ dtg)
114 
115       case id if id.name == Z2Index.name =>
116         require(id.version <= 2, s"Expected index version of 1 or 2 but got: $id")
117         indices += id.copy(attributes = geom, version = id.version + 2)
118         tableNameKeys += ((s"table.${Z2Index.name}.v${id.version}", s"table.${Z2Index.name}.v${id.version + 2}"))
119 
120       case id if id.name == XZ2Index.name =>
121         require(id.version == 1, s"Expected index version of 1 but got: $id")
122         indices += id.copy(attributes = geom)
123 
124       case id if id.name == AttributeIndex.name =>
125         require(id.version <= 2, s"Expected index version of 1 or 2 but got: $id")
126         val fields = geom ++ dtg
127         sft.getAttributeDescriptors.asScala.foreach { d =>
128           val index = d.getUserData.remove(AttributeOptions.OptIndex).asInstanceOf[String]
129           if (index == null || index.equalsIgnoreCase("false") || index.equalsIgnoreCase(IndexCoverage.NONE.toString)) {
130             // no-op
131           } else if (java.lang.Boolean.valueOf(index) || index.equalsIgnoreCase(IndexCoverage.FULL.toString) ||
132               index.equalsIgnoreCase(IndexCoverage.JOIN.toString)) {
133             indices += id.copy(attributes = Seq(d.getLocalName) ++ fields, version = id.version + 4)
134           } else {
135             throw new IllegalStateException(s"Expected an index coverage or boolean but got: $index")
136           }
137         }
138         tableNameKeys ++=
139             Seq(s"table.${AttributeIndex.name}.v${id.version}", "tables.idx.attr.name")
140                 .map((_, s"table.${AttributeIndex.name}.v${id.version + 4}"))
141     }
142 
143     sft.setIndices(indices.result)
144 
145     // update metadata keys for tables
146     tableNameKeys.result.foreach { case (from, to) =>
147       metadata.scan(sft.getTypeName, from, cache = false).foreach { case (key, name) =>
148         metadata.insert(sft.getTypeName, to + key.substring(from.length), name)
149         metadata.remove(sft.getTypeName, key)
150       }
151     }
152   }
