33 class CassandraDataStore(val session: CqlSession, override val config: CassandraDataStoreConfig)
34     extends GeoMesaDataStore[CassandraDataStore](config) with LocalLocking {
35

40   override val metadata: GeoMesaMetadata[String] =
41     new CassandraBackedMetadata(session, config.catalog, MetadataStringSerializer, config.queries.consistency)
42 
43   override val adapter: CassandraIndexAdapter = new CassandraIndexAdapter(this)
44 
45   override val stats: GeoMesaStats = new RunnableStats(this)
46 
47   override def getQueryPlan(query: Query, index: Option[String], explainer: Explainer): Seq[CassandraQueryPlan] =
48     super.getQueryPlan(query, index, explainer).asInstanceOf[Seq[CassandraQueryPlan]]
49 
50   override def delete(): Unit = {
51     val tables = getTypeNames.flatMap(getAllIndexTableNames).distinct :+ config.catalog
52     tables.toList.map(t => CachedThreadPool.submit(() => session.execute(schemaStatement(s"drop table $t")))).foreach(_.get)
53   }
54 
55   override protected def preSchemaUpdate(sft: SimpleFeatureType, previous: SimpleFeatureType): Unit = {
56     val rename = sft.getUserData.remove(SimpleFeatureTypes.Configs.UpdateRenameTables)
57     if (FastConverter.convert(rename, classOf[java.lang.Boolean]) == java.lang.Boolean.TRUE) {
58       logger.warn("Cassandra does not support renaming tables - disabling rename option")
59     }
60     sft.getUserData.put(SimpleFeatureTypes.Configs.UpdateRenameTables, java.lang.Boolean.FALSE)
61     super.preSchemaUpdate(sft, previous)
62   }
63 
64   override protected def transitionIndices(sft: SimpleFeatureType): Unit = {
65     val dtg = sft.getDtgField.toSeq
66     val geom = Option(sft.getGeomField).toSeq
67 
68     val indices = Seq.newBuilder[IndexId]
69     val tableNameKeys = Seq.newBuilder[(String, String)]
70 
71     sft.getIndices.foreach {
72       case id if id.name == IdIndex.name =>
73         require(id.version == 1, s"Expected index version of 1 but got: $id")
74         indices += id.copy(version = 3)
75         tableNameKeys += ((s"table.${IdIndex.name}.v1", s"table.${IdIndex.name}.v3"))
76 
77       case id if id.name == Z3Index.name =>
78         require(id.version <= 2, s"Expected index version of 1 or 2 but got: $id")
79         indices += id.copy(attributes = geom ++ dtg, version = id.version + 3)
80         tableNameKeys += ((s"table.${Z3Index.name}.v${id.version}", s"table.${Z3Index.name}.v${id.version + 3}"))
81 
82       case id if id.name == XZ3Index.name =>
83         require(id.version == 1, s"Expected index version of 1 but got: $id")
84         indices += id.copy(attributes = geom ++ dtg)
85 
86       case id if id.name == Z2Index.name =>
87         require(id.version <= 2, s"Expected index version of 1 or 2 but got: $id")
88         indices += id.copy(attributes = geom, version = id.version + 2)
89         tableNameKeys += ((s"table.${Z2Index.name}.v${id.version}", s"table.${Z2Index.name}.v${id.version + 2}"))
90 
91       case id if id.name == XZ2Index.name =>
92         require(id.version == 1, s"Expected index version of 1 but got: $id")
93         indices += id.copy(attributes = geom)
94 
95       case id if id.name == AttributeIndex.name =>
96         require(id.version <= 2, s"Expected index version of 1 or 2 but got: $id")
97         val fields = geom ++ dtg
98         sft.getAttributeDescriptors.asScala.foreach { d =>
99           val index = d.getUserData.remove(AttributeOptions.OptIndex).asInstanceOf[String]
100           if (index == null || index.equalsIgnoreCase("false") || index.equalsIgnoreCase(IndexCoverage.NONE.toString)) {
101             // no-op
102           } else if (java.lang.Boolean.valueOf(index) || index.equalsIgnoreCase(IndexCoverage.FULL.toString) ||
103               index.equalsIgnoreCase(IndexCoverage.JOIN.toString)) {
104             indices += id.copy(attributes = Seq(d.getLocalName) ++ fields, version = id.version + 4)
105           } else {
106             throw new IllegalStateException(s"Expected an index coverage or boolean but got: $index")
107           }
108         }
109         tableNameKeys ++=
110             Seq(s"table.${AttributeIndex.name}.v${id.version}", "tables.idx.attr.name")
111                 .map((_, s"table.${AttributeIndex.name}.v${id.version + 4}"))
112     }
113 
114     sft.setIndices(indices.result)
115 
116     // update metadata keys for tables
117     tableNameKeys.result.foreach { case (from, to) =>
118       metadata.scan(sft.getTypeName, from, cache = false).foreach { case (key, name) =>
119         metadata.insert(sft.getTypeName, to + key.substring(from.length), name)
120         metadata.remove(sft.getTypeName, key)
121       }
122     }
123   }
