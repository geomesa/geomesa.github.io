55   override protected def preSchemaCreate(sft: SimpleFeatureType): Unit = {
56     // feature ids are stored in a text column, so uuids can't be stored in their encoded (binary) form
57     if (sft.isUuidEncoded) {
58       logger.warn("UUID feature ID encoding is not supported by the Cassandra data store - disabling UUID encoding")
59       sft.getUserData.put(SimpleFeatureTypes.Configs.FidsAreUuidEncoded, "false")
60     }
61     super.preSchemaCreate(sft)
62     CassandraIndexAdapter.timeToLive(sft) // validate any feature expiration
63   }

65   override protected def preSchemaUpdate(sft: SimpleFeatureType, previous: SimpleFeatureType): Unit = {
66     val rename = sft.getUserData.remove(SimpleFeatureTypes.Configs.UpdateRenameTables)
67     if (FastConverter.convert(rename, classOf[java.lang.Boolean]) == java.lang.Boolean.TRUE) {
68       logger.warn("Cassandra does not support renaming tables - disabling rename option")
69     }
70     sft.getUserData.put(SimpleFeatureTypes.Configs.UpdateRenameTables, java.lang.Boolean.FALSE)
71     super.preSchemaUpdate(sft, previous)
72     // validate any feature expiration - only if changed, so that existing schemas with an unsupported expiration
73     // can still be updated
74     if (sft.getFeatureExpiration != previous.getFeatureExpiration) {
75       // noinspection ScalaDeprecation
76       if (CassandraIndexAdapter.timeToLive(sft) > 0 && sft.isTableSharing) {
77         throw new IllegalArgumentException("Feature expiration is not supported for schemas with shared tables")
78       }
79     }
80   }

82   override protected def onSchemaUpdated(sft: SimpleFeatureType, previous: SimpleFeatureType): Unit = {
83     super.onSchemaUpdated(sft, previous)
84     // update the time-to-live of existing tables - note that this only affects rows written afterwards
85     if (sft.getFeatureExpiration != previous.getFeatureExpiration) {
86       val ttl = CassandraIndexAdapter.timeToLive(sft)
87       val keyspace = session.getKeyspace.flatMap(session.getMetadata.getKeyspace)
88       manager.indices(sft).flatMap(_.getTableNames()).distinct.foreach { table =>
89         if (keyspace.flatMap(_.getTable(table)).isPresent) {
90           val alter = s"ALTER TABLE $table WITH default_time_to_live = $ttl"
91           logger.debug(alter)
92           session.execute(schemaStatement(alter))
93         }
94       }
95     }
96   }

98   override protected def transitionIndices(sft: SimpleFeatureType): Unit = {
99     val dtg = sft.getDtgField.toSeq
100     val geom = Option(sft.getGeomField).toSeq
101 
102     val indices = Seq.newBuilder[IndexId]
103     val tableNameKeys = Seq.newBuilder[(String, String)]
104 
105     sft.getIndices.foreach {
106       case id if id.name == IdIndex.name =>
107         require(id.version == 1, s"Expected index version of 1 but got: $id")
108         indices += id.copy(version = 3)
109         tableNameKeys += ((s"table.${IdIndex.name}.v1", s"table.${IdIndex.name}.v3"))
110 
111       case id if id.name == Z3Index.name =>
112         require(id.version <= 2, s"Expected index version of 1 or 2 but got: $id")
113         indices += id.copy(attributes = geom ++ dtg, version = id.version + 3)
114         tableNameKeys += ((s"table.${Z3Index.name}.v${id.version}", s"table.${Z3Index.name}.v${id.version + 3}"))
115 
116       case id if id.name == XZ3Index.name =>
117         require(id.version == 1, s"Expected index version of 1 but got: $id")
118         indices += id.copy(attributes = geom ++ dtg)
119 
120       case id if id.name == Z2Index.name =>
121         require(id.version <= 2, s"Expected index version of 1 or 2 but got: $id")
122         indices += id.copy(attributes = geom, version = id.version + 2)
123         tableNameKeys += ((s"table.${Z2Index.name}.v${id.version}", s"table.${Z2Index.name}.v${id.version + 2}"))
124 
125       case id if id.name == XZ2Index.name =>
126         require(id.version == 1, s"Expected index version of 1 but got: $id")
127         indices += id.copy(attributes = geom)
128 
129       case id if id.name == AttributeIndex.name =>
130         require(id.version <= 2, s"Expected index version of 1 or 2 but got: $id")
131         val fields = geom ++ dtg
132         sft.getAttributeDescriptors.asScala.foreach { d =>
133           val index = d.getUserData.remove(AttributeOptions.OptIndex).asInstanceOf[String]
134           if (index == null || index.equalsIgnoreCase("false") || index.equalsIgnoreCase(IndexCoverage.NONE.toString)) {
135             // no-op
136           } else if (java.lang.Boolean.valueOf(index) || index.equalsIgnoreCase(IndexCoverage.FULL.toString) ||
137               index.equalsIgnoreCase(IndexCoverage.JOIN.toString)) {
138             indices += id.copy(attributes = Seq(d.getLocalName) ++ fields, version = id.version + 4)
139           } else {
140             throw new IllegalStateException(s"Expected an index coverage or boolean but got: $index")
141           }
142         }
143         tableNameKeys ++=
144             Seq(s"table.${AttributeIndex.name}.v${id.version}", "tables.idx.attr.name")
145                 .map((_, s"table.${AttributeIndex.name}.v${id.version + 4}"))
146     }
147 
148     sft.setIndices(indices.result)
149 
150     // update metadata keys for tables
151     tableNameKeys.result.foreach { case (from, to) =>
152       metadata.scan(sft.getTypeName, from, cache = false).foreach { case (key, name) =>
153         metadata.insert(sft.getTypeName, to + key.substring(from.length), name)
154         metadata.remove(sft.getTypeName, key)
155       }
156     }
157   }
