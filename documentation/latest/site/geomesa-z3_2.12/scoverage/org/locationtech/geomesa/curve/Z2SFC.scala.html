18 /**
19   * z2 space-filling curve
20   *
21   * @param precision number of bits used per dimension - note sum must be less than 64
22   */
23 class Z2SFC(val precision: Int) extends SpaceFillingCurve {
24 
25   private val hexFormat = HexFormat.of()
26 
27   val lon: NormalizedDimension = NormalizedLon(precision)
28   val lat: NormalizedDimension = NormalizedLat(precision)
29 
30   override def index(x: Double, y: Double, lenient: Boolean = false): Long = {
31     try {
32       require(x >= lon.min && x <= lon.max && y >= lat.min && y <= lat.max,
33         s"Value(s) out of bounds ([${lon.min},${lon.max}], [${lat.min},${lat.max}]): $x, $y")
34       Z2(lon.normalize(x), lat.normalize(y)).z
35     } catch {
36       case _: IllegalArgumentException if lenient => lenientIndex(x, y)
37     }
38   }
39 
40   protected def lenientIndex(x: Double, y: Double): Long = {
41     val bx = if (x < lon.min) { lon.min } else if (x > lon.max) { lon.max } else { x }
42     val by = if (y < lat.min) { lat.min } else if (y > lat.max) { lat.max } else { y }
43     Z2(lon.normalize(bx), lat.normalize(by)).z
44   }
45 
46   override def invert(z: Long): (Double, Double) = {
47     val (x, y) = Z2(z).decode
48     (lon.denormalize(x), lat.denormalize(y))
49   }
50 
51   override def ranges(xy: Seq[(Double, Double, Double, Double)],
52                       precision: Int,
53                       maxRanges: Option[Int]): Seq[IndexRange] = {
54     val zbounds = xy.map { case (xmin, ymin, xmax, ymax) => ZRange(index(xmin, ymin), index(xmax, ymax)) }
55     Z2.zranges(zbounds.toArray, precision, maxRanges)
56   }
57 
58   /**
59    * Encodes a z value into hex, bit-shifting left as necessary (2 unused leading bits) so that prefix matching works
60    *
61    * @param x x value
62    * @param y y value
63    * @return hex-encoded string
64    */
65   def hexEncode(x: Double, y: Double): String = hexEncode(index(x, y))
66 
67   /**
68    * Encodes a z value into hex, bit-shifting left as necessary (2 unused leading bits) so that prefix matching works
69    *
70    * @param z z value
71    * @return hex-encoded string
72    */
73   def hexEncode(z: Long): String = hexFormat.toHexDigits(z << 2)
74 
75   /**
76    * Decodes a previously encoded z value from hex
77    *
78    * @param hex hex string from `hexEncode`
79    * @return original z value
80    */
81   def hexDecode(hex: String): Long = HexFormat.fromHexDigitsToLong(hex) >>> 2
82 }
