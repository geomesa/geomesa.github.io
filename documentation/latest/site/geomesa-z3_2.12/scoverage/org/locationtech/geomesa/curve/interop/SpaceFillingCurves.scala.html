19 /**
20  * Java interop-friendly facade over the Z2SFC and XZ2SFC space-filling curves.
21  *
22  * Inputs are clamped to the WGS84 domain, matching the curves' lenient
23  * handling of out-of-bounds coordinates.
24  *
25  * Hex outputs delegate to `Z2SFC.hexEncode`/`XZ2SFC.hexEncode`, and
26  * left-align the significant bits so lexicographic (truncate-prefix)
27  * comparison over the encoded strings matches numeric comparison of the
28  * underlying index values.
29  */
30 object SpaceFillingCurves {
31 
32   /**
33    * A contiguous, inclusive range of hex-encoded curve values —
34    * Java-interop stand-in for Scala's `IndexRange`.
35    *
36    * @param lower inclusive lower bound
37    * @param upper inclusive upper bound
38    */
39   case class HexRange(lower: String, upper: String)
40 
41   private def clampLon(x: Double): Double = math.min(lon.max, math.max(lon.min, x))
42   private def clampLat(y: Double): Double = math.min(lat.max, math.max(lat.min, y))
43 
44   /**
45    * Z2 cell index for a single (lon, lat) point at the default Z2SFC precision
46    * (31 bits/axis, 62-bit positive Long).
47    *
48    * @param lon longitude, clamped to [-180, 180]
49    * @param lat latitude, clamped to [-90, 90]
50    * @return the Z2SFC index value
51    */
52   def z2Index(lon: Double, lat: Double): Long =
53     Z2SFC.index(clampLon(lon), clampLat(lat))
54 
55   /**
56    * Z2 cell value for a single (lon, lat) point, hex-encoded via
57    * `Z2SFC.hexEncode` (left-shifted).
58    *
59    * @param lon longitude, clamped to [-180, 180]
60    * @param lat latitude, clamped to [-90, 90]
61    * @return the hex-encoded Z2SFC index value
62    */
63   def z2Hex(lon: Double, lat: Double): String =
64     Z2SFC.hexEncode(z2Index(lon, lat))
65 
66   /**
67    * Z2 index ranges covering the provided lat/lon bounds, hex-encoded via
68    * `Z2SFC.hexEncode`. Returns inclusive `HexRange` pairs;
69    * stored values produced by `z2Hex` that compare lexicographically.
70    *
71    * @param minLon envelope min longitude, clamped to [-180, 180]
72    * @param minLat envelope min latitude, clamped to [-90, 90]
73    * @param maxLon envelope max longitude, clamped to [-180, 180]
74    * @param maxLat envelope max latitude, clamped to [-90, 90]
75    * @param maxRanges rough upper bound on the number of ranges returned; the SFC
76    *                  coarsens (merges) past it, so the cover remains a superset of
77    *                  the envelope — pruning gets less selective, never lossy
78    * @return inclusive hex range pairs
79    */
80   def z2HexRanges(minLon: Double, minLat: Double, maxLon: Double, maxLat: Double, maxRanges: Int): jList[HexRange] =
81     Z2SFC.ranges((clampLon(minLon), clampLon(maxLon)), (clampLat(minLat), clampLat(maxLat)), FullPrecision, Some(maxRanges)).iterator
82       .map(r => HexRange(Z2SFC.hexEncode(r.lower), Z2SFC.hexEncode(r.upper)))
83       .toList.asJava
84 
85   /**
86    * XZ2 cell index for lat/lon bounds at the given `g` resolution.
87    * Returns the appropriate XZ2SFC sequence-code.
88    *
89    * @param minLon envelope min longitude, clamped to [-180, 180]
90    * @param minLat envelope min latitude, clamped to [-90, 90]
91    * @param maxLon envelope max longitude, clamped to [-180, 180]
92    * @param maxLat envelope max latitude, clamped to [-90, 90]
93    * @param g XZ2 quad-tree resolution
94    * @return the XZ2SFC sequence-code value
95    */
96   def xz2Index(minLon: Double, minLat: Double, maxLon: Double, maxLat: Double, g: Short): Long =
97     XZ2SFC(g).index(clampLon(minLon), clampLat(minLat), clampLon(maxLon), clampLat(maxLat))
98 
99   /**
100    * XZ2 cell value for a geometry's envelope at the given `g` resolution,
101    * hex-encoded via `XZ2SFC.hexEncode` (bit-shifted left so the significant
102    * bits are left-aligned and truncate-prefix matching works).
103    *
104    * @param minLon envelope min longitude, clamped to [-180, 180]
105    * @param minLat envelope min latitude, clamped to [-90, 90]
106    * @param maxLon envelope max longitude, clamped to [-180, 180]
107    * @param maxLat envelope max latitude, clamped to [-90, 90]
108    * @param g XZ2 quad-tree resolution
109    * @return the hex-encoded XZ2SFC sequence-code value
110    */
111   def xz2Hex(minLon: Double, minLat: Double, maxLon: Double, maxLat: Double, g: Short): String =
112     XZ2SFC(g).hexEncode(xz2Index(minLon, minLat, maxLon, maxLat, g))
113 
114   /**
115    * XZ2 index ranges covering the query envelope at the given `g` resolution,
116    * hex-encoded via `XZ2SFC.hexEncode`. Returns inclusive `HexRange` pairs;
117    * stored values produced by `xz2Hex` compare lexicographically within them.
118    *
119    * @param minLon envelope min longitude, clamped to [-180, 180]
120    * @param minLat envelope min latitude, clamped to [-90, 90]
121    * @param maxLon envelope max longitude, clamped to [-180, 180]
122    * @param maxLat envelope max latitude, clamped to [-90, 90]
123    * @param g XZ2 quad-tree resolution
124    * @param maxRanges rough upper bound on the number of ranges returned; the SFC
125    *                  coarsens (merges) past it, so the cover remains a superset of
126    *                  the envelope — pruning gets less selective, never lossy
127    * @return inclusive hex range pairs
128    */
129   def xz2HexRanges(minLon: Double, minLat: Double, maxLon: Double, maxLat: Double, g: Short, maxRanges: Int): jList[HexRange] = {
130     val sfc = XZ2SFC(g)
131     sfc.ranges((clampLon(minLon), clampLat(minLat), clampLon(maxLon), clampLat(maxLat)), Some(maxRanges)).iterator
132       .map(r => HexRange(sfc.hexEncode(r.lower), sfc.hexEncode(r.upper)))
133       .toList.asJava
134   }
135 }
