24 class GeoToolsUpdateSchemaCommand extends DataStoreCommand[DataStore] with GeoToolsDataStoreCommand {
25 
26   import scala.collection.JavaConverters._
27 
28   override val name = "update-schema"
29   override val params = new GeoToolsUpdateSchemaParams()
30 
31   override def execute(): Unit = withDataStore(update)
32 
33   protected def update(ds: DataStore): Unit = {
34 
35     // ensure we have an operation
36     params.validate().foreach(e => throw e)
37 
38     val sft = try { ds.getSchema(params.featureName) } catch { case _: IOException => null }
39     if (sft == null) {
40       throw new ParameterException(s"Schema '${params.featureName}' does not exist in the data store")
41     }
42 
43     var n = 0
44     // numbering for our prompts
45     def number: Int = { n += 1; n }
46     val prompts = new StringBuilder()
47 
48     val updated = SimpleFeatureTypes.copy(sft)
49 
50     if (!params.userData.isEmpty) {
51       params.userData.asScala.foreach { ud =>
52         ud.split(":", 2) match {
53           case Array(k, v) =>
54             updated.getUserData.put(k, v) match {
55               case null => prompts.append(s"\n  $number: Adding user data: '$k=$v'")
56               case old  => prompts.append(s"\n  $number: Updating user data: '$k=$v' (was '$old')")
57             }
58           case _ => throw new ParameterException(s"Invalid user data entry - expected 'key:value': $ud")
59         }
60       }
61     }
62 
63     Command.user.info(s"Preparing to update schema '${sft.getTypeName}':$prompts")
64     if (params.force || Prompt.confirm("Continue (y/n)? ")) {
65       Command.user.info("Updating, please wait...")
66       ds match {
67         case jdbc: PartitionedPostgisDataStore => jdbc.upgrade(updated)
68         case _ =>
69           try { ds.updateSchema(sft.getTypeName, updated) } catch {
70             case _: UnsupportedOperationException =>
71               throw new RuntimeException(s"${ds.getClass.getSimpleName} does not support schema updates")
72           }
73       }
74 
75       Command.user.info("Update complete")
76     }
77   }
78 }
