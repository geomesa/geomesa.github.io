12 /**
13  * Removes retired numeric write-ahead tables when readers permit it. The retirement marker hides
14  * their rows from new queries; rotation can delete the originals and reuse a table before cleanup.
15  */
16 object CleanWriteAheadRetirements extends SqlProcedure with CronSchedule {
17 
18   override def name(info: TypeInfo): FunctionName = FunctionName(s"${info.typeIdentifier}_clean_wa_retirements")
19 
20   override def jobName(info: TypeInfo): SqlLiteral = SqlLiteral(s"${info.typeIdentifier}-clean-wa-retirements")
21 
22   override protected def schedule(info: TypeInfo): SqlLiteral = SqlLiteral("* * * * *")
23 
24   override protected def invocation(info: TypeInfo): SqlLiteral =
25     SqlLiteral(s"CALL ${info.schema.quoted}.${name(info).quoted}()")
26 
27   override protected def createStatements(info: TypeInfo): Seq[String] = Seq(proc(info)) ++ super.createStatements(info)
28 
29   private def proc(info: TypeInfo): String =
30     s"""CREATE OR REPLACE PROCEDURE ${info.schema.quoted}.${name(info).quoted}() LANGUAGE plpgsql AS
31        |  $$BODY$$
32        |    DECLARE
33        |      retired record;
34        |      is_attached boolean;
35        |      is_retired boolean;
36        |      table_exists boolean;
37        |    BEGIN
38        |      FOR retired IN
39        |        SELECT * FROM ${info.tables.writeAheadRetirements.name.qualified} ORDER BY source_partition
40        |      LOOP
41        |        -- first check if the table is attached and try to detach it - this requires ACCESS EXCLUSIVE
42        |        SELECT EXISTS (SELECT FROM pg_inherits
43        |            WHERE inhparent = ${info.tables.writeAhead.name.asRegclass}
44        |              AND inhrelid = to_regclass(retired.source_partition)) INTO is_attached;
45        |
46        |        IF is_attached THEN
47        |          BEGIN
48        |            -- match log rotation's lock order, and never queue behind a reader or another cleanup.
49        |            -- we need EXCLUSIVE on both parent and child to NO INHERIT it, but that is the same as DROP requires.
50        |            -- note that queries will still lock the child table, even though it is excluded due to _wa_retirements
51        |            LOCK TABLE ONLY ${info.tables.writeAhead.name.qualified} IN ACCESS EXCLUSIVE MODE NOWAIT;
52        |            EXECUTE 'LOCK TABLE ONLY ' || retired.source_partition || ' IN ACCESS EXCLUSIVE MODE NOWAIT';
53        |            -- re-check existence now that we have our locks
54        |            SELECT EXISTS (
55        |              SELECT FROM ${info.tables.writeAheadRetirements.name.qualified}
56        |                WHERE partition_oid = retired.partition_oid AND enqueued = retired.enqueued)
57        |              INTO is_retired;
58        |            SELECT COALESCE(to_regclass(retired.source_partition)::oid = retired.partition_oid, false)
59        |              INTO table_exists;
60        |            SELECT EXISTS (SELECT FROM pg_inherits
61        |              WHERE inhparent = ${info.tables.writeAhead.name.asRegclass} AND inhrelid = to_regclass(retired.source_partition))
62        |              INTO is_attached;
63        |            IF is_retired AND table_exists AND is_attached THEN
64        |              EXECUTE 'ALTER TABLE ' || retired.source_partition || ' NO INHERIT ${info.tables.writeAhead.name.qualified}';
65        |              is_attached := false;
66        |            END IF;
67        |          EXCEPTION WHEN lock_not_available OR undefined_table THEN
68        |            RAISE DEBUG 'Deferring removal of numeric write-ahead table %', retired.source_partition;
69        |          END;
70        |          COMMIT; -- release locks
71        |        END IF;
72        |
73        |        -- now try to drop the table, if it's no longer attached.
74        |        -- there shouldn't be any contention on the child table at this point
75        |        IF NOT is_attached THEN
76        |          -- serialize with rotation and cleanup procedures
77        |          LOCK TABLE ONLY ${info.tables.writeAhead.name.qualified} IN SHARE UPDATE EXCLUSIVE MODE;
78        |          -- verify again that table exists and isn't attached
79        |          SELECT EXISTS (SELECT FROM ${info.tables.writeAheadRetirements.name.qualified}
80        |            WHERE partition_oid = retired.partition_oid AND enqueued = retired.enqueued)
81        |            INTO is_retired;
82        |          SELECT COALESCE(to_regclass(retired.source_partition)::oid = retired.partition_oid, false)
83        |            INTO table_exists;
84        |          SELECT EXISTS (
85        |            SELECT FROM pg_inherits
86        |              WHERE inhparent = ${info.tables.writeAhead.name.asRegclass} AND inhrelid = to_regclass(retired.source_partition))
87        |            INTO is_attached;
88        |          IF is_retired AND NOT is_attached THEN
89        |            IF table_exists THEN
90        |              EXECUTE 'DROP TABLE IF EXISTS' || retired.source_partition;
91        |            END IF;
92        |            DELETE FROM ${info.tables.writeAheadRetirements.name.qualified}
93        |              WHERE partition_oid = retired.partition_oid AND enqueued = retired.enqueued;
94        |          END IF;
95        |        END IF;
96        |        COMMIT; -- release locks
97        |      END LOOP;
98        |    END;
99        |  $$BODY$$;""".stripMargin
