16 /**
17  * Drops the previous generation of a feature type, once all of its data has aged off.
18  *
19  * The main view and triggers for the current generation depend on the view for the previous generation, so they
20  * are re-created without it. Replacing the view requires an exclusive lock on it - if the lock can't be acquired
21  * quickly (due to queries against the view), then we give up and try again the next time the procedure is run.
22  *
23  * If the previous generation can't be checked or dropped for any other reason (for example, if something outside of
24  * our control depends on its main view, or one of its tables has been modified), then we log a warning and try again
25  * the next time the procedure is run, so that partition maintenance for the current generation isn't affected.
26  *
27  * Writes to the previous generation are blocked while we verify that it is empty and drop it, so that we don't drop
28  * anything that is written (or committed) after we check.
29  *
30  * Once the previous generation has been retired, the procedure drops itself, and removes the call to itself from
31  * partition maintenance. The previous generation is only identified by name, so otherwise the procedure would go
32  * on to drop anything that was subsequently created with the same name.
33  *
34  * Age-off is configured per generation, so until the previous generation is retired, the procedure also keeps its
35  * max partitions in sync with the current generation, in case the configuration is changed directly in the user
36  * data table.
37  */
38 object RetirePreviousGeneration extends Sql {
39 
40   def name(info: TypeInfo): FunctionName = FunctionName(s"${info.typeIdentifier}_retire_previous")
41 
42   override def create(info: TypeInfo)(implicit ex: ExecutionContext): Unit = {
43     info.previous match {
44       case None => drop(info)
45       case Some(prev) => ex.execute(proc(info, prev, statements(info, prev)))
46     }
47   }
48 
49   override def dropSql(info: TypeInfo): Seq[String] =
50     Seq(s"DROP PROCEDURE IF EXISTS ${info.schema.quoted}.${name(info).quoted};")
51 
52   /**
53    * Gets the statements required to retire the previous generation.
54    *
55    * Note that the statements are executed later, by the procedure, so they are generated without reference to the
56    * connection that is creating the procedure. They also don't take the advisory locks that are used when dropping
57    * a schema through the data store - the locks are there to synchronize creating the shared tables, and the
58    * procedure only deletes rows from them.
59    *
60    * @param info type info
61    * @param prev previous generation type info
62    * @return
63    */
64   private def statements(info: TypeInfo, prev: TypeInfo): Seq[String] = {
65     // re-create the things that reference the previous generation
66     val current = info.copy(previous = None)
67     val create = Seq(MainView, UpdateTrigger, DeleteTrigger).flatMap(_.createSql(current))
68     // drop the previous generation - this is the same as removing a schema through the data store
69     val drop = PartitionedPostgisDialect.dropSql(prev)
70     // the previous generation only needs to be retired once, so remove this procedure and the call to it - otherwise
71     // it would also retire anything that is subsequently created with the same name as the previous generation
72     // note: we only replace the maintenance procedure, we don't need to re-schedule the cron job that invokes it
73     create ++ drop ++ Seq(PartitionMaintenance.proc(current)) ++ dropSql(info)
74   }
75 
76   private def proc(info: TypeInfo, prev: TypeInfo, statements: Seq[String]): String = {
77     val userData = s"${info.schema.quoted}.${UserDataTable.Name.quoted}"
78     val tables = Seq(prev.tables.writeAhead, prev.tables.writeAheadPartitions, prev.tables.mainPartitions, prev.tables.spillPartitions)
79     val maxPartitions = literal(SftUserData.MaxPartitions.key)
80     val notEmpty = tables.map(t => s"EXISTS (SELECT FROM ${t.name.qualified})").mkString("\n          OR ")
81     // note: the statements contain $$ and $BODY$ quoting, so we use different tags here
82     s"""CREATE OR REPLACE PROCEDURE ${info.schema.quoted}.${name(info).quoted}() LANGUAGE plpgsql AS
83        |  $$RETIRE$$
84        |    DECLARE
85        |      cur_max text;
86        |      prev_max text;
87        |    BEGIN
88        |      -- check if the previous generation has already been retired
89        |      IF to_regclass(${literal(prev.tables.view.name.qualified)}) IS NULL THEN
90        |        RETURN;
91        |      END IF;
92        |      -- note: everything else runs in a block, so that an error doesn't fail partition maintenance for the
93        |      -- current generation
94        |      BEGIN
95        |        -- age-off is configured per generation, so apply our current configuration to the previous generation, in
96        |        -- case it has been changed directly in the user data table
97        |        -- note: we only write to the user data table if the configuration is out of sync, to avoid taking row locks
98        |        SELECT value INTO cur_max FROM $userData WHERE type_name = ${literal(info.typeName)} AND key = $maxPartitions;
99        |        SELECT value INTO prev_max FROM $userData WHERE type_name = ${literal(prev.typeName)} AND key = $maxPartitions;
100        |        IF cur_max IS NULL AND prev_max IS NOT NULL THEN
101        |          DELETE FROM $userData WHERE type_name = ${literal(prev.typeName)} AND key = $maxPartitions;
102        |        ELSIF cur_max IS DISTINCT FROM prev_max THEN
103        |          INSERT INTO $userData (type_name, key, value, mutable)
104        |            SELECT ${literal(prev.typeName)}, c.key, c.value, c.mutable FROM $userData c
105        |              WHERE c.type_name = ${literal(info.typeName)} AND c.key = $maxPartitions
106        |            ON CONFLICT (type_name, key) DO UPDATE SET value = EXCLUDED.value;
107        |        END IF;
108        |        -- the previous generation has to retire any generations that came before it first
109        |        IF EXISTS (SELECT FROM $userData
110        |            WHERE type_name = ${literal(prev.typeName)} AND key = ${literal(SftUserData.PreviousView.key)}) THEN
111        |          RETURN;
112        |        END IF;
113        |        -- wait until all the data in the previous generation has aged off
114        |        IF $notEmpty THEN
115        |          RETURN;
116        |        END IF;
117        |        -- don't block queries if we can't get a lock on the main view, we'll try again next time
118        |        SET LOCAL lock_timeout = '1s';
119        |        -- block any writes to the previous generation, and wait for any that are in progress, then
120        |        -- check again to make sure that nothing was written since our first check
121        |        LOCK TABLE ${tables.map(_.name.qualified).mkString(", ")} IN SHARE ROW EXCLUSIVE MODE;
122        |        IF NOT ($notEmpty) THEN
123        |${statements.map(s => s"          EXECUTE $$STATEMENT$$$s$$STATEMENT$$;").mkString("\n")}
124        |          DELETE FROM $userData
125        |            WHERE type_name = ${literal(info.typeName)} AND key = ${literal(SftUserData.PreviousView.key)};
126        |          RAISE NOTICE 'Retired previous generation %', ${literal(prev.typeName)};
127        |        END IF;
128        |      EXCEPTION
129        |        WHEN lock_not_available OR deadlock_detected THEN
130        |          RAISE INFO 'Could not acquire locks to retire previous generation %, will retry', ${literal(prev.typeName)};
131        |        WHEN OTHERS THEN
132        |          -- don't fail partition maintenance for the current generation - for example, if a user-created view
133        |          -- depends on the previous main view, then it can't be dropped until that view is removed
134        |          RAISE WARNING 'Could not retire previous generation %, will retry: % (%)',
135        |            ${literal(prev.typeName)}, SQLERRM, SQLSTATE;
136        |      END;
137        |      COMMIT;
138        |    END;
139        |  $$RETIRE$$;
140        |""".stripMargin
141   }
