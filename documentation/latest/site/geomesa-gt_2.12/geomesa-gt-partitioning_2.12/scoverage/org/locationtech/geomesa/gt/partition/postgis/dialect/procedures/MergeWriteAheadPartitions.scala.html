14 /**
15  * Merge recent write-ahead partitions and move them into the main partition table
16  */
17 object MergeWriteAheadPartitions extends SqlProcedure {
18 
19   override def name(info: TypeInfo): FunctionName = FunctionName(s"${info.typeIdentifier}_merge_wa_partitions")
20 
21   override protected def createStatements(info: TypeInfo): Seq[String] = Seq(proc(info))
22 
23   private def proc(info: TypeInfo): String = {
24     s"""CREATE OR REPLACE PROCEDURE ${info.schema.quoted}.${name(info).quoted}(cur_time timestamp without time zone) LANGUAGE plpgsql AS
25        |  $$BODY$$
26        |    DECLARE
27        |      partition_size int;                          -- size of the partition, in hours
28        |      min_dtg timestamp without time zone;         -- min date in our partitioned tables
29        |      main_cutoff timestamp without time zone;     -- max age of the records for main tables
30        |      partition_start timestamp without time zone; -- start bounds for the partition we're writing
31        |      partition_end timestamp without time zone;   -- end bounds for the partition we're writing
32        |      partition_name text;                         -- partition table name
33        |      partition_parent text;                       -- partition table to attach to
34        |      partition_tablespace text;                   -- partition tablespace
35        |      index_tablespace text;                       -- index tablespace
36        |      table_wa_logging text;                       -- wa log options
37        |      write_ahead_partitions text[];               -- names of the partitions we're migrating
38        |      write_ahead_partition text;                  -- name of current partition
39        |      pexists boolean;                             -- table exists check
40        |      unsorted_count bigint;
41        |    BEGIN
42        |      -- constants
43        |      ${UserDataConfig.select(info,
44                 "partition_size" -> SftUserData.IntervalHours,
45                 "partition_tablespace" -> SftUserData.MainTableSpace,
46                 "table_wa_logging" -> SftUserData.WalLogEnabled)}
47        |      partition_size := COALESCE(partition_size, ${SftUserData.IntervalHours.default});
48        |      ${UserDataConfig.tablespaceClauses("partition_tablespace", "index_tablespace")}
49        |      ${UserDataConfig.walLoggingClause("table_wa_logging")}
50        |      ${UserDataConfig.release}
51        |
52        |      main_cutoff := ${info.schema.quoted}.truncate_to_partition(cur_time, partition_size) - make_interval(hours => partition_size);
53        |
54        |      -- move data from the write ahead partitions to the main partitions
55        |      LOOP
56        |        -- find the range of dates in the write ahead partition tables
57        |        SELECT min(write_ahead_partition.${info.cols.dtg.quoted}) INTO min_dtg
58        |          FROM ${info.tables.writeAheadPartitions.name.qualified} write_ahead_partition
59        |          WHERE write_ahead_partition.${info.cols.dtg.quoted} < main_cutoff
60        |            AND NOT EXISTS (
61        |              SELECT FROM ${info.tables.writeAheadMigrations.name.qualified} migration
62        |              WHERE write_ahead_partition.${info.cols.dtg.quoted} >= migration.partition_start
63        |                AND write_ahead_partition.${info.cols.dtg.quoted} < migration.partition_end
64        |            );
65        |        EXIT WHEN min_dtg IS NULL;
66        |
67        |        partition_start := ${info.schema.quoted}.truncate_to_partition(min_dtg, partition_size);
68        |        partition_end := partition_start + make_interval(hours => partition_size);
69        |        partition_parent := ${info.tables.mainPartitions.name.asLiteral};
70        |        partition_name := partition_parent || '_' || to_char(partition_start, 'YYYY_MM_DD_HH24');
71        |
72        |        SELECT EXISTS(SELECT FROM pg_tables WHERE schemaname = ${info.schema.asLiteral} AND tablename = partition_name)
73        |          INTO pexists;
74        |
75        |        -- if the partition already exists, write to the spill partition instead to avoid messing up the BRIN index
76        |        IF pexists THEN
77        |          partition_parent := ${info.tables.spillPartitions.name.asLiteral};
78        |          partition_name := partition_parent || '_' || to_char(partition_start, 'YYYY_MM_DD_HH24');
79        |          SELECT EXISTS(SELECT FROM pg_tables WHERE schemaname = ${info.schema.asLiteral} AND tablename = partition_name)
80        |            INTO pexists;
81        |        END IF;
82        |
83        |        -- find the write ahead partitions we're copying from
84        |        -- order the results to ensure we get locks in a consistent order to avoid deadlocks
85        |        write_ahead_partitions := Array(
86        |          SELECT '${info.schema.quoted}.' || quote_ident(pg_class.relname)
87        |            FROM pg_catalog.pg_inherits
88        |            INNER JOIN pg_catalog.pg_class ON (pg_inherits.inhrelid = pg_class.oid)
89        |            INNER JOIN pg_catalog.pg_namespace ON (pg_class.relnamespace = pg_namespace.oid)
90        |            WHERE inhparent = ${info.tables.writeAheadPartitions.name.asRegclass}
91        |              AND pg_class.relname >= ${info.tables.writeAheadPartitions.name.asLiteral} || '_' || to_char(partition_start, 'YYYY_MM_DD_HH24_MI')
92        |              AND pg_class.relname < ${info.tables.writeAheadPartitions.name.asLiteral} || '_' || to_char(partition_end, 'YYYY_MM_DD_HH24_MI')
93        |            ORDER BY 1
94        |          );
95        |
96        |        -- get a lock on the tables - this mode won't prevent reads but will prevent writes
97        |        -- (there shouldn't be any writes though) and will synchronize this method
98        |        -- TODO we really just need to sync this method for safety in manual invocations
99        |        -- FOREACH write_ahead_partition IN ARRAY write_ahead_partitions LOOP
100        |        --   EXECUTE 'LOCK TABLE ' || write_ahead_partition || ' IN SHARE ROW EXCLUSIVE MODE';
101        |        --   RAISE INFO '% Locked write ahead partition % for migration', timeofday()::timestamp, write_ahead_partition;
102        |        -- END LOOP;
103        |
104        |        -- create a view from the tables so that we can sort the result by an expression (geohash)
105        |        EXECUTE 'CREATE OR REPLACE TEMP VIEW ' || quote_ident(partition_name || '_tmp_migrate') ||
106        |          ' AS SELECT * FROM ' || array_to_string(write_ahead_partitions, ' UNION ALL SELECT * FROM ');
107        |
108        |        -- copy rows from write ahead partitions to main partition table
109        |        IF pexists THEN
110        |          RAISE INFO '% Copying rows to partition %', timeofday()::timestamp, partition_name;
111        |          EXECUTE 'INSERT INTO ${info.schema.quoted}.' || quote_ident(partition_name) ||
112        |            ' SELECT * FROM ' || quote_ident(partition_name || '_tmp_migrate') ||
113        |            '   ORDER BY _st_sortablehash(${info.cols.geom.quoted})' ||
114        |            '   ON CONFLICT DO NOTHING';
115        |          GET DIAGNOSTICS unsorted_count := ROW_COUNT;
116        |        ELSE
117        |          RAISE INFO '% Creating partition with insert % (unattached)', timeofday()::timestamp, partition_name;
118        |
119        |          -- upper bounds are exclusive
120        |          -- this won't have any indices until we attach it to the parent partition table
121        |          -- use "create table as" (vs create then insert) for performance benefits related to WAL skipping
122        |          -- we need a "select distinct" to avoid primary key conflicts - this should be fairly cheap since
123        |          --   we're already sorting and there should be few or no conflicts
124        |          -- create the partition table with a 'create as' for improved performance
125        |          EXECUTE 'CREATE ' || table_wa_logging || 'TABLE ${info.schema.quoted}.' || quote_ident(partition_name) ||
126        |            '${info.tables.mainPartitions.storage.opts}' || partition_tablespace || ' AS SELECT DISTINCT ON' ||
127        |            ' (_st_sortablehash(${info.cols.geom.quoted}), ${info.cols.fid.quoted}, ${info.cols.dtg.quoted}) * FROM ' ||
128        |            quote_ident(partition_name || '_tmp_migrate') || ' ORDER BY _st_sortablehash(${info.cols.geom.quoted})';
129        |          GET DIAGNOSTICS unsorted_count := ROW_COUNT;
130        |          EXECUTE 'ALTER TABLE ${info.schema.quoted}.' || quote_ident(partition_name) ||
131        |            ' ADD CONSTRAINT ' || quote_ident(partition_name || '_pkey') ||
132        |            ' PRIMARY KEY (${info.cols.fid.quoted}, ${info.cols.dtg.quoted})' || index_tablespace;
133        |          -- creating a constraint allows it to be attached to the parent without any additional checks
134        |          EXECUTE 'ALTER TABLE  ${info.schema.quoted}.' || quote_ident(partition_name) ||
135        |            ' ADD CONSTRAINT ' || quote_ident(partition_name || '_constraint') ||
136        |            ' CHECK ( ${info.cols.dtg.quoted} >= ' || quote_literal(partition_start) ||
137        |            ' AND ${info.cols.dtg.quoted} < ' || quote_literal(partition_end) || ' )';
138        |        END IF;
139        |        RAISE INFO '% Done writing % rows to partition %', timeofday()::timestamp, unsorted_count, partition_name;
140        |
141        |        IF partition_parent = ${info.tables.spillPartitions.name.asLiteral} THEN
142        |          -- store record of unsorted row counts which could negatively impact BRIN index scans
143        |          INSERT INTO ${info.tables.sortQueue.name.qualified}(partition_name, unsorted_count, enqueued)
144        |            VALUES (partition_name, unsorted_count, now());
145        |          RAISE NOTICE 'Inserting % rows into spill partition %, queries may be impacted',
146        |                unsorted_count, partition_name;
147        |        END IF;
148        |
149        |        IF NOT pexists THEN
150        |          EXECUTE 'ALTER TABLE ${info.schema.quoted}.' || quote_ident(partition_parent) ||
151        |            ' ATTACH PARTITION ${info.schema.quoted}.' || quote_ident(partition_name) ||
152        |            ' FOR VALUES FROM (' || quote_literal(partition_start) ||
153        |            ') TO (' || quote_literal(partition_end) || ' );';
154        |          -- now that we've attached the table we can drop the redundant constraint
155        |          -- however, this requires ACCESS EXCLUSIVE - since constraints are only checked on inserts
156        |          -- or updates, and partition tables are 'immutable' (only written to once), it shouldn't
157        |          -- affect anything to leave it. note that for 'spill' tables, there may be some redundant checks
158        |          -- EXECUTE 'ALTER TABLE ${info.schema.quoted}.' || quote_ident(partition_name) ||
159        |          --  ' DROP CONSTRAINT ' || quote_ident(partition_name || '_constraint');
160        |          RAISE NOTICE 'A partition has been created %', partition_name;
161        |        END IF;
162        |
163        |        EXECUTE 'DROP VIEW ' || quote_ident(partition_name || '_tmp_migrate');
164        |        FOREACH write_ahead_partition IN ARRAY write_ahead_partitions LOOP
165        |          INSERT INTO ${info.tables.writeAheadMigrations.name.qualified}
166        |            (source_partition, partition_start, partition_end, enqueued)
167        |            VALUES (write_ahead_partition, partition_start, partition_end, now())
168        |            ON CONFLICT (source_partition) DO UPDATE
169        |              SET partition_start = EXCLUDED.partition_start,
170        |                  partition_end = EXCLUDED.partition_end,
171        |                  enqueued = EXCLUDED.enqueued;
172        |
173        |          RAISE NOTICE 'A partition has been queued for detachment %', write_ahead_partition;
174        |        END LOOP;
175        |
176        |        -- mark the partition to be analyzed in a separate thread
177        |        INSERT INTO ${info.tables.analyzeQueue.name.qualified}(partition_name, enqueued)
178        |          VALUES (partition_name, now());
179        |
180        |        -- commit after each move, also releases the table locks
181        |        COMMIT;
182        |
183        |      END LOOP;
184        |      CALL ${info.schema.quoted}.${CleanWriteAheadMigrations.name(info).quoted}();
185        |    END;
186        |  $$BODY$$;
187        |""".stripMargin
188   }
