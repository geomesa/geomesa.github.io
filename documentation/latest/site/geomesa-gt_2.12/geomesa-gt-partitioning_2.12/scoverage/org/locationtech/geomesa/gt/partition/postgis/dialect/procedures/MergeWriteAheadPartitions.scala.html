15 /**
16  * Merge recent write-ahead partitions and move them into the main partition table
17  */
18 object MergeWriteAheadPartitions extends SqlProcedure {
19 
20   override def name(info: TypeInfo): FunctionName = FunctionName(s"${info.typeIdentifier}_merge_wa_partitions")
21 
22   override protected def createStatements(info: TypeInfo): Seq[String] = Seq(proc(info))
23 
24   private def proc(info: TypeInfo): String = {
25     s"""CREATE OR REPLACE PROCEDURE ${info.schema.quoted}.${name(info).quoted}(cur_time timestamp without time zone) LANGUAGE plpgsql AS
26        |  $$BODY$$
27        |    DECLARE
28        |      partition_size int;                          -- size of the partition, in hours
29        |      min_dtg timestamp without time zone;         -- min date in our partitioned tables
30        |      main_cutoff timestamp without time zone;     -- max age of the records for main tables
31        |      partition_start timestamp without time zone; -- start bounds for the partition we're writing
32        |      partition_end timestamp without time zone;   -- end bounds for the partition we're writing
33        |      partition_name text;                         -- partition table name
34        |      partition_parent text;                       -- partition table to attach to
35        |      partition_tablespace text;                   -- partition tablespace
36        |      index_tablespace text;                       -- index tablespace
37        |      table_wa_logging text;                       -- wa log options
38        |      write_ahead_partitions text[];               -- names of the partitions we're migrating
39        |      write_ahead_partition text;                  -- name of current partition
40        |      pexists boolean;                             -- table exists check
41        |      unsorted_count bigint;
42        |    BEGIN
43        |      -- constants
44        |      SELECT COALESCE(
45        |          (SELECT value::int FROM ${info.schema.quoted}.${UserDataTable.Name.quoted}
46        |            WHERE type_name = ${literal(info.typeName)} AND key = ${literal(SftUserData.IntervalHours.key)}),
47        |          ${SftUserData.IntervalHours.default}
48        |        ) INTO partition_size;
49        |      main_cutoff := ${info.schema.quoted}.truncate_to_partition(cur_time, partition_size) - make_interval(hours => partition_size);
50        |
51        |      -- move data from the write ahead partitions to the main partitions
52        |      LOOP
53        |        -- find the range of dates in the write ahead partition tables
54        |        SELECT min(write_ahead_partition.${info.cols.dtg.quoted}) INTO min_dtg
55        |          FROM ${info.tables.writeAheadPartitions.name.qualified} write_ahead_partition
56        |          WHERE write_ahead_partition.${info.cols.dtg.quoted} < main_cutoff
57        |            AND NOT EXISTS (
58        |              SELECT FROM ${info.tables.writeAheadMigrations.name.qualified} migration
59        |              WHERE write_ahead_partition.${info.cols.dtg.quoted} >= migration.partition_start
60        |                AND write_ahead_partition.${info.cols.dtg.quoted} < migration.partition_end
61        |            );
62        |        EXIT WHEN min_dtg IS NULL;
63        |
64        |        partition_start := ${info.schema.quoted}.truncate_to_partition(min_dtg, partition_size);
65        |        partition_end := partition_start + make_interval(hours => partition_size);
66        |        partition_parent := ${info.tables.mainPartitions.name.asLiteral};
67        |        partition_name := partition_parent || '_' || to_char(partition_start, 'YYYY_MM_DD_HH24');
68        |
69        |        SELECT EXISTS(SELECT FROM pg_tables WHERE schemaname = ${info.schema.asLiteral} AND tablename = partition_name)
70        |          INTO pexists;
71        |
72        |        -- if the partition already exists, write to the spill partition instead to avoid messing up the BRIN index
73        |        IF pexists THEN
74        |          partition_parent := ${info.tables.spillPartitions.name.asLiteral};
75        |          partition_name := partition_parent || '_' || to_char(partition_start, 'YYYY_MM_DD_HH24');
76        |          SELECT EXISTS(SELECT FROM pg_tables WHERE schemaname = ${info.schema.asLiteral} AND tablename = partition_name)
77        |            INTO pexists;
78        |        END IF;
79        |
80        |        -- find the write ahead partitions we're copying from
81        |        -- order the results to ensure we get locks in a consistent order to avoid deadlocks
82        |        write_ahead_partitions := Array(
83        |          SELECT '${info.schema.quoted}.' || quote_ident(pg_class.relname)
84        |            FROM pg_catalog.pg_inherits
85        |            INNER JOIN pg_catalog.pg_class ON (pg_inherits.inhrelid = pg_class.oid)
86        |            INNER JOIN pg_catalog.pg_namespace ON (pg_class.relnamespace = pg_namespace.oid)
87        |            WHERE inhparent = ${info.tables.writeAheadPartitions.name.asRegclass}
88        |              AND pg_class.relname >= ${info.tables.writeAheadPartitions.name.asLiteral} || '_' || to_char(partition_start, 'YYYY_MM_DD_HH24_MI')
89        |              AND pg_class.relname < ${info.tables.writeAheadPartitions.name.asLiteral} || '_' || to_char(partition_end, 'YYYY_MM_DD_HH24_MI')
90        |            ORDER BY 1
91        |          );
92        |
93        |        -- get a lock on the tables - this mode won't prevent reads but will prevent writes
94        |        -- (there shouldn't be any writes though) and will synchronize this method
95        |        -- TODO we really just need to sync this method for safety in manual invocations
96        |        -- FOREACH write_ahead_partition IN ARRAY write_ahead_partitions LOOP
97        |        --   EXECUTE 'LOCK TABLE ' || write_ahead_partition || ' IN SHARE ROW EXCLUSIVE MODE';
98        |        --   RAISE INFO '% Locked write ahead partition % for migration', timeofday()::timestamp, write_ahead_partition;
99        |        -- END LOOP;
100        |
101        |        -- create a view from the tables so that we can sort the result by an expression (geohash)
102        |        EXECUTE 'CREATE OR REPLACE TEMP VIEW ' || quote_ident(partition_name || '_tmp_migrate') ||
103        |          ' AS SELECT * FROM ' || array_to_string(write_ahead_partitions, ' UNION ALL SELECT * FROM ');
104        |
105        |        -- copy rows from write ahead partitions to main partition table
106        |        IF pexists THEN
107        |          RAISE INFO '% Copying rows to partition %', timeofday()::timestamp, partition_name;
108        |          EXECUTE 'INSERT INTO ${info.schema.quoted}.' || quote_ident(partition_name) ||
109        |            ' SELECT * FROM ' || quote_ident(partition_name || '_tmp_migrate') ||
110        |            '   ORDER BY _st_sortablehash(${info.cols.geom.quoted})' ||
111        |            '   ON CONFLICT DO NOTHING';
112        |          GET DIAGNOSTICS unsorted_count := ROW_COUNT;
113        |        ELSE
114        |          RAISE INFO '% Creating partition with insert % (unattached)', timeofday()::timestamp, partition_name;
115        |
116        |          SELECT value FROM ${info.schema.quoted}.${UserDataTable.Name.quoted}
117        |            WHERE type_name = ${literal(info.typeName)} AND key = ${literal(SftUserData.MainTableSpace.key)}
118        |            INTO partition_tablespace;
119        |          IF partition_tablespace IS NULL THEN
120        |            index_tablespace := '';
121        |            partition_tablespace := '';
122        |          ELSE
123        |            index_tablespace := ' USING INDEX TABLESPACE '|| quote_ident(partition_tablespace);
124        |            partition_tablespace := ' TABLESPACE ' || quote_ident(partition_tablespace);
125        |          END IF;
126        |
127        |          SELECT COALESCE(
128        |            (SELECT value FROM ${info.schema.quoted}.${UserDataTable.Name.quoted}
129        |              WHERE type_name = ${literal(info.typeName)} AND key = ${literal(SftUserData.WalLogEnabled.key)}),
130        |              '${SftUserData.WalLogEnabled.default}'
131        |            ) INTO table_wa_logging;
132        |          IF table_wa_logging IS NULL OR table_wa_logging::boolean THEN
133        |            table_wa_logging := '';
134        |          ELSE
135        |            table_wa_logging := 'UNLOGGED ';
136        |          END IF;
137        |
138        |          -- upper bounds are exclusive
139        |          -- this won't have any indices until we attach it to the parent partition table
140        |          -- use "create table as" (vs create then insert) for performance benefits related to WAL skipping
141        |          -- we need a "select distinct" to avoid primary key conflicts - this should be fairly cheap since
142        |          --   we're already sorting and there should be few or no conflicts
143        |          -- create the partition table with a 'create as' for improved performance
144        |          EXECUTE 'CREATE ' || table_wa_logging || 'TABLE ${info.schema.quoted}.' || quote_ident(partition_name) ||
145        |            '${info.tables.mainPartitions.storage.opts}' || partition_tablespace || ' AS SELECT DISTINCT ON' ||
146        |            ' (_st_sortablehash(${info.cols.geom.quoted}), ${info.cols.fid.quoted}, ${info.cols.dtg.quoted}) * FROM ' ||
147        |            quote_ident(partition_name || '_tmp_migrate') || ' ORDER BY _st_sortablehash(${info.cols.geom.quoted})';
148        |          GET DIAGNOSTICS unsorted_count := ROW_COUNT;
149        |          EXECUTE 'ALTER TABLE ${info.schema.quoted}.' || quote_ident(partition_name) ||
150        |            ' ADD CONSTRAINT ' || quote_ident(partition_name || '_pkey') ||
151        |            ' PRIMARY KEY (${info.cols.fid.quoted}, ${info.cols.dtg.quoted})' || index_tablespace;
152        |          -- creating a constraint allows it to be attached to the parent without any additional checks
153        |          EXECUTE 'ALTER TABLE  ${info.schema.quoted}.' || quote_ident(partition_name) ||
154        |            ' ADD CONSTRAINT ' || quote_ident(partition_name || '_constraint') ||
155        |            ' CHECK ( ${info.cols.dtg.quoted} >= ' || quote_literal(partition_start) ||
156        |            ' AND ${info.cols.dtg.quoted} < ' || quote_literal(partition_end) || ' )';
157        |        END IF;
158        |        RAISE INFO '% Done writing % rows to partition %', timeofday()::timestamp, unsorted_count, partition_name;
159        |
160        |        IF partition_parent = ${info.tables.spillPartitions.name.asLiteral} THEN
161        |          -- store record of unsorted row counts which could negatively impact BRIN index scans
162        |          INSERT INTO ${info.tables.sortQueue.name.qualified}(partition_name, unsorted_count, enqueued)
163        |            VALUES (partition_name, unsorted_count, now());
164        |          RAISE NOTICE 'Inserting % rows into spill partition %, queries may be impacted',
165        |                unsorted_count, partition_name;
166        |        END IF;
167        |
168        |        IF NOT pexists THEN
169        |          EXECUTE 'ALTER TABLE ${info.schema.quoted}.' || quote_ident(partition_parent) ||
170        |            ' ATTACH PARTITION ${info.schema.quoted}.' || quote_ident(partition_name) ||
171        |            ' FOR VALUES FROM (' || quote_literal(partition_start) ||
172        |            ') TO (' || quote_literal(partition_end) || ' );';
173        |          -- now that we've attached the table we can drop the redundant constraint
174        |          -- however, this requires ACCESS EXCLUSIVE - since constraints are only checked on inserts
175        |          -- or updates, and partition tables are 'immutable' (only written to once), it shouldn't
176        |          -- affect anything to leave it. note that for 'spill' tables, there may be some redundant checks
177        |          -- EXECUTE 'ALTER TABLE ${info.schema.quoted}.' || quote_ident(partition_name) ||
178        |          --  ' DROP CONSTRAINT ' || quote_ident(partition_name || '_constraint');
179        |          RAISE NOTICE 'A partition has been created %', partition_name;
180        |        END IF;
181        |
182        |        EXECUTE 'DROP VIEW ' || quote_ident(partition_name || '_tmp_migrate');
183        |        FOREACH write_ahead_partition IN ARRAY write_ahead_partitions LOOP
184        |          INSERT INTO ${info.tables.writeAheadMigrations.name.qualified}
185        |            (source_partition, partition_start, partition_end, enqueued)
186        |            VALUES (write_ahead_partition, partition_start, partition_end, now())
187        |            ON CONFLICT (source_partition) DO UPDATE
188        |              SET partition_start = EXCLUDED.partition_start,
189        |                  partition_end = EXCLUDED.partition_end,
190        |                  enqueued = EXCLUDED.enqueued;
191        |
192        |          RAISE NOTICE 'A partition has been queued for detachment %', write_ahead_partition;
193        |        END LOOP;
194        |
195        |        -- mark the partition to be analyzed in a separate thread
196        |        INSERT INTO ${info.tables.analyzeQueue.name.qualified}(partition_name, enqueued)
197        |          VALUES (partition_name, now());
198        |
199        |        -- commit after each move, also releases the table locks
200        |        COMMIT;
201        |
202        |      END LOOP;
203        |      CALL ${info.schema.quoted}.${CleanWriteAheadMigrations.name(info).quoted}();
204        |    END;
205        |  $$BODY$$;
206        |""".stripMargin
207   }
