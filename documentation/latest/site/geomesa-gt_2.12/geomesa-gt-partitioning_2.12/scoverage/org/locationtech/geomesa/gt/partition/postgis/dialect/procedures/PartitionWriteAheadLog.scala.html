27        |    DECLARE
28        |      seq_val smallint;
29        |      write_partition text;                        -- current partition receiving writes
30        |      min_dtg timestamp without time zone;         -- min date in our partitioned tables
31        |      main_cutoff timestamp without time zone;     -- max age of the records for main tables
32        |      write_ahead record;
33        |      partition_size int;                          -- size of the partition, in hours
34        |      partition_start timestamp without time zone; -- start bounds for the partition we're writing
35        |      partition_end timestamp without time zone;   -- end bounds for the partition we're writing
36        |      partition_name text;                         -- partition table name
37        |      partition_parent text;                       -- partition parent table name
38        |      partition_tablespace text;                   -- partition tablespace
39        |      index_tablespace text;                       -- index tablespace
40        |      table_wa_logging text;                       -- wa log options
41        |      pexists boolean;                             -- table exists check
42        |      unsorted_count bigint;
43        |    BEGIN
44        |      -- constants
45        |      SELECT COALESCE(
46        |        (SELECT value::int FROM ${info.schema.quoted}.${UserDataTable.Name.quoted}
47        |          WHERE type_name = ${literal(info.typeName)} AND key = ${literal(SftUserData.IntervalHours.key)}),
48        |        ${SftUserData.IntervalHours.default})
49        |        INTO partition_size;
50        |      main_cutoff := ${info.schema.quoted}.truncate_to_partition(cur_time, partition_size) - make_interval(hours => partition_size);
51        |
52        |      SELECT value from ${info.schema.quoted}.${SequenceTable.Name.quoted}
53        |        WHERE type_name = ${literal(info.typeName)} INTO seq_val;
54        |      write_partition := ${literal(info.tables.writeAhead.name.raw + "_")} || lpad(seq_val::text, 3, '0');
55        |
56        |      -- check for write ahead partitions and move the data into the time partitioned tables
57        |      FOR write_ahead IN
58        |        SELECT pg_class.relname AS name, pg_class.oid AS oid
59        |          FROM pg_catalog.pg_inherits
60        |          INNER JOIN pg_catalog.pg_class ON (pg_inherits.inhrelid = pg_class.oid)
61        |          INNER JOIN pg_catalog.pg_namespace ON (pg_class.relnamespace = pg_namespace.oid)
62        |          WHERE inhparent = ${info.tables.writeAhead.name.asRegclass}
63        |          AND relname != write_partition
64        |          AND NOT EXISTS (SELECT FROM ${info.tables.writeAheadRetirements.name.qualified} retired
65        |            WHERE retired.partition_oid = pg_class.oid)
66        |          ORDER BY name
67        |      LOOP
68        |
69        |        RAISE INFO '% Checking write ahead table %', timeofday()::timestamp, write_ahead.name;
70        |        -- get a lock on the table - this mode won't prevent reads but will prevent writes
71        |        -- (there shouldn't be any writes though) and will synchronize this method
72        |        LOCK TABLE ONLY ${info.tables.mainPartitions.name.qualified} IN SHARE UPDATE EXCLUSIVE MODE;
73        |        LOCK TABLE ONLY ${info.tables.spillPartitions.name.qualified} IN SHARE UPDATE EXCLUSIVE MODE;
74        |        -- Recheck after acquiring the parent locks: another migration may have retired this table.
75        |        IF NOT EXISTS (SELECT FROM pg_class WHERE oid = write_ahead.oid)
76        |          OR EXISTS (SELECT FROM ${info.tables.writeAheadRetirements.name.qualified} WHERE partition_oid = write_ahead.oid) THEN
77        |          COMMIT;
78        |          CONTINUE;
79        |        END IF;
80        |        EXECUTE 'LOCK TABLE ${info.schema.quoted}.' || quote_ident(write_ahead.name) ||
81        |          ' IN SHARE ROW EXCLUSIVE MODE';
82        |        -- Rotation may have reused a retired table since this loop enumerated the children.
83        |        SELECT value FROM ${info.schema.quoted}.${SequenceTable.Name.quoted}
84        |          WHERE type_name = ${literal(info.typeName)} INTO seq_val;
85        |        IF write_ahead.name = ${literal(info.tables.writeAhead.name.raw + "_")} || lpad(seq_val::text, 3, '0') THEN
86        |          COMMIT;
87        |          CONTINUE;
88        |        END IF;
89        |        RAISE INFO '% Locked write ahead table % for migration', timeofday()::timestamp, write_ahead.name;
90        |
91        |        -- wait until the table doesn't contain any recent records
92        |        EXECUTE 'SELECT EXISTS(SELECT 1 FROM ${info.schema.quoted}.' || quote_ident(write_ahead.name) ||
93        |          ' WHERE ${info.cols.dtg.quoted} >= ' || quote_literal(${info.schema.quoted}.truncate_to_ten_minutes(cur_time)) || ')'
94        |          INTO pexists;
95        |
96        |        IF pexists THEN
97        |          -- should only happen if data is inserted with timestamps from the future
98        |          RAISE NOTICE '% Skipping write ahead table % due to min date', timeofday()::timestamp, write_ahead.name;
99        |        ELSE
100        |          partition_end := '-infinity'::timestamp without time zone;
101        |          LOOP
102        |            -- find the range of dates in the write ahead partition
103        |            EXECUTE 'SELECT min(${info.cols.dtg.quoted}) FROM ${info.schema.quoted}.' || quote_ident(write_ahead.name) ||
104        |              ' WHERE ${info.cols.dtg.quoted} >= ' || quote_literal(partition_end) INTO min_dtg;
105        |            EXIT WHEN min_dtg IS NULL;
106        |
107        |            -- calculate the partition bounds for the min date
108        |            IF min_dtg < main_cutoff THEN
109        |              partition_start := ${info.schema.quoted}.truncate_to_partition(min_dtg, partition_size);
110        |              partition_end := partition_start + make_interval(hours => partition_size);
111        |              partition_parent := ${info.tables.mainPartitions.name.asLiteral};
112        |              partition_name := partition_parent || '_' || to_char(partition_start, 'YYYY_MM_DD_HH24');
113        |              SELECT value FROM ${info.schema.quoted}.${UserDataTable.Name.quoted}
114        |                WHERE type_name = ${literal(info.typeName)} AND key = ${literal(SftUserData.MainTableSpace.key)}
115        |                INTO partition_tablespace;
116        |              IF partition_tablespace IS NULL THEN
117        |                index_tablespace := '';
118        |                partition_tablespace := '${info.tables.mainPartitions.storage.opts}';
119        |              ELSE
120        |                index_tablespace := ' USING INDEX TABLESPACE '|| quote_ident(partition_tablespace);
121        |                partition_tablespace := '${info.tables.mainPartitions.storage.opts} TABLESPACE ' ||
122        |                  quote_ident(partition_tablespace);
123        |              END IF;
124        |
125        |              -- check for existing partition and switch to spill table instead
126        |              SELECT EXISTS(SELECT FROM pg_tables WHERE schemaname = ${info.schema.asLiteral} AND tablename = partition_name)
127        |                INTO pexists;
128        |              IF pexists THEN
129        |                partition_parent := ${info.tables.spillPartitions.name.asLiteral};
130        |                partition_name := partition_parent || '_' || to_char(partition_start, 'YYYY_MM_DD_HH24');
131        |              END IF;
132        |            ELSE
133        |              partition_start := ${info.schema.quoted}.truncate_to_ten_minutes(min_dtg);
134        |              partition_end := partition_start + INTERVAL '10 MINUTES';
135        |              partition_parent := ${info.tables.writeAheadPartitions.name.asLiteral};
136        |              partition_name := partition_parent || '_' || to_char(partition_start, 'YYYY_MM_DD_HH24_MI');
137        |              SELECT value FROM ${info.schema.quoted}.${UserDataTable.Name.quoted}
138        |                WHERE type_name = ${literal(info.typeName)} AND key = ${literal(SftUserData.WriteAheadPartitionsTableSpace.key)}
139        |                INTO partition_tablespace;
140        |              IF partition_tablespace IS NULL THEN
141        |                index_tablespace := '';
142        |                partition_tablespace := '${info.tables.writeAheadPartitions.storage.opts}';
143        |              ELSE
144        |                index_tablespace := ' USING INDEX TABLESPACE '|| quote_ident(partition_tablespace);
145        |                partition_tablespace := '${info.tables.writeAheadPartitions.storage.opts} TABLESPACE ' ||
146        |                  quote_ident(partition_tablespace);
147        |              END IF;
148        |            END IF;
149        |
150        |            RAISE INFO '% Writing to partition %', timeofday()::timestamp, partition_name;
151        |
152        |            SELECT EXISTS(SELECT FROM pg_tables WHERE schemaname = ${info.schema.asLiteral} AND tablename = partition_name)
153        |              INTO pexists;
154        |
155        |            -- create the partition table if it doesn't exist
156        |            -- note: normally the partition will not exist unless time-latent data was inserted
157        |            -- we create it unattached so that it doesn't lock the _recent table on insert
158        |            -- then we attach it after inserting the rows
159        |            -- since this is all within a transaction it should all happen "at once"
160        |            -- see https://www.postgresql.org/docs/13/ddl-partitioning.html#DDL-PARTITIONING-DECLARATIVE-MAINTENANCE
161        |
162        |            IF pexists THEN
163        |              -- copy rows from write ahead table to partition table
164        |              RAISE INFO '% Copying rows to partition %', timeofday()::timestamp, partition_name;
165        |              EXECUTE 'INSERT INTO ${info.schema.quoted}.' || quote_ident(partition_name) ||
166        |                ' SELECT * FROM ' || quote_ident(write_ahead.name) ||
167        |                '   WHERE ${info.cols.dtg.quoted} >= ' || quote_literal(partition_start) ||
168        |                '     AND ${info.cols.dtg.quoted} < ' || quote_literal(partition_end) ||
169        |                '   ORDER BY _st_sortablehash(${info.cols.geom.quoted})' ||
170        |                '   ON CONFLICT DO NOTHING';
171        |              GET DIAGNOSTICS unsorted_count := ROW_COUNT;
172        |            ELSE
173        |              RAISE INFO '% Creating partition with insert % (unattached)', timeofday()::timestamp, partition_name;
174        |
175        |              SELECT value FROM ${info.schema.quoted}.${UserDataTable.Name.quoted}
176        |                WHERE type_name = ${literal(info.typeName)} AND key = ${literal(SftUserData.WalLogEnabled.key)}
177        |                INTO table_wa_logging;
178        |              IF table_wa_logging IS NULL OR table_wa_logging::boolean THEN
179        |                table_wa_logging := '';
180        |              ELSE
181        |                table_wa_logging := 'UNLOGGED ';
182        |              END IF;
183        |
184        |              -- upper bounds are exclusive
185        |              -- use "create table as" (vs create then insert) for performance benefits related to WAL skipping
186        |              EXECUTE 'CREATE ' || table_wa_logging || 'TABLE ${info.schema.quoted}.' || quote_ident(partition_name) ||
187        |                partition_tablespace || ' AS SELECT * FROM ' || quote_ident(write_ahead.name) ||
188        |                '   WHERE ${info.cols.dtg.quoted} >= ' || quote_literal(partition_start) ||
189        |                '     AND ${info.cols.dtg.quoted} < ' || quote_literal(partition_end) ||
190        |                '   ORDER BY _st_sortablehash(${info.cols.geom.quoted})';
191        |              GET DIAGNOSTICS unsorted_count := ROW_COUNT;
192        |              EXECUTE 'ALTER TABLE ${info.schema.quoted}.' || quote_ident(partition_name) ||
193        |                ' ADD CONSTRAINT ' || quote_ident(partition_name || '_pkey') ||
194        |                ' PRIMARY KEY (${info.cols.fid.quoted}, ${info.cols.dtg.quoted})' || index_tablespace;
195        |              -- creating a constraint allows it to be attached to the parent without any additional checks
196        |              EXECUTE 'ALTER TABLE  ${info.schema.quoted}.' || quote_ident(partition_name) ||
197        |                ' ADD CONSTRAINT ' || quote_ident(partition_name || '_constraint') ||
198        |                ' CHECK ( ${info.cols.dtg.quoted} >= ' || quote_literal(partition_start) ||
199        |                ' AND ${info.cols.dtg.quoted} < ' || quote_literal(partition_end) || ' )';
200        |            END IF;
201        |            RAISE INFO '% Done writing % rows to partition %', timeofday()::timestamp, unsorted_count, partition_name;
202        |
203        |            IF partition_parent = ${info.tables.spillPartitions.name.asLiteral} THEN
204        |              -- store record of unsorted row counts which could negatively impact BRIN index scans
205        |              INSERT INTO ${info.tables.sortQueue.name.qualified}(partition_name, unsorted_count, enqueued)
206        |                VALUES (partition_name, unsorted_count, now());
207        |              RAISE NOTICE 'Inserting % rows into spill partition %, queries may be impacted',
208        |                unsorted_count, partition_name;
209        |            END IF;
210        |
211        |            -- attach the partition table to the parent
212        |            IF NOT pexists THEN
213        |              RAISE INFO '% Attaching partition % to parent', timeofday()::timestamp, partition_name;
214        |              EXECUTE 'ALTER TABLE ${info.schema.quoted}.' || quote_ident(partition_parent) ||
215        |                ' ATTACH PARTITION ${info.schema.quoted}.' || quote_ident(partition_name) ||
216        |                ' FOR VALUES FROM (' || quote_literal(partition_start) ||
217        |                ') TO (' || quote_literal(partition_end) || ' );';
218        |              -- once the table is attached we can drop the redundant constraint
219        |              -- however, this requires ACCESS EXCLUSIVE - since constraints are only checked on inserts
220        |              -- or updates, and partition tables are 'immutable' (only written to once), it shouldn't
221        |              -- affect anything to leave it. note that for 'spill' tables, there may be some redundant checks
222        |              -- EXECUTE 'ALTER TABLE ${info.schema.quoted}.' || quote_ident(partition_name) ||
223        |              --  ' DROP CONSTRAINT ' || quote_ident(partition_name || '_constraint');
224        |              RAISE NOTICE 'A partition has been created %', partition_name;
225        |            END IF;
226        |
227        |            -- mark the partition to be analyzed in a separate thread
228        |            INSERT INTO ${info.tables.analyzeQueue.name.qualified}(partition_name, enqueued)
229        |              VALUES (partition_name, now());
230        |          END LOOP;
231        |
232        |          -- Commit the copies and retirement marker together. Cleanup drops the originals,
233        |          -- or rotation deletes them before reusing the table.
234        |          INSERT INTO ${info.tables.writeAheadRetirements.name.qualified}(source_partition, partition_oid, enqueued)
235        |            VALUES (${literal(info.schema.quoted + ".")} || quote_ident(write_ahead.name), write_ahead.oid, now());
236        |
237        |        END IF;
238        |
239        |        COMMIT; -- releases the lock
240        |      END LOOP;
241        |      CALL ${info.schema.quoted}.${CleanWriteAheadRetirements.name(info).quoted}();
242        |    END;
243        |  $$BODY$$;
244        |""".stripMargin
