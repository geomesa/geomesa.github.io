30        |      min_dtg timestamp without time zone;         -- min date in our partitioned tables
31        |      main_cutoff timestamp without time zone;     -- max age of the records for main tables
32        |      write_ahead record;
33        |      partition_size int;                          -- size of the partition, in hours
34        |      partition_start timestamp without time zone; -- start bounds for the partition we're writing
35        |      partition_end timestamp without time zone;   -- end bounds for the partition we're writing
36        |      partition_name text;                         -- partition table name
37        |      partition_parent text;                       -- partition parent table name
38        |      partition_tablespace text;                   -- partition tablespace
39        |      main_tablespace text;                        -- configured tablespace for main partitions
40        |      wa_tablespace text;                          -- configured tablespace for write ahead partitions
41        |      index_tablespace text;                       -- index tablespace
42        |      table_wa_logging text;                       -- wa log options
43        |      pexists boolean;                             -- table exists check
44        |      unsorted_count bigint;
45        |    BEGIN
46        |      -- constants
47        |      ${UserDataConfig.select(info,
48                 "partition_size" -> SftUserData.IntervalHours,
49                 "main_tablespace" -> SftUserData.MainTableSpace,
50                 "wa_tablespace" -> SftUserData.WriteAheadPartitionsTableSpace,
51                 "table_wa_logging" -> SftUserData.WalLogEnabled)}
52        |      partition_size := COALESCE(partition_size, ${SftUserData.IntervalHours.default});
53        |      ${UserDataConfig.walLoggingClause("table_wa_logging")}
54        |      ${UserDataConfig.release}
55        |
56        |      main_cutoff := ${info.schema.quoted}.truncate_to_partition(cur_time, partition_size) - make_interval(hours => partition_size);
57        |
58        |      SELECT value from ${info.schema.quoted}.${SequenceTable.Name.quoted}
59        |        WHERE type_name = ${literal(info.typeName)} INTO seq_val;
60        |      write_partition := ${literal(info.tables.writeAhead.name.raw + "_")} || lpad(seq_val::text, 3, '0');
61        |
62        |      -- check for write ahead partitions and move the data into the time partitioned tables
63        |      FOR write_ahead IN
64        |        SELECT pg_class.relname AS name, pg_class.oid AS oid
65        |          FROM pg_catalog.pg_inherits
66        |          INNER JOIN pg_catalog.pg_class ON (pg_inherits.inhrelid = pg_class.oid)
67        |          INNER JOIN pg_catalog.pg_namespace ON (pg_class.relnamespace = pg_namespace.oid)
68        |          WHERE inhparent = ${info.tables.writeAhead.name.asRegclass}
69        |          AND relname != write_partition
70        |          AND NOT EXISTS (SELECT FROM ${info.tables.writeAheadRetirements.name.qualified} retired
71        |            WHERE retired.partition_oid = pg_class.oid)
72        |          ORDER BY name
73        |      LOOP
74        |
75        |        RAISE INFO '% Checking write ahead table %', timeofday()::timestamp, write_ahead.name;
76        |        -- lock parent tables up-front to avoid deadlocks when attaching children - this won't prevent reads or writes
77        |        LOCK TABLE ONLY ${info.tables.writeAheadPartitions.name.qualified} IN SHARE UPDATE EXCLUSIVE MODE;
78        |        LOCK TABLE ONLY ${info.tables.mainPartitions.name.qualified} IN SHARE UPDATE EXCLUSIVE MODE;
79        |        LOCK TABLE ONLY ${info.tables.spillPartitions.name.qualified} IN SHARE UPDATE EXCLUSIVE MODE;
80        |        -- Recheck after acquiring the parent locks: another migration may have retired this table.
81        |        IF NOT EXISTS (SELECT FROM pg_class WHERE oid = write_ahead.oid)
82        |          OR EXISTS (SELECT FROM ${info.tables.writeAheadRetirements.name.qualified} WHERE partition_oid = write_ahead.oid) THEN
83        |          COMMIT;
84        |          CONTINUE;
85        |        END IF;
86        |        -- get a lock on the table - this mode won't prevent reads but will prevent writes
87        |        -- (there shouldn't be any writes though) and will synchronize this method
88        |        EXECUTE 'LOCK TABLE ${info.schema.quoted}.' || quote_ident(write_ahead.name) ||
89        |          ' IN SHARE ROW EXCLUSIVE MODE';
90        |        -- Rotation may have reused a retired table since this loop enumerated the children.
91        |        SELECT value FROM ${info.schema.quoted}.${SequenceTable.Name.quoted}
92        |          WHERE type_name = ${literal(info.typeName)} INTO seq_val;
93        |        IF write_ahead.name = ${literal(info.tables.writeAhead.name.raw + "_")} || lpad(seq_val::text, 3, '0') THEN
94        |          COMMIT;
95        |          CONTINUE;
96        |        END IF;
97        |        RAISE INFO '% Locked write ahead table % for migration', timeofday()::timestamp, write_ahead.name;
98        |
99        |        -- wait until the table doesn't contain any recent records
100        |        EXECUTE 'SELECT EXISTS(SELECT 1 FROM ${info.schema.quoted}.' || quote_ident(write_ahead.name) ||
101        |          ' WHERE ${info.cols.dtg.quoted} >= ' || quote_literal(${info.schema.quoted}.truncate_to_ten_minutes(cur_time)) || ')'
102        |          INTO pexists;
103        |
104        |        IF pexists THEN
105        |          -- should only happen if data is inserted with timestamps from the future
106        |          RAISE NOTICE '% Skipping write ahead table % due to min date', timeofday()::timestamp, write_ahead.name;
107        |        ELSE
108        |          partition_end := '-infinity'::timestamp without time zone;
109        |          LOOP
110        |            -- find the range of dates in the write ahead partition
111        |            EXECUTE 'SELECT min(${info.cols.dtg.quoted}) FROM ${info.schema.quoted}.' || quote_ident(write_ahead.name) ||
112        |              ' WHERE ${info.cols.dtg.quoted} >= ' || quote_literal(partition_end) INTO min_dtg;
113        |            EXIT WHEN min_dtg IS NULL;
114        |
115        |            -- calculate the partition bounds for the min date
116        |            IF min_dtg < main_cutoff THEN
117        |              partition_start := ${info.schema.quoted}.truncate_to_partition(min_dtg, partition_size);
118        |              partition_end := partition_start + make_interval(hours => partition_size);
119        |              partition_parent := ${info.tables.mainPartitions.name.asLiteral};
120        |              partition_name := partition_parent || '_' || to_char(partition_start, 'YYYY_MM_DD_HH24');
121        |              partition_tablespace := main_tablespace;
122        |              IF partition_tablespace IS NULL THEN
123        |                index_tablespace := '';
124        |                partition_tablespace := '${info.tables.mainPartitions.storage.opts}';
125        |              ELSE
126        |                index_tablespace := ' USING INDEX TABLESPACE '|| quote_ident(partition_tablespace);
127        |                partition_tablespace := '${info.tables.mainPartitions.storage.opts} TABLESPACE ' ||
128        |                  quote_ident(partition_tablespace);
129        |              END IF;
130        |
131        |              -- check for existing partition and switch to spill table instead
132        |              SELECT EXISTS(SELECT FROM pg_tables WHERE schemaname = ${info.schema.asLiteral} AND tablename = partition_name)
133        |                INTO pexists;
134        |              IF pexists THEN
135        |                partition_parent := ${info.tables.spillPartitions.name.asLiteral};
136        |                partition_name := partition_parent || '_' || to_char(partition_start, 'YYYY_MM_DD_HH24');
137        |              END IF;
138        |            ELSE
139        |              partition_start := ${info.schema.quoted}.truncate_to_ten_minutes(min_dtg);
140        |              partition_end := partition_start + INTERVAL '10 MINUTES';
141        |              partition_parent := ${info.tables.writeAheadPartitions.name.asLiteral};
142        |              partition_name := partition_parent || '_' || to_char(partition_start, 'YYYY_MM_DD_HH24_MI');
143        |              partition_tablespace := wa_tablespace;
144        |              IF partition_tablespace IS NULL THEN
145        |                index_tablespace := '';
146        |                partition_tablespace := '${info.tables.writeAheadPartitions.storage.opts}';
147        |              ELSE
148        |                index_tablespace := ' USING INDEX TABLESPACE '|| quote_ident(partition_tablespace);
149        |                partition_tablespace := '${info.tables.writeAheadPartitions.storage.opts} TABLESPACE ' ||
150        |                  quote_ident(partition_tablespace);
151        |              END IF;
152        |            END IF;
153        |
154        |            RAISE INFO '% Writing to partition %', timeofday()::timestamp, partition_name;
155        |
156        |            SELECT EXISTS(SELECT FROM pg_tables WHERE schemaname = ${info.schema.asLiteral} AND tablename = partition_name)
157        |              INTO pexists;
158        |
159        |            -- create the partition table if it doesn't exist
160        |            -- note: normally the partition will not exist unless time-latent data was inserted
161        |            -- we create it unattached so that it doesn't lock the _recent table on insert
162        |            -- then we attach it after inserting the rows
163        |            -- since this is all within a transaction it should all happen "at once"
164        |            -- see https://www.postgresql.org/docs/13/ddl-partitioning.html#DDL-PARTITIONING-DECLARATIVE-MAINTENANCE
165        |
166        |            IF pexists THEN
167        |              -- copy rows from write ahead table to partition table
168        |              RAISE INFO '% Copying rows to partition %', timeofday()::timestamp, partition_name;
169        |              EXECUTE 'INSERT INTO ${info.schema.quoted}.' || quote_ident(partition_name) ||
170        |                ' SELECT * FROM ' || quote_ident(write_ahead.name) ||
171        |                '   WHERE ${info.cols.dtg.quoted} >= ' || quote_literal(partition_start) ||
172        |                '     AND ${info.cols.dtg.quoted} < ' || quote_literal(partition_end) ||
173        |                '   ORDER BY _st_sortablehash(${info.cols.geom.quoted})' ||
174        |                '   ON CONFLICT DO NOTHING';
175        |              GET DIAGNOSTICS unsorted_count := ROW_COUNT;
176        |            ELSE
177        |              RAISE INFO '% Creating partition with insert % (unattached)', timeofday()::timestamp, partition_name;
178        |
179        |              -- upper bounds are exclusive
180        |              -- use "create table as" (vs create then insert) for performance benefits related to WAL skipping
181        |              EXECUTE 'CREATE ' || table_wa_logging || 'TABLE ${info.schema.quoted}.' || quote_ident(partition_name) ||
182        |                partition_tablespace || ' AS SELECT * FROM ' || quote_ident(write_ahead.name) ||
183        |                '   WHERE ${info.cols.dtg.quoted} >= ' || quote_literal(partition_start) ||
184        |                '     AND ${info.cols.dtg.quoted} < ' || quote_literal(partition_end) ||
185        |                '   ORDER BY _st_sortablehash(${info.cols.geom.quoted})';
186        |              GET DIAGNOSTICS unsorted_count := ROW_COUNT;
187        |              EXECUTE 'ALTER TABLE ${info.schema.quoted}.' || quote_ident(partition_name) ||
188        |                ' ADD CONSTRAINT ' || quote_ident(partition_name || '_pkey') ||
189        |                ' PRIMARY KEY (${info.cols.fid.quoted}, ${info.cols.dtg.quoted})' || index_tablespace;
190        |              -- creating a constraint allows it to be attached to the parent without any additional checks
191        |              EXECUTE 'ALTER TABLE  ${info.schema.quoted}.' || quote_ident(partition_name) ||
192        |                ' ADD CONSTRAINT ' || quote_ident(partition_name || '_constraint') ||
193        |                ' CHECK ( ${info.cols.dtg.quoted} >= ' || quote_literal(partition_start) ||
194        |                ' AND ${info.cols.dtg.quoted} < ' || quote_literal(partition_end) || ' )';
195        |            END IF;
196        |            RAISE INFO '% Done writing % rows to partition %', timeofday()::timestamp, unsorted_count, partition_name;
197        |
198        |            IF partition_parent = ${info.tables.spillPartitions.name.asLiteral} THEN
199        |              -- store record of unsorted row counts which could negatively impact BRIN index scans
200        |              INSERT INTO ${info.tables.sortQueue.name.qualified}(partition_name, unsorted_count, enqueued)
201        |                VALUES (partition_name, unsorted_count, now());
202        |              RAISE NOTICE 'Inserting % rows into spill partition %, queries may be impacted',
203        |                unsorted_count, partition_name;
204        |            END IF;
205        |
206        |            -- attach the partition table to the parent
207        |            IF NOT pexists THEN
208        |              RAISE INFO '% Attaching partition % to parent', timeofday()::timestamp, partition_name;
209        |              EXECUTE 'ALTER TABLE ${info.schema.quoted}.' || quote_ident(partition_parent) ||
210        |                ' ATTACH PARTITION ${info.schema.quoted}.' || quote_ident(partition_name) ||
211        |                ' FOR VALUES FROM (' || quote_literal(partition_start) ||
212        |                ') TO (' || quote_literal(partition_end) || ' );';

214        |              -- however, this requires ACCESS EXCLUSIVE - since constraints are only checked on inserts
215        |              -- or updates, and partition tables are 'immutable' (only written to once), it shouldn't
216        |              -- affect anything to leave it. note that for 'spill' tables, there may be some redundant checks
217        |              -- EXECUTE 'ALTER TABLE ${info.schema.quoted}.' || quote_ident(partition_name) ||
218        |              --  ' DROP CONSTRAINT ' || quote_ident(partition_name || '_constraint');
219        |              RAISE NOTICE 'A partition has been created %', partition_name;
220        |            END IF;
221        |
222        |            -- mark the partition to be analyzed in a separate thread
223        |            INSERT INTO ${info.tables.analyzeQueue.name.qualified}(partition_name, enqueued)
224        |              VALUES (partition_name, now());
225        |          END LOOP;
226        |
227        |          -- Commit the copies and retirement marker together. Cleanup drops the originals,
228        |          -- or rotation deletes them before reusing the table.
229        |          INSERT INTO ${info.tables.writeAheadRetirements.name.qualified}(source_partition, partition_oid, enqueued)
230        |            VALUES (${literal(info.schema.quoted + ".")} || quote_ident(write_ahead.name), write_ahead.oid, now());
231        |
232        |        END IF;
233        |
234        |        COMMIT; -- releases the lock
235        |      END LOOP;
236        |      CALL ${info.schema.quoted}.${CleanWriteAheadRetirements.name(info).quoted}();
237        |    END;
238        |  $$BODY$$;
