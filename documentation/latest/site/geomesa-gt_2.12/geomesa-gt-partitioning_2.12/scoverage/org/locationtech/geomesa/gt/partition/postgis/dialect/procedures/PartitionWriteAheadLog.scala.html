28        |      seq_val smallint;
29        |      write_partition text;                        -- current partition receiving writes
30        |      min_dtg timestamp without time zone;         -- min date in our partitioned tables
31        |      main_cutoff timestamp without time zone;     -- max age of the records for main tables
32        |      write_ahead record;
33        |      partition_size int;                          -- size of the partition, in hours
34        |      partition_start timestamp without time zone; -- start bounds for the partition we're writing
35        |      partition_end timestamp without time zone;   -- end bounds for the partition we're writing
36        |      partition_name text;                         -- partition table name
37        |      partition_parent text;                       -- partition parent table name
38        |      partition_tablespace text;                   -- partition tablespace
39        |      index_tablespace text;                       -- index tablespace
40        |      table_wa_logging text;                       -- wa log options
41        |      pexists boolean;                             -- table exists check
42        |      unsorted_count bigint;
43        |    BEGIN
44        |      -- constants
45        |      SELECT COALESCE(
46        |        (SELECT value::int FROM ${info.schema.quoted}.${UserDataTable.Name.quoted}
47        |          WHERE type_name = ${literal(info.typeName)} AND key = ${literal(SftUserData.IntervalHours.key)}),
48        |        ${SftUserData.IntervalHours.default})
49        |        INTO partition_size;
50        |      main_cutoff := ${info.schema.quoted}.truncate_to_partition(cur_time, partition_size) - make_interval(hours => partition_size);
51        |
52        |      SELECT value from ${info.schema.quoted}.${SequenceTable.Name.quoted}
53        |        WHERE type_name = ${literal(info.typeName)} INTO seq_val;
54        |      write_partition := ${literal(info.tables.writeAhead.name.raw + "_")} || lpad(seq_val::text, 3, '0');
55        |
56        |      -- check for write ahead partitions and move the data into the time partitioned tables
57        |      FOR write_ahead IN
58        |        SELECT pg_class.relname AS name, pg_class.oid AS oid
59        |          FROM pg_catalog.pg_inherits
60        |          INNER JOIN pg_catalog.pg_class ON (pg_inherits.inhrelid = pg_class.oid)
61        |          INNER JOIN pg_catalog.pg_namespace ON (pg_class.relnamespace = pg_namespace.oid)
62        |          WHERE inhparent = ${info.tables.writeAhead.name.asRegclass}
63        |          AND relname != write_partition
64        |          AND NOT EXISTS (SELECT FROM ${info.tables.writeAheadRetirements.name.qualified} retired
65        |            WHERE retired.partition_oid = pg_class.oid)
66        |          ORDER BY name
67        |      LOOP
68        |
69        |        RAISE INFO '% Checking write ahead table %', timeofday()::timestamp, write_ahead.name;
70        |        -- lock parent tables up-front to avoid deadlocks when attaching children - this won't prevent reads or writes
71        |        LOCK TABLE ONLY ${info.tables.writeAheadPartitions.name.qualified} IN SHARE UPDATE EXCLUSIVE MODE;
72        |        LOCK TABLE ONLY ${info.tables.mainPartitions.name.qualified} IN SHARE UPDATE EXCLUSIVE MODE;
73        |        LOCK TABLE ONLY ${info.tables.spillPartitions.name.qualified} IN SHARE UPDATE EXCLUSIVE MODE;
74        |        -- Recheck after acquiring the parent locks: another migration may have retired this table.
75        |        IF NOT EXISTS (SELECT FROM pg_class WHERE oid = write_ahead.oid)
76        |          OR EXISTS (SELECT FROM ${info.tables.writeAheadRetirements.name.qualified} WHERE partition_oid = write_ahead.oid) THEN
77        |          COMMIT;
78        |          CONTINUE;
79        |        END IF;

81        |        -- (there shouldn't be any writes though) and will synchronize this method
82        |        EXECUTE 'LOCK TABLE ${info.schema.quoted}.' || quote_ident(write_ahead.name) ||
83        |          ' IN SHARE ROW EXCLUSIVE MODE';
84        |        -- Rotation may have reused a retired table since this loop enumerated the children.
85        |        SELECT value FROM ${info.schema.quoted}.${SequenceTable.Name.quoted}
86        |          WHERE type_name = ${literal(info.typeName)} INTO seq_val;
87        |        IF write_ahead.name = ${literal(info.tables.writeAhead.name.raw + "_")} || lpad(seq_val::text, 3, '0') THEN
88        |          COMMIT;
89        |          CONTINUE;
90        |        END IF;

92        |
93        |        -- wait until the table doesn't contain any recent records
94        |        EXECUTE 'SELECT EXISTS(SELECT 1 FROM ${info.schema.quoted}.' || quote_ident(write_ahead.name) ||
95        |          ' WHERE ${info.cols.dtg.quoted} >= ' || quote_literal(${info.schema.quoted}.truncate_to_ten_minutes(cur_time)) || ')'
96        |          INTO pexists;
97        |
98        |        IF pexists THEN
99        |          -- should only happen if data is inserted with timestamps from the future
100        |          RAISE NOTICE '% Skipping write ahead table % due to min date', timeofday()::timestamp, write_ahead.name;
101        |        ELSE
102        |          partition_end := '-infinity'::timestamp without time zone;
103        |          LOOP
104        |            -- find the range of dates in the write ahead partition
105        |            EXECUTE 'SELECT min(${info.cols.dtg.quoted}) FROM ${info.schema.quoted}.' || quote_ident(write_ahead.name) ||
106        |              ' WHERE ${info.cols.dtg.quoted} >= ' || quote_literal(partition_end) INTO min_dtg;
107        |            EXIT WHEN min_dtg IS NULL;
108        |
109        |            -- calculate the partition bounds for the min date
110        |            IF min_dtg < main_cutoff THEN
111        |              partition_start := ${info.schema.quoted}.truncate_to_partition(min_dtg, partition_size);
112        |              partition_end := partition_start + make_interval(hours => partition_size);
113        |              partition_parent := ${info.tables.mainPartitions.name.asLiteral};
114        |              partition_name := partition_parent || '_' || to_char(partition_start, 'YYYY_MM_DD_HH24');
115        |              SELECT value FROM ${info.schema.quoted}.${UserDataTable.Name.quoted}
116        |                WHERE type_name = ${literal(info.typeName)} AND key = ${literal(SftUserData.MainTableSpace.key)}
117        |                INTO partition_tablespace;
118        |              IF partition_tablespace IS NULL THEN
119        |                index_tablespace := '';
120        |                partition_tablespace := '${info.tables.mainPartitions.storage.opts}';
121        |              ELSE
122        |                index_tablespace := ' USING INDEX TABLESPACE '|| quote_ident(partition_tablespace);
123        |                partition_tablespace := '${info.tables.mainPartitions.storage.opts} TABLESPACE ' ||
124        |                  quote_ident(partition_tablespace);
125        |              END IF;
126        |
127        |              -- check for existing partition and switch to spill table instead
128        |              SELECT EXISTS(SELECT FROM pg_tables WHERE schemaname = ${info.schema.asLiteral} AND tablename = partition_name)
129        |                INTO pexists;
130        |              IF pexists THEN
131        |                partition_parent := ${info.tables.spillPartitions.name.asLiteral};
132        |                partition_name := partition_parent || '_' || to_char(partition_start, 'YYYY_MM_DD_HH24');
133        |              END IF;
134        |            ELSE
135        |              partition_start := ${info.schema.quoted}.truncate_to_ten_minutes(min_dtg);
136        |              partition_end := partition_start + INTERVAL '10 MINUTES';
137        |              partition_parent := ${info.tables.writeAheadPartitions.name.asLiteral};
138        |              partition_name := partition_parent || '_' || to_char(partition_start, 'YYYY_MM_DD_HH24_MI');
139        |              SELECT value FROM ${info.schema.quoted}.${UserDataTable.Name.quoted}
140        |                WHERE type_name = ${literal(info.typeName)} AND key = ${literal(SftUserData.WriteAheadPartitionsTableSpace.key)}
141        |                INTO partition_tablespace;
142        |              IF partition_tablespace IS NULL THEN
143        |                index_tablespace := '';
144        |                partition_tablespace := '${info.tables.writeAheadPartitions.storage.opts}';
145        |              ELSE
146        |                index_tablespace := ' USING INDEX TABLESPACE '|| quote_ident(partition_tablespace);
147        |                partition_tablespace := '${info.tables.writeAheadPartitions.storage.opts} TABLESPACE ' ||
148        |                  quote_ident(partition_tablespace);
149        |              END IF;
150        |            END IF;
151        |
152        |            RAISE INFO '% Writing to partition %', timeofday()::timestamp, partition_name;
153        |
154        |            SELECT EXISTS(SELECT FROM pg_tables WHERE schemaname = ${info.schema.asLiteral} AND tablename = partition_name)
155        |              INTO pexists;
156        |
157        |            -- create the partition table if it doesn't exist
158        |            -- note: normally the partition will not exist unless time-latent data was inserted
159        |            -- we create it unattached so that it doesn't lock the _recent table on insert
160        |            -- then we attach it after inserting the rows
161        |            -- since this is all within a transaction it should all happen "at once"
162        |            -- see https://www.postgresql.org/docs/13/ddl-partitioning.html#DDL-PARTITIONING-DECLARATIVE-MAINTENANCE
163        |
164        |            IF pexists THEN
165        |              -- copy rows from write ahead table to partition table
166        |              RAISE INFO '% Copying rows to partition %', timeofday()::timestamp, partition_name;
167        |              EXECUTE 'INSERT INTO ${info.schema.quoted}.' || quote_ident(partition_name) ||
168        |                ' SELECT * FROM ' || quote_ident(write_ahead.name) ||
169        |                '   WHERE ${info.cols.dtg.quoted} >= ' || quote_literal(partition_start) ||
170        |                '     AND ${info.cols.dtg.quoted} < ' || quote_literal(partition_end) ||
171        |                '   ORDER BY _st_sortablehash(${info.cols.geom.quoted})' ||
172        |                '   ON CONFLICT DO NOTHING';
173        |              GET DIAGNOSTICS unsorted_count := ROW_COUNT;
174        |            ELSE
175        |              RAISE INFO '% Creating partition with insert % (unattached)', timeofday()::timestamp, partition_name;
176        |
177        |              SELECT value FROM ${info.schema.quoted}.${UserDataTable.Name.quoted}
178        |                WHERE type_name = ${literal(info.typeName)} AND key = ${literal(SftUserData.WalLogEnabled.key)}
179        |                INTO table_wa_logging;
180        |              IF table_wa_logging IS NULL OR table_wa_logging::boolean THEN
181        |                table_wa_logging := '';
182        |              ELSE
183        |                table_wa_logging := 'UNLOGGED ';
184        |              END IF;
185        |
186        |              -- upper bounds are exclusive
187        |              -- use "create table as" (vs create then insert) for performance benefits related to WAL skipping
188        |              EXECUTE 'CREATE ' || table_wa_logging || 'TABLE ${info.schema.quoted}.' || quote_ident(partition_name) ||
189        |                partition_tablespace || ' AS SELECT * FROM ' || quote_ident(write_ahead.name) ||
190        |                '   WHERE ${info.cols.dtg.quoted} >= ' || quote_literal(partition_start) ||
191        |                '     AND ${info.cols.dtg.quoted} < ' || quote_literal(partition_end) ||
192        |                '   ORDER BY _st_sortablehash(${info.cols.geom.quoted})';
193        |              GET DIAGNOSTICS unsorted_count := ROW_COUNT;
194        |              EXECUTE 'ALTER TABLE ${info.schema.quoted}.' || quote_ident(partition_name) ||
195        |                ' ADD CONSTRAINT ' || quote_ident(partition_name || '_pkey') ||
196        |                ' PRIMARY KEY (${info.cols.fid.quoted}, ${info.cols.dtg.quoted})' || index_tablespace;
197        |              -- creating a constraint allows it to be attached to the parent without any additional checks
198        |              EXECUTE 'ALTER TABLE  ${info.schema.quoted}.' || quote_ident(partition_name) ||
199        |                ' ADD CONSTRAINT ' || quote_ident(partition_name || '_constraint') ||
200        |                ' CHECK ( ${info.cols.dtg.quoted} >= ' || quote_literal(partition_start) ||
201        |                ' AND ${info.cols.dtg.quoted} < ' || quote_literal(partition_end) || ' )';
202        |            END IF;
203        |            RAISE INFO '% Done writing % rows to partition %', timeofday()::timestamp, unsorted_count, partition_name;
204        |
205        |            IF partition_parent = ${info.tables.spillPartitions.name.asLiteral} THEN
206        |              -- store record of unsorted row counts which could negatively impact BRIN index scans
207        |              INSERT INTO ${info.tables.sortQueue.name.qualified}(partition_name, unsorted_count, enqueued)

210        |                unsorted_count, partition_name;
211        |            END IF;
212        |
213        |            -- attach the partition table to the parent
214        |            IF NOT pexists THEN
215        |              RAISE INFO '% Attaching partition % to parent', timeofday()::timestamp, partition_name;
216        |              EXECUTE 'ALTER TABLE ${info.schema.quoted}.' || quote_ident(partition_parent) ||
217        |                ' ATTACH PARTITION ${info.schema.quoted}.' || quote_ident(partition_name) ||
218        |                ' FOR VALUES FROM (' || quote_literal(partition_start) ||
219        |                ') TO (' || quote_literal(partition_end) || ' );';
220        |              -- once the table is attached we can drop the redundant constraint
221        |              -- however, this requires ACCESS EXCLUSIVE - since constraints are only checked on inserts
222        |              -- or updates, and partition tables are 'immutable' (only written to once), it shouldn't

225        |              --  ' DROP CONSTRAINT ' || quote_ident(partition_name || '_constraint');
226        |              RAISE NOTICE 'A partition has been created %', partition_name;
227        |            END IF;
228        |
229        |            -- mark the partition to be analyzed in a separate thread
230        |            INSERT INTO ${info.tables.analyzeQueue.name.qualified}(partition_name, enqueued)
231        |              VALUES (partition_name, now());

234        |          -- Commit the copies and retirement marker together. Cleanup drops the originals,
235        |          -- or rotation deletes them before reusing the table.
236        |          INSERT INTO ${info.tables.writeAheadRetirements.name.qualified}(source_partition, partition_oid, enqueued)
237        |            VALUES (${literal(info.schema.quoted + ".")} || quote_ident(write_ahead.name), write_ahead.oid, now());
238        |
239        |        END IF;
240        |
241        |        COMMIT; -- releases the lock
242        |      END LOOP;
243        |      CALL ${info.schema.quoted}.${CleanWriteAheadRetirements.name(info).quoted}();
244        |    END;
245        |  $$BODY$$;
