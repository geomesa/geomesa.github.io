12 /**
13  * Schedules concurrent detachments, recovers interrupted detachments, and drops detached expired partitions
14  */
15 object CleanAgedOffPartitions extends SqlProcedure with CronSchedule {
16 
17   override def name(info: TypeInfo): FunctionName = FunctionName(s"${info.typeIdentifier}_clean_age_off")
18 
19   override def jobName(info: TypeInfo): SqlLiteral = SqlLiteral(s"${info.typeIdentifier}-clean-age-off")
20 
21   def detachJobName(info: TypeInfo, parent: TableConfig): SqlLiteral =
22     SqlLiteral(s"${info.typeIdentifier}-age-off-${parent.name.raw}")
23 
24   override protected def schedule(info: TypeInfo): SqlLiteral = SqlLiteral("* * * * *")
25 
26   override protected def invocation(info: TypeInfo): SqlLiteral =
27     SqlLiteral(s"CALL ${info.schema.quoted}.${name(info).quoted}()")
28 
29   override protected def createStatements(info: TypeInfo): Seq[String] =
30     Seq(proc(info)) ++ super.createStatements(info)
31 
32   override protected def dropStatements(info: TypeInfo): Seq[String] = {
33     val jobs = Seq(info.tables.mainPartitions, info.tables.spillPartitions)
34         .map(parent => unscheduleSql(detachJobName(info, parent).quoted)).mkString("\n")
35     Seq(s"DO $$$$ BEGIN $jobs END$$$$;") ++ super.dropStatements(info)
36   }
37 
38   private def proc(info: TypeInfo): String = {
39     val scheduleParents = Seq(info.tables.mainPartitions, info.tables.spillPartitions).map { parent =>
40       val job = detachJobName(info, parent).quoted
41       s"""SELECT queue.source_partition, inherits.inhdetachpending
42          |  INTO partition_name, detach_pending
43          |  FROM ${info.tables.ageOffQueue.name.qualified} queue
44          |  JOIN pg_catalog.pg_inherits inherits ON inherits.inhrelid = queue.partition_oid
45          |    AND inherits.inhparent = queue.parent_oid
46          |  WHERE queue.parent_oid = ${parent.name.asRegclass}
47          |  -- Only one detach may be pending per parent; finish it before starting another.
48          |  ORDER BY inherits.inhdetachpending DESC, queue.enqueued, queue.source_partition
49          |  LIMIT 1;
50          |IF FOUND THEN
51          |  detach_invocation := ${literal(s"ALTER TABLE ${parent.name.qualified} DETACH PARTITION ")} || partition_name ||
52          |    CASE WHEN detach_pending THEN ' FINALIZE' ELSE ' CONCURRENTLY' END;
53          |  PERFORM cron.schedule($job, '*/5 * * * *', detach_invocation);
54          |ELSIF EXISTS (SELECT FROM cron.job WHERE jobname = $job) THEN
55          |  PERFORM cron.unschedule($job);
56          |END IF;""".stripMargin
57     }.mkString("\n")
58     s"""CREATE OR REPLACE PROCEDURE ${info.schema.quoted}.${name(info).quoted}() LANGUAGE plpgsql AS
59        |  $$BODY$$
60        |    DECLARE
61        |      partition_name text;
62        |      detach_pending boolean;
63        |      detach_invocation text;
64        |    BEGIN
65        |      FOR partition_name IN
66        |        SELECT queue.source_partition FROM ${info.tables.ageOffQueue.name.qualified} queue
67        |        WHERE NOT EXISTS (SELECT FROM pg_catalog.pg_inherits WHERE inhrelid = queue.partition_oid)
68        |        ORDER BY queue.source_partition
69        |      LOOP
70        |        EXECUTE 'DROP TABLE IF EXISTS ' || partition_name;
71        |        DELETE FROM ${info.tables.ageOffQueue.name.qualified} WHERE source_partition = partition_name;
72        |      END LOOP;
73        |$scheduleParents
74        |    END;
75        |  $$BODY$$;
76        |""".stripMargin
77   }
78 }
