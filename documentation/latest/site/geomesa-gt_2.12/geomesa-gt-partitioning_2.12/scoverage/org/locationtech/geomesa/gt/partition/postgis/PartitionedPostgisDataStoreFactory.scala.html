39  *
40  * Delegates to a standard JDBC-based factory to create the underlying `JDBCDataStore`, then wraps it
41  * in a `PartitionedPostgisDataStore` to manage the hidden per-row visibility column. We can't just
42  * subclass the JDBC factory and wrap in `createDataStoreInternal`, as `JDBCDataStoreFactory.createDataStore`
43  * is final and returns `JDBCDataStore` (which is itself final)

113     override protected def createDataSource(params: java.util.Map[String, _], dialect: SQLDialect): DataSource = {
114       val auths = AuthUtils.getProvider(params, AuthsParam.lookupOpt(params).getOrElse("").split(",").filterNot(_.isEmpty).toSeq)
115 
116       val dataSource = super.createDataSource(params, dialect)
117 
118       val metrics = MetricsRegistryParam.lookupRegistry(params).flatMap { registry =>
119         var metrics: MetricsDataSource = null
120         try {
121           // try to re-create the datasource as a commons-dbcp2 instance so we can instrument it
122           val basic = dataSource match {
123             case ds: AbstractManageableDataSource => ds.unwrap(classOf[BasicDataSource])
124             case ds: BasicDataSource => ds
125             case ds => throw new UnsupportedOperationException(s"Unexpected data source class: ${ds.getClass.getName} $ds")
126           }
127           metrics = new MetricsDataSource("postgis")
128 
129           // these are the options that are configured in our parent factories
130           metrics.setDriverClassName(basic.getDriverClassName)
131           metrics.setUrl(basic.getUrl)
132           metrics.setUsername(basic.getUsername)
133           metrics.setPassword(basic.getPassword)
134           metrics.setMaxWait(Duration.ofMillis(basic.getMaxWait))
135           metrics.setMinIdle(basic.getMinIdle)
136           metrics.setMaxTotal(basic.getMaxActive)
137           metrics.setTestOnBorrow(basic.getTestOnBorrow)
138           metrics.setValidationQuery(basic.getValidationQuery)
139           metrics.setTestWhileIdle(basic.getTestWhileIdle)
140           metrics.setDurationBetweenEvictionRuns(Duration.ofMillis(basic.getTimeBetweenEvictionRunsMillis))
141           metrics.setMinEvictableIdle(Duration.ofMillis(basic.getMinEvictableIdleTimeMillis))
142           metrics.setNumTestsPerEvictionRun(basic.getNumTestsPerEvictionRun)
143           metrics.setAccessToUnderlyingConnectionAllowed(basic.isAccessToUnderlyingConnectionAllowed)
144           metrics.setMaxOpenPreparedStatements(basic.getMaxOpenPreparedStatements)
145           metrics.setPoolPreparedStatements(basic.isPoolPreparedStatements)
146 
147           // re-add connection properties, as we can't access those from the BasicDataSource
148           createConnectionOptions(params).foreach { case (k, v) => metrics.addConnectionProperty(k, v) }
149 
150           val registration = registry.register()
151 
152           tryBindPostgresMetrics(metrics, params)
153 
154           // we attach the registry to the ManagedDataSource b/c that's the only hook we have into datastore.dispose()
155           Some(new Dbcp2ManagedDataSource(metrics, registration))
156         } catch {
157           case NonFatal(e) =>
158             logger.warn("Unable to instrument data source, metrics will not be available:", e)
159             CloseWithLogging(Option(metrics))
160             None
161         }
162       }
163 
164       new SessionDataSource(metrics.getOrElse(dataSource), auths)
165     }

167     private def tryBindPostgresMetrics(dataSource: MetricsDataSource, params: java.util.Map[String, _]): Unit = {
168       try {
169         val conf =
170           MetricsRegistryConfigParam.lookupOpt(params)
171             .getOrElse(ConfigFactory.empty)
172             .withFallback(ConfigFactory.load("postgres-metrics"))
173             .getConfig("postgres")
174         if (conf.getBoolean("enabled")) {
175           val tags = Tags.of(conf.getConfig("tags").root().unwrapped().asScala.map { case (k, v) => Tag.of(k, v.toString) }.asJava)
176           val db =
177             Option(DATABASE.lookUp(params).asInstanceOf[String])
178               .orElse(Option(USER.lookUp(params).asInstanceOf[String]))
179               .getOrElse(throw new IllegalArgumentException(s"${DATABASE.key} was not found in the parameter map"))
180           if (PartitionedPostgisDataStoreFactory.bindings.add(db -> tags)) {
181             logger.info("Adding postgres metrics")
182             new PostgreSQLDatabaseMetrics(dataSource, db, tags).bindTo(Metrics.globalRegistry)
183           }
184         }
185       } catch {
186         case NonFatal(e) => logger.warn("Error binding postgres metrics:", e)
187       }
188     }

196     override protected def createDataStoreInternal(store: JDBCDataStore, baseParams: java.util.Map[String, _]): JDBCDataStore = {
197       val params = new java.util.HashMap[String, Any](baseParams)
198       // default to using prepared statements, if not specified
199       if (!params.containsKey(PreparedStatements.key)) {
200         params.put(PreparedStatements.key, java.lang.Boolean.TRUE)
201       }
202       // set default schema, if not specified - postgis store doesn't actually use its own default
203       if (!params.containsKey(PostgisNGDataStoreFactory.SCHEMA.key)) {
204         // need to set it in the store, as the key has already been processed
205         store.setDatabaseSchema("public")
206         // also set in the params for consistency, although it's not used anywhere
207         params.put(PostgisNGDataStoreFactory.SCHEMA.key, "public")
208       }
209       val roles = Option(ReadAccessRoles.lookUp(params).asInstanceOf[String]).toSeq.flatMap(_.split(",")).map(r => RoleName(r.trim))
210 
211       val ds = super.createDataStoreInternal(store, params)
212       val dialect = new PartitionedPostgisDialect(ds, roles)
213 
214       ds.getSQLDialect match {
215         case d: PostGISDialect =>
216           dialect.setEncodeBBOXFilterAsEnvelope(d.isEncodeBBOXFilterAsEnvelope)
217           dialect.setEstimatedExtentsEnabled(d.isEstimatedExtentsEnabled)
218           dialect.setFunctionEncodingEnabled(d.isFunctionEncodingEnabled)
219           dialect.setLooseBBOXEnabled(d.isLooseBBOXEnabled)
220           dialect.setSimplifyEnabled(d.isSimplifyEnabled)
221           ds.setSQLDialect(dialect)
222 
223         case d: PostGISPSDialect =>
224           dialect.setEncodeBBOXFilterAsEnvelope(d.isEncodeBBOXFilterAsEnvelope)
225           dialect.setFunctionEncodingEnabled(d.isFunctionEncodingEnabled)
226           dialect.setLooseBBOXEnabled(d.isLooseBBOXEnabled)
227 
228           // these configs aren't exposed through the PS dialect so re-calculate them from the params
229           val est = PostgisNGDataStoreFactory.ESTIMATED_EXTENTS.lookUp(params)
230           dialect.setEstimatedExtentsEnabled(est == null || est == java.lang.Boolean.TRUE)
231           val simplify = PostgisNGDataStoreFactory.SIMPLIFY.lookUp(params)
232           dialect.setSimplifyEnabled(simplify == null || simplify == java.lang.Boolean.TRUE)
233 
234           ds.setSQLDialect(new PartitionedPostgisPsDialect(ds, dialect))
235 
236         case d => throw new IllegalArgumentException(s"Expected PostGISDialect but got: ${d.getClass.getName}")
237       }
238 
239       ds
240     }

247     private[postgis] def createConnectionOptions(params: java.util.Map[String, _]): Map[String, String] = {
248       val options =
249         Seq(IdleInTransactionTimeout)
250           .flatMap(p => p.opt(params).map(t => s"-c ${p.key}=${t.millis}"))
251 
252       logger.debug(s"Connection options: ${options.mkString(" ")}")
253 
254       val serverOptions =
255         if (options.isEmpty) { Map.empty[String, String] } else { Map("options" -> options.mkString(" ")) }
256 
257       // prepareThreshold is a pgjdbc driver connection property (not a server GUC), so it is added
258       // directly rather than through the '-c <guc>' options string. -1 forces binary result wire
259       // format on the first execution, which one-shot queries never reach with the default of 5.
260       val driverOptions =
261         Option(PrepareThreshold.lookUp(params).asInstanceOf[Integer])
262           .map(v => "prepareThreshold" -> v.toString)
263           .toMap
264 
265       serverOptions ++ driverOptions
266     }
