92   /**
93    * Gets the max size of the connection pool, if it's limited - creating, updating, upgrading and removing schemas
94    * each require two connections at the same time, and will hang with a smaller pool. The data store uses this to
95    * fail those operations up front, without affecting clients that only read or write features.
96    *
97    * Note that we can only check a pool that is configured through the data store parameters - if a data source
98    * is passed in directly, then the pool size parameter is not used.
99    *
100    * @param params data store params
101    * @return
102    */
103   private def maxConnections(params: java.util.Map[String, _]): Option[Int] = {
104     import JDBCDataStoreFactory.{DATASOURCE, MAXCONN}
105     if (params.get(DATASOURCE.key) != null) { None } else {
106       // note: a value of zero or less means that the pool is not limited
107       Option(MAXCONN.lookUp(params).asInstanceOf[Integer]).map(_.intValue).filter(_ > 0)
108     }
109   }

135       val dataSource = super.createDataSource(params, dialect)
136 
137       val metrics = MetricsRegistryParam.lookupRegistry(params).flatMap { registry =>
138         var metrics: MetricsDataSource = null
139         try {
140           // try to re-create the datasource as a commons-dbcp2 instance so we can instrument it
141           val basic = dataSource match {
142             case ds: AbstractManageableDataSource => ds.unwrap(classOf[BasicDataSource])
143             case ds: BasicDataSource => ds
144             case ds => throw new UnsupportedOperationException(s"Unexpected data source class: ${ds.getClass.getName} $ds")
145           }
146           metrics = new MetricsDataSource("postgis")
147 
148           // these are the options that are configured in our parent factories
149           metrics.setDriverClassName(basic.getDriverClassName)
150           metrics.setUrl(basic.getUrl)
151           metrics.setUsername(basic.getUsername)
152           metrics.setPassword(basic.getPassword)
153           metrics.setMaxWait(Duration.ofMillis(basic.getMaxWait))
154           metrics.setMinIdle(basic.getMinIdle)
155           metrics.setMaxTotal(basic.getMaxActive)
156           metrics.setTestOnBorrow(basic.getTestOnBorrow)
157           metrics.setValidationQuery(basic.getValidationQuery)
158           metrics.setTestWhileIdle(basic.getTestWhileIdle)
159           metrics.setDurationBetweenEvictionRuns(Duration.ofMillis(basic.getTimeBetweenEvictionRunsMillis))
160           metrics.setMinEvictableIdle(Duration.ofMillis(basic.getMinEvictableIdleTimeMillis))
161           metrics.setNumTestsPerEvictionRun(basic.getNumTestsPerEvictionRun)
162           metrics.setAccessToUnderlyingConnectionAllowed(basic.isAccessToUnderlyingConnectionAllowed)
163           metrics.setMaxOpenPreparedStatements(basic.getMaxOpenPreparedStatements)
164           metrics.setPoolPreparedStatements(basic.isPoolPreparedStatements)
165 
166           // re-add connection properties, as we can't access those from the BasicDataSource
167           createConnectionOptions(params).foreach { case (k, v) => metrics.addConnectionProperty(k, v) }
168 
169           val registration = registry.register()
170 
171           tryBindPostgresMetrics(metrics, params)
172 
173           // we attach the registry to the ManagedDataSource b/c that's the only hook we have into datastore.dispose()
174           Some(new Dbcp2ManagedDataSource(metrics, registration))
175         } catch {
176           case NonFatal(e) =>
177             logger.warn("Unable to instrument data source, metrics will not be available:", e)
178             CloseWithLogging(Option(metrics))
179             None
180         }
181       }
182 
183       sessionDataSource(metrics.getOrElse(dataSource), params)
184     }

186     private def tryBindPostgresMetrics(dataSource: MetricsDataSource, params: java.util.Map[String, _]): Unit = {
187       try {
188         val conf =
189           MetricsRegistryConfigParam.lookupOpt(params)
190             .getOrElse(ConfigFactory.empty)
191             .withFallback(ConfigFactory.load("postgres-metrics"))
192             .getConfig("postgres")
193         if (conf.getBoolean("enabled")) {
194           val tags = Tags.of(conf.getConfig("tags").root().unwrapped().asScala.map { case (k, v) => Tag.of(k, v.toString) }.asJava)
195           val db =
196             Option(DATABASE.lookUp(params).asInstanceOf[String])
197               .orElse(Option(USER.lookUp(params).asInstanceOf[String]))
198               .getOrElse(throw new IllegalArgumentException(s"${DATABASE.key} was not found in the parameter map"))
199           if (PartitionedPostgisDataStoreFactory.bindings.add(db -> tags)) {
200             logger.info("Adding postgres metrics")
201             new PostgreSQLDatabaseMetrics(dataSource, db, tags).bindTo(Metrics.globalRegistry)
202           }
203         }
204       } catch {
205         case NonFatal(e) => logger.warn("Error binding postgres metrics:", e)
206       }
207     }

220     override protected def createDataStoreInternal(store: JDBCDataStore, baseParams: java.util.Map[String, _]): JDBCDataStore = {
221       val params = new java.util.HashMap[String, Any](baseParams)
222       // default to using prepared statements, if not specified
223       if (!params.containsKey(PreparedStatements.key)) {
224         params.put(PreparedStatements.key, java.lang.Boolean.TRUE)
225       }
226       // set default schema, if not specified - postgis store doesn't actually use its own default
227       if (!params.containsKey(PostgisNGDataStoreFactory.SCHEMA.key)) {
228         // need to set it in the store, as the key has already been processed
229         store.setDatabaseSchema("public")
230         // also set in the params for consistency, although it's not used anywhere
231         params.put(PostgisNGDataStoreFactory.SCHEMA.key, "public")
232       }
233       val roles = Option(ReadAccessRoles.lookUp(params).asInstanceOf[String]).toSeq.flatMap(_.split(",")).map(r => RoleName(r.trim))
234 
235       val ds = super.createDataStoreInternal(store, params)
236       val dialect = new PartitionedPostgisDialect(ds, roles)
237 
238       ds.getSQLDialect match {
239         case d: PostGISDialect =>
240           dialect.setEncodeBBOXFilterAsEnvelope(d.isEncodeBBOXFilterAsEnvelope)
241           dialect.setEstimatedExtentsEnabled(d.isEstimatedExtentsEnabled)
242           dialect.setFunctionEncodingEnabled(d.isFunctionEncodingEnabled)
243           dialect.setLooseBBOXEnabled(d.isLooseBBOXEnabled)
244           dialect.setSimplifyEnabled(d.isSimplifyEnabled)
245           ds.setSQLDialect(dialect)
246 
247         case d: PostGISPSDialect =>
248           dialect.setEncodeBBOXFilterAsEnvelope(d.isEncodeBBOXFilterAsEnvelope)
249           dialect.setFunctionEncodingEnabled(d.isFunctionEncodingEnabled)
250           dialect.setLooseBBOXEnabled(d.isLooseBBOXEnabled)
251 
252           // these configs aren't exposed through the PS dialect so re-calculate them from the params
253           val est = PostgisNGDataStoreFactory.ESTIMATED_EXTENTS.lookUp(params)
254           dialect.setEstimatedExtentsEnabled(est == null || est == java.lang.Boolean.TRUE)
255           val simplify = PostgisNGDataStoreFactory.SIMPLIFY.lookUp(params)
256           dialect.setSimplifyEnabled(simplify == null || simplify == java.lang.Boolean.TRUE)
257 
258           ds.setSQLDialect(new PartitionedPostgisPsDialect(ds, dialect))
259 
260         case d => throw new IllegalArgumentException(s"Expected PostGISDialect but got: ${d.getClass.getName}")
261       }
262 
263       // if dataSource was specified directly by a parameter, wrap it here to provide session stamping and authorizations
264       ds.getDataSource match {
265         case _: SessionDataSource => // no-op
266         case s => ds.setDataSource(sessionDataSource(s, params))
267       }
268 
269       ds
270     }

277     private[postgis] def createConnectionOptions(params: java.util.Map[String, _]): Map[String, String] = {
278       val options =
279         Seq(IdleInTransactionTimeout)
280           .flatMap(p => p.opt(params).map(t => s"-c ${p.key}=${t.millis}"))
281 
282       logger.debug(s"Connection options: ${options.mkString(" ")}")
283 
284       val serverOptions =
285         if (options.isEmpty) { Map.empty[String, String] } else { Map("options" -> options.mkString(" ")) }
286 
287       // prepareThreshold is a pgjdbc driver connection property (not a server GUC), so it is added
288       // directly rather than through the '-c <guc>' options string. -1 forces binary result wire
289       // format on the first execution, which one-shot queries never reach with the default of 5.
290       val driverOptions =
291         Option(PrepareThreshold.lookUp(params).asInstanceOf[Integer])
292           .map(v => "prepareThreshold" -> v.toString)
293           .toMap
294 
295       serverOptions ++ driverOptions
296     }
