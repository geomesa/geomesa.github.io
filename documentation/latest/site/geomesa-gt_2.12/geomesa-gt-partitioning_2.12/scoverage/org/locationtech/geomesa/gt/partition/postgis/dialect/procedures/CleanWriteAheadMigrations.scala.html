43   private def proc(info: TypeInfo): String =
44     s"""CREATE OR REPLACE PROCEDURE ${info.schema.quoted}.${name(info).quoted}() LANGUAGE plpgsql AS
45        |  $$BODY$$
46        |    DECLARE
47        |      migrated record;
48        |      partition_name text;
49        |      detach_pending boolean;
50        |      legacy_job_name text;
51        |      detach_invocation text;
52        |    BEGIN
53        |      FOR migrated IN
54        |        SELECT source_partition FROM ${info.tables.writeAheadMigrations.name.qualified} ORDER BY source_partition
55        |      LOOP
56        |        -- Replace the old per-partition jobs with one worker for the parent.
57        |        legacy_job_name := ${literal(info.typeIdentifier + "-detach-")} || split_part(migrated.source_partition, '.', 2);
58        |        IF EXISTS (SELECT FROM cron.job WHERE jobname = legacy_job_name) THEN
59        |          PERFORM cron.unschedule(legacy_job_name);
60        |        END IF;
61        |        IF NOT EXISTS (SELECT FROM pg_inherits
62        |            WHERE inhparent = ${info.tables.writeAheadPartitions.name.asRegclass}
63        |              AND inhrelid = to_regclass(migrated.source_partition)) THEN
64        |          BEGIN
65        |            IF to_regclass(migrated.source_partition) IS NOT NULL THEN
66        |              EXECUTE 'LOCK TABLE ONLY ' || migrated.source_partition || ' IN ACCESS EXCLUSIVE MODE NOWAIT';
67        |              EXECUTE 'DROP TABLE ' || migrated.source_partition;
68        |            END IF;
69        |            DELETE FROM ${info.tables.writeAheadMigrations.name.qualified}
70        |              WHERE source_partition = migrated.source_partition;
71        |          EXCEPTION WHEN lock_not_available THEN
72        |            RAISE DEBUG 'Deferring removal of detached write-ahead partition %', migrated.source_partition;
73        |          END;
74        |        END IF;
75        |        COMMIT;
76        |      END LOOP;
77        |
78        |      SELECT migration.source_partition, inherits.inhdetachpending INTO partition_name, detach_pending
79        |        FROM ${info.tables.writeAheadMigrations.name.qualified} migration
80        |        JOIN pg_inherits inherits ON inherits.inhrelid = to_regclass(migration.source_partition)
81        |          AND inherits.inhparent = ${info.tables.writeAheadPartitions.name.asRegclass}
82        |        ORDER BY inherits.inhdetachpending DESC, migration.enqueued, migration.source_partition
83        |        LIMIT 1;
84        |      IF FOUND THEN
85        |        detach_invocation := ${literal(s"ALTER TABLE ${info.tables.writeAheadPartitions.name.qualified} DETACH PARTITION ")} ||
86        |          partition_name || CASE WHEN detach_pending THEN ' FINALIZE' ELSE ' CONCURRENTLY' END;
87        |        PERFORM cron.schedule(${detachJobName(info).quoted}, '* * * * *', detach_invocation);
88        |      ELSIF EXISTS (SELECT FROM cron.job WHERE jobname = ${detachJobName(info).quoted}) THEN
89        |        PERFORM cron.unschedule(${detachJobName(info).quoted});
90        |      END IF;
91        |    END;
92        |  $$BODY$$;""".stripMargin
