95   override def visit(filter: Or, extraData: AnyRef): AnyRef = {
96     // the super-class implementation merges ORs into INs
97     // for array-types, skip the super class as it breaks array OR queries
98     // for json-path expressions, skip the super class as it breaks json path predicates
99     // for other types, keep the super handling as INs may be more efficient that ORs
100     def skipOrInProcessing(attribute: String): Boolean =
101       attribute.startsWith("$") || Option(featureType.getDescriptor(attribute)).exists(_.getType.getBinding.isArray)
102 
103     val names = FilterHelper.propertyNames(filter)
104     if (names.exists(skipOrInProcessing)) {
105       visit(filter.asInstanceOf[BinaryLogicOperator], "OR")
106     } else {
107       super.visit(filter, extraData)
108     }
109   }

111   override protected def visitBinaryComparisonOperator(filter: BinaryComparisonOperator, extraData: AnyRef): Unit = {
112     try {
113       (filter.getExpression1, filter.getExpression2) match {
114         case (f: JsonPathFilterFunction, lit: Literal) =>
115           writePath(f.path(null), lit, filter, flipped = false, extraData)
116 
117         case (lit: Literal, f: JsonPathFilterFunction) =>
118           writePath(f.path(null), lit, filter, flipped = true, extraData)
119 
120         case (p: PropertyName, lit: Literal) if p.getPropertyName.startsWith("$") =>
121           writePath(p.getPropertyName, lit, filter, flipped = false, extraData)
122 
123         case (lit: Literal, p: PropertyName) if p.getPropertyName.startsWith("$") =>
124           writePath(p.getPropertyName, lit, filter, flipped = true, extraData)
125 
126         case _ => super.visitBinaryComparisonOperator(filter, extraData)
127       }
128     } catch {
129       case NonFatal(e) => throw new RuntimeException(s"Error encoding filter into SQL: $filter", e)
130     }
131   }

133   private def writePath(pathString: String, lit: Literal, filter: BinaryComparisonOperator, flipped: Boolean, extraData: AnyRef): AnyRef = {
134     val op = filter match {
135       case _: PropertyIsEqualTo => "=="
136       case _: PropertyIsGreaterThan if flipped => "<"
137       case _: PropertyIsGreaterThan => ">"
138       case _: PropertyIsGreaterThanOrEqualTo if flipped => "<="
139       case _: PropertyIsGreaterThanOrEqualTo => ">="
140       case _: PropertyIsLessThan if flipped => ">"
141       case _: PropertyIsLessThan => "<"
142       case _: PropertyIsLessThanOrEqualTo if flipped => ">="
143       case _: PropertyIsLessThanOrEqualTo => "<="
144       case _: PropertyIsNotEqualTo => "!="
145       case _ => throw new UnsupportedOperationException(s"Unexpected binary comparison op: $filter")
146     }
147     val path = JsonPathPropertyAccessor.Paths.get(pathString)
148     path.head match {
149       case a: PathAttribute => writePath(a.name, path.tail, lit, op, extraData)
150       case e => throw new UnsupportedOperationException(s"Json paths must start with an attribute: $e")
151     }
152   }
153 
154   private def writePath(attribute: String, path: JsonPath, lit: Literal, op: String, extraData: AnyRef): AnyRef = {
155     if (!isPrepareEnabled) {
156       throw new UnsupportedOperationException(s"Json paths expressions are only supported when using prepared statements")
157     }
158     if (path.elements.contains(PathDeepScan)) {
159       // helps prevent denial of service
160       throw new UnsupportedOperationException(s"Json paths deep scan expressions (..) are not supported")
161     }
162     visit(FilterHelper.ff.property(attribute), extraData)
163     out.write(" @@ ")
164     val expression = lit.getValue match {
165       case s: String => '"' + StringEscapeUtils.escapeJava(s) + '"'
166       case n: Number => String.valueOf(n)
167       case b: java.lang.Boolean => String.valueOf(b)
168       case null => "null"
169       case v => throw new UnsupportedOperationException(s"Received unsupported literal in JSON path expression: ${v.getClass} $v")
170     }
171     visit(FilterHelper.ff.literal(s"$path $op $expression"), extraData)
172     out.write("::jsonpath")
173     extraData
174   }
