95   override def visit(filter: Or, extraData: AnyRef): AnyRef = {
96     // the super-class implementation merges ORs into INs
97     // for array-types, skip the super class as it breaks array OR queries
98     // for json-path expressions, skip the super class as it breaks json path predicates
99     // for other types, keep the super handling as INs may be more efficient that ORs
100     def skipOrInProcessing(attribute: String): Boolean =
101       attribute.startsWith("$") || Option(featureType.getDescriptor(attribute)).exists(_.getType.getBinding.isArray)
102 
103     val names = FilterHelper.propertyNames(filter)
104     if (names.exists(skipOrInProcessing)) {
105       visit(filter.asInstanceOf[BinaryLogicOperator], "OR")
106     } else {
107       super.visit(filter, extraData)
108     }
109   }

111   override def visit(filter: Id, extraData: AnyRef): AnyRef = {
112     if (filter.getIdentifiers.size < 2) {
113       super.visit(filter, extraData)
114     } else {
115       // fix bug in super class where multiple OR clauses are not grouped together
116       out.write("(")
117       val result = super.visit(filter, extraData)
118       out.write(")")
119       result
120     }
121   }

123   override protected def visitBinaryComparisonOperator(filter: BinaryComparisonOperator, extraData: AnyRef): Unit = {
124     try {
125       (filter.getExpression1, filter.getExpression2) match {
126         case (f: JsonPathFilterFunction, lit: Literal) =>
127           writePath(f.path(null), lit, filter, flipped = false, extraData)
128 
129         case (lit: Literal, f: JsonPathFilterFunction) =>
130           writePath(f.path(null), lit, filter, flipped = true, extraData)
131 
132         case (p: PropertyName, lit: Literal) if p.getPropertyName.startsWith("$") =>
133           writePath(p.getPropertyName, lit, filter, flipped = false, extraData)
134 
135         case (lit: Literal, p: PropertyName) if p.getPropertyName.startsWith("$") =>
136           writePath(p.getPropertyName, lit, filter, flipped = true, extraData)
137 
138         case _ => super.visitBinaryComparisonOperator(filter, extraData)
139       }
140     } catch {
141       case NonFatal(e) => throw new RuntimeException(s"Error encoding filter into SQL: $filter", e)
142     }
143   }

145   private def writePath(pathString: String, lit: Literal, filter: BinaryComparisonOperator, flipped: Boolean, extraData: AnyRef): AnyRef = {
146     val op = filter match {
147       case _: PropertyIsEqualTo => "=="
148       case _: PropertyIsGreaterThan if flipped => "<"
149       case _: PropertyIsGreaterThan => ">"
150       case _: PropertyIsGreaterThanOrEqualTo if flipped => "<="
151       case _: PropertyIsGreaterThanOrEqualTo => ">="
152       case _: PropertyIsLessThan if flipped => ">"
153       case _: PropertyIsLessThan => "<"
154       case _: PropertyIsLessThanOrEqualTo if flipped => ">="
155       case _: PropertyIsLessThanOrEqualTo => "<="
156       case _: PropertyIsNotEqualTo => "!="
157       case _ => throw new UnsupportedOperationException(s"Unexpected binary comparison op: $filter")
158     }
159     val path = JsonPathPropertyAccessor.Paths.get(pathString)
160     path.head match {
161       case a: PathAttribute => writePath(a.name, path.tail, lit, op, extraData)
162       case e => throw new UnsupportedOperationException(s"Json paths must start with an attribute: $e")
163     }
164   }
165 
166   private def writePath(attribute: String, path: JsonPath, lit: Literal, op: String, extraData: AnyRef): AnyRef = {
167     if (!isPrepareEnabled) {
168       throw new UnsupportedOperationException(s"Json paths expressions are only supported when using prepared statements")
169     }
170     if (path.elements.contains(PathDeepScan)) {
171       // helps prevent denial of service
172       throw new UnsupportedOperationException(s"Json paths deep scan expressions (..) are not supported")
173     }
174     visit(FilterHelper.ff.property(attribute), extraData)
175     out.write(" @@ ")
176     val expression = lit.getValue match {
177       case s: String => '"' + StringEscapeUtils.escapeJava(s) + '"'
178       case n: Number => String.valueOf(n)
179       case b: java.lang.Boolean => String.valueOf(b)
180       case null => "null"
181       case v => throw new UnsupportedOperationException(s"Received unsupported literal in JSON path expression: ${v.getClass} $v")
182     }
183     visit(FilterHelper.ff.literal(s"$path $op $expression"), extraData)
184     out.write("::jsonpath")
185     extraData
186   }
