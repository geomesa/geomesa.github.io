39   /**
40    * Read partition table spaces for the specified type
41    *
42    * @param cx connection
43    * @param metadata metadata
44    * @param schema database schema name
45    * @param typeName feature type name
46    * @return
47    */
48   def read(cx: Connection, metadata: DatabaseMetaData, schema: String, typeName: String): Map[String, String] = {
49     val map = Map.newBuilder[String, String]
50     if (exists(schema, metadata)) {
51       val tablespaceSql =
52         s"select table_space, table_type from " +
53           s"${escape(schema)}.${PartitionTablespacesTable.Name.quoted} where type_name = ?"
54       WithClose(cx.prepareStatement(tablespaceSql)) { statement =>
55         statement.setString(1, typeName)
56         WithClose(statement.executeQuery()) { rs =>
57           while (rs.next()) {
58             val ts = rs.getString(1)
59             if (ts != null && ts.nonEmpty) {
60               rs.getString(2) match {
61                 case WriteAheadTableSuffix.raw => map += SftUserData.WriteAheadTableSpace.key -> ts
62                 case PartitionedWriteAheadTableSuffix.raw => map += SftUserData.WriteAheadPartitionsTableSpace.key -> ts
63                 case PartitionedTableSuffix.raw => map += SftUserData.MainTableSpace.key -> ts
64                 case s => logger.warn(s"Ignoring unexpected tablespace table: $s")
65               }
66             }
67           }
68         }
69       }
70     }
71     map.result()
72   }
73 
74   private def exists(schema: String, metadata: DatabaseMetaData): Boolean =
75     WithClose(metadata.getTables(null, schema, Name.raw, null))(_.next())
