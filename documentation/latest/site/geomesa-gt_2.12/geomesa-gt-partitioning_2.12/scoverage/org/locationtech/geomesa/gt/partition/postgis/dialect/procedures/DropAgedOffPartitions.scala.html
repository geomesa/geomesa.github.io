15 /**
16  * Queues partitions older than the configured maximum for concurrent detachment and deletion
17  */
18 object DropAgedOffPartitions extends SqlProcedure {
19 
20   override def name(info: TypeInfo): FunctionName = FunctionName(s"${info.typeIdentifier}_age_off")
21 
22   override protected def createStatements(info: TypeInfo): Seq[String] = Seq(proc(info))
23 
24   private def proc(info: TypeInfo): String = {
25     s"""CREATE OR REPLACE PROCEDURE ${info.schema.quoted}.${name(info).quoted}(cur_time timestamp without time zone) LANGUAGE plpgsql AS
26        |  $$BODY$$
27        |    DECLARE
28        |      num_partitions int;                          -- number of partitions to keep
29        |      partition_size int;                          -- size of the partition, in hours
30        |      main_cutoff timestamp without time zone;     -- max age of the records for main tables
31        |      partition_start timestamp without time zone; -- start bounds for the partition we're writing
32        |    BEGIN
33        |      SELECT value::int FROM ${info.schema.quoted}.${UserDataTable.Name.quoted}
34        |        WHERE key = ${literal(SftUserData.MaxPartitions.key)}
35        |          AND type_name = ${literal(info.typeName)}
36        |        INTO num_partitions;
37        |      IF FOUND THEN
38        |        SELECT COALESCE(
39        |          (SELECT value::int FROM ${info.schema.quoted}.${UserDataTable.Name.quoted}
40        |            WHERE type_name = ${literal(info.typeName)} AND key = ${literal(SftUserData.IntervalHours.key)}),
41        |          ${SftUserData.IntervalHours.default})
42        |          INTO partition_size;
43        |        main_cutoff := ${info.schema.quoted}.truncate_to_partition(cur_time, partition_size) - make_interval(hours => partition_size);
44        |        -- remove any partitions that have aged out
45        |        partition_start := main_cutoff - (make_interval(hours => partition_size) * num_partitions);
46        |        INSERT INTO ${info.tables.ageOffQueue.name.qualified}
47        |          (source_partition, partition_oid, parent_oid, enqueued)
48        |          SELECT quote_ident(namespace.nspname) || '.' || quote_ident(partition.relname),
49        |              partition.oid, inherits.inhparent, now()
50        |            FROM pg_catalog.pg_inherits inherits
51        |            JOIN pg_catalog.pg_class partition ON partition.oid = inherits.inhrelid
52        |            JOIN pg_catalog.pg_namespace namespace ON namespace.oid = partition.relnamespace
53        |            WHERE (inherits.inhparent = ${info.tables.mainPartitions.name.asRegclass}
54        |              AND partition.relname <= ${literal(info.tables.mainPartitions.name.raw + "_")} || to_char(partition_start, 'YYYY_MM_DD_HH24'))
55        |            OR (inherits.inhparent = ${info.tables.spillPartitions.name.asRegclass}
56        |              AND partition.relname <= ${literal(info.tables.spillPartitions.name.raw + "_")} || to_char(partition_start, 'YYYY_MM_DD_HH24'))
57        |          ON CONFLICT (source_partition) DO NOTHING;
58        |      END IF;
59        |      CALL ${info.schema.quoted}.${CleanAgedOffPartitions.name(info).quoted}();
60        |    END;
61        |  $$BODY$$;
62        |""".stripMargin
63   }
64 }
