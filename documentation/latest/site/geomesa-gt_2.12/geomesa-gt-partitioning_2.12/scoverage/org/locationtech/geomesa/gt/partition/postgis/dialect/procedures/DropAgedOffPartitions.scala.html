14 /**
15  * Queues partitions older than the configured maximum for concurrent detachment and deletion
16  */
17 object DropAgedOffPartitions extends SqlProcedure {
18 
19   override def name(info: TypeInfo): FunctionName = FunctionName(s"${info.typeIdentifier}_age_off")
20 
21   override protected def createStatements(info: TypeInfo): Seq[String] = Seq(proc(info))
22 
23   private def proc(info: TypeInfo): String = {
24     s"""CREATE OR REPLACE PROCEDURE ${info.schema.quoted}.${name(info).quoted}(cur_time timestamp without time zone) LANGUAGE plpgsql AS
25        |  $$BODY$$
26        |    DECLARE
27        |      num_partitions int;                          -- number of partitions to keep
28        |      partition_size int;                          -- size of the partition, in hours
29        |      main_cutoff timestamp without time zone;     -- max age of the records for main tables
30        |      partition_start timestamp without time zone; -- start bounds for the partition we're writing
31        |    BEGIN
32        |      ${UserDataConfig.select(info,
33                 "num_partitions" -> SftUserData.MaxPartitions,
34                 "partition_size" -> SftUserData.IntervalHours)}
35        |      partition_size := COALESCE(partition_size, ${SftUserData.IntervalHours.default});
36        |      ${UserDataConfig.release}
37        |
38        |      IF num_partitions IS NOT NULL THEN
39        |        main_cutoff := ${info.schema.quoted}.truncate_to_partition(cur_time, partition_size) - make_interval(hours => partition_size);
40        |        -- remove any partitions that have aged out
41        |        partition_start := main_cutoff - (make_interval(hours => partition_size) * num_partitions);
42        |        INSERT INTO ${info.tables.ageOffQueue.name.qualified}
43        |          (source_partition, partition_oid, parent_oid, enqueued)
44        |          SELECT quote_ident(namespace.nspname) || '.' || quote_ident(partition.relname),
45        |              partition.oid, inherits.inhparent, now()
46        |            FROM pg_catalog.pg_inherits inherits
47        |            JOIN pg_catalog.pg_class partition ON partition.oid = inherits.inhrelid
48        |            JOIN pg_catalog.pg_namespace namespace ON namespace.oid = partition.relnamespace
49        |            WHERE (inherits.inhparent = ${info.tables.mainPartitions.name.asRegclass}
50        |              AND partition.relname <= ${literal(info.tables.mainPartitions.name.raw + "_")} || to_char(partition_start, 'YYYY_MM_DD_HH24'))
51        |            OR (inherits.inhparent = ${info.tables.spillPartitions.name.asRegclass}
52        |              AND partition.relname <= ${literal(info.tables.spillPartitions.name.raw + "_")} || to_char(partition_start, 'YYYY_MM_DD_HH24'))
53        |          ON CONFLICT (source_partition) DO NOTHING;
54        |      END IF;
55        |      CALL ${info.schema.quoted}.${CleanAgedOffPartitions.name(info).quoted}();
56        |    END;
57        |  $$BODY$$;
58        |""".stripMargin
59   }
