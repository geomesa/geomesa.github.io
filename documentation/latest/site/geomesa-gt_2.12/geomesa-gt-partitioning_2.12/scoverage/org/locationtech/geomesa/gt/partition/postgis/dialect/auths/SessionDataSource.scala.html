20 /**
21  * DataSource wrapper that puts the current user's authorizations into a Postgres
22  * session variable (GUC) every time a connection is borrowed, so that row-level
23  * security policies keyed off the variable (e.g. `current_setting('geomesa.auths', true)`)
24  * filter rows correctly.
25  *
26  * The auths are read from the AuthorizationProvider, and the variable is set
27  * unconditionally on every borrow, so a pooled connection can never leak a previous
28  * user's auths (a thread with no user sets it to the empty string). Postgres-family
29  * databases only.
30  *
31  * @param delegate the underlying data source (usually the store's pooled data source)
32  * @param provider authorizations provider
33  */
34 class SessionDataSource(delegate: DataSource, provider: AuthorizationsProvider) extends ManageableDataSource {
35 
36   import scala.collection.JavaConverters._
37 
38   override def getConnection: Connection = prepare(delegate.getConnection)
39 
40   override def getConnection(username: String, password: String): Connection =
41     prepare(delegate.getConnection(username, password))
42 
43   private def prepare(cx: Connection): Connection = {
44     try {
45       val auths = provider.getAuthorizations.asScala.sorted.mkString(",")
46       // current_setting('geomesa.auths', true)
47       WithClose(cx.prepareStatement(SessionDataSource.SetSessionSql)) { st =>
48         st.setString(1, auths)
49         st.executeQuery().close()
50       }
51       cx
52     } catch {
53       case e: Throwable =>
54         try { cx.close() } catch { case suppressed: Throwable => e.addSuppressed(suppressed) }
55         throw e
56     }
57   }
58 
59   override def close(): Unit = {
60     delegate match {
61       case c: AutoCloseable => c.close()
62       case _ => // no-op
63     }
64   }
65 
66   override def getLogWriter: PrintWriter = delegate.getLogWriter
67   override def setLogWriter(out: PrintWriter): Unit = delegate.setLogWriter(out)
68   override def getLoginTimeout: Int = delegate.getLoginTimeout
69   override def setLoginTimeout(seconds: Int): Unit = delegate.setLoginTimeout(seconds)
70   override def getParentLogger: Logger = delegate.getParentLogger
71 
72   // drill into the delegate first (matching AbstractDecorator semantics), so that unwrapping to a general
73   // interface like DataSource returns the underlying (pooled/metrics) source rather than this auth wrapper
74   override def unwrap[T](iface: Class[T]): T = {
75     if (delegate.isWrapperFor(iface)) {
76       delegate.unwrap(iface)
77     } else if (iface.isInstance(delegate)) {
78       iface.cast(delegate)
79     } else if (iface.isInstance(this)) {
80       iface.cast(this)
81     } else {
82       delegate.unwrap(iface)
83     }
84   }
85   override def isWrapperFor(iface: Class[_]): Boolean =
86     delegate.isWrapperFor(iface) || iface.isInstance(delegate) || iface.isInstance(this)
87 }
