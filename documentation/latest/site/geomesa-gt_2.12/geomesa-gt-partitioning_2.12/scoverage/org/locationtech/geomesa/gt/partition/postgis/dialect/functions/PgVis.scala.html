60   /**
61    * Converts an Accumulo expressions to a tsvector expression
62    *
63    * @param expression visibility expression
64    * @return
65    */
66   private def toPostgis(expression: String): String = {
67     // parse first to ensure a valid expression
68     val parsed = access.newParsedExpression(expression)
69     if (parsed.getType == ExpressionType.EMPTY) {
70       throw new IllegalArgumentException("Invalid empty visibility label")
71     }
72     // but if possible, just use the original expression
73     if (!needsEscaping(expression)) {
74       expression
75     } else {
76       // we have to use tsvector quoting/escaping instead of accumulo
77       toPostgis(parsed)
78     }
79   }
80 
81   private def needsEscaping(expression: String): Boolean = {
82     var i = 0
83     while (i < expression.length) {
84       val c = expression.charAt(i)
85       if ((c >= 'A' && c <= 'Z') ||
86           (c >= 'a' && c <= 'z') ||
87           (c >= '0' && c <= '9') ||
88           c == '_'|| c == '(' || c == ')' || c == '&' || c == '|') {
89         i += 1
90       } else {
91         return true
92       }
93     }
94     false
95   }
96 
97   /**
98    * Converts an Accumulo expressions to a tsvector expression. Expressions are wrapped in single quotes, and single quotes
99    * and back-slashes within a term are doubled to escape them.
100    *
101    * @param expression visibility expression
102    * @return
103    */
104   private def toPostgis(expression: ParsedAccessExpression): String = {
105     expression.getType match {
106       case ExpressionType.AUTHORIZATION => s"'${access.unquote(expression.getExpression).replaceAll("(['\\\\])", "$0$0")}'"
107       case ExpressionType.AND => expression.getChildren.asScala.map(toPostgis).filter(_.nonEmpty).mkString("(", " & ", ")")
108       case ExpressionType.OR => expression.getChildren.asScala.map(toPostgis).filter(_.nonEmpty).mkString("(", " | ", ")")
109       case ExpressionType.EMPTY => ""
110     }
111   }
112 
113   /**
114    * Converts a tsvector expression back to an Accumulo expression.
115    *
116    * @param expression expression
117    * @return
118    */
119   private def fromPostgis(expression: String): String = {
120     if (!PostgisVisPattern.matcher(expression).find()) {
121       // assumed to be an original, non-escaped expression
122       return expression
123     }
124     val result = new StringBuilder(expression.length)
125     var index = 0
126     while (index < expression.length) {
127       expression.charAt(index) match {
128         case c if Character.isWhitespace(c) =>
129           index += 1
130 
131         case '(' | ')' | '&' | '|' =>
132           result.append(expression.charAt(index))
133           index += 1
134 
135         case '\'' =>
136           val authorization = new StringBuilder()
137           index += 1
138           var closed = false
139           while (index < expression.length && !closed) {
140             expression.charAt(index) match {
141               case '\'' if index + 1 < expression.length && expression.charAt(index + 1) == '\'' =>
142                 authorization.append('\'')
143                 index += 2
144               case '\'' =>
145                 closed = true
146                 index += 1
147               case '\\' if index + 1 < expression.length && expression.charAt(index + 1) == '\\' =>
148                 authorization.append('\\')
149                 index += 2
150               case '\\' =>
151                 throw new IllegalArgumentException(s"Invalid escape sequence at index $index: $expression")
152               case character =>
153                 authorization.append(character)
154                 index += 1
155             }
156           }
157           if (!closed) {
158             throw new IllegalArgumentException(s"Unterminated authorization at index $index: $expression")
159           }
160           result.append(access.quote(authorization.toString))
161 
162         case _ =>
163           throw new IllegalArgumentException(s"Unexpected character at index $index: $expression")
164       }
165     }
166     result.toString
167   }
