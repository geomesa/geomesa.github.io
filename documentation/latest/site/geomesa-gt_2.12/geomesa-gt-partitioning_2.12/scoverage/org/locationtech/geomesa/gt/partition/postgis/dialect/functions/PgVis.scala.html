27         |-- Note that visibility strings are expected to be well formed - invalid strings will not raise errors but will

30         |-- A valid expression is a sequence of tokens (chars from [A-Za-z0-9_-.:/], or any chars if quoted with ' or ")

200         |        -- pop 2 values off the result stack and put the resulting AND back on the stack
201         |        result_stack := (result_stack[1] AND result_stack[2]) || result_stack[3:];
202         |      ELSIF output_stack[i] = '|' THEN
203         |        -- pop 2 values off the result stack and put the resulting OR back on the stack

213         |      RETURN false;

216         |    RETURN result_stack[1];
217         |  EXCEPTION
218         |    -- any error indicates an invalid visibility expression - evaluate to 'false' rather than raising
219         |    WHEN OTHERS THEN
220         |      RAISE WARNING 'Invalid visibility expression: %', vis;
221         |      RETURN false;
222         |  END;
223         |$BODY$ LANGUAGE plpgsql IMMUTABLE PARALLEL SAFE;
