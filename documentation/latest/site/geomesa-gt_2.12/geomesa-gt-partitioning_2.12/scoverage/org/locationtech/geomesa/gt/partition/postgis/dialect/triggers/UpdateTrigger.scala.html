12 /**
13  * Trigger to delegate updates from the main view to the sub tables
14  */
15 object UpdateTrigger extends SqlTriggerFunction {
16 
17   override def name(info: TypeInfo): FunctionName = FunctionName(s"update_to_${info.typeIdentifier}")
18 
19   override protected def table(info: TypeInfo): TableIdentifier = info.tables.view.name
20 
21   override protected def action: String = "INSTEAD OF UPDATE"
22 
23   override protected def createStatements(info: TypeInfo): Seq[String] =
24     Seq(function(info)) ++ super.createStatements(info)
25 
26   private def function(info: TypeInfo): String = {
27     val updateFields = info.cols.all.map(c => s"${c.quoted} = NEW.${c.quoted}").mkString(",")
28     val where = s"${info.cols.fid.quoted} = OLD.${info.cols.fid.quoted} AND ${info.cols.dtg.quoted} = OLD.${info.cols.dtg.quoted}"
29     // Migrated write-ahead rows are hidden by the view; mutate their visible main/spill copies instead.
30     val notMigrated =
31       s"""NOT EXISTS (
32          |  SELECT FROM ${info.tables.writeAheadMigrations.name.qualified} migration
33          |  WHERE OLD.${info.cols.dtg.quoted} >= migration.partition_start
34          |    AND OLD.${info.cols.dtg.quoted} < migration.partition_end
35          |)""".stripMargin
36 
37     val notRetired = s"NOT EXISTS (SELECT FROM ${info.tables.writeAheadRetirements.name.qualified} retired " +
38         "WHERE retired.partition_oid = target.tableoid)"
39     val notAgedOff = s"NOT EXISTS (SELECT FROM ${info.tables.ageOffQueue.name.qualified} queue " +
40         "WHERE queue.partition_oid = target.tableoid)"
41 
42     s"""CREATE OR REPLACE FUNCTION ${info.schema.quoted}.${name(info).quoted}() RETURNS trigger AS
43        |  $$BODY$$
44        |    DECLARE
45        |      del_count integer;
46        |    BEGIN
47        |      UPDATE ${info.tables.writeAhead.name.qualified} target SET $updateFields WHERE $where AND $notRetired;
48        |      IF NOT FOUND THEN
49        |        DELETE FROM ${info.tables.writeAheadPartitions.name.qualified} WHERE $where AND $notMigrated;
50        |        GET DIAGNOSTICS del_count := ROW_COUNT;
51        |        IF del_count > 0 THEN
52        |          INSERT INTO ${info.tables.writeAhead.name.qualified} VALUES(NEW.*);
53        |        ELSE
54        |          DELETE FROM ${info.tables.mainPartitions.name.qualified} target WHERE $where AND $notAgedOff;
55        |          GET DIAGNOSTICS del_count := ROW_COUNT;
56        |          IF del_count > 0 THEN
57        |            INSERT INTO ${info.tables.writeAhead.name.qualified} VALUES(NEW.*);
58        |          ELSE
59        |            DELETE FROM ${info.tables.spillPartitions.name.qualified} target WHERE $where AND $notAgedOff;
60        |            GET DIAGNOSTICS del_count := ROW_COUNT;
61        |            IF del_count > 0 THEN
62        |              INSERT INTO ${info.tables.writeAhead.name.qualified} VALUES(NEW.*);
63        |            ELSE
64        |              RETURN NULL;
65        |            END IF;
66        |          END IF;
67        |        END IF;
68        |      END IF;
69        |      RETURN NEW;
70        |    END;
71        |  $$BODY$$
72        |LANGUAGE plpgsql VOLATILE
73        |COST 100;""".stripMargin
74   }
75 }
