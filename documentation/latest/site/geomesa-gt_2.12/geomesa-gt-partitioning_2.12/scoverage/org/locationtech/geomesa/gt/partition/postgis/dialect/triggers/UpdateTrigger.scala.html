12 /**
13  * Trigger to delegate updates from the main view to the sub tables
14  */
15 object UpdateTrigger extends SqlTriggerFunction {
16 
17   override def name(info: TypeInfo): FunctionName = FunctionName(s"update_to_${info.typeIdentifier}")
18 
19   override protected def table(info: TypeInfo): TableIdentifier = info.tables.view.name
20 
21   override protected def action: String = "INSTEAD OF UPDATE"
22 
23   override protected def createStatements(info: TypeInfo): Seq[String] =
24     Seq(function(info)) ++ super.createStatements(info)
25 
26   private def function(info: TypeInfo): String = {
27     val updateFields = info.cols.all.map(c => s"${c.quoted} = NEW.${c.quoted}").mkString(",")
28     val where = s"${info.cols.fid.quoted} = OLD.${info.cols.fid.quoted} AND ${info.cols.dtg.quoted} = OLD.${info.cols.dtg.quoted}"
29     // Migrated write-ahead rows are hidden by the view; mutate their visible main/spill copies instead.
30     val notMigrated =
31       s"""NOT EXISTS (
32          |  SELECT FROM ${info.tables.writeAheadMigrations.name.qualified} migration
33          |  WHERE OLD.${info.cols.dtg.quoted} >= migration.partition_start
34          |    AND OLD.${info.cols.dtg.quoted} < migration.partition_end
35          |)""".stripMargin
36 
37     val notRetired = s"NOT EXISTS (SELECT FROM ${info.tables.writeAheadRetirements.name.qualified} retired " +
38         "WHERE retired.partition_oid = target.tableoid)"
39     val notAgedOff = s"NOT EXISTS (SELECT FROM ${info.tables.ageOffQueue.name.qualified} queue " +
40         "WHERE queue.partition_oid = target.tableoid)"
41 
42     // check the previous generation of this type, if there is one - this will invoke the triggers on the previous view
43     val previous = info.previous match {
44       case None => "RETURN NULL;"
45       case Some(prev) =>
46         s"""DELETE FROM ${prev.tables.view.name.qualified}
47            |                WHERE ${prev.cols.fid.quoted} = OLD.${info.cols.fid.quoted} AND ${prev.cols.dtg.quoted} = OLD.${info.cols.dtg.quoted};
48            |              GET DIAGNOSTICS del_count := ROW_COUNT;
49            |              IF del_count > 0 THEN
50            |                INSERT INTO ${info.tables.writeAhead.name.qualified} VALUES(NEW.*);
51            |              ELSE
52            |                RETURN NULL;
53            |              END IF;""".stripMargin
54     }
55 
56     s"""CREATE OR REPLACE FUNCTION ${info.schema.quoted}.${name(info).quoted}() RETURNS trigger AS
57        |  $$BODY$$
58        |    DECLARE
59        |      del_count integer;
60        |    BEGIN
61        |      UPDATE ${info.tables.writeAhead.name.qualified} target SET $updateFields WHERE $where AND $notRetired;
62        |      IF NOT FOUND THEN
63        |        DELETE FROM ${info.tables.writeAheadPartitions.name.qualified} WHERE $where AND $notMigrated;
64        |        GET DIAGNOSTICS del_count := ROW_COUNT;
65        |        IF del_count > 0 THEN
66        |          INSERT INTO ${info.tables.writeAhead.name.qualified} VALUES(NEW.*);
67        |        ELSE
68        |          DELETE FROM ${info.tables.mainPartitions.name.qualified} target WHERE $where AND $notAgedOff;
69        |          GET DIAGNOSTICS del_count := ROW_COUNT;
70        |          IF del_count > 0 THEN
71        |            INSERT INTO ${info.tables.writeAhead.name.qualified} VALUES(NEW.*);
72        |          ELSE
73        |            DELETE FROM ${info.tables.spillPartitions.name.qualified} target WHERE $where AND $notAgedOff;
74        |            GET DIAGNOSTICS del_count := ROW_COUNT;
75        |            IF del_count > 0 THEN
76        |              INSERT INTO ${info.tables.writeAhead.name.qualified} VALUES(NEW.*);
77        |            ELSE
78        |              $previous
79        |            END IF;
80        |          END IF;
81        |        END IF;
82        |      END IF;
83        |      RETURN NEW;
84        |    END;
85        |  $$BODY$$
86        |LANGUAGE plpgsql VOLATILE
87        |COST 100;""".stripMargin
88   }
89 }
