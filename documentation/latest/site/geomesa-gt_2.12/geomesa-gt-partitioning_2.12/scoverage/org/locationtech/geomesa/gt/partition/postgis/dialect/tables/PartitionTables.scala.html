28   private def statements(info: TypeInfo, table: TableConfig, indexType: String): Seq[String] = {
29     // note: don't include storage opts since these are parent partition tables
30     val (tableTs, indexTs) = table.tablespace match {
31       case None => ("", "")
32       case Some(ts) => (s" TABLESPACE ${ts.quoted}", s" USING INDEX TABLESPACE ${ts.quoted}")
33     }
34 
35     val logging = if (table.logged) { "" } else { "UNLOGGED" }
36     val create =
37       s"""CREATE $logging TABLE IF NOT EXISTS ${table.name.qualified} (
38          |  LIKE ${info.tables.writeAhead.name.qualified} INCLUDING DEFAULTS INCLUDING CONSTRAINTS,
39          |  CONSTRAINT ${escape(table.name.raw, "pkey")} PRIMARY KEY (${info.cols.fid.quoted}, ${info.cols.dtg.quoted})$indexTs
40          |) PARTITION BY RANGE(${info.cols.dtg.quoted})$tableTs;""".stripMargin
41     val pagesPerRange =
42       if (indexType == "brin") { s" with (pages_per_range = ${info.partitions.pagesPerRange})" } else { "" }
43     // note: brin doesn't support 'include' cols
44     val geomIndex =
45       s"""CREATE INDEX IF NOT EXISTS ${escape(table.name.raw, info.cols.geom.raw)}
46          |  ON ${table.name.qualified}
47          |  USING $indexType(${info.cols.geom.quoted})$pagesPerRange$tableTs;""".stripMargin
48     val dtgIndex =
49       s"""CREATE INDEX IF NOT EXISTS ${escape(table.name.raw, info.cols.dtg.raw)}
50          |  ON ${table.name.qualified} (${info.cols.dtg.quoted})$tableTs;""".stripMargin
51     val indices = info.cols.indexed.map { col =>
52       s"""CREATE INDEX IF NOT EXISTS ${escape(table.name.raw, col.raw)}
53          |  ON ${table.name.qualified} (${col.quoted})$tableTs;""".stripMargin
54     }
55     Seq(create, geomIndex, dtgIndex) ++ indices
56   }
57 
58   override protected def dropStatements(info: TypeInfo): Seq[String] =
