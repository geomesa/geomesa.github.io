25   private def statements(info: TypeInfo, table: TableConfig): Seq[String] = {
26     // note: don't include storage opts since these are parent partition tables
27     val (tableTs, indexTs) = table.tablespace match {
28       case None => ("", "")
29       case Some(ts) => (s"TABLESPACE ${ts.quoted}", s"USING INDEX TABLESPACE ${ts.quoted}")
30     }
31 
32     val logging = if (table.logged) { "" } else { "UNLOGGED" }
33     val create =
34       s"""CREATE $logging TABLE IF NOT EXISTS ${table.name.qualified} (
35          |  LIKE ${info.tables.writeAhead.name.qualified} INCLUDING DEFAULTS INCLUDING CONSTRAINTS,
36          |  CONSTRAINT ${escape(table.name.raw, "pkey")} PRIMARY KEY (${info.cols.fid.quoted}, ${info.cols.dtg.quoted}) $indexTs
37          |) PARTITION BY RANGE(${info.cols.dtg.quoted}) $tableTs;""".stripMargin
38     // note: partitions inherit these indices when they're attached, so they must be declared here
39     val indices = info.cols.filter(table.name).indices.map { index =>
40       s"""CREATE INDEX IF NOT EXISTS ${escape(table.name.raw, index.cols.map(_.raw).mkString("_"))}
41          |  ON ${table.name.qualified}
42          |  ${index.using} (${index.cols.map(_.quoted).mkString(", ")} ${index.opclass}) ${index.includes} ${index.storageOpts} $tableTs;""".stripMargin
43     }
44     Seq(create) ++ indices
45   }
46 
47   override protected def dropStatements(info: TypeInfo): Seq[String] =
48     tables(info).map(table => s"DROP TABLE IF EXISTS ${table.name.qualified};")
