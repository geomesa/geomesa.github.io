25   private def statements(info: TypeInfo, table: TableConfig): Seq[String] = {
26     // note: don't include storage opts since these are parent partition tables
27     val (tableTs, indexTs) = table.tablespace match {
28       case None => ("", "")
29       case Some(ts) => (s"TABLESPACE ${ts.quoted}", s"USING INDEX TABLESPACE ${ts.quoted}")
30     }
31 
32     val logging = if (table.logged) { "" } else { "UNLOGGED" }
33     val create =
34       s"""CREATE $logging TABLE IF NOT EXISTS ${table.name.qualified} (
35          |  LIKE ${info.tables.writeAhead.name.qualified} INCLUDING DEFAULTS INCLUDING CONSTRAINTS,
36          |  CONSTRAINT ${escape(table.name.raw, "pkey")} PRIMARY KEY (${info.cols.fid.quoted}, ${info.cols.dtg.quoted}) $indexTs
37          |) PARTITION BY RANGE(${info.cols.dtg.quoted}) $tableTs;""".stripMargin
38     // note: partitions inherit these indices when they're attached, so they must be declared here
39     // note: we check for the index first, as CREATE INDEX IF NOT EXISTS locks the table before checking if the index
40     // exists, which will block behind any partition maintenance (and will then block any updates or deletes)
41     val indices = info.cols.filter(table.name).indices.map { index =>
42       val name = TableIdentifier(info.schema.raw, s"${table.name.raw}_${index.cols.map(_.raw).mkString("_")}")
43       s"""DO $$$$
44          |BEGIN
45          |  IF to_regclass(${literal(name.qualified)}) IS NULL THEN
46          |    CREATE INDEX IF NOT EXISTS ${name.quoted}
47          |      ON ${table.name.qualified}
48          |      ${index.using} (${index.cols.map(_.quoted).mkString(", ")} ${index.opclass}) ${index.includes} ${index.storageOpts} $tableTs;
49          |  END IF;
50          |END$$$$;""".stripMargin
51     }
52     Seq(create) ++ indices
53   }
54 
55   override protected def dropStatements(info: TypeInfo): Seq[String] =
56     tables(info).map(table => s"DROP TABLE IF EXISTS ${table.name.qualified};")
