35   override def setValue(
36       value: AnyRef,
37       binding: Class[_],
38       att: AttributeDescriptor,
39       ps: PreparedStatement,
40       column: Int,
41       cx: Connection): Unit = {
42     // json columns are string type in geotools, but we have to use setObject or else we get a binding error
43     if (binding == classOf[String] && att != null && att.isJson) {
44       ps.setObject(column, value, Types.OTHER)
45     } else if (binding.isArray || binding == classOf[java.util.List[_]]) {
46       // handle bug in jdbc store not calling setArrayValue in update statements
47       value match {
48         case null =>
49           ps.setNull(column, Types.ARRAY)
50 
51         case list: java.util.Collection[_] =>
52           if (list.isEmpty) {
53             ps.setNull(column, Types.ARRAY)
54           } else {
55             setArrayValue(list.toArray(), att, ps, column, cx)
56           }
57 
58         case array: Array[_] =>
59           if (array.isEmpty) {
60             ps.setNull(column, Types.ARRAY)
61           } else {
62             setArrayValue(array, att, ps, column, cx)
63           }
64 
65         case singleton =>
66           setArrayValue(Array(singleton), att, ps, column, cx)
67       }
68     } else {
69       super.setValue(value, binding, att, ps, column, cx)
70     }
71   }
72 
73   override protected def convertToArray(
74       value: Any, componentTypeName: String, componentType: Class[_], connection: Connection): java.sql.Array = {
75     val array = value match {

77       case _ => value
78     }
79     super.convertToArray(array, componentTypeName, componentType, connection)
80   }
