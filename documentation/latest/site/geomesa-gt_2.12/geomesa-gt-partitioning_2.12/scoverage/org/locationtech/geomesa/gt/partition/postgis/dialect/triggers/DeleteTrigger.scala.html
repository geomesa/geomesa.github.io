12 /**
13  * Trigger to delegate deletes from the main view to the sub tables
14  */
15 object DeleteTrigger extends SqlTriggerFunction {
16 
17   override def name(info: TypeInfo): FunctionName = FunctionName(s"delete_from_${info.typeIdentifier}")
18 
19   override protected def table(info: TypeInfo): TableIdentifier = info.tables.view.name
20 
21   override protected def action: String = "INSTEAD OF DELETE"
22 
23   override protected def createStatements(info: TypeInfo): Seq[String] =
24     Seq(function(info)) ++ super.createStatements(info)
25 
26   private def function(info: TypeInfo): String = {
27     // Migrated write-ahead rows are hidden by the view; mutate their visible main/spill copies instead.
28     val notMigrated =
29       s"""NOT EXISTS (
30          |  SELECT FROM ${info.tables.writeAheadMigrations.name.qualified} migration
31          |  WHERE OLD.${info.cols.dtg.quoted} >= migration.partition_start
32          |    AND OLD.${info.cols.dtg.quoted} < migration.partition_end
33          |)""".stripMargin
34     val notRetired = s"NOT EXISTS (SELECT FROM ${info.tables.writeAheadRetirements.name.qualified} retired " +
35         "WHERE retired.partition_oid = target.tableoid)"
36     val notAgedOff = s"NOT EXISTS (SELECT FROM ${info.tables.ageOffQueue.name.qualified} queue " +
37         "WHERE queue.partition_oid = target.tableoid)"
38     def delete(table: TableConfig): String = {
39       val filter = if (table == info.tables.writeAhead) {
40         s" AND $notRetired"
41       } else if (table == info.tables.writeAheadPartitions) {
42         s" AND $notMigrated"
43       } else if (table == info.tables.mainPartitions || table == info.tables.spillPartitions) {
44         s" AND $notAgedOff"
45       } else {
46         ""
47       }
48       s"DELETE FROM ${table.name.qualified} target WHERE ${info.cols.fid.quoted} = OLD.${info.cols.fid.quoted} AND ${info.cols.dtg.quoted} = OLD.${info.cols.dtg.quoted}$filter"
49     }
50 
51     s"""CREATE OR REPLACE FUNCTION ${info.schema.quoted}.${name(info).quoted}() RETURNS trigger AS
52        |  $$BODY$$
53        |    DECLARE
54        |      del_count integer;
55        |    BEGIN
56        |      ${delete(info.tables.writeAhead)};
57        |      GET DIAGNOSTICS del_count := ROW_COUNT;
58        |      IF del_count = 0 THEN
59        |        ${delete(info.tables.writeAheadPartitions)};
60        |        GET DIAGNOSTICS del_count := ROW_COUNT;
61        |        IF del_count = 0 THEN
62        |          ${delete(info.tables.mainPartitions)};
63        |          GET DIAGNOSTICS del_count := ROW_COUNT;
64        |          IF del_count = 0 THEN
65        |            ${delete(info.tables.spillPartitions)};
66        |            GET DIAGNOSTICS del_count := ROW_COUNT;
67        |            IF del_count = 0 THEN
68        |              RETURN NULL;
69        |            END IF;
70        |          END IF;
71        |        END IF;
72        |      END IF;
73        |      RETURN OLD;
74        |    END;
75        |  $$BODY$$
76        |LANGUAGE plpgsql VOLATILE
77        |COST 100;""".stripMargin
78   }
79 }
