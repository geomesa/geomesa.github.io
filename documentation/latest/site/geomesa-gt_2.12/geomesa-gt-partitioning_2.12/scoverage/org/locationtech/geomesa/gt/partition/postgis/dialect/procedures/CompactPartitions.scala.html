14 /**
15  * Re-sorts a partition table. Can be used for maintenance if time-latent data comes in and degrades performance
16  * of the BRIN index
17  */
18 object CompactPartitions extends SqlProcedure {
19 
20   override def name(info: TypeInfo): FunctionName = FunctionName(s"${info.typeIdentifier}_compact_partitions")
21 
22   override protected def createStatements(info: TypeInfo): Seq[String] = Seq(proc(info))
23 
24   private def proc(info: TypeInfo): String = {
25     s"""CREATE OR REPLACE PROCEDURE ${info.schema.quoted}.${name(info).quoted}(for_date timestamp without time zone) LANGUAGE plpgsql AS
26        |  $$BODY$$
27        |    DECLARE
28        |      partition_name text;
29        |      spill_partition text;
30        |      min_dtg timestamp without time zone;         -- min date in our partitioned tables
31        |      partition_start timestamp without time zone; -- start bounds for the partition we're writing
32        |      partition_end timestamp without time zone;   -- end bounds for the partition we're writing
33        |      partition_size int;                          -- size of the partition, in hours
34        |      partition_tablespace text;
35        |      index_tablespace text;                       -- index tablespace
36        |      table_wa_logging text;                       -- wa log options
37        |      unsorted_count bigint;
38        |      pexists boolean;                             -- table exists check
39        |    BEGIN
40        |      IF for_date IS NULL THEN
41        |        RAISE EXCEPTION 'date is null' USING HINT = 'Please use a valid date';
42        |      END IF;
43        |
44        |      ${UserDataConfig.select(info,
45                 "partition_size" -> SftUserData.IntervalHours,
46                 "partition_tablespace" -> SftUserData.MainTableSpace,
47                 "table_wa_logging" -> SftUserData.WalLogEnabled)}
48        |      partition_size := COALESCE(partition_size, ${SftUserData.IntervalHours.default});
49        |      ${UserDataConfig.tablespaceClauses("partition_tablespace", "index_tablespace")}
50        |      ${UserDataConfig.walLoggingClause("table_wa_logging")}
51        |      ${UserDataConfig.release}
52        |
53        |      partition_name := ${info.tables.mainPartitions.name.asLiteral} || '_' ||
54        |        to_char(${info.schema.quoted}.truncate_to_partition(for_date, partition_size), 'YYYY_MM_DD_HH24');
55        |
56        |      RAISE INFO '% Compacting partition table %', timeofday()::timestamp, partition_name;
57        |      LOCK TABLE ONLY ${info.tables.mainPartitions.name.qualified} IN SHARE UPDATE EXCLUSIVE MODE;
58        |      -- lock the child table to prevent any inserts that would be lost
59        |      EXECUTE 'LOCK TABLE ${info.schema.quoted}.' || quote_ident(partition_name) ||
60        |        ' IN SHARE ROW EXCLUSIVE MODE';
61        |      EXECUTE 'SELECT ${info.cols.dtg.quoted} FROM ${info.schema.quoted}.' || quote_ident(partition_name) ||
62        |        ' LIMIT 1' INTO min_dtg;
63        |      partition_start := ${info.schema.quoted}.truncate_to_partition(min_dtg, partition_size);
64        |      partition_end := partition_start + make_interval(hours => partition_size);
65        |      spill_partition := ${info.tables.spillPartitions.name.asLiteral} || '_' || to_char(partition_start, 'YYYY_MM_DD_HH24');
66        |
67        |      SELECT EXISTS(SELECT FROM pg_tables WHERE schemaname = ${info.schema.asLiteral} AND tablename = spill_partition)
68        |        INTO pexists;
69        |
70        |      -- use "create table as" (vs create then insert) for performance benefits related to WAL skipping
71        |      EXECUTE 'DROP TABLE IF EXISTS ${info.schema.quoted}.' || quote_ident(partition_name || '_tmp_sort');
72        |      IF pexists THEN
73        |        -- lock the child table to prevent any inserts that would be lost
74        |        EXECUTE 'LOCK TABLE ${info.schema.quoted}.' || quote_ident(spill_partition) ||
75        |          ' IN SHARE ROW EXCLUSIVE MODE';
76        |        EXECUTE 'CREATE ' || table_wa_logging || 'TABLE ${info.schema.quoted}.' || quote_ident(partition_name || '_tmp_sort') ||
77        |          '${info.tables.mainPartitions.storage.opts}' || partition_tablespace || ' AS SELECT * FROM' ||
78        |          ' (SELECT * FROM ${info.schema.quoted}.' || quote_ident(partition_name) ||
79        |          ' UNION ALL SELECT * FROM ${info.schema.quoted}.' || quote_ident(spill_partition) ||
80        |          ') results' ||
81        |          ' ORDER BY _st_sortablehash(${info.cols.geom.quoted})';
82        |        GET DIAGNOSTICS unsorted_count := ROW_COUNT;
83        |      ELSE
84        |        EXECUTE 'CREATE ' || table_wa_logging || 'TABLE ${info.schema.quoted}.' || quote_ident(partition_name || '_tmp_sort') ||
85        |          '${info.tables.mainPartitions.storage.opts}' || partition_tablespace || ' AS SELECT * FROM ' || quote_ident(partition_name) ||
86        |          ' ORDER BY _st_sortablehash(${info.cols.geom.quoted})';
87        |        GET DIAGNOSTICS unsorted_count := ROW_COUNT;
88        |      END IF;
89        |
90        |      EXECUTE 'ALTER TABLE ${info.schema.quoted}.' || quote_ident(partition_name || '_tmp_sort') ||
91        |        ' ADD CONSTRAINT ' || quote_ident(partition_name || '_pkey_tmp_sort') ||
92        |        ' PRIMARY KEY (${info.cols.fid.quoted}, ${info.cols.dtg.quoted})' || index_tablespace;
93        |      -- creating a constraint allows it to be attached to the parent without any additional checks
94        |      EXECUTE 'ALTER TABLE  ${info.schema.quoted}.' || quote_ident(partition_name || '_tmp_sort') ||
95        |        ' ADD CONSTRAINT ' || quote_ident(partition_name || '_constraint_tmp_sort') ||
96        |        ' CHECK ( ${info.cols.dtg.quoted} >= ' || quote_literal(partition_start) ||
97        |        ' AND ${info.cols.dtg.quoted} < ' || quote_literal(partition_end) || ' )';
98        |
99        |      -- create indices before attaching to minimize time to attach, copied from PartitionTables code
100        |${info.cols.filter(info.tables.mainPartitions.name).indices.map { index =>
101     s"""      EXECUTE 'CREATE INDEX IF NOT EXISTS ' || quote_ident(partition_name || ${literal(index.cols.map(_.raw).mkString("_", "_", "_tmp_sort"))}) ||
102        |        ' ON ${info.schema.quoted}.' || quote_ident(partition_name || '_tmp_sort') ||
103        |        ' ${index.using} (${index.cols.map(_.quoted).mkString(", ")} ${index.opclass}) ${index.includes} ${index.storageOpts}' || partition_tablespace;""".stripMargin}.mkString("\n")}
104        |
105        |      RAISE INFO '% Dropping old partition table (queries will be blocked) %', timeofday()::timestamp, partition_name;
106        |      EXECUTE 'DROP TABLE IF EXISTS ${info.schema.quoted}.' || quote_ident(partition_name);
107        |      IF pexists THEN
108        |        EXECUTE 'DROP TABLE IF EXISTS ${info.schema.quoted}.' || quote_ident(spill_partition);
109        |      END IF;
110        |
111        |      RAISE INFO '% Renaming newly sorted partition table %', timeofday()::timestamp, partition_name;
112        |      EXECUTE 'ALTER TABLE ${info.schema.quoted}.' || quote_ident(partition_name || '_tmp_sort') ||
113        |        ' RENAME TO ' || quote_ident(partition_name);
114        |      EXECUTE 'ALTER TABLE ${info.schema.quoted}.' || quote_ident(partition_name) ||
115        |        ' RENAME CONSTRAINT ' || quote_ident(partition_name || '_pkey_tmp_sort') ||
116        |        ' TO ' || quote_ident(partition_name || '_pkey');
117        |${info.cols.filter(info.tables.mainPartitions.name).indices.map { index =>
118     s"""      EXECUTE 'ALTER INDEX ' || quote_ident(partition_name || ${literal(index.cols.map(_.raw).mkString("_", "_", "_tmp_sort"))}) ||
119        |        ' RENAME TO ' || quote_ident(partition_name || ${literal(index.cols.map(_.raw).mkString("_", "_", ""))});""".stripMargin}.mkString("\n")}
120        |
121        |      RAISE INFO '% Attaching newly sorted partition table %', timeofday()::timestamp, partition_name;
122        |      EXECUTE 'ALTER TABLE ${info.tables.mainPartitions.name.qualified}' ||
123        |        ' ATTACH PARTITION ${info.schema.quoted}.' || quote_ident(partition_name) ||
124        |        ' FOR VALUES FROM (' || quote_literal(partition_start) || ') TO (' ||
125        |        quote_literal(partition_end) || ' );';
126        |      RAISE INFO '% Done compacting partition table %', timeofday()::timestamp, partition_name;
127        |
128        |      -- now that we've attached the table we can drop the redundant constraint
129        |      RAISE INFO '% Dropping constraint for partition table %', timeofday()::timestamp, partition_name;
130        |      EXECUTE 'ALTER TABLE ${info.schema.quoted}.' || quote_ident(partition_name) ||
131        |        ' DROP CONSTRAINT ' || quote_ident(partition_name || '_constraint_tmp_sort');
132        |
133        |      -- mark the partition to be analyzed in a separate thread
134        |      INSERT INTO ${info.tables.analyzeQueue.name.qualified}(partition_name, enqueued)
135        |        VALUES (partition_name, now());
136        |
137        |      -- commit to release our locks
138        |      COMMIT;
139        |      RAISE INFO '% Done compacting % rows in partition %', timeofday()::timestamp, unsorted_count, partition_name;
140        |
141        |    END;
142        |  $$BODY$$;
143        |""".stripMargin
144   }
