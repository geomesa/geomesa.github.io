15 /**
16  * Re-sorts a partition table. Can be used for maintenance if time-latent data comes in and degrades performance
17  * of the BRIN index
18  */

25   private def proc(info: TypeInfo): String = {
26     s"""CREATE OR REPLACE PROCEDURE ${info.schema.quoted}.${name(info).quoted}(for_date timestamp without time zone) LANGUAGE plpgsql AS
27        |  $$BODY$$
28        |    DECLARE
29        |      partition_name text;
30        |      spill_partition text;
31        |      min_dtg timestamp without time zone;         -- min date in our partitioned tables
32        |      partition_start timestamp without time zone; -- start bounds for the partition we're writing
33        |      partition_end timestamp without time zone;   -- end bounds for the partition we're writing
34        |      partition_size int;                          -- size of the partition, in hours
35        |      partition_tablespace text;
36        |      index_tablespace text;                       -- index tablespace
37        |      table_wa_logging text;                       -- wa log options
38        |      unsorted_count bigint;
39        |      pexists boolean;                             -- table exists check
40        |    BEGIN
41        |      IF for_date IS NULL THEN
42        |        RAISE EXCEPTION 'date is null' USING HINT = 'Please use a valid date';
43        |      END IF;
44        |
45        |      SELECT COALESCE(
46        |        (SELECT value::int FROM ${info.schema.quoted}.${UserDataTable.Name.quoted}
47        |          WHERE type_name = ${literal(info.typeName)} AND key = ${literal(SftUserData.IntervalHours.key)}),
48        |          ${SftUserData.IntervalHours.default})
49        |        INTO partition_size;
50        |      partition_name := ${info.tables.mainPartitions.name.asLiteral} || '_' ||
51        |        to_char(${info.schema.quoted}.truncate_to_partition(for_date, partition_size), 'YYYY_MM_DD_HH24');
52        |
53        |      SELECT value FROM ${info.schema.quoted}.${UserDataTable.Name.quoted}
54        |        WHERE type_name = ${literal(info.typeName)} AND key = ${literal(SftUserData.MainTableSpace.key)}
55        |        INTO partition_tablespace;
56        |      IF partition_tablespace IS NULL THEN
57        |        index_tablespace := '';
58        |        partition_tablespace := '';
59        |      ELSE
60        |        index_tablespace := ' USING INDEX TABLESPACE '|| quote_ident(partition_tablespace);
61        |        partition_tablespace := ' TABLESPACE ' || quote_ident(partition_tablespace);
62        |      END IF;
63        |
64        |      SELECT COALESCE(
65        |        (SELECT value FROM ${info.schema.quoted}.${UserDataTable.Name.quoted}
66        |          WHERE type_name = ${literal(info.typeName)} AND key = ${literal(SftUserData.WalLogEnabled.key)}),
67        |          '${SftUserData.WalLogEnabled.default}'
68        |        ) INTO table_wa_logging;
69        |      IF table_wa_logging IS NULL OR table_wa_logging::boolean THEN
70        |        table_wa_logging := '';
71        |      ELSE
72        |        table_wa_logging := 'UNLOGGED ';
73        |      END IF;
74        |
75        |      RAISE INFO '% Compacting partition table %', timeofday()::timestamp, partition_name;
76        |      LOCK TABLE ONLY ${info.tables.mainPartitions.name.qualified} IN SHARE UPDATE EXCLUSIVE MODE;
77        |      -- lock the child table to prevent any inserts that would be lost
78        |      EXECUTE 'LOCK TABLE ${info.schema.quoted}.' || quote_ident(partition_name) ||
79        |        ' IN SHARE ROW EXCLUSIVE MODE';
80        |      EXECUTE 'SELECT ${info.cols.dtg.quoted} FROM ${info.schema.quoted}.' || quote_ident(partition_name) ||
81        |        ' LIMIT 1' INTO min_dtg;
82        |      partition_start := ${info.schema.quoted}.truncate_to_partition(min_dtg, partition_size);
83        |      partition_end := partition_start + make_interval(hours => partition_size);
84        |      spill_partition := ${info.tables.spillPartitions.name.asLiteral} || '_' || to_char(partition_start, 'YYYY_MM_DD_HH24');
85        |
86        |      SELECT EXISTS(SELECT FROM pg_tables WHERE schemaname = ${info.schema.asLiteral} AND tablename = spill_partition)
87        |        INTO pexists;
88        |
89        |      -- use "create table as" (vs create then insert) for performance benefits related to WAL skipping
90        |      EXECUTE 'DROP TABLE IF EXISTS ${info.schema.quoted}.' || quote_ident(partition_name || '_tmp_sort');
91        |      IF pexists THEN
92        |        -- lock the child table to prevent any inserts that would be lost
93        |        EXECUTE 'LOCK TABLE ${info.schema.quoted}.' || quote_ident(spill_partition) ||
94        |          ' IN SHARE ROW EXCLUSIVE MODE';
95        |        EXECUTE 'CREATE ' || table_wa_logging || 'TABLE ${info.schema.quoted}.' || quote_ident(partition_name || '_tmp_sort') ||
96        |          '${info.tables.mainPartitions.storage.opts}' || partition_tablespace || ' AS SELECT * FROM' ||
97        |          ' (SELECT * FROM ${info.schema.quoted}.' || quote_ident(partition_name) ||
98        |          ' UNION ALL SELECT * FROM ${info.schema.quoted}.' || quote_ident(spill_partition) ||
99        |          ') results' ||
100        |          ' ORDER BY _st_sortablehash(${info.cols.geom.quoted})';
101        |        GET DIAGNOSTICS unsorted_count := ROW_COUNT;
102        |      ELSE
103        |        EXECUTE 'CREATE ' || table_wa_logging || 'TABLE ${info.schema.quoted}.' || quote_ident(partition_name || '_tmp_sort') ||
104        |          '${info.tables.mainPartitions.storage.opts}' || partition_tablespace || ' AS SELECT * FROM ' || quote_ident(partition_name) ||
105        |          ' ORDER BY _st_sortablehash(${info.cols.geom.quoted})';
106        |        GET DIAGNOSTICS unsorted_count := ROW_COUNT;
107        |      END IF;
108        |
109        |      EXECUTE 'ALTER TABLE ${info.schema.quoted}.' || quote_ident(partition_name || '_tmp_sort') ||
110        |        ' ADD CONSTRAINT ' || quote_ident(partition_name || '_pkey_tmp_sort') ||
111        |        ' PRIMARY KEY (${info.cols.fid.quoted}, ${info.cols.dtg.quoted})' || index_tablespace;
112        |      -- creating a constraint allows it to be attached to the parent without any additional checks
113        |      EXECUTE 'ALTER TABLE  ${info.schema.quoted}.' || quote_ident(partition_name || '_tmp_sort') ||
114        |        ' ADD CONSTRAINT ' || quote_ident(partition_name || '_constraint_tmp_sort') ||
115        |        ' CHECK ( ${info.cols.dtg.quoted} >= ' || quote_literal(partition_start) ||
116        |        ' AND ${info.cols.dtg.quoted} < ' || quote_literal(partition_end) || ' )';
117        |
118        |      -- create indices before attaching to minimize time to attach, copied from PartitionTables code
119        |${info.cols.filter(info.tables.mainPartitions.name).indices.map { index =>
120     s"""      EXECUTE 'CREATE INDEX IF NOT EXISTS ' || quote_ident(partition_name || ${literal(index.cols.map(_.raw).mkString("_", "_", "_tmp_sort"))}) ||
121        |        ' ON ${info.schema.quoted}.' || quote_ident(partition_name || '_tmp_sort') ||
122        |        ' ${index.using} (${index.cols.map(_.quoted).mkString(", ")} ${index.opclass}) ${index.includes} ${index.storageOpts}' || partition_tablespace;""".stripMargin}.mkString("\n")}
123        |
124        |      RAISE INFO '% Dropping old partition table (queries will be blocked) %', timeofday()::timestamp, partition_name;
125        |      EXECUTE 'DROP TABLE IF EXISTS ${info.schema.quoted}.' || quote_ident(partition_name);
126        |      IF pexists THEN
127        |        EXECUTE 'DROP TABLE IF EXISTS ${info.schema.quoted}.' || quote_ident(spill_partition);
128        |      END IF;
129        |
130        |      RAISE INFO '% Renaming newly sorted partition table %', timeofday()::timestamp, partition_name;
131        |      EXECUTE 'ALTER TABLE ${info.schema.quoted}.' || quote_ident(partition_name || '_tmp_sort') ||
132        |        ' RENAME TO ' || quote_ident(partition_name);
133        |      EXECUTE 'ALTER TABLE ${info.schema.quoted}.' || quote_ident(partition_name) ||
134        |        ' RENAME CONSTRAINT ' || quote_ident(partition_name || '_pkey_tmp_sort') ||
135        |        ' TO ' || quote_ident(partition_name || '_pkey');
136        |${info.cols.filter(info.tables.mainPartitions.name).indices.map { index =>
137     s"""      EXECUTE 'ALTER INDEX ' || quote_ident(partition_name || ${literal(index.cols.map(_.raw).mkString("_", "_", "_tmp_sort"))}) ||
138        |        ' RENAME TO ' || quote_ident(partition_name || ${literal(index.cols.map(_.raw).mkString("_", "_", ""))});""".stripMargin}.mkString("\n")}
139        |
140        |      RAISE INFO '% Attaching newly sorted partition table %', timeofday()::timestamp, partition_name;
141        |      EXECUTE 'ALTER TABLE ${info.tables.mainPartitions.name.qualified}' ||
142        |        ' ATTACH PARTITION ${info.schema.quoted}.' || quote_ident(partition_name) ||
143        |        ' FOR VALUES FROM (' || quote_literal(partition_start) || ') TO (' ||
144        |        quote_literal(partition_end) || ' );';
145        |      RAISE INFO '% Done compacting partition table %', timeofday()::timestamp, partition_name;
146        |
147        |      -- now that we've attached the table we can drop the redundant constraint
148        |      RAISE INFO '% Dropping constraint for partition table %', timeofday()::timestamp, partition_name;
149        |      EXECUTE 'ALTER TABLE ${info.schema.quoted}.' || quote_ident(partition_name) ||
150        |        ' DROP CONSTRAINT ' || quote_ident(partition_name || '_constraint_tmp_sort');
151        |
152        |      -- mark the partition to be analyzed in a separate thread
153        |      INSERT INTO ${info.tables.analyzeQueue.name.qualified}(partition_name, enqueued)
154        |        VALUES (partition_name, now());
155        |
156        |      -- commit to release our locks
157        |      COMMIT;
158        |      RAISE INFO '% Done compacting % rows in partition %', timeofday()::timestamp, unsorted_count, partition_name;
159        |
160        |    END;
161        |  $$BODY$$;
162        |""".stripMargin
163   }
