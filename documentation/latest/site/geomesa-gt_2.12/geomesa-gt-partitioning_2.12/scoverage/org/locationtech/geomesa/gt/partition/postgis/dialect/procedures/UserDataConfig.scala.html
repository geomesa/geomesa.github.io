15 /**
16  * Sql snippets for reading feature type user data in a procedure.
17  *
18  * The user data table is shared between all feature types, and reading from it takes a lock that is held until
19  * the end of the transaction. Any ddl on the table will block behind that lock, and will then block all other
20  * readers of the table (i.e. all partition maintenance and schema creation). To prevent that, procedures should read
21  * all their configuration up front, and then release the lock before doing any long-running work.
22  *
23  * Snippets are indented to be used in the top-level block of a procedure.
24  */
25 private[procedures] object UserDataConfig {
26 
27   private val Indent = "\n      "
28 
29   /**
30    * Reads user data values into variables, with a single query. Variables will be null if the user data is not set.
31    *
32    * @param info type info
33    * @param values pairs of the variable name to read into, and the user data to read
34    * @return
35    */
36   def select(info: TypeInfo, values: (String, SftUserData[_])*): String = {
37     val keys = values.map { case (_, config) => literal(config.key) }
38     val cols = keys.map(key => s"max(value) FILTER (WHERE key = $key)")
39     Seq(
40       s"SELECT ${cols.mkString(", ")}",
41       s"  FROM ${info.schema.quoted}.${UserDataTable.Name.quoted}",
42       s"  WHERE type_name = ${literal(info.typeName)} AND key IN (${keys.mkString(", ")})",
43       s"  INTO ${values.map(_._1).mkString(", ")};"
44     ).mkString(Indent)
45   }
46 
47   /**
48    * Converts a tablespace name into the sql clauses for creating tables and indices in that tablespace
49    *
50    * @param tablespace name of the variable holding the tablespace name, will be updated with the table clause
51    * @param index name of the variable to hold the index clause
52    * @return
53    */
54   def tablespaceClauses(tablespace: String, index: String): String = {
55     Seq(
56       s"IF $tablespace IS NULL THEN",
57       s"  $index := '';",
58       s"  $tablespace := '';",
59       s"ELSE",
60       s"  $index := ' USING INDEX TABLESPACE ' || quote_ident($tablespace);",
61       s"  $tablespace := ' TABLESPACE ' || quote_ident($tablespace);",
62       s"END IF;"
63     ).mkString(Indent)
64   }
65 
66   /**
67    * Converts the wal logging flag into the sql clause for creating tables
68    *
69    * @param variable name of the variable holding the value of `SftUserData.WalLogEnabled`, will be updated
70    *                 with the table clause
71    * @return
72    */
73   def walLoggingClause(variable: String): String = {
74     Seq(
75       s"IF $variable IS NULL OR $variable::boolean THEN",
76       s"  $variable := '';",
77       s"ELSE",
78       s"  $variable := 'UNLOGGED ';",
79       s"END IF;"
80     ).mkString(Indent)
81   }
82 
83   /**
84    * Releases the lock on the user data table
85    *
86    * @return
87    */
88   def release: String = {
89     Seq(
90       "-- release our lock on the user data table, so that we don't hold it while doing any long-running work -",
91       "-- otherwise any ddl on the (shared) user data table will block, and then block all other readers of that table",
92       "COMMIT;"
93     ).mkString(Indent)
94   }
95 }
