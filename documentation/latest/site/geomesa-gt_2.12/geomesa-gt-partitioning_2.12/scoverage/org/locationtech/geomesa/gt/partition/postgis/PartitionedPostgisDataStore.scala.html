46 /**
47  * Data store wrapper that manages a hidden per-row visibility column (`_vis`).
48  *
49  * The physical schema carries `_vis` as an ordinary attribute so that it flows through the sealed
50  * `JDBCDataStore` insert/read machinery for free. This wrapper hides that column from callers:
51  *
52  *  - on `createSchema`, if the `pg.vis.enabled` user-data flag is explicitly `true`, a nullable
53  *    `_vis` attribute is appended to the type before it's created
54  *  - on read, features are re-typed to drop `_vis`; the MainView already filters rows by the caller's
55  *    authorizations, and the visibility expression is surfaced into the feature user data
56  *  - on write, the feature's visibility (from user data) is copied into `_vis`
57  *
58  * For any type that has no physical `_vis` column (pre-existing schemas, or vis explicitly disabled)
59  * the wrapper is fully transparent
60  *
61  * @param delegate wrapped jdbc data store
62  */
63 class PartitionedPostgisDataStore(delegate: JDBCDataStore) extends DecoratingDataStore(delegate) with HasAuthorizationsProvider {

73           val underlying = delegate.getSchema(typeName)

83               SchemaWithVis(removeVisFromSchema(underlying), underlying)
84             }
85           }

97   /**
98    * Re-create the PLPG/SQL procedures associated with a feature type. This can be used
99    * to 'upgrade in place' if the code is changed.
100    *
101    * This can be used to alter mutable user data settings, but *cannot* be used to modify the attributes
102    * of the feature type
103    *
104    * @param sft updated feature type
105    */
106   def upgrade(sft: SimpleFeatureType): Unit = {
107     val existing = loadSchema(sft.getTypeName)
108     if (existing == null) {
109       throw new IllegalArgumentException(s"Schema does not exist: ${sft.getTypeName}")
110     }
111     val dialect = this.dialect match {
112       case d: PartitionedPostgisDialect => d
113       case d: PartitionedPostgisPsDialect => d.delegate
114     }
115     val upgrade = existing match {
116       case _: SchemaWithoutVis => sft
117       case s: SchemaWithVis =>
118         val copy = SimpleFeatureTypes.copy(s.underlying)
119         copy.getUserData.putAll(sft.getUserData)
120         copy
121     }
122     WithClose(new DefaultTransaction()) { tx =>
123       WithClose(getConnection(tx)) { cx =>
124         dialect.postCreateTable(getDatabaseSchema, upgrade, cx)
125         tx.commit()
126       }
127     }
128   }

232    * Adds visibility to a schema, for underlying types
233    *
234    * @param sft user-facing feature type
235    * @return
236    */
237   private def addVisToSchema(sft: SimpleFeatureType): SimpleFeatureType = {
238     require(sft.indexOf(VisCol) == -1, s"'$VisCol' is a reserved attribute name")
239     // add the _vis column into the underlying table
240     val builder = new SimpleFeatureTypeBuilder()
241     builder.init(sft)
242     builder.nillable(true)
243     builder.add(PartitionedPostgisDialect.VisCol, classOf[String])
244     val result = builder.buildFeatureType()
245     result.getUserData.putAll(sft.getUserData)
246     result
247   }

265   /**
266    * Project a physical feature onto the user schema, dropping the `_vis` column but surfacing its
267    * value into the feature's user data as the visibility expression (if non-null/non-empty).
268    *
269    * @param from physical feature (with `_vis`)
270    * @param userFacingType user schema (without `_vis`)
271    * @return user feature with visibility populated in user data
272    */
273   private def toUserFacing(from: SimpleFeature, userFacingType: SimpleFeatureType): SimpleFeature = {
274     val to = new ScalaSimpleFeature(userFacingType, from.getID)
275     var i = 0
276     while (i < userFacingType.getAttributeCount) {
277       to.setAttributeNoConvert(i, from.getAttribute(i))
278       i += 1
279     }
280     val pgVis = from.getAttribute(i).asInstanceOf[String]
281     if (pgVis != null) {
282       val vis =
283         try { PgVis.fromPostgisVisibility(pgVis) } catch {
284           case NonFatal(e) =>
285             logger.warn(s"Error converting visibilities back to standard syntax - may be pre-release data: $pgVis", e)
286             pgVis
287         }
288       SecurityUtils.setFeatureVisibility(to, vis)
289     }
290 
291     to.getUserData.putAll(from.getUserData)
292     to
293   }

295   /**
296    * Gets the visibility from a user-facing feature, ensuring it is not null or empty
297    *
298    * @param sf feature
299    * @return
300    */
301   private def getVisibility(sf: SimpleFeature): String = {
302     val visibility = SecurityUtils.getVisibility(sf)
303     if (visibility == null || visibility.isBlank) {
304       throw new IllegalArgumentException(
305         s"Feature ${sf.getID} does not contain a visibility label under the user data key ${SecurityUtils.FEATURE_VISIBILITY}")
306     }
307     try {
308       PgVis.toPostgisVisibility(visibility)
309     } catch {
310       case NonFatal(e) =>
311         throw new IllegalArgumentException(s"Feature ${sf.getID} does not contain a valid visibility label: $visibility", e)
312     }
313   }

353   /**
354    * Feature writer that exposes the user schema but maps visibility into `_vis` on write. The
355    * delegate feature returned by `next()` is retained so that `write()` can copy the caller's
356    * attributes (including visibility) back into it before persisting.
357    */
358   private class VisFeatureWriter(delegate: FeatureWriter[SimpleFeatureType, SimpleFeature], userFacingType: SimpleFeatureType)
359       extends FeatureWriter[SimpleFeatureType, SimpleFeature] {
360 
361     private var delegateFeature: SimpleFeature = _
362     private var userFeature: SimpleFeature = _
363 
364     override def getFeatureType: SimpleFeatureType = userFacingType
365 
366     override def hasNext: Boolean = delegate.hasNext
367 
368     override def next(): SimpleFeature = {
369       delegateFeature = delegate.next()
370       // present a user-schema view (without '_vis') for the caller to populate
371       userFeature = toUserFacing(delegateFeature, userFacingType)
372       userFeature
373     }
374 
375     override def write(): Unit = {
376       var i = 0
377       while (i < userFacingType.getAttributeCount) {
378         delegateFeature.setAttribute(i, userFeature.getAttribute(i))
379         i += 1
380       }
381       delegateFeature.setAttribute(i, getVisibility(userFeature))
382       delegateFeature.getUserData.putAll(userFeature.getUserData)
383       val fid = Option(userFeature.getUserData.get(Hints.PROVIDED_FID).asInstanceOf[String]).getOrElse(userFeature.getID)
384       // propagate a caller-provided fid - the jdbc writer reads the id off the delegate feature itself
385       delegateFeature.getIdentifier match {
386         case id: FeatureIdImpl => id.setID(fid)
387         case _ => // no-op
388       }
389       delegate.write()
390     }
391 
392     override def remove(): Unit = delegate.remove()
393 
394     override def close(): Unit = delegate.close()
395   }

480   /**
481    * Replaces the JDBCFeatureSource with overrides to correctly resolve json path attributes. The method comes
482    * from ContentFeatureSource, so we can't override it in our delegating vis store/source (above). Since JDBCFeatureStore is
483    * a final class, write methods are re-routed back to the data store - this means that doing updates based on json-path
484    * expressions will not work.
485    *
486    * @param ds data store
487    * @param entry content entry
488    * @param query query
489    */
490   private class JsonPathJdbcFeatureStore(ds: JDBCDataStore, entry: ContentEntry, query: Query)
491       extends JDBCFeatureSource(entry, query) with SimpleFeatureStore {
492 
493     /**
494      * Resolves filter property names against the schema, but preserves JSON paths.
495      *
496      * The base class runs `DataUtilities.resolvePropertyNames`, whose `PropertyNameResolvingVisitor`
497      * evaluates each property name against the feature type and rewrites it to the resolved attribute's
498      * local name. A JSON path like `$.props.name` evaluates (via GeoMesa's JSON property accessor) to
499      * the `props` descriptor, so the default would collapse the whole path to `props` - losing the nested
500      * field that IcebergFilterConverter needs to push down. We keep `$`-prefixed names verbatim and
501      * resolve everything else as usual.
502      *
503      * @param query the query being planned
504      * @return the query with non-JSON-path property names resolved
505      */
506     override protected def resolvePropertyNames(query: Query): Query = {
507       val filter = query.getFilter
508       if (filter == null || filter == Filter.INCLUDE || filter == Filter.EXCLUDE) { query } else {
509         val resolved = filter.accept(new JsonPathPropertyNameResolver(getSchema), null).asInstanceOf[Filter]
510         if (resolved == filter) { query } else {
511           val newQuery = new Query(query)
512           newQuery.setFilter(resolved)
513           newQuery
514         }
515       }
516     }
517 
518     override def setFeatures(reader: FeatureReader[SimpleFeatureType, SimpleFeature]): Unit = {
519       WithClose(ds.getFeatureWriter(entry.getTypeName, Filter.INCLUDE, getTransaction)) { writer =>
520         while (writer.hasNext) {
521           writer.next()
522           writer.remove()
523         }
524       }
525       WithClose(ds.getFeatureWriterAppend(entry.getTypeName, getTransaction)) { writer =>
526         while (reader.hasNext) {
527           FeatureUtils.write(writer, reader.next())
528         }
529       }
530     }
531 
532     override def addFeatures(features: FeatureCollection[SimpleFeatureType, SimpleFeature]): java.util.List[FeatureId] = {
533       val ids = new util.ArrayList[FeatureId]()
534       WithClose(ds.getFeatureWriterAppend(entry.getTypeName, getTransaction)) { writer =>
535         WithClose(features.features()) { iter =>
536           while (iter.hasNext) {
537             ids.add(FeatureUtils.write(writer, iter.next()).getIdentifier)
538           }
539         }
540       }
541       ids
542     }
543 
544     override def removeFeatures(filter: Filter): Unit = {
545       WithClose(ds.getFeatureWriter(entry.getTypeName, filter, getTransaction)) { writer =>
546         while (writer.hasNext) {
547           writer.next()
548           writer.remove()
549         }
550       }
551     }

553     override def modifyFeatures(attributeName: String, attributeValue: AnyRef, filter: Filter): Unit =
554       modifyFeatures(Array(attributeName), Array(attributeValue), filter)
555 
556     override def modifyFeatures(attributeNames: Array[Name], attributeValues: Array[AnyRef], filter: Filter): Unit =
557       modifyFeatures(attributeNames.map(_.getLocalPart), attributeValues, filter)
558 
559     override def modifyFeatures(attributeName: Name, attributeValue: AnyRef, filter: Filter): Unit =
560       modifyFeatures(Array(attributeName.getLocalPart), Array(attributeValue), filter)
561 
562     override def modifyFeatures(attributeNames: Array[String], attributeValues: Array[AnyRef], filter: Filter): Unit = {
563       if(attributeNames.length != attributeValues.length) {
564         throw new IllegalArgumentException(
565           s"Attribute names and values do not align: ${attributeNames.length} names (${attributeNames.mkString(", ")}) vs " +
566             s"${attributeValues.length} values (${attributeValues.mkString(", ")})")
567       }
568       WithClose(ds.getFeatureWriter(entry.getTypeName, filter, getTransaction)) { writer =>
569         while (writer.hasNext) {
570           val sf = writer.next()
571           var i = 0
572           while (i < attributeNames.length) {
573             sf.setAttribute(attributeNames(i), attributeValues(i))
574             i += 1
575           }
576           writer.write()
577         }
578       }
579     }
