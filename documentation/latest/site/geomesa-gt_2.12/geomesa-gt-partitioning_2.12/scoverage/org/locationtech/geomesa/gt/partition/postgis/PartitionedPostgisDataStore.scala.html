52 /**
53  * Data store wrapper for partitioned postgis feature types. This wrapper handles:
54  *
55  * <b>Visibilities</b>
56  *
57  * The physical schema carries a hidden per-row visibility column (`_vis`) as an ordinary attribute so that it flows
58  * through the sealed `JDBCDataStore` insert/read machinery for free. This wrapper hides that column from callers:
59  *
60  *  - on `createSchema`, if the `pg.vis.enabled` user-data flag is explicitly `true`, a nullable
61  *    `_vis` attribute is appended to the type before it's created
62  *  - on read, features are re-typed to drop `_vis`; the MainView already filters rows by the caller's
63  *    authorizations, and the visibility expression is surfaced into the feature user data
64  *  - on write, the feature's visibility (from user data) is copied into `_vis`
65  *
66  * <b>Feature type names</b>
67  *
68  * The wrapped `JDBCDataStore` uses the name of the main view as the feature type name. Normally they are the same,
69  * but they will be different if a feature type has been renamed, or if attributes have been added or removed,
70  * which creates a new generation of the feature type backed by a new view. This wrapper maps feature type names
71  * to the current view, and hides the views for any previous generations.
72  *
73  * <b>Feature ids</b>
74  *
75  * The wrapped `JDBCDataStore` prefixes feature ids with the name of the main view. This wrapper removes the prefix,
76  * so that ids are stable across generations. Note that an id which itself starts with the name of the view is not
77  * supported, and may not be matched correctly.
78  *
79  * <b>Default geometries</b>
80  *
81  * The wrapped `JDBCDataStore` always uses the first geometry attribute as the default geometry. The default geometry
82  * that a feature type was created with is stored in the `pg.geom.col` user data, and this wrapper restores it in
83  * the feature types and features that it returns, and in any filters that reference the default geometry.
84  *
85  * For any type that has no physical `_vis` column (pre-existing schemas, or vis explicitly disabled), whose
86  * name matches its view, and whose default geometry is the first geometry attribute, features are passed through
87  * without being re-typed
88  *
89  * <b>Multiple data store instances</b>
90  *
91  * The mapping of feature type names to views is cached by each data store instance, for the duration configured
92  * by the `geomesa.metadata.expiry` system property. After `updateSchema` creates a new generation of a feature
93  * type, other data store instances (including ones in other processes) will keep reading from and writing to the
94  * view for the previous generation until their cache expires. The previous view does not expose the new generation,
95  * so those instances will not see anything written through the new view, although anything that they write will be
96  * visible through both views. Any feature source, reader or writer is bound to the view that was current when it
97  * was created, for as long as it is used, and will start to fail once that generation has been dropped.
98  *
99  * The mapping used to list feature type names is cached in the same way, but is re-loaded whenever the views in the
100  * database change - so a feature type that is renamed through a different data store instance will be listed
101  * under its old name until the cache expires.
102  *
103  * @param delegate wrapped jdbc data store
104  * @param maxConnections max size of the connection pool, if known and limited
105  */
106 class PartitionedPostgisDataStore(delegate: JDBCDataStore, maxConnections: Option[Int] = None)

149   /**
150    * Re-create the PLPG/SQL procedures associated with a feature type. This can be used
151    * to 'upgrade in place' if the code is changed.
152    *
153    * This can be used to alter mutable user data settings, but *cannot* be used to modify the attributes
154    * of the feature type, to add indices, or to rename it - use `updateSchema` instead. Any attributes in the
155    * feature type are ignored, including any index flags - indices are never added to the existing tables, as that
156    * would block writes while the index is built.
157    *
158    * Only the current generation of the feature type is upgraded. Any previous generations (created by adding or
159    * removing attributes through `updateSchema`) keep the procedures and triggers that they were created with,
160    * until they are dropped - only changes to the age-off configuration are applied to them.
161    *
162    * An upgrade requires two connections from the connection pool at the same time - one to hold a lock on the
163    * feature type name, and one to make the changes.
164    *
165    * @param sft updated feature type
166    */
167   def upgrade(sft: SimpleFeatureType): Unit = {
168     val typeName = sft.getTypeName
169     // prevent the feature type from being updated or removed while we're upgrading it, including from other
170     // data store instances - otherwise we could end up upgrading a generation that has been superseded
171     withUpdateLock(typeName) { cx =>
172       val viewExists = this.viewExists(cx) _
173       // now that we have the lock, re-load the schema in case it has been updated since we cached it
174       // note: we always start from the stored type, as the dialect depends on the internal user data that is loaded
175       // with it (identifier alias, previous generation, etc), which may not be present in the type that's passed in
176       val existing = reloadSchema(typeName, generations(cx), viewExists)
177       val updated = SimpleFeatureTypes.copy(sft)
178       InternalKeys.foreach(updated.getUserData.remove)
179       validateUserData(existing, updated)
180       warnIgnoredIndices(existing, updated)
181       try { updateConfig(typeName, existing, updated, viewExists) } finally {
182         invalidate(typeName)
183       }
184     }
185   }

202   /**
203    * Creates a new schema.
204    *
205    * Creating a schema requires two connections from the connection pool at the same time - one to hold a lock on the
206    * feature type name, and one to create things.
207    *
208    * @param featureType feature type
209    */
210   override def createSchema(featureType: SimpleFeatureType): Unit = {
211     val typeName = featureType.getTypeName
212     val sft = SimpleFeatureTypes.copy(featureType)
213     InternalKeys.foreach(sft.getUserData.remove)
214     // note: the dialect only locks on the name of the view, which may not be the same as the type name, so we lock on
215     // the type name in order to prevent the name from being used concurrently - either by the same feature type being
216     // created under a different view, or by an existing feature type being renamed
217     withUpdateLock(typeName) { cx =>
218       try {
219         val viewExists = this.viewExists(cx) _
220         val generations = this.generations(cx)
221         if (generations.view(typeName).exists(viewExists)) {
222           throw new IllegalArgumentException(s"Schema '$typeName' already exists")
223         }
224         val named = if (!viewExists(typeName)) { sft } else {
225           // the name is being used by a view for a different feature type, so we have to use a different name for our view
226           val view = WithClose(getDataSource.getConnection)(partitionedDialect.nextViewName(getDatabaseSchema, typeName, _))
227           val renamed = SimpleFeatureTypes.renameSft(sft, view)
228           renamed.getUserData.put(SftUserData.TypeName.key, typeName)
229           renamed
230         }
231         val view = named.getTypeName
232         // check if there's already a (non-view) table with our name, as the jdbc store will re-use it, in which case
233         // we don't want to drop it if we fail
234         val tableExists = partitionedDialect.relationExists(getDatabaseSchema, view, cx)
235         // the view may have been dropped outside of the jdbc store (for example, a previous generation that was
236         // retired), in which case the jdbc store will still have it cached, and won't let us re-use the name
237         if (!tableExists) {
238           evict(delegate, view)
239         }
240         try {
241           delegate.createSchema(if (SftUserData.VisEnabled.get(named)) { addVisToSchema(named) } else { named })
242         } catch {
243           case NonFatal(e) =>
244             // clean up the initial table created by the jdbc store, so that it doesn't get re-used by a later attempt
245             // note: the rest of the feature type is created in a single transaction, so nothing else should be left behind
246             try {
247               if (!tableExists && !viewExists(view)) {
248                 WithClose(getDataSource.getConnection)(partitionedDialect.dropInitialTable(getDatabaseSchema, view, _))
249               }
250             } catch {
251               case NonFatal(suppressed) => e.addSuppressed(suppressed)
252             }
253             throw e
254         }
255       } finally {
256         invalidate(typeName)
257       }
258     }
259   }

286   /**
287    * Re-loads a schema from the database, clearing anything that we (or the delegate store) have cached for it
288    *
289    * @param typeName name of the feature type
290    * @param generations current mapping of feature type names to views
291    * @param viewExists checks if a view currently exists
292    * @return the underlying feature type
293    */
294   private def reloadSchema(typeName: String, generations: Generations, viewExists: String => Boolean): SimpleFeatureType = {
295     invalidate(typeName)
296     // note: the delegate store may still have the feature type cached, if it was removed through a different
297     // data store instance, so we have to verify that the view still exists
298     val view = generations.view(typeName).filter(viewExists).getOrElse {
299       throw new IOException(s"Schema '$typeName' does not exist")
300     }
301     flush(view)
302     // note: we already have the current view, so we bypass the cache loader, which would use another connection
303     // to look it up again, while we're already holding the connection for the update lock
304     schemas.get(typeName, (_: String) => loadSchemaType(typeName, view)) match {
305       case null => throw new IOException(s"Schema '$typeName' does not exist")
306       case s: Passthrough => s.sft
307       case s: Mapped => s.mapping.underlying
308     }
309   }

322   /**
323    * Updates an existing schema. The following changes are supported:
324    *
325    *  - adding new attributes at the end of the feature type - new attributes must be nillable, and will be null for
326    *    any existing features
327    *  - removing existing attributes, other than the default date and default geometry
328    *  - adding indices to existing attributes - only features written after the update will be indexed
329    *  - renaming the feature type
330    *  - updating user data configuration, for any configuration that isn't fixed when the schema is created - any
331    *    configuration that is not specified in the updated feature type will be left unchanged
332    *
333    * Re-ordering or renaming attributes, changing attribute types, or changing the default date or geometry are
334    * not supported.
335    *
336    * Adding or removing attributes, or adding an index, will create a new generation of the feature type, with a new set of tables and a
337    * new main view which also exposes the data from the existing tables. This does not require any locks on the
338    * existing view or tables, so it will not block (or be blocked by) any queries, writes or partition maintenance.
339    * The previous generation will be dropped automatically once all of its data has aged off. The new generation
340    * is created with the indices specified in the updated feature type - the tables for the previous generation
341    * keep the indices that they were created with.
342    *
343    * Renaming a feature type only changes the name that is exposed by this data store - the underlying view,
344    * tables, etc will keep their existing names.
345    *
346    * Updating user data configuration will re-create the PLPG/SQL procedures associated with the feature type, same
347    * as with `upgrade`.
348    *
349    * Other data store instances that have already loaded the feature type will keep using the previous generation
350    * until their cached schema expires, and any feature sources, readers or writers that have already been created
351    * will keep using it for as long as they are used - see the class docs for details.
352    *
353    * An update requires two connections from the connection pool at the same time - one to hold a lock on the
354    * feature type name (and the new name, if the feature type is being renamed), and one to make the changes.
355    *
356    * @param typeName name of the feature type to update
357    * @param featureType updated feature type
358    */
359   override def updateSchema(typeName: String, featureType: SimpleFeatureType): Unit = {
360     require(featureType.indexOf(VisCol) == -1, s"'$VisCol' is a reserved attribute name")
361     // prevent concurrent updates to the same feature type, including from other data store instances - otherwise
362     // each update could create its own generation, based off the same previous generation
363     // note: if the feature type is being renamed, we also lock on the new name, so that a feature type with that
364     // name can't be created at the same time
365     withUpdateLock(typeName, featureType.getTypeName) { cx =>
366       updateSchemaWithLock(typeName, featureType, this.generations(cx), viewExists(cx))
367     }
368   }

370   /**
371    * Updates a schema - the caller must be holding the update lock for the feature type name, and for the new name
372    * if the feature type is being renamed
373    *
374    * @param typeName name of the feature type to update
375    * @param featureType updated feature type
376    * @param generations current mapping of feature type names to views
377    * @param viewExists checks if a view currently exists
378    */
379   private def updateSchemaWithLock(
380       typeName: String,
381       featureType: SimpleFeatureType,
382       generations: Generations,
383       viewExists: String => Boolean): Unit = {
384     // now that we have the lock, re-load the schema in case it has been updated since we cached it
385     val existing = reloadSchema(typeName, generations, viewExists)
386     val updated = SimpleFeatureTypes.copy(featureType)
387     InternalKeys.foreach(updated.getUserData.remove)
388     validateUpdate(existing, updated)
389 
390     val name = updated.getTypeName
391     if (name != typeName && generations.view(name).exists(viewExists)) {
392       throw new IllegalArgumentException(s"Schema '$name' already exists")
393     }
394 
395     val attributes = existing.getAttributeDescriptors.asScala.map(_.getLocalName).filter(_ != VisCol)
396     // an index can't be added to the existing tables without blocking writes, so it's added through a new generation
397     // note: the index flags are missing from the existing type if they couldn't be loaded, so we verify them against
398     // the database (which will fail if they still can't be loaded), to avoid creating a generation that isn't needed
399     lazy val addsIndex = {
400       val added = addedIndices(existing, updated)
401       added.nonEmpty && {
402         val indexed =
403           WithClose(getDataSource.getConnection)(partitionedDialect.indexedColumns(getDatabaseSchema, existing, _))
404         added.exists(!indexed.contains(_))
405       }
406     }
407     if (attributes != updated.getAttributeDescriptors.asScala.map(_.getLocalName) || addsIndex) {
408       // create a new generation with the updated attributes and indices, which will include the existing data
409       // through its view
410       val view = WithClose(getDataSource.getConnection)(partitionedDialect.nextViewName(getDatabaseSchema, name, _))
411       val generation = SimpleFeatureTypes.renameSft(updated, view)
412       generation.getAttributeDescriptors.asScala.foreach { d =>
413         if (d.isJson() || d.isList) {
414           // these types get special handling when creating a table, which only works based off the attribute binding
415           d.getUserData.remove(JDBCDataStore.JDBC_NATIVE_TYPENAME)
416           d.getUserData.remove(JDBCDataStore.JDBC_NATIVE_TYPE)
417         }
418       }
419       // retain any existing user data that isn't being modified
420       existing.getUserData.asScala.foreach { case (k, v) =>
421         if (!InternalKeys.contains(k)) { generation.getUserData.putIfAbsent(k, v) }
422       }
423       generation.getUserData.put(SftUserData.TypeName.key, name)
424       generation.getUserData.put(SftUserData.PreviousView.key, existing.getTypeName)
425       // the view name is unused, but the jdbc store may still have it cached, if it was dropped outside of the store
426       evict(delegate, view)
427       try {
428         delegate.createSchema(if (existing.indexOf(VisCol) == -1) { generation } else { addVisToSchema(generation) })
429       } catch {
430         case NonFatal(e) =>
431           // clean up anything that was partially created, so that it doesn't interfere with the existing generation
432           // note: the generation is created in a single transaction, so normally the only thing left behind will
433           // be the initial table created by the jdbc store - we hold the lock and the view name was unused, so
434           // anything with that name is ours
435           try {
436             if (viewExists(view)) {
437               dropGenerations(view, Seq.empty)
438             } else {
439               WithClose(getDataSource.getConnection)(partitionedDialect.dropInitialTable(getDatabaseSchema, view, _))
440             }
441           } catch {
442             case NonFatal(suppressed) => e.addSuppressed(suppressed)
443           }
444           throw e
445       }
446     } else {
447       updateConfig(typeName, existing, updated, viewExists)
448     }
449     invalidate(typeName, name)
450   }

452   /**
453    * Updates the user data configuration and/or the name of a schema, without changing its attributes. Unless the
454    * schema is only being renamed, this re-creates the PLPG/SQL procedures associated with it.
455    *
456    * The caller is responsible for validating the update, and for invalidating our cached schema afterwards.
457    *
458    * @param typeName name of the feature type to update
459    * @param existing existing (underlying) feature type, as currently stored in the database
460    * @param updated updated (user-facing) feature type, without any internal user data - only the name and
461    *                user data are used
462    * @param viewExists checks if a view currently exists
463    */
464   private def updateConfig(
465       typeName: String,
466       existing: SimpleFeatureType,
467       updated: SimpleFeatureType,
468       viewExists: String => Boolean): Unit = {
469     val name = updated.getTypeName
470     val changed = updated.getUserData.asScala.exists { case (k, v) =>
471       String.valueOf(v) != String.valueOf(existing.getUserData.get(k))
472     }
473     loadPreviousGeneration(existing, viewExists)
474     WithClose(new DefaultTransaction()) { tx =>
475       WithClose(getConnection(tx)) { cx =>
476         if (changed || name == typeName) {
477           val underlying = SimpleFeatureTypes.copy(existing)
478           underlying.getUserData.putAll(updated.getUserData)
479           if (name != typeName) {
480             // note: the name is persisted along with the rest of the user data, in the same transaction - otherwise
481             // a failure could leave the configuration updated, but the feature type not renamed
482             underlying.getUserData.put(SftUserData.TypeName.key, name)
483           }
484           partitionedDialect.postCreateTable(getDatabaseSchema, underlying, cx)
485         } else {
486           // only the name is changing
487           partitionedDialect.setTypeName(getDatabaseSchema, existing.getTypeName, name, cx)
488         }
489         tx.commit()
490       }
491     }
492     // clear the cached feature type, so that it gets re-loaded from the database
493     flush(existing.getTypeName)
494   }

521   private def removeSchemaWithLock(typeName: String, generations: Generations, viewExists: String => Boolean): Unit = {
522     // note: there won't be a view if the name is being used by the view for a different feature type, or for a
523     // previous generation of a feature type - those must not be removed through their view names
524     // note: the delegate store may still have the feature type cached, if it was removed through a different
525     // data store instance, so we have to verify that the view still exists
526     val view = generations.view(typeName).filter(viewExists).getOrElse {
527       throw new IllegalArgumentException(s"Schema '$typeName' does not exist")
528     }
529     // also remove any previous generations, which are no longer accessible - they're dropped in the same
530     // transaction as the current generation, so if anything fails then the feature type is left intact
531     val previous = generations.previous(view)
532     try {
533       // load the previous generations, so that they're cached by the delegate store - otherwise the dialect has to
534       // load them with an additional connection, while we're already holding the connection being used to drop things
535       previous.filter(viewExists).foreach(delegate.getSchema)
536       dropGenerations(view, previous)
537     } finally {
538       invalidate(typeName)
539     }
540   }

544    *
545    * Note that we don't go through the delegate store - it hard-codes a "DROP TABLE" on the main view, and doesn't
546    * drop things atomically.
547    *
548    * @param view name of the main view for the generation to drop
549    * @param previous names of the main views for any previous generations to drop, most recent first

619   /**
620    * Runs an operation while holding the update locks for one or more feature type names, which prevents the names
621    * from being created, updated or removed concurrently, including by other data store instances.
622    *
623    * The locks are held by a dedicated connection, which is passed to the operation so that it can be used to read
624    * the current state without requiring another connection - it must not be used for anything else.
625    *
626    * Operations are run one at a time for this data store instance, so that concurrent operations can't exhaust
627    * the connection pool while each waiting for the additional connection that they need to make their changes.
628    *
629    * @param typeNames feature type names
630    * @param op operation to run
631    */
632   private def withUpdateLock(typeNames: String*)(op: Connection => Unit): Unit = {
633     // fail fast instead of hanging while waiting for a second connection
634     maxConnections.filter(_ < 2).foreach { max =>
635       throw new IllegalStateException(
636         s"Creating, updating, upgrading and removing schemas each require two connections at the same time, but " +
637             s"the connection pool is limited to $max - set '${JDBCDataStoreFactory.MAXCONN.key}' to at least 2")
638     }
639     updateLock.lock()
640     try {
641       WithClose(getDataSource.getConnection) { cx =>
642         partitionedDialect.withUpdateLock(getDatabaseSchema, typeNames, cx)(op(cx))
643       }
644     } finally {
645       updateLock.unlock()
646     }
647   }

817   /**
818    * Verifies that the changes to a schema are supported by `updateSchema`
819    *
820    * @param existing existing (underlying) feature type
821    * @param updated updated (user-facing) feature type
822    */
823   private def validateUpdate(existing: SimpleFeatureType, updated: SimpleFeatureType): Unit = {
824     def unsupported(msg: String): Nothing = throw new UnsupportedOperationException(msg)
825 
826     val names = updated.getAttributeDescriptors.asScala.map(_.getLocalName).toList
827     val (retained, added) = names.span(existing.getDescriptor(_) != null)
828     if (added.exists(existing.getDescriptor(_) != null)) {
829       unsupported(s"New attributes must be added after any existing attributes: ${added.mkString(", ")}")
830     } else if (retained != existing.getAttributeDescriptors.asScala.map(_.getLocalName).filter(retained.contains)) {
831       unsupported(s"Re-ordering attributes is not supported: ${retained.mkString(", ")}")
832     }
833     retained.foreach { name =>
834       // note: bindings loaded from the database may be sub-types of the 'standard' bindings, e.g. java.sql.Timestamp
835       val from = existing.getDescriptor(name)
836       val to = updated.getDescriptor(name)
837       def changed(what: String, fromValue: Any, toValue: Any): Nothing =
838         unsupported(s"Changing attribute types is not supported: $name $what $fromValue -> $toValue")
839       val binding = to.getType.getBinding
840       if (!binding.isAssignableFrom(from.getType.getBinding) || (from.isInstanceOf[GeometryDescriptor] && binding != from.getType.getBinding)) {
841         changed("type", from.getType.getBinding.getSimpleName, binding.getSimpleName)
842       }
843       // the binding doesn't capture everything about the column type, so check the rest explicitly
844       (from, to) match {
845         case (f: GeometryDescriptor, t: GeometryDescriptor) =>
846           // note: an unknown srid may be -1 or 0
847           val srids = Seq(f, t).map(_.getSrid.filter(_ > 0))
848           if (srids.head != srids.last) {
849             changed("srid", srids.head.getOrElse("none"), srids.last.getOrElse("none"))
850           }
851           val dims = Seq(f, t).map(_.getCoordinateDimensions.filter(_ > 0).getOrElse(2))
852           if (dims.head != dims.last) {
853             changed("coordinate dimensions", dims.head, dims.last)
854           }
855 
856         case _ if from.isJson() != to.isJson() =>
857           changed("json", from.isJson(), to.isJson())
858 
859         case _ if from.isList =>
860           // note: if we couldn't determine the existing list type, then we can't check it
861           val (f, t) = (from.getListType(), to.getListType())
862           if (f != null && (t == null || !t.isAssignableFrom(f))) {
863             changed("list type", f.getSimpleName, Option(t).map(_.getSimpleName).getOrElse("none"))
864           }
865 
866         case _ if classOf[String].isAssignableFrom(binding) =>
867           // note: an unlimited length may be -1, 0 or max int
868           val lengths = Seq(from, to).map(d => Some(FeatureTypes.getFieldLength(d)).filter(n => n > 0 && n < Int.MaxValue))
869           if (lengths.head != lengths.last) {
870             changed("length", lengths.head.getOrElse("unlimited"), lengths.last.getOrElse("unlimited"))
871           }
872 
873         case _ => // no-op
874       }
875     }
876     added.map(updated.getDescriptor).foreach { d =>
877       if (d.getMinOccurs > 0 && !d.isNillable) {
878         unsupported(s"Attributes added to an existing schema must be nillable: ${d.getLocalName}")
879       }
880     }
881     // note: when a feature type is loaded from the database, the default geometry will be the first geometry column,
882     // so we use the default geometry that was recorded when the feature type was created, if available
883     val geom = SftUserData.GeomField.get(existing).orElse(Option(existing.getGeometryDescriptor).map(_.getLocalName))
884     geom.foreach { geom =>
885       if (Option(updated.getGeometryDescriptor).forall(_.getLocalName != geom)) {
886         unsupported(s"Removing or changing the default geometry attribute is not supported: $geom")
887       }
888     }
889     SftUserData.DtgField.get(existing).foreach { dtg =>
890       if (updated.getDescriptor(dtg) == null) {
891         unsupported(s"Removing the default date attribute is not supported: $dtg")
892       }
893     }
894     validateUserData(existing, updated)
895   }
