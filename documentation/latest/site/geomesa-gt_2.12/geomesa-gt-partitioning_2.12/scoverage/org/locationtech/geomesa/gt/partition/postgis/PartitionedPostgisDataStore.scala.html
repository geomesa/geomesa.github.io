40 /**
41  * Data store wrapper that manages a hidden per-row visibility column (`_vis`).
42  *
43  * The physical schema carries `_vis` as an ordinary attribute so that it flows through the sealed
44  * `JDBCDataStore` insert/read machinery for free. This wrapper hides that column from callers:
45  *
46  *  - on `createSchema`, if the `pg.vis.enabled` user-data flag is explicitly `true`, a nullable
47  *    `_vis` attribute is appended to the type before it's created
48  *  - on read, features are re-typed to drop `_vis`; the MainView already filters rows by the caller's
49  *    authorizations, and the visibility expression is surfaced into the feature user data
50  *  - on write, the feature's visibility (from user data) is copied into `_vis`
51  *
52  * For any type that has no physical `_vis` column (pre-existing schemas, or vis explicitly disabled)
53  * the wrapper is fully transparent
54  *
55  * @param delegate wrapped jdbc data store
56  */
57 class PartitionedPostgisDataStore(delegate: JDBCDataStore) extends DecoratingDataStore(delegate) {

66         override def load(typeName: String): SchemaType = {
67           val underlying = delegate.getSchema(typeName)
68           if (underlying == null) {
69             null
70           } else {
71             val visCol = underlying.indexOf(VisCol)
72             if (visCol == -1) {
73               SchemaWithoutVis(underlying)
74             } else {
75               // verify vis is the last attribute, logic in other places depends on that
76               require(visCol == underlying.getAttributeCount - 1, s"Expected $VisCol to be last attribute, but it was not")
77               SchemaWithVis(removeVisFromSchema(underlying), underlying)
78             }
79           }
80         }
81       })

89   /**
90    * Re-create the PLPG/SQL procedures associated with a feature type. This can be used
91    * to 'upgrade in place' if the code is changed.
92    *
93    * This can be used to alter mutable user data settings, but *cannot* be used to modify the attributes
94    * of the feature type
95    *
96    * @param sft updated feature type
97    */
98   def upgrade(sft: SimpleFeatureType): Unit = {
99     val existing = loadSchema(sft.getTypeName)
100     if (existing == null) {
101       throw new IllegalArgumentException(s"Schema does not exist: ${sft.getTypeName}")
102     }
103     val dialect = this.dialect match {
104       case d: PartitionedPostgisDialect => d
105       case d: PartitionedPostgisPsDialect => d.delegate
106     }
107     val upgrade = existing match {
108       case _: SchemaWithoutVis => sft
109       case s: SchemaWithVis =>
110         val copy = SimpleFeatureTypes.copy(s.underlying)
111         copy.getUserData.putAll(sft.getUserData)
112         copy
113     }
114     WithClose(new DefaultTransaction()) { tx =>
115       WithClose(getConnection(tx)) { cx =>
116         dialect.postCreateTable(getDatabaseSchema, upgrade, cx)
117         tx.commit()
118       }
119     }
120   }

210   /**
211    * Remove visibility from a schema, for user-facing types
212    *
213    * @param sft underlying feature type
214    * @return
215    */
216   private def removeVisFromSchema(sft: SimpleFeatureType): SimpleFeatureType = {
217     val builder = new SimpleFeatureTypeBuilder()
218     builder.init(sft)
219     builder.remove(PartitionedPostgisDialect.VisCol)
220     val result = builder.buildFeatureType()
221     result.getUserData.putAll(sft.getUserData)
222     result
223   }

225   /**
226    * Adds visibility to a schema, for underlying types
227    *
228    * @param sft user-facing feature type
229    * @return
230    */
231   private def addVisToSchema(sft: SimpleFeatureType): SimpleFeatureType = {
232     require(sft.indexOf(VisCol) == -1, s"'$VisCol' is a reserved attribute name")
233     // add the _vis column into the underlying table
234     val builder = new SimpleFeatureTypeBuilder()
235     builder.init(sft)
236     builder.nillable(true)
237     builder.add(PartitionedPostgisDialect.VisCol, classOf[String])
238     val result = builder.buildFeatureType()
239     result.getUserData.putAll(sft.getUserData)
240     result
241   }

243   /**
244    * Add the _vis col to any transforms so that it is always retrieved
245    *
246    * @param query query
247    * @return same query, with updated property transform
248    */
249   private def addVisToTransform(query: Query): Query = {
250     if (query.retrieveAllProperties()) {
251       query
252     } else {
253       val updated = new Query(query)
254       updated.setPropertyNames(query.getPropertyNames :+ PartitionedPostgisDialect.VisCol: _*)
255       updated
256     }
257   }

259   /**
260    * Project a physical feature onto the user schema, dropping the `_vis` column but surfacing its
261    * value into the feature's user data as the visibility expression (if non-null/non-empty).
262    *
263    * @param from physical feature (with `_vis`)
264    * @param userFacingType user schema (without `_vis`)
265    * @return user feature with visibility populated in user data
266    */
267   private def toUserFacing(from: SimpleFeature, userFacingType: SimpleFeatureType): SimpleFeature = {
268     val to = new ScalaSimpleFeature(userFacingType, from.getID)
269     var i = 0
270     while (i < userFacingType.getAttributeCount) {
271       to.setAttributeNoConvert(i, from.getAttribute(i))
272       i += 1
273     }
274     SecurityUtils.setFeatureVisibility(to, from.getAttribute(i).asInstanceOf[String])
275     to.getUserData.putAll(from.getUserData)
276     to
277   }
278 
279   /**
280    * Gets the visibility from a user-facing feature, ensuring it is not null or empty
281    *
282    * @param sf feature
283    * @return
284    */
285   private def getVisibility(sf: SimpleFeature): String = {
286     val visibility = SecurityUtils.getVisibility(sf)
287     if (visibility == null || visibility.isBlank) {
288       throw new IllegalArgumentException(
289         s"Feature ${sf.getID} does not contain a visibility label under the user data key ${SecurityUtils.FEATURE_VISIBILITY}")
290     }
291     visibility
292   }

302   /**
303    * Feature reader that re-types features to the user schema, dropping `_vis` (but surfacing its

332   /**
333    * Feature writer that exposes the user schema but maps visibility into `_vis` on write. The
334    * delegate feature returned by `next()` is retained so that `write()` can copy the caller's
335    * attributes (including visibility) back into it before persisting.
336    */
337   private class VisFeatureWriter(delegate: FeatureWriter[SimpleFeatureType, SimpleFeature], userFacingType: SimpleFeatureType)
338       extends FeatureWriter[SimpleFeatureType, SimpleFeature] {
339 
340     private var delegateFeature: SimpleFeature = _
341     private var userFeature: SimpleFeature = _
342 
343     override def getFeatureType: SimpleFeatureType = userFacingType
344 
345     override def hasNext: Boolean = delegate.hasNext
346 
347     override def next(): SimpleFeature = {
348       delegateFeature = delegate.next()
349       // present a user-schema view (without '_vis') for the caller to populate
350       userFeature = toUserFacing(delegateFeature, userFacingType)
351       userFeature
352     }
353 
354     override def write(): Unit = {
355       var i = 0
356       while (i < userFacingType.getAttributeCount) {
357         delegateFeature.setAttribute(i, userFeature.getAttribute(i))
358         i += 1
359       }
360       delegateFeature.setAttribute(i, getVisibility(userFeature))
361       delegateFeature.getUserData.putAll(userFeature.getUserData)
362       val fid = Option(userFeature.getUserData.get(Hints.PROVIDED_FID).asInstanceOf[String]).getOrElse(userFeature.getID)
363       // propagate a caller-provided fid - the jdbc writer reads the id off the delegate feature itself
364       delegateFeature.getIdentifier match {
365         case id: FeatureIdImpl => id.setID(fid)
366         case _ => // no-op
367       }
368       delegate.write()
369     }
370 
371     override def remove(): Unit = delegate.remove()
372 
373     override def close(): Unit = delegate.close()
374   }
