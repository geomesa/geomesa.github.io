42 /**
43  * Data store wrapper that manages a hidden per-row visibility column (`_vis`).
44  *
45  * The physical schema carries `_vis` as an ordinary attribute so that it flows through the sealed
46  * `JDBCDataStore` insert/read machinery for free. This wrapper hides that column from callers:
47  *
48  *  - on `createSchema`, if the `pg.vis.enabled` user-data flag is explicitly `true`, a nullable
49  *    `_vis` attribute is appended to the type before it's created
50  *  - on read, features are re-typed to drop `_vis`; the MainView already filters rows by the caller's
51  *    authorizations, and the visibility expression is surfaced into the feature user data
52  *  - on write, the feature's visibility (from user data) is copied into `_vis`
53  *
54  * For any type that has no physical `_vis` column (pre-existing schemas, or vis explicitly disabled)
55  * the wrapper is fully transparent
56  *
57  * @param delegate wrapped jdbc data store
58  */
59 class PartitionedPostgisDataStore(delegate: JDBCDataStore) extends DecoratingDataStore(delegate) {

68         override def load(typeName: String): SchemaType = {
69           val underlying = delegate.getSchema(typeName)
70           if (underlying == null) {
71             null
72           } else {
73             val visCol = underlying.indexOf(VisCol)
74             if (visCol == -1) {
75               SchemaWithoutVis(underlying)
76             } else {
77               // verify vis is the last attribute, logic in other places depends on that
78               require(visCol == underlying.getAttributeCount - 1, s"Expected $VisCol to be last attribute, but it was not")
79               SchemaWithVis(removeVisFromSchema(underlying), underlying)
80             }
81           }
82         }
83       })

91   /**
92    * Re-create the PLPG/SQL procedures associated with a feature type. This can be used
93    * to 'upgrade in place' if the code is changed.
94    *
95    * This can be used to alter mutable user data settings, but *cannot* be used to modify the attributes
96    * of the feature type
97    *
98    * @param sft updated feature type
99    */
100   def upgrade(sft: SimpleFeatureType): Unit = {
101     val existing = loadSchema(sft.getTypeName)
102     if (existing == null) {
103       throw new IllegalArgumentException(s"Schema does not exist: ${sft.getTypeName}")
104     }
105     val dialect = this.dialect match {
106       case d: PartitionedPostgisDialect => d
107       case d: PartitionedPostgisPsDialect => d.delegate
108     }
109     val upgrade = existing match {
110       case _: SchemaWithoutVis => sft
111       case s: SchemaWithVis =>
112         val copy = SimpleFeatureTypes.copy(s.underlying)
113         copy.getUserData.putAll(sft.getUserData)
114         copy
115     }
116     WithClose(new DefaultTransaction()) { tx =>
117       WithClose(getConnection(tx)) { cx =>
118         dialect.postCreateTable(getDatabaseSchema, upgrade, cx)
119         tx.commit()
120       }
121     }
122   }

226    * Adds visibility to a schema, for underlying types
227    *
228    * @param sft user-facing feature type
229    * @return
230    */
231   private def addVisToSchema(sft: SimpleFeatureType): SimpleFeatureType = {
232     require(sft.indexOf(VisCol) == -1, s"'$VisCol' is a reserved attribute name")
233     // add the _vis column into the underlying table
234     val builder = new SimpleFeatureTypeBuilder()
235     builder.init(sft)
236     builder.nillable(true)
237     builder.add(PartitionedPostgisDialect.VisCol, classOf[String])
238     val result = builder.buildFeatureType()
239     result.getUserData.putAll(sft.getUserData)
240     result

244    * Add the _vis col to any transforms so that it is always retrieved
245    *
246    * @param query query

250     if (query.retrieveAllProperties()) {

259   /**
260    * Project a physical feature onto the user schema, dropping the `_vis` column but surfacing its
261    * value into the feature's user data as the visibility expression (if non-null/non-empty).
262    *
263    * @param from physical feature (with `_vis`)
264    * @param userFacingType user schema (without `_vis`)
265    * @return user feature with visibility populated in user data
266    */
267   private def toUserFacing(from: SimpleFeature, userFacingType: SimpleFeatureType): SimpleFeature = {
268     val to = new ScalaSimpleFeature(userFacingType, from.getID)
269     var i = 0
270     while (i < userFacingType.getAttributeCount) {
271       to.setAttributeNoConvert(i, from.getAttribute(i))
272       i += 1
273     }
274     SecurityUtils.setFeatureVisibility(to, from.getAttribute(i).asInstanceOf[String])
275     to.getUserData.putAll(from.getUserData)
276     to
277   }

279   /**
280    * Gets the visibility from a user-facing feature, ensuring it is not null or empty
281    *
282    * @param sf feature
283    * @return
284    */
285   private def getVisibility(sf: SimpleFeature): String = {
286     val visibility = SecurityUtils.getVisibility(sf)
287     if (visibility == null || visibility.isBlank) {
288       throw new IllegalArgumentException(
289         s"Feature ${sf.getID} does not contain a visibility label under the user data key ${SecurityUtils.FEATURE_VISIBILITY}")
290     }
291     visibility
292   }

332   /**
333    * Feature writer that exposes the user schema but maps visibility into `_vis` on write. The
334    * delegate feature returned by `next()` is retained so that `write()` can copy the caller's
335    * attributes (including visibility) back into it before persisting.
336    */
337   private class VisFeatureWriter(delegate: FeatureWriter[SimpleFeatureType, SimpleFeature], userFacingType: SimpleFeatureType)
338       extends FeatureWriter[SimpleFeatureType, SimpleFeature] {
339 
340     private var delegateFeature: SimpleFeature = _
341     private var userFeature: SimpleFeature = _
342 
343     override def getFeatureType: SimpleFeatureType = userFacingType
344 
345     override def hasNext: Boolean = delegate.hasNext
346 
347     override def next(): SimpleFeature = {
348       delegateFeature = delegate.next()
349       // present a user-schema view (without '_vis') for the caller to populate
350       userFeature = toUserFacing(delegateFeature, userFacingType)
351       userFeature
352     }
353 
354     override def write(): Unit = {
355       var i = 0
356       while (i < userFacingType.getAttributeCount) {
357         delegateFeature.setAttribute(i, userFeature.getAttribute(i))
358         i += 1
359       }
360       delegateFeature.setAttribute(i, getVisibility(userFeature))
361       delegateFeature.getUserData.putAll(userFeature.getUserData)
362       val fid = Option(userFeature.getUserData.get(Hints.PROVIDED_FID).asInstanceOf[String]).getOrElse(userFeature.getID)
363       // propagate a caller-provided fid - the jdbc writer reads the id off the delegate feature itself
364       delegateFeature.getIdentifier match {
365         case id: FeatureIdImpl => id.setID(fid)
366         case _ => // no-op
367       }
368       delegate.write()
369     }
370 
371     override def remove(): Unit = delegate.remove()
372 
373     override def close(): Unit = delegate.close()
374   }

459   /**
460    * Replaces the JDBCFeatureSource with overrides to correctly resolve json path attributes. The method comes
461    * from ContentFeatureSource, so we can't override it in our delegating vis store/source (above). Since JDBCFeatureStore is
462    * a final class, write methods are re-routed back to the data store - this means that doing updates based on json-path
463    * expressions will not work.
464    *
465    * @param ds data store
466    * @param entry content entry
467    * @param query query
468    */
469   private class JsonPathJdbcFeatureStore(ds: JDBCDataStore, entry: ContentEntry, query: Query)
470       extends JDBCFeatureSource(entry, query) with SimpleFeatureStore {
471 
472     /**
473      * Resolves filter property names against the schema, but preserves JSON paths.
474      *
475      * The base class runs `DataUtilities.resolvePropertyNames`, whose `PropertyNameResolvingVisitor`
476      * evaluates each property name against the feature type and rewrites it to the resolved attribute's
477      * local name. A JSON path like `$.props.name` evaluates (via GeoMesa's JSON property accessor) to
478      * the `props` descriptor, so the default would collapse the whole path to `props` - losing the nested
479      * field that IcebergFilterConverter needs to push down. We keep `$`-prefixed names verbatim and
480      * resolve everything else as usual.
481      *
482      * @param query the query being planned
483      * @return the query with non-JSON-path property names resolved
484      */
485     override protected def resolvePropertyNames(query: Query): Query = {
486       val filter = query.getFilter
487       if (filter == null || filter == Filter.INCLUDE || filter == Filter.EXCLUDE) { query } else {
488         val resolved = filter.accept(new JsonPathPropertyNameResolver(getSchema), null).asInstanceOf[Filter]
489         if (resolved == filter) { query } else {
490           val newQuery = new Query(query)
491           newQuery.setFilter(resolved)
492           newQuery
493         }
494       }
495     }
496 
497     override def setFeatures(reader: FeatureReader[SimpleFeatureType, SimpleFeature]): Unit = {
498       WithClose(ds.getFeatureWriter(entry.getTypeName, Filter.INCLUDE, getTransaction)) { writer =>
499         while (writer.hasNext) {
500           writer.next()
501           writer.remove()
502         }
503       }
504       WithClose(ds.getFeatureWriterAppend(entry.getTypeName, getTransaction)) { writer =>
505         while (reader.hasNext) {
506           FeatureUtils.write(writer, reader.next())
507         }
508       }
509     }
510 
511     override def addFeatures(features: FeatureCollection[SimpleFeatureType, SimpleFeature]): java.util.List[FeatureId] = {
512       val ids = new util.ArrayList[FeatureId]()
513       WithClose(ds.getFeatureWriterAppend(entry.getTypeName, getTransaction)) { writer =>
514         WithClose(features.features()) { iter =>
515           while (iter.hasNext) {
516             ids.add(FeatureUtils.write(writer, iter.next()).getIdentifier)
517           }
518         }
519       }
520       ids
521     }
522 
523     override def removeFeatures(filter: Filter): Unit = {
524       WithClose(ds.getFeatureWriter(entry.getTypeName, filter, getTransaction)) { writer =>
525         while (writer.hasNext) {
526           writer.next()
527           writer.remove()
528         }
529       }
530     }
531 
532     override def modifyFeatures(attributeName: String, attributeValue: AnyRef, filter: Filter): Unit =
533       modifyFeatures(Array(attributeName), Array(attributeValue), filter)
534 
535     override def modifyFeatures(attributeNames: Array[Name], attributeValues: Array[AnyRef], filter: Filter): Unit =
536       modifyFeatures(attributeNames.map(_.getLocalPart), attributeValues, filter)
537 
538     override def modifyFeatures(attributeName: Name, attributeValue: AnyRef, filter: Filter): Unit =
539       modifyFeatures(Array(attributeName.getLocalPart), Array(attributeValue), filter)
540 
541     override def modifyFeatures(attributeNames: Array[String], attributeValues: Array[AnyRef], filter: Filter): Unit = {
542       if(attributeNames.length != attributeValues.length) {
543         throw new IllegalArgumentException(
544           s"Attribute names and values do not align: ${attributeNames.length} names (${attributeNames.mkString(", ")}) vs " +
545             s"${attributeValues.length} values (${attributeValues.mkString(", ")})")
546       }
547       WithClose(ds.getFeatureWriter(entry.getTypeName, filter, getTransaction)) { writer =>
548         while (writer.hasNext) {
549           val sf = writer.next()
550           var i = 0
551           while (i < attributeNames.length) {
552             sf.setAttribute(attributeNames(i), attributeValues(i))
553             i += 1
554           }
555           writer.write()
556         }
557       }
558     }
