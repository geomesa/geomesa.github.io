45 /**
46  * Data store wrapper that manages a hidden per-row visibility column (`_vis`).
47  *
48  * The physical schema carries `_vis` as an ordinary attribute so that it flows through the sealed
49  * `JDBCDataStore` insert/read machinery for free. This wrapper hides that column from callers:
50  *
51  *  - on `createSchema`, if the `pg.vis.enabled` user-data flag is explicitly `true`, a nullable
52  *    `_vis` attribute is appended to the type before it's created
53  *  - on read, features are re-typed to drop `_vis`; the MainView already filters rows by the caller's
54  *    authorizations, and the visibility expression is surfaced into the feature user data
55  *  - on write, the feature's visibility (from user data) is copied into `_vis`
56  *
57  * For any type that has no physical `_vis` column (pre-existing schemas, or vis explicitly disabled)
58  * the wrapper is fully transparent
59  *
60  * @param delegate wrapped jdbc data store
61  */
62 class PartitionedPostgisDataStore(delegate: JDBCDataStore) extends DecoratingDataStore(delegate) {

71         override def load(typeName: String): SchemaType = {
72           val underlying = delegate.getSchema(typeName)
73           if (underlying == null) {

75           } else {
76             val visCol = underlying.indexOf(VisCol)

80               // verify vis is the last attribute, logic in other places depends on that
81               require(visCol == underlying.getAttributeCount - 1, s"Expected $VisCol to be last attribute, but it was not")
82               SchemaWithVis(removeVisFromSchema(underlying), underlying)
83             }
84           }
85         }
86       })

94   /**
95    * Re-create the PLPG/SQL procedures associated with a feature type. This can be used
96    * to 'upgrade in place' if the code is changed.
97    *
98    * This can be used to alter mutable user data settings, but *cannot* be used to modify the attributes
99    * of the feature type
100    *
101    * @param sft updated feature type
102    */
103   def upgrade(sft: SimpleFeatureType): Unit = {
104     val existing = loadSchema(sft.getTypeName)
105     if (existing == null) {
106       throw new IllegalArgumentException(s"Schema does not exist: ${sft.getTypeName}")
107     }
108     val dialect = this.dialect match {
109       case d: PartitionedPostgisDialect => d
110       case d: PartitionedPostgisPsDialect => d.delegate
111     }
112     val upgrade = existing match {
113       case _: SchemaWithoutVis => sft
114       case s: SchemaWithVis =>
115         val copy = SimpleFeatureTypes.copy(s.underlying)
116         copy.getUserData.putAll(sft.getUserData)
117         copy
118     }
119     WithClose(new DefaultTransaction()) { tx =>
120       WithClose(getConnection(tx)) { cx =>
121         dialect.postCreateTable(getDatabaseSchema, upgrade, cx)
122         tx.commit()
123       }
124     }
125   }

229    * Adds visibility to a schema, for underlying types
230    *
231    * @param sft user-facing feature type
232    * @return
233    */
234   private def addVisToSchema(sft: SimpleFeatureType): SimpleFeatureType = {
235     require(sft.indexOf(VisCol) == -1, s"'$VisCol' is a reserved attribute name")
236     // add the _vis column into the underlying table
237     val builder = new SimpleFeatureTypeBuilder()
238     builder.init(sft)
239     builder.nillable(true)
240     builder.add(PartitionedPostgisDialect.VisCol, classOf[String])
241     val result = builder.buildFeatureType()
242     result.getUserData.putAll(sft.getUserData)
243     result

262   /**
263    * Project a physical feature onto the user schema, dropping the `_vis` column but surfacing its
264    * value into the feature's user data as the visibility expression (if non-null/non-empty).
265    *
266    * @param from physical feature (with `_vis`)
267    * @param userFacingType user schema (without `_vis`)
268    * @return user feature with visibility populated in user data
269    */
270   private def toUserFacing(from: SimpleFeature, userFacingType: SimpleFeatureType): SimpleFeature = {
271     val to = new ScalaSimpleFeature(userFacingType, from.getID)
272     var i = 0
273     while (i < userFacingType.getAttributeCount) {
274       to.setAttributeNoConvert(i, from.getAttribute(i))
275       i += 1
276     }
277     val pgVis = from.getAttribute(i).asInstanceOf[String]
278     if (pgVis != null) {
279       val vis =
280         try { PgVis.fromPostgisVisibility(pgVis) } catch {
281           case NonFatal(e) =>
282             logger.warn(s"Error converting visibilities back to standard syntax - may be pre-release data: $pgVis", e)
283             pgVis
284         }
285       SecurityUtils.setFeatureVisibility(to, vis)
286     }
287 
288     to.getUserData.putAll(from.getUserData)
289     to
290   }

292   /**
293    * Gets the visibility from a user-facing feature, ensuring it is not null or empty
294    *
295    * @param sf feature
296    * @return
297    */
298   private def getVisibility(sf: SimpleFeature): String = {
299     val visibility = SecurityUtils.getVisibility(sf)
300     if (visibility == null || visibility.isBlank) {
301       throw new IllegalArgumentException(
302         s"Feature ${sf.getID} does not contain a visibility label under the user data key ${SecurityUtils.FEATURE_VISIBILITY}")
303     }
304     try {
305       PgVis.toPostgisVisibility(visibility)
306     } catch {
307       case NonFatal(e) =>
308         throw new IllegalArgumentException(s"Feature ${sf.getID} does not contain a valid visibility label: $visibility", e)
309     }
310   }

350   /**
351    * Feature writer that exposes the user schema but maps visibility into `_vis` on write. The
352    * delegate feature returned by `next()` is retained so that `write()` can copy the caller's
353    * attributes (including visibility) back into it before persisting.
354    */
355   private class VisFeatureWriter(delegate: FeatureWriter[SimpleFeatureType, SimpleFeature], userFacingType: SimpleFeatureType)
356       extends FeatureWriter[SimpleFeatureType, SimpleFeature] {
357 
358     private var delegateFeature: SimpleFeature = _
359     private var userFeature: SimpleFeature = _
360 
361     override def getFeatureType: SimpleFeatureType = userFacingType
362 
363     override def hasNext: Boolean = delegate.hasNext
364 
365     override def next(): SimpleFeature = {
366       delegateFeature = delegate.next()
367       // present a user-schema view (without '_vis') for the caller to populate
368       userFeature = toUserFacing(delegateFeature, userFacingType)
369       userFeature
370     }
371 
372     override def write(): Unit = {
373       var i = 0
374       while (i < userFacingType.getAttributeCount) {
375         delegateFeature.setAttribute(i, userFeature.getAttribute(i))
376         i += 1
377       }
378       delegateFeature.setAttribute(i, getVisibility(userFeature))
379       delegateFeature.getUserData.putAll(userFeature.getUserData)
380       val fid = Option(userFeature.getUserData.get(Hints.PROVIDED_FID).asInstanceOf[String]).getOrElse(userFeature.getID)
381       // propagate a caller-provided fid - the jdbc writer reads the id off the delegate feature itself
382       delegateFeature.getIdentifier match {
383         case id: FeatureIdImpl => id.setID(fid)
384         case _ => // no-op
385       }
386       delegate.write()
387     }
388 
389     override def remove(): Unit = delegate.remove()
390 
391     override def close(): Unit = delegate.close()
392   }

477   /**
478    * Replaces the JDBCFeatureSource with overrides to correctly resolve json path attributes. The method comes
479    * from ContentFeatureSource, so we can't override it in our delegating vis store/source (above). Since JDBCFeatureStore is
480    * a final class, write methods are re-routed back to the data store - this means that doing updates based on json-path
481    * expressions will not work.
482    *
483    * @param ds data store
484    * @param entry content entry
485    * @param query query
486    */
487   private class JsonPathJdbcFeatureStore(ds: JDBCDataStore, entry: ContentEntry, query: Query)
488       extends JDBCFeatureSource(entry, query) with SimpleFeatureStore {
489 
490     /**
491      * Resolves filter property names against the schema, but preserves JSON paths.
492      *
493      * The base class runs `DataUtilities.resolvePropertyNames`, whose `PropertyNameResolvingVisitor`
494      * evaluates each property name against the feature type and rewrites it to the resolved attribute's
495      * local name. A JSON path like `$.props.name` evaluates (via GeoMesa's JSON property accessor) to
496      * the `props` descriptor, so the default would collapse the whole path to `props` - losing the nested
497      * field that IcebergFilterConverter needs to push down. We keep `$`-prefixed names verbatim and
498      * resolve everything else as usual.
499      *
500      * @param query the query being planned
501      * @return the query with non-JSON-path property names resolved
502      */
503     override protected def resolvePropertyNames(query: Query): Query = {
504       val filter = query.getFilter
505       if (filter == null || filter == Filter.INCLUDE || filter == Filter.EXCLUDE) { query } else {
506         val resolved = filter.accept(new JsonPathPropertyNameResolver(getSchema), null).asInstanceOf[Filter]
507         if (resolved == filter) { query } else {
508           val newQuery = new Query(query)
509           newQuery.setFilter(resolved)
510           newQuery
511         }
512       }
513     }
514 
515     override def setFeatures(reader: FeatureReader[SimpleFeatureType, SimpleFeature]): Unit = {
516       WithClose(ds.getFeatureWriter(entry.getTypeName, Filter.INCLUDE, getTransaction)) { writer =>
517         while (writer.hasNext) {
518           writer.next()
519           writer.remove()
520         }
521       }
522       WithClose(ds.getFeatureWriterAppend(entry.getTypeName, getTransaction)) { writer =>
523         while (reader.hasNext) {
524           FeatureUtils.write(writer, reader.next())
525         }
526       }
527     }
528 
529     override def addFeatures(features: FeatureCollection[SimpleFeatureType, SimpleFeature]): java.util.List[FeatureId] = {
530       val ids = new util.ArrayList[FeatureId]()
531       WithClose(ds.getFeatureWriterAppend(entry.getTypeName, getTransaction)) { writer =>
532         WithClose(features.features()) { iter =>
533           while (iter.hasNext) {
534             ids.add(FeatureUtils.write(writer, iter.next()).getIdentifier)
535           }
536         }
537       }
538       ids
539     }
540 
541     override def removeFeatures(filter: Filter): Unit = {
542       WithClose(ds.getFeatureWriter(entry.getTypeName, filter, getTransaction)) { writer =>
543         while (writer.hasNext) {
544           writer.next()
545           writer.remove()
546         }
547       }
548     }
549 
550     override def modifyFeatures(attributeName: String, attributeValue: AnyRef, filter: Filter): Unit =
551       modifyFeatures(Array(attributeName), Array(attributeValue), filter)
552 
553     override def modifyFeatures(attributeNames: Array[Name], attributeValues: Array[AnyRef], filter: Filter): Unit =
554       modifyFeatures(attributeNames.map(_.getLocalPart), attributeValues, filter)
555 
556     override def modifyFeatures(attributeName: Name, attributeValue: AnyRef, filter: Filter): Unit =
557       modifyFeatures(Array(attributeName.getLocalPart), Array(attributeValue), filter)
558 
559     override def modifyFeatures(attributeNames: Array[String], attributeValues: Array[AnyRef], filter: Filter): Unit = {
560       if(attributeNames.length != attributeValues.length) {
561         throw new IllegalArgumentException(
562           s"Attribute names and values do not align: ${attributeNames.length} names (${attributeNames.mkString(", ")}) vs " +
563             s"${attributeValues.length} values (${attributeValues.mkString(", ")})")
564       }
565       WithClose(ds.getFeatureWriter(entry.getTypeName, filter, getTransaction)) { writer =>
566         while (writer.hasNext) {
567           val sf = writer.next()
568           var i = 0
569           while (i < attributeNames.length) {
570             sf.setAttribute(attributeNames(i), attributeValues(i))
571             i += 1
572           }
573           writer.write()
574         }
575       }
576     }
