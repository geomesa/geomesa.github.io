19   import SessionDataSource.AuthConfigName

21   override protected def createStatements(info: TypeInfo): Seq[String] = {
22     // if visibilities are enabled, filter each branch by evaluating the hidden '_vis' column against the
23     // caller's authorizations, stamped into the 'geomesa.auths' session variable by SessionDataSource.
24     // the auths array is built in an uncorrelated sub-select so that it's evaluated once per query (as an
25     // init plan) rather than re-running current_setting/string_to_array for every row
26     val visibilityFilter =
27       info.cols.vis.map(vis => s"pg_vis(${vis.quoted}, (SELECT string_to_array(current_setting('$AuthConfigName', true), ',')))")
28     def filter(conditions: Seq[String]): String =
29       if (conditions.isEmpty) { "" } else { conditions.mkString(" WHERE ", " AND ", "") }
30     val writeAheadPartitionsFilter = filter(
31       Seq(
32         s"""NOT EXISTS (
33            |  SELECT FROM ${info.tables.writeAheadMigrations.name.qualified} migration
34            |  WHERE write_ahead_partition.${info.cols.dtg.quoted} >= migration.partition_start
35            |    AND write_ahead_partition.${info.cols.dtg.quoted} < migration.partition_end
36            |)""".stripMargin
37       ) ++ visibilityFilter
38     )
39     val numericFilter = filter(
40       Seq(s"NOT EXISTS (SELECT FROM ${info.tables.writeAheadRetirements.name.qualified} retired " +
41           "WHERE retired.partition_oid = write_ahead.tableoid)") ++ visibilityFilter
42     )
43     def ageOffFilter(alias: String): String = filter(
44       Seq(s"NOT EXISTS (SELECT FROM ${info.tables.ageOffQueue.name.qualified} queue " +
45           s"WHERE queue.partition_oid = $alias.tableoid)") ++ visibilityFilter
46     )
47     Seq(
48       s"""CREATE OR REPLACE VIEW ${info.tables.view.name.qualified} AS
49          |  SELECT * FROM ${info.tables.writeAhead.name.qualified} write_ahead$numericFilter UNION ALL
50          |  SELECT * FROM ${info.tables.writeAheadPartitions.name.qualified} write_ahead_partition$writeAheadPartitionsFilter UNION ALL
51          |  SELECT * FROM ${info.tables.mainPartitions.name.qualified} main_partition${ageOffFilter("main_partition")} UNION ALL
52          |  SELECT * FROM ${info.tables.spillPartitions.name.qualified} spill_partition${ageOffFilter("spill_partition")};""".stripMargin
53     )
54   }
55 
56   override protected def dropStatements(info: TypeInfo): Seq[String] =
57     Seq(s"DROP VIEW IF EXISTS ${info.tables.view.name.qualified};")
