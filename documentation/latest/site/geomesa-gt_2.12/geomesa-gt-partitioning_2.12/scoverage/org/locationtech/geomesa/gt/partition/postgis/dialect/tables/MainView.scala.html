23   /**
24    * Where visible clause. Assumes user auths are accessible via `user_auths.vis`
25    *
26    * @param table table or alias
27    * @param col vis column
28    * @return
29    */
30   def whereVisible(table: String, col: ColumnName): String =
31     s"$table.${col.quoted} IS NOT NULL AND $table.${col.quoted} <> '' AND $table.${col.quoted}::tsquery @@ user_auths.vis"
32 
33   override protected def createStatements(info: TypeInfo): Seq[String] = {
34     // if visibilities are enabled, filter each branch by evaluating the hidden '_vis' column against the
35     // caller's authorizations, stamped into the 'geomesa.auths' session variable by SessionDataSource
36 
37     val withUserAuths = info.cols.vis.fold("") { _ =>
38       s"""  -- compute the user's authorizations as a native tsvector
39          |  -- use NOT MATERIALIZED so that it can be pushed into each table scan
40          |  WITH user_auths AS NOT MATERIALIZED (
41          |    SELECT array_to_tsvector(string_to_array(current_setting('$AuthConfigName', true), ',')) AS vis
42          |  )""".stripMargin
43     }
44     val joinUserAuths = info.cols.vis.fold("")(_ => "CROSS JOIN user_auths")
45     def andWhereVisible(alias: String): String = info.cols.vis.fold("")(v => s"AND ${whereVisible(alias, v)}")
46 
47     // include the data from the previous generation of this type, if there is one. columns are matched by name,
48     // with nulls for any columns that don't exist in the previous generation
49     // note: the previous view already filters on visibilities and chains to any generations before it
50     val previous = info.previous.fold("") { prev =>
51       val existing = prev.cols.all.map(_.raw).toSet
52       val cols = Seq(s"prev.${prev.cols.fid.quoted}") ++ info.cols.all.map { col =>
53         // note: the null is typed based on the write ahead table, so that the view column keeps its type modifiers
54         if (existing.contains(col.raw)) { s"prev.${col.quoted}" } else { s"(NULL::${info.tables.writeAhead.name.qualified}).${col.quoted}" }
55       }
56       s"""
57          |  UNION ALL
58          |  SELECT ${cols.mkString(", ")} FROM ${prev.tables.view.name.qualified} prev""".stripMargin
59     }
60 
61     val select =
62       s"""$withUserAuths
63          |  SELECT wa.* FROM ${info.tables.writeAhead.name.qualified} wa $joinUserAuths WHERE
64          |    NOT EXISTS (
65          |      SELECT FROM ${info.tables.writeAheadRetirements.name.qualified} retired
66          |        WHERE retired.partition_oid = wa.tableoid
67          |    ) ${andWhereVisible("wa")}
68          |  UNION ALL
69          |  SELECT wap.* FROM ${info.tables.writeAheadPartitions.name.qualified} wap $joinUserAuths WHERE
70          |    NOT EXISTS (
71          |      SELECT FROM ${info.tables.writeAheadMigrations.name.qualified} migration
72          |        WHERE wap.${info.cols.dtg.quoted} >= migration.partition_start
73          |        AND wap.${info.cols.dtg.quoted} < migration.partition_end
74          |    ) ${andWhereVisible("wap")}
75          |  UNION ALL
76          |  SELECT mp.* FROM ${info.tables.mainPartitions.name.qualified} mp $joinUserAuths WHERE
77          |    NOT EXISTS (
78          |      SELECT FROM ${info.tables.ageOffQueue.name.qualified} queue
79          |        WHERE queue.partition_oid = mp.tableoid
80          |    ) ${andWhereVisible("mp")}
81          |  UNION ALL
82          |  SELECT sp.* FROM ${info.tables.spillPartitions.name.qualified} sp $joinUserAuths WHERE
83          |    NOT EXISTS (
84          |      SELECT FROM ${info.tables.ageOffQueue.name.qualified} queue
85          |        WHERE queue.partition_oid = sp.tableoid
86          |    ) ${andWhereVisible("sp")}$previous""".stripMargin
87 
88     // replacing a view requires an exclusive lock on it, which will block behind any queries against the view (and
89     // will then block any new queries). to avoid that, we only replace the view if the definition has changed, which
90     // we determine by comparing it to a temporary view
91     // note: if we don't have the privileges to create temporary views, we fall back to always replacing the view
92     val create =
93       s"""DO $$VIEW$$
94          |DECLARE
95          |  existing_def text;
96          |  updated_def text;
97          |BEGIN
98          |  IF to_regclass(${literal(info.tables.view.name.qualified)}) IS NOT NULL THEN
99          |    BEGIN
100          |      CREATE TEMP VIEW $TempView AS
101          |$select;
102          |      existing_def := pg_get_viewdef(${info.tables.view.name.asRegclass});
103          |      updated_def := pg_get_viewdef('pg_temp.$TempView'::regclass);
104          |      DROP VIEW pg_temp.$TempView;
105          |    EXCEPTION WHEN insufficient_privilege THEN
106          |      existing_def := NULL;
107          |    END;
108          |  END IF;
109          |  IF existing_def IS NULL OR existing_def <> updated_def THEN
110          |    CREATE OR REPLACE VIEW ${info.tables.view.name.qualified} AS
111          |$select;
112          |  END IF;
113          |END$$VIEW$$;""".stripMargin
114 
115     Seq(create)
116   }
