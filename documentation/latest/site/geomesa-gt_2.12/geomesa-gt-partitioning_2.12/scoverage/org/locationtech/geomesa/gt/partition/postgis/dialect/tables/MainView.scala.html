15  * Main view of all the partitions and write ahead table. This should accept and reads and writes.

17 object MainView extends SqlStatements {
18 
19   import SessionDataSource.AuthConfigName
20 
21   override protected def createStatements(info: TypeInfo): Seq[String] = {
22     // if visibilities are enabled, filter each branch by evaluating the hidden '_vis' column against the
23     // caller's authorizations, stamped into the 'geomesa.auths' session variable by SessionDataSource.
24     // the auths array is built in an uncorrelated sub-select so that it's evaluated once per query (as an
25     // init plan) rather than re-running current_setting/string_to_array for every row
26     val filter =
27       info.cols.vis.fold("")(vis => s" WHERE pg_vis(${vis.quoted}, (SELECT string_to_array(current_setting('$AuthConfigName', true), ',')))")
28     Seq(
29       s"""CREATE OR REPLACE VIEW ${info.tables.view.name.qualified} AS
30          |  SELECT * FROM ${info.tables.writeAhead.name.qualified}$filter UNION ALL
31          |  SELECT * FROM ${info.tables.writeAheadPartitions.name.qualified}$filter UNION ALL
32          |  SELECT * FROM ${info.tables.mainPartitions.name.qualified}$filter UNION ALL
33          |  SELECT * FROM ${info.tables.spillPartitions.name.qualified}$filter;""".stripMargin
34     )
35   }
36 
37   override protected def dropStatements(info: TypeInfo): Seq[String] =
38     Seq(s"DROP VIEW IF EXISTS ${info.tables.view.name.qualified};")
