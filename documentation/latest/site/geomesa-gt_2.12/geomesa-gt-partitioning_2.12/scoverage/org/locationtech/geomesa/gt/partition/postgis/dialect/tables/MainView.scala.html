21   /**
22    * Where visible clause. Assumes user auths are accessible via `user_auths.vis`
23    *
24    * @param table table or alias
25    * @param col vis column
26    * @return
27    */
28   def whereVisible(table: String, col: ColumnName): String =
29     s"$table.${col.quoted} IS NOT NULL AND $table.${col.quoted} <> '' AND $table.${col.quoted}::tsquery @@ user_auths.vis"
30 
31   override protected def createStatements(info: TypeInfo): Seq[String] = {
32     // if visibilities are enabled, filter each branch by evaluating the hidden '_vis' column against the
33     // caller's authorizations, stamped into the 'geomesa.auths' session variable by SessionDataSource
34 
35     val withUserAuths = info.cols.vis.fold("") { _ =>
36       s"""  -- compute the user's authorizations as a native tsvector
37          |  -- use NOT MATERIALIZED so that it can be pushed into each table scan
38          |  WITH user_auths AS NOT MATERIALIZED (
39          |    SELECT array_to_tsvector(string_to_array(current_setting('$AuthConfigName', true), ',')) AS vis
40          |  )""".stripMargin
41     }
42     val joinUserAuths = info.cols.vis.fold("")(_ => "CROSS JOIN user_auths")
43     def andWhereVisible(alias: String): String = info.cols.vis.fold("")(v => s"AND ${whereVisible(alias, v)}")
44 
45     Seq(
46       s"""CREATE OR REPLACE VIEW ${info.tables.view.name.qualified} AS
47          |$withUserAuths
48          |  SELECT wa.* FROM ${info.tables.writeAhead.name.qualified} wa $joinUserAuths WHERE
49          |    NOT EXISTS (
50          |      SELECT FROM ${info.tables.writeAheadRetirements.name.qualified} retired
51          |        WHERE retired.partition_oid = wa.tableoid
52          |    ) ${andWhereVisible("wa")}
53          |  UNION ALL
54          |  SELECT wap.* FROM ${info.tables.writeAheadPartitions.name.qualified} wap $joinUserAuths WHERE
55          |    NOT EXISTS (
56          |      SELECT FROM ${info.tables.writeAheadMigrations.name.qualified} migration
57          |        WHERE wap.${info.cols.dtg.quoted} >= migration.partition_start
58          |        AND wap.${info.cols.dtg.quoted} < migration.partition_end
59          |    ) ${andWhereVisible("wap")}
60          |  UNION ALL
61          |  SELECT mp.* FROM ${info.tables.mainPartitions.name.qualified} mp $joinUserAuths WHERE
62          |    NOT EXISTS (
63          |      SELECT FROM ${info.tables.ageOffQueue.name.qualified} queue
64          |        WHERE queue.partition_oid = mp.tableoid
65          |    ) ${andWhereVisible("mp")}
66          |  UNION ALL
67          |  SELECT sp.* FROM ${info.tables.spillPartitions.name.qualified} sp $joinUserAuths WHERE
68          |    NOT EXISTS (
69          |      SELECT FROM ${info.tables.ageOffQueue.name.qualified} queue
70          |        WHERE queue.partition_oid = sp.tableoid
71          |    ) ${andWhereVisible("sp")}
72          |;""".stripMargin
73     )
74   }
75 
76   override protected def dropStatements(info: TypeInfo): Seq[String] =
77     Seq(s"DROP VIEW IF EXISTS ${info.tables.view.name.qualified};")
