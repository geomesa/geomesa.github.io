15 /**
16  * Moves the current write ahead table to a sequential name, and creates a new write ahead table
17  * to accept writes going forward
18  */
19 object RollWriteAheadLog extends SqlProcedure with CronSchedule {
20 
21   override def name(info: TypeInfo): FunctionName = FunctionName(s"${info.typeIdentifier}_roll_wa")
22 
23   override def jobName(info: TypeInfo): SqlLiteral = SqlLiteral(s"${info.typeIdentifier}-roll-wa")
24 
25   override protected def schedule(info: TypeInfo): SqlLiteral = SqlLiteral("9,19,29,39,49,59 * * * *")
26 
27   override protected def invocation(info: TypeInfo): SqlLiteral = SqlLiteral(s"CALL ${name(info).quoted}()")
28 
29   override protected def createStatements(info: TypeInfo): Seq[String] =
30     Seq(proc(info)) ++ super.createStatements(info)
31 
32   private def proc(info: TypeInfo): String = {
33     s"""CREATE OR REPLACE PROCEDURE ${info.schema.quoted}.${name(info).quoted}() LANGUAGE plpgsql AS
34        |  $$BODY$$
35        |    DECLARE
36        |      seq_val smallint;
37        |      attempt integer;
38        |      next_oid oid;
39        |      cur_partition text;        -- rename of the _writes table
40        |      next_partition text;       -- next partition that will be created from the _writes table
41        |      partition_tablespace text; -- table space command
42        |      index_space text;          -- index table space command
43        |      table_wa_logging text;     -- wa log options
44        |      pexists boolean;           -- table exists check
45        |    BEGIN
46        |
47        |      -- Serialize slot selection with rotation and cleanup before reading the current sequence.
48        |      LOCK TABLE ONLY ${info.tables.writeAhead.name.qualified} IN SHARE UPDATE EXCLUSIVE MODE;
49        |      SELECT value FROM ${info.schema.quoted}.${SequenceTable.Name.quoted}
50        |        WHERE type_name = ${literal(info.typeName)} INTO seq_val;
51        |      cur_partition := ${literal(info.tables.writeAhead.name.raw + "_")} || lpad(seq_val::text, 3, '0');
52        |
53        |      -- don't re-create the table if there hasn't been any data inserted
54        |      EXECUTE 'SELECT EXISTS(SELECT 1 FROM ${info.schema.quoted}.' || quote_ident(cur_partition) || ')'
55        |          INTO pexists;
56        |      IF pexists THEN
57        |        -- find the next slot, skipping any tables queued for retirement but not yet deleted
58        |        FOR attempt IN 1..999 LOOP
59        |          seq_val := (seq_val + 1) % 1000;
60        |          next_partition := ${literal(info.tables.writeAhead.name.raw + "_")} || lpad(seq_val::text, 3, '0');
61        |          SELECT c.oid INTO next_oid FROM pg_class c JOIN pg_namespace n ON n.oid = c.relnamespace
62        |            WHERE n.nspname = ${info.schema.asLiteral} AND c.relname = next_partition;
63        |          pexists := FOUND;
64        |          CONTINUE WHEN EXISTS (SELECT FROM ${info.tables.writeAheadRetirements.name.qualified}
65        |            WHERE partition_oid = next_oid);
66        |          EXIT WHEN NOT pexists;
67        |        END LOOP;
68        |        IF pexists THEN
69        |          RAISE EXCEPTION 'No available numeric write-ahead slots remain for %', ${literal(info.typeName)};
70        |        END IF;
71        |        UPDATE ${info.schema.quoted}.${SequenceTable.Name.quoted} SET value = seq_val
72        |          WHERE type_name = ${literal(info.typeName)};
73        |
74        |        SELECT value FROM ${info.schema.quoted}.${UserDataTable.Name.quoted}
75        |          WHERE type_name = ${literal(info.typeName)} AND key = ${literal(SftUserData.WriteAheadTableSpace.key)}
76        |          INTO partition_tablespace;
77        |        IF partition_tablespace IS NULL THEN
78        |          partition_tablespace := '';
79        |          index_space := '';
80        |        ELSE
81        |          partition_tablespace := ' TABLESPACE ' || quote_ident(partition_tablespace);
82        |          index_space := ' USING INDEX' || partition_tablespace;
83        |        END IF;
84        |
85        |        SELECT value FROM ${info.schema.quoted}.${UserDataTable.Name.quoted}
86        |          WHERE type_name = ${literal(info.typeName)} AND key = ${literal(SftUserData.WalLogEnabled.key)}
87        |          INTO table_wa_logging;
88        |        IF table_wa_logging IS NULL OR table_wa_logging::boolean THEN
89        |          table_wa_logging := '';
90        |        ELSE
91        |          table_wa_logging := 'UNLOGGED ';
92        |        END IF;
93        |
94        |        -- requires SHARE UPDATE EXCLUSIVE
95        |        EXECUTE 'CREATE ' || table_wa_logging || 'TABLE IF NOT EXISTS ${info.schema.quoted}.' || quote_ident(next_partition) || '(' ||
96        |          'CONSTRAINT ' || quote_ident(next_partition || '_pkey') ||
97        |          ' PRIMARY KEY (${info.cols.fid.quoted}, ${info.cols.dtg.quoted})' || index_space || ')' ||
98        |          ' INHERITS (${info.tables.writeAhead.name.qualified})${info.tables.writeAhead.storage.opts}' || partition_tablespace;
99        |${info.cols.filter(info.tables.writeAhead.name).indices.map { index =>
100     s"""        EXECUTE 'CREATE INDEX IF NOT EXISTS ' || quote_ident(next_partition || ${literal(index.cols.map(_.raw).mkString("_", "_", ""))}) ||
101        |          ' ON ${info.schema.quoted}.' || quote_ident(next_partition) ||
102        |          ' ${index.using} (${index.cols.map(_.quoted).mkString(", ")} ${index.opclass}) ${index.includes} ${index.storageOpts}' || partition_tablespace;""".stripMargin}.mkString("\n")}
103        |        COMMIT; -- releases our locks
104        |
105        |        EXECUTE 'ANALYZE ' || quote_ident(cur_partition);
106        |
107        |      END IF;
108        |    END;
109        |  $$BODY$$;""".stripMargin
110   }
