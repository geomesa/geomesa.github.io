15 /**
16  * Moves the current write ahead table to a sequential name, and creates a new write ahead table
17  * to accept writes going forward
18  */
19 object RollWriteAheadLog extends SqlProcedure with CronSchedule {
20 
21   override def name(info: TypeInfo): FunctionName = FunctionName(s"${info.typeIdentifier}_roll_wa")
22 
23   override def jobName(info: TypeInfo): SqlLiteral = SqlLiteral(s"${info.typeIdentifier}-roll-wa")
24 
25   override protected def schedule(info: TypeInfo): SqlLiteral = SqlLiteral("9,19,29,39,49,59 * * * *")
26 
27   override protected def invocation(info: TypeInfo): SqlLiteral = SqlLiteral(s"CALL ${name(info).quoted}()")
28 
29   override protected def createStatements(info: TypeInfo): Seq[String] =
30     Seq(proc(info)) ++ super.createStatements(info)
31 
32   private def proc(info: TypeInfo): String = {
33     s"""CREATE OR REPLACE PROCEDURE ${info.schema.quoted}.${name(info).quoted}() LANGUAGE plpgsql AS
34        |  $$BODY$$
35        |    DECLARE
36        |      seq_val smallint;
37        |      attempt integer;
38        |      next_oid oid;
39        |      cur_partition text;        -- rename of the _writes table
40        |      next_partition text;       -- next partition that will be created from the _writes table
41        |      partition_tablespace text; -- table space command
42        |      index_space text;          -- index table space command
43        |      table_wa_logging text;     -- wa log options
44        |      pexists boolean;           -- table exists check
45        |    BEGIN
46        |
47        |      ${UserDataConfig.select(info,
48                 "partition_tablespace" -> SftUserData.WriteAheadTableSpace,
49                 "table_wa_logging" -> SftUserData.WalLogEnabled)}
50        |      ${UserDataConfig.tablespaceClauses("partition_tablespace", "index_space")}
51        |      ${UserDataConfig.walLoggingClause("table_wa_logging")}
52        |      ${UserDataConfig.release}
53        |
54        |      -- Serialize slot selection with rotation and cleanup before reading the current sequence.
55        |      LOCK TABLE ONLY ${info.tables.writeAhead.name.qualified} IN SHARE UPDATE EXCLUSIVE MODE;
56        |      SELECT value FROM ${info.schema.quoted}.${SequenceTable.Name.quoted}
57        |        WHERE type_name = ${literal(info.typeName)} INTO seq_val;
58        |      cur_partition := ${literal(info.tables.writeAhead.name.raw + "_")} || lpad(seq_val::text, 3, '0');
59        |
60        |      -- don't re-create the table if there hasn't been any data inserted
61        |      EXECUTE 'SELECT EXISTS(SELECT 1 FROM ${info.schema.quoted}.' || quote_ident(cur_partition) || ')'
62        |          INTO pexists;
63        |      IF pexists THEN
64        |        -- find the next slot, skipping any tables queued for retirement but not yet deleted
65        |        FOR attempt IN 1..999 LOOP
66        |          seq_val := (seq_val + 1) % 1000;
67        |          next_partition := ${literal(info.tables.writeAhead.name.raw + "_")} || lpad(seq_val::text, 3, '0');
68        |          SELECT c.oid INTO next_oid FROM pg_class c JOIN pg_namespace n ON n.oid = c.relnamespace
69        |            WHERE n.nspname = ${info.schema.asLiteral} AND c.relname = next_partition;
70        |          pexists := FOUND;
71        |          CONTINUE WHEN EXISTS (SELECT FROM ${info.tables.writeAheadRetirements.name.qualified}
72        |            WHERE partition_oid = next_oid);
73        |          EXIT WHEN NOT pexists;
74        |        END LOOP;
75        |        IF pexists THEN
76        |          RAISE EXCEPTION 'No available numeric write-ahead slots remain for %', ${literal(info.typeName)};
77        |        END IF;
78        |        UPDATE ${info.schema.quoted}.${SequenceTable.Name.quoted} SET value = seq_val
79        |          WHERE type_name = ${literal(info.typeName)};
80        |
81        |        -- requires SHARE UPDATE EXCLUSIVE
82        |        EXECUTE 'CREATE ' || table_wa_logging || 'TABLE IF NOT EXISTS ${info.schema.quoted}.' || quote_ident(next_partition) || '(' ||
83        |          'CONSTRAINT ' || quote_ident(next_partition || '_pkey') ||
84        |          ' PRIMARY KEY (${info.cols.fid.quoted}, ${info.cols.dtg.quoted})' || index_space || ')' ||
85        |          ' INHERITS (${info.tables.writeAhead.name.qualified})${info.tables.writeAhead.storage.opts}' || partition_tablespace;
86        |${info.cols.filter(info.tables.writeAhead.name).indices.map { index =>
87     s"""        EXECUTE 'CREATE INDEX IF NOT EXISTS ' || quote_ident(next_partition || ${literal(index.cols.map(_.raw).mkString("_", "_", ""))}) ||
88        |          ' ON ${info.schema.quoted}.' || quote_ident(next_partition) ||
89        |          ' ${index.using} (${index.cols.map(_.quoted).mkString(", ")} ${index.opclass}) ${index.includes} ${index.storageOpts}' || partition_tablespace;""".stripMargin}.mkString("\n")}
90        |        COMMIT; -- releases our locks
91        |
92        |        EXECUTE 'ANALYZE ' || quote_ident(cur_partition);
93        |
94        |      END IF;
95        |    END;
96        |  $$BODY$$;""".stripMargin
97   }
