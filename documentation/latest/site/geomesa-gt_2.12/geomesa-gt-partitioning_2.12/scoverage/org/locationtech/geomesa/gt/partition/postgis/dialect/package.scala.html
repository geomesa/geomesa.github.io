95     /**
96      * Execute a sql command
97      *
98      * @param sql sql string
99      */
100     def execute(sql: String): Unit = {
101       try {
102         logger.debug(sql)
103         statement.execute(sql)
104       } catch {
105         case NonFatal(e) => rollback = true; throw e
106       }
107     }
108 
109     /**
110      * Execute an update statements
111      *
112      * @param sql sql string, with arguments denoted by '?'
113      * @param args arguments to the update statement, must match the argument placeholders in the sql string
114      */
115     def executeUpdate(sql: String, args: Seq[Any]): Unit = {
116       val ps: PreparedStatement = cx.prepareStatement(sql)
117       try {
118         var i = 1
119         args.foreach { arg =>
120           ps.setObject(i, arg)
121           i += 1
122         }
123         logger.debug(s"$sql, ${args.mkString(", ")}")
124         ps.executeUpdate()
125       } catch {
126         case NonFatal(e) => rollback = true; throw e
127       } finally {
128         ps.close()
129       }
130     }
131 
132     override def close(): Unit = {
133       if (!cx.getAutoCommit) {
134         if (rollback) {
135           Try(cx.rollback())
136         } else {
137           cx.commit()
138         }
139       }
140       statement.close()
141     }

326     def apply(original: SimpleFeatureType): Columns = {
327       val sft = SimpleFeatureTypes.copy(original)

329       val fid = ColumnName(SftUserData.FidColumn.get(sft))

337       // TODO we could use covering indices, but BRIN indices don't support them

339       //    Seq(dtg) ++ sft.getAttributeDescriptors.asScala.collect { case d if d.isIndexValue() => ColumnName(d) }

345             !i.isBlank && !"false".equalsIgnoreCase(i) && !"none".equalsIgnoreCase(i)

591            |      $action ON ${tgTable.qualified}
592            |      FOR EACH ROW EXECUTE PROCEDURE ${info.schema.quoted}.${name(info).quoted}();
593            |  END IF;
