95     /**
96      * Execute a sql command

100     def execute(sql: String): Unit = {
101       try {
102         logger.debug(sql)
103         statement.execute(sql)
104       } catch {
105         case NonFatal(e) => rollback = true; throw e
106       }
107     }

109     /**
110      * Execute an update statements
111      *
112      * @param sql sql string, with arguments denoted by '?'
113      * @param args arguments to the update statement, must match the argument placeholders in the sql string
114      */
115     def executeUpdate(sql: String, args: Seq[Any]): Unit = {
116       val ps: PreparedStatement = cx.prepareStatement(sql)
117       try {
118         var i = 1
119         args.foreach { arg =>
120           ps.setObject(i, arg)
121           i += 1
122         }
123         logger.debug(s"$sql, ${args.mkString(", ")}")
124         ps.executeUpdate()
125       } catch {
126         case NonFatal(e) => rollback = true; throw e
127       } finally {
128         ps.close()
129       }
130     }

132     override def close(): Unit = {
133       if (!cx.getAutoCommit) {
134         if (rollback) {
135           Try(cx.rollback())
136         } else {
137           cx.commit()
138         }
139       }
140       statement.close()
141     }

666   /**
667    * Uses a postgres advisory lock to synchronize create and drop statements
668    */
669   trait AdvisoryLock extends Sql {
670 
671     /**
672      * The lock id associated with this object
673      *
674      * @return
675      */
676     protected def lockId: Long
677 
678     // session-scoped locks
679     private def sessionLock: String = s"SELECT pg_advisory_lock($lockId);"
680     private def sessionUnlock: String = s"SELECT pg_advisory_unlock($lockId);"
681     // transaction-scoped lock, automatically released when the enclosing transaction commits or rolls back
682     private def xactLock: String = s"SELECT pg_advisory_xact_lock($lockId);"
683 
684     /**
685      * Acquire the advisory lock and run the given operation, releasing the lock afterward.
686      *
687      * The catalog changes made by `op` are not committed until the enclosing command sequence finishes. When the
688      * connection is in a transaction (auto-commit off), we use a transaction-scoped lock so that the lock is held
689      * until commit/rollback - a session-scoped lock would be released as soon as `op` returns, allowing a concurrent
690      * transaction to modify the same catalog tuples before this one commits, resulting in 'tuple concurrently
691      * updated' errors. When auto-commit is on, each statement commits immediately so there is no open transaction to
692      * scope the lock to, and we fall back to a session-scoped lock with an explicit unlock.
693      */
694     private def locked(op: => Unit)(implicit ex: ExecutionContext): Unit = {
695       if (ex.cx.getAutoCommit) {
696         ex.execute(sessionLock)
697         try { op } finally {
698           ex.execute(sessionUnlock)
699         }
700       } else {
701         ex.execute(xactLock)
702         op
703       }
704     }
705 
706     abstract override def create(info: TypeInfo)(implicit ex: ExecutionContext): Unit =
707       locked(super.create(info))
708 
709     abstract override def drop(info: TypeInfo)(implicit ex: ExecutionContext): Unit =
710       locked(super.drop(info))
711   }
