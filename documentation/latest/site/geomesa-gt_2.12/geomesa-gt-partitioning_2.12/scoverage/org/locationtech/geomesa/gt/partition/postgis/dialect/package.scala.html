84   /**
85    * Wrapper to facilitate jdbc calls.
86    *
87    * Note: does not do any validation, sql must be sanitized by the caller.
88    *
89    * @param cx connection
90    */
91   class ExecutionContext(val cx: Connection) extends Closeable with StrictLogging {
92 
93     private val statement = cx.createStatement()
94     private var rollback = false
95 
96     /**
97      * Execute a sql command
98      *
99      * @param sql sql string
100      */
101     def execute(sql: String): Unit = {
102       try {
103         logger.debug(sql)
104         statement.execute(sql)
105       } catch {
106         case NonFatal(e) => rollback = true; throw e
107       }
108     }
109 
110     /**
111      * Execute an update statements
112      *
113      * @param sql sql string, with arguments denoted by '?'
114      * @param args arguments to the update statement, must match the argument placeholders in the sql string
115      */
116     def executeUpdate(sql: String, args: Seq[Any]): Unit = {
117       val ps: PreparedStatement = cx.prepareStatement(sql)
118       try {
119         var i = 1
120         args.foreach { arg =>
121           ps.setObject(i, arg)
122           i += 1
123         }
124         logger.debug(s"$sql, ${args.mkString(", ")}")
125         ps.executeUpdate()
126       } catch {
127         case NonFatal(e) => rollback = true; throw e
128       } finally {
129         ps.close()
130       }
131     }
132 
133     override def close(): Unit = {
134       if (!cx.getAutoCommit) {
135         if (rollback) {
136           Try(cx.rollback())
137         } else {
138           cx.commit()
139         }
140       }
141       statement.close()
142     }
143   }

216   object Tables {
217     def apply(sft: SimpleFeatureType, schema: String, typeIdentifier: String): Tables = {
218       val logged = SftUserData.WalLogEnabled.get(sft)
219       val view = TableConfig(schema, sft.getTypeName, None)
220       val tablePrefix = TableConfig(schema, typeIdentifier, None).name.raw
221       // we disable autovacuum for write ahead tables, as they are transient and get dropped fairly quickly
222       val writeAhead = TableConfig(schema, tablePrefix + WriteAheadTableSuffix.raw, SftUserData.WriteAheadTableSpace.get(sft), vacuum = false, logged)
223       val writeAheadPartitions = TableConfig(schema, tablePrefix + PartitionedWriteAheadTableSuffix.raw, SftUserData.WriteAheadPartitionsTableSpace.get(sft), vacuum = false, logged)
224       val toastTupleTarget = SftUserData.ToastTupleTarget.get(sft)
225       require(toastTupleTarget.forall(t => t >= 128 && t <= 8160),
226         s"Toast tuple target must be between 128 and 8160: ${toastTupleTarget.orNull}")
227       val mainPartitions = TableConfig(schema, tablePrefix + PartitionedTableSuffix.raw, SftUserData.MainTableSpace.get(sft), logged = logged, toastTupleTarget = toastTupleTarget)
228       val spillPartitions = TableConfig(schema, tablePrefix + SpillTableSuffix.raw, SftUserData.MainTableSpace.get(sft), logged = logged, toastTupleTarget = toastTupleTarget)
229       val analyzeQueue = TableConfig(schema, tablePrefix + AnalyzeTableSuffix.raw, None, logged = logged)
230       val sortQueue = TableConfig(schema, tablePrefix + SortTableSuffix.raw, None, logged = logged)
231       Tables(view, writeAhead, writeAheadPartitions, mainPartitions, spillPartitions, analyzeQueue, sortQueue)
232     }
233   }

312   /**
313    * Columns names for the feature type
314    *
315    * @param dtg primary dtg column
316    * @param geom primary geometry column
317    * @param geoms all geometry cols (including primary)
318    * @param indexed any indexed columns
319    * @param vis hidden visibility column, if present
320    * @param all all columns
321    * @param dtgIncludes columns carried by the dtg index, if it is covering
322    */
323   case class Columns(
324       fid: ColumnName,
325       dtg: ColumnName,
326       geom: ColumnName,
327       geoms: Seq[ColumnName],
328       indexed: Seq[ColumnName],
329       vis: Option[ColumnName],
330       all: Seq[ColumnName],
331       dtgIncludes: Seq[ColumnName] = Seq.empty
332     ) {
333 
334     /**
335      * The keys and INCLUDE clause of the dtg index, e.g. `("dtg", "_vis") INCLUDE ("geom")`
336      */
337     def dtgIndexColumns: String = {
338       // carry visibility as a trailing key so repeated (dtg, visibility) pairs can be deduplicated
339       val keys = (Seq(dtg) ++ vis).map(_.quoted).mkString("(", ", ", ")")
340       if (dtgIncludes.isEmpty) { keys } else { s"$keys INCLUDE (${dtgIncludes.map(_.quoted).mkString(", ")})" }
341     }
342   }

344   object Columns {
345 
346     import org.locationtech.geomesa.utils.geotools.RichSimpleFeatureType.RichSimpleFeatureType
347 
348     import scala.collection.JavaConverters._
349 
350     def apply(original: SimpleFeatureType): Columns = {
351       val sft = SimpleFeatureTypes.copy(original)
352       TemporalIndexCheck.validateDtgField(sft)
353       val fid = ColumnName(SftUserData.FidColumn.get(sft))
354       val dtg = sft.getDtgField.map(ColumnName.apply).getOrElse {
355         throw new IllegalArgumentException("Must include a date-type attribute when using a partitioned store")
356       }
357       val geom = Option(sft.getGeomField).map(ColumnName.apply).getOrElse {
358         throw new IllegalArgumentException("Must include a geometry-type attribute when using a partitioned store")
359       }
360       val geoms = sft.getAttributeDescriptors.asScala.collect { case d: GeometryDescriptor => ColumnName(d) }
361       // TODO we could use covering indices, but BRIN indices don't support them
362       //  val includes =
363       //    Seq(dtg) ++ sft.getAttributeDescriptors.asScala.collect { case d if d.isIndexValue() => ColumnName(d) }
364       //    .distinct
365       def indexed(d: AttributeDescriptor): Boolean =
366         d.getLocalName != geom.raw && d.getLocalName != dtg.raw &&
367           Option(d.getUserData.get(AttributeOptions.OptIndex).asInstanceOf[String]).exists { i =>
368             // accept any index flags, so config can be consistent across data stores
369             !i.isBlank && !"false".equalsIgnoreCase(i) && !"none".equalsIgnoreCase(i)
370           }
371 
372       val indices = sft.getAttributeDescriptors.asScala.collect { case d if indexed(d) => ColumnName(d) }
373       val vis = sft.getAttributeDescriptors.asScala.collectFirst {
374         case d if d.getLocalName == PartitionedPostgisDialect.VisCol => ColumnName(d)
375       }
376       val all = sft.getAttributeDescriptors.asScala.map(ColumnName.apply)
377       // the geometry only when it's a point: an index row is limited to ~2.7kB, and a larger geometry
378       // would fail the insert
379       val dtgIncludes =
380         if (SftUserData.CoveringDtgIndex.get(sft) &&
381             classOf[Point].isAssignableFrom(sft.getGeometryDescriptor.getType.getBinding)) { Seq(geom) } else { Seq.empty }
382       Columns(fid, dtg, geom, geoms.toSeq, indices.toSeq, vis, all.toSeq, dtgIncludes)
383     }
384   }

401    *
402    * @param hoursPerPartition number of hours to keep in each partition, must be a divisor of 24
403    * @param pagesPerRange number of pages to add to each range in the BRIN index
404    * @param maxPartitions max number of partitions to keep
405    * @param cronMinute minute of each 10 minute chunk that partition maintenance will run

414   object PartitionInfo {
415     def apply(sft: SimpleFeatureType): PartitionInfo = {
416       val hours = SftUserData.IntervalHours.get(sft)
417       require(hours > 0 && hours <= 24, s"Partition interval must be between 1 and 24 hours: $hours hours")
418       require(24 % hours == 0, s"Partition interval must be a divisor of 24 hours: $hours hours")
419       val cronMinute = SftUserData.CronMinute.get(sft)
420       require(cronMinute.forall(m => m >= 0 && m < 9), s"Cron minute must be between 0 and 8: ${cronMinute.orNull}")
421       val pagesPerRange = SftUserData.PagesPerRange.get(sft)
422       require(pagesPerRange >= 0, s"Pages per range but be a positive number: $pagesPerRange")
423       val maxPartitions = SftUserData.MaxPartitions.get(sft)
424       PartitionInfo(hours, pagesPerRange, maxPartitions, cronMinute)
425     }

586   /**
587    * A table trigger with an associated function definition
588    */
589   trait SqlTriggerFunction extends SqlFunction {
590 
591     /**
592      * Name of the trigger
593      *
594      * @param info type info
595      * @return
596      */
597     def triggerName(info: TypeInfo): TriggerName = TriggerName(s"${name(info).raw}_trigger")
598 
599     /**
600      * Table that the trigger is applied to
601      *
602      * @param info type info
603      * @return
604      */
605     protected def table(info: TypeInfo): TableIdentifier
606 
607     /**
608      * Triggered action (e.g. "INSTEAD OF INSERT", "BEFORE INSERT", "INSTEAD OF DELETE", etc)
609      *
610      * @return
611      */
612     protected def action: String
613 
614     override protected def createStatements(info: TypeInfo): Seq[String] = {
615       // there is no "create or replace" for triggers so we have to wrap it in a check
616       val tgName = triggerName(info)
617       val tgTable = table(info)
618       val create =
619         s"""DO $$$$
620            |BEGIN
621            |  IF NOT EXISTS (SELECT FROM pg_trigger WHERE tgname = ${tgName.asLiteral} AND tgrelid = ${tgTable.asRegclass}) THEN
622            |    CREATE TRIGGER ${tgName.quoted}
623            |      $action ON ${tgTable.qualified}
624            |      FOR EACH ROW EXECUTE PROCEDURE ${info.schema.quoted}.${name(info).quoted}();
625            |  END IF;
626            |END$$$$;""".stripMargin
627       Seq(create)
628     }
629 
630     override protected def dropStatements(info: TypeInfo): Seq[String] =
631       Seq(dropTrigger(info)) ++ super.dropStatements(info)
632 
633     private def dropTrigger(info: TypeInfo): String =
634       s"DROP TRIGGER IF EXISTS ${triggerName(info).quoted} ON ${table(info).qualified};"
635   }

669         s"""DO $$$$

681            |BEGIN

683            |END$$$$;""".stripMargin

685     }

688       s"""  IF EXISTS (SELECT FROM cron.job WHERE jobname = $quotedName) THEN
689          |    PERFORM cron.unschedule($quotedName);
