87   /**
88    * Wrapper to facilitate jdbc calls.
89    *
90    * Note: does not do any validation, sql must be sanitized by the caller.
91    *
92    * @param cx connection
93    */
94   class ExecutionContext(val cx: Connection) extends Closeable with StrictLogging {
95 
96     private val statement = cx.createStatement()
97     private var rollback = false
98 
99     /**
100      * Execute a sql command
101      *
102      * @param sql sql string
103      */
104     def execute(sql: String): Unit = {
105       try {
106         logger.debug(sql)
107         statement.execute(sql)
108       } catch {
109         case NonFatal(e) => rollback = true; throw e
110       }
111     }
112 
113     /**
114      * Execute a statement with variable binding
115      *
116      * @param sql sql string, with arguments denoted by '?'
117      * @param args arguments to the update statement, must match the argument placeholders in the sql string
118      */
119     def execute(sql: String, args: Seq[Any]): Unit = execute(sql, args, _.execute())
120 
121     /**
122      * Execute an update statement with variable binding
123      *
124      * @param sql sql string, with arguments denoted by '?'
125      * @param args arguments to the update statement, must match the argument placeholders in the sql string
126      */
127     def executeUpdate(sql: String, args: Seq[Any]): Unit = execute(sql, args, _.executeUpdate())
128 
129     private def execute(sql: String, args: Seq[Any], op: PreparedStatement => Unit): Unit = {
130       val ps: PreparedStatement = cx.prepareStatement(sql)
131       try {
132         var i = 1
133         args.foreach { arg =>
134           ps.setObject(i, arg)
135           i += 1
136         }
137         logger.debug(s"$sql, ${args.mkString(", ")}")
138         op(ps)
139       } catch {
140         case NonFatal(e) => rollback = true; throw e
141       } finally {
142         ps.close()
143       }
144     }
145 
146     override def close(): Unit = {
147       if (!cx.getAutoCommit) {
148         if (rollback) {
149           Try(cx.rollback())
150         } else {
151           cx.commit()
152         }
153       }
154       statement.close()
155     }
156   }

235   object Tables {
236     def apply(sft: SimpleFeatureType, schema: String, typeIdentifier: String): Tables = {
237       val logged = SftUserData.WalLogEnabled.get(sft)
238       val view = TableConfig(schema, sft.getTypeName, None)
239       val tablePrefix = TableConfig(schema, typeIdentifier, None).name.raw
240       // we disable autovacuum for write ahead tables, as they are transient and get dropped fairly quickly
241       val writeAhead = TableConfig(schema, tablePrefix + WriteAheadTableSuffix.raw, SftUserData.WriteAheadTableSpace.get(sft), vacuum = false, logged)
242       val writeAheadPartitions = TableConfig(schema, tablePrefix + PartitionedWriteAheadTableSuffix.raw, SftUserData.WriteAheadPartitionsTableSpace.get(sft), vacuum = false, logged)
243       val toastTupleTarget = SftUserData.ToastTupleTarget.get(sft)
244       require(toastTupleTarget.forall(t => t >= 128 && t <= 8160),
245         s"Toast tuple target must be between 128 and 8160: ${toastTupleTarget.orNull}")
246       val mainPartitions = TableConfig(schema, tablePrefix + PartitionedTableSuffix.raw, SftUserData.MainTableSpace.get(sft), logged = logged, toastTupleTarget = toastTupleTarget)
247       val spillPartitions = TableConfig(schema, tablePrefix + SpillTableSuffix.raw, SftUserData.MainTableSpace.get(sft), logged = logged, toastTupleTarget = toastTupleTarget)
248       val analyzeQueue = TableConfig(schema, tablePrefix + AnalyzeTableSuffix.raw, None, logged = logged)
249       val sortQueue = TableConfig(schema, tablePrefix + SortTableSuffix.raw, None, logged = logged)
250       val writeAheadMigrations = TableConfig(schema, tablePrefix + WriteAheadMigrationsTableSuffix.raw, None)
251       val ageOffQueue = TableConfig(schema, tablePrefix + AgeOffQueueTableSuffix.raw, None)
252       val writeAheadRetirements = TableConfig(schema, tablePrefix + WriteAheadRetirementsTableSuffix.raw, None)
253       Tables(view, writeAhead, writeAheadPartitions, mainPartitions, spillPartitions, analyzeQueue, sortQueue,
254         writeAheadMigrations, ageOffQueue, writeAheadRetirements)
255     }

363   object Columns {
364 
365     import org.locationtech.geomesa.utils.geotools.RichAttributeDescriptors.RichAttributeDescriptor
366     import org.locationtech.geomesa.utils.geotools.RichSimpleFeatureType.RichSimpleFeatureType
367 
368     import scala.collection.JavaConverters._
369 
370     def apply(original: SimpleFeatureType, tables: Tables, partitions: PartitionInfo): Columns = {
371       val sft = SimpleFeatureTypes.copy(original)
372       TemporalIndexCheck.validateDtgField(sft)
373       val fid = ColumnName(SftUserData.FidColumn.get(sft))
374       val dtg = sft.getDtgField.map(ColumnName.apply).getOrElse {
375         throw new IllegalArgumentException("Must include a date-type attribute when using a partitioned store")
376       }
377       val geom = Option(sft.getGeomField).map(ColumnName.apply).getOrElse {
378         throw new IllegalArgumentException("Must include a geometry-type attribute when using a partitioned store")
379       }
380       val vis = sft.getAttributeDescriptors.asScala.collectFirst {
381         case d if d.getLocalName == PartitionedPostgisDialect.VisCol => ColumnName(d)
382       }
383       val all = sft.getAttributeDescriptors.asScala.map(ColumnName.apply)
384 
385       def indexed(d: AttributeDescriptor): Boolean =
386         Option(d.getUserData.get(AttributeOptions.OptIndex).asInstanceOf[String]).exists { i =>
387           // accept any index flags, so config can be consistent across data stores
388           !i.isBlank && !"false".equalsIgnoreCase(i) && !"none".equalsIgnoreCase(i)
389         }
390 
391       // TODO we could use covering indices, but BRIN indices don't support them
392       //  val includes =
393       //    Seq(dtg) ++ sft.getAttributeDescriptors.asScala.collect { case d if d.isIndexValue() => ColumnName(d) }
394       //    .distinct
395 
396       val indices = sft.getAttributeDescriptors.asScala.flatMap { d =>
397         if (d.getLocalName == dtg.raw) {
398           // the geometry only when it's a point: an index row is limited to ~2.7kB, and a larger geometry would fail the insert
399           val includes =
400             if (SftUserData.CoveringDtgIndex.get(sft) &&
401               classOf[Point].isAssignableFrom(sft.getGeometryDescriptor.getType.getBinding)) { s"INCLUDE (${geom.quoted})" } else { "" }
402           // carry visibility as a trailing key so repeated (dtg, visibility) pairs can be deduplicated
403           Seq(ColumnIndex(Seq(dtg) ++ vis, includes = includes))
404         } else if (d.isInstanceOf[GeometryDescriptor]) {
405           val col = Seq(ColumnName(d))
406           val brinTables = Seq(tables.mainPartitions.name)
407           val gistTables = Seq(tables.writeAhead, tables.writeAheadPartitions, tables.spillPartitions).map(_.name)
408           val brin =
409             ColumnIndex(col, using = "USING brin", storageOpts = s"WITH (pages_per_range = ${partitions.pagesPerRange})", tables = Some(brinTables))
410           val gist = ColumnIndex(col, using = "USING gist", tables = Some(gistTables))
411           Seq(gist, brin)
412         } else if (indexed(d)) {
413           if (d.isJson()) {
414             Seq(ColumnIndex(Seq(ColumnName(d)), using = "USING gin", opclass = "jsonb_path_ops"))
415           } else {
416             Seq(ColumnIndex(Seq(ColumnName(d))))
417           }
418         } else {
419           Seq.empty
420         }
421       }
422 
423       Columns(fid, dtg, geom, vis, all.toSeq, indices.toSeq)
424     }
425   }

664     override protected def createStatements(info: TypeInfo): Seq[String] = {
665       // there is no "create or replace" for triggers so we have to wrap it in a check
666       val tgName = triggerName(info)
667       val tgTable = table(info)
668       val create =
669         s"""DO $$$$
670            |BEGIN
671            |  IF NOT EXISTS (SELECT FROM pg_trigger WHERE tgname = ${tgName.asLiteral} AND tgrelid = ${tgTable.asRegclass}) THEN
672            |    CREATE TRIGGER ${tgName.quoted}
673            |      $action ON ${tgTable.qualified}
674            |      FOR EACH ROW EXECUTE PROCEDURE ${info.schema.quoted}.${name(info).quoted}();
675            |  END IF;
676            |END$$$$;""".stripMargin
677       Seq(create)
678     }

770     /**
771      * Acquire the advisory lock and run the given operation, releasing the lock afterward.
772      *
773      * The catalog changes made by `op` are not committed until the enclosing command sequence finishes. When the
774      * connection is in a transaction (auto-commit off), we use a transaction-scoped lock so that the lock is held
775      * until commit/rollback - a session-scoped lock would be released as soon as `op` returns, allowing a concurrent
776      * transaction to modify the same catalog tuples before this one commits, resulting in 'tuple concurrently
777      * updated' errors. When auto-commit is on, each statement commits immediately so there is no open transaction to
778      * scope the lock to, and we fall back to a session-scoped lock with an explicit unlock.
