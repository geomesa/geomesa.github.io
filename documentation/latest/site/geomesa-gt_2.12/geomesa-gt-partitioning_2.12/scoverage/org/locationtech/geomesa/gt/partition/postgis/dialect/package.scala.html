140     private def execute(sql: String, args: Seq[Any], op: PreparedStatement => Unit): Unit = {
141       val ps: PreparedStatement = cx.prepareStatement(sql)
142       try {
143         var i = 1
144         args.foreach { arg =>
145           ps.setObject(i, arg)
146           i += 1
147         }
148         logger.debug(s"$sql, ${args.mkString(", ")}")
149         op(ps)
150       } catch {
151         case NonFatal(e) => rollback = true; throw e
152       } finally {
153         ps.close()
154       }
155     }

157     override def close(): Unit = {
158       if (commit && !cx.getAutoCommit) {
159         if (rollback) {
160           Try(cx.rollback())
161         } else {
162           cx.commit()
163         }
164       }
165       if (created != null) {
166         created.close()
167       }
168     }

296     def apply(sft: SimpleFeatureType, schema: String, typeIdentifier: String): Tables = {
297       val logged = SftUserData.WalLogEnabled.get(sft)
298       val view = TableConfig(schema, sft.getTypeName, None)

301       val writeAhead = TableConfig(schema, tablePrefix + WriteAheadTableSuffix.raw, SftUserData.WriteAheadTableSpace.get(sft), vacuum = false, logged)

307       val spillPartitions = TableConfig(schema, tablePrefix + SpillTableSuffix.raw, SftUserData.MainTableSpace.get(sft), logged = logged, toastTupleTarget = toastTupleTarget)
308       val analyzeQueue = TableConfig(schema, tablePrefix + AnalyzeTableSuffix.raw, None, logged = logged)
309       val sortQueue = TableConfig(schema, tablePrefix + SortTableSuffix.raw, None, logged = logged)
310       val writeAheadMigrations = TableConfig(schema, tablePrefix + WriteAheadMigrationsTableSuffix.raw, None)
311       val ageOffQueue = TableConfig(schema, tablePrefix + AgeOffQueueTableSuffix.raw, None)
312       val writeAheadRetirements = TableConfig(schema, tablePrefix + WriteAheadRetirementsTableSuffix.raw, None)

430     /**
431      * Checks if an attribute is flagged for indexing - accepts any index flags, so config can be consistent
432      * across data stores
433      *
434      * @param d attribute descriptor
435      * @return
436      */
437     def indexed(d: AttributeDescriptor): Boolean = {
438       Option(d.getUserData.get(AttributeOptions.OptIndex)).map(String.valueOf).exists { i =>
439         !i.isBlank && !"false".equalsIgnoreCase(i) && !"none".equalsIgnoreCase(i)
440       }
441     }
442 
443     def apply(original: SimpleFeatureType, tables: Tables, partitions: PartitionInfo): Columns = {
444       val sft = SimpleFeatureTypes.copy(original)
445       TemporalIndexCheck.validateDtgField(sft)
446       val fid = ColumnName(SftUserData.FidColumn.get(sft))
447       val dtg = sft.getDtgField.map(ColumnName.apply).getOrElse {
448         throw new IllegalArgumentException("Must include a date-type attribute when using a partitioned store")
449       }
450       // note: a feature type loaded by the jdbc store always has the first geometry as its default geometry, so
451       // for an existing type we use the default geometry that was recorded when it was created
452       val geom = SftUserData.GeomField.get(sft).orElse(Option(sft.getGeomField)).map(ColumnName.apply).getOrElse {
453         throw new IllegalArgumentException("Must include a geometry-type attribute when using a partitioned store")
454       }
455       val geomBinding = sft.getDescriptor(geom.raw) match {
456         case d: GeometryDescriptor => d.getType.getBinding
457         case _ => throw new IllegalArgumentException(s"Default geometry '${geom.raw}' is not a geometry-type attribute")
458       }
459       val vis = sft.getAttributeDescriptors.asScala.collectFirst {
460         case d if d.getLocalName == PartitionedPostgisDialect.VisCol => ColumnName(d)
461       }
462       val all = sft.getAttributeDescriptors.asScala.map(ColumnName.apply)
463 
464       // TODO we could use covering indices, but BRIN indices don't support them
465       //  val includes =
466       //    Seq(dtg) ++ sft.getAttributeDescriptors.asScala.collect { case d if d.isIndexValue() => ColumnName(d) }
467       //    .distinct
468 
469       val indices = sft.getAttributeDescriptors.asScala.flatMap { d =>
470         if (d.getLocalName == dtg.raw) {
471           // the geometry only when it's a point: an index row is limited to ~2.7kB, and a larger geometry would fail the insert
472           val includes =
473             if (SftUserData.CoveringDtgIndex.get(sft) &&
474               classOf[Point].isAssignableFrom(geomBinding)) { s"INCLUDE (${geom.quoted})" } else { "" }
475           // carry visibility as a trailing key so repeated (dtg, visibility) pairs can be deduplicated
476           Seq(ColumnIndex(Seq(dtg) ++ vis, includes = includes))
477         } else if (d.isInstanceOf[GeometryDescriptor]) {
478           val col = Seq(ColumnName(d))
479           val brinTables = Seq(tables.mainPartitions.name)
480           val gistTables = Seq(tables.writeAhead, tables.writeAheadPartitions, tables.spillPartitions).map(_.name)
481           val brin =
482             ColumnIndex(col, using = "USING brin", storageOpts = s"WITH (pages_per_range = ${partitions.pagesPerRange})", tables = Some(brinTables))
483           val gist = ColumnIndex(col, using = "USING gist", tables = Some(gistTables))
484           Seq(gist, brin)
485         } else if (indexed(d)) {
486           if (d.isJson()) {
487             Seq(ColumnIndex(Seq(ColumnName(d)), using = "USING gin", opclass = "jsonb_path_ops"))
488           } else {
489             Seq(ColumnIndex(Seq(ColumnName(d))))
490           }
491         } else {
492           Seq.empty
493         }
494       }
495 
496       Columns(fid, dtg, geom, vis, all.toSeq, indices.toSeq)
497     }
498   }

849      *

856     private def sessionUnlock: String = s"SELECT pg_advisory_unlock($lockId);"
857     // transaction-scoped lock, automatically released when the enclosing transaction commits or rolls back
858     private def xactLock: String = s"SELECT pg_advisory_xact_lock($lockId);"
859 
860     /**
861      * Acquire the advisory lock and run the given operation, releasing the lock afterward.
862      *
863      * The catalog changes made by `op` are not committed until the enclosing command sequence finishes. When the
864      * connection is in a transaction (auto-commit off), we use a transaction-scoped lock so that the lock is held
865      * until commit/rollback - a session-scoped lock would be released as soon as `op` returns, allowing a concurrent
866      * transaction to modify the same catalog tuples before this one commits, resulting in 'tuple concurrently
867      * updated' errors. When auto-commit is on, each statement commits immediately so there is no open transaction to
868      * scope the lock to, and we fall back to a session-scoped lock with an explicit unlock.
869      */
870     def locked(op: => Unit)(implicit ex: ExecutionContext): Unit = {
871       if (ex.cx.getAutoCommit) {
872         ex.execute(sessionLock)
873         try { op } finally {
874           ex.execute(sessionUnlock)
875         }
876       } else {
877         ex.execute(xactLock)
878         op
879       }
880     }
