84   /**
85    * Wrapper to facilitate jdbc calls.
86    *
87    * Note: does not do any validation, sql must be sanitized by the caller.
88    *
89    * @param cx connection
90    */
91   class ExecutionContext(val cx: Connection) extends Closeable with StrictLogging {
92 
93     private val statement = cx.createStatement()
94     private var rollback = false
95 
96     /**
97      * Execute a sql command
98      *
99      * @param sql sql string
100      */
101     def execute(sql: String): Unit = {
102       try {
103         logger.debug(sql)
104         statement.execute(sql)
105       } catch {
106         case NonFatal(e) => rollback = true; throw e
107       }
108     }
109 
110     /**
111      * Execute an update statements
112      *
113      * @param sql sql string, with arguments denoted by '?'
114      * @param args arguments to the update statement, must match the argument placeholders in the sql string
115      */
116     def executeUpdate(sql: String, args: Seq[Any]): Unit = {
117       val ps: PreparedStatement = cx.prepareStatement(sql)
118       try {
119         var i = 1
120         args.foreach { arg =>
121           ps.setObject(i, arg)
122           i += 1
123         }
124         logger.debug(s"$sql, ${args.mkString(", ")}")
125         ps.executeUpdate()
126       } catch {
127         case NonFatal(e) => rollback = true; throw e
128       } finally {
129         ps.close()
130       }
131     }
132 
133     override def close(): Unit = {
134       if (!cx.getAutoCommit) {
135         if (rollback) {
136           Try(cx.rollback())
137         } else {
138           cx.commit()
139         }
140       }
141       statement.close()
142     }
143   }

216   object Tables {
217     def apply(sft: SimpleFeatureType, schema: String, typeIdentifier: String): Tables = {
218       val logged = SftUserData.WalLogEnabled.get(sft)
219       val view = TableConfig(schema, sft.getTypeName, None)
220       val tablePrefix = TableConfig(schema, typeIdentifier, None).name.raw
221       // we disable autovacuum for write ahead tables, as they are transient and get dropped fairly quickly
222       val writeAhead = TableConfig(schema, tablePrefix + WriteAheadTableSuffix.raw, SftUserData.WriteAheadTableSpace.get(sft), vacuum = false, logged)
223       val writeAheadPartitions = TableConfig(schema, tablePrefix + PartitionedWriteAheadTableSuffix.raw, SftUserData.WriteAheadPartitionsTableSpace.get(sft), vacuum = false, logged)
224       val toastTupleTarget = SftUserData.ToastTupleTarget.get(sft)
225       require(toastTupleTarget.forall(t => t >= 128 && t <= 8160),
226         s"Toast tuple target must be between 128 and 8160: ${toastTupleTarget.orNull}")
227       val mainPartitions = TableConfig(schema, tablePrefix + PartitionedTableSuffix.raw, SftUserData.MainTableSpace.get(sft), logged = logged, toastTupleTarget = toastTupleTarget)
228       val spillPartitions = TableConfig(schema, tablePrefix + SpillTableSuffix.raw, SftUserData.MainTableSpace.get(sft), logged = logged, toastTupleTarget = toastTupleTarget)
229       val analyzeQueue = TableConfig(schema, tablePrefix + AnalyzeTableSuffix.raw, None, logged = logged)
230       val sortQueue = TableConfig(schema, tablePrefix + SortTableSuffix.raw, None, logged = logged)
231       Tables(view, writeAhead, writeAheadPartitions, mainPartitions, spillPartitions, analyzeQueue, sortQueue)
232     }
233   }

340   object Columns {
341 
342     import org.locationtech.geomesa.utils.geotools.RichAttributeDescriptors.RichAttributeDescriptor
343     import org.locationtech.geomesa.utils.geotools.RichSimpleFeatureType.RichSimpleFeatureType
344 
345     import scala.collection.JavaConverters._
346 
347     def apply(original: SimpleFeatureType, tables: Tables, partitions: PartitionInfo): Columns = {
348       val sft = SimpleFeatureTypes.copy(original)
349       TemporalIndexCheck.validateDtgField(sft)
350       val fid = ColumnName(SftUserData.FidColumn.get(sft))
351       val dtg = sft.getDtgField.map(ColumnName.apply).getOrElse {
352         throw new IllegalArgumentException("Must include a date-type attribute when using a partitioned store")
353       }
354       val geom = Option(sft.getGeomField).map(ColumnName.apply).getOrElse {
355         throw new IllegalArgumentException("Must include a geometry-type attribute when using a partitioned store")
356       }
357       val vis = sft.getAttributeDescriptors.asScala.collectFirst {
358         case d if d.getLocalName == PartitionedPostgisDialect.VisCol => ColumnName(d)
359       }
360       val all = sft.getAttributeDescriptors.asScala.map(ColumnName.apply)
361 
362       def indexed(d: AttributeDescriptor): Boolean =
363         Option(d.getUserData.get(AttributeOptions.OptIndex).asInstanceOf[String]).exists { i =>
364           // accept any index flags, so config can be consistent across data stores
365           !i.isBlank && !"false".equalsIgnoreCase(i) && !"none".equalsIgnoreCase(i)
366         }
367 
368       // TODO we could use covering indices, but BRIN indices don't support them
369       //  val includes =
370       //    Seq(dtg) ++ sft.getAttributeDescriptors.asScala.collect { case d if d.isIndexValue() => ColumnName(d) }
371       //    .distinct
372 
373       val indices = sft.getAttributeDescriptors.asScala.flatMap { d =>
374         if (d.getLocalName == dtg.raw) {
375           // the geometry only when it's a point: an index row is limited to ~2.7kB, and a larger geometry would fail the insert
376           val includes =
377             if (SftUserData.CoveringDtgIndex.get(sft) &&
378               classOf[Point].isAssignableFrom(sft.getGeometryDescriptor.getType.getBinding)) { s"INCLUDE (${geom.quoted})" } else { "" }
379           // carry visibility as a trailing key so repeated (dtg, visibility) pairs can be deduplicated
380           Seq(ColumnIndex(Seq(dtg) ++ vis, includes = includes))
381         } else if (d.isInstanceOf[GeometryDescriptor]) {
382           val col = Seq(ColumnName(d))
383           val brinTables = Seq(tables.mainPartitions.name)
384           val gistTables = Seq(tables.writeAhead, tables.writeAheadPartitions, tables.spillPartitions).map(_.name)
385           val brin =
386             ColumnIndex(col, using = "USING brin", storageOpts = s"WITH (pages_per_range = ${partitions.pagesPerRange})", tables = Some(brinTables))
387           val gist = ColumnIndex(col, using = "USING gist", tables = Some(gistTables))
388           Seq(gist, brin)
389         } else if (indexed(d)) {
390           if (d.isJson()) {
391             Seq(ColumnIndex(Seq(ColumnName(d)), using = "USING gin", opclass = "jsonb_path_ops"))
392           } else {
393             Seq(ColumnIndex(Seq(ColumnName(d))))
394           }
395         } else {
396           Seq.empty
397         }
398       }
399 
400       Columns(fid, dtg, geom, vis, all.toSeq, indices.toSeq)
401     }
402   }

441   object PartitionInfo {
442     def apply(sft: SimpleFeatureType): PartitionInfo = {
443       val hours = SftUserData.IntervalHours.get(sft)
444       require(hours > 0 && hours <= 24, s"Partition interval must be between 1 and 24 hours: $hours hours")

449       require(pagesPerRange >= 0, s"Pages per range but be a positive number: $pagesPerRange")
450       val maxPartitions = SftUserData.MaxPartitions.get(sft)
451       PartitionInfo(hours, pagesPerRange, maxPartitions, cronMinute)

613   /**
614    * A table trigger with an associated function definition
615    */
616   trait SqlTriggerFunction extends SqlFunction {
617 
618     /**
619      * Name of the trigger
620      *
621      * @param info type info
622      * @return
623      */
624     def triggerName(info: TypeInfo): TriggerName = TriggerName(s"${name(info).raw}_trigger")
625 
626     /**
627      * Table that the trigger is applied to
628      *
629      * @param info type info
630      * @return
631      */
632     protected def table(info: TypeInfo): TableIdentifier
633 
634     /**
635      * Triggered action (e.g. "INSTEAD OF INSERT", "BEFORE INSERT", "INSTEAD OF DELETE", etc)
636      *
637      * @return
638      */

640 
641     override protected def createStatements(info: TypeInfo): Seq[String] = {
642       // there is no "create or replace" for triggers so we have to wrap it in a check
643       val tgName = triggerName(info)
644       val tgTable = table(info)
645       val create =
646         s"""DO $$$$
647            |BEGIN
648            |  IF NOT EXISTS (SELECT FROM pg_trigger WHERE tgname = ${tgName.asLiteral} AND tgrelid = ${tgTable.asRegclass}) THEN
649            |    CREATE TRIGGER ${tgName.quoted}
650            |      $action ON ${tgTable.qualified}
651            |      FOR EACH ROW EXECUTE PROCEDURE ${info.schema.quoted}.${name(info).quoted}();
652            |  END IF;
653            |END$$$$;""".stripMargin
654       Seq(create)
655     }
656 
657     override protected def dropStatements(info: TypeInfo): Seq[String] =
658       Seq(dropTrigger(info)) ++ super.dropStatements(info)
659 
660     private def dropTrigger(info: TypeInfo): String =
661       s"DROP TRIGGER IF EXISTS ${triggerName(info).quoted} ON ${table(info).qualified};"
662   }
