12 /**
13  * Write ahead log, partitioned using inheritance to avoid any restraints on data overlap between partitions.
14  * All writes get directed to the main partition, which is identified with the suffix `_writes`. Every 10 minutes,
15  * the current writes partition is renamed based on a sequence number, and a new writes partition is created
16  */
17 object WriteAheadTable extends SqlStatements {

19   override protected def createStatements(info: TypeInfo): Seq[String] = {
20     val table = info.tables.writeAhead
21     val (tableTs, indexTs) = table.tablespace match {
22       case None => ("", "")
23       case Some(ts) => (s" TABLESPACE ${ts.quoted}", s" USING INDEX TABLESPACE ${ts.quoted}")
24     }
25     val logging = if (info.tables.writeAhead.logged) { "" } else { "UNLOGGED" }
26     val renameMainTable =
27       s"""DO $$$$
28          |DECLARE
29          |  main_table_exists boolean;
30          |  wa_table_exists boolean;
31          |BEGIN
32          |  SELECT EXISTS (
33          |    SELECT 1
34          |    FROM pg_catalog.pg_class c
35          |    JOIN pg_catalog.pg_namespace n ON n.oid = c.relnamespace
36          |    WHERE n.nspname = ${literal(info.schema.raw)}
37          |      AND c.relname = ${literal(info.tables.view.name.raw)}
38          |      AND c.relkind = 'r' -- indicates a table, not a view
39          |  ) INTO main_table_exists;
40          |  SELECT EXISTS (
41          |    SELECT 1
42          |    FROM pg_catalog.pg_class c
43          |    JOIN pg_catalog.pg_namespace n ON n.oid = c.relnamespace
44          |    WHERE n.nspname = ${literal(info.schema.raw)}
45          |      AND c.relname = ${literal(table.name.raw)}
46          |      AND c.relkind = 'r' -- indicates a table, not a view
47          |  ) INTO wa_table_exists;
48          |
49          |  IF main_table_exists THEN
50          |    IF wa_table_exists THEN
51          |      -- schema was partially deleted but _wa table is still here
52          |      DROP TABLE ${info.tables.view.name.qualified};
53          |    ELSE
54          |      ALTER TABLE ${info.tables.view.name.qualified} RENAME TO ${table.name.quoted};
55          |    END IF;
56          |  END IF;
57          |END $$$$;""".stripMargin
58     val move = table.tablespace.toSeq.map { ts =>
59       s"ALTER TABLE ${table.name.qualified} SET TABLESPACE ${ts.quoted};\n"
60     }
61     val createFirstPartition =
62       s"""DO $$$$
63          |DECLARE
64          |  seq_val smallint;
65          |  partition text;
66          |BEGIN
67          |  SELECT value from ${info.schema.quoted}.${SequenceTable.Name.quoted}
68          |    WHERE type_name = ${literal(info.typeName)} INTO seq_val;
69          |  partition := ${literal(table.name.raw + "_")} || lpad(seq_val::text, 3, '0');
70          |
71          |  EXECUTE 'CREATE $logging TABLE IF NOT EXISTS ${info.schema.quoted}.' || quote_ident(partition) || '(' ||
72          |    'CONSTRAINT ' || quote_ident(partition || '_pkey') ||
73          |    ' PRIMARY KEY (${info.cols.fid.quoted}, ${info.cols.dtg.quoted})$indexTs ' ||
74          |    ') INHERITS (${table.name.qualified})${table.storage.opts}$tableTs';
75          |  EXECUTE 'CREATE INDEX IF NOT EXISTS ' || quote_ident(partition || '_' || ${info.cols.dtg.asLiteral}) ||
76          |    ' ON ${info.schema.quoted}.' || quote_ident(partition) || ' (${info.cols.dtg.quoted})$tableTs';
77          |${info.cols.geoms.map { col =>
78       s"""  EXECUTE 'CREATE INDEX IF NOT EXISTS ' || quote_ident(partition || '_spatial_' || ${col.asLiteral}) ||
79          |    ' ON ${info.schema.quoted}.' || quote_ident(partition) || ' USING gist(${col.quoted})$tableTs';""".stripMargin}.mkString("\n")}
80          |${info.cols.indexed.map { col =>
81       s"""  EXECUTE 'CREATE INDEX IF NOT EXISTS ' || quote_ident(partition || '_' || ${col.asLiteral}) ||
82          |    ' ON ${info.schema.quoted}.' || quote_ident(partition) || '(${col.quoted})$tableTs';""".stripMargin}.mkString("\n")}
83          |END $$$$;""".stripMargin
84 
85     Seq(renameMainTable) ++ move ++ Seq(createFirstPartition)
86   }

92     )
93   }
