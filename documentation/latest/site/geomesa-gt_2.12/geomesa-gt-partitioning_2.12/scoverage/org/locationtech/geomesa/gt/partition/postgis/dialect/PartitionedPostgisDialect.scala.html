105   override def postCreateTable(schemaName: String, original: SimpleFeatureType, cx: Connection): Unit = {
106 
107     // note: we skip the call to `super`, which creates a spatial index (that we don't want), and which
108     // alters the geometry column types (which we handle in the create statement)
109 
110     val sft = SimpleFeatureTypes.copy(original)
111 
112     // create everything in a single transaction, so that a failure part way through doesn't leave behind a
113     // partially created feature type - in particular, a main view and user data without any partition maintenance
114     // note: the jdbc store creates the initial table before calling this method, outside of our transaction
115     val autoCommit = cx.getAutoCommit
116     if (autoCommit) {
117       cx.setAutoCommit(false)
118     }
119     implicit val ex: ExecutionContext = new ExecutionContext(cx)
120     try {
121       typeLock(schemaName, sft.getTypeName).locked {
122         val info = typeInfo(schemaName, sft, cx)
123         // note: we have to check this before we create anything
124         val isNew = !isView(cx, info.tables.view.name)
125         create(info, isNew)
126       }
127     } catch {
128       case NonFatal(e) =>
129         // note: the execution context only rolls back if one of its statements fails
130         try { cx.rollback() } catch {
131           case NonFatal(suppressed) => e.addSuppressed(suppressed)
132         }
133         throw e
134     } finally {
135       try { ex.close() } finally {
136         if (autoCommit) {
137           cx.setAutoCommit(true)
138         }
139       }
140     }
141   }

220   /**
221    * Runs an operation while holding advisory locks for changes to one or more feature type names, so that different
222    * data store instances don't try to create, update or remove the same feature type concurrently.
223    *
224    * Note that the locks are held by the connection passed in to this method, which must not be used for anything
225    * that commits or rolls back while the operation is running, and that the locks are distinct from the lock used
226    * by `postCreateTable`, so that the operation can create things.
227    *
228    * @param schemaName database schema
229    * @param typeNames feature type names
230    * @param cx connection used to hold the locks
231    * @param op operation to run
232    */
233   def withUpdateLock(schemaName: String, typeNames: Seq[String], cx: Connection)(op: => Unit): Unit = {
234     WithClose(new ExecutionContext(cx)) { implicit ex =>
235       // note: the locks are always acquired in the same order, so that concurrent callers can't deadlock
236       val locks = typeNames.distinct.sorted.map(typeLock(schemaName, _, seed = 1))
237       def locked(remaining: Seq[AdvisoryLock]): Unit =
238         if (remaining.isEmpty) { op } else { remaining.head.locked(locked(remaining.tail)) }
239       locked(locks)
240     }
241   }

243   /**
244    * Drops a feature type (or a generation of a feature type), optionally along with previous generations of it.
245    * Everything is dropped in a single transaction, so if anything fails then all of the generations are left
246    * intact - in particular, we don't leave behind a main view without any partition maintenance.
247    *
248    * The previous generations must be passed in explicitly - they are not dropped when the current generation is
249    * dropped on its own, so that a new generation that failed to be created can be cleaned up without affecting the
250    * existing ones.
251    *
252    * Note that this bypasses the jdbc store, so the caller is responsible for clearing anything that the store
253    * has cached for the views that are dropped.
254    *
255    * @param schemaName database schema
256    * @param sft feature type to drop, as loaded from the jdbc store (i.e. named for its main view)
257    * @param previous names of the main views for any previous generations to drop, most recent first
258    * @param cx connection
259    */
260   def dropSchema(schemaName: String, sft: SimpleFeatureType, previous: Seq[String], cx: Connection): Unit = {
261     val info = TypeInfo(schemaName, sft)
262     val autoCommit = cx.getAutoCommit
263     if (autoCommit) {
264       cx.setAutoCommit(false)
265     }
266     try {
267       try {
268         WithClose(new ExecutionContext(cx, commit = false)) { implicit ex =>
269           // note: load the previous generations up front, before we've dropped anything
270           val (existing, retired) = previous.partition(v => exists(cx, TableIdentifier(schemaName, v)))
271           val generations = existing.map(view => TypeInfo(schemaName, store.getSchema(view)))
272           // note: we use the drop statements directly, without the advisory locks that are used to synchronize creating
273           // the shared tables - we only delete rows from them, and holding the locks until we commit could deadlock
274           // with another feature type being created, which takes them in the opposite order
275           PartitionedPostgisDialect.dropSql(info).foreach(ex.execute)
276           // note: the previous generations have to be dropped after the current view, which depends on their views
277           generations.flatMap(PartitionedPostgisDialect.dropSql(_)).foreach(ex.execute)
278           // clean up any user data left behind for generations that no longer exist
279           retired.foreach(UserDataTable.delete(schemaName, _))
280         }
281         // rename the sft so that configuration is applied to the write ahead table
282         super.postDropTable(schemaName, SimpleFeatureTypes.renameSft(sft, info.tables.writeAhead.name.raw), cx)
283         cx.commit()
284       } catch {
285         case NonFatal(e) =>
286           try { cx.rollback() } catch {
287             case NonFatal(suppressed) => e.addSuppressed(suppressed)
288           }
289           throw e
290       }
291     } finally {
292       if (autoCommit) {
293         cx.setAutoCommit(true)
294       }
295     }
296   }

298   /**
299    * Gets the next unique identifier for a feature type
300    *
301    * @param schemaName database schema
302    * @param typeName feature type name
303    * @param ex execution context
304    * @return
305    */
306   private def nextIdentifier(schemaName: String, typeName: String)(implicit ex: ExecutionContext): String = {
307     // note: 'if not exists' still fails if the sequence is created concurrently, so we synchronize creating it
308     if (!exists(ex.cx, TableIdentifier(schemaName, PartitionedPostgisDialect.SftSeqName))) {
309       PartitionedPostgisDialect.SftSeqLock.locked {
310         ex.execute(
311           s"CREATE SEQUENCE IF NOT EXISTS ${escape(schemaName)}.${PartitionedPostgisDialect.SftSeqName} " +
312             s"AS integer MINVALUE 0 MAXVALUE 65535")
313       }
314     }
315     val sql = s"SELECT nextval(${literal(s"$schemaName.${PartitionedPostgisDialect.SftSeqName}")})"
316     val nextVal = WithClose(ex.cx.prepareStatement(sql)) { st =>
317       WithClose(st.executeQuery()) { rs =>
318         rs.next()
319         rs.getInt(1)
320       }
321     }
322     if (nextVal > 0xFFFF) {
323       throw new IllegalStateException(
324         s"Sequence ${PartitionedPostgisDialect.SftSeqName} has exceeded maximum supported value of 65535 unique feature types")
325     }
326     typeName.take(26) + f"_$nextVal%04x" // 4-character hex-encoded padded string
327   }

329   /**
330    * Gets an advisory lock for a feature type, so that multiple threads don't try to create things concurrently,
331    * which can cause errors
332    *
333    * @param schemaName database schema
334    * @param typeName feature type name
335    * @return
336    */
337   private def typeLock(schemaName: String, typeName: String, seed: Int = 0): AdvisoryLock = {
338     new AdvisoryLock {
339       override protected val lockId: Long = {
340         val bytes = s"$schemaName.$typeName".getBytes(StandardCharsets.UTF_8)
341         MurmurHash3.hash128x64(bytes, 0, bytes.length, seed).head
342       }
343     }
344   }

346   /**
347    * Gets the type info for a feature type, populating any identifiers that are derived from the database
348    *
349    * @param schemaName database schema
350    * @param sft feature type, will be updated with derived identifiers
351    * @param cx connection
352    * @param ex execution context
353    * @return
354    */
355   private def typeInfo(schemaName: String, sft: SimpleFeatureType, cx: Connection)(implicit ex: ExecutionContext): TypeInfo = {
356     // if the sft name is longer than 31 characters, use an alias for delegate tables to avoid character limits
357     // 31 is the max length, based on the current length of our sql identifiers (tables, etc)
358     // note: for an existing type that is being upgraded, we need to re-use the alias that was assigned when it was created
359     if (sft.getTypeName.length() >= 32 && !sft.getUserData.containsKey(SftUserData.IdentAlias.key)) {
360       sft.getUserData.put(SftUserData.IdentAlias.key, nextIdentifier(schemaName, sft.getTypeName))
361     }
362     // get the first column as the fid col, which may or may not be called 'fid'
363     WithClose(cx.getMetaData.getColumns(null, schemaName, sft.getTypeName, null)) { cols =>
364       if (cols.next()) {
365         val name = cols.getString("COLUMN_NAME")
366         if (name != null) {
367           sft.getUserData.put(SftUserData.FidColumn.key, name)
368         }
369       }
370     }
371     val info = TypeInfo(schemaName, sft)
372     // load the previous generation of this type, if there is one and it hasn't been retired yet
373     val previous = SftUserData.PreviousView.get(sft).filter(view => exists(cx, TableIdentifier(schemaName, view))).map { view =>
374       TypeInfo(schemaName, store.getSchema(view))
375     }
376     info.copy(previous = previous)
377   }

379   /**
380    * Creates (or re-creates) all the tables, procedures, etc for a feature type
381    *
382    * @param info type info
383    * @param isNew whether the feature type (or generation) is being created for the first time
384    * @param ex execution context
385    */
386   private def create(info: TypeInfo, isNew: Boolean)(implicit ex: ExecutionContext): Unit = {
387     PartitionedPostgisDialect.Commands.foreach(_.create(info))
388     if (grants.nonEmpty) {
389       val roles = grants.map(_.quoted).mkString(", ")
390       val tables =
391         Seq(
392           info.tables.view.name,
393           info.tables.writeAhead.name,
394           info.tables.writeAheadPartitions.name,
395           info.tables.mainPartitions.name,
396           info.tables.spillPartitions.name,
397           TableIdentifier(info.schema.raw, PrimaryKeyTable.Name.raw),
398           TableIdentifier(info.schema.raw, UserDataTable.Name.raw)
399         )
400       tables.foreach { table =>
401         ex.execute(s"GRANT SELECT ON ${table.qualified} TO $roles;")
402       }
403     }
404     info.previous.foreach { previous =>
405       // carry over any privileges that were granted on the previous view
406       // note: we only do this when the generation is created, so that we don't restore privileges
407       // that are subsequently revoked from the current view
408       if (isNew) {
409         viewPrivileges(previous.tables.view.name, info.tables.view.name).foreach(ex.execute)
410       }
411       // age-off is configured per-generation, so apply the current configuration to all the previous generations
412       // note: the retire procedure for each generation also keeps its previous generation in sync, in case the
413       // configuration is changed directly in the user data table
414       // note: we only write the rows that are out of sync, to avoid taking row locks if nothing has changed
415       val table = s"${info.schema.quoted}.${UserDataTable.Name.quoted}"
416       val generations =
417         s"""WITH RECURSIVE generations(view) AS (
418            |  SELECT ?::text
419            |  UNION
420            |  SELECT u.value FROM $table u JOIN generations g ON u.type_name = g.view WHERE u.key = ?
421            |)""".stripMargin
422       val key = SftUserData.MaxPartitions.key
423       info.partitions.maxPartitions match {
424         case Some(max) =>
425           val value = String.valueOf(max)
426           ex.executeUpdate(
427             s"""$generations
428                |INSERT INTO $table (type_name, key, value, mutable)
429                |  SELECT g.view, ?, ?, true FROM generations g
430                |    WHERE NOT EXISTS (SELECT FROM $table t WHERE t.type_name = g.view AND t.key = ? AND t.value = ?)
431                |  ON CONFLICT (type_name, key) DO UPDATE SET value = EXCLUDED.value;""".stripMargin,
432             Seq(previous.typeName, SftUserData.PreviousView.key, key, value, key, value))
433 
434         case None =>
435           ex.executeUpdate(
436             s"""$generations
437                |DELETE FROM $table WHERE key = ? AND type_name IN (SELECT view FROM generations);""".stripMargin,
438             Seq(previous.typeName, SftUserData.PreviousView.key, key))
439       }
440     }
441   }

443   /**
444    * Gets grant statements that will restore any privileges on the main view, so that they can be re-applied if
445    * the view is dropped and re-created
446    *
447    * @param view existing view
448    * @param target view that the privileges will be applied to
449    * @param ex execution context
450    * @return
451    */
452   private def viewPrivileges(view: TableIdentifier, target: TableIdentifier)(implicit ex: ExecutionContext): Seq[String] = {
453     val sql =
454       s"""SELECT r.rolname, acl.privilege_type, acl.is_grantable
455          |  FROM pg_class c
456          |  CROSS JOIN LATERAL aclexplode(c.relacl) acl
457          |  LEFT JOIN pg_roles r ON r.oid = acl.grantee
458          |  WHERE c.oid = to_regclass(${literal(view.qualified)}) AND acl.grantee <> c.relowner;""".stripMargin
459     WithClose(ex.cx.createStatement()) { st =>
460       WithClose(st.executeQuery(sql)) { rs =>
461         Iterator.continually(rs).takeWhile(_.next()).map { row =>
462           val role = Option(row.getString(1)).map(escape).getOrElse("PUBLIC") // a null role indicates public
463           val option = if (row.getBoolean(3)) { " WITH GRANT OPTION" } else { "" }
464           s"GRANT ${row.getString(2)} ON ${target.qualified} TO $role$option;"
465         }.toList
466       }
467     }
468   }

501           logger.warn(s"Found a list-type attribute but database type was not an array for ${att.getLocalName}")

507   override def postCreateFeatureType(
508       sft: SimpleFeatureType,
509       metadata: DatabaseMetaData,
510       schemaName: String,
511       cx: Connection): Unit = {
512 
513     // normally views get set to read-only, override that here since we use triggers to delegate writes
514     sft.getUserData.remove(JDBCDataStore.JDBC_READ_ONLY)
515 
516     // populate tablespaces (deprecated)
517     PartitionTablespacesTable.read(cx, metadata, schemaName, sft.getTypeName).foreach { case (k, v) => sft.getUserData.put(k, v) }
518 
519     // populate user data
520     UserDataTable.read(cx, schemaName, sft.getTypeName).foreach { case (k, v) => sft.getUserData.put(k, v) }
521 
522     // the default geometry wasn't always recorded in the user data, and the jdbc store always uses the first geometry
523     if (!sft.getUserData.containsKey(SftUserData.GeomField.key)) {
524       legacyDefaultGeometry(sft, schemaName, cx).foreach(sft.getUserData.put(SftUserData.GeomField.key, _))
525     }
526 
527     // populate flags on indexed attributes
528     getIndexedColumns(cx, TypeInfo(schemaName, sft)) match {
529       case Success(cols) =>
530         cols.foreach { col =>
531           val i = sft.indexOf(col)
532           if (i == -1) {
533             logger.debug(
534               s"Found unexpected indexed column not in feature type: $col for ${sft.getTypeName}=${SimpleFeatureTypes.encodeType(sft)}")
535           } else {
536             sft.getDescriptor(i).getUserData.put(AttributeOptions.OptIndex, "true")
537           }
538         }
539 
540       case Failure(e) => logger.warn(s"Error loading indexed columns for feature type ${sft.getTypeName}:", e)
541     }
542   }

544   /**
545    * Gets the default geometry for a feature type that was created before the default geometry was recorded in the
546    * user data. The default geometry is used to sort the data in each partition, so we determine it based on the
547    * column that the existing procedures are sorting by.
548    *
549    * @param sft feature type, with user data populated
550    * @param schemaName database schema
551    * @param cx connection
552    * @return the default geometry, if it could be determined and there is more than one candidate
553    */
554   private def legacyDefaultGeometry(sft: SimpleFeatureType, schemaName: String, cx: Connection): Option[String] = {
555     val geoms = sft.getAttributeDescriptors.asScala.collect { case d: GeometryDescriptor => d.getLocalName }
