116   override def postCreateTable(schemaName: String, original: SimpleFeatureType, cx: Connection): Unit = {
117 
118     // note: we skip the call to `super`, which creates a spatial index (that we don't want), and which
119     // alters the geometry column types (which we handle in the create statement)
120 
121     val sft = SimpleFeatureTypes.copy(original)
122 
123     implicit val ex: ExecutionContext = new ExecutionContext(cx)
124     try {
125       // if the sft name is longer than 31 characters, use an alias for delegate tables to avoid character limits
126       // 31 is the max length, based on the current length of our sql identifiers (tables, etc)
127       if (sft.getTypeName.length() >= 32) {
128         ex.execute(
129           s"CREATE SEQUENCE IF NOT EXISTS ${escape(schemaName)}.${PartitionedPostgisDialect.SftSeqName} " +
130             s"AS integer MINVALUE 0 MAXVALUE 65535")
131         val sql = s"SELECT nextval(${literal(s"$schemaName.${PartitionedPostgisDialect.SftSeqName}")})"
132         val nextVal = WithClose(cx.prepareStatement(sql)) { st =>
133           WithClose(st.executeQuery()) { rs =>
134             rs.next()
135             rs.getInt(1)
136           }
137         }
138         if (nextVal > 0xFFFF) {
139           throw new IllegalStateException(
140             s"Sequence ${PartitionedPostgisDialect.SftSeqName} has exceeded maximum supported value of 65535 unique feature types")
141         }
142         val id = sft.getTypeName.substring(0, 26) + f"_$nextVal%04x" // 4-character hex-encoded padded string
143         sft.getUserData.put(SftUserData.IdentAlias.key, id)
144       }
145       // get the first column as the fid col, which may or may not be called 'fid'
146       WithClose(cx.getMetaData.getColumns(null, schemaName, sft.getTypeName, null)) { cols =>
147         if (cols.next()) {
148           val name = cols.getString("COLUMN_NAME")
149           if (name != null) {
150             sft.getUserData.put(SftUserData.FidColumn.key, name)
151           }
152         }
153       }
154       val info = TypeInfo(schemaName, sft)
155       PartitionedPostgisDialect.Commands.foreach(_.create(info))
156       if (grants.nonEmpty) {
157         val roles = grants.map(_.quoted).mkString(", ")
158         val tables =
159           Seq(
160             info.tables.view.name,
161             info.tables.writeAhead.name,
162             info.tables.writeAheadPartitions.name,
163             info.tables.mainPartitions.name,
164             info.tables.spillPartitions.name,
165             TableIdentifier(schemaName, PrimaryKeyTable.Name.raw),
166             TableIdentifier(schemaName, UserDataTable.Name.raw)
167           )
168         tables.foreach { table =>
169           ex.execute(s"GRANT SELECT ON ${table.qualified} TO $roles;")
170         }
171       }
172     } finally {
173       ex.close()
174     }
175   }

177   override def postCreateAttribute(
178       att: AttributeDescriptor,
179       tableName: String,
180       schemaName: String,
181       cx: Connection): Unit = {
182 
183     def withCol(fn: ResultSet => Unit): Unit = {
184       val meta = cx.getMetaData
185       def escape(name: String): String = store.escapeNamePattern(meta, name)
186       WithClose(meta.getColumns(cx.getCatalog, escape(schemaName), escape(tableName), escape(att.getLocalName))) { cols =>
187         if (cols.next()) {
188           fn(cols)
189         } else {
190           logger.warn(s"Could not retrieve column metadata for attribute ${att.getLocalName}")
191         }
192       }
193     }
194 
195     if (classOf[String].isAssignableFrom(att.getType.getBinding)) {
196       withCol { cols =>
197         val typeName = cols.getString("TYPE_NAME")
198         if ("json".equalsIgnoreCase(typeName) || "jsonb".equalsIgnoreCase(typeName)) {
199           att.getUserData.put(SimpleFeatureTypes.AttributeOptions.OptJson, "true")

201       }
202     } else if (classOf[java.util.List[_]].isAssignableFrom(att.getType.getBinding)) {
203       withCol { cols =>
204         val arrayType = super.getMapping(cols, cx)
205         if (arrayType.isArray) {
206           att.getUserData.put(SimpleFeatureTypes.AttributeConfigs.UserDataListType, arrayType.getComponentType.getName)
207         } else {
208           logger.warn(s"Found a list-type attribute but database type was not an array for ${att.getLocalName}")
209         }
210       }

219 
220     // normally views get set to read-only, override that here since we use triggers to delegate writes

240         }
241 
242       case Failure(e) => logger.warn(s"Error loading indexed columns for feature type ${sft.getTypeName}:", e)
243     }

246   override def preDropTable(schemaName: String, sft: SimpleFeatureType, cx: Connection): Unit = {
247     // due to the JDBCDataStore hard-coding "DROP TABLE" we have to redirect it away from the main view,
248     // and we can't drop the write ahead table so that it has something to drop
249     val info = TypeInfo(schemaName, sft)
250     dropping.set(info)
251 
252     implicit val ex: ExecutionContext = new ExecutionContext(cx)
253     try {
254       PartitionedPostgisDialect.Commands.reverse.filter(_ != WriteAheadTable).foreach(_.drop(info))
255       PartitionTablespacesTable.drop(info)
256     } finally {
257       ex.close()
258     }
259   }

297   override def encodePostColumnCreateTable(att: AttributeDescriptor, sql: StringBuffer): Unit = {
298     import PartitionedPostgisDialect.GeometryAttributeConversions
299     att match {
300       case gd: GeometryDescriptor =>
301         val nullable = gd.getMinOccurs <= 0 || gd.isNillable
302         val i = sql.lastIndexOf("geometry")
303         // expect `geometry NOT NULL` or `geometry` depending on nullable flag
304         if (i == -1 || (nullable && i != sql.length() - 8) || (!nullable && i != sql.length() - 17)) {
305           logger.warn(s"Found geometry-type attribute but no geometry column binding: $sql")
306         } else {
307           val srid = gd.getSrid.getOrElse(-1)
308           val geomType = PartitionedPostgisDialect.GeometryMappings.getOrElse(gd.getType.getBinding, "GEOMETRY")
309           val geomTypeWithDims = gd.getCoordinateDimensions match {
310             case None | Some(2) => geomType
311             case Some(3) => s"${geomType}Z"
312             case Some(4) => s"${geomType}ZM"
313             case Some(d) =>
314               throw new IllegalArgumentException(
315                 s"PostGIS only supports geometries with 2, 3 and 4 dimensions, but found: $d")
316           }
317           sql.insert(i + 8, s" ($geomTypeWithDims, $srid)")
318         }
319 
320       case _ if att.isJson() =>
321         // replace 'VARCHAR' with jsonb
322         val i = sql.lastIndexOf(" VARCHAR")
323         if (i == sql.length() - 8) {
324           sql.replace(i + 1, i + 8, "JSONB")
325         } else {
326           logger.warn(s"Found JSON-type attribute but no CHARACTER VARYING column binding: $sql")
327         }
328 
329       case _ if att.isList =>
330         // go back and encode the array type in the CQL create statement
331         val i = sql.lastIndexOf(" ARRAY")
332         if (i == sql.length() - 6) {
333           sql.insert(i, " " + getListTypeMapping(att.getListType()))
334         } else {
335           logger.warn(s"Found list-type attribute but no ARRAY column binding: $sql")
336         }
337 
338       case _ => // no-op
339     }
340   }

352    * Gets the array type for a list/array column
353    *
354    * @param binding list-type binding
355    * @return
356    */
357   private def getListTypeMapping(binding: Class[_]): String = {
358     val mappings = new java.util.HashMap[String, Class[_]]()
359     registerSqlTypeNameToClassMappings(mappings)
360     var mapping: String = null
361     var partial: String = null
362     val iter = mappings.asScala.iterator
363     while (iter.hasNext && mapping == null) {
364       val (name, clas) = iter.next
365       if (clas == binding) {
366         mapping = name
367       } else if (partial == null && clas.isAssignableFrom(binding)) {
368         partial = name
369       }
370     }
371     if (mapping != null) {
372       mapping
373     } else if (partial != null) {
374       partial
375     } else {
376       logger.warn(s"Could not find list-type column for type ${binding.getName}")
377       "text"
378     }
379   }
