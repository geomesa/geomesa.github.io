129   override def postCreateTable(schemaName: String, sft: SimpleFeatureType, cx: Connection): Unit = {
130 
131     // note: we skip the call to `super`, which creates a spatial index (that we don't want), and which
132     // alters the geometry column types (which we handle in the create statement)
133 
134     val sftWithUserData = SimpleFeatureTypes.copy(sft)
135 
136     implicit val ex: ExecutionContext = new ExecutionContext(cx)
137     try {
138       // if the sft name is longer than 31 characters, use an alias for delegate tables to avoid character limits
139       // 31 is the max length, based on the current length of our sql identifiers (tables, etc)
140       if (sft.getTypeName.length() >= 32) {
141         ex.execute(
142           s"CREATE SEQUENCE IF NOT EXISTS ${escape(schemaName)}.${PartitionedPostgisDialect.SftSeqName} " +
143             s"AS integer MINVALUE 0 MAXVALUE 65535")
144         val sql = s"SELECT nextval(${literal(s"$schemaName.${PartitionedPostgisDialect.SftSeqName}")})"
145         val nextVal = WithClose(cx.prepareStatement(sql)) { st =>
146           WithClose(st.executeQuery()) { rs =>
147             rs.next()
148             rs.getInt(1)
149           }
150         }
151         if (nextVal > 0xFFFF) {
152           throw new IllegalStateException(
153             s"Sequence ${PartitionedPostgisDialect.SftSeqName} has exceeded maximum supported value of 65535 unique feature types")
154         }
155         val id = sft.getTypeName.substring(0, 26) + f"_$nextVal%04x" // 4-character hex-encoded padded string
156         sftWithUserData.getUserData.put(SftUserData.IdentAlias.key, id)
157       }
158       // get the first column as the fid col, which may or may not be called 'fid'
159       Option(creating.get()).foreach { tableName =>
160         WithClose(cx.getMetaData.getColumns(null, schemaName, tableName, null)) { cols =>
161           if (cols.next()) {
162             val name = cols.getString("COLUMN_NAME")
163             if (name != null) {
164               sftWithUserData.getUserData.put(SftUserData.FidColumn.key, name)
165             }
166           }
167         }
168       }
169       val info = TypeInfo(schemaName, sftWithUserData)
170       PartitionedPostgisDialect.Commands.foreach(_.create(info))
171       if (grants.nonEmpty) {
172         val roles = grants.map(_.quoted).mkString(", ")
173         val tables =
174           Seq(
175             info.tables.view.name,
176             info.tables.writeAhead.name,
177             info.tables.writeAheadPartitions.name,
178             info.tables.mainPartitions.name,
179             info.tables.spillPartitions.name,
180             TableIdentifier(schemaName, PrimaryKeyTable.Name.raw),
181             TableIdentifier(schemaName, UserDataTable.Name.raw)
182           )
183         tables.foreach { table =>
184           ex.execute(s"GRANT SELECT ON ${table.qualified} TO $roles;")
185         }
186       }
187     } finally {
188       creating.remove()
189       ex.close()
190     }
191   }

262   override def preDropTable(schemaName: String, sft: SimpleFeatureType, cx: Connection): Unit = {
263     // due to the JDBCDataStore hard-coding "DROP TABLE" we have to redirect it away from the main view,
264     // and we can't drop the write ahead table so that it has something to drop
265     val info = TypeInfo(schemaName, sft)
266     dropping.set(info)
267 
268     implicit val ex: ExecutionContext = new ExecutionContext(cx)
269     try {
270       PartitionedPostgisDialect.Commands.reverse.filter(_ != WriteAheadTable).foreach(_.drop(info))
271       PartitionTablespacesTable.drop(info)
272     } finally {
273       ex.close()
274     }

313   override def encodePostColumnCreateTable(att: AttributeDescriptor, sql: StringBuffer): Unit = {
314     import PartitionedPostgisDialect.GeometryAttributeConversions
315     att match {
316       case gd: GeometryDescriptor =>
317         val nullable = gd.getMinOccurs <= 0 || gd.isNillable
318         val i = sql.lastIndexOf("geometry")
319         // expect `geometry NOT NULL` or `geometry` depending on nullable flag
320         if (i == -1 || (nullable && i != sql.length() - 8) || (!nullable && i != sql.length() - 17)) {
321           logger.warn(s"Found geometry-type attribute but no geometry column binding: $sql")
322         } else {
323           val srid = gd.getSrid.getOrElse(-1)
324           val geomType = PartitionedPostgisDialect.GeometryMappings.getOrElse(gd.getType.getBinding, "GEOMETRY")
325           val geomTypeWithDims = gd.getCoordinateDimensions match {
326             case None | Some(2) => geomType
327             case Some(3) => s"${geomType}Z"
328             case Some(4) => s"${geomType}ZM"
329             case Some(d) =>
330               throw new IllegalArgumentException(
331                 s"PostGIS only supports geometries with 2, 3 and 4 dimensions, but found: $d")
332           }
333           sql.insert(i + 8, s" ($geomTypeWithDims, $srid)")
334         }
335 
336       case _ if att.isJson() =>
337         // replace 'VARCHAR' with jsonb
338         val i = sql.lastIndexOf(" VARCHAR")
339         if (i == sql.length() - 8) {
340           sql.replace(i + 1, i + 8, "JSONB")
341         } else {
342           logger.warn(s"Found JSON-type attribute but no CHARACTER VARYING column binding: $sql")
343         }
344 
345       case _ if att.isList =>
346         // go back and encode the array type in the CQL create statement
347         val i = sql.lastIndexOf(" ARRAY")
348         if (i == sql.length() - 6) {
349           sql.insert(i, " " + getListTypeMapping(att.getListType()))
350         } else {
351           logger.warn(s"Found list-type attribute but no ARRAY column binding: $sql")
352         }
353 
354       case _ => // no-op
355     }
356   }

367   /**
368    * Gets the array type for a list/array column
369    *
370    * @param binding list-type binding
371    * @return
372    */
373   private def getListTypeMapping(binding: Class[_]): String = {
374     val mappings = new java.util.HashMap[String, Class[_]]()
375     registerSqlTypeNameToClassMappings(mappings)
376     var mapping: String = null
377     var partial: String = null
378     val iter = mappings.asScala.iterator
379     while (iter.hasNext && mapping == null) {
380       val (name, clas) = iter.next
381       if (clas == binding) {
382         mapping = name
383       } else if (partial == null && clas.isAssignableFrom(binding)) {
384         partial = name
385       }
386     }
387     if (mapping != null) {
388       mapping
389     } else if (partial != null) {
390       partial
391     } else {
392       logger.warn(s"Could not find list-type column for type ${binding.getName}")
393       "text"
394     }
395   }

514     val sql =

516          |from pg_class obj
517          |join pg_index idx on idx.indrelid = obj.oid
518          |join pg_attribute att on att.attrelid = obj.oid and att.attnum = any(idx.indkey)
519          |where obj.relname = ${literal(info.tables.mainPartitions.name.raw)}
