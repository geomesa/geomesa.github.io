128   override def postCreateTable(schemaName: String, sft: SimpleFeatureType, cx: Connection): Unit = {
129 
130     // note: we skip the call to `super`, which creates a spatial index (that we don't want), and which
131     // alters the geometry column types (which we handle in the create statement)
132 
133     implicit val ex: ExecutionContext = new ExecutionContext(cx)
134     try {
135       // if the sft name is longer than 31 characters, use an alias for delegate tables to avoid character limits
136       // 31 is the max length, based on the current length of our sql identifiers (tables, etc)
137       val typeId = if (sft.getTypeName.length() < 32) { None } else {
138         ex.execute(
139           s"CREATE SEQUENCE IF NOT EXISTS ${escape(schemaName)}.${PartitionedPostgisDialect.SftSeqName} " +
140             s"AS integer MINVALUE 0 MAXVALUE 65535")
141         val sql = s"SELECT nextval(${literal(s"$schemaName.${PartitionedPostgisDialect.SftSeqName}")})"
142         val nextVal = WithClose(cx.prepareStatement(sql)) { st =>
143           WithClose(st.executeQuery()) { rs =>
144             rs.next()
145             rs.getInt(1)
146           }
147         }
148         if (nextVal > 0xFFFF) {
149           throw new IllegalStateException(
150             s"Sequence ${PartitionedPostgisDialect.SftSeqName} has exceeded maximum supported value of 65535 unique feature types")
151         }
152         Some(sft.getTypeName.substring(0, 26) + f"_$nextVal%04x") // 4-character hex-encoded padded string
153       }
154 
155       val info = TypeInfo(schemaName, typeId.fold(sft) { id =>
156         val copy = SimpleFeatureTypes.copy(sft)
157         copy.getUserData.put(SftUserData.IdentAlias.key, id)
158         copy
159       })
160       PartitionedPostgisDialect.Commands.foreach(_.create(info))
161       if (grants.nonEmpty) {
162         val roles = grants.map(_.quoted).mkString(", ")
163         val tables =
164           Seq(
165             info.tables.view.name,
166             info.tables.writeAhead.name,
167             info.tables.writeAheadPartitions.name,
168             info.tables.mainPartitions.name,
169             info.tables.spillPartitions.name,
170             TableIdentifier(schemaName, PrimaryKeyTable.Name.raw),
171             TableIdentifier(schemaName, UserDataTable.Name.raw)
172           )
173         tables.foreach { table =>
174           ex.execute(s"GRANT SELECT ON ${table.qualified} TO $roles;")
175         }
176       }
177     } finally {
178       ex.close()
179     }
180   }

253   override def preDropTable(schemaName: String, sft: SimpleFeatureType, cx: Connection): Unit = {
254     // due to the JDBCDataStore hard-coding "DROP TABLE" we have to redirect it away from the main view,
255     // and we can't drop the write ahead table so that it has something to drop
256     dropping.set(true)
257     val info = TypeInfo(schemaName, sft)
258 
259     implicit val ex: ExecutionContext = new ExecutionContext(cx)
260     try {
261       PartitionedPostgisDialect.Commands.reverse.filter(_ != WriteAheadTable).foreach(_.drop(info))
262       PartitionTablespacesTable.drop(info)
263     } finally {
264       ex.close()
265     }

304   override def encodePostColumnCreateTable(att: AttributeDescriptor, sql: StringBuffer): Unit = {
305     import PartitionedPostgisDialect.GeometryAttributeConversions
306     att match {
307       case gd: GeometryDescriptor =>
308         val nullable = gd.getMinOccurs <= 0 || gd.isNillable
309         val i = sql.lastIndexOf("geometry")
310         // expect `geometry NOT NULL` or `geometry` depending on nullable flag
311         if (i == -1 || (nullable && i != sql.length() - 8) || (!nullable && i != sql.length() - 17)) {
312           logger.warn(s"Found geometry-type attribute but no geometry column binding: $sql")
313         } else {
314           val srid = gd.getSrid.getOrElse(-1)
315           val geomType = PartitionedPostgisDialect.GeometryMappings.getOrElse(gd.getType.getBinding, "GEOMETRY")
316           val geomTypeWithDims = gd.getCoordinateDimensions match {
317             case None | Some(2) => geomType
318             case Some(3) => s"${geomType}Z"
319             case Some(4) => s"${geomType}ZM"
320             case Some(d) =>
321               throw new IllegalArgumentException(
322                 s"PostGIS only supports geometries with 2, 3 and 4 dimensions, but found: $d")
323           }
324           sql.insert(i + 8, s" ($geomTypeWithDims, $srid)")
325         }
326 
327       case _ if att.isJson() =>
328         // replace 'VARCHAR' with jsonb
329         val i = sql.lastIndexOf(" VARCHAR")
330         if (i == sql.length() - 8) {
331           sql.replace(i + 1, i + 8, "JSONB")
332         } else {
333           logger.warn(s"Found JSON-type attribute but no CHARACTER VARYING column binding: $sql")
334         }
335 
336       case _ if att.isList =>
337         // go back and encode the array type in the CQL create statement
338         val i = sql.lastIndexOf(" ARRAY")
339         if (i == sql.length() - 6) {
340           sql.insert(i, " " + getListTypeMapping(att.getListType()))
341         } else {
342           logger.warn(s"Found list-type attribute but no ARRAY column binding: $sql")
343         }
344 
345       case _ => // no-op
346     }
347   }

358   /**
359    * Gets the array type for a list/array column
360    *
361    * @param binding list-type binding
362    * @return
363    */
364   private def getListTypeMapping(binding: Class[_]): String = {
365     val mappings = new java.util.HashMap[String, Class[_]]()
366     registerSqlTypeNameToClassMappings(mappings)
367     var mapping: String = null
368     var partial: String = null
369     val iter = mappings.asScala.iterator
370     while (iter.hasNext && mapping == null) {
371       val (name, clas) = iter.next
372       if (clas == binding) {
373         mapping = name
374       } else if (partial == null && clas.isAssignableFrom(binding)) {
375         partial = name
376       }
377     }
378     if (mapping != null) {
379       mapping
380     } else if (partial != null) {
381       partial
382     } else {
383       logger.warn(s"Could not find list-type column for type ${binding.getName}")
384       "text"
385     }
386   }

505          |from pg_class obj

507          |join pg_attribute att on att.attrelid = obj.oid and att.attnum = any(idx.indkey)
