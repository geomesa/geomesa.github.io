95     val typeInfo = dropping.get

118   override def postCreateTable(schemaName: String, original: SimpleFeatureType, cx: Connection): Unit = {
119 
120     // note: we skip the call to `super`, which creates a spatial index (that we don't want), and which
121     // alters the geometry column types (which we handle in the create statement)
122 
123     val sft = SimpleFeatureTypes.copy(original)
124 
125     // get an advisory lock so that multiple threads don't try to create things concurrently, which can cause errors
126     val lock = new AdvisoryLock {
127       override protected val lockId: Long = {
128         val bytes = s"$schemaName.${sft.getTypeName}".getBytes(StandardCharsets.UTF_8)
129         MurmurHash3.hash128x64(bytes, 0, bytes.length, 0).head
130       }
131     }
132 
133     implicit val ex: ExecutionContext = new ExecutionContext(cx)
134     try {
135       lock.locked {
136         // if the sft name is longer than 31 characters, use an alias for delegate tables to avoid character limits
137         // 31 is the max length, based on the current length of our sql identifiers (tables, etc)
138         if (sft.getTypeName.length() >= 32) {
139           ex.execute(
140             s"CREATE SEQUENCE IF NOT EXISTS ${escape(schemaName)}.${PartitionedPostgisDialect.SftSeqName} " +
141               s"AS integer MINVALUE 0 MAXVALUE 65535")
142           val sql = s"SELECT nextval(${literal(s"$schemaName.${PartitionedPostgisDialect.SftSeqName}")})"
143           val nextVal = WithClose(cx.prepareStatement(sql)) { st =>
144             WithClose(st.executeQuery()) { rs =>
145               rs.next()
146               rs.getInt(1)
147             }
148           }
149           if (nextVal > 0xFFFF) {
150             throw new IllegalStateException(
151               s"Sequence ${PartitionedPostgisDialect.SftSeqName} has exceeded maximum supported value of 65535 unique feature types")
152           }
153           val id = sft.getTypeName.substring(0, 26) + f"_$nextVal%04x" // 4-character hex-encoded padded string
154           sft.getUserData.put(SftUserData.IdentAlias.key, id)
155         }
156         // get the first column as the fid col, which may or may not be called 'fid'
157         WithClose(cx.getMetaData.getColumns(null, schemaName, sft.getTypeName, null)) { cols =>
158           if (cols.next()) {
159             val name = cols.getString("COLUMN_NAME")
160             if (name != null) {
161               sft.getUserData.put(SftUserData.FidColumn.key, name)
162             }
163           }
164         }
165         val info = TypeInfo(schemaName, sft)
166         PartitionedPostgisDialect.Commands.foreach(_.create(info))
167         if (grants.nonEmpty) {
168           val roles = grants.map(_.quoted).mkString(", ")
169           val tables =
170             Seq(
171               info.tables.view.name,
172               info.tables.writeAhead.name,
173               info.tables.writeAheadPartitions.name,
174               info.tables.mainPartitions.name,
175               info.tables.spillPartitions.name,
176               TableIdentifier(schemaName, PrimaryKeyTable.Name.raw),
177               TableIdentifier(schemaName, UserDataTable.Name.raw)
178             )
179           tables.foreach { table =>
180             ex.execute(s"GRANT SELECT ON ${table.qualified} TO $roles;")
181           }
182         }
183       }
184     } finally {
185       ex.close()
186     }
187   }

189   override def postCreateAttribute(
190       att: AttributeDescriptor,
191       tableName: String,
192       schemaName: String,
193       cx: Connection): Unit = {
194 
195     def withCol(fn: ResultSet => Unit): Unit = {
196       val meta = cx.getMetaData
197       def escape(name: String): String = store.escapeNamePattern(meta, name)
198       WithClose(meta.getColumns(cx.getCatalog, escape(schemaName), escape(tableName), escape(att.getLocalName))) { cols =>
199         if (cols.next()) {
200           fn(cols)
201         } else {
202           logger.warn(s"Could not retrieve column metadata for attribute ${att.getLocalName}")
203         }
204       }
205     }
206 
207     if (classOf[String].isAssignableFrom(att.getType.getBinding)) {
208       withCol { cols =>
209         val typeName = cols.getString("TYPE_NAME")
210         if ("json".equalsIgnoreCase(typeName) || "jsonb".equalsIgnoreCase(typeName)) {
211           att.getUserData.put(SimpleFeatureTypes.AttributeOptions.OptJson, "true")
212         }
213       }
214     } else if (classOf[java.util.List[_]].isAssignableFrom(att.getType.getBinding)) {
215       withCol { cols =>
216         val arrayType = super.getMapping(cols, cx)
217         if (arrayType.isArray) {
218           att.getUserData.put(SimpleFeatureTypes.AttributeConfigs.UserDataListType, arrayType.getComponentType.getName)
219         } else {
220           logger.warn(s"Found a list-type attribute but database type was not an array for ${att.getLocalName}")
221         }
222       }
223     }
224   }

226   override def postCreateFeatureType(
227       sft: SimpleFeatureType,
228       metadata: DatabaseMetaData,
229       schemaName: String,
230       cx: Connection): Unit = {
231 
232     // normally views get set to read-only, override that here since we use triggers to delegate writes
233     sft.getUserData.remove(JDBCDataStore.JDBC_READ_ONLY)
234 
235     // populate tablespaces (deprecated)
236     PartitionTablespacesTable.read(cx, metadata, schemaName, sft.getTypeName).foreach { case (k, v) => sft.getUserData.put(k, v) }
237 
238     // populate user data
239     UserDataTable.read(cx, schemaName, sft.getTypeName).foreach { case (k, v) => sft.getUserData.put(k, v) }
240 
241     // populate flags on indexed attributes
242     getIndexedColumns(cx, TypeInfo(schemaName, sft)) match {
243       case Success(cols) =>
244         cols.foreach { col =>
245           val i = sft.indexOf(col)
246           if (i == -1) {
247             logger.debug(
248               s"Found unexpected indexed column not in feature type: $col for ${sft.getTypeName}=${SimpleFeatureTypes.encodeType(sft)}")
249           } else {
250             sft.getDescriptor(i).getUserData.put(AttributeOptions.OptIndex, "true")
251           }
252         }
253 
254       case Failure(e) => logger.warn(s"Error loading indexed columns for feature type ${sft.getTypeName}:", e)
255     }
256   }

258   override def preDropTable(schemaName: String, sft: SimpleFeatureType, cx: Connection): Unit = {
259     // due to the JDBCDataStore hard-coding "DROP TABLE" we have to redirect it away from the main view,
260     // and we can't drop the write ahead table so that it has something to drop
261     val info = TypeInfo(schemaName, sft)
262     dropping.set(info)
263 
264     implicit val ex: ExecutionContext = new ExecutionContext(cx)
265     try {
266       PartitionedPostgisDialect.Commands.reverse.filter(_ != WriteAheadTable).foreach(_.drop(info))
267       PartitionTablespacesTable.drop(info)
268     } finally {
269       ex.close()
270     }
271   }

273   override def postDropTable(schemaName: String, sft: SimpleFeatureType, cx: Connection): Unit = {
274     val info = TypeInfo(schemaName, sft)
275 
276     implicit val ex: ExecutionContext = new ExecutionContext(cx)
277     try {
278       WriteAheadTable.drop(info) // drop the write ahead name sequence

284     super.postDropTable(schemaName, SimpleFeatureTypes.renameSft(sft, info.tables.writeAhead.name.raw), cx)

309   override def encodePostColumnCreateTable(att: AttributeDescriptor, sql: StringBuffer): Unit = {
310     import PartitionedPostgisDialect.GeometryAttributeConversions
311     att match {
312       case gd: GeometryDescriptor =>
313         val nullable = gd.getMinOccurs <= 0 || gd.isNillable
314         val i = sql.lastIndexOf("geometry")
315         // expect `geometry NOT NULL` or `geometry` depending on nullable flag
316         if (i == -1 || (nullable && i != sql.length() - 8) || (!nullable && i != sql.length() - 17)) {
317           logger.warn(s"Found geometry-type attribute but no geometry column binding: $sql")
318         } else {
319           val srid = gd.getSrid.getOrElse(-1)
320           val geomType = PartitionedPostgisDialect.GeometryMappings.getOrElse(gd.getType.getBinding, "GEOMETRY")
321           val geomTypeWithDims = gd.getCoordinateDimensions match {
322             case None | Some(2) => geomType
323             case Some(3) => s"${geomType}Z"
324             case Some(4) => s"${geomType}ZM"
325             case Some(d) =>
326               throw new IllegalArgumentException(
327                 s"PostGIS only supports geometries with 2, 3 and 4 dimensions, but found: $d")
328           }
329           sql.insert(i + 8, s" ($geomTypeWithDims, $srid)")
330         }
331 
332       case _ if att.isJson() =>
333         // replace 'VARCHAR' with jsonb
334         val i = sql.lastIndexOf(" VARCHAR")
335         if (i == sql.length() - 8) {
336           sql.replace(i + 1, i + 8, "JSONB")
337         } else {
338           logger.warn(s"Found JSON-type attribute but no CHARACTER VARYING column binding: $sql")
339         }
340 
341       case _ if att.isList =>
342         // go back and encode the array type in the CQL create statement
343         val i = sql.lastIndexOf(" ARRAY")
344         if (i == sql.length() - 6) {
345           sql.insert(i, " " + getListTypeMapping(att.getListType()))
346         } else {
347           logger.warn(s"Found list-type attribute but no ARRAY column binding: $sql")
348         }
349 
350       case _ => // no-op
351     }
352   }

363   /**
364    * Gets the array type for a list/array column
365    *
366    * @param binding list-type binding
367    * @return
368    */
369   private def getListTypeMapping(binding: Class[_]): String = {
370     val mappings = new java.util.HashMap[String, Class[_]]()
371     registerSqlTypeNameToClassMappings(mappings)
372     var mapping: String = null
373     var partial: String = null
374     val iter = mappings.asScala.iterator
375     while (iter.hasNext && mapping == null) {
376       val (name, clas) = iter.next
377       if (clas == binding) {
378         mapping = name
379       } else if (partial == null && clas.isAssignableFrom(binding)) {
380         partial = name
381       }
382     }
383     if (mapping != null) {
384       mapping
385     } else if (partial != null) {
386       partial
387     } else {
388       logger.warn(s"Could not find list-type column for type ${binding.getName}")
389       "text"
390     }
391   }

521   /**
522    * Get a list of indexed columns for the given SimpleFeatureType
523    *
524    * @param cx connection
525    * @param info type info
526    * @return a sequence of SimpleFeatureType attribute names which have an index
527    */
528   private def getIndexedColumns(cx: Connection, info: TypeInfo): Try[List[String]] = {
529     // only key columns: indkey also lists the INCLUDE columns of a covering index, after the first indnkeyatts
530     val sql =
531       s"""select distinct(att.attname) as indexed_attribute_name
532          |from pg_class obj
533          |join pg_index idx on idx.indrelid = obj.oid
534          |join pg_attribute att on att.attrelid = obj.oid
535          |  and exists (select from generate_series(0, idx.indnkeyatts - 1) k where idx.indkey[k] = att.attnum)
536          |where obj.relname = ${literal(info.tables.mainPartitions.name.raw)}
537          |order by att.attname;""".stripMargin
538     Try {
539       WithClose(cx.createStatement()) { st =>
