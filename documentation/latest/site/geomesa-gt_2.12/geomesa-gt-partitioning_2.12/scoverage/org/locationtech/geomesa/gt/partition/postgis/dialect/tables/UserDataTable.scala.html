55       ex.executeUpdate(insertSql, Seq(info.typeName, config.key, value, config.mutable))

68     info.tables.writeAheadPartitions.tablespace.foreach(ts => insert(SftUserData.WriteAheadPartitionsTableSpace, ts.raw))

90    * @param cx connection
91    * @param schema database schema
92    * @param typeName feature type name
93    * @return
94    */
95   def read(cx: Connection, schema: String, typeName: String): Map[String, String] = {
96     val map = Map.newBuilder[String, String]
97     val userDataSql = s"select key, value from ${escape(schema)}.${UserDataTable.Name.quoted} where type_name = ?"
98     WithClose(cx.prepareStatement(userDataSql)) { statement =>
99       statement.setString(1, typeName)
100       WithClose(statement.executeQuery()) { rs =>
101         while (rs.next()) {
102           map += rs.getString(1) -> rs.getString(2)
103         }
104       }
105     }
106     map.result()
107   }
