28   override def create(info: TypeInfo)(implicit ex: ExecutionContext): Unit = {

33     val create =
34       s"""CREATE TABLE IF NOT EXISTS ${table.quoted} (

38          |  mutable boolean default false
39          |);""".stripMargin

42          |BEGIN
43          |  IF NOT EXISTS (SELECT FROM pg_constraint WHERE conname = ${cName.asLiteral} AND conrelid = ${table.asRegclass}) THEN
44          |    ALTER TABLE ${table.quoted} ADD CONSTRAINT ${cName.quoted} PRIMARY KEY (type_name, key);
45          |  END IF;
46          |END$$$$;""".stripMargin

48     Seq(alter, create, constraint).foreach(ex.execute)

52           s"ON CONFLICT (type_name, key) DO UPDATE SET value = EXCLUDED.value;"

54     def insert(config: SftUserData[_], value: String): Unit =
55       ex.executeUpdate(insertSql, Seq(info.typeName, config.key, value, config.mutable))

57     if (info.typeIdentifier != info.typeName) {

59     }

67     info.partitions.cronMinute.foreach(v => insert(SftUserData.CronMinute, String.valueOf(v)))

69     info.tables.writeAheadPartitions.tablespace.foreach(ts => insert(SftUserData.WriteAheadPartitionsTableSpace, ts.raw))
70     info.tables.mainPartitions.tablespace.foreach(ts => insert(SftUserData.MainTableSpace, ts.raw))

74     }

76       info.userData.get(config.key).foreach(v => insert(config, v))
77     }

85     }
86   }

91    * @param cx connection
92    * @param schema database schema
93    * @param typeName feature type name
94    * @return
95    */
96   def read(cx: Connection, schema: String, typeName: String): Map[String, String] = {
97     val map = Map.newBuilder[String, String]
98     val userDataSql = s"select key, value from ${escape(schema)}.${UserDataTable.Name.quoted} where type_name = ?"
99     WithClose(cx.prepareStatement(userDataSql)) { statement =>
100       statement.setString(1, typeName)
101       WithClose(statement.executeQuery()) { rs =>
102         while (rs.next()) {
103           map += rs.getString(1) -> rs.getString(2)
104         }
105       }
106     }
107     map.result()
108   }
