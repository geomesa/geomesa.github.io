31   override def create(info: TypeInfo)(implicit ex: ExecutionContext): Unit = {
32     val table = TableIdentifier(info.schema.raw, Name.raw)
33     val cName = TableName(Name.raw + "_pkey")
34     val create =
35       s"""CREATE TABLE IF NOT EXISTS ${table.qualified} (
36          |  type_name text not null,
37          |  key text not null,
38          |  value text not null,
39          |  mutable boolean default false
40          |);""".stripMargin
41     // migrate tables created before the mutable column was added
42     // note: we check for the column first, as an ALTER TABLE requires an ACCESS EXCLUSIVE lock even if it's a no-op,
43     // which will block behind (and then block) any partition maintenance that is reading from this table
44     val alter =
45       s"""DO $$$$
46          |BEGIN
47          |  IF NOT EXISTS (SELECT FROM pg_attribute WHERE attrelid = ${table.asRegclass} AND attname = 'mutable' AND NOT attisdropped) THEN
48          |    ALTER TABLE ${table.qualified} ADD COLUMN IF NOT EXISTS mutable boolean default false;
49          |  END IF;
50          |END$$$$;""".stripMargin
51     val constraint =
52       s"""DO $$$$
53          |BEGIN
54          |  IF NOT EXISTS (SELECT FROM pg_constraint WHERE conname = ${cName.asLiteral} AND conrelid = ${table.asRegclass}) THEN
55          |    ALTER TABLE ${table.qualified} ADD CONSTRAINT ${cName.quoted} PRIMARY KEY (type_name, key);
56          |  END IF;
57          |END$$$$;""".stripMargin
58 
59     Seq(create, alter, constraint).foreach(ex.execute)
60 
61     def insert(config: SftUserData[_], value: String): Unit = write(info.schema.raw, info.typeName, config, value)
62 
63     if (info.typeIdentifier != info.typeName) {
64       insert(SftUserData.IdentAlias, info.typeIdentifier)
65     }
66     insert(SftUserData.FidColumn, info.cols.fid.raw)
67     insert(SftUserData.DtgField, info.cols.dtg.raw)
68     insert(SftUserData.GeomField, info.cols.geom.raw)
69     insert(SftUserData.IntervalHours, String.valueOf(info.partitions.hoursPerPartition))
70     if (info.partitions.pagesPerRange != SftUserData.PagesPerRange.default) {
71       insert(SftUserData.PagesPerRange, String.valueOf(info.partitions.pagesPerRange))
72     }
73     info.partitions.maxPartitions.foreach(v => insert(SftUserData.MaxPartitions, String.valueOf(v)))
74     info.partitions.cronMinute.foreach(v => insert(SftUserData.CronMinute, String.valueOf(v)))
75     info.tables.writeAhead.tablespace.foreach(ts => insert(SftUserData.WriteAheadTableSpace, ts.raw))
76     info.tables.writeAheadPartitions.tablespace.foreach(ts => insert(SftUserData.WriteAheadPartitionsTableSpace, ts.raw))
77     info.tables.mainPartitions.tablespace.foreach(ts => insert(SftUserData.MainTableSpace, ts.raw))
78     // note: all tables key off the same user data field for logged/unlogged
79     if (!info.tables.mainPartitions.logged) {
80       insert(SftUserData.WalLogEnabled, "false")
81     }
82     if (info.cols.vis.isDefined) {
83       insert(SftUserData.VisEnabled, "true")
84     }
85     Seq(SftUserData.FilterWholeWorld, SftUserData.QueryInterceptors, SftUserData.ToastTupleTarget,
86         SftUserData.CoveringDtgIndex, SftUserData.TypeName).foreach { config =>
87       info.userData.get(config.key).foreach(v => insert(config, v))
88     }
89     // note: we don't use the user data for this, as it may be stale if the previous generation has been retired
90     info.previous.foreach(prev => insert(SftUserData.PreviousView, prev.typeName))
91   }

116     ex.executeUpdate(

146   /**
147    * Read the values for particular user data keys, across all types
148    *
149    * @param cx connection
150    * @param schema database schema
151    * @param configs user data to read
152    * @return (type name, key, value)
153    */
154   def read(cx: Connection, schema: String, configs: Seq[SftUserData[_]]): Seq[(String, String, String)] = {
155     val table = TableIdentifier(schema, Name.raw)
156     val select =
157       s"SELECT type_name, key, value FROM ${table.qualified} WHERE key IN (${configs.map(c => literal(c.key)).mkString(", ")});"
158     def query(): Seq[(String, String, String)] = {
159       WithClose(cx.createStatement()) { st =>
160         WithClose(st.executeQuery(select)) { rs =>
161           Iterator.continually(rs).takeWhile(_.next()).map(r => (r.getString(1), r.getString(2), r.getString(3))).toList
162         }
163       }
164     }
165     if (cx.getAutoCommit) {
166       // the table normally exists, so skip the check for it and handle the error if it doesn't
167       try { query() } catch {
168         case e: SQLException if e.getSQLState == UndefinedTable => Seq.empty
169       }
170     } else if (exists(cx, table)) {
171       // note: we can't rely on the error in a transaction, as it would cause the transaction to be aborted
172       query()
173     } else {
174       Seq.empty
175     }
176   }

178   /**
179    * Read user data for the given type
180    *
181    * @param cx connection
182    * @param schema database schema
183    * @param typeName feature type name
184    * @return
185    */
186   def read(cx: Connection, schema: String, typeName: String): Map[String, String] = {
187     val map = Map.newBuilder[String, String]
188     val userDataSql = s"select key, value from ${escape(schema)}.${UserDataTable.Name.quoted} where type_name = ?"
189     WithClose(cx.prepareStatement(userDataSql)) { statement =>
190       statement.setString(1, typeName)
191       WithClose(statement.executeQuery()) { rs =>
192         while (rs.next()) {
193           map += rs.getString(1) -> rs.getString(2)
194         }
195       }
196     }
197     map.result()
198   }
