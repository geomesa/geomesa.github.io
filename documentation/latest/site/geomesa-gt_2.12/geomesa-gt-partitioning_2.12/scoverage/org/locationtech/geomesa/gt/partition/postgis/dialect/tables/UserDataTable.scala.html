99   def read(cx: Connection, schema: String, typeName: String): Map[String, String] = {
100     val map = Map.newBuilder[String, String]
101     val userDataSql = s"select key, value from ${escape(schema)}.${UserDataTable.Name.quoted} where type_name = ?"
102     WithClose(cx.prepareStatement(userDataSql)) { statement =>
103       statement.setString(1, typeName)
104       WithClose(statement.executeQuery()) { rs =>
105         while (rs.next()) {
106           map += rs.getString(1) -> rs.getString(2)
107         }
108       }
109     }
110     map.result()
111   }
