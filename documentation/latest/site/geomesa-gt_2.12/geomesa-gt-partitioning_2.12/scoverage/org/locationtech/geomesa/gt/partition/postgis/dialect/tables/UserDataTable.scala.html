28   override def create(info: TypeInfo)(implicit ex: ExecutionContext): Unit = {
29     val table = TableIdentifier(info.schema.raw, Name.raw)
30     val cName = TableName(Name.raw + "_pkey")
31     val create =
32       s"""CREATE TABLE IF NOT EXISTS ${table.qualified} (
33          |  type_name text not null,
34          |  key text not null,
35          |  value text not null,
36          |  mutable boolean default false
37          |);""".stripMargin
38     // migrate tables created before the mutable column was added
39     // note: we check for the column first, as an ALTER TABLE requires an ACCESS EXCLUSIVE lock even if it's a no-op,
40     // which will block behind (and then block) any partition maintenance that is reading from this table
41     val alter =
42       s"""DO $$$$
43          |BEGIN
44          |  IF NOT EXISTS (SELECT FROM pg_attribute WHERE attrelid = ${table.asRegclass} AND attname = 'mutable' AND NOT attisdropped) THEN
45          |    ALTER TABLE ${table.qualified} ADD COLUMN IF NOT EXISTS mutable boolean default false;
46          |  END IF;
47          |END$$$$;""".stripMargin
48     val constraint =
49       s"""DO $$$$
50          |BEGIN
51          |  IF NOT EXISTS (SELECT FROM pg_constraint WHERE conname = ${cName.asLiteral} AND conrelid = ${table.asRegclass}) THEN
52          |    ALTER TABLE ${table.qualified} ADD CONSTRAINT ${cName.quoted} PRIMARY KEY (type_name, key);
53          |  END IF;
54          |END$$$$;""".stripMargin
55 
56     Seq(create, alter, constraint).foreach(ex.execute)
57 
58     val insertSql =
59       s"INSERT INTO ${table.qualified} (type_name, key, value, mutable) VALUES (?, ?, ?, ?) " +
60           s"ON CONFLICT (type_name, key) DO UPDATE SET value = EXCLUDED.value;"
61 
62     def insert(config: SftUserData[_], value: String): Unit =
63       ex.executeUpdate(insertSql, Seq(info.typeName, config.key, value, config.mutable))
64 
65     if (info.typeIdentifier != info.typeName) {
66       insert(SftUserData.IdentAlias, info.typeIdentifier)
67     }
68     insert(SftUserData.FidColumn, info.cols.fid.raw)
69     insert(SftUserData.DtgField, info.cols.dtg.raw)
70     insert(SftUserData.IntervalHours, String.valueOf(info.partitions.hoursPerPartition))
71     if (info.partitions.pagesPerRange != SftUserData.PagesPerRange.default) {
72       insert(SftUserData.PagesPerRange, String.valueOf(info.partitions.pagesPerRange))
73     }
74     info.partitions.maxPartitions.foreach(v => insert(SftUserData.MaxPartitions, String.valueOf(v)))
75     info.partitions.cronMinute.foreach(v => insert(SftUserData.CronMinute, String.valueOf(v)))
76     info.tables.writeAhead.tablespace.foreach(ts => insert(SftUserData.WriteAheadTableSpace, ts.raw))
77     info.tables.writeAheadPartitions.tablespace.foreach(ts => insert(SftUserData.WriteAheadPartitionsTableSpace, ts.raw))
78     info.tables.mainPartitions.tablespace.foreach(ts => insert(SftUserData.MainTableSpace, ts.raw))
79     // note: all tables key off the same user data field for logged/unlogged
80     if (!info.tables.mainPartitions.logged) {
81       insert(SftUserData.WalLogEnabled, "false")
82     }
83     if (info.cols.vis.isDefined) {
84       insert(SftUserData.VisEnabled, "true")
85     }
86     Seq(SftUserData.FilterWholeWorld, SftUserData.QueryInterceptors, SftUserData.ToastTupleTarget,
87         SftUserData.CoveringDtgIndex).foreach { config =>
88       info.userData.get(config.key).foreach(v => insert(config, v))
89     }
90   }

92   override def drop(info: TypeInfo)(implicit ex: ExecutionContext): Unit = {
93     val rs = ex.cx.getMetaData.getTables(null, info.schema.raw, Name.raw, null)
94     val exists = try { rs.next() } finally { rs.close() }
95     if (exists) {
96       ex.executeUpdate(s"DELETE FROM ${info.schema.quoted}.${Name.quoted} WHERE type_name = ?;", Seq(info.typeName))
97     }
98   }

100   /**
101    * Read user data for the given type
102    *
103    * @param cx connection
104    * @param schema database schema
105    * @param typeName feature type name
106    * @return
107    */
108   def read(cx: Connection, schema: String, typeName: String): Map[String, String] = {
109     val map = Map.newBuilder[String, String]
110     val userDataSql = s"select key, value from ${escape(schema)}.${UserDataTable.Name.quoted} where type_name = ?"
111     WithClose(cx.prepareStatement(userDataSql)) { statement =>
112       statement.setString(1, typeName)
113       WithClose(statement.executeQuery()) { rs =>
114         while (rs.next()) {
115           map += rs.getString(1) -> rs.getString(2)
116         }
117       }
118     }
119     map.result()
120   }
