28   override def create(info: TypeInfo)(implicit ex: ExecutionContext): Unit = {
29     val table = TableIdentifier(info.schema.raw, Name.raw)
30     val cName = TableName(Name.raw + "_pkey")

32       s"""ALTER TABLE IF EXISTS ${table.quoted} ADD COLUMN IF NOT EXISTS mutable boolean default false;"""
33     val create =
34       s"""CREATE TABLE IF NOT EXISTS ${table.quoted} (
35          |  type_name text not null,
36          |  key text not null,
37          |  value text not null,
38          |  mutable boolean default false
39          |);""".stripMargin
40     val constraint =
41       s"""DO $$$$
42          |BEGIN
43          |  IF NOT EXISTS (SELECT FROM pg_constraint WHERE conname = ${cName.asLiteral} AND conrelid = ${table.asRegclass}) THEN
44          |    ALTER TABLE ${table.quoted} ADD CONSTRAINT ${cName.quoted} PRIMARY KEY (type_name, key);
45          |  END IF;
46          |END$$$$;""".stripMargin
47 
48     Seq(alter, create, constraint).foreach(ex.execute)

50     val insertSql =
51       s"INSERT INTO ${table.quoted} (type_name, key, value, mutable) VALUES (?, ?, ?, ?) " +
52           s"ON CONFLICT (type_name, key) DO UPDATE SET value = EXCLUDED.value;"
53 
54     def insert(config: SftUserData[_], value: String): Unit =
55       ex.executeUpdate(insertSql, Seq(info.typeName, config.key, value, config.mutable))
56 
57     if (info.typeIdentifier != info.typeName) {
58       insert(SftUserData.IdentAlias, info.typeIdentifier)
59     }
60     insert(SftUserData.FidColumn, info.cols.fid.raw)
61     insert(SftUserData.DtgField, info.cols.dtg.raw)
62     insert(SftUserData.IntervalHours, String.valueOf(info.partitions.hoursPerPartition))

65     }
66     info.partitions.maxPartitions.foreach(v => insert(SftUserData.MaxPartitions, String.valueOf(v)))
67     info.partitions.cronMinute.foreach(v => insert(SftUserData.CronMinute, String.valueOf(v)))
68     info.tables.writeAhead.tablespace.foreach(ts => insert(SftUserData.WriteAheadTableSpace, ts.raw))
69     info.tables.writeAheadPartitions.tablespace.foreach(ts => insert(SftUserData.WriteAheadPartitionsTableSpace, ts.raw))
70     info.tables.mainPartitions.tablespace.foreach(ts => insert(SftUserData.MainTableSpace, ts.raw))
71     // note: all tables key off the same user data field for logged/unlogged
72     if (!info.tables.mainPartitions.logged) {
73       insert(SftUserData.WalLogEnabled, "false")
74     }
75     if (info.cols.vis.isDefined) {

77     }

79         SftUserData.CoveringDtgIndex).foreach { config =>

81     }
82   }

84   override def drop(info: TypeInfo)(implicit ex: ExecutionContext): Unit = {
85     val rs = ex.cx.getMetaData.getTables(null, info.schema.raw, Name.raw, null)

87     if (exists) {
88       ex.executeUpdate(s"DELETE FROM ${info.schema.quoted}.${Name.quoted} WHERE type_name = ?;", Seq(info.typeName))
89     }
90   }

92   /**
93    * Read user data for the given type
94    *
95    * @param cx connection
96    * @param schema database schema
97    * @param typeName feature type name
98    * @return
99    */
100   def read(cx: Connection, schema: String, typeName: String): Map[String, String] = {
101     val map = Map.newBuilder[String, String]
102     val userDataSql = s"select key, value from ${escape(schema)}.${UserDataTable.Name.quoted} where type_name = ?"
103     WithClose(cx.prepareStatement(userDataSql)) { statement =>
104       statement.setString(1, typeName)
105       WithClose(statement.executeQuery()) { rs =>
106         while (rs.next()) {
107           map += rs.getString(1) -> rs.getString(2)
108         }
109       }
110     }
111     map.result()
112   }
