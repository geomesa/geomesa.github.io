106   private object NormalizedExpression  {
107 
108     /**
109      * <p>
110      * For a given access expression this example will deduplicate, sort, flatten, and remove unneeded
111      * parentheses or quotes in the expressions. The following list gives examples of what each
112      * normalization step does.
113      *
114      * <ul>
115      * <li>As an example of flattening, the expression {@code A&(B&C)} flattens to {@code
116      * A&B&C}.</li>
117      * <li>As an example of sorting, the expression {@code (Z&Y)|(C&B)} sorts to {@code
118      * (B&C)|(Y&Z)}</li>
119      * <li>As an example of deduplication, the expression {@code X&Y&X} normalizes to {@code X&Y}</li>
120      * <li>As an example of unneeded quotes, the expression {@code "ABC"&"XYZ"} normalizes to
121      * {@code ABC&XYZ}</li>
122      * <li>As an example of unneeded parentheses, the expression {@code (((ABC)|(XYZ)))} normalizes to
123      * {@code ABC|XYZ}</li>
124      * </ul>
125      *
126      * <p>
127      * This algorithm attempts to have the same behavior as the one in the Accumulo 2.1
128      * ColumnVisibility class. However the implementation is very different.
129      */
130     def normalize(parsed: ParsedAccessExpression): NormalizedExpression = {
131       if (parsed.getType == ExpressionType.AUTHORIZATION) {
132         // If the authorization is quoted and it does not need to be quoted then the following two
133         // lines will remove the unnecessary quoting.
134         val unquoted = access.unquote(parsed.getExpression)
135         val quoted = access.quote(unquoted)
136         new NormalizedExpression(quoted, parsed.getType)
137       } else {
138         // The tree set does the work of sorting and deduplicating sub expressions.
139         val normalizedChildren = new java.util.TreeSet[NormalizedExpression]()
140         parsed.getChildren.forEach(child => flatten(parsed.getType, child, normalizedChildren))
141 
142         if (normalizedChildren.size() == 1) {
143           normalizedChildren.first()
144         } else {
145           val operator = if (parsed.getType == ExpressionType.AND) { "&" } else { "|" }
146           var sep = ""
147 
148           val builder = new StringBuilder()
149 
150           normalizedChildren.forEach { child =>
151             builder.append(sep)
152             if (child.expressionType == ExpressionType.AUTHORIZATION) {
153               builder.append(child.expression)
154             } else {
155               builder.append("(")
156               builder.append(child.expression)
157               builder.append(")")
158             }
159             sep = operator
160           }
161           new NormalizedExpression(builder.toString(), parsed.getType)
162         }
163       }
164     }
165 
166     /**
167      * This method helps with the flattening aspect of normalization by recursing down as far as
168      * possible the parse tree in the case when the expression type is the same. As long as the type
169      * is the same in the sub expression, keep using the same set.
170      */
171     private def flatten(
172       parentType: ExpressionType,
173       parsed: ParsedAccessExpression,
174       normalizedExpressions: java.util.SortedSet[NormalizedExpression]): Unit = {
175       if (parsed.getType == parentType) {
176         parsed.getChildren.forEach(child => flatten(parentType, child, normalizedExpressions))
177       } else {
178         // The type changed, so start again on the subexpression.
179         normalizedExpressions.add(normalize(parsed))
180       }
181     }
182 
183     // determines the sort order of different kinds of subexpressions.
184     private def typeOrder(expressionType: ExpressionType): Int = expressionType match {
185       case ExpressionType.AUTHORIZATION => 1
186       case ExpressionType.OR            => 2
187       case ExpressionType.AND           => 3
188       case ExpressionType.EMPTY => throw new IllegalArgumentException(s"Unexpected type: $expressionType")
189     }
190   }
191   /**
192    * As part of normalizing access expression this class is used to sort and dedupe sub-expressions
193    * in a tree set.
194    */
195   private case class NormalizedExpression(expression: String, expressionType: ExpressionType)
196       extends Comparable[NormalizedExpression] {
197 
198     override def compareTo(o: NormalizedExpression): Int = {
199       var cmp = NormalizedExpression.typeOrder(expressionType) - NormalizedExpression.typeOrder(o.expressionType)
200       if (cmp == 0) {
201         if (expressionType == ExpressionType.AUTHORIZATION) {
202           // sort based on the unquoted and unescaped form of the authorization
203           cmp = access.unquote(expression).compareTo(access.unquote(o.expression))
204         } else {
205           cmp = expression.compareTo(o.expression)
206         }
207       }
208       cmp
209     }
210 
211     override def equals(obj: Any): Boolean =
212       this == obj || (canEqual(obj) && obj.asInstanceOf[NormalizedExpression].compareTo(this) == 0)
213 
214     override def hashCode(): Int = {
215       if (expressionType == ExpressionType.AUTHORIZATION) {
216         access.unquote(expression).hashCode
217       } else {
218         expression.hashCode
219       }
220     }
221   }
