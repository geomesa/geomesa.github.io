23 /**
24  * Function to compare auths with vis. Note that this function "fails open", in that if visibilities are not found
25  * it will count as visible
26  */
27 class IsVisibleFilterFunction extends FunctionExpressionImpl(IsVisibleFilterFunction.Name) {
28 
29   override def evaluate(obj: Object): Object = {
30     val vis = getExpression(1).evaluate(obj).asInstanceOf[String]
31     if (vis == null || vis.isBlank) {
32       java.lang.Boolean.TRUE
33     } else {
34       val auths = getExpression(0).evaluate(obj, classOf[String])
35       if (auths == null || auths.isBlank) {
36         java.lang.Boolean.FALSE
37       } else {
38         IsVisibleFilterFunction.visibilityCache.get(auths -> vis)
39       }
40     }
41   }
42 }

77   }
