28 /**
29   * Access values from a json-type string field. Syntax must start with '$.'. The first part of the path
30   * selects the simple feature attribute, and the rest of the path selects within the json contained in
31   * that attribute.
32   *
33   * Note: this class is optimized for `KryoBufferSimpleFeature`s. It will work on standard simple features,
34   * but will incur a serialization cost.
35   */
36 trait JsonPathPropertyAccessor extends PropertyAccessor with LazyLogging {
37 
38   import org.locationtech.geomesa.utils.geotools.RichAttributeDescriptors.RichAttributeDescriptor
39 
40   import scala.collection.JavaConverters._
41 
42   override def canHandle(obj: AnyRef, xpath: String, target: Class[_]): Boolean = {
43     val path =
44       try {
45         JsonPathPropertyAccessor.paths.get(xpath)
46       } catch {
47         case NonFatal(e) =>
48           logger.warn(s"Error parsing path: $xpath", e)
49           JsonPath(Seq.empty, None)
50       }
51 
52     if (path.isEmpty) { false } else {
53       path.head match {
54         case PathAttribute(name: String, _) =>
55           val descriptor = obj match {
56             case s: SimpleFeature => s.getFeatureType.getDescriptor(name)
57             case s: SimpleFeatureType => s.getDescriptor(name)
58             case _ => null
59           }
60           descriptor != null && descriptor.getType.getBinding == classOf[String]
61 
62         case PathAttributeWildCard | PathDeepScan =>
63           val sft = obj match {
64             case s: SimpleFeature => s.getFeatureType
65             case s: SimpleFeatureType => s
66             case _ => null
67           }
68           sft != null && sft.getAttributeDescriptors.asScala.exists(_.getType.getBinding == classOf[String])
69 
70         case _ => false
71       }
72     }
73   }
74 
75   override def get[T](obj: AnyRef, xpath: String, target: Class[T]): T = {
76 
77     val path = JsonPathPropertyAccessor.paths.get(xpath)
78 
79     val result = obj match {
80       case s: JsonAwareFeature =>
81         val i = attribute(s.getFeatureType, path.head)
82         if (s.getFeatureType.getDescriptor(i).isJson()) {
83           s.readJsonPath(i, path.tail)
84         } else {
85           JsonPathPropertyAccessor.evaluateJsonPath(s.getAttribute(i).asInstanceOf[String], path.tail)
86         }
87 
88       case s: SimpleFeature =>
89         JsonPathPropertyAccessor.evaluateJsonPath(s.getAttribute(attribute(s.getFeatureType, path.head)).asInstanceOf[String], path.tail)
90 
91       case s: SimpleFeatureType =>
92         // remove the json flag, so that transform serializations don't try to serialize
93         // json path results that aren't valid json objects or arrays
94         val descriptor = s.getDescriptor(attribute(s, path.head))
95         val builder = new AttributeTypeBuilder()
96         builder.init(descriptor)
97         val result = builder.buildDescriptor(descriptor.getLocalName)
98         result.getUserData.remove(SimpleFeatureTypes.AttributeOptions.OptJson)
99         result
100     }
101 
102     if (target == null) {
103       result.asInstanceOf[T]
104     } else {
105       FastConverter.convert(result, target)
106     }
107   }
108 
109   override def set[T](obj: Any, xpath: String, value: T, target: Class[T]): Unit = throw new UnsupportedOperationException()
110 
111   private def attribute(sft: SimpleFeatureType, head: PathElement): Int = {
112     head match {
113       case PathAttribute(name: String, _) =>
114         sft.indexOf(name)
115 
116       case _ =>
117         // we know it will be a wildcard due to canHandle
118         // prioritize fields marked json over generic strings
119         // note: will only match first json attribute if more than 1
120         val i = sft.getAttributeDescriptors.asScala.indexWhere(_.isJson())
121         if (i != -1) { i } else {
122           sft.getAttributeDescriptors.asScala.indexWhere(_.getType.getBinding == classOf[String])
123         }
124     }
125   }
126 }

163   /**
164    * Evaluate a json path using the json-path lib. This method is a fallback for our custom kryo-deserialization,
165    * and requires parsing the input into the json-path AST. The result has been modified to be consistent
166    * with our custom serialization (for better interop with geotools), i.e. json objects are serialized to json
167    * strings.
168    *
169    * @param json json doc
170    * @param path json path
171    * @return
172    */
173   def evaluateJsonPath(json: String, path: JsonPath): AnyRef = {
174     if (json == null) {
175       return null
176     }
177     val parsed = com.jayway.jsonpath.JsonPath.using(pathConfig).parse(json)
178     val list = parsed.read[java.util.List[AnyRef]](JsonPathParser.print(path))
179     if (list == null || list.isEmpty) {
180       // special handling to return 0 for length functions without matches
181       path.function match {
182         case Some(PathFunction.LengthFunction) => Int.box(0)
183         case _ => null
184       }
185     } else {
186       val transformed = list.asScala.map {
187         case o: java.util.Map[String, AnyRef] => gson.toJson(o)
188         case a: java.util.List[AnyRef] => unwrapArray(a)
189         case p => p
190       }
191       if (transformed.lengthCompare(1) == 0) { transformed.head } else { transformed.asJava }
192     }
193   }
194 
195   private def unwrapArray(array: java.util.List[AnyRef]): java.util.List[AnyRef] = {
196     array.asScala.map {
197       case o: java.util.Map[String, AnyRef] => gson.toJson(o)
198       case a: java.util.List[AnyRef] => unwrapArray(a)
199       case p => p
200     }.asJava
201   }
