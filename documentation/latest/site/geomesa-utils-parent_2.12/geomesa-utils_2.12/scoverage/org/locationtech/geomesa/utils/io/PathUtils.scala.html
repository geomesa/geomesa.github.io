43   /**
44     * Checks to see if the path uses a URL pattern and then if it is *not* file://
45     *
46     * @param path Input resource path
47     * @return     Whether or not the resource is remote.

61   /**
62     * Gets the base file name and the extension. Useful for adding unique ids to a common file name,
63     * while preserving the extension
64     *
65     * @param path path
66     * @param includeDot if true, the '.' will be preserved in the extension, otherwise it will be dropped
67     * @return (base name including path prefix, extension)
68     */
69   def getBaseNameAndExtension(path: String, includeDot: Boolean = true): (String, String) = {
70     def dotIndex(base: Int): Int = if (includeDot) { base } else { base + 1}
71     val split = FilenameUtils.indexOfExtension(path)
72     if (split == -1) { (path, "") } else {
73       val withoutExtension = path.substring(0, split)
74       // look for file names like 'foo.tar.gz'
75       val secondSplit = FilenameUtils.indexOfExtension(withoutExtension)
76       if (secondSplit != -1 && CompressionUtils.isCompressedFilename(path)) {
77         (path.substring(0, secondSplit), path.substring(dotIndex(secondSplit)))
78       } else {
79         (withoutExtension, path.substring(dotIndex(split)))
80       }
81     }
82   }
83 
84   /**
85     * Wrap the input stream in a decompressor, if the file is compressed
86     *
87     * @param is input stream
88     * @param filename filename (used to determine compression)
89     * @return
90     */
91   def handleCompression(is: InputStream, filename: String): InputStream = {
92     CompressionUtils.Utils.find(_.isCompressedFilename(filename)) match {
93       case None => is
94       case Some(utils) => utils.decompress(is)
95     }
96   }

98   /**

101     * @param path path
102     */
103   def deleteRecursively(path: Path): Unit = Files.walkFileTree(path, new DeleteFileVisitor())
104 
105   private def chooseDelegate(path: String): FileSystemDelegate =
106     if (hadoopDelegate != null && uriRegex.matcher(path).matches()) { hadoopDelegate } else { localDelegate }
107 
108   /**
109     * File visitor to delete nested paths
110     */
111   private class DeleteFileVisitor extends FileVisitor[Path] {
112 
113     override def visitFileFailed(file: Path, exc: IOException): FileVisitResult = FileVisitResult.CONTINUE
114 
115     override def visitFile(file: Path, attrs: BasicFileAttributes): FileVisitResult = {
116       if (!attrs.isDirectory) {
117         Files.delete(file)
118       }
119       FileVisitResult.CONTINUE
120     }
121 
122     override def preVisitDirectory(dir: Path, attrs: BasicFileAttributes): FileVisitResult = FileVisitResult.CONTINUE
123 
124     override def postVisitDirectory(dir: Path, exc: IOException): FileVisitResult = {
125       Files.delete(dir)
126       FileVisitResult.CONTINUE
127     }
128   }
