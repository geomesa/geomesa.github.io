27   @throws(classOf[ParsingException])
28   def parse(path: String, report: Boolean = true): JsonPath = {
29     if (path == null) {
30       throw new IllegalArgumentException("Path must not be null")
31     }
32     val runner = if (report) { ReportingParseRunner(Parser.Path) } else { BasicParseRunner(Parser.Path) }
33     val fixedPath = if (path.startsWith("$")) { path } else s"$$.$path"
34     val parsing = runner.run(fixedPath)
35     parsing.result.getOrElse {
36       throw new ParsingException(s"Invalid json path: ${ErrorUtils.printParseErrors(parsing)}")
37     }
38   }
39 
40   /**
41    * Prints a parsed path back into a path string
42    *
43    * @param path path
44    * @param prefix prefix
45    * @return
46    */
47   def print(path: JsonPath, prefix: Char = '$'): String = {
48     require(path.elements.nonEmpty, "Path must not be empty")
49 
50     val builder = new StringBuilder()
51 
52     def append(e: Any): Unit = {
53       val s = e.toString
54       // handle deep scans followed by dot selectors
55       if (s.startsWith(".") && builder.endsWith(Seq('.', '.'))) {
56         builder.append(s.substring(1))
57       } else {
58         builder.append(s)
59       }
60     }
61 
62     builder.append(prefix)
63     path.elements.foreach(append)
64     path.function.foreach(append)
65     builder.toString()
66   }

74   case class JsonPath(elements: Seq[PathElement], function: Option[PathFunction] = None) {

123     /**
124      * Function over a seq of numbers
125      *
126      * @param name name of the function
127      * @param op operation
128      */
129     abstract class NumericFunction(name: String, op: Seq[Double] => Double) extends PathFunction {
130       private def toNum(v: Any): Double = v match {
131         case n: Number => n.doubleValue
132         case null => 0.0
133         case n => n.toString.toDouble
134       }
135       override def apply(args: Any): Any = {
136         args match {
137           case s: java.util.List[_] if !s.isEmpty => op(s.asScala.toSeq.map(toNum))
138           case s: Seq[_] if s.nonEmpty => op(s.map(toNum))
139           case _ => null
140         }
141       }
142       override def toString: String = s".$name()"
143     }
144 
145     case object MinFunction extends NumericFunction("min", _.min)
146     case object MaxFunction extends NumericFunction("max", _.max)
147     case object SumFunction extends NumericFunction("sum", _.sum)
148     case object AvgFunction extends NumericFunction("avg", a => a.sum / a.length)
149 
150     case object LengthFunction extends PathFunction {
151       override def apply(obj: Any): Any = obj match {
152         case s: java.util.List[_] => s.size
153         case s: Seq[_] => s.size
154         case s: String => s.length
155         case _ => null
156       }
157       override def toString: String = ".length()"
158     }
159 
160     case object FirstFunction extends PathFunction {
161       override def apply(obj: Any): Any = obj match {
162         case s: java.util.List[_] if !s.isEmpty => s.get(0)
163         case s: Seq[_] if s.nonEmpty => s.head
164         case s: String if s.nonEmpty => s.substring(0, 1)
165         case _ => null
166       }
167       override def toString: String = ".first()"
168     }
169 
170     case object LastFunction extends PathFunction {
171       override def apply(obj: Any): Any = obj match {
172         case s: java.util.List[_] if !s.isEmpty => s.get(s.size() - 1)
173         case s: Seq[_] if s.nonEmpty => s.last
174         case s: String if s.nonEmpty => s.substring(s.length - 1, s.length)
175         case _ => null
176       }
177       override def toString: String = ".last()"
178     }
179 
180     // note: negative indices count backwards from the end
181     case class IndexFunction(i: Int) extends PathFunction {
182       override def apply(obj: Any): Any = obj match {
183         case s: java.util.List[_] if i >= 0 && i < s.size => s.get(i)
184         case s: java.util.List[_] if i < 0 && s.size + i >= 0 => s.get(s.size + i)
185         case s: Seq[_] if i >= 0 && i < s.size => s(i)
186         case s: Seq[_] if i < 0 && s.size + i >= 0 => s(s.size + i)
187         case s: String if i >= 0 && i < s.length => s.substring(i, i + 1)
188         case s: String if i < 0 && s.length + i >= 0 => s.substring(0, s.length + i + 1)
189         case _ => null
190       }
191       override def toString: String = s".index($i)"
192     }
193   }

240     // =~ left matches regular expression [?(@.name =~ /foo.*?/i)]
241     case class RegexOp(left: PathExpression, regex: Pattern) extends FilterOp {
242 
243       override def toString: String = {
244         val flag = if ((regex.flags() & Pattern.CASE_INSENSITIVE) != 0) { "i" } else { "" }
245         s"$left =~ /${regex.pattern()}/$flag"
246       }
247 
248       // note: we implement equals and hash code since Pattern does not (mainly for unit tests)
249       override def hashCode(): Int = Objects.hash(left, regex.pattern(), Int.box(regex.flags()))
250 
251       override def equals(obj: Any): Boolean = obj match {
252         case r: RegexOp => left == r.left && regex.pattern() == r.regex.pattern() && regex.flags() == r.regex.flags()
253         case _ => false
254       }
255 
256       override def canEqual(that: Any): Boolean = that.isInstanceOf[RegexOp]
257     }

354   // main parsing rule
355   def Path: Rule1[JsonPath] =
356     rule { "$" ~ zeroOrMore(Element) ~ optional(Function) ~~> ((e, f) => JsonPath(e, f)) ~ EOI }
357 
358   def Element: Rule1[PathElement] = rule {
359     Attribute | ArrayIndices | ArrayIndexRange | BracketedAttribute | AttributeWildCard | IndexWildCard | DeepScan | FilterOp
360   }
361 
362   def IndexWildCard: Rule1[PathElement] = rule { "[*]" ~ push(PathIndexWildCard) }
363 
364   def AttributeWildCard: Rule1[PathElement] = rule { ".*" ~ push(PathAttributeWildCard) }
365 
366   // we have to push the deep scan directly onto the stack as there is no forward matching and
367   // it's ridiculous trying to combine Rule1's and Rule2's
368   def DeepScan: Rule1[PathElement] = rule { "." ~ toRunAction(pushDeepScan) ~ (Attribute | BracketedAttribute | AttributeWildCard) }
369 
370   // note: this assumes that we are inside a zeroOrMore, which is currently the case
371   // the zeroOrMore will have pushed a single list onto the value stack - we append our value to that
372   private def pushDeepScan(context: Context[Any]): Unit =
373     context.getValueStack.push(PathDeepScan :: context.getValueStack.pop.asInstanceOf[List[_]])
374 
375   def ArrayIndices: Rule1[PathIndices] =
376     rule { "[" ~ oneOrMore(int, ",") ~ "]" ~~> PathIndices.apply }
377 
378   def ArrayIndexRange: Rule1[PathIndexRange] =
379     rule { "[" ~ optional(int) ~ ":" ~ optional(int) ~ "]" ~~> ((from, to) => PathIndexRange(from, to)) }
380 
381   def Attribute: Rule1[PathAttribute] = rule { "." ~ oneOrMore(char) ~> { s => PathAttribute(s) } ~ !"(" }
382 
383   def BracketedAttribute: Rule1[PathAttribute] =
384     rule { "[" ~ (unquotedString | singleQuotedString) ~~> { s => PathAttribute(s, bracketed = true) } ~ "]" }
385 
386   def Function: Rule1[PathFunction] =
387     functionMin | functionMax | functionAvg | functionLength | functionSum | functionFirst | functionLast | functionIndex
388 
389   private def functionMin: Rule1[PathFunction] = rule { ".min()" ~ push(PathFunction.MinFunction) }
390   private def functionMax: Rule1[PathFunction] = rule { ".max()" ~ push(PathFunction.MaxFunction) }
391   private def functionAvg: Rule1[PathFunction] = rule { ".avg()" ~ push(PathFunction.AvgFunction) }
392   private def functionLength: Rule1[PathFunction] = rule { ".length()" ~ push(PathFunction.LengthFunction) }
393   private def functionSum: Rule1[PathFunction] = rule { ".sum()" ~ push(PathFunction.SumFunction) }
394   private def functionFirst: Rule1[PathFunction] = rule { ".first()" ~ push(PathFunction.FirstFunction) }
395   private def functionLast: Rule1[PathFunction] = rule { ".last()" ~ push(PathFunction.LastFunction) }
396   private def functionIndex: Rule1[PathFunction] = rule { ".index(" ~ int ~ ")" ~~> PathFunction.IndexFunction.apply }
397 
398   /*
399    * from jayway's impl:
400    *
401    *  LogicalOR               = LogicalAND { '||' LogicalAND }
402    *  LogicalAND              = LogicalANDOperand { '&&' LogicalANDOperand }
403    *  LogicalANDOperand       = RelationalExpression | '(' LogicalOR ')' | '!' LogicalANDOperand
404    *  RelationalExpression    = Value [ RelationalOperator Value ]
405    */
406 
407   // filter op on a path
408   def FilterOp: Rule1[PathFilter] = rule { "[?(" ~ whitespace ~ logicalOr ~ whitespace ~ ")]" ~~> PathFilter.apply  }
409 
410   private def logicalOr: Rule1[FilterOp] = rule {
411     oneOrMore(logicalAnd, whitespace ~ "||" ~ whitespace) ~~> { ops =>
412       if (ops.lengthCompare(1) == 0) { ops.head } else { OrFilterOp(ops) }
413     }
414   }
415   private def logicalAnd: Rule1[FilterOp] = rule {
416     oneOrMore(andOperand, whitespace ~ "&&" ~ whitespace) ~~> { ops =>
417       if (ops.lengthCompare(1) == 0) { ops.head } else { AndFilterOp(ops) }
418     }
419   }
420   private def andOperand: Rule1[FilterOp] = rule { filter | "(" ~ whitespace ~ logicalOr ~ whitespace ~ ")" | logicalNot }
421   private def logicalNot: Rule1[FilterOp] = rule { "!" ~ whitespace ~ andOperand ~~> NotFilterOp.apply  }
422 
423   private def filter: Rule1[FilterOp] =
424     equalsFilter | notEqualsFilter | lessThanOrEqualsFilter | lessThanFilter | greaterThanOrEqualsFilter |
425       greaterThanFilter | inFilter | ninFilter | subsetFilter | anyOfFilter | noneOfFilter | sizeFilter |
426       regexFilter | isEmptyFilter | existsFilter
427 
428   private def equalsFilter: Rule1[EqualsOp] = rule { binaryFilter("==", EqualsOp.apply) }
429   private def notEqualsFilter: Rule1[NotEqualsOp] = rule { binaryFilter("!=", NotEqualsOp.apply) }
430   private def lessThanOrEqualsFilter: Rule1[LessThanOrEqualsOp] = rule { binaryFilter("<=", LessThanOrEqualsOp.apply) }
431   private def lessThanFilter: Rule1[LessThanOp] = rule { binaryFilter("<", LessThanOp.apply) }
432   private def greaterThanOrEqualsFilter: Rule1[GreaterThanOrEqualsOp] = rule { binaryFilter(">=", GreaterThanOrEqualsOp.apply) }
433   private def greaterThanFilter: Rule1[GreaterThanOp] = rule { binaryFilter(">", GreaterThanOp.apply) }
434   private def inFilter: Rule1[InOp] = rule { binaryStringOpFilter("in", InOp.apply) }
435   private def ninFilter: Rule1[NotInOp] = rule { binaryStringOpFilter("nin", NotInOp.apply) }
436   private def subsetFilter: Rule1[IsSubsetOp] = rule { binaryStringOpFilter("subsetof", IsSubsetOp.apply) }
437   private def anyOfFilter: Rule1[AnyOfOp] = rule { binaryStringOpFilter("anyof", AnyOfOp.apply) }
438   private def noneOfFilter: Rule1[NoneOfOp] = rule { binaryStringOpFilter("noneof", NoneOfOp.apply) }
439   private def sizeFilter: Rule1[EqualSizeOp] = rule { binaryStringOpFilter("size", EqualSizeOp.apply) }
440   private def isEmptyFilter: Rule1[IsEmptyOp] = rule { binaryStringOpFilter("empty", IsEmptyOp.apply) }
441 
442   private def binaryFilter[T](op: String, builder: (FilterExpression, FilterExpression) => T) =
443     (filterExpression ~ whitespace ~ op ~ whitespace ~ filterExpression) ~~> { (left, right) => builder(left, right) }
444   private def binaryStringOpFilter[T](op: String, builder: (FilterExpression, FilterExpression) => T) =
445     (filterExpression ~ space ~ op ~ space ~ filterExpression) ~~> { (left, right) => builder(left, right) }
446 
447   private def regexFilter: Rule1[RegexOp] = rule {
448     (pathFilterExpression ~ whitespace ~ "=~" ~ whitespace ~ "/" ~ regexPattern ~ "/" ~ optional("i" ~ push(true))) ~~> {
449       (left, pattern, i) => RegexOp(left, Pattern.compile(pattern, if (i.contains(true)) { Pattern.CASE_INSENSITIVE } else { 0 }))
450     }
451   }
452   private def regexPattern: Rule1[String] = rule {
453     oneOrMore((noneOf("/") ~? notControlChar) | escapedChar) ~> StringEscapeUtils.unescapeJava
454   }
455 
456   private def existsFilter: Rule1[ExistsOp] = rule { filterExpression ~~> ExistsOp.apply }
457 
458   // an argument in a filter - a path or literal
459   private def filterExpression: Rule1[FilterExpression] =
460     pathFilterExpression | literalFilterString | literalFilterNumber | literalFilterBoolean |
461       literalFilterStringArray | literalFilterNumberArray | literalFilterBooleanArray
462 
463   private def pathFilterExpression: Rule1[PathExpression] = rule {
464     ("$"  ~ push(true) | "@" ~ push(false)) ~ zeroOrMore(Element) ~ optional(Function) ~~> { (absolute, e, f) =>
465       PathExpression(JsonPath(e, f), absolute = absolute)
466     }
467   }
468 
469   private def literalFilterNumber: Rule1[NumericLiteral] =
470     (double | float | long | int) ~~> (n => NumericLiteral(BigDecimal(n.toString)))
471   private def literalFilterString: Rule1[StringLiteral] = (quotedString | singleQuotedString) ~~> StringLiteral.apply
472   private def literalFilterBoolean: Rule1[BooleanLiteral] = boolean ~~> BooleanLiteral.apply
473   private def literalFilterNumberArray: Rule1[ArrayLiteral[NumericLiteral]] =
474     "[" ~ whitespace ~ oneOrMore(literalFilterNumber, "," ~ whitespace) ~ whitespace ~ "]" ~~> ArrayLiteral.apply
475   private def literalFilterStringArray: Rule1[ArrayLiteral[StringLiteral]] =
476     "[" ~ whitespace ~ oneOrMore(literalFilterString, "," ~ whitespace) ~ whitespace ~ "]" ~~> ArrayLiteral.apply
477   private def literalFilterBooleanArray: Rule1[ArrayLiteral[BooleanLiteral]] =
478     "[" ~ whitespace ~ oneOrMore(literalFilterBoolean, "," ~ whitespace) ~ whitespace ~ "]" ~~> ArrayLiteral.apply
