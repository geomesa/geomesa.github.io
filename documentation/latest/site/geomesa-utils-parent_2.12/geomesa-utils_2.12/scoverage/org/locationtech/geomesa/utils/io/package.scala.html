58   /**
59    * Similar to java's try-with-resources, or scala 2.13's Using, allows for using an object then closing in a finally block
60    */
61   object WithClose {
62 
63     def apply[C : IsCloseable, T](c: C)(fn: C => T): T = {
64       if (c == null) {
65         return fn(c)
66       }
67       val ev: IsCloseable[C] = implicitly[IsCloseable[C]]
68       var toThrow: Throwable = null
69       try {
70         fn(c)
71       } catch {
72         case t: Throwable =>
73           toThrow = t
74           null.asInstanceOf[T] // compiler doesn't know `finally` will throw
75       } finally {
76         if (toThrow eq null) {
77           ev.close(c).get
78         } else {
79           try {
80             ev.close(c).get
81           } catch {
82             case other: Throwable => toThrow.addSuppressed(other)
83           } finally {
84             throw toThrow
85           }
86         }
87       }
88     }
89 
90     def apply[C1 : IsCloseable, C2 : IsCloseable, T](c1: C1, c2: => C2)(fn: (C1, C2) => T): T =
91       apply(c1) { c1 => apply(c2) { c2 => fn(c1, c2) } }
92   }

94   /**
95    * Creates unique file names generated from a base name, by appending a sequence number
96    * before the file extension.
97    *
98    * For example, given the file name 'foo.txt', will return 'foo_000.txt', 'foo_001.txt', etc.
99    * If the iterator exceeds the specified number of digits, it will start to append additional
100    * digits to ensure uniqueness, e.g. 'foo_999.txt', 'foo_1000.txt', 'foo_1001.txt', etc.
101    *
102    * @param path file name path
103    * @param digits number of digits used to format the sequence number
104    */
105   class IncrementingFileName(path: String, digits: Int = 3) extends Iterator[String] {
106 
107     private val i = new AtomicInteger(0)
108     private val format = s"_%0${digits}d"
109     private val (prefix, suffix) = PathUtils.getBaseNameAndExtension(path)
110 
111     override def hasNext: Boolean = true
112 
113     override def next(): String = s"$prefix${format.format(i.getAndIncrement())}$suffix"
114   }
