23   extends FunctionExpressionImpl(JsonPathFilterFunction.Name) with VolatileFunction with AttributeAwareExpression with LazyLogging {
24 
25   import JsonPathFilterFunction.ff

27   private val cache = new ConcurrentHashMap[String, PropertyAccessor]
28 
29   override def evaluate(obj: Object): AnyRef = {
30     val sf = obj match {
31       case sf: SimpleFeature => sf
32       case _ =>
33         throw new IllegalArgumentException(
34           s"Expected SimpleFeature, but received ${obj.getClass}. Only simple features are supported: $obj")
35     }
36     val base = params.get(0) match {
37       case p: PropertyName => p.getPropertyName // for property name expressions, we want the attribute name
38       case p => p.evaluate(sf).asInstanceOf[String] // for literals, we want to evaluate the expression
39     }
40     val path = if (params.size() < 2) { base } else { s"$$.$base.${params.get(1).evaluate(sf)}" }
41     var accessor = cache.get(path)
42     if (accessor == null) {
43       accessor = SimpleFeaturePropertyAccessor.getAccessor(sf, path).getOrElse {
44         throw new RuntimeException(s"Can't handle property '$path' for feature type " +
45             s"${sf.getFeatureType.getTypeName} ${SimpleFeatureTypes.encodeType(sf.getFeatureType)}")
46       }
47       cache.put(path, accessor)
48     }
49     accessor.get(sf, path, classOf[AnyRef])
50   }
51 
52   override def visit(visitor: FilterAttributeExtractor, data: AnyRef): AnyRef = {
53     val prop = params.get(0) match {
54       case p: PropertyName => p
55       case p => ff.property(p.evaluate(null).asInstanceOf[String])
56     }
57     visitor.visit(prop, data)
58   }
59 }

68     )
