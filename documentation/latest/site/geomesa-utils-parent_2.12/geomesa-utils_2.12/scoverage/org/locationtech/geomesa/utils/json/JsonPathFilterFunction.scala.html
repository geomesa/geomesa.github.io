22 class JsonPathFilterFunction
23   extends FunctionExpressionImpl(JsonPathFilterFunction.Name) with VolatileFunction with AttributeAwareExpression with LazyLogging {
24 
25   import JsonPathFilterFunction.ff
26 
27   private val cache = new ConcurrentHashMap[String, PropertyAccessor]
28 
29   override def evaluate(obj: AnyRef): AnyRef = {
30     val sf = obj match {
31       case sf: SimpleFeature => sf
32       case _ =>
33         throw new IllegalArgumentException(
34           s"Expected SimpleFeature, but received ${obj.getClass}. Only simple features are supported: $obj")
35     }
36     val path = this.path(sf)
37     var accessor = cache.get(path)
38     if (accessor == null) {
39       accessor = SimpleFeaturePropertyAccessor.getAccessor(sf, path).getOrElse {
40         throw new RuntimeException(s"Can't handle property '$path' for feature type " +
41             s"${sf.getFeatureType.getTypeName} ${SimpleFeatureTypes.encodeType(sf.getFeatureType)}")
42       }
43       cache.put(path, accessor)
44     }
45     accessor.get(sf, path, classOf[AnyRef])
46   }
47 
48   /**
49    * Get the json path from the parameters
50    *
51    * @param obj object being evaluated
52    * @return
53    */
54   def path(obj: AnyRef): String = {
55     val base = params.get(0) match {
56       case p: PropertyName => p.getPropertyName // for property name expressions, we want the attribute name
57       case p => p.evaluate(obj).asInstanceOf[String] // for literals, we want to evaluate the expression
58     }
59     if (params.size() < 2) { base } else { s"$$.$base.${params.get(1).evaluate(obj)}" }
60   }
61 
62   override def visit(visitor: FilterAttributeExtractor, data: AnyRef): AnyRef = {
63     val prop = params.get(0) match {
64       case p: PropertyName => p
65       case p => ff.property(p.evaluate(null).asInstanceOf[String])
66     }
67     visitor.visit(prop, data)
68   }
69 }

71 object JsonPathFilterFunction {
72   val Name =

75       FunctionNameImpl.parameter("value", classOf[String]),
76       FunctionNameImpl.parameter("path", classOf[String]), // can be a full path expression, OR an attribute name
77       FunctionNameImpl.parameter("nested-path", classOf[String], 0, 1) // (optional) if path is an attribute name, the nested path into the attribute
78     )
79 
80   private val ff = CommonFactoryFinder.getFilterFactory
