24 /**

26  */
27 sealed trait Transform {
28 
29   /**
30    * The name of the attribute
31    *
32    * @return
33    */
34   def name: String
35 
36   /**
37    * The type of the attribute
38    *
39    * @return

52 /**
53   * Convenience methods for working with relational query projections (aka transforms)
54   */
55 object Transform {
56 
57   val DefinitionDelimiter = ";"
58 
59   /**
60    * A simple relational projection (selection)
61    *
62    * @param name name
63    * @param binding binding
64    * @param i index of the original attribute being projected
65    */
66   case class PropertyTransform(name: String, binding: Class[_], i: Int) extends Transform {
67     override def evaluate(feature: SimpleFeature): AnyRef = feature.getAttribute(i)
68   }
69 
70   /**
71    * A simple relational projection, but also renamed
72    *
73    * @param name name
74    * @param binding binding
75    * @param original name of the original attribute being projected
76    * @param i index of the original attribute being projected
77    */
78   case class RenameTransform(name: String, binding: Class[_], original: String, i: Int) extends Transform {
79     override def evaluate(feature: SimpleFeature): AnyRef = feature.getAttribute(i)
80   }
81 
82   /**
83    * An attribute comprised of an expression (function, math transform, json-path, etc)
84    *
85    * @param name name
86    * @param binding binding
87    * @param expression expression
88    */
89   case class ExpressionTransform(name: String, binding: Class[_], expression: Expression) extends Transform {
90     override def evaluate(feature: SimpleFeature): AnyRef = expression.evaluate(feature)
91   }
92 
93   object Transforms {
94 
95     /**
96      * Create transform definitions for a delimited expression string
97      *
98      * @param sft original simple feature type
99      * @param transforms semi-colon delimited transform expressions
100      * @return
101      */
102     def apply(sft: SimpleFeatureType, transforms: String): Seq[Transform] =
103       if (transforms.isEmpty) { Seq.empty } else { apply(sft, transforms.split(DefinitionDelimiter)) }
104 
105     /**
106      * Create transform definitions for a sequence of expression strings
107      *
108      * @param sft original simple feature type
109      * @param transforms transform expressions
110      * @return
111      */
112     def apply(sft: SimpleFeatureType, transforms: Seq[String]): Seq[Transform] =
113       transforms.map(definition(sft, _))
114 
115     /**
116      * Create a transform definition from an expression
117      *
118      * @param sft simple feature type
119      * @param transform transform expression
120      * @return
121      */
122     private def definition(sft: SimpleFeatureType, transform: String): Transform = {
123       val equals = transform.indexOf('=')
124       if (equals == -1) {
125         val expression = transform.trim()
126         val name = StringUtils.strip(expression, "\"")
127         val i = sft.indexOf(name)
128         if (i == -1) {
129           attributeExpression(sft, name, expression)
130         } else {
131           PropertyTransform(name, sft.getDescriptor(i).getType.getBinding, i)
132         }
133       } else {
134         val name = transform.substring(0, equals).trim()
135         val exp = transform.substring(equals + 1).trim()
136         val expression = try { ECQL.toExpression(exp) } catch {
137           case NonFatal(e) => throw new IllegalArgumentException(s"Unable to parse expression $transform:", e)
138         }
139         expression match {
140           case f: Function =>
141             ExpressionTransform(name, f.getFunctionName.getReturn.getType, f)
142 
143           case f: MathExpressionImpl =>
144             ExpressionTransform(name, classOf[java.lang.Double], f) // math ops always return doubles?
145 
146           case p: PropertyName =>
147             val orig = p.getPropertyName
148             val i = sft.indexOf(orig)
149             if (i == -1) {
150               attributeExpression(sft, name, p)
151             } else if (name == orig) {
152               PropertyTransform(name, sft.getDescriptor(i).getType.getBinding, i)
153             } else {
154               RenameTransform(name, sft.getDescriptor(i).getType.getBinding, orig, i)
155             }
156 
157           // TODO: Add support for LiteralExpressionImpl and/or ClassificationFunction?
158           case _ =>
159             throw new IllegalArgumentException(s"Unable to handle transform expression: $expression")
160         }
161       }
162     }
163 
164     /**
165      * This handles custom attribute accessors (e.g. json-path) and transforms that don't correspond
166      * to any attribute (e.g. in AttributeKeyPlusValueIterator)
167      *
168      * @param sft simple feature type
169      * @param name transform name
170      * @param e expression
171      * @return
172      */
173     private def attributeExpression(sft: SimpleFeatureType, name: String, e: String): ExpressionTransform = {
174       val expression = try { ECQL.toExpression(e) } catch {
175         case NonFatal(e) => throw new IllegalArgumentException(s"Unable to parse expression '$e':", e)
176       }
177       attributeExpression(sft, name, expression)
178     }
179 
180     /**
181      * This handles custom attribute accessors (e.g. json-path) and transforms that don't correspond
182      * to any attribute (e.g. in AttributeKeyPlusValueIterator)
183      *
184      * @param sft simple feature type
185      * @param name transform name
186      * @param expression expression
187      * @return
188      */
189     private def attributeExpression(
190         sft: SimpleFeatureType,
191         name: String,
192         expression: Expression): ExpressionTransform = {
193       val descriptor = expression.evaluate(sft).asInstanceOf[AttributeDescriptor]
194       val binding = if (descriptor == null) { classOf[String] } else { descriptor.getType.getBinding }
195       ExpressionTransform(name, binding, expression)
196     }
197 
198     /**
199      * Create the feature type corresponding a transform
200      *
201      * @param sft original simple feature type
202      * @param transforms transforms
203      * @return
204      */
205     def schema(sft: SimpleFeatureType, transforms: Seq[Transform]): SimpleFeatureType = {
206       val schema = new java.util.ArrayList[AttributeDescriptor]()
207       var geom: GeometryDescriptor = null
208 
209       lazy val typeBuilder = new AttributeTypeBuilder()
210 
211       transforms.foreach { t =>
212         val descriptor = t match {
213           case t: PropertyTransform =>
214             val d = sft.getDescriptor(t.i)
215             val im = SimpleFeatureTypes.immutable(d)
216             if (d == sft.getGeometryDescriptor) {
217               geom = im.asInstanceOf[GeometryDescriptor]
218             }
219             im
220 
221           case t: RenameTransform =>
222             sft.getDescriptor(t.i) match {
223               case d: GeometryDescriptor =>
224                 val im = new ImmutableGeometryDescriptor(d.getType, new NameImpl(t.name), d.getMinOccurs,
225                   d.getMaxOccurs, d.isNillable, d.getDefaultValue, d.getUserData)
226                 if (d == sft.getGeometryDescriptor) {
227                   geom = im
228                 }
229                 im
230               case d: AttributeDescriptor =>
231                 new ImmutableAttributeDescriptor(d.getType, new NameImpl(t.name), d.getMinOccurs, d.getMaxOccurs,
232                   d.isNillable, d.getDefaultValue, d.getUserData)
233             }
234 
235           case t: ExpressionTransform =>
236             typeBuilder.setBinding(t.binding)
237             if (classOf[Geometry].isAssignableFrom(t.binding)) {
238               typeBuilder.crs(CRS_EPSG_4326)
239               val typ = typeBuilder.buildGeometryType()
240               new ImmutableGeometryDescriptor(typ, new NameImpl(t.name), 0, 1, true, null, Collections.emptyMap())
241             } else {
242               val typ = typeBuilder.buildType()
243               new ImmutableAttributeDescriptor(typ, new NameImpl(t.name), 0, 1, true, null, Collections.emptyMap())
244             }
245         }
246 
247         schema.add(descriptor)
248       }
249 
250       var i = 0
251       while (geom == null && i < schema.size) {
252         schema.get(i) match {
253           case d: ImmutableGeometryDescriptor => geom = d
254           case _ => // no-op
255         }
256         i += 1
257       }
258 
259       // TODO reconsider default field user data?
260       new ImmutableSimpleFeatureType(sft.getName, schema, geom, sft.isAbstract, sft.getRestrictions, sft.getSuper,
261         sft.getDescription, sft.getUserData)
262     }
263   }
