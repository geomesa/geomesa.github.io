346   /**
347     * Creates an immutable copy of the simple feature type
348     *
349     * Note that immutable types will never be `equals` to a mutable type, due to specific class checking
350     * in the geotools implementation
351     *
352     * Note that some parts of the feature type may still be mutable - in particular AttributeType,
353     * GeometryType and SuperType are not used by geomesa so we don't bother with them. In addition,
354     * user data keys and values may be mutable objects, so while the user data map will not change,
355     * the values inside may
356     *
357     * @param sft simple feature type
358     * @param extraData additional user data to add to the simple feature type
359     * @return immutable copy of the simple feature type
360     */
361   def immutable(
362       sft: SimpleFeatureType,
363       extraData: java.util.Map[_ <: AnyRef, _ <: AnyRef] = null): SimpleFeatureType = {
364     sft match {
365       case immutable: ImmutableSimpleFeatureType if extraData == null || extraData.isEmpty => immutable
366       case _ =>
367         val schema = new java.util.ArrayList[AttributeDescriptor](sft.getAttributeCount)
368         var geom: GeometryDescriptor = null
369         var i = 0
370         while (i < sft.getAttributeCount) {
371           val descriptor = sft.getDescriptor(i)
372           val im = immutable(descriptor)
373           if (descriptor == sft.getGeometryDescriptor) {
374             geom = im.asInstanceOf[GeometryDescriptor]
375           }
376           schema.add(im)
377           i += 1
378         }
379         val userData = Option(extraData).filterNot(_.isEmpty).map { data =>
380           val map = new java.util.HashMap[AnyRef, AnyRef](data.size() + sft.getUserData.size)
381           map.putAll(sft.getUserData)
382           map.putAll(data)
383           map
384         }
385 
386         new ImmutableSimpleFeatureType(sft.getName, schema, geom, sft.isAbstract, sft.getRestrictions, sft.getSuper,
387           sft.getDescription, userData.getOrElse(sft.getUserData))
388     }
389   }

468   /**
469     * Compare two feature types. This method compares the the schemas of each feature type, i.e. the
470     * names and type bindings of the attributes. The result will be:
471     *
472     * <ul>
473     *   <li>a positive int if type A is a sub type or reorder of type B</li>
474     *   <li>zero if type A has the same schema as type B</li>
475     *   <li>a negative int if none of the above</li>
476     * </ul>
477     *
478     * Note that this shouldn't be used for sorting.
479     *
480     * Note: GeoTools DataUtilities has a similar method that does not work correctly
481     *
482     * @param a schema A
483     * @param b schema B
484     * @return comparison
485     */
486   def compare(a: SimpleFeatureType, b: SimpleFeatureType): Int = {
487     if (a.getAttributeCount > b.getAttributeCount) {
488       return -1 // not equals, subtype or reorder
489     }
490 
491     var exact = a.getAttributeCount == b.getAttributeCount
492 
493     var i = 0
494     while (i < a.getAttributeCount) {
495       val ad = a.getDescriptor(i)
496       val bd = b.getDescriptor(i)
497       if (ad.getLocalName == bd.getLocalName) {
498         if (ad.getType.getBinding != bd.getType.getBinding) {
499           if (bd.getType.getBinding.isAssignableFrom(ad.getType.getBinding)) {
500             exact = false // we've seen a subtyping
501           } else {
502             return -1 // not a subtype/reorder
503           }
504         }
505       } else {
506         val reorder = b.getDescriptor(ad.getLocalName)
507         if (reorder == null || !reorder.getType.getBinding.isAssignableFrom(ad.getType.getBinding)) {
508           return -1 // not a subtype/reorder
509         } else {
510           exact = false // we've seen a reordering
511         }
512       }
513       i += 1
514     }
515 
516     if (exact) { 0 } else {
517       1  // we haven't hit an absolute inequality, but we've found a reordering or subtype
518     }
519   }

521   /**
522    * Compares that index tables are the same format (key and value) between the two feature types
523    *
524    * @param a first type
525    * @param b second type
526    * @return 0 if indices are the same, non-zero otherwise

528   def compareIndexConfigs(a: SimpleFeatureType, b: SimpleFeatureType): Int = {
529     def compareAttributes(i: Int): Boolean = {
530       val (ad, bd) = (a.getDescriptor(i), b.getDescriptor(i))
531       ad.getType.getBinding == bd.getType.getBinding && AttributeChecks.forall(c => c(ad) == c(bd))
532     }
533     if (a.getAttributeCount == b.getAttributeCount &&
534         IndexChecks.forall(c => c(a) == c(b)) &&
535         Range(0, a.getAttributeCount).forall(compareAttributes)) {
536       0
537     } else {
538       1
539     }
540   }

542   private def createFeatureType(namespace: String,
543                                 name: String,
544                                 spec: SimpleFeatureSpec,
545                                 factory: Option[FeatureTypeFactory] = None): SimpleFeatureType = {
546     import AttributeOptions.OptDefault
547 
548     val defaultGeom = {
549       val geomAttributes = spec.attributes.collect { case g: GeomAttributeSpec => g }
550       geomAttributes.find(_.options.get(OptDefault).exists(_.toBoolean))
551           .orElse(geomAttributes.headOption)
552           .map(_.name)
553     }
554     val defaultDate = if (spec.options.get(IndexIgnoreDtg).exists(toBoolean)) { None } else {
555       val dateAttributes = spec.attributes.filter(a => classOf[Date].isAssignableFrom(a.clazz))
556       spec.options.get(DefaultDtgField).flatMap(dtg => dateAttributes.find(_.name == dtg))
557           .orElse(dateAttributes.find(_.options.get(OptDefault).exists(_.toBoolean)))
558           .orElse(dateAttributes.headOption)
559           .map(_.name)
560     }
561 
562     val b = factory.map(new SimpleFeatureTypeBuilder(_)).getOrElse(new SimpleFeatureTypeBuilder())
563     b.setNamespaceURI(namespace)
564     b.setName(name)
565     b.addAll(spec.attributes.map(_.toDescriptor).asJava)
566     defaultGeom.foreach(b.setDefaultGeometry)
567 
568     val sft = b.buildFeatureType()
569     sft.getUserData.putAll(spec.options.asJava)
570     defaultDate.foreach(sft.setDtgField)
571     sft
572   }
