14 /**
15  * Helper class that can extract attributes from non-standard expressions
16  *
17  * @param sft simple feature type
18  */
19 class AttributeExtractingVisitor(sft: SimpleFeatureType) extends FilterAttributeExtractor(sft) {
20   override def visit(expression: org.geotools.api.filter.expression.Function, data: AnyRef): AnyRef = {
21     expression match {
22       case e: AttributeAwareExpression => e.visit(this, data)
23       case _ => super.visit(expression, data)
24     }
25   }
