15 /**
16  * Resolves property names against the schema, but leaves `$`-prefixed JSON paths untouched so that
17  * IcebergFilterConverter can translate them into pushed-down predicates on nested struct fields.
18  *
19  * @param featureType feature type
20  */
21 class JsonPathPropertyNameResolver(featureType: SimpleFeatureType) extends PropertyNameResolvingVisitor(featureType) {
22   override def visit(expression: PropertyName, extraData: AnyRef): AnyRef = {
23     val name = expression.getPropertyName
24     if (name != null && name.startsWith("$")) {
25       getFactory(extraData).property(name)
26     } else {
27       super.visit(expression, extraData)
28     }
29   }
30 }
