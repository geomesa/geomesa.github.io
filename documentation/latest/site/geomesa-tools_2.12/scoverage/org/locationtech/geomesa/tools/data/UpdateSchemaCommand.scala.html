41   override def execute(): Unit = withDataStore(update)

43   protected def update(ds: DS): Unit = {
44     // ensure we have an operation
45     params.validate().foreach(e => throw e)
46 
47     val sft = try { ds.getSchema(params.featureName) } catch { case _: IOException => null }
48     if (sft == null) {
49       throw new ParameterException(s"Schema '${params.featureName}' does not exist in the data store")
50     }
51 
52     var n = 0
53     // numbering for our prompts
54     def number: Int = { n += 1; n }
55     val prompts = new StringBuilder()
56     var canRenameTables = false
57 
58     val builder = new SimpleFeatureTypeBuilder()
59     builder.init(sft)
60 
61     Option(params.rename).foreach { rename =>
62       rename.indexOf(':') match {
63         case -1 => builder.setName(rename)
64         case i  => builder.setNamespaceURI(rename.substring(0, i)); builder.setName(rename.substring(i + 1))
65       }
66       prompts.append(s"\n  $number: Renaming schema to '$rename'")
67       canRenameTables = true
68     }
69 
70     params.renameAttributes.asScala.grouped(2).foreach { case Seq(from: String, to: String) =>
71       val i = sft.indexOf(from)
72       if (i == -1) {
73         throw new ParameterException(s"Attribute '$from' does not exist in the schema")
74       }
75       val attribute = new AttributeTypeBuilder()
76       attribute.init(sft.getDescriptor(i))
77       builder.set(i, attribute.buildDescriptor(to))
78       if (sft.getGeomField == from) {
79         builder.setDefaultGeometry(to)
80       }
81       prompts.append(s"\n  $number: Renaming attribute '$from' to '$to'")
82       canRenameTables = canRenameTables || sft.getIndices.exists(_.attributes.contains(from))
83     }
84 
85     val renameTables =
86       canRenameTables && Option(params).collect { case p: RenameTablesParam => p.renameTables }.getOrElse(false)
87     if (renameTables) {
88       prompts.append(s"\n  $number: Renaming index tables (WARNING may be expensive)")
89     }
90 
91     params.attributes.asScala.foreach { attribute =>
92       val spec = try { SimpleFeatureSpecParser.parseAttribute(attribute) } catch {
93         case NonFatal(e) => throw new ParameterException(s"Invalid attribute spec: $attribute", e)
94       }
95       builder.add(spec.toDescriptor)
96       prompts.append(s"\n  $number: Adding attribute '${spec.name}' of type ${spec.clazz.getName}")
97     }
98 
99     val updated = builder.buildFeatureType()
100     updated.getUserData.putAll(sft.getUserData)
101 
102     if (!params.plusKeywords.isEmpty) {
103       val keywords = params.plusKeywords.asScala
104       updated.addKeywords(keywords.toSet)
105       prompts.append(s"\n  $number: Adding keywords: '${keywords.mkString("', '")}'")
106     }
107     if (!params.minusKeywords.isEmpty) {
108       val keywords = params.minusKeywords.asScala
109       updated.removeKeywords(keywords.toSet)
110       prompts.append(s"\n  $number: Removing keywords: '${keywords.mkString("', '")}'")
111     }
112     if (!params.userData.isEmpty) {
113       params.userData.asScala.foreach { case (k, v) =>
114         updated.getUserData.put(k, v) match {
115           case null => prompts.append(s"\n  $number: Adding user data: '$k=$v'")
116           case old  => prompts.append(s"\n  $number: Updating user data: '$k=$v' (was '$old')")
117         }
118       }
119     }
120 
121     if (renameTables) {
122       updated.getUserData.put(SimpleFeatureTypes.Configs.UpdateRenameTables, java.lang.Boolean.TRUE)
123     }
124     if (Option(params).collect { case p: NoBackupParam => p.noBackup }.getOrElse(false)) {
125       updated.getUserData.put(SimpleFeatureTypes.Configs.UpdateBackupMetadata, java.lang.Boolean.FALSE)
126     }
127 
128     storeSpecificUpdates(sft).foreach(message => prompts.append(s"\n  $number: $message"))
129 
130     Command.user.info(s"Preparing to update schema '${sft.getTypeName}':$prompts")
131     if (params.force || Prompt.confirm("Continue (y/n)? ")) {
132       Command.user.info("Updating, please wait...")
133       doUpdate(ds, sft.getTypeName, updated)
134       Command.user.info("Update complete")
135     }
136   }
137 
138   protected def storeSpecificUpdates(sft: SimpleFeatureType): Seq[String] = Seq.empty
139   protected def doUpdate(ds: DS, typeName: String, updated: SimpleFeatureType): Unit = ds.updateSchema(typeName, updated)
140 }
