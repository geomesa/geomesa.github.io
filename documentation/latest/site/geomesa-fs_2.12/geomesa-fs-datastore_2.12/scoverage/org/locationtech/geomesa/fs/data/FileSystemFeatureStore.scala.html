38   override def getWriterInternal(query: Query, flags: Int): FeatureWriter[SimpleFeatureType, SimpleFeature] = {
39     // note: check update first as sometimes we get ADD | UPDATE
40     if ((flags & WRITER_UPDATE) == WRITER_UPDATE) {
41       new FileSystemFeatureWriterModify(storage, query.getFilter, config.readThreads)
42     } else if ((flags & WRITER_ADD) == WRITER_ADD) {
43       new FileSystemFeatureWriterAppend(storage)
44     } else {
45       throw new IllegalArgumentException(s"Expected one of $WRITER_ADD or $WRITER_UPDATE, but got: $flags")
46     }
47   }

53   override def getCountInternal(query: Query): Int = {
54     val count = storage.getCount(query.getFilter, config.readThreads)

58   override def getReaderInternal(original: Query): FeatureReader[SimpleFeatureType, SimpleFeature] = {
59     import org.locationtech.geomesa.index.conf.QueryHints._
60 
61     val query = new Query(original)
62     // The type name can sometimes be empty such as Query.ALL
63     query.setTypeName(storage.sft.getTypeName)
64 
65     val reader = config.queryTimeout match {
66       case None => storage.getReader(query, threads = config.readThreads)
67       case Some(timeout) =>
68         val filter = Option(query.getFilter).filter(_ != Filter.INCLUDE)
69         new ManagedScan(new FileSystemScanner(storage, query, config.readThreads), Timeout(timeout), query.getTypeName, filter)
70     }
71 
72     // get a closeable java iterator that DelegateSimpleFeatureIterator will process correctly
73     val iter = new FileSystemFeatureIterator(reader)
74 
75     // transforms will be set after getting the iterator
76     val transformSft = query.getHints.getTransformSchema.getOrElse(storage.sft)
77 
78     // note: DelegateSimpleFeatureIterator will close the iterator by checking that it implements Closeable
79     new DelegateSimpleFeatureReader(transformSft, new DelegateSimpleFeatureIterator(iter))
80   }

82   /**
83    * Resolves filter property names against the schema, but preserves JSON paths.
84    *
85    * The base class runs `DataUtilities.resolvePropertyNames`, whose `PropertyNameResolvingVisitor`
86    * evaluates each property name against the feature type and rewrites it to the resolved attribute's
87    * local name. A JSON path like `$.props.name` evaluates (via GeoMesa's JSON property accessor) to
88    * the `props` descriptor, so the default would collapse the whole path to `props` - losing the nested
89    * field that IcebergFilterConverter needs to push down. We keep `$`-prefixed names verbatim and
90    * resolve everything else as usual.
91    *
92    * @param query the query being planned
93    * @return the query with non-JSON-path property names resolved
94    */
95   override protected def resolvePropertyNames(query: Query): Query = {
96     val filter = query.getFilter
97     if (filter == null || filter == Filter.INCLUDE || filter == Filter.EXCLUDE) { query } else {
98       val resolved = filter.accept(new JsonPathPropertyNameResolver(getSchema), null).asInstanceOf[Filter]
99       if (resolved == filter) { query } else {
100         val newQuery = new Query(query)
101         newQuery.setFilter(resolved)
102         newQuery
103       }
104     }
105   }
