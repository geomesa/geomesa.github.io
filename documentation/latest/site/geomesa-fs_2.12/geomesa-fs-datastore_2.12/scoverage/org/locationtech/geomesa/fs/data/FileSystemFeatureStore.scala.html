41   override def getWriterInternal(query: Query, flags: Int): FeatureWriter[SimpleFeatureType, SimpleFeature] = {
42     val transaction = storageTransaction()
43     // note: check update first as sometimes we get ADD | UPDATE
44     if ((flags & WRITER_UPDATE) == WRITER_UPDATE) {
45       new FileSystemFeatureWriterModify(storage, transaction, query.getFilter, config.readThreads)
46     } else if ((flags & WRITER_ADD) == WRITER_ADD) {
47       new FileSystemFeatureWriterAppend(storage, transaction)
48     } else {
49       throw new IllegalArgumentException(s"Expected one of $WRITER_ADD or $WRITER_UPDATE, but got: $flags")
50     }
51   }

53   /**
54    * Gets the storage transaction associated with the current geotools transaction
55    *
56    * @return storage transaction
57    */
58   private def storageTransaction(): StorageTransaction = {
59     val transaction = getTransaction
60     if (transaction == null || transaction == Transaction.AUTO_COMMIT) { StorageTransaction.AutoCommit } else {
61       // note: geotools tracks its own state using the content entry as a key, so we have to use something different
62       val key = TransactionKey(getDataStore)
63       val state = transaction.synchronized {
64         transaction.getState(key) match {
65           case null =>
66             val state = new FileSystemTransactionState()
67             transaction.putState(key, state)
68             state
69 
70           case state: FileSystemTransactionState => state
71           case state => throw new IllegalArgumentException(s"Found non-FileSystem state in transaction: $state")
72         }
73       }
74       state.transaction(storage)
75     }
76   }

111   /**
112    * Resolves filter property names against the schema, but preserves JSON paths.
113    *
114    * The base class runs `DataUtilities.resolvePropertyNames`, whose `PropertyNameResolvingVisitor`
115    * evaluates each property name against the feature type and rewrites it to the resolved attribute's
116    * local name. A JSON path like `$.props.name` evaluates (via GeoMesa's JSON property accessor) to
117    * the `props` descriptor, so the default would collapse the whole path to `props` - losing the nested
118    * field that IcebergFilterConverter needs to push down. We keep `$`-prefixed names verbatim and
119    * resolve everything else as usual.
120    *
121    * @param query the query being planned
122    * @return the query with non-JSON-path property names resolved
123    */
124   override protected def resolvePropertyNames(query: Query): Query = {
125     val filter = query.getFilter
126     if (filter == null || filter == Filter.INCLUDE || filter == Filter.EXCLUDE) { query } else {
127       val resolved = filter.accept(new JsonPathPropertyNameResolver(getSchema), null).asInstanceOf[Filter]
128       if (resolved == filter) { query } else {
129         val newQuery = new Query(query)
130         newQuery.setFilter(resolved)
131         newQuery
132       }
133     }
134   }

169   /**
170    * Ties storage transactions to a geotools transaction. A geotools transaction can be committed or rolled back multiple
171    * times, while a storage transaction is single-use - so a new storage transaction is started for each set of changes.
172    * Any uncommitted changes are rolled back when the geotools transaction is closed.
173    *
174    * A single state tracks all the feature types in a data store, as geotools hides the type of any failure if
175    * more than one state fails to commit.
176    */
177   private class FileSystemTransactionState extends Transaction.State {
178 
179     // the current storage transactions, if there are any pending changes, keyed by feature type name
180     private val current = scala.collection.mutable.LinkedHashMap.empty[String, TransactionInstance]
181 
182     /**
183      * Gets the current storage transaction, starting a new one if needed
184      *
185      * @param storage storage instance
186      * @return
187      */
188     def transaction(storage: FileSystemStorage): StorageTransaction = synchronized {
189       val typeName = storage.sft.getTypeName
190       current.get(typeName) match {
191         case None =>
192           val tx = storage.newTransaction()
193           current.put(typeName, TransactionInstance(storage, tx))
194           tx
195 
196         case Some(instance) if instance.storage.eq(storage) => instance.tx
197 
198         case _ =>
199           throw new IllegalStateException(
200             s"Schema '$typeName' was modified during the transaction - the transaction must be rolled back")
201       }
202     }
203 
204     override def setTransaction(transaction: Transaction): Unit = {
205       if (transaction == null) {
206         // the transaction has been closed
207         rollback()
208       }
209     }
210 
211     override def addAuthorization(authID: String): Unit = {}
212 
213     override def commit(): Unit = synchronized {
214       val failures = ArrayBuffer.empty[Throwable]
215       current.toList.foreach { case (typeName, instance) =>
216         // note: if the commit fails, the storage transaction is generally left open so it can be re-tried or rolled back
217         try { instance.tx.commit() } catch {
218           case NonFatal(e) => failures += e
219         } finally {
220           if (!instance.tx.isOpen) {
221             current.remove(typeName)
222           }
223         }
224       }
225       raise(failures.toSeq)
226     }
227 
228     override def rollback(): Unit = synchronized {
229       val failures = ArrayBuffer.empty[Throwable]
230       try {
231         current.values.foreach { instance =>
232           try { instance.tx.rollback() } catch {
233             case NonFatal(e) => failures += e
234           }
235         }
236       } finally {
237         current.clear()
238       }
239       raise(failures.toSeq)
240     }
241 
242     /**
243      * Throws a single exception for any failures. Conflict exceptions take priority, so that they can be
244      * handled by the caller - any other failures are added as suppressed exceptions.
245      *
246      * @param failures failures
247      */
248     private def raise(failures: Seq[Throwable]): Unit = {
249       if (failures.nonEmpty) {
250         val primary = failures.find(_.isInstanceOf[ConflictException]).getOrElse(failures.head)
251         val e = primary match {
252           case e: IOException => e
253           case e => new IOException(e.getMessage, e)
254         }
255         failures.filterNot(_.eq(primary)).foreach(e.addSuppressed)
256         throw e
257       }
258     }
259   }
