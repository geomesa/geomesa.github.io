20 /**

22  *
23  * @param ds datastore
24  */
25 class FileSystemStats(ds: FileSystemDataStore) extends UnoptimizedRunnableStats(ds) {
26 
27   override def getCount(
28       sft: SimpleFeatureType,
29       filter: Filter,
30       exact: Boolean,
31       queryHints: Hints): Option[Long] = {
32     if (!exact || filter == Filter.INCLUDE) {
33       Some(ds.storage(sft.getTypeName).getCount(filter, 1))
34     } else {
35       super.getCount(sft, filter, exact, queryHints)
36     }
37   }
38 
39   override def getMinMax[T](
40       sft: SimpleFeatureType,
41       attribute: String,
42       filter: Filter,
43       exact: Boolean): Option[MinMax[T]] = {
44     if (!exact || filter == Filter.INCLUDE) {
45       val (min, max) = ds.storage(sft.getTypeName).getBounds[T](attribute, filter, 1)
46       val minMax = Stat(sft, Stat.MinMax(attribute)).asInstanceOf[MinMax[T]]
47       val sf = new ScalaSimpleFeature(sft, "")
48       Seq(min, max).foreach { value =>
49         sf.setAttribute(attribute, value.asInstanceOf[AnyRef])
50         minMax.observe(sf)
51       }
52       Some(minMax)
53     } else {
54       super.getMinMax(sft, attribute, filter, exact)
55     }
56   }
57 }
