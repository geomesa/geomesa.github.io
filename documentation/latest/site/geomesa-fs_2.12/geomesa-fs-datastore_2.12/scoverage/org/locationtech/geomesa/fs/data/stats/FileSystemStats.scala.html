22 /**
23  * Optimized stats using per-file bounds for non-exact cases
24  *
25  * @param ds datastore
26  */
27 class FileSystemStats(ds: FileSystemDataStore) extends UnoptimizedRunnableStats(ds) {
28 
29   override def getCount(
30       sft: SimpleFeatureType,
31       filter: Filter,
32       exact: Boolean,
33       queryHints: Hints): Option[Long] = {
34     if (!exact || filter == Filter.INCLUDE) {
35       Some(ds.storage(sft.getTypeName).metadata.files().forFilter(filter).scan().map(_.recordCount()).sum)
36     } else {
37       super.getCount(sft, filter, exact, queryHints)
38     }
39   }
40 
41   override def getMinMax[T](
42       sft: SimpleFeatureType,
43       attribute: String,
44       filter: Filter,
45       exact: Boolean): Option[MinMax[T]] = {
46     if (!exact || filter == Filter.INCLUDE) {
47       val minMax = Stat(sft, Stat.MinMax(attribute)).asInstanceOf[MinMax[T]]
48       val storage = ds.storage(sft.getTypeName)
49       val sf = new ScalaSimpleFeature(sft, "")
50       val i = sft.indexOf(attribute)
51       val field = storage.schema.schema.findField(ColumnName.encode(attribute))
52       val fieldId = field.fieldId()
53       val fieldType = field.`type`()
54       ds.storage(sft.getTypeName).metadata.files().includeFileStats().forFilter(filter).scan().foreach { f =>
55         Seq(f.lowerBounds().get(fieldId), f.upperBounds().get(fieldId)).foreach { buffer =>
56           if (buffer != null) {
57             sf.setAttribute(i, Conversions.fromByteBuffer[AnyRef](fieldType, buffer))
58             minMax.observe(sf)
59           }
60         }
61       }
62       Some(minMax)
63     } else {
64       super.getMinMax(sft, attribute, filter, exact)
65     }
66   }
67 }
