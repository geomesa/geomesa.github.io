23  *
24  * @param ds datastore
25  */
26 class FileSystemStats(ds: FileSystemDataStore) extends UnoptimizedRunnableStats(ds) {
27 
28   import org.locationtech.geomesa.fs.storage.core.RichSimpleFeatureType
29 
30   override def getCount(
31       sft: SimpleFeatureType,
32       filter: Filter,
33       exact: Boolean,
34       queryHints: Hints): Option[Long] = {
35     if (!exact || filter == Filter.INCLUDE) {
36       Some(ds.storage(sft.getTypeName).metadata.getFiles(filter).map(_.count).sum)
37     } else {
38       super.getCount(sft, filter, exact, queryHints)
39     }
40   }
41 
42   override def getMinMax[T](
43       sft: SimpleFeatureType,
44       attribute: String,
45       filter: Filter,
46       exact: Boolean): Option[MinMax[T]] = {
47     val i = sft.indexOf(attribute)
48     if ((!exact || filter == Filter.INCLUDE) && sft.columnBounds().contains(i)) {
49       var min: String = null
50       var max: String = null
51       ds.storage(sft.getTypeName).metadata.getFiles(filter).foreach { file =>
52         file.bounds.filter(_.attribute == i).foreach { bounds =>
53           if (max == null || max < bounds.upper) {
54             max = bounds.upper
55           }
56           if (min == null || min > bounds.lower) {
57             min = bounds.lower
58           }
59         }
60       }
61       val minMax = new MinMax[T](sft, sft.getDescriptor(i).getLocalName)(MinMaxDefaults(sft.getDescriptor(i).getType.getBinding))
62       if (min != null) {
63         val sf = new ScalaSimpleFeature(sft, "")
64         ColumnBounds(i, min, max).decode(sft).productIterator.foreach { value =>
65           sf.setAttribute(i, value.asInstanceOf[AnyRef])
66           minMax.observe(sf)
67         }
68       }
69       Some(minMax)
70     } else {
71       super.getMinMax(sft, attribute, filter, exact)
72     }
73   }
