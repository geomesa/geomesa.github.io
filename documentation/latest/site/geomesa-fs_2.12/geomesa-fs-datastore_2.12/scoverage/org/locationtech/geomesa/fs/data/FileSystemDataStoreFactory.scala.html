34   override def createDataStore(params: java.util.Map[String, _]): DataStore = {
35     val xml = ConfigsParam.lookupOpt(params)
36     val resources = ConfigPathsParam.lookupOpt(params).toSeq.flatMap(_.split(',')).map(_.trim).filterNot(_.isEmpty)
37 
38     val conf = if (xml.isEmpty && resources.isEmpty) { FileSystemDataStoreFactory.configuration } else {
39       val conf = new Configuration(FileSystemDataStoreFactory.configuration)
40       // add the explicit props first, they may be needed for loading the path resources
41       xml.foreach(x => conf.addResource(new ByteArrayInputStream(x.getBytes(StandardCharsets.UTF_8))))
42       resources.foreach(HadoopUtils.addResource(conf, _))
43       conf
44     }
45 
46     val path = new Path(PathParam.lookup(params))
47 
48     // Need to do more tuning here. On a local system 1 thread (so basic producer/consumer) was best
49     // because Parquet is also threading the reads underneath I think. using prod/cons pattern was
50     // about 30% faster but increasing beyond 1 thread slowed things down. This could be due to the
51     // cost of serializing simple features though. need to investigate more.
52     //
53     // However, if you are doing lots of filtering it appears that bumping the threads up high
54     // can be very useful. Seems possibly numcores/2 might is a good setting (which is a standard idea)
55 
56     val readThreads = ReadThreadsParam.lookup(params)
57     val writeTimeout = WriteTimeoutParam.lookup(params)
58     val queryTimeout = QueryTimeoutParam.lookupOpt(params).filter(_.isFinite)
59 
60     AuthProviderParam.lookupOpt(params).foreach(p => conf.set(AuthsParam.key, p.getClass.getName))
61     AuthsParam.lookupOpt(params).foreach(conf.set(AuthsParam.key, _))
62 
63     val namespace = NamespaceParam.lookupOpt(params)
64 
65     val encoding = EncodingParam.lookup(params)
66     val storageFactory = FileSystemStorageFactory.factories.find(_.encoding == encoding).getOrElse {
67       // this shouldn't happen since encoding param lookup should fail if it doesn't match a factory
68       throw new RuntimeException(
69         s"Invalid encoding parameter, does not match a storage factory instance: $encoding\n" +
70           s"  Available factories: ${FileSystemStorageFactory.factories.map(_.encoding).mkString(", ")}")
71     }
72     val fs = FileSystem.get(path.toUri, conf)
73     val context = FileSystemContext(fs, conf, path, namespace)
74     val config = FileSystemDataStoreConfig(context, readThreads, writeTimeout, queryTimeout)
75 
76     val metadataConfig = MetadataConfigParam.lookupOpt(params).getOrElse(new Properties()).asScala.toMap
77     val metadataType =
78       if (ConverterStorageFactory.Encoding.equalsIgnoreCase(encoding) && !MetadataTypeParam.exists(params)) {
79         // for back compatibility, don't require metadata type if using converter storage
80         ConverterMetadata.MetadataType
81       } else {
82         MetadataTypeParam.lookup(params)
83       }
84     // note: closeable, ensure we create this last in case there is an error in the other params (or clean it up if there is an error)
85     val metadata = StorageMetadataCatalog(context, metadataType, metadataConfig)
86 
87     new FileSystemDataStore(storageFactory, metadata, config)
88   }
