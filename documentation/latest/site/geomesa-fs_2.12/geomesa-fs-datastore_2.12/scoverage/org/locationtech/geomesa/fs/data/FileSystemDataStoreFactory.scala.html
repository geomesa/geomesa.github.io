39   override def createDataStore(params: java.util.Map[String, _]): DataStore = {
40     // Need to do more tuning here. On a local system 1 thread (so basic producer/consumer) was best
41     // because Parquet is also threading the reads underneath I think. using prod/cons pattern was
42     // about 30% faster but increasing beyond 1 thread slowed things down. This could be due to the
43     // cost of serializing simple features though. need to investigate more.
44     //
45     // However, if you are doing lots of filtering it appears that bumping the threads up high
46     // can be very useful. Seems possibly numcores/2 might is a good setting (which is a standard idea)
47 
48     val readThreads = QueryThreadsParam.lookup(params)
49     val queryTimeout = QueryTimeoutParam.lookupOpt(params).filter(_.isFinite)
50 
51     val conf = buildConf(params)
52     val config = FileSystemDataStoreConfig(conf, readThreads, queryTimeout)
53 
54     lazy val encodingType =
55       if (params.get("fs.encoding") == "converter") {
56         logger.warn(s"Using deprecated parameter 'fs.encoding' - please switch to '${CatalogTypeParam.key}' instead")
57         true
58       } else {
59         false
60       }
61 
62     val catalogType = conf.getOrElse(CatalogUtil.ICEBERG_CATALOG_TYPE, null)
63     val catalog =
64       if (catalogType == ConverterCatalog.CatalogType || (catalogType == null && encodingType)) {
65         new ConverterCatalog(conf)
66       } else {
67         new IcebergCatalog(conf)
68       }
69 
70     new FileSystemDataStore(catalog, config)
71   }

88   private def buildConf(params: java.util.Map[String, _]): Map[String, String] = {
89     val builder = Map.newBuilder[String, String]
90     // pick up any hadoop props, to e.g. make it a bit easier to configure s3 access based on s3a settings
91     // only do this if hadoop file exist on the classpath, to avoid a hard runtime dependency on hadoop
92     if (Seq("core-site.xml", "hdfs-site.xml").exists(getClass.getClassLoader.getResource(_) != null)) {
93       FileSystemDataStoreFactory.configuration.forEach { e =>
94         val key = e.getKey
95         if (key.startsWith("geomesa.") || key.startsWith("fs.") || key.startsWith("parquet.")) {
96           builder += e.getKey -> FileSystemDataStoreFactory.configuration.get(e.getKey) // use .get to resolve envs
97         }
98       }
99     }
100     ConfigXmlParam.lookupOpt(params).foreach { xml =>
101       logger.warn(s"Parameter '${ConfigXmlParam.key}' is deprecated, please use '${ConfigParam.key}' instead")
102       val conf = new Configuration(false)
103       conf.addResource(new ByteArrayInputStream(xml.getBytes(StandardCharsets.UTF_8)))
104       // note: need to call iterator() to force loading of the resource
105       conf.iterator().forEachRemaining { e => builder += e.getKey -> conf.get(e.getKey) } // use .get to resolve envs
106     }
107     ConfigPathsParam.lookupOpt(params).map(_.split(',').map(_.trim).filterNot(_.isEmpty)).filterNot(_.isEmpty).foreach { resources =>
108       logger.warn(s"Parameter '${ConfigPathsParam.key}' is deprecated, please use '${ConfigFileParam.key}' instead")
109       val conf = new Configuration(false)
110       resources.foreach(HadoopUtils.addResource(conf, _))
111       // note: need to call iterator() to force loading of the resource
112       conf.iterator().forEachRemaining { e => builder += e.getKey -> conf.get(e.getKey) }  // use .get to resolve envs
113     }
114     ConfigFileParam.lookupOpt(params).foreach { f =>
115       var uri = new URI(f)
116       if (uri.getScheme == null) {
117         uri = new URI("file", uri.getHost, uri.getPath, uri.getFragment)
118       }
119       val asString = try { WithClose(uri.toURL.openStream())(IOUtils.toString(_, StandardCharsets.UTF_8)) } catch {
120         case NonFatal(e) =>
121           throw new IllegalArgumentException(s"Invalid parameter ${ConfigFileParam.key} - could not open file: $f", e)
122       }
123       // use our properties parameter parsing to evaluate env vars
124       ConfigParam.lookup(java.util.Map.of(ConfigParam.key, asString)).asScala.foreach { case (k, v) => builder += k -> v }
125     }
126     ConfigParam.lookupOpt(params).foreach(_.asScala.foreach { case (k, v) => builder += k -> v })
127     AuthProviderParam.lookupOpt(params).foreach(p => builder += (AuthsParam.key -> p.getClass.getName))
128     AuthsParam.lookupOpt(params).foreach(auths => builder += (AuthsParam.key -> auths))
129     CatalogTypeParam.lookupOpt(params).filter(_.nonEmpty).foreach(t => builder += (CatalogUtil.ICEBERG_CATALOG_TYPE -> t))
130     WritersMaxOpenPartitionsParam.lookupOpt(params).foreach(m => builder += (WritersMaxOpenPartitionsParam.key -> m.toString))
131     NamespaceParam.lookupOpt(params).foreach(n => builder += StorageCatalog.NamespaceConfigKey -> n)
132     builder.result()
133   }
134 }
