39   override def createDataStore(params: java.util.Map[String, _]): DataStore = {
40     // Need to do more tuning here. On a local system 1 thread (so basic producer/consumer) was best
41     // because Parquet is also threading the reads underneath I think. using prod/cons pattern was
42     // about 30% faster but increasing beyond 1 thread slowed things down. This could be due to the
43     // cost of serializing simple features though. need to investigate more.
44     //
45     // However, if you are doing lots of filtering it appears that bumping the threads up high
46     // can be very useful. Seems possibly numcores/2 might is a good setting (which is a standard idea)
47 
48     val readThreads = QueryThreadsParam.lookup(params)
49     val queryTimeout = QueryTimeoutParam.lookupOpt(params).filter(_.isFinite)
50     val metrics = MetricsRegistryParam.lookupRegistry(params)
51 
52     val conf = buildConf(params)
53     val config = FileSystemDataStoreConfig(conf, readThreads, queryTimeout, metrics)
54 
55     lazy val encodingType =
56       if (params.get("fs.encoding") == "converter") {
57         logger.warn(s"Using deprecated parameter 'fs.encoding' - please switch to '${CatalogTypeParam.key}' instead")
58         true
59       } else {
60         false
61       }
62 
63     val catalogType = conf.getOrElse(CatalogUtil.ICEBERG_CATALOG_TYPE, null)
64     val catalog =
65       if (catalogType == ConverterCatalog.CatalogType || (catalogType == null && encodingType)) {
66         new ConverterCatalog(conf)
67       } else {
68         new IcebergCatalog(conf)
69       }
70 
71     new FileSystemDataStore(catalog, config)
72   }

89   private def buildConf(params: java.util.Map[String, _]): Map[String, String] = {
90     val builder = Map.newBuilder[String, String]
91     // pick up any hadoop props, to e.g. make it a bit easier to configure s3 access based on s3a settings
92     // only do this if hadoop file exist on the classpath, to avoid a hard runtime dependency on hadoop
93     if (Seq("core-site.xml", "hdfs-site.xml").exists(getClass.getClassLoader.getResource(_) != null)) {
94       FileSystemDataStoreFactory.configuration.forEach { e =>
95         val key = e.getKey
96         if (key.startsWith("geomesa.") || key.startsWith("fs.") || key.startsWith("parquet.")) {
97           builder += e.getKey -> FileSystemDataStoreFactory.configuration.get(e.getKey) // use .get to resolve envs
98         }
99       }
100     }
101     ConfigXmlParam.lookupOpt(params).foreach { xml =>
102       logger.warn(s"Parameter '${ConfigXmlParam.key}' is deprecated, please use '${ConfigParam.key}' instead")
103       val conf = new Configuration(false)
104       conf.addResource(new ByteArrayInputStream(xml.getBytes(StandardCharsets.UTF_8)))
105       // note: need to call iterator() to force loading of the resource
106       conf.iterator().forEachRemaining { e => builder += e.getKey -> conf.get(e.getKey) } // use .get to resolve envs
107     }
108     ConfigPathsParam.lookupOpt(params).map(_.split(',').map(_.trim).filterNot(_.isEmpty)).filterNot(_.isEmpty).foreach { resources =>
109       logger.warn(s"Parameter '${ConfigPathsParam.key}' is deprecated, please use '${ConfigFileParam.key}' instead")
110       val conf = new Configuration(false)
111       resources.foreach(HadoopUtils.addResource(conf, _))
112       // note: need to call iterator() to force loading of the resource
113       conf.iterator().forEachRemaining { e => builder += e.getKey -> conf.get(e.getKey) }  // use .get to resolve envs
114     }
115     ConfigFileParam.lookupOpt(params).foreach { f =>
116       var uri = new URI(f)
117       if (uri.getScheme == null) {
118         uri = new URI("file", uri.getHost, uri.getPath, uri.getFragment)
119       }
120       val asString = try { WithClose(uri.toURL.openStream())(IOUtils.toString(_, StandardCharsets.UTF_8)) } catch {
121         case NonFatal(e) =>
122           throw new IllegalArgumentException(s"Invalid parameter ${ConfigFileParam.key} - could not open file: $f", e)
123       }
124       // use our properties parameter parsing to evaluate env vars
125       ConfigParam.lookup(java.util.Map.of(ConfigParam.key, asString)).asScala.foreach { case (k, v) => builder += k -> v }
126     }
127     ConfigParam.lookupOpt(params).foreach(_.asScala.foreach { case (k, v) => builder += k -> v })
128     AuthProviderParam.lookupOpt(params).foreach(p => builder += (AuthsParam.key -> p.getClass.getName))
129     AuthsParam.lookupOpt(params).foreach(auths => builder += (AuthsParam.key -> auths))
130     CatalogTypeParam.lookupOpt(params).filter(_.nonEmpty).foreach(t => builder += (CatalogUtil.ICEBERG_CATALOG_TYPE -> t))
131     WritersMaxOpenPartitionsParam.lookupOpt(params).foreach(m => builder += (WritersMaxOpenPartitionsParam.key -> m.toString))
132     NamespaceParam.lookupOpt(params).foreach(n => builder += StorageCatalog.NamespaceConfigKey -> n)
133     builder.result()
134   }
135 }
