30 /**
31  * File system data store
32  *
33  * @param catalog storage catalog
34  * @param config config
35  */
36 class FileSystemDataStore(catalog: StorageCatalog, config: FileSystemDataStoreConfig)
37     extends DecoratingDataStore(new FileSystemDelegateDataStore(catalog, config))
38       with HasGeoMesaStats with LazyLogging {
39 
40   // note: we use a decorating store as an indirection layer in order to override final methods in ContentDataStore
41 
42   override val stats: GeoMesaStats = new FileSystemStats(this)
43 
44   /**
45    * Get a handle on the underlying storage instance for a given simple feature type
46    *
47    * @param typeName simple feature type name
48    * @return
49    */
50   def storage(typeName: String): FileSystemStorage = delegate.asInstanceOf[FileSystemDelegateDataStore].storage(typeName)
51 
52   /**
53    * Run a schema update operation
54    *
55    * @param typeName simple feature type name
56    * @return
57    */
58   def update(typeName: String): CatalogUpdate = delegate.asInstanceOf[FileSystemDelegateDataStore].update(typeName)
59 
60   override def updateSchema(typeName: Name, featureType: SimpleFeatureType): Unit =
61     updateSchema(typeName.getLocalPart, featureType)
62 
63   override def updateSchema(typeName: String, featureType: SimpleFeatureType): Unit =
64     throw new UnsupportedOperationException("Use `update(typeName)` for schema changes")
65 
66   override def dispose(): Unit = try { super.dispose() } finally { stats.close() }
67 }

69 object FileSystemDataStore {
70 
71   /**
72    * Config options
73    *
74    * @param conf file system configuration props
75    * @param readThreads number of threads per read
76    * @param queryTimeout read timeout
77    */
78   case class FileSystemDataStoreConfig(
79     conf: Map[String, String],
80     readThreads: Int,
81     queryTimeout: Option[Duration],
82     metrics: Option[MetricsConfig],
83   )
84 
85   /**
86    * Base data store implementation
87    *
88    * @param catalog storage catalog
89    * @param config config
90    */
91   private class FileSystemDelegateDataStore(catalog: StorageCatalog, config: FileSystemDataStoreConfig)
92       extends ContentDataStore with LazyLogging {
93 
94     import org.locationtech.geomesa.fs.storage.core.RichSimpleFeatureType
95 
96     import scala.collection.JavaConverters._
97 
98     private val cache = Caffeine.newBuilder().build(
99       new CacheLoader[String, FileSystemStorage]() {
100         override def load(typeName: String): FileSystemStorage = catalog.load(typeName)
101       }
102     )
103 
104     private val registry = config.metrics.map(_.register())
105 
106     this.namespaceURI = config.conf.getOrElse(StorageCatalog.NamespaceConfigKey, null)
107 
108     override def createTypeNames(): java.util.List[Name] = {
109       val names = new java.util.ArrayList[Name]()
110       catalog.getTypeNames.foreach(name => names.add(new NameImpl(getNamespaceURI, name)))
111       names
112     }
113 
114     override def createSchema(original: SimpleFeatureType): Unit = {
115       if (catalog.getTypeNames.contains(original.getTypeName)) {
116         logger.warn(
117           s"Schema already exists: ${SimpleFeatureTypes.encodeType(cache.get(original.getTypeName).sft, includeUserData = true)}")
118       } else {
119         // copy the feature type so that we don't affect it when removing user data, below
120         val sft = SimpleFeatureTypes.copy(original)
121 
122         GeoMesaSchemaValidator.validate(sft)
123         // remove the configs in the user data as they're persisted separately
124         val scheme = sft.removeScheme().getOrElse {
125           throw new IllegalArgumentException("Partition scheme must be specified in the SimpleFeatureType user data")
126         }
127         val fileSize = sft.removeTargetFileSize()
128         val bloomFilters = sft.removeBloomFilters()
129         cache.put(sft.getTypeName, catalog.create(sft, scheme, fileSize, bloomFilters))
130       }
131     }
132 
133     override def removeSchema(typeName: Name): Unit = removeSchema(typeName.getLocalPart)
134     override def removeSchema(typeName: String): Unit = {
135       clearCaches(typeName)
136       catalog.delete(typeName, purge = true)
137     }
138 
139     override def createFeatureSource(entry: ContentEntry): ContentFeatureSource =
140       new FileSystemFeatureStore(storage(entry.getTypeName), entry, Query.ALL, config)
141 
142     /**
143      * Get a handle on the underlying storage instance for a given simple feature type
144      *
145      * @param typeName simple feature type name
146      * @return
147      */
148     def storage(typeName: String): FileSystemStorage = {
149       val storage = cache.get(typeName)
150       if (storage == null) {
151         throw new IllegalArgumentException(s"Schema '$typeName' doesn't exist or could not be loaded")
152       }
153       storage
154     }
155 
156     /**
157      * Run a schema update operation
158      *
159      * @param typeName simple feature type name
160      * @return
161      */
162     def update(typeName: String): CatalogUpdate = {
163       clearCaches(typeName)
164       catalog.update(typeName)
165     }
166 
167     override def dispose(): Unit = {
168       try {
169         CloseWithLogging(cache.asMap().asScala.values)
170         CloseWithLogging(Seq(catalog) ++ registry)
171         cache.invalidateAll()
172       } finally {
173         super.dispose()
174       }
175     }
176 
177     private def clearCaches(typeName: String): Unit = {
178       if (cache.asMap().containsKey(typeName)) {
179         val existing = cache.get(typeName)
180         cache.invalidate(typeName)
181         existing.close()
182       }
183       Seq(new NameImpl(typeName), new NameImpl(namespaceURI, typeName)).foreach { key =>
184         val entry = entries.remove(key)
185         if (entry != null) {
186           entry.dispose()
187         }
188       }
189     }
190   }
