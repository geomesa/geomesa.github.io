29 /**
30  * File system data store
31  *
32  * @param catalog storage catalog
33  * @param config config
34  */
35 class FileSystemDataStore(catalog: StorageCatalog, config: FileSystemDataStoreConfig)
36     extends ContentDataStore with HasGeoMesaStats with LazyLogging {
37 
38   import scala.collection.JavaConverters._
39 
40   private val cache = Caffeine.newBuilder().build(
41     new CacheLoader[String, FileSystemStorage]() {
42       override def load(typeName: String): FileSystemStorage = catalog.load(typeName)
43     }
44   )
45 
46   override val stats: GeoMesaStats = new FileSystemStats(this)
47 
48   private val registry = config.metrics.map(_.register())
49 
50   this.namespaceURI = config.conf.getOrElse(StorageCatalog.NamespaceConfigKey, null)
51 
52   override def createTypeNames(): java.util.List[Name] = {
53     val names = new java.util.ArrayList[Name]()
54     catalog.getTypeNames.foreach(name => names.add(new NameImpl(getNamespaceURI, name)))
55     names
56   }
57 
58   override def createSchema(original: SimpleFeatureType): Unit = {
59     import org.locationtech.geomesa.fs.storage.core.RichSimpleFeatureType
60     if (catalog.getTypeNames.contains(original.getTypeName)) {
61       logger.warn(
62         s"Schema already exists: ${SimpleFeatureTypes.encodeType(cache.get(original.getTypeName).sft, includeUserData = true)}")
63     } else {
64       // copy the feature type so that we don't affect it when removing user data, below
65       val sft = SimpleFeatureTypes.copy(original)
66 
67       GeoMesaSchemaValidator.validate(sft)
68       // remove the configs in the user data as they're persisted separately
69       val scheme = sft.removeScheme().getOrElse {
70         throw new IllegalArgumentException("Partition scheme must be specified in the SimpleFeatureType user data")
71       }
72       val fileSize = sft.removeTargetFileSize()
73       val bloomFilters = sft.removeBloomFilters()
74       cache.put(sft.getTypeName, catalog.create(sft, scheme, fileSize, bloomFilters))
75     }
76   }
77 
78   override def createFeatureSource(entry: ContentEntry): ContentFeatureSource =
79     new FileSystemFeatureStore(storage(entry.getTypeName), entry, Query.ALL, config)
80 
81   /**
82    * Get a handle on the underlying storage instance for a given simple feature type
83    *
84    * @param typeName simple feature type name
85    * @return
86    */
87   def storage(typeName: String): FileSystemStorage = {
88     val storage = cache.get(typeName)
89     if (storage == null) {
90       throw new IllegalArgumentException(s"Schema '$typeName' doesn't exist or could not be loaded")
91     }
92     storage
93   }
94 
95   override def dispose(): Unit = {
96     try {
97       CloseWithLogging(cache.asMap().asScala.values)
98       CloseWithLogging(Seq(catalog) ++ registry)
99       cache.invalidateAll()
100     } finally {
101       super.dispose()
102     }
103   }
104 }
