28 /**
29  * File system data store
30  *
31  * @param catalog storage catalog
32  * @param config config
33  */
34 class FileSystemDataStore(catalog: StorageCatalog, config: FileSystemDataStoreConfig)
35     extends ContentDataStore with HasGeoMesaStats with LazyLogging {
36 
37   import scala.collection.JavaConverters._
38 
39   private val cache = Caffeine.newBuilder().build(
40     new CacheLoader[String, FileSystemStorage]() {
41       override def load(typeName: String): FileSystemStorage = catalog.load(typeName)
42     }
43   )
44 
45   override val stats: GeoMesaStats = new FileSystemStats(this)
46 
47   this.namespaceURI = config.conf.getOrElse(StorageCatalog.NamespaceConfigKey, null)
48 
49   override def createTypeNames(): java.util.List[Name] = {
50     val names = new java.util.ArrayList[Name]()
51     catalog.getTypeNames.foreach(name => names.add(new NameImpl(getNamespaceURI, name)))
52     names
53   }
54 
55   override def createSchema(original: SimpleFeatureType): Unit = {
56     import org.locationtech.geomesa.fs.storage.core.RichSimpleFeatureType
57     if (catalog.getTypeNames.contains(original.getTypeName)) {
58       logger.warn(
59         s"Schema already exists: ${SimpleFeatureTypes.encodeType(cache.get(original.getTypeName).sft, includeUserData = true)}")
60     } else {
61       // copy the feature type so that we don't affect it when removing user data, below
62       val sft = SimpleFeatureTypes.copy(original)
63 
64       GeoMesaSchemaValidator.validate(sft)
65       // remove the configs in the user data as they're persisted separately
66       val scheme = sft.removeScheme().getOrElse {
67         throw new IllegalArgumentException("Partition scheme must be specified in the SimpleFeatureType user data")
68       }
69       val fileSize = sft.removeTargetFileSize()
70       cache.put(sft.getTypeName, catalog.create(sft, scheme, fileSize))
71     }
72   }
73 
74   override def createFeatureSource(entry: ContentEntry): ContentFeatureSource =
75     new FileSystemFeatureStore(storage(entry.getTypeName), entry, Query.ALL, config)
76 
77   /**
78    * Get a handle on the underlying storage instance for a given simple feature type
79    *
80    * @param typeName simple feature type name
81    * @return
82    */
83   def storage(typeName: String): FileSystemStorage = {
84     val storage = cache.get(typeName)
85     if (storage == null) {
86       throw new IllegalArgumentException(s"Schema '$typeName' doesn't exist or could not be loaded")
87     }
88     storage
89   }
90 
91   override def dispose(): Unit = {
92     try {
93       CloseWithLogging(cache.asMap().asScala.values)
94       CloseWithLogging(catalog)
95       cache.invalidateAll()
96     } finally {
97       super.dispose()
98     }
99   }
100 }
