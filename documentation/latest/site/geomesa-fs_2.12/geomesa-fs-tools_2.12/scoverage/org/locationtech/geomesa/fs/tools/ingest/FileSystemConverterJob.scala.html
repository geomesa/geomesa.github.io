39 abstract class FileSystemConverterJob(
40     dsParams: Map[String, String],
41     sft: SimpleFeatureType,
42     converterConfig: Config,
43     paths: Seq[String],
44     libjarsFiles: Seq[String],
45     libjarsPaths: Iterator[() => Seq[File]],
46     reducers: Int,
47     root: URI,
48     schemes: Seq[PartitionScheme],
49     tmpPath: Option[Path],
50     targetFileSize: Option[Long]
51   ) extends ConverterIngestJob(dsParams, sft, converterConfig, paths, libjarsFiles, libjarsPaths)
52     with StorageConfiguration with LazyLogging {
53 
54   override protected def reduceCounters(job: Job): Seq[(String, Long)] =
55     Seq((IngestCounters.Persisted, job.getCounters.findCounter(OutputCounters.Group, "reduced").getValue))
56 
57   override def configureJob(job: Job): Unit = {
58     super.configureJob(job)
59 
60     job.setMapperClass(classOf[FsIngestMapper])
61     job.setMapOutputKeyClass(classOf[Text])
62     job.setMapOutputValueClass(classOf[BytesWritable])
63     job.setOutputKeyClass(classOf[Void])
64     job.setOutputValueClass(classOf[SimpleFeature])
65     job.setReducerClass(classOf[DummyReducer])
66     job.setNumReduceTasks(reducers)
67     // Ensure that the reducers don't start too early
68     // (default is at 0.05 which takes all the map slots and isn't needed)
69     job.getConfiguration.set("mapreduce.job.reduce.slowstart.completedmaps", ".90")
70 
71     StorageConfiguration.setRootPath(job.getConfiguration, root)
72     StorageConfiguration.setPartitionScheme(job.getConfiguration, schemes)
73     targetFileSize.foreach(StorageConfiguration.setTargetFileSize(job.getConfiguration, _))
74 
75     FileOutputFormat.setOutputPath(job, tmpPath.getOrElse(new Path(S3ObjectStore.s3aUri(root))))
76 
77     configureOutput(sft, job)
78   }
79 
80   override def await(reporter: StatusCallback): JobResult = {
81     super.await(reporter).merge {
82       tmpPath.map { tp =>
83         reporter.reset()
84         new DistributedCopy().copy(Seq(tp), new Path(S3ObjectStore.s3aUri(root)), reporter) match {
85           case JobSuccess(message, counts) =>
86             Command.user.info(message)
87             JobSuccess("", counts)
88 
89           case j => j
90         }
91       }
92     }
93   }
94 }

114   class FsIngestMapper extends Mapper[LongWritable, SimpleFeature, Text, BytesWritable] with LazyLogging {
115 
116     type Context = Mapper[LongWritable, SimpleFeature, Text, BytesWritable]#Context
117 
118     private var serializer: KryoFeatureSerializer = _
119     private var scheme: Seq[PartitionScheme] = _
120 
121     var mapped: Counter = _
122     var written: Counter = _
123     var failed: Counter = _
124 
125     override def setup(context: Context): Unit = {
126       val sft = StorageConfiguration.getSft(context.getConfiguration)
127       serializer = KryoFeatureSerializer(sft, SerializationOption.defaults)
128       scheme = StorageConfiguration.getPartitionScheme(context.getConfiguration, sft)
129 
130       mapped = context.getCounter(OutputCounters.Group, "mapped")
131       written = context.getCounter(OutputCounters.Group, OutputCounters.Written)
132       failed = context.getCounter(OutputCounters.Group, OutputCounters.Failed)
133     }
134 
135     override def map(key: LongWritable, sf: SimpleFeature, context: Context): Unit = {
136       // partitionKey is important because this needs to be correct for the parquet file
137       try {
138         mapped.increment(1)
139         val sfWithFid = GeoMesaFeatureWriter.featureWithFid(sf)
140         val partitionKey = new Text(Partition(scheme.map(_.getPartition(sfWithFid))).toString)
141         context.write(partitionKey, new BytesWritable(serializer.serialize(sfWithFid)))
142         written.increment(1)
143       } catch {
144         case e: Throwable =>
145           logger.error(s"Failed to write '${DataUtilities.encodeFeature(sf)}'", e)
146           failed.increment(1)
147       }
148     }
149   }
150

164     }
165 
166     override def reduce(key: Text, values: java.lang.Iterable[BytesWritable], context: Context): Unit = {
167       values.asScala.foreach { bw =>
168         val sf = serializer.deserialize(bw.getBytes)
169         context.write(null, sf)
170         reduced.increment(1)
171       }
172     }
