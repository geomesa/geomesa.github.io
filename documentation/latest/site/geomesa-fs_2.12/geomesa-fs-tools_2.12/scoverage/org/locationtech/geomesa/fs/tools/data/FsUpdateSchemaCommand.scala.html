34   override protected def doUpdate(ds: FileSystemDataStore, typeName: String, updated: SimpleFeatureType): Unit = {
35     // parse the new attributes up front, so that we don't start an update if the parameters are invalid
36     val descriptors = params.attributes.asScala.map { attribute =>
37       try { SimpleFeatureSpecParser.parseAttribute(attribute).toDescriptor } catch {
38         case NonFatal(e) => throw new ParameterException(s"Invalid attribute spec: $attribute", e)
39       }
40     }
41     // stage all the changes, so that they're validated up front and then applied in a single transaction
42     val update = ds.update(typeName)
43     params.renameAttributes.asScala.grouped(2).foreach { case Seq(from: String, to: String) =>
44       update.columns().rename(from, to)
45     }
46     descriptors.foreach(update.columns().add)
47     if (params.scheme != null) {
48       update.table().setPartitions(params.scheme.split(",").toSeq)
49     }
50     if (params.targetFileSize != null) {
51       // note: Memory.bytes parsing was validated earlier
52       update.metadata().setTargetFileSize(Some(Memory.bytes(params.targetFileSize).get))
53     }
54     if (!params.plusKeywords.isEmpty || !params.minusKeywords.isEmpty) {
55       val keywords = Option(updated.getUserData.get(Configs.Keywords)).map(_.toString).filterNot(_.isBlank)
56       update.metadata().setUserData(Configs.Keywords, keywords)
57     }
58     params.userData.asScala.foreach { case (k, v) =>
59       update.metadata().setUserData(k, Option(v))
60     }
61     Option(params.rename).foreach { rename =>
62       update.table().rename(rename)
63     }
64     update.apply()
65   }
66 
67   override protected def storeSpecificUpdates(sft: SimpleFeatureType): Seq[String] = {
68     val scheme = Option(params.scheme).map { s =>
69       sft.setScheme(s)
70       s"Changing partition scheme to '$s'"
71     }
72     val fileSize = Option(params.targetFileSize).map { s =>
73       sft.setTargetFileSize(s)
74       s"Setting target file size to '$s'"
75     }
76     scheme.toSeq ++ fileSize
77   }
78 }
