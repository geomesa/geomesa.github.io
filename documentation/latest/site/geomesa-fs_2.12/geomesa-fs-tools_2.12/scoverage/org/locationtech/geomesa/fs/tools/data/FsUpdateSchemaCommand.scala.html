34   override protected def doUpdate(ds: FileSystemDataStore, typeName: String, updated: SimpleFeatureType): Unit = {
35     params.renameAttributes.asScala.grouped(2).foreach { case Seq(from: String, to: String) =>
36       ds.update(typeName).columns().rename(from, to)
37     }
38     params.attributes.asScala.foreach { attribute =>
39       val descriptor = try { SimpleFeatureSpecParser.parseAttribute(attribute).toDescriptor } catch {
40         case NonFatal(e) => throw new ParameterException(s"Invalid attribute spec: $attribute", e)
41       }
42       ds.update(typeName).columns().add(descriptor)
43     }
44     if (params.scheme != null) {
45       ds.update(typeName).table().setPartitions(params.scheme.split(",").toSeq)
46     }
47     if (params.targetFileSize != null) {
48       // note: Memory.bytes parsing was validated earlier
49       ds.update(typeName).metadata().setTargetFileSize(Some(Memory.bytes(params.targetFileSize).get))
50     }
51     if (!params.plusKeywords.isEmpty || !params.minusKeywords.isEmpty) {
52       val keywords = Option(updated.getUserData.get(Configs.Keywords)).map(_.toString).filterNot(_.isBlank)
53       ds.update(typeName).metadata().setUserData(Configs.Keywords, keywords)
54     }
55     params.userData.asScala.foreach { case (k, v) =>
56       ds.update(typeName).metadata().setUserData(k, Option(v))
57     }
58     // do this last so type name change doesn't affect the other ops
59     Option(params.rename).foreach { rename =>
60       ds.update(typeName).table().rename(rename)
61     }
62   }
63 
64   override protected def storeSpecificUpdates(sft: SimpleFeatureType): Seq[String] = {
65     val scheme = Option(params.scheme).map { s =>
66       sft.setScheme(s)
67       s"Changing partition scheme to '$s'"
68     }
69     val fileSize = Option(params.targetFileSize).map { s =>
70       sft.setTargetFileSize(s)
71       s"Setting target file size to '$s'"
72     }
73     scheme.toSeq ++ fileSize
74   }
75 }
