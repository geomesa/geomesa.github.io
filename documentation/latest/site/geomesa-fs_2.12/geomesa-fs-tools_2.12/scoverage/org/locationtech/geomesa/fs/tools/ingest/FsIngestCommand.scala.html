35 class FsIngestCommand extends IngestCommand[FileSystemDataStore] with FsDistributedCommand {
36 
37   import scala.collection.JavaConverters._
38 
39   override val params = new FsIngestParams
40 
41   override protected def setBackendSpecificOptions(sft: SimpleFeatureType): Unit =
42     FsCreateSchemaCommand.setOptions(sft, params)
43 
44   override protected def startIngest(
45       mode: RunMode,
46       ds: FileSystemDataStore,
47       sft: SimpleFeatureType,
48       converter: Config,
49       inputs: Inputs): Awaitable = {
50     mode match {
51       case RunModes.Local =>
52         super.startIngest(mode, ds, sft, converter, inputs)
53 
54       case RunModes.Distributed =>
55         Command.user.info(s"Running ingestion in distributed ${if (params.combineInputs) "combine " else "" }mode")
56         val reducers = Option(params.reducers).filter(_ > 0).getOrElse {
57           throw new ParameterException("Please specify --num-reducers for distributed ingest")
58         }
59         val storage = ds.storage(sft.getTypeName)
60         val tmpPath = Option(params.tempPath).map(new URI(_))
61         val targetFileSize = storage.metadata.get(Metadata.TargetFileSize).map(_.toLong)
62 
63         tmpPath.foreach { tp =>
64           WithClose(ObjectStore(FileSystemContext.create(tp, storage.context.conf))) { fs =>
65             if (fs.exists(tp)) {
66               Command.user.info(s"Deleting temp path $tp")
67               val toCheck = new java.util.LinkedList[URI]()
68               toCheck.addAll(fs.list(tp).toList.asJava)
69               while (!toCheck.isEmpty) {
70                 val head = toCheck.remove()
71                 if (head.toString.endsWith("/")) {
72                   toCheck.addAll(fs.list(head).toList.asJava)
73                 } else {
74                   fs.delete(head)
75                 }
76               }
77             }
78           }
79         }
80 
81         new ParquetConverterJob(
82           connection, sft, converter, inputs.paths, libjarsFiles, libjarsPaths, reducers,
83           storage.context.root, storage.metadata.schemes, tmpPath.map(new Path(_)), targetFileSize) {
84           override def configureJob(job: Job): Unit = {
85             super.configureJob(job)
86             if (params.combineInputs) {
87               job.setInputFormatClass(classOf[ConverterCombineInputFormat])
88               Option(params.maxSplitSize).foreach(s => FileInputFormat.setMaxInputSplitSize(job, s.toLong))
89             }
90           }
91         }
92 
93       case _ =>
94         throw new UnsupportedOperationException(s"Missing implementation for mode $mode")
95     }
96   }
97 }
