37 class FsRegisterIcebergCommand extends FsDataStoreCommand with LazyLogging {
38 
39   import scala.collection.JavaConverters._
40 
41   override val params = new FsRegisterIcebergParams()
42 
43   override val name: String = "register-iceberg-files"
44 
45   override def execute(): Unit = {
46     if (params.icebergConfigFile == null && params.icebergConfig.isEmpty) {
47       throw new ParameterException("At least one of --iceberg-config or --iceberg-config-file must be specified")
48     }
49 
50     val icebergProps = new Properties()
51     if (params.icebergConfigFile != null) {
52       WithClose(new FileReader(params.icebergConfigFile, StandardCharsets.UTF_8))(icebergProps.load)
53     }
54     params.icebergConfig.asScala.foreach { case (k, v) => icebergProps.put(k, v) }
55     if (!icebergProps.containsKey("catalog-impl")) {
56       throw new ParameterException("Iceberg properties must specify a 'catalog-impl'")
57     }
58 
59     val catalog = try {
60       Class.forName(icebergProps.getProperty("catalog-impl")).getConstructor().newInstance().asInstanceOf[Catalog]
61     } catch {
62       case e: Throwable => throw new ParameterException(s"Could not instantiate catalog class '$name':", e)
63     }
64 
65     catalog.initialize("geomesa", Collections.unmodifiableMap(icebergProps.asInstanceOf[java.util.Map[String, String]]))
66 
67     withDataStore(execute(_, catalog, icebergProps.asInstanceOf[java.util.Map[String, String]]))
68   }
69 
70   private def execute(ds: FileSystemDataStore, catalog: Catalog, icebergProps: java.util.Map[String, String]): Unit = {
71     val storage = try { ds.storage(params.featureName) } catch {
72       case NonFatal(e) => throw new IllegalArgumentException(s"No schema exists with type name ${params.featureName}", e)
73     }
74     if (storage.encoding != ParquetFileSystemStorage.Encoding) {
75       throw new UnsupportedOperationException(s"Iceberg is only implemented for Parquet storage: found ${storage.encoding}")
76     }
77 
78     params.partitions.asScala.foreach { p =>
79       storage.metadata.schemes.foreach { s =>
80         if (!p.values.exists(_.name == s.name)) {
81           throw new IllegalArgumentException(
82             s"Specified partition '$p' does not include all required scheme(s): " +
83               storage.metadata.schemes.map(_.name).toSeq.sorted.mkString(", "))
84         }
85       }
86     }
87 
88     val iceberg = new IcebergMapper(storage)
89     logger.debug(s"Iceberg schema: ${SchemaParser.toJson(iceberg.schema)}")
90 
91     val namespace = Namespace.of(params.icebergNamespace)
92     // TODO valid identifiers vary based on the catalog... this is for glue and not comprehensive
93     val tableId = TableIdentifier.of(namespace, storage.metadata.sft.getTypeName.toLowerCase(Locale.US).replaceAll("[^a-z0-9]+", "_"))
94     val table = if (catalog.tableExists(tableId)) {
95       Command.user.info("Found existing table")
96       catalog.loadTable(tableId)
97       // TODO compare with existing schema and partitioning
98     } else {
99       Command.user.info("Creating new iceberg table")
100       val warehouse = icebergProps.get("warehouse")
101       if (warehouse == null) {
102         throw new IllegalArgumentException("Iceberg properties must specify a 'warehouse' location")
103       }
104       // file format v3 lets us use native geometries - but it's not yet supported in spark or trino
105       // icebergProps.put(TableProperties.FORMAT_VERSION, "3")
106       catalog.createTable(tableId, iceberg.schema, iceberg.spec, warehouse, icebergProps)
107     }
108     logger.debug(s"Iceberg table: $table")
109 
110     val existingFiles = if (params.allowDuplicates) { Set.empty[String] } else {
111       val builder = Set.newBuilder[String]
112       // TODO we could probably limit this to just files that match our partition with a filter
113       WithClose(table.newScan().planFiles()) { tasks =>
114         tasks.asScala.foreach(task => builder += task.file().location())
115       }
116       builder.result()
117     }
118 
119     val files = params.partitions.asScala.flatMap { p =>
120       storage.metadata.getFiles(p).flatMap { f =>
121         val df = iceberg.toDataFile(table, f)
122         if (existingFiles.contains(df.location())) {
123           Command.user.info(s"Skipping already registered file: ${df.location()}")
124           None
125         } else {
126           Command.user.info(s"Registering file: ${df.location()}")
127           Some(df)
128         }
129       }
130     }
131 
132     logger.debug(s"Files:\n  ${files.mkString("\n  ")}")
133 
134     if (files.nonEmpty) {
135       Command.user.info("Updating table")
136       val append = table.newAppend()
137       files.foreach(append.appendFile)
138       append.commit()
139     }
140     Command.user.info("Complete")
141   }
142 }
