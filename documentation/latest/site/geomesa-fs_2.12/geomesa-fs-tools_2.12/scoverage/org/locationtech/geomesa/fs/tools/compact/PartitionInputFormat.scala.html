26 /**
27   * An Input format that creates splits based on FSDS Partitions. This is used for compaction, when we want a single
28   * split per partition. Otherwise, use OrcSimpleFeatureInputFormat/ParquetSimpleFeatureInputFormat as those are
29   * more efficient
30   */
31 class PartitionInputFormat extends InputFormat[Void, SimpleFeature] {
32 
33   override def getSplits(context: JobContext): java.util.List[InputSplit] = {
34     val hadoopConf = context.getConfiguration
35     val conf = {
36       val builder = Map.newBuilder[String, String]
37       hadoopConf.forEach(e => builder += e.getKey -> hadoopConf.get(e.getKey)) // use .get to resolve envs
38       builder.result()
39     }
40 
41     val root = StorageConfiguration.getRootPath(hadoopConf)
42     val fileSize = StorageConfiguration.getTargetFileSize(hadoopConf)
43     val typeName = StorageConfiguration.getSftName(hadoopConf)
44     val encoding = StorageConfiguration.getEncoding(hadoopConf)
45 
46     val fsc = FileSystemContext.create(root, conf)
47     val catalog = StorageMetadataCatalog(fsc)
48     val factory = FileSystemStorageFactory(encoding)
49     WithClose(factory.apply(fsc, catalog.load(typeName))) { storage =>
50       val sizeCheck = fileSize.orElse(storage.sizer.targetSize).map(t => (p: URI) => storage.sizer.fileIsSized(p, t))
51       val splits = StorageConfiguration.getPartitions(hadoopConf).map { partition =>
52         var size = 0L
53         val files = storage.metadata.getFiles(partition).filter { f =>
54           if (sizeCheck.exists(_.apply(fsc.root.resolve(f.file)))) { false } else {
55             size += storage.fs.size(fsc.root.resolve(f.file))
56             true
57           }
58         }
59         new PartitionInputSplit(partition.toString, files, size)
60       }
61       java.util.Arrays.asList(splits: _*)
62     }
63   }
64 
65   override def createRecordReader(split: InputSplit, context: TaskAttemptContext): RecordReader[Void, SimpleFeature] = {
66     val psplit = split.asInstanceOf[PartitionInputSplit]
67     new PartitionRecordReader(psplit.getFiles)
68   }
69 }

73   /**
74     * InputSplit corresponding to a single FileSystemDataStore PartitionScheme partition
75     */
76   class PartitionInputSplit extends InputSplit with Writable {
77 
78     private var name: String = _
79     private var files: Seq[StorageFile] = _
80     private var length: java.lang.Long = _
81 
82     def this(name: String, files: Seq[StorageFile], length: Long) = {
83       this()
84       this.name = name
85       this.files = files
86       this.length = length
87     }
88 
89     /**
90       * @return the name of this partition
91       */
92     def getName: String = name
93 
94     def getFiles: Seq[StorageFile] = files
95 
96     override def getLength: Long = length
97 
98     // TODO attempt to optimize the locations where this should run in the case of HDFS
99     // With S3 this won't really matter
100     override def getLocations: Array[String] = Array.empty[String]
101 
102     override def write(out: DataOutput): Unit = {
103       out.writeUTF(name)
104       out.writeLong(length)
105       out.writeInt(files.length)
106       // note: we don't store bounds, sort, etc as they're not used
107       files.foreach { file =>
108         out.writeUTF(file.file)
109         out.writeUTF(file.partition.toString)
110         out.writeLong(file.count)
111         out.writeUTF(file.action.toString)
112         out.writeLong(file.timestamp)
113       }
114     }
115 
116     override def readFields(in: DataInput): Unit = {
117       this.name = in.readUTF()
118       this.length = in.readLong()
119       this.files = Seq.fill(in.readInt) {
120         StorageFile(in.readUTF(), Partition(in.readUTF()), in.readLong, StorageFileAction.withName(in.readUTF()), Seq.empty, Seq.empty, in.readLong())
121       }
122     }
123   }

125   class PartitionRecordReader(files: Seq[StorageFile]) extends RecordReader[Void, SimpleFeature] {
126 
127     private var storage: FileSystemStorage = _
128     private var reader: CloseableFeatureIterator = _
129 
130     private var curValue: SimpleFeature = _
131 
132     override def initialize(split: InputSplit, context: TaskAttemptContext): Unit = {
133       val hadoopConf = context.getConfiguration
134       val conf = {
135         val builder = Map.newBuilder[String, String]
136         hadoopConf.forEach(e => builder += e.getKey -> e.getValue)
137         builder.result()
138       }
139       val root = StorageConfiguration.getRootPath(hadoopConf)
140       val fsc = FileSystemContext.create(root, conf)
141       val sft = StorageConfiguration.getSft(hadoopConf)
142       val encoding = StorageConfiguration.getEncoding(hadoopConf)
143 
144       // use a cached metadata impl instead of reloading
145       val cached = new StaticMetadata(sft, files)
146       storage = FileSystemStorageFactory(encoding).apply(fsc, cached)
147       reader = storage.getReader(new Query("", Filter.INCLUDE))
148     }
149 
150     // TODO look at how the ParquetInputFormat provides progress and utilize something similar
151     override def getProgress: Float = 0.0f
152 
153     override def nextKeyValue(): Boolean = {
154       if (reader.hasNext) {
155         curValue = reader.next()
156         true
157       } else {
158         curValue = null
159         false
160       }
161     }
162 
163     override def getCurrentKey: Void = null
164     override def getCurrentValue: SimpleFeature = curValue
165 
166     override def close(): Unit = {
167       if (reader != null) {
168         CloseWithLogging(reader)
169       }
170       if (storage != null) {
171         CloseWithLogging(storage)
172       }
173     }
174   }

179     override def getFiles(partition: Partition): Seq[StorageFile] = files.filter(_.partition == partition)

181       // note: should only be called with filter.include
182       if (filter != Filter.INCLUDE) {
183         logger.warn(s"Unexpected filter: $filter")
184       }

186     }
187     override def addFile(file: StorageFile): Unit = throw new UnsupportedOperationException()
188     override def addFiles(files: Seq[StorageFile]): Unit = throw new UnsupportedOperationException()
189     override def removeFile(file: StorageFile): Unit = throw new UnsupportedOperationException()
190     override def replaceFiles(existing: Seq[StorageFile], replacements: Seq[StorageFile]): Unit =
191       throw new UnsupportedOperationException()
192     override def schemes: Set[PartitionScheme] = throw new UnsupportedOperationException()
193     override def close(): Unit = {}
194   }
