21 class FsGeneratePartitionFiltersCommand extends FsDataStoreCommand {
22 
23   import org.locationtech.geomesa.filter.andFilters
24 
25   override val params = new FsGeneratePartitionFiltersParams()
26 
27   override val name: String = "generate-partition-filters"
28 
29   override def execute(): Unit = withDataStore { ds =>
30     if (params.date == null && params.loadedPartitions.isEmpty) {
31       throw new ParameterException("At least one of --partition, --partition-file, or --date must be specified")
32     }
33 
34     val storage = ds.storage(params.featureName)
35 
36     val fromDate = Option(params.date).toSeq.flatMap(PartitionScheme.enumerate(storage.schemes, _))
37 
38     val partitions = if (params.loadedPartitions.isEmpty) { fromDate } else { (params.loadedPartitions ++ fromDate).distinct }
39 
40     Command.user.info(s"Generating filters for ${partitions.size} partitions")
41     if (!params.noHeader) {
42       Command.output.info("Partition\tFilter")
43     }
44 
45     partitions.sortBy(_.toString).foreach { partition =>
46       val filters = partition.values.flatMap(v => storage.schemes.find(_.name == v.name).map(_.getCoveringFilter(v)))
47       val filter = ECQL.toCQL(andFilters(filters))
48       Command.output.info(s"$partition\t$filter")
49     }
50   }
51 }
