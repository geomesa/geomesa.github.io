37   override def connection: Map[String, String] = {
38     val builder = Map.newBuilder[String, String]
39     if (params.catalogType == null && params.configuration.isEmpty && params.configFile == null) {
40       throw new ParameterException("Must specify at least one of --catalog-type, --config or --config-file")
41     }
42     if (params.catalogType != null) {
43       builder += (FileSystemDataStoreParams.CatalogTypeParam.key -> params.catalogType)
44     }
45     if (!params.configuration.isEmpty) {
46       val props = new Properties()
47       params.configuration.asScala.foreach { case (k, v) => props.put(k, v) }
48       val out = new StringWriter()
49       props.store(out, null)
50       builder += (FileSystemDataStoreParams.ConfigParam.key -> out.toString)
51     }
52     if (params.configFile != null) {
53       builder += (FileSystemDataStoreParams.ConfigFileParam.key -> params.configFile.getAbsolutePath)
54     }
55     if (params.auths != null) {
56       builder += (FileSystemDataStoreParams.AuthsParam.key -> params.auths)
57     }
58     val maxOpenPartitions =
59       Option(params).collect { case p: FsIngestParams if p.maxOpenPartitions != null => p.maxOpenPartitions.toString }.orNull
60     if (maxOpenPartitions != null) {
61       builder += (FileSystemDataStoreParams.WritersMaxOpenPartitionsParam.key -> maxOpenPartitions)
62     }
63     builder.result()
64   }
65 }
