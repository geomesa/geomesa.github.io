32 trait FsDataStoreCommand extends DataStoreCommand[FileSystemDataStore] {
33 
34   import scala.collection.JavaConverters._

36   override def params: FsParams

38   override def connection: Map[String, String] = {
39     val builder = Map.newBuilder[String, String]
40     builder += (FileSystemDataStoreParams.PathParam.getName -> params.path)
41     builder += (FileSystemDataStoreParams.MetadataTypeParam.getName -> params.metadataType)
42     val metadataProps = new Properties()
43     if (params.metadataConfigFile != null) {
44       WithClose(new FileReader(params.metadataConfigFile))(metadataProps.load)
45     }
46     if (!params.metadataConfig.isEmpty) {
47       params.metadataConfig.asScala.foreach { case (k, v) => metadataProps.put(k, v) }
48     }
49     if (!metadataProps.isEmpty) {
50       val out = new StringWriter()
51       metadataProps.store(out, null)
52       builder += (FileSystemDataStoreParams.MetadataConfigParam.getName -> out.toString)
53     }
54 
55     if (!params.configuration.isEmpty) {
56       val xml = {
57         val conf = new Configuration(false)
58         params.configuration.asScala.foreach { case (k, v) => conf.set(k, v) }
59         val out = new StringWriter()
60         conf.writeXml(out)
61         out.toString
62       }
63       builder += (FileSystemDataStoreParams.ConfigsParam.getName -> xml)
64     }
65     if (params.auths != null) {
66       builder += (FileSystemDataStoreParams.AuthsParam.getName -> params.auths)
67     }
68     if (params.encoding != null) {
69       builder += (FileSystemDataStoreParams.EncodingParam.getName -> params.encoding)
70     }
71     builder.result()
72   }
73 }
