18 class FsGetFilesCommand extends FsDataStoreCommand {
19 
20   override val params = new FSGetFilesParams
21 
22   override val name: String = "get-files"
23 
24   override def execute(): Unit = withDataStore { ds =>
25     val metadata = ds.storage(params.featureName).metadata
26 
27     lazy val fromFilter = {
28       Command.user.info(s"Listing files for filter: ${ECQL.toCQL(params.cqlFilter)}")
29       metadata.files().forFilter(params.cqlFilter).scan()
30     }
31     lazy val fromPartitions = {
32       Command.user.info(s"Listing files for partition(s): ${params.loadedPartitions.mkString(", ")}")
33       params.loadedPartitions.flatMap(metadata.files().forPartition(_).scan())
34     }
35 
36     val files =
37       if (params.cqlFilter == null && params.loadedPartitions.isEmpty) {
38         Command.user.info("Listing files for all partitions")
39         metadata.files().scan()
40       } else if (params.loadedPartitions.isEmpty) {
41         fromFilter
42       } else if (params.cqlFilter == null) {
43         fromPartitions
44       } else {
45         (fromFilter ++ fromPartitions).distinct
46       }
47 
48     files.groupBy(f => metadata.partition(f)).toSeq.sortBy(_._1.toString).foreach { case (p, files) =>
49       Command.output.info(s"$p:")
50       // sort by record count descending
51       files.sortBy(_.recordCount())(Ordering[Long].reverse).foreach { f =>
52         Command.output.info(s"  ${f.location()} ${f.recordCount()} features")
53       }
54     }
55   }
56 }
