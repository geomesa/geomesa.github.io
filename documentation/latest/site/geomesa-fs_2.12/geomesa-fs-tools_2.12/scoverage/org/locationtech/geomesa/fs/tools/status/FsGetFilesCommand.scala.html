18 class FsGetFilesCommand extends FsDataStoreCommand {
19 
20   import scala.collection.JavaConverters._
21 
22   override val params = new FSGetFilesParams
23 
24   override val name: String = "get-files"
25 
26   override def execute(): Unit = withDataStore { ds =>
27     val metadata = ds.storage(params.featureName).metadata
28 
29     lazy val fromFilter = {
30       Command.user.info(s"Listing files for filter: ${ECQL.toCQL(params.cqlFilter)}")
31       metadata.files().forFilter(params.cqlFilter).scan()
32     }
33     lazy val fromPartitions = {
34       Command.user.info(s"Listing files for partition(s): ${params.loadedPartitions.mkString(", ")}")
35       params.loadedPartitions.flatMap(metadata.files().forPartition(_).scan())
36     }
37 
38     val files =
39       if (params.cqlFilter == null && params.loadedPartitions.isEmpty) {
40         Command.user.info("Listing files for all partitions")
41         metadata.files().scan()
42       } else if (params.loadedPartitions.isEmpty) {
43         fromFilter
44       } else if (params.cqlFilter == null) {
45         fromPartitions
46       } else {
47         (fromFilter ++ fromPartitions).distinct
48       }
49 
50     files.groupBy(f => metadata.partition(f.file())).toSeq.sortBy(_._1.toString).foreach { case (p, files) =>
51       Command.output.info(s"$p:")
52       // sort by record count descending
53       files.sortBy(_.file().recordCount())(Ordering[Long].reverse).foreach { f =>
54         val deletes = f.deletes().asScala.map(_.recordCount()).sum
55         val count = f.file().recordCount() - deletes
56         Command.output.info(s"  ${f.file().location()} $count features${if (deletes > 0) { s" (and $deletes deleted features)" } else { "" }}")
57       }
58     }
59   }
60 }
