34 trait FileSystemCompactionJob extends StorageConfiguration with JobWithLibJars {
35 
36   import FileSystemCompactionJob.{FailedCounter, MappedCounter}

38   def run(
39       storage: FileSystemStorage,
40       partitions: Seq[Partition],
41       tempPath: Option[Path],
42       libjarsFiles: Seq[String],
43       libjarsPaths: Iterator[() => Seq[File]],
44       statusCallback: StatusCallback): JobResult = {
45 
46     val job = {
47       val conf = new Configuration()
48       storage.conf.foreach { case (k, v) => conf.set(k, v) }
49       Job.getInstance(conf, "GeoMesa Storage Compaction")
50     }
51 
52     setLibJars(job, libjarsFiles, libjarsPaths)
53     job.setJarByClass(this.getClass)
54 
55     // InputFormat and Mappers
56     job.setInputFormatClass(classOf[ParquetPartitionInputFormat])
57     job.setMapperClass(classOf[CompactionMapper])
58 
59     // No reducers - Mapper will read/write its own things
60     job.setNumReduceTasks(0)
61 
62     job.setMapOutputKeyClass(classOf[Void])
63     job.setMapOutputValueClass(classOf[SimpleFeature])
64     job.setOutputKeyClass(classOf[Void])
65     job.setOutputValueClass(classOf[SimpleFeature])
66 
67     StorageConfiguration.setSft(job.getConfiguration, storage.sft)
68     StorageConfiguration.setPartitions(job.getConfiguration, partitions)
69     storage.sizer.targetSize.foreach(StorageConfiguration.setTargetFileSize(job.getConfiguration, _))
70 
71     val outputPath = new Path(S3ObjectStore.s3aUri(s"${storage.table.location()}/tmp"))
72     FileOutputFormat.setOutputPath(job, tempPath.getOrElse(outputPath))
73 
74     // MapReduce options
75     job.getConfiguration.set("mapred.map.tasks.speculative.execution", "false")
76     job.getConfiguration.set("mapreduce.job.user.classpath.first", "true")
77 
78     configureOutput(storage.sft, job)
79 
80     // save the existing files so we can delete them afterward
81     // mimic the filtering done in PartitionInputFormat
82     val sizeCheck = storage.sizer.targetSize.map(t => (p: DataFile) => storage.sizer.fileIsSized(p, t))
83     val existingDataFiles = partitions.toList.flatMap { p =>
84       val files = storage.metadata.files().forPartition(p).scan().filterNot(f => sizeCheck.exists(_.apply(f)))
85       // TODO get counts right... use m/r counters?
86       if (files.isEmpty) { None } else { Some(p -> files) }
87     }
88 
89     def mapCounters: Seq[(String, Long)] = Seq((MappedCounter, written(job)), (FailedCounter, failed(job)))
90 
91     val result = JobRunner.run(job, statusCallback, mapCounters, Seq.empty)
92     // TODO this doesn't work b/c the output format registers the files before we can distcp them
93     /*.merge {
94       tempPath.map { tp =>
95         new DistributedCopy().copy(Seq(tp), outputPath, statusCallback)
96       }
97     }*/
98 
99     result match {
100       case JobSuccess(message, counts) =>
101         if (message.nonEmpty) {
102           Command.user.info(message)
103         }
104         Command.user.info("Removing old files")
105         existingDataFiles.foreach { case (partition, files) =>
106           val name = partition.values.map(_.name).mkString(",")
107           val counter = StorageConfiguration.Counters.partition(name)
108           val count = Option(job.getCounters.findCounter(StorageConfiguration.Counters.Group, counter)).map(_.getValue)
109           val delete = storage.table.newDelete()
110           files.foreach(delete.deleteFile)
111           delete.commit()
112           files.foreach(f => storage.table.io().deleteFile(f.location()))
113           val removed = count.map(c => s"containing $c features ").getOrElse("")
114           Command.user.info(s"Removed ${TextTools.getPlural(files.size, "file")} ${removed}in partition $name")
115         }
116         JobSuccess("", counts)
117 
118       case j => j
119     }
120   }
121 
122   private def written(job: Job): Long =
123     job.getCounters.findCounter(OutputCounters.Group, OutputCounters.Written).getValue
124 
125   private def failed(job: Job): Long =
126     job.getCounters.findCounter(OutputCounters.Group, OutputCounters.Failed).getValue
127 }
