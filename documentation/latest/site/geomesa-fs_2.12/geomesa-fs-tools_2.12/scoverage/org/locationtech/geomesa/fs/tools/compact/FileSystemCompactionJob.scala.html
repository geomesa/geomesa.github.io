35 trait FileSystemCompactionJob extends StorageConfiguration with JobWithLibJars {
36 
37   import FileSystemCompactionJob.{FailedCounter, MappedCounter}
38 
39   def run(
40       storage: FileSystemStorage,
41       partitions: Seq[Partition],
42       targetFileSize: Option[Long],
43       tempPath: Option[Path],
44       libjarsFiles: Seq[String],
45       libjarsPaths: Iterator[() => Seq[File]],
46       statusCallback: StatusCallback): JobResult = {
47 
48     val job = {
49       val conf = new Configuration()
50       storage.context.conf.foreach { case (k, v) => conf.set(k, v) }
51       S3ObjectStore.s3aConfigs(storage.context.conf).foreach { case (k, v) => conf.set(k, v) }
52       Job.getInstance(conf, "GeoMesa Storage Compaction")
53     }
54 
55     setLibJars(job, libjarsFiles, libjarsPaths)
56     job.setJarByClass(this.getClass)
57 
58     // InputFormat and Mappers
59     job.setInputFormatClass(classOf[PartitionInputFormat])
60     job.setMapperClass(classOf[CompactionMapper])
61 
62     // No reducers - Mapper will read/write its own things
63     job.setNumReduceTasks(0)
64 
65     job.setMapOutputKeyClass(classOf[Void])
66     job.setMapOutputValueClass(classOf[SimpleFeature])
67     job.setOutputKeyClass(classOf[Void])
68     job.setOutputValueClass(classOf[SimpleFeature])
69 
70     StorageConfiguration.setRootPath(job.getConfiguration, storage.context.root)
71     StorageConfiguration.setEncoding(job.getConfiguration, storage.encoding)
72     StorageConfiguration.setSft(job.getConfiguration, storage.metadata.sft)
73     StorageConfiguration.setPartitions(job.getConfiguration, partitions)
74     StorageConfiguration.setFileType(job.getConfiguration, FileType.Compacted)
75     targetFileSize.foreach(StorageConfiguration.setTargetFileSize(job.getConfiguration, _))
76 
77     FileOutputFormat.setOutputPath(job, tempPath.getOrElse { new Path(S3ObjectStore.s3aUri(storage.context.root)) })
78 
79     // MapReduce options
80     job.getConfiguration.set("mapred.map.tasks.speculative.execution", "false")
81     job.getConfiguration.set("mapreduce.job.user.classpath.first", "true")
82 
83     configureOutput(storage.metadata.sft, job)
84 
85     // save the existing files so we can delete them afterwards
86     // mimic the filtering done in PartitionInputFormat
87     val sizeCheck = targetFileSize.orElse(storage.sizer.targetSize).map(t => (p: URI) => storage.sizer.fileIsSized(p, t))
88     val existingDataFiles = partitions.toList.flatMap { p =>
89       val files = storage.metadata.getFiles(p).filterNot(f => sizeCheck.exists(_.apply(storage.context.root.resolve(f.file))))
90       // TODO get counts right... use m/r counters?
91       if (files.isEmpty) { None } else { Some(p -> files) }
92     }
93 
94     def mapCounters = Seq((MappedCounter, written(job)), (FailedCounter, failed(job)))
95 
96     val result = JobRunner.run(job, statusCallback, mapCounters, Seq.empty).merge {
97       tempPath.map { tp =>
98         new DistributedCopy().copy(Seq(tp), new Path(storage.context.root), statusCallback)
99       }
100     }
101 
102     result match {
103       case JobSuccess(message, counts) =>
104         if (message.nonEmpty) {
105           Command.user.info(message)
106         }
107         Command.user.info("Removing old files")
108         existingDataFiles.foreach { case (partition, files) =>
109           val name = partition.values.map(_.name).mkString(",")
110           val counter = StorageConfiguration.Counters.partition(name)
111           val count = Option(job.getCounters.findCounter(StorageConfiguration.Counters.Group, counter)).map(_.getValue)
112           files.foreach { f =>
113             storage.metadata.removeFile(f)
114             storage.fs.delete(storage.context.root.resolve(f.file))
115           }
116           val removed = count.map(c => s"containing $c features ").getOrElse("")
117           Command.user.info(s"Removed ${TextTools.getPlural(files.size, "file")} ${removed}in partition $name")
118         }
119         JobSuccess("", counts)
120 
121       case j => j
122     }
123   }
124 
125   private def written(job: Job): Long =
126     job.getCounters.findCounter(OutputCounters.Group, OutputCounters.Written).getValue
127 
128   private def failed(job: Job): Long =
129     job.getCounters.findCounter(OutputCounters.Group, OutputCounters.Failed).getValue
130 }
