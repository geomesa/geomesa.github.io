40   def run(
41       storage: FileSystemStorage,
42       partitions: Seq[Partition],
43       tempPath: Option[Path],
44       libjarsFiles: Seq[String],
45       libjarsPaths: Iterator[() => Seq[File]],
46       statusCallback: StatusCallback): JobResult = {
47 
48     val job = {
49       val conf = new Configuration()
50       storage.conf.foreach { case (k, v) => conf.set(k, v) }
51       Job.getInstance(conf, "GeoMesa Storage Compaction")
52     }
53 
54     setLibJars(job, libjarsFiles, libjarsPaths)
55     job.setJarByClass(this.getClass)
56 
57     // InputFormat and Mappers
58     job.setInputFormatClass(classOf[ParquetPartitionInputFormat])
59     job.setMapperClass(classOf[CompactionMapper])
60 
61     // No reducers - Mapper will read/write its own things
62     job.setNumReduceTasks(0)
63 
64     job.setMapOutputKeyClass(classOf[Void])
65     job.setMapOutputValueClass(classOf[SimpleFeature])
66     job.setOutputKeyClass(classOf[Void])
67     job.setOutputValueClass(classOf[SimpleFeature])
68 
69     StorageConfiguration.setSft(job.getConfiguration, storage.sft)
70     StorageConfiguration.setPartitions(job.getConfiguration, partitions)
71     storage.sizer.targetSize.foreach(StorageConfiguration.setTargetFileSize(job.getConfiguration, _))
72 
73     val outputPath = new Path(S3ObjectStore.s3aUri(s"${storage.table.location()}/tmp"))
74     FileOutputFormat.setOutputPath(job, tempPath.getOrElse(outputPath))
75 
76     // MapReduce options
77     job.getConfiguration.set("mapred.map.tasks.speculative.execution", "false")
78     job.getConfiguration.set("mapreduce.job.user.classpath.first", "true")
79 
80     configureOutput(storage.sft, job)
81 
82     // save the existing files so we can delete them afterward
83     // mimic the filtering done in PartitionInputFormat
84     val sizeCheck = storage.sizer.targetSize.map(t => (p: DataFile) => storage.sizer.fileIsSized(p, t))
85     val existingDataFiles = partitions.toList.flatMap { p =>
86       val files = storage.metadata.files().forPartition(p).scan().filterNot(f => sizeCheck.exists(_.apply(f.file())))
87       // TODO get counts right... use m/r counters?
88       if (files.isEmpty) { None } else { Some(p -> files) }
89     }
90 
91     def mapCounters: Seq[(String, Long)] = Seq((MappedCounter, written(job)), (FailedCounter, failed(job)))
92 
93     val result = JobRunner.run(job, statusCallback, mapCounters, Seq.empty)
94     // TODO this doesn't work b/c the output format registers the files before we can distcp them
95     /*.merge {
96       tempPath.map { tp =>
97         new DistributedCopy().copy(Seq(tp), outputPath, statusCallback)
98       }
99     }*/
100 
101     result match {
102       case JobSuccess(message, counts) =>
103         if (message.nonEmpty) {
104           Command.user.info(message)
105         }
106         Command.user.info("Removing old files")
107         existingDataFiles.foreach { case (partition, files) =>
108           val name = partition.values.map(_.name).mkString(",")
109           val counter = StorageConfiguration.Counters.partition(name)
110           val count = Option(job.getCounters.findCounter(StorageConfiguration.Counters.Group, counter)).map(_.getValue)
111           val delete = storage.table.newDelete()
112           files.foreach(f => delete.deleteFile(f.file()))
113           delete.commit()
114           files.foreach { f =>
115             storage.table.io().deleteFile(f.file().location())
116             f.deletes().asScala.foreach(d => storage.table.io().deleteFile(d.location()))
117           }
118           val removed = count.map(c => s"containing $c features ").getOrElse("")
119           Command.user.info(s"Removed ${TextTools.getPlural(files.size, "file")} ${removed}in partition $name")
120         }
121         JobSuccess("", counts)
122 
123       case j => j
124     }
125   }
126

129 
130   private def failed(job: Job): Long =
131     job.getCounters.findCounter(OutputCounters.Group, OutputCounters.Failed).getValue
