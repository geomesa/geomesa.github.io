38   private class RegisterCommand extends FsDataStoreCommand {
39 
40     override val name = "register"
41     override val params = new RegisterParams()
42 
43     override def execute(): Unit = withDataStore { ds =>
44       val storage = ds.storage(params.featureName)
45 
46       val paths = params.files.asScala.map { file =>
47         val path = {
48           val root = URI.create(storage.table.location())
49           val tmp = new URI(file)
50           if (tmp.getScheme == null || tmp.getScheme.isEmpty) {
51             new URI(root.getScheme, tmp.getHost, tmp.getPath, tmp.getFragment)
52           } else if (tmp.getScheme == root.getScheme) {
53             tmp
54           } else {
55             throw new IllegalArgumentException(s"File $file must be in the same filesystem as the storage context: $root")
56           }
57         }
58         if (!storage.table.io().newInputFile(path.toString).exists()) {
59           throw new IllegalArgumentException(s"File $path does not exist")
60         }
61         path
62       }
63 
64       def outputResult(file: DataFile): Unit = {
65         Command.user.info(s"Registered file ${file.location()} containing ${file.recordCount()} known features")
66         // TODO: extract and display bounds from DataFile.lowerBounds()/upperBounds()
67       }
68 
69       try {
70         paths.foreach { path =>
71           outputResult(storage.metadata.register(path))
72           if (params.delete) {
73             storage.table.io().deleteFile(path.toString)
74           }
75         }
76       } catch {
77         case NonFatal(e) => throw new RuntimeException("Error registering file:", e)
78       }
79     }
80   }

108         val current = Metadata.get(table, key)
109         Metadata.set(table, k, value.orNull)
110         Command.output.info(s"Updated $key from '${current.getOrElse("<unset>")}' to '${value.getOrElse("<unset>")}'")
111       }
112     }
