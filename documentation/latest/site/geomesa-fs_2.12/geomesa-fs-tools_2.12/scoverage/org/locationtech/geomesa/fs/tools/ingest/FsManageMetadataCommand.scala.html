55   private class RegisterCommand extends FsDataStoreCommand {
56 
57     override val name = "register"
58     override val params = new RegisterParams()
59 
60     override def execute(): Unit = withDataStore { ds =>
61       val storage = ds.storage(params.featureName)
62       val metadata = storage.metadata
63 
64       val paths = params.files.asScala.map { file =>
65         val path = {
66           val tmp = new URI(file)
67           if (tmp.getScheme == null || tmp.getScheme.isEmpty) {
68             new URI(storage.context.root.getScheme, tmp.getHost, tmp.getPath, tmp.getFragment)
69           } else if (tmp.getScheme == storage.context.root.getScheme) {
70             tmp
71           } else {
72             throw new IllegalArgumentException(
73               s"File $file must have the same scheme as the storage context: ${storage.context.root}")
74           }
75         }
76         if (!storage.fs.exists(path)) {
77           throw new IllegalArgumentException(s"File $path does not exist")
78         }
79         path
80       }
81 
82       def outputResult(file: StorageFile): Unit = {
83         Command.user.info(s"Registered file ${storage.context.root.resolve(file.file)} containing ${file.count} known features")
84         val bounds =
85           file.spatialBounds.map(b => s"${metadata.sft.getDescriptor(b.attribute).getLocalName} [${b.xmin},${b.ymin},${b.xmax},${b.ymax}]") ++
86             file.attributeBounds.map { b =>
87               val descriptor = metadata.sft.getDescriptor(b.attribute)
88               val alias = AttributeIndexKey.alias(descriptor.getType.getBinding)
89               val lower = AttributeIndexKey.decode(alias, b.lower) match {
90                 case null => ""
91                 case d: Date => DateParsing.formatDate(d)
92                 case v => v.toString
93               }
94               val upper = AttributeIndexKey.decode(alias, b.upper) match {
95                 case null => ""
96                 case d: Date => DateParsing.formatDate(d)
97                 case v => v.toString
98               }
99               s"${descriptor.getLocalName} [$lower,$upper]"
100             }
101         Command.user.info(s"File bounds:\n  ${bounds.mkString("\n  ")}")
102       }
103 
104       try {
105         paths.foreach { path =>
106           outputResult(storage.register(path))
107           if (params.delete) {
108             storage.fs.delete(path)
109           }
110         }
111       } catch {
112         case NonFatal(e) => throw new RuntimeException("Error registering file:", e)
113       }
114     }

187     class ConsistencyChecker(storage: FileSystemStorage, otherTypes: Seq[FileSystemStorage], threads: Int)
188         extends Runnable with Closeable with LazyLogging {
189 
190       private val pool = new CachedThreadPool(threads)
191       private val onDisk = Collections.newSetFromMap(new ConcurrentHashMap[URI, java.lang.Boolean]())
192 
193       override def run(): Unit = {
194         // list out the files currently in the root directory, results go into onDisk
195         listRoot()
196 
197         val inconsistencies = ArrayBuffer.empty[Inconsistency]
198         val checked = scala.collection.mutable.Set.empty[String]
199 
200         // compare the files known to the metadata to the ones on disk
201         storage.metadata.getFiles().foreach { file =>
202           if (onDisk.remove(storage.context.root.resolve(file.file))) {
203             // metadata and file are consistent
204             checked.add(file.file)
205           } else {
206             // file is missing from metadata vs file is in metadata more than once
207             val duplicate = !checked.add(file.file)
208             inconsistencies += Inconsistency(file, duplicate)
209           }
210         }
211 
212         if (onDisk.isEmpty && inconsistencies.isEmpty) {
213           Command.user.info("No inconsistencies detected")
214         } else {
215           if (inconsistencies.nonEmpty) {
216             lazy val strings = inconsistencies.map { i =>
217               s"${i.file.file} (${if (i.duplicate) { "duplicate" } else { "missing" }})"
218             }
219             val msg = if (inconsistencies.size == 1) { "1 metadata entry that does" } else { s"${inconsistencies.size} metadata entries that do"}
220             Command.user.warn(s"Found $msg not correspond to a data file:")
221             Command.output.info(strings.sorted.mkString("  ", "\n  ", ""))
222           }
223           if (!onDisk.isEmpty) {
224             // filter out files from other feature types in the same root
225             otherTypes.foreach { s =>
226               s.metadata.getFiles().foreach { f =>
227                 onDisk.remove(s.context.root.resolve(f.file))
228               }
229             }
230             if (!onDisk.isEmpty) {
231               val msg = if (onDisk.size() == 1) { "1 data file that does" } else { s"${onDisk.size} data files that do"}
232               Command.user.warn(s"Found $msg not have a metadata entry:")
233               if (otherTypes.nonEmpty) {
234                 Command.user.warn(s"Note: they may belong to one of the other registered feature types in this root")
235               }
236               Command.output.info(onDisk.asScala.map(_.toString).toSeq.sorted.mkString("  ", "\n  ", ""))
237             }
238           }
239         }
240       }
241 
242       override def close(): Unit = pool.shutdown()
243 
244       /**
245        * List the files in the root directory. Excludes known file-based metadata files. Results are
246        * added to onDisk
247        */
248       private def listRoot(): Unit = {
249         val iter = storage.fs.list(storage.context.root)
250         // use a phaser to track worker thread completion
251         val phaser = new Phaser(2) // 1 for this thread + 1 for the worker
252         pool.submit(new TopLevelListWorker(phaser, iter))
253         // wait for the worker threads to complete
254         phaser.awaitAdvanceInterruptibly(phaser.arrive())
255       }
256 
257       private class TopLevelListWorker(phaser: Phaser, list: CloseableIterator[URI]) extends Runnable {
258         override def run(): Unit = {
259           try {
260             var i = phaser.getRegisteredParties + 1
261             while (list.hasNext && i < PhaserUtils.MaxParties) {
262               val path = list.next
263               if (path.toString.endsWith("/")) { // .isDirectory
264                 if (!path.toString.endsWith(s"/${FileBasedMetadataCatalog.MetadataDirectory}/")) {
265                   i += 1
266                   // use a tiered phaser on each directory avoid the limit of 65535 registered parties
267                   pool.submit(new ListWorker(new Phaser(phaser, 1), storage.fs.list(path)))
268                 }
269               } else if (!storage.fs.filename(path).startsWith(".")) { // ignore "hidden" files
270                 onDisk.add(path)
271               }
272             }
273             if (list.hasNext) {
274               pool.submit(new TopLevelListWorker(new Phaser(phaser, 1), list))
275             } else {
276               list.close()
277             }
278           } catch {
279             case NonFatal(e) => Try(list.close()); throw e
280           } finally {
281             phaser.arriveAndDeregister()
282           }
283         }
284       }
285 
286       private class ListWorker(phaser: Phaser, listDirectory: => Iterator[URI]) extends Runnable {
287         override def run(): Unit = {
288           try {
289             var i = phaser.getRegisteredParties + 1
290             val iter = listDirectory
291             while (iter.hasNext && i < PhaserUtils.MaxParties) {
292               val path = iter.next
293               if (path.toString.endsWith("/")) { // .isDirectory
294                 i += 1
295                 // use a tiered phaser on each directory avoid the limit of 65535 registered parties
296                 pool.submit(new ListWorker(new Phaser(phaser, 1), storage.fs.list(path)))
297               } else if (!storage.fs.filename(path).startsWith(".")) { // ignore "hidden" files
298                 onDisk.add(path)
299               }
300             }
301             if (iter.hasNext) {
302               pool.submit(new ListWorker(new Phaser(phaser, 1), iter))
303             }
304           } catch {
305             case _: FileNotFoundException => // the partition dir was deleted... just return
306             case NonFatal(e) => logger.error("Error scanning metadata directory:", e)
307           } finally {
308             phaser.arriveAndDeregister() // notify that this thread is done
309           }
310         }
311       }
312     }
