33   def compact(ds: FileSystemDataStore): Unit = {
34     Command.user.info("Beginning compaction process...")
35 
36     val storage = ds.storage(params.featureName)
37 
38     val toCompact = if (params.loadedPartitions.isEmpty) { storage.metadata.partitions() } else {
39       val filtered = params.loadedPartitions.filter(storage.metadata.files().forPartition(_).scan().nonEmpty)
40       if (filtered.isEmpty) {
41         throw new ParameterException(s"Partition(s) did not match any files: ${params.loadedPartitions.mkString(", ")}")
42       } else if (filtered.size != params.loadedPartitions.size) {
43         val unmatched = params.loadedPartitions.filterNot(filtered.contains)
44         Command.user.warn(s"Some partition did not match any files: ${unmatched.mkString(", ")}")
45       }
46       filtered
47     }
48 
49     Command.user.info(s"Compacting ${toCompact.size} partitions")
50 
51     val start = System.currentTimeMillis()
52     val status = TerminalCallback()
53 
54     val total = toCompact.length
55     val latch = new CountDownLatch(total)
56     val failure = new AtomicReference[Throwable]()
57     val executor = Executors.newFixedThreadPool(math.max(1, math.min(params.threads, total)))
58 
59     try {
60       toCompact.foreach { p =>
61         executor.submit(
62           new Runnable() {
63             override def run(): Unit = {
64               try {
65                 logger.info(s"Compacting $p")
66                 storage.metadata.compact(p)
67               } catch {
68                 case NonFatal(e) =>
69                   failure.compareAndSet(null, e)
70                   logger.error(s"Error processing partition '$p':", e)
71               } finally {
72                 latch.countDown()
73               }
74             }
75           }
76         )
77       }
78     } finally {
79       executor.shutdown()
80     }
81 
82     while (latch.getCount > 0) {
83       Thread.sleep(1000)
84       status("", 1f - latch.getCount.toFloat / total, Seq.empty, done = false)
85     }
86     status("", 1f, Seq.empty, done = true)
87     Option(failure.get()).foreach(e => throw new CommandException("Compaction failed").initCause(e))
88     Command.user.info(s"Compaction complete in ${TextTools.getTime(start)}")
89   }
90 }
