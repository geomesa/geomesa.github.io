45     def compact(ds: FileSystemDataStore): Unit = {
46       Command.user.info("Beginning compaction process...")
47 
48       val storage = ds.storage(params.featureName)
49 
50       val toCompact = if (params.loadedPartitions.isEmpty) { storage.metadata.partitions() } else {
51         val filtered = params.loadedPartitions.filter(storage.metadata.files().forPartition(_).scan().nonEmpty)
52         if (filtered.isEmpty) {
53           throw new ParameterException(s"Partition(s) did not match any files: ${params.loadedPartitions.mkString(", ")}")
54         } else if (filtered.size != params.loadedPartitions.size) {
55           val unmatched = params.loadedPartitions.filterNot(filtered.contains)
56           Command.user.warn(s"Some partition did not match any files: ${unmatched.mkString(", ")}")
57         }
58         filtered
59       }
60 
61       val mode = params.mode.getOrElse(RunModes.Distributed)
62 
63       Command.user.info(s"Compacting ${toCompact.size} files in ${mode.toString.toLowerCase(Locale.US)} mode")
64 
65       val start = System.currentTimeMillis()
66       val status = TerminalCallback()
67 
68       mode match {
69         case RunModes.Local =>
70           val total = toCompact.length
71           val latch = new CountDownLatch(total)
72           val executor = Executors.newFixedThreadPool(math.max(1, math.min(params.threads, total)))
73 
74           try {
75             toCompact.foreach { p =>
76               executor.submit(
77                 new Runnable() {
78                   override def run(): Unit = {
79                     try {
80                       logger.info(s"Compacting $p")
81                       storage.metadata.compact(p)
82                     } catch {
83                       case NonFatal(e) => logger.error(s"Error processing partition '$p':", e)
84                     } finally {
85                       latch.countDown()
86                     }
87                   }
88                 }
89               )
90             }
91           } finally {
92             executor.shutdown()
93           }
94 
95           while (latch.getCount > 0) {
96             Thread.sleep(1000)
97             status("", 1f - latch.getCount.toFloat / total, Seq.empty, done = false)
98           }
99           status("", 1f, Seq.empty, done = true)
100           Command.user.info(s"Local compaction complete in ${TextTools.getTime(start)}")
101 
102         case RunModes.Distributed =>
103           val job = new ParquetCompactionJob()
104           val tempDir = Option(params.tempPath).map(t => new Path(t))
105           job.run(storage, toCompact.toSeq, tempDir, libjarsFiles, libjarsPaths, status) match {
106             case JobSuccess(message, counts) =>
107               Command.user.info(s"Distributed compaction complete in ${TextTools.getTime(start)}")
108               val success = counts(FileSystemCompactionJob.MappedCounter)
109               val failed = counts(FileSystemCompactionJob.FailedCounter)
110               Command.user.info(IngestCommand.getStatInfo(success, failed, "Compacted", message))
111 
112             case JobFailure(message) =>
113               Command.user.error(s"Distributed compaction failed in ${TextTools.getTime(start)}")
114               throw new CommandException(message)
115           }
116       }
117     }
118   }
