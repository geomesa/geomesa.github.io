49     def compact(ds: FileSystemDataStore): Unit = {
50       Command.user.info("Beginning compaction process...")
51 
52       val storage = ds.storage(params.featureName)
53 
54       val toCompact = if (params.partitions.isEmpty) { storage.metadata.getFiles().map(_.partition).distinct } else {
55         val filtered = params.partitions.asScala.filter(storage.metadata.getFiles(_).nonEmpty)
56         if (filtered.isEmpty) {
57           throw new ParameterException(s"Partition(s) did not match any files: ${params.partitions.asScala.mkString(", ")}")
58         } else if (filtered.size != params.partitions.size) {
59           val unmatched = params.partitions.asScala.filterNot(filtered.contains)
60           Command.user.warn(s"Some partition did not match any files: ${unmatched.mkString(", ")}")
61         }
62         filtered
63       }
64 
65       val mode = params.mode.getOrElse {
66         if (PathUtils.isRemote(storage.context.root.toString)) { RunModes.Distributed } else { RunModes.Local }
67       }
68       val fileSize = Option(params.targetFileSize).map(_.longValue)
69 
70       Command.user.info(s"Compacting ${toCompact.size} files in ${mode.toString.toLowerCase(Locale.US)} mode")
71 
72       val start = System.currentTimeMillis()
73       val status = TerminalCallback()
74 
75       mode match {
76         case RunModes.Local =>
77           val total = toCompact.length
78           val latch = new CountDownLatch(total)
79           val executor = Executors.newFixedThreadPool(math.max(1, math.min(params.threads, total)))
80 
81           try {
82             toCompact.foreach { p =>
83               executor.submit(
84                 new Runnable() {
85                   override def run(): Unit = {
86                     try {
87                       logger.info(s"Compacting $p")
88                       storage.compact(p, fileSize)
89                     } catch {
90                       case NonFatal(e) => logger.error(s"Error processing partition '$p':", e)
91                     } finally {
92                       latch.countDown()
93                     }
94                   }
95                 }
96               )
97             }
98           } finally {
99             executor.shutdown()
100           }
101 
102           while (latch.getCount > 0) {
103             Thread.sleep(1000)
104             status("", 1f - latch.getCount.toFloat / total, Seq.empty, done = false)
105           }
106           status("", 1f, Seq.empty, done = true)
107           Command.user.info(s"Local compaction complete in ${TextTools.getTime(start)}")
108 
109         case RunModes.Distributed =>
110           val job = new ParquetCompactionJob()
111           val tempDir = Option(params.tempPath).map(t => new Path(t))
112           val metadataConfig = FileSystemDataStoreParams.MetadataConfigParam.lookupOpt(connection.asJava)
113           job.run(storage, metadataConfig, toCompact.toSeq, fileSize, tempDir, libjarsFiles, libjarsPaths, status) match {
114             case JobSuccess(message, counts) =>
115               Command.user.info(s"Distributed compaction complete in ${TextTools.getTime(start)}")
116               val success = counts(FileSystemCompactionJob.MappedCounter)
117               val failed = counts(FileSystemCompactionJob.FailedCounter)
118               Command.user.info(IngestCommand.getStatInfo(success, failed, "Compacted", message))
119 
120             case JobFailure(message) =>
121               Command.user.error(s"Distributed compaction failed in ${TextTools.getTime(start)}")
122               throw new CommandException(message)
123           }
124       }
125     }
126   }
