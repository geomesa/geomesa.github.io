24 class ParquetSimpleFeatureOutputFormat extends ParquetOutputFormat[SimpleFeature] {
25 
26   private var committer: OutputCommitter = _
27 
28   // overridden to write metadata for each file instead of single file parquet expects
29   override def getOutputCommitter(context: TaskAttemptContext): OutputCommitter = {
30     if (committer == null) {
31       val output = FileOutputFormat.getOutputPath(context)
32       committer = new ParquetMultiFileOutputCommitter(output, context)
33     }
34     committer
35   }
36 }

38 object ParquetSimpleFeatureOutputFormat {
39 
40   class ParquetMultiFileOutputCommitter(outputPath: Path, context: TaskAttemptContext)
41       extends FileOutputCommitter(outputPath, context) with LazyLogging {
42     // based on parquetOutputCommitter, but for multiple output files
43     @throws[IOException]
44     override def commitJob(jobContext: JobContext): Unit = {
45       super.commitJob(jobContext)
46       val conf = ContextUtil.getConfiguration(jobContext)
47       listFiles(outputPath, conf).map(_.getParent).distinct.foreach { path =>
48         ParquetOutputCommitter.writeMetaDataFile(conf, path)
49         logger.info(s"Wrote metadata file for path $path")
50       }
51     }
52   }
53 
54   private def listFiles(path: Path, conf: Configuration): Seq[Path] = {
55     val fs = path.getFileSystem(conf)
56     val listing = fs.listFiles(path, true)
57 
58     val result = mutable.ListBuffer.empty[Path]
59     while (listing.hasNext) {
60       val next = listing.next()
61       if (next.isFile) {
62         val p = next.getPath
63         if (p.getName.endsWith(".parquet")) {
64           result += p
65         }
66       }
67     }
68 
69     result.toSeq
70   }
