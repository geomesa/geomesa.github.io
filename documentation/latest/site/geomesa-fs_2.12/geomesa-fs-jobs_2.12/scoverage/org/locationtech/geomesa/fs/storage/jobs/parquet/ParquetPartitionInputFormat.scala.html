25 /**
26   * An Input format that creates splits based on FSDS Partitions. This is used for compaction, when we want a single
27   * split per partition. Otherwise, use ParquetSimpleFeatureInputFormat as that is more efficient
28   */
29 class ParquetPartitionInputFormat extends InputFormat[Void, SimpleFeature] {
30 
31   override def getSplits(context: JobContext): java.util.List[InputSplit] = {
32     val hadoopConf = context.getConfiguration
33     val conf = {
34       val builder = Map.newBuilder[String, String]
35       hadoopConf.forEach(e => builder += e.getKey -> hadoopConf.get(e.getKey)) // use .get to resolve envs
36       builder.result()
37     }
38 
39     val fileSize = StorageConfiguration.getTargetFileSize(hadoopConf)
40     val typeName = StorageConfiguration.getSftName(hadoopConf)
41 
42     WithClose(StorageCatalog(conf)) { catalog =>
43       WithClose(catalog.load(typeName)) { storage =>
44         val sizeCheck = fileSize.orElse(storage.sizer.targetSize).map(t => (f: DataFile) => storage.sizer.fileIsSized(f, t))
45         val splits = StorageConfiguration.getPartitions(hadoopConf).map { partition =>
46           var size = 0L
47           val files = storage.metadata.files().forPartition(partition).scan().filter { f =>
48             if (sizeCheck.exists(_.apply(f))) { false } else {
49               size += f.fileSizeInBytes()
50               true
51             }
52           }
53           new PartitionInputSplit(partition.toString, files.map(_.location()), size)
54         }
55         java.util.Arrays.asList(splits: _*)
56       }
57     }
58   }
59 
60   override def createRecordReader(split: InputSplit, context: TaskAttemptContext): RecordReader[Void, SimpleFeature] =
61     new PartitionRecordReader(split.asInstanceOf[PartitionInputSplit].getFiles)
62 }

66   /**
67     * InputSplit corresponding to a single FileSystemDataStore PartitionScheme partition
68     */
69   class PartitionInputSplit extends InputSplit with Writable {
70 
71     private var name: String = _
72     private var files: Seq[String] = _
73     private var length: java.lang.Long = _
74 
75     def this(name: String, files: Seq[String], length: Long) = {
76       this()
77       this.name = name
78       this.files = files
79       this.length = length
80     }
81 
82     /**
83       * @return the name of this partition
84       */
85     def getName: String = name
86 
87     def getFiles: Seq[String] = files
88 
89     override def getLength: Long = length
90 
91     // TODO attempt to optimize the locations where this should run in the case of HDFS
92     // With S3 this won't really matter
93     override def getLocations: Array[String] = Array.empty[String]
94 
95     override def write(out: DataOutput): Unit = {
96       out.writeUTF(name)
97       out.writeLong(length)
98       out.writeInt(files.length)
99       files.foreach(out.writeUTF)
100     }
101 
102     override def readFields(in: DataInput): Unit = {
103       this.name = in.readUTF()
104       this.length = in.readLong()
105       val count = in.readInt
106       this.files = Seq.fill(count)(in.readUTF())
107     }
108   }
109 
110   class PartitionRecordReader(files: Seq[String]) extends RecordReader[Void, SimpleFeature] {
111 
112     private var catalog: StorageCatalog = _
113     private var storage: FileSystemStorage = _
114     private var reader: CloseableIterator[SimpleFeature] = _
115 
116     private var curValue: SimpleFeature = _
117 
118     private def filterFiles(location: String): Boolean = files.contains(location)
119 
120     override def initialize(split: InputSplit, context: TaskAttemptContext): Unit = {
121       val hadoopConf = context.getConfiguration
122       val conf = {
123         val builder = Map.newBuilder[String, String]
124         hadoopConf.forEach(e => builder += e.getKey -> e.getValue)
125         builder.result()
126       }
127       val typeName = StorageConfiguration.getSftName(hadoopConf)
128       catalog = StorageCatalog(conf)
129       storage = catalog.load(typeName)
130 
131       val readSchema = storage.schema.read(None, Set.empty)
132       reader = new IcebergParquetScan(storage.table, readSchema, Expressions.alwaysTrue(), math.min(8, files.size), Some(filterFiles))
133     }
134 
135     // TODO look at how the ParquetInputFormat provides progress and utilize something similar
136     override def getProgress: Float = 0.0f
137 
138     override def nextKeyValue(): Boolean = {
139       if (reader.hasNext) {
140         curValue = reader.next()
141         true
142       } else {
143         curValue = null
144         false
145       }
146     }
147 
148     override def getCurrentKey: Void = null
149     override def getCurrentValue: SimpleFeature = curValue
150 
151     override def close(): Unit = CloseWithLogging(Seq(reader, storage, catalog).filter(_ != null))
152   }
