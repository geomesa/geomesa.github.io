25 /**
26   * Output format that writes to multiple partition files
27   */
28 class ParquetPartitionOutputFormat extends OutputFormat[Void, SimpleFeature] {
29 
30   private val delegate = new ParquetSimpleFeatureOutputFormat()
31 
32   override def getRecordWriter(context: TaskAttemptContext): RecordWriter[Void, SimpleFeature] =
33     new PartitionSchemeRecordWriter(context)
34 
35   override def getOutputCommitter(context: TaskAttemptContext): OutputCommitter =
36     delegate.getOutputCommitter(context)
37 
38   // same as FileOutputFormat, but doesn't require that output directory doesn't exist
39   override def checkOutputSpecs(job: JobContext): Unit = {
40     // Ensure that the output directory is set
41     val outDir = FileOutputFormat.getOutputPath(job)
42     if (outDir == null) {
43       throw new InvalidJobConfException("Output directory not set")
44     }
45     // get delegation token for outDir's file system
46     TokenCache.obtainTokensForNamenodes(job.getCredentials, Array[Path](outDir), job.getConfiguration)
47   }
48 
49   private class PartitionSchemeRecordWriter(context: TaskAttemptContext)
50       extends RecordWriter[Void, SimpleFeature] with LazyLogging {
51 
52     import StorageConfiguration.Counters.{Features, Group}
53 
54     private val conf = {
55       val hadoopConf = context.getConfiguration
56       val builder = Map.newBuilder[String, String]
57       hadoopConf.forEach(e => builder += e.getKey -> hadoopConf.get(e.getKey)) // use .get to resolve envs
58       builder.result()
59     }
60 
61     private val catalog = StorageCatalog(conf)
62     private val storage = catalog.load(StorageConfiguration.getSftName(context.getConfiguration))
63 
64     private val fileSize = StorageConfiguration.getTargetFileSize(context.getConfiguration)
65 
66     private val counter = context.getCounter(Group, Features)
67     private val cache = scala.collection.mutable.Map.empty[Seq[PartitionKey], PartitionState]
68 
69     private val workPath = delegate.getOutputCommitter(context).asInstanceOf[FileOutputCommitter].getWorkPath
70 
71     private def newState(partitions: Seq[PartitionKey]): PartitionState = {
72       val estimator = fileSize.map(storage.sizer.estimator)
73       estimator match {
74         case None => new SinglePartitionState(Partition(partitions))
75         case Some(e) => new ChunkedPartitionState(Partition(partitions), e, context)
76       }
77     }
78 
79     override def write(key: Void, value: SimpleFeature): Unit = {
80       val partition = storage.schemes.map(_.getPartition(value))
81       val state = cache.getOrElseUpdate(partition, newState(partition))
82       state.write(key, value)
83       counter.increment(1)
84     }
85 
86     override def close(context: TaskAttemptContext): Unit = {
87       cache.foreach { case (partition, state) =>
88         logger.debug(s"Closing writer for $partition")
89         state.close(context)
90         storage.metadata.register(Map(Partition(partition) -> state.getFiles()))
91       }
92       CloseWithLogging(Seq(storage, catalog))
93     }
94 
95     private sealed abstract class PartitionState(partition: Partition) {
96 
97       import scala.collection.JavaConverters._
98 
99       private val files = new CopyOnWriteArrayList[URI]()
100       private val counter = context.getCounter(Group, StorageConfiguration.Counters.partition(partition.toString))
101 
102       private var currentFile: Path = _
103       private var featureCount: Long = 0
104 
105       // noinspection AccessorLikeMethodIsEmptyParen
106       def getFiles(): Seq[URI] = files.asScala.toSeq
107 
108       def write(key: Void, value: SimpleFeature): Unit = featureCount += 1
109 
110       def close(context: TaskAttemptContext): Unit = closeCurrentFile()
111 
112       protected def newWriter(): (Path, RecordWriter[Void, SimpleFeature]) = {
113         closeCurrentFile()
114         // TODO is this in the right fs?
115         currentFile = new Path(workPath, FileSystemStorage.newFilePath(storage.sft.getTypeName))
116         featureCount = 0
117         logger.debug(s"Creating record writer at path $currentFile")
118         // noinspection LanguageFeature
119         (currentFile, delegate.getRecordWriter(context, currentFile))
120       }
121 
122       private def closeCurrentFile(): Unit = {
123         if (currentFile != null) {
124           files.add(currentFile.toUri)
125           counter.increment(featureCount)
126           currentFile = null
127           featureCount = 0
128         }
129       }
130     }
131 
132     private class SinglePartitionState(partition: Partition) extends PartitionState(partition) {
133 
134       private val writer = newWriter()._2
135 
136       override def write(key: Void, value: SimpleFeature): Unit = {
137         writer.write(key, value)
138         super.write(key, value)
139       }
140 
141       override def close(context: TaskAttemptContext): Unit = {
142         writer.close(context)
143         super.close(context)
144       }
145     }
146 
147     private class ChunkedPartitionState(partition: Partition, estimator: FileSizeEstimator, context: TaskAttemptContext)
148         extends PartitionState(partition) {
149 
150       private var count = 0L // number of features written
151       private var total = 0L // sum size of all finished chunks
152       private var remaining = estimator.estimate(0L)
153 
154       logger.debug(s"Initial estimate: $remaining")
155 
156       private var path: Path = _
157       private var writer: RecordWriter[Void, SimpleFeature] = _
158 
159       override def write(key: Void, value: SimpleFeature): Unit = {
160         if (writer == null) {
161           val (p, w) = newWriter()
162           path = p
163           writer = w
164         }
165         writer.write(key, value)
166         count += 1
167         remaining -= 1
168         if (remaining == 0) {
169           // TODO if we can access the underlying parquet writer we can check the size without closing the file
170           writer.close(context)
171           val length = storage.table.io().newInputFile(path.toString).getLength
172           logger.debug(s"File length: $length $path")
173           writer = null
174           // adjust our estimate to account for the actual bytes written
175           total += length
176           estimator.update(total, count)
177           remaining = estimator.estimate(0L)
178           logger.debug(s"New estimate: $remaining")
179         }
180         super.write(key, value)
181       }
182 
183       override def close(context: TaskAttemptContext): Unit = {
184         if (writer != null) {
185           writer.close(context)
186         }
187         super.close(context)
188       }
189     }
190   }
