25 /**
26   * Output format that writes to multiple partition files
27   */
28 class ParquetPartitionOutputFormat extends OutputFormat[Void, SimpleFeature] {
29 
30   private val delegate = new ParquetSimpleFeatureOutputFormat()
31 
32   override def getRecordWriter(context: TaskAttemptContext): RecordWriter[Void, SimpleFeature] =
33     new PartitionSchemeRecordWriter(context)
34 
35   override def getOutputCommitter(context: TaskAttemptContext): OutputCommitter =
36     delegate.getOutputCommitter(context)
37 
38   // same as FileOutputFormat, but doesn't require that output directory doesn't exist
39   override def checkOutputSpecs(job: JobContext): Unit = {
40     // Ensure that the output directory is set
41     val outDir = FileOutputFormat.getOutputPath(job)
42     if (outDir == null) {
43       throw new InvalidJobConfException("Output directory not set")
44     }
45     // get delegation token for outDir's file system
46     TokenCache.obtainTokensForNamenodes(job.getCredentials, Array[Path](outDir), job.getConfiguration)
47   }

49   private class PartitionSchemeRecordWriter(context: TaskAttemptContext)
50       extends RecordWriter[Void, SimpleFeature] with LazyLogging {
51 
52     import StorageConfiguration.Counters.{Features, Group}
53 
54     private val fsc = {
55       val hadoopConf = context.getConfiguration
56       val conf = {
57         val builder = Map.newBuilder[String, String]
58         hadoopConf.forEach(e => builder += e.getKey -> hadoopConf.get(e.getKey)) // use .get to resolve envs
59         builder.result()
60       }
61       val root = StorageConfiguration.getRootPath(hadoopConf)
62       FileSystemContext.create(root, conf)
63     }
64 
65     private val catalog = StorageCatalog(fsc)
66     private val storage = catalog.load(StorageConfiguration.getSftName(context.getConfiguration))
67 
68     private val fileSize = StorageConfiguration.getTargetFileSize(context.getConfiguration)
69 
70     private val counter = context.getCounter(Group, Features)
71     private val cache = scala.collection.mutable.Map.empty[Seq[PartitionKey], PartitionState]
72 
73     private val workPath = delegate.getOutputCommitter(context).asInstanceOf[FileOutputCommitter].getWorkPath
74 
75     private def newState(partitions: Seq[PartitionKey]): PartitionState = {
76       val estimator = fileSize.map(storage.sizer.estimator)
77       estimator match {
78         case None => new SinglePartitionState(Partition(partitions))
79         case Some(e) => new ChunkedPartitionState(Partition(partitions), e, context)
80       }
81     }
82 
83     override def write(key: Void, value: SimpleFeature): Unit = {
84       val partition = storage.schemes.map(_.getPartition(value))
85       val state = cache.getOrElseUpdate(partition, newState(partition))
86       state.write(key, value)
87       counter.increment(1)
88     }
89 
90     override def close(context: TaskAttemptContext): Unit = {
91       cache.foreach { case (partition, state) =>
92         logger.debug(s"Closing writer for $partition")
93         state.close(context)
94         storage.metadata.register(Map(Partition(partition) -> state.getFiles()))
95       }
96       CloseWithLogging(Seq(storage, catalog))
97     }
98 
99     private sealed abstract class PartitionState(partition: Partition) {
100 
101       import scala.collection.JavaConverters._
102 
103       private val files = new CopyOnWriteArrayList[URI]()
104       private val counter = context.getCounter(Group, StorageConfiguration.Counters.partition(partition.toString))
105 
106       private var currentFile: Path = _
107       private var featureCount: Long = 0
108 
109       // noinspection AccessorLikeMethodIsEmptyParen
110       def getFiles(): Seq[URI] = files.asScala.toSeq
111 
112       def write(key: Void, value: SimpleFeature): Unit = featureCount += 1
113 
114       def close(context: TaskAttemptContext): Unit = closeCurrentFile()
115 
116       protected def newWriter(): (Path, RecordWriter[Void, SimpleFeature]) = {
117         closeCurrentFile()
118         // TODO is this in the right fs?
119         currentFile = new Path(workPath, FileSystemStorage.newFilePath(storage.sft.getTypeName))
120         featureCount = 0
121         logger.debug(s"Creating record writer at path $currentFile")
122         // noinspection LanguageFeature
123         (currentFile, delegate.getRecordWriter(context, currentFile))
124       }
125 
126       private def closeCurrentFile(): Unit = {
127         if (currentFile != null) {
128           files.add(currentFile.toUri)
129           counter.increment(featureCount)
130           currentFile = null
131           featureCount = 0
132         }
133       }
134     }
135 
136     private class SinglePartitionState(partition: Partition) extends PartitionState(partition) {
137 
138       private val writer = newWriter()._2
139 
140       override def write(key: Void, value: SimpleFeature): Unit = {
141         writer.write(key, value)
142         super.write(key, value)
143       }
144 
145       override def close(context: TaskAttemptContext): Unit = {
146         writer.close(context)
147         super.close(context)
148       }
149     }

151     private class ChunkedPartitionState(partition: Partition, estimator: FileSizeEstimator, context: TaskAttemptContext)
152         extends PartitionState(partition) {
153 
154       private var count = 0L // number of features written
155       private var total = 0L // sum size of all finished chunks
156       private var remaining = estimator.estimate(0L)
157 
158       logger.debug(s"Initial estimate: $remaining")
159 
160       private var path: Path = _
161       private var writer: RecordWriter[Void, SimpleFeature] = _
162 
163       override def write(key: Void, value: SimpleFeature): Unit = {
164         if (writer == null) {
165           val (p, w) = newWriter()
166           path = p
167           writer = w
168         }
169         writer.write(key, value)
170         count += 1
171         remaining -= 1
172         if (remaining == 0) {
173           // TODO if we can access the underlying parquet writer we can check the size without closing the file
174           writer.close(context)
175           val length = storage.table.io().newInputFile(path.toString).getLength
176           logger.debug(s"File length: $length $path")
177           writer = null
178           // adjust our estimate to account for the actual bytes written
179           total += length
180           estimator.update(total, count)
181           remaining = estimator.estimate(0L)
182           logger.debug(s"New estimate: $remaining")
183         }
184         super.write(key, value)
185       }
186 
187       override def close(context: TaskAttemptContext): Unit = {
188         if (writer != null) {
189           writer.close(context)
190         }
191         super.close(context)
192       }
193     }
194   }
