184   /**
185    * Immutable representation of an envelope
186    *
187    * Note that conversions to/from 'null' envelopes should be handled carefully, as envelopes are considered
188    * null if xmin > xmax, however, when instantiating an envelope it will re-order the coordinates:
189    *
190    * {{{
191    *   val env = new Envelope()
192    *   val copy = new Envelope(env.getMinX, env.getMinY, env.getMaxX, env.getMaxY)
193    *   copy == env // false
194    * }}}
195    *
196    * Thus, ensure that 'null' envelopes are converted to `None` and not directly to a bounds object. See
197    * `PartitionBounds.apply`
198    *
199    * @param attribute index of the attribute being bounded
200    * @param xmin min x dimension
201    * @param ymin min y dimension
202    * @param xmax max x dimension
203    * @param ymax max y dimension
204    */
205   case class SpatialBounds(attribute: Int, xmin: Double, ymin: Double, xmax: Double, ymax: Double) {
206 
207     /**
208      * Calculate the minimal bounds encompassing both bounds
209      *
210      * @param b other bounds
211      * @return
212      */
213     def +(b: SpatialBounds): SpatialBounds = {
214       require(attribute == b.attribute, "Trying to merge bounds from different attributes")
215       SpatialBounds(attribute, math.min(xmin, b.xmin), math.min(ymin, b.ymin), math.max(xmax, b.xmax), math.max(ymax, b.ymax))
216     }
217 
218     /**
219      * Convert to a mutable envelope
220      *
221      * @return
222      */
223     def envelope: Envelope = new Envelope(xmin, xmax, ymin, ymax)
224   }

255      * Json serializer for partitions

257     private object PartitionSerializer extends JsonSerializer[Partition] with JsonDeserializer[Partition] {
258       override def serialize(src: Partition, typeOfSrc: Type, context: JsonSerializationContext): JsonElement = {
259         val array = new JsonArray(src.values.size)
260         src.values.toSeq.sortBy(k => (k.name, k.value)).foreach { value =>
261           array.add(context.serialize(value))
262         }
263         array
264       }
265 
266       override def deserialize(json: JsonElement, typeOfT: Type, context: JsonDeserializationContext): Partition = {
267         val array = json.getAsJsonArray
268         val values = Set.newBuilder[PartitionKey]
269         var i = 0
270         while (i < array.size()) {
271           values += context.deserialize(array.get(i), classOf[PartitionKey])
272           i += 1
273         }
274         Partition(values.result())
275       }
276     }
