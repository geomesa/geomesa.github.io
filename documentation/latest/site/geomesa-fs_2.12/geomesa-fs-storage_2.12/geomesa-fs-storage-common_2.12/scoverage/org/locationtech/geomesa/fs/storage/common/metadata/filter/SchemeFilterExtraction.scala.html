31   protected def getFilters(filter: Filter): Seq[SchemeFilter] = {
32     val iter = schemes.iterator
33     val start = iter.map(_.getRangesForFilter(filter)).collectFirst { case Some(f) => f }
34     val filters = start match {
35       case None =>
36         // filter did not constrain partitions at all
37         Seq(SchemeFilter(filter, Seq.empty, extractGeometries(filter), extractAttributes(filter)))
38 
39       case Some(ranges) =>
40         // set up the initial scheme filters, without any spatial/attribute bounds
41         val initialRanges = ranges.map(r => Seq(r))
42         // add in the remaining partitions
43         val permutations = iter.foldLeft(initialRanges) { case (ranges, scheme) => addPartition(scheme, ranges, filter) }
44         // add in the spatial/attribute bounds based on the remaining filter for each scheme filter
45         val spatialBounds = extractGeometries(filter)
46         val attributeBounds = extractAttributes(filter)
47         permutations.map(ranges => SchemeFilter(filter, ranges, spatialBounds, attributeBounds))
48     }
49 
50     logger.debug(s"Extracted filters from ${ECQL.toCQL(filter)}:\n  ${filters.mkString("\n  ")}")
51     filters
52   }
53 
54   // add the next partition to the scheme filter
55   private def addPartition(scheme: PartitionScheme, ranges: Seq[Seq[PartitionRange]], filter: Filter): Seq[Seq[PartitionRange]] = {
56     val result = Seq.newBuilder[Seq[PartitionRange]]
57     ranges.foreach { filesTableFilter =>
58       scheme.getRangesForFilter(filter) match {
59         case None => result += filesTableFilter
60         case Some(ranges) =>
61           ranges.foreach { range =>
62             result += filesTableFilter :+ range
63           }
64       }
65     }
66     result.result()
67   }
68 
69   private def extractGeometries(filter: Filter): Seq[SpatialOr] = {
70     sft.spatialBounds().flatMap { i =>
71       val ors = FilterHelper.extractGeometries(filter, sft.getDescriptor(i).getLocalName).values.flatMap { g =>
72         SpatialBound(g.getEnvelopeInternal)
73       }
74       if (ors.isEmpty) { None } else { Some(SpatialOr(i, ors)) }
75     }
76   }
77 
78   private def extractAttributes(filter: Filter): Seq[AttributeOr] = {
79     sft.nonSpatialBounds().flatMap { i =>
80       val d = sft.getDescriptor(i)
81       val ors = FilterHelper.extractAttributeBounds(filter, d.getLocalName, d.getType.getBinding).values.map { b =>
82         val lower = b.lower.value.map(AttributeIndexKey.typeEncode).getOrElse("")
83         val upper = b.upper.value.map(AttributeIndexKey.typeEncode).getOrElse("zzz")
84         AttributeBound(lower, upper)
85       }
86       if (ors.isEmpty) { None } else { Some(AttributeOr(i, ors)) }
87     }
88   }
89 }

91 object SchemeFilterExtraction {
92 
93   /**
94    * Predicates extracted from a filter, based on the partition scheme
95    *
96    * @param filter remaining ECQL filter that isn't accounted for with the other bounds
97    * @param partitions list of partition bound predicates (implicit AND between each element in the seq)
98    * @param spatialBounds list of spatial bound predicates (implicit AND between each element in the seq)
99    * @param attributeBounds list of attribute bound predicates (implicit AND between each element in the seq)
100    */
101   case class SchemeFilter(
102     filter: Filter,
103     partitions: Seq[PartitionRange],
104     spatialBounds: Seq[SpatialOr],
105     attributeBounds: Seq[AttributeOr]
106   )
107 
108   case class SpatialOr(attribute: Int, bounds: Seq[SpatialBound])
109   case class AttributeOr(attribute: Int, bounds: Seq[AttributeBound])
110 
111   case class SpatialBound(xmin: Double, ymin: Double, xmax: Double, ymax: Double) {
112     def intersects(other: SpatialBounds): Boolean =
113       other.xmin <= xmax && other.xmax >= xmin && other.ymin <= ymax && other.ymax >= ymin
114   }
115 
116   object SpatialBound {
117 
118     /**
119      * Converts an envelope to a bounds, handling 'null' (empty) envelopes
120      *
121      * @param env envelope
122      * @return
123      */
124     def apply(env: Envelope): Option[SpatialBound] =
125       if (env.isNull) { None } else { Some(SpatialBound(env.getMinX, env.getMinY, env.getMaxX, env.getMaxY)) }
126   }
127 
128   case class AttributeBound(lower: String, upper: String) {
129     def intersects(other: AttributeBounds): Boolean = other.lower <= upper && other.upper >= lower
130   }
