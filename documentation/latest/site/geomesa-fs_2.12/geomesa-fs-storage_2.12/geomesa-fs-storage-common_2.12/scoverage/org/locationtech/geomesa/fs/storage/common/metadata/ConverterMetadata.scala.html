23 /**
24  * Synthetic metadata that just lists files on the filesystem
25  *
26  * @param context file system
27  * @param sft simple feature type
28  * @param orderedSchemes partition schemes, ordered according to the file system directory layout
29  * @param leafStorage if true, the final partition scheme is a prefix of the file name, instead of a regular directory
30  */
31 class ConverterMetadata(
32     context: FileSystemContext,
33     val sft: SimpleFeatureType,
34     val orderedSchemes: Seq[PartitionScheme],
35     val leafStorage: Boolean
36   ) extends StorageMetadata with CachedMetadata with LazyLogging {
37 
38   override val `type`: String = ConverterMetadata.MetadataType
39 
40   override val schemes: Set[PartitionScheme] = orderedSchemes.toSet
41 
42   private val depth = orderedSchemes.map {
43     case s: HierarchicalDateTimeScheme => s.depth
44     case _ => 1
45   }.sum - (if (leafStorage) { 1 } else { 0 })
46 
47   filesCache.refresh(BoxedUnit.UNIT) // kick off the initial load asynchronously
48 
49   override def addFile(file: StorageFile): Unit = throw new UnsupportedOperationException()
50   override def removeFile(file: StorageFile): Unit = throw new UnsupportedOperationException()
51   override def replaceFiles(existing: Seq[StorageFile], replacements: Seq[StorageFile]): Unit =
52     throw new UnsupportedOperationException()
53 
54   override protected def buildFileList(): Seq[StorageFile] = {
55     logger.debug("Building file list")
56     val start = System.currentTimeMillis()
57     val result = Seq.newBuilder[StorageFile]
58     try {
59       val toCheck = new java.util.LinkedList[FileStatus]()
60       PathCache.list(context.fs, context.root).foreach(toCheck.add)
61       val maxDepth = depth + context.root.depth()
62       val rootPathLength = context.fs.resolvePath(context.root).toString.length
63       while (!toCheck.isEmpty) {
64         val status = toCheck.remove()
65         logger.debug(s"Checking ${status.getPath}")
66         if (status.isDirectory) {
67           if (status.getPath.depth() <= maxDepth) {
68             PathCache.list(context.fs, status.getPath).foreach(toCheck.add)
69           }
70         } else {
71           val relativePath = status.getPath.toString.substring(rootPathLength + 1)
72           var partitionParts = relativePath.split("/")
73           if (leafStorage) {
74             partitionParts = partitionParts :+ new Path(relativePath).getName.takeWhile(_ != '_')
75           }
76           var i = 0
77           val partitions = orderedSchemes.map {
78             case s: HierarchicalDateTimeScheme =>
79               val key = PartitionKey(s.name, partitionParts.slice(i, i + s.depth).mkString("/"))
80               i += s.depth
81               key
82             case s =>
83               val key = PartitionKey(s.name, partitionParts(i))
84               i += 1
85               key
86           }
87           result += StorageFile(relativePath, Partition(partitions.toSet), -1L)
88         }
89       }
90     } catch {
91       case NonFatal(e) => logger.error("Error building file list:", e); throw e
92     }
93     val list = result.result()
94     logger.debug(s"Found ${list.size} files in ${System.currentTimeMillis() - start}ms")
95     logger.whenTraceEnabled(list.foreach(f => logger.trace(f.toString)))
96     list
97   }
98 }
