19 class HashScheme[T](
20     attribute: String,
21     index: Int,
22     buckets: Int,
23     hasher: MurmurHashFunction.Hashing[T],
24   ) extends PartitionScheme {
25 
26   import FilterHelper.ff
27 
28   override val name: String = s"${HashScheme.Name}:attribute=$attribute:buckets=$buckets"
29 
30   private val format = s"%0${(buckets - 1).toString.length}d"
31   private val default = PartitionKey(name, format.format(0))
32 
33   override def getPartition(feature: SimpleFeature): PartitionKey = {
34     val value = feature.getAttribute(index)
35     if (value == null) {
36       default
37     } else {
38       PartitionKey(name, toPartition(value.asInstanceOf[T]))
39     }
40   }
41 
42   override def getRangesForFilter(filter: Filter): Option[Seq[PartitionRange]] = {
43     getBounds(filter).map { bounds =>
44       val builder = new RangeBuilder()
45       bounds.foreach { bound =>
46         // note: these are all equality values, as determined in getBounds
47         val lower = toPartition(bound.lower.value.get)
48         val upper = lower + ZeroChar
49         builder += PartitionRange(name, lower, upper)
50       }
51       builder.result()
52     }
53   }
54 
55   override def getPartitionsForFilter(filter: Filter): Option[Seq[PartitionKey]] = {
56     getBounds(filter).map { bounds =>
57       bounds.map { bound =>
58         // note: these are all equality values, as determined in getBounds
59         PartitionKey(name, toPartition(bound.lower.value.get))
60       }
61     }
62   }
63 
64   private def getBounds(filter: Filter): Option[Seq[Bounds[T]]] = {
65     val bounds = FilterHelper.extractAttributeBounds(filter, attribute, hasher.binding)
66     if (bounds.isEmpty || bounds.exists(_.isRange)) {
67       None
68     } else if (bounds.disjoint) {
69       Some(Seq.empty)
70     } else {
71       Some(bounds.values)
72     }
73   }
74 
75   private def toPartition(value: T): String = format.format((hasher(value) & Int.MaxValue) % buckets)
76 
77   override def getCoveringFilter(partition: PartitionKey): Filter = {
78     val fn = BucketHashFunction.Name.getFunctionName
79     ff.equals(ff.function(fn, ff.property(attribute), ff.literal(buckets)), ff.literal(partition.value.toInt))
80   }
81 }

87   class HashPartitionSchemeFactory extends PartitionSchemeFactory {
88     override def load(sft: SimpleFeatureType, scheme: String): Option[PartitionScheme] = {
89       val opts = SchemeOpts(scheme)
90       if (opts.name != Name) { None } else {
91         val attribute = opts.getSingle("attribute").orNull
92         require(attribute != null, s"Hash scheme requires an attribute to be specified with 'attribute=<attribute>'")
93         val index = attributeIndex(sft, attribute)
94         val binding = sft.getDescriptor(index).getType.getBinding
95         val hasher = MurmurHashFunction.Hashers.find(_.binding.isAssignableFrom(binding)).orNull
96         require(hasher != null, s"Invalid type binding '${binding.getName}' of attribute '$attribute'")
97         val buckets = opts.getSingle("buckets").orNull
98         require(buckets != null, s"Hash scheme requires a number of buckets to be specified with 'buckets=<n>'")
99         Some(new HashScheme(attribute, index, buckets.toInt, hasher))
100       }
101     }
102   }
