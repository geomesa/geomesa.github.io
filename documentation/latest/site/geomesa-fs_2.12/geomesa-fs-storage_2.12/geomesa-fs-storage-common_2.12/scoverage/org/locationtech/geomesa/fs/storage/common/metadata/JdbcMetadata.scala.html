31 /**
32  * Storage metadata implementation backed by a SQL database. Currently compatible with Postgres - other
33  * databases will likely have incompatibilities in the SQL syntax.
34  *
35  * Schema consists of five tables (note: `storage_` is the default table prefix, but may be customized):
36  *
37  * `storage_meta`
38  *
39  * * Holds the base metadata (simple feature type, partition scheme, encoding) as a serialized clob
40  *
41  * ** root varchar(256) not null
42  * ** typeName varchar(256) not null
43  * ** meta text not null
44  * ** primary key (root, typeName)
45  *
46  * `storage_files`
47  *
48  * * Holds file-level metadata for each storage file
49  *
50  * ** id bigint primary key (generated identity)
51  * ** root varchar(256) not null
52  * ** file varchar(256) not null
53  * ** count bigint not null
54  * ** action char(1) not null (A=Append, M=Modify, D=Delete)
55  * ** sort integer[]
56  * ** ts timestamp without time zone not null
57  *
58  * `storage_partitions`
59  *
60  * * Holds partition key-value pairs for each file
61  *
62  * ** file_id bigint not null (foreign key to storage_files)
63  * ** name varchar(64) not null
64  * ** value varchar(64) not null
65  * ** primary key (file_id, name)
66  *
67  * `storage_spatial_bounds`
68  *
69  * * Holds spatial bounds for each file by attribute
70  *
71  * ** file_id bigint not null (foreign key to storage_files)
72  * ** attribute smallint not null
73  * ** x_min double precision
74  * ** x_max double precision
75  * ** y_min double precision
76  * ** y_max double precision
77  * ** primary key (file_id, attribute)
78  *
79  * `storage_attr_bounds`
80  *
81  * * Holds attribute bounds for each file by attribute
82  *
83  * ** file_id bigint not null (foreign key to storage_files)
84  * ** attribute smallint not null
85  * ** lower text
86  * ** upper text
87  * ** primary key (file_id, attribute)
88  *
89  * @param pool connection pool
90  * @param root storage root path
91  * @param schema database schema name
92  * @param tablePrefix table name prefix
93  * @param meta basic metadata config
94  **/

185   override def getFiles(filter: Filter): Seq[StorageFile] = {
186     if (filter == Filter.INCLUDE) {
187       getFiles()
188     } else {
189       val filters = getFilters(filter)
190       if (filters.isEmpty) {
191         Seq.empty // no intersecting partitions
192       } else {
193         WithClose(pool.getConnection()) { cx =>
194           filters.flatMap { f =>
195             filesTable.select(cx, root, f.partitions, f.spatialBounds, f.attributeBounds)
196           }
197         }
198       }
199     }
200   }

271     def create(connection: Connection): Unit = {
272       WithClose(connection.createStatement()) { st =>

287         st.setString(1, root)
288         st.setString(2, metadata.sft.getTypeName)
289         st.setString(3, new String(serialized.toByteArray, StandardCharsets.UTF_8))

291       }

344       WithClose(cx.createStatement()) { st =>
345         st.executeUpdate(
346           s"""CREATE TABLE IF NOT EXISTS $filesTable (
347              |  id BIGINT PRIMARY KEY GENERATED BY DEFAULT AS IDENTITY,
348              |  root VARCHAR(256) NOT NULL,
349              |  file VARCHAR(256) NOT NULL,
350              |  count BIGINT NOT NULL,
351              |  action CHAR(1) NOT NULL,
352              |  sort INTEGER[],
353              |  ts TIMESTAMP WITHOUT TIME ZONE NOT NULL

357           s"""CREATE TABLE IF NOT EXISTS $partitionsTable (
358              |  file_id BIGINT NOT NULL,
359              |  name VARCHAR(64) NOT NULL,
360              |  value VARCHAR(64) NOT NULL,
361              |  PRIMARY KEY (file_id, name),
362              |  CONSTRAINT fk_storage_file

366              |);""".stripMargin
367         )
368         st.executeUpdate(
369           s"""CREATE TABLE IF NOT EXISTS $spatialBoundsTable (
370              |  file_id BIGINT NOT NULL,
371              |  attribute SMALLINT NOT NULL,
372              |  x_min DOUBLE PRECISION,
373              |  x_max DOUBLE PRECISION,
374              |  y_min DOUBLE PRECISION,
375              |  y_max DOUBLE PRECISION,
376              |  PRIMARY KEY (file_id, attribute),
377              |  CONSTRAINT fk_storage_file
378              |    FOREIGN KEY (file_id)

380              |    ON DELETE CASCADE

382         )
383         st.executeUpdate(

385              |  file_id BIGINT NOT NULL,
386              |  attribute SMALLINT NOT NULL,
387              |  lower TEXT,
388              |  upper TEXT,
389              |  PRIMARY KEY (file_id, attribute),
390              |  CONSTRAINT fk_storage_file
391              |    FOREIGN KEY (file_id)

398         st.executeUpdate(s"""CREATE INDEX IF NOT EXISTS ${tablePrefix}spatial_bounds_idx_bounds ON $spatialBoundsTable(attribute, x_min, x_max, y_min, y_max);""")

402     def insert(cx: Connection, root: String, file: StorageFile): Unit = {
403       val id = WithClose(cx.prepareStatement(
404         s"INSERT INTO $filesTable (root, file, count, action, sort, ts) " +
405           "VALUES (?, ?, ?, ?, ?, ?) RETURNING id", Statement.RETURN_GENERATED_KEYS)) { st =>
406         st.setString(1, root)
407         st.setString(2, file.file)
408         st.setLong(3, file.count)
409         st.setString(4, file.action.toString.substring(0, 1))
410         if (file.sort.isEmpty) {
411           st.setNull(5, java.sql.Types.ARRAY)
412         } else {
413           st.setArray(5, cx.createArrayOf("integer", file.sort.map(Int.box).toArray[AnyRef]))
414         }
415         st.setTimestamp(6, Timestamp.from(Instant.ofEpochMilli(file.timestamp)))
416         st.executeUpdate()
417         WithClose(st.getGeneratedKeys) { rs =>
418           if (rs.next()) {
419             rs.getLong(1)
420           } else {
421             throw new RuntimeException("Failed to retrieve generated key")
422           }
423         }
424       }
425       WithClose(cx.prepareStatement(s"INSERT INTO $partitionsTable (file_id, name, value) VALUES (?, ?, ?)")) { st =>
426         file.partition.values.foreach { p =>
427           st.setLong(1, id)
428           st.setString(2, p.name)
429           st.setString(3, p.value)
430           st.executeUpdate()
431         }
432       }
433       if (file.spatialBounds.nonEmpty) {
434         WithClose(cx.prepareStatement(s"INSERT INTO $spatialBoundsTable (file_id, attribute, x_min, x_max, y_min, y_max) VALUES (?, ?, ?, ?, ?, ?)")) { st =>
435           file.spatialBounds.foreach { bounds =>
436             st.setLong(1, id)
437             st.setInt(2, bounds.attribute)
438             st.setDouble(3, bounds.xmin)
439             st.setDouble(4, bounds.xmax)
440             st.setDouble(5, bounds.ymin)
441             st.setDouble(6, bounds.ymax)
442             st.executeUpdate()
443           }
444         }
445       }
446       if (file.attributeBounds.nonEmpty) {
447         WithClose(cx.prepareStatement(s"INSERT INTO $attrBoundsTable (file_id, attribute, lower, upper) VALUES (?, ?, ?, ?)")) { st =>
448           file.attributeBounds.foreach { bounds =>
449             st.setLong(1, id)
450             st.setInt(2, bounds.attribute)
451             st.setString(3, bounds.lower)
452             st.setString(4, bounds.upper)
453             st.executeUpdate()
454           }
455         }
456       }
457     }

467     def select(
468         cx: Connection,
469         root: String,
470         partitions: Seq[PartitionRange],
471         spatialBounds: Seq[SpatialOr],
472         attributeBounds: Seq[AttributeOr],
473       ): Seq[StorageFile] = {
474 
475       // build query with multiple joins - one for each partition filter (AND logic)
476       val partitionJoins = partitions.zipWithIndex.map {
477         case (PartitionRange(name, lower, upper), i) => PartitionJoin(name, lower, upper, s"sp_filter_$i")
478       }
479 
480       // build spatial bound filters - one LEFT JOIN per attribute with OR logic for bounds
481       val spatialBoundJoins = spatialBounds.zipWithIndex.map {
482         case (SpatialOr(attribute, bounds), i) => SpatialBoundsJoin(attribute, bounds, s"sb_filter_$i")
483       }
484 
485       // build attribute bound filters - one LEFT JOIN per attribute with OR logic for bounds
486       val attrBoundJoins = attributeBounds.zipWithIndex.map {
487         case (AttributeOr(attribute, bounds), i) => AttributeBoundsJoin(attribute, bounds, s"ab_filter_$i")
488       }
489 
490       val joinClause =
491         partitionJoins.map { j =>
492           s"JOIN $partitionsTable ${j.tableAlias} ON sf.id = ${j.tableAlias}.file_id AND ${j.onClause}"
493         } ++
494         spatialBoundJoins.map { j =>
495           s"LEFT JOIN $spatialBoundsTable ${j.tableAlias} ON sf.id = ${j.tableAlias}.file_id AND ${j.tableAlias}.attribute = ?"
496         } ++
497         attrBoundJoins.map { j =>
498           s"LEFT JOIN $attrBoundsTable ${j.tableAlias} ON sf.id = ${j.tableAlias}.file_id AND ${j.tableAlias}.attribute = ?"
499         }
500 
501       // build WHERE clause for spatial and attribute bounds
502       // if bounds don't exist (NULL), we count it as a match
503       val spatialWhereClause = spatialBoundJoins.map { j =>
504         s"(${j.tableAlias}.file_id IS NULL OR ${j.whereClause})"
505       }
506       val attrWhereClause = attrBoundJoins.map { j =>
507         s"(${j.tableAlias}.file_id IS NULL OR ${j.whereClause})"
508       }
509       val allWhereClauses = spatialWhereClause ++ attrWhereClause
510 
511       // note: all child tables (partitions, spatial bounds, attribute bounds) are pre-aggregated
512       // in subqueries. This ensures each file returns exactly one row with no Cartesian products.
513       // The sp_filter, sb_filter, and ab_filter joins are used for filtering and must remain on the raw tables.
514       val whereClause = if (allWhereClauses.isEmpty) {
515         "WHERE sf.root = ?"
516       } else {
517         s"WHERE sf.root = ? AND ${allWhereClauses.mkString(" AND ")}"
518       }
519 
520       val query =
521         s"""SELECT sf.id, sf.file, sf.count, sf.action, sf.sort, sf.ts,
522            |  sp.partition_names, sp.partition_values,
523            |  sb.sb_attributes, sb.sb_x_mins, sb.sb_x_maxs, sb.sb_y_mins, sb.sb_y_maxs,
524            |  ab.ab_attributes, ab.ab_lowers, ab.ab_uppers
525            |FROM $filesTable sf
526            |${joinClause.mkString("\n")}
527            |LEFT JOIN (
528            |  SELECT file_id,
529            |    array_agg(name ORDER BY name) as partition_names,
530            |    array_agg(value ORDER BY name) as partition_values
531            |  FROM $partitionsTable
532            |  GROUP BY file_id
533            |) sp ON sf.id = sp.file_id
534            |LEFT JOIN (
535            |  SELECT file_id,
536            |    array_agg(attribute ORDER BY attribute) as sb_attributes,
537            |    array_agg(x_min ORDER BY attribute) as sb_x_mins,
538            |    array_agg(x_max ORDER BY attribute) as sb_x_maxs,
539            |    array_agg(y_min ORDER BY attribute) as sb_y_mins,
540            |    array_agg(y_max ORDER BY attribute) as sb_y_maxs
541            |  FROM $spatialBoundsTable
542            |  GROUP BY file_id
543            |) sb ON sf.id = sb.file_id
544            |LEFT JOIN (
545            |  SELECT file_id,
546            |    array_agg(attribute ORDER BY attribute) as ab_attributes,
547            |    array_agg(lower ORDER BY attribute) as ab_lowers,
548            |    array_agg(upper ORDER BY attribute) as ab_uppers
549            |  FROM $attrBoundsTable
550            |  GROUP BY file_id
551            |) ab ON sf.id = ab.file_id
552            |$whereClause
553            |ORDER BY sf.id DESC""".stripMargin
554 
555       WithClose(cx.prepareStatement(query)) { st =>
556         var paramIndex = 1
557         // set partition join parameters
558         partitionJoins.foreach { join =>
559           paramIndex += join.apply(st, paramIndex)
560         }
561         // set spatial bound join parameters (attribute IDs in JOIN clauses)
562         spatialBoundJoins.foreach { join =>
563           st.setInt(paramIndex, join.attribute)
564           paramIndex += 1
565         }
566         // set attribute bound join parameters (attribute IDs in JOIN clauses)
567         attrBoundJoins.foreach { join =>
568           st.setInt(paramIndex, join.attribute)
569           paramIndex += 1
570         }
571         // set root parameter
572         st.setString(paramIndex, root)
573         paramIndex += 1
574         // set spatial bound WHERE clause parameters
575         spatialBoundJoins.foreach { join =>
576           paramIndex += join.apply(st, paramIndex)
577         }
578         // set attribute bound WHERE clause parameters
579         attrBoundJoins.foreach { join =>
580           paramIndex += join.apply(st, paramIndex)
581         }
582         WithClose(st.executeQuery())(toStorageFiles)
583       }
584     }

586     private def toStorageFiles(rs: ResultSet): Seq[StorageFile] = {
587       val result = Seq.newBuilder[StorageFile]
588 
589       while (rs.next()) {
590         val file = rs.getString(2)
591         val count = rs.getLong(3)
592         val action = rs.getString(4).charAt(0) match {
593           case 'A' => StorageMetadata.StorageFileAction.Append
594           case 'M' => StorageMetadata.StorageFileAction.Modify
595           case 'D' => StorageMetadata.StorageFileAction.Delete
596           case c => throw new IllegalStateException(s"Unknown action: $c")
597         }
598         val sort = Option(rs.getArray(5)).fold(Seq.empty[Int])(_.getArray.asInstanceOf[Array[Integer]].map(_.intValue()).toSeq)
599         val timestamp = rs.getTimestamp(6).toInstant.toEpochMilli
600
