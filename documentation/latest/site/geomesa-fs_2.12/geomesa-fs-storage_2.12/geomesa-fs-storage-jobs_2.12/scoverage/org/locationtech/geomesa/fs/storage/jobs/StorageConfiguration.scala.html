129   /**
130     * Key used for merging feature updates.
131     *
132     * Implements hadoop writable for m/r, kryo serializable for spark, and comparable to sort in
133     * reverse chronological order
134     */
135   class SimpleFeatureAction extends Writable with KryoSerializable with Comparable[SimpleFeatureAction] {
136 
137     private var _id: String = _
138     private var _timestamp: Long = _
139     private var _action: StorageFileAction = _
140 
141     def this(id: String, timestamp: Long, action: StorageFileAction) = {
142       this()
143       this._id = id
144       this._timestamp = timestamp
145       this._action = action
146     }
147 
148     def id: String = _id
149     def timestamp: Long = _timestamp
150     def action: StorageFileAction = _action
151 
152     override def compareTo(o: SimpleFeatureAction): Int = {
153       var res = _id.compareTo(o.id)
154       if (res == 0) {
155         res = _timestamp.compareTo(o.timestamp) * -1 // note: reverse chronological sort
156         if (res == 0) {
157           res = _action.compareTo(o.action)
158         }
159       }
160       res
161     }
162 
163     override def write(out: DataOutput): Unit = {
164       out.writeUTF(_id)
165       out.writeLong(_timestamp)
166       out.writeUTF(_action.toString)
167     }
168 
169     override def readFields(in: DataInput): Unit = {
170       _id = in.readUTF()
171       _timestamp = in.readLong()
172       _action = StorageFileAction.withName(in.readUTF())
173     }
174 
175     override def write(kryo: Kryo, output: Output): Unit = {
176       output.writeString(_id)
177       output.writeLong(_timestamp)
178       output.writeString(_action.toString)
179     }
180 
181     override def read(kryo: Kryo, input: Input): Unit = {
182       _id = input.readString()
183       _timestamp = input.readLong()
184       _action = StorageFileAction.withName(input.readString())
185     }
