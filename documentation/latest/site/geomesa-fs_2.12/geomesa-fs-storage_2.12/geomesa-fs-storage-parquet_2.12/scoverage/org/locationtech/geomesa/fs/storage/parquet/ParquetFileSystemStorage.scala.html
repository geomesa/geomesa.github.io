38 class ParquetFileSystemStorage(context: FileSystemContext, metadata: StorageMetadata)
39     extends FileSystemStorage(context, metadata, ParquetFileSystemStorage.FileExtension) {
40 
41   import scala.collection.JavaConverters._
42 
43   private val authProvider: AuthorizationsProvider =
44     AuthUtils.getProvider(
45       context.conf.get(AuthProviderParam.key).map(p => AuthProviderParam.key -> p).toMap.asJava,
46       context.conf.getOrElse(AuthsParam.key, "").split(",").toSeq.filter(_.nonEmpty)
47     )
48 
49   override val encoding: String = ParquetFileSystemStorage.Encoding
50 
51   override protected def createWriter(file: URI, partition: Partition, observer: FileSystemObserver): FileSystemWriter = {
52     val conf = context.conf ++ Map(SimpleFeatureParquetSchema.PartitionKey -> partition.toString)
53 
54     val observers =
55       if (FileValidationEnabled.toBoolean.get) {
56         CompositeObserver(Seq(observer, FileValidationObserver(file)))
57       } else {
58         observer
59       }
60     new ParquetFileSystemWriter(fs, conf, metadata.sft, file, observers)
61   }
62 
63   override protected def createReader(
64       filter: Option[Filter],
65       transform: Option[(String, SimpleFeatureType)]): FileSystemPathReader = {
66     // readSft has all the fields needed for filtering and return
67     val ReadSchema(readSft, readTransform) = ReadSchema(metadata.sft, filter, transform)
68     val ReadFilter(fc, residualFilter) = ReadFilter(readSft, filter)
69     val parquetFilter = fc.map(FilterCompat.get).getOrElse(FilterCompat.NOOP)
70     val gtFilter = residualFilter.map(FastFilterFactory.optimize(readSft, _))
71     val visFilter = VisibilityUtils.visible(authProvider)
72 
73     logger.debug(
74       s"    Parquet filter: ${parquetFilter match { case f: FilterPredicateCompat => f.getFilterPredicate; case f => f }} " +
75         s"and modified gt filter: ${gtFilter.getOrElse(Filter.INCLUDE)}")
76 
77     new ParquetFileSystemReader(fs, context, readSft, parquetFilter, gtFilter, visFilter, readTransform)
78   }
79 }

81 object ParquetFileSystemStorage extends LazyLogging {
82 
83   val Encoding = "parquet"

85 
86   val ParquetCompressionOpt = "parquet.compression"

88   /**
89    * Validate a file by reading it back
90    *
91    * @param file file to validate
92    */
93   case class FileValidationObserver(file: URI) extends FileSystemObserver {
94     override def apply(feature: SimpleFeature): Unit = {}
95     override def flush(): Unit = {}
96     override def close(): Unit = {
97       try {
98         WithClose(ParquetReader.builder(new GroupReadSupport(), new Path(file)).build()) { reader =>
99           var record = reader.read()
100           while (record != null) {
101             // Process the record
102             record = reader.read()
103           }
104           logger.trace(s"$file is a valid Parquet file")
105         }
106       } catch {
107         case NonFatal(e) => throw new RuntimeException(s"File appears to be corrupted: $file", e)
108       }
109     }
110   }
