35   private def reduce(
36       sft: SimpleFeatureType
37     )(result: (Option[FilterPredicate], Option[Filter]),
38       name: String): (Option[FilterPredicate], Option[Filter]) = {
39     val (parquet, geotools) = result
40     val filter = geotools.orNull
41     if (filter == null) {
42       return result // no more filter to evaluate
43     }
44 
45     val bindings = ObjectType.selectType(sft.getDescriptor(name))
46     val col = StringSerialization.alphaNumericSafeString(name)
47 
48     val (predicate, remaining): (Option[FilterPredicate], Option[Filter]) = bindings.head match {
49       // note: non-points use repeated values, which aren't supported in parquet predicates
50       case ObjectType.GEOMETRY => spatial(sft, name, filter, col, bindings.last)
51       case ObjectType.DATE     => attribute(sft, name, filter, FilterApi.longColumn(col), toMicros)
52       case ObjectType.STRING   => attribute(sft, name, filter, FilterApi.binaryColumn(col), Binary.fromString)
53       case ObjectType.INT      => attribute(sft, name, filter, FilterApi.intColumn(col), identity[java.lang.Integer])
54       case ObjectType.LONG     => attribute(sft, name, filter, FilterApi.longColumn(col), identity[java.lang.Long])
55       case ObjectType.FLOAT    => attribute(sft, name, filter, FilterApi.floatColumn(col), identity[java.lang.Float])
56       case ObjectType.DOUBLE   => attribute(sft, name, filter, FilterApi.doubleColumn(col), identity[java.lang.Double])
57       case ObjectType.BOOLEAN  => boolean(sft, name, filter, FilterApi.booleanColumn(col))
58       case _ => (None, Some(filter))
59     }
60 
61     ((predicate.toSeq ++ parquet).reduceLeftOption(FilterApi.and), remaining)
62   }

64   private def spatial(
65       sft: SimpleFeatureType,
66       name: String,
67       filter: Filter,
68       col: String,
69       typed: ObjectType): (Option[FilterPredicate], Option[Filter]) = {
70     val (spatial, _) = FilterExtractingVisitor(filter, name, sft, SpatialFilterStrategy.spatialCheck)
71     val xyBounds = spatial.map(FilterHelper.extractGeometries(_, name)).flatMap { extracted =>
72       Some(extracted).filter(e => e.nonEmpty && !e.disjoint).map { e =>
73         e.values.map(GeometryUtils.bounds).reduce { (a, b) =>
74           (math.min(a._1, b._1), math.min(a._2, b._2), math.max(a._3, b._3), math.max(a._4, b._4))
75         }
76       }
77     }
78 
79     val predicate = xyBounds.map { case (xmin, ymin, xmax, ymax) =>
80       // filter against the bbox field
81       val bboxGroup = BoundingBoxField(col, encoded = true).bbox
82       val xminCol = FilterApi.floatColumn(s"$bboxGroup.${BoundingBoxField.XMin}")
83       val yminCol = FilterApi.floatColumn(s"$bboxGroup.${BoundingBoxField.YMin}")
84       val xmaxCol = FilterApi.floatColumn(s"$bboxGroup.${BoundingBoxField.XMax}")
85       val ymaxCol = FilterApi.floatColumn(s"$bboxGroup.${BoundingBoxField.YMax}")
86       Seq[FilterPredicate](
87         FilterApi.ltEq(xminCol, Float.box(xmax.toFloat)),
88         FilterApi.gtEq(xmaxCol, Float.box(xmin.toFloat)),
89         FilterApi.ltEq(yminCol, Float.box(ymax.toFloat)),
90         FilterApi.gtEq(ymaxCol, Float.box(ymin.toFloat))
91       ).reduce(FilterApi.and)
92     }
93     // since we don't know what the actual file encoding is up front, we always have to evaluate the full predicate post-read
94     (predicate, Some(filter))
95   }

97   private def attribute[T : ClassTag, U <: Comparable[U]](
98       sft: SimpleFeatureType,
99       name: String,
100       filter: Filter,
101       col: Column[U] with SupportsLtGt,
102       conversion: T => U): (Option[FilterPredicate], Option[Filter]) = {
103     val (attribute, nonAttribute) = FilterExtractingVisitor(filter, name, sft)
104     val binding = implicitly[ClassTag[T]].runtimeClass.asInstanceOf[Class[T]]
105     val bounds = attribute.map(FilterHelper.extractAttributeBounds(_, name, binding))
106     val predicate = bounds.flatMap { extracted =>
107       Some(extracted).filter(e => e.nonEmpty && !e.disjoint && e.values.forall(_.isBounded)).map { e =>
108         val filters = e.values.map { bounds =>
109           if (bounds.isEquals) {
110             FilterApi.eq(col, conversion(bounds.lower.value.get))
111           } else {
112             val lower: Option[FilterPredicate] = bounds.lower.value.map { value =>
113               val converted = conversion(value)
114               if (bounds.lower.inclusive) { FilterApi.gtEq(col, converted) } else { FilterApi.gt(col, converted) }
115             }
116             val upper: Option[FilterPredicate] = bounds.upper.value.map { value =>
117               val converted = conversion(value)
118               if (bounds.upper.inclusive) { FilterApi.ltEq(col, converted) } else { FilterApi.lt(col, converted) }
119             }
120             (lower, upper) match {
121               case (Some(lo), Some(hi)) => FilterApi.and(lo, hi)
122               case (Some(f), None) => f
123               case (None, Some(f)) => f
124               case (None, None) => throw new IllegalStateException() // shouldn't happen due to checks above
125             }
126           }
127         }
128         merge(filters)
129       }
130     }
131     val remaining = if (bounds.exists(_.precise)) { nonAttribute } else { Some(filter) }
132     (predicate, remaining)
133   }

153   /**
154     * Merge OR'd filters
155     *
156     * Detect and re-write "not equals" filters to handle null values
157     *
158     * FilterHelper methods make "foo != x" comes out as "foo < x OR foo > x"
159    *
160     * that won't return null values - not normally a concern since we don't index them in key-value dbs,
161     * but we want to handle that case here
162     *
163     * @param filters filters
164     * @return combined filter
165     */
166   private def merge[T <: Comparable[T]](filters: Seq[FilterPredicate]): FilterPredicate = {
167     lazy val values = filters.collect {
168       case f: Lt[T] => (f.getColumn.asInstanceOf[Column[T] with SupportsEqNotEq], f.getValue)
169       case f: Gt[T] => (f.getColumn.asInstanceOf[Column[T] with SupportsEqNotEq], f.getValue)
170     }.distinct
171 
172     if (filters.lengthCompare(2) == 0 &&
173         filters.exists(_.isInstanceOf[Lt[_]]) &&
174         filters.exists(_.isInstanceOf[Gt[_]]) &&
175         values.lengthCompare(1) == 0) {
176       FilterApi.notEq(values.head._1, values.head._2)
177     } else {
178       filters.reduce(FilterApi.or)
179     }
180   }
