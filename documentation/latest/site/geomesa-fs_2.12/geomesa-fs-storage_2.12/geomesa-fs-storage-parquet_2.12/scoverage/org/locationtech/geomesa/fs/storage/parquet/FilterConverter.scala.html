37   private def reduce(
38       sft: SimpleFeatureType
39     )(result: (Option[FilterPredicate], Option[Filter]),
40       name: String): (Option[FilterPredicate], Option[Filter]) = {
41     val (parquet, geotools) = result
42     val filter = geotools.orNull
43     if (filter == null) {
44       return result // no more filter to evaluate
45     }
46 
47     val bindings = ObjectType.selectType(sft.getDescriptor(name))
48     val col = StringSerialization.alphaNumericSafeString(name)
49 
50     val (predicate, remaining): (Option[FilterPredicate], Option[Filter]) = bindings.head match {
51       // note: non-points use repeated values, which aren't supported in parquet predicates
52       case ObjectType.GEOMETRY => spatial(sft, name, filter, col, bindings.last)
53       case ObjectType.DATE     => attribute(sft, name, filter, FilterApi.longColumn(col), toMicros)
54       case ObjectType.STRING   => attribute(sft, name, filter, FilterApi.binaryColumn(col), Binary.fromString)
55       case ObjectType.INT      => attribute(sft, name, filter, FilterApi.intColumn(col), identity[java.lang.Integer])
56       case ObjectType.LONG     => attribute(sft, name, filter, FilterApi.longColumn(col), identity[java.lang.Long])
57       case ObjectType.FLOAT    => attribute(sft, name, filter, FilterApi.floatColumn(col), identity[java.lang.Float])
58       case ObjectType.DOUBLE   => attribute(sft, name, filter, FilterApi.doubleColumn(col), identity[java.lang.Double])
59       case ObjectType.BOOLEAN  => boolean(sft, name, filter, FilterApi.booleanColumn(col))
60       case _ => (None, Some(filter))
61     }
62 
63     ((predicate.toSeq ++ parquet).reduceLeftOption(FilterApi.and), remaining)
64   }

66   private def spatial(
67       sft: SimpleFeatureType,
68       name: String,
69       filter: Filter,
70       col: String,
71       typed: ObjectType): (Option[FilterPredicate], Option[Filter]) = {
72     val (spatial, _) = FilterExtractingVisitor(filter, name, sft, SpatialFilterStrategy.spatialCheck)
73     val predicate = spatial.map(FilterHelper.extractGeometries(_, name)).flatMap { extracted =>
74       Some(extracted).filter(e => e.nonEmpty && !e.disjoint).map { e =>
75         // compute our ranges based on the coarse bounds for our query
76         val multiplier = QueryProperties.PolygonDecompMultiplier.toInt.get
77         val bits = QueryProperties.PolygonDecompBits.toInt.get
78         val bounds = e.values.flatMap(GeometryUtils.bounds(_, multiplier, bits))
79         val (field, ranges) =
80           if (typed == ObjectType.POINT) {
81             val field = ZValueField.z2(col, encoded = true).zValue
82             // TODO make max ranges configurable
83             val ranges = Z2Encoder.ranges(bounds, Some(8))
84             (field, ranges)
85           } else {
86             val field = ZValueField.xz2(col, encoded = true).zValue
87             val ranges = XZ2Encoder.ranges(bounds, Some(8))
88             (field, ranges)
89           }
90         val zcol = FilterApi.binaryColumn(field)
91         val filters = ranges.map { case (lower, upper) =>
92           FilterApi.and(FilterApi.gtEq(zcol, Binary.fromString(lower)), FilterApi.ltEq(zcol, Binary.fromString(upper)))
93         }
94         filters.reduce(FilterApi.or)
95       }
96     }
97     // since we don't know what the actual file encoding is up front, we always have to evaluate the full predicate post-read
98     (predicate, Some(filter))
99   }

101   private def attribute[T : ClassTag, U <: Comparable[U]](
102       sft: SimpleFeatureType,
103       name: String,
104       filter: Filter,
105       col: Column[U] with SupportsLtGt,
106       conversion: T => U): (Option[FilterPredicate], Option[Filter]) = {
107     val (attribute, nonAttribute) = FilterExtractingVisitor(filter, name, sft)
108     val binding = implicitly[ClassTag[T]].runtimeClass.asInstanceOf[Class[T]]
109     val bounds = attribute.map(FilterHelper.extractAttributeBounds(_, name, binding))
110     val predicate = bounds.flatMap { extracted =>
111       Some(extracted).filter(e => e.nonEmpty && !e.disjoint && e.values.forall(_.isBounded)).map { e =>
112         val filters = e.values.map { bounds =>
113           if (bounds.isEquals) {
114             FilterApi.eq(col, conversion(bounds.lower.value.get))
115           } else {
116             val lower: Option[FilterPredicate] = bounds.lower.value.map { value =>
117               val converted = conversion(value)
118               if (bounds.lower.inclusive) { FilterApi.gtEq(col, converted) } else { FilterApi.gt(col, converted) }
119             }
120             val upper: Option[FilterPredicate] = bounds.upper.value.map { value =>
121               val converted = conversion(value)
122               if (bounds.upper.inclusive) { FilterApi.ltEq(col, converted) } else { FilterApi.lt(col, converted) }
123             }
124             (lower, upper) match {
125               case (Some(lo), Some(hi)) => FilterApi.and(lo, hi)
126               case (Some(f), None) => f
127               case (None, Some(f)) => f
128               case (None, None) => throw new IllegalStateException() // shouldn't happen due to checks above
129             }
130           }
131         }
132         merge(filters)
133       }
134     }
135     val remaining = if (bounds.exists(_.precise)) { nonAttribute } else { Some(filter) }
136     (predicate, remaining)
137   }

157   /**
158     * Merge OR'd filters
159     *
160     * Detect and re-write "not equals" filters to handle null values
161     *
162     * FilterHelper methods make "foo != x" comes out as "foo < x OR foo > x"
163    *
164     * that won't return null values - not normally a concern since we don't index them in key-value dbs,
165     * but we want to handle that case here
166     *
167     * @param filters filters
168     * @return combined filter
169     */
170   private def merge[T <: Comparable[T]](filters: Seq[FilterPredicate]): FilterPredicate = {
171     lazy val values = filters.collect {
172       case f: Lt[T] => (f.getColumn.asInstanceOf[Column[T] with SupportsEqNotEq], f.getValue)
173       case f: Gt[T] => (f.getColumn.asInstanceOf[Column[T] with SupportsEqNotEq], f.getValue)
174     }.distinct
175 
176     if (filters.lengthCompare(2) == 0 &&
177         filters.exists(_.isInstanceOf[Lt[_]]) &&
178         filters.exists(_.isInstanceOf[Gt[_]]) &&
179         values.lengthCompare(1) == 0) {
180       FilterApi.notEq(values.head._1, values.head._2)
181     } else {
182       filters.reduce(FilterApi.or)
183     }
184   }
