62   /**
63    * Convert geomesa file metadata to iceberg file metadata
64    *
65    * @param table iceberg table
66    * @param file file
67    * @return
68    */
69   def toDataFile(table: Table, file: StorageFile): DataFile = {
70     val uri = storage.context.root.resolve(file.file).toString
71     logger.whenDebugEnabled {
72       val inputFile = ParquetFileSystemReader.inputFile(storage.fs, new URI(uri))
73       val footer = WithClose(inputFile.newStream())(ParquetFileReader.readFooter(inputFile, ParquetReadOptions.builder().build(), _))
74       logger.debug(s"Parquet schema for ${file.file}: ${ParquetMetadata.toJSON(footer)}")
75     }
76     val inputFile = table.io().newInputFile(uri)
77     val metrics = ParquetUtil.fileMetrics(inputFile, metricsConfigs.computeIfAbsent(table.name(), _ => MetricsConfig.forTable(table)), null)
78     val partitions = partitionValues(file.partition).asJava
79     // TODO withSort(f.sort)
80     DataFiles.builder(table.spec())
81       .withPath(inputFile.location())
82       .withFormat(FileFormat.PARQUET)
83       .withFileSizeInBytes(inputFile.getLength)
84       .withMetrics(metrics)
85       .withPartitionValues(partitions)
86       .withRecordCount(file.count)
87       .build()
88   }
89 
90   /**
91    * Gets the iceberg partition values that correspond to a given partition
92    *
93    * @param partition partition
94    * @return
95    */
96   def partitionValues(partition: Partition): Seq[String] = {
97     mappers.map { m =>
98       val key = partition.values.find(_.name == m.scheme.name).getOrElse {
99         throw new IllegalArgumentException(
100           s"Could not find associated partition: ${m.scheme.name} out of ${partition.values.mkString(", ")}")
101       }
102       m.toIceberg(key.value)
103     }
104   }

107 object IcebergMapper {
108 
109   /**
110    * Maps a partition scheme to iceberg
111    */
112   private trait SchemeMapper {
113 
114     /**
115      * The partition scheme being mapped
116      *
117      * @return
118      */
119     def scheme: PartitionScheme
120 
121     /**
122      * Creates the iceberg partition spec
123      *
124      * @param b spec builder
125      * @return
126      */
127     def spec(b: PartitionSpec.Builder): PartitionSpec.Builder
128 
129     /**
130      * Gets the iceberg partition value for a given geomesa partition value
131      *
132      * @param partitionValue geomesa partition value
133      * @return iceberg partition value
134      */
135     def toIceberg(partitionValue: String): String
136   }
137 
138   private object SchemeMapper {
139 
140     /**
141      * Maps a partition scheme to iceberg
142      *
143      * @param scheme the scheme to map
144      * @return a mapping, if the scheme is supported by iceberg
145      */
146     def apply(scheme: PartitionScheme, binding: Class[_]): Option[SchemeMapper] = scheme match {
147       case s: DateTimeScheme if s.step == 1 && s.unit == ChronoUnit.HOURS => Some(HourMapper(s))
148       case s: DateTimeScheme if s.step == 1 && s.unit == ChronoUnit.DAYS => Some(DayMapper(s))
149       case s: DateTimeScheme if s.step == 1 && s.unit == ChronoUnit.MONTHS => Some(MonthMapper(s))
150       case s: DateTimeScheme if s.step == 1 && s.unit == ChronoUnit.YEARS => Some(YearMapper(s))
151 
152       case s: Z2Scheme if s.bits % 4 == 0 => Some(Z2Mapper(s))
153       case s: XZ2Scheme if s.bits % 4 == 0 => Some(XZ2Mapper(s))
154 
155       case s: HashScheme[_] => Some(HashMapper(s))
156 
157       case s: AttributeScheme[_] if classOf[String].isAssignableFrom(binding) =>
158         s.bucketing match {
159           case None => Some(IdentityMapper(s, LexiTypeEncoders.stringEncoder()))
160           case Some(w: WidthBucketing) => Some(TruncateMapper(s, LexiTypeEncoders.stringEncoder(), w.max))
161         }
162 
163       case s: AttributeScheme[_] if classOf[Integer].isAssignableFrom(binding) =>
164         s.bucketing match {
165           case None => Some(IdentityMapper(s, LexiTypeEncoders.integerEncoder()))
166           case Some(i: IntegralBucketing[Int]) => Some(TruncateMapper(s, LexiTypeEncoders.integerEncoder(), i.divisor))
167         }
168 
169       case s: AttributeScheme[_] if classOf[java.lang.Long].isAssignableFrom(binding) =>
170         s.bucketing match {
171           case None => Some(IdentityMapper(s, LexiTypeEncoders.longEncoder()))
172           case Some(i: IntegralBucketing[Long]) => Some(TruncateMapper(s, LexiTypeEncoders.longEncoder(), i.divisor.toInt))
173         }
174 
175       case _ => None
176     }
177   }
178 
179   private case class HourMapper(scheme: DateTimeScheme) extends SchemeMapper {
180     override def spec(b: PartitionSpec.Builder): PartitionSpec.Builder = b.hour(scheme.attribute)
181     override def toIceberg(key: String): String = LexiTypeEncoders.integerEncoder().decode(key).toString
182   }
183 
184   private case class DayMapper(scheme: DateTimeScheme) extends SchemeMapper {
185     override def spec(b: PartitionSpec.Builder): PartitionSpec.Builder = b.day(scheme.attribute)
186     // note: days are handled differently from other types, and expect an ISO_LOCAL_DATE formatted string
187     override def toIceberg(key: String): String = {
188       val days = LexiTypeEncoders.integerEncoder().decode(key)
189       DateTimeFormatter.ISO_LOCAL_DATE.format(LocalDate.EPOCH.plusDays(days.longValue()))
190     }
191   }
192 
193   private case class MonthMapper(scheme: DateTimeScheme) extends SchemeMapper {
194     override def spec(b: PartitionSpec.Builder): PartitionSpec.Builder = b.month(scheme.attribute)
195     override def toIceberg(key: String): String = LexiTypeEncoders.integerEncoder().decode(key).toString
196   }
197 
198   private case class YearMapper(scheme: DateTimeScheme) extends SchemeMapper {
199     override def spec(b: PartitionSpec.Builder): PartitionSpec.Builder = b.year(scheme.attribute)
200     override def toIceberg(key: String): String = LexiTypeEncoders.integerEncoder().decode(key).toString
201   }
202 
203   private case class Z2Mapper(scheme: Z2Scheme) extends SchemeMapper {
204     override def spec(b: PartitionSpec.Builder): PartitionSpec.Builder =
205       b.truncate(ZValueField.z2(scheme.attribute).zValue, scheme.bits / 4)
206     override def toIceberg(partitionValue: String): String = Z2Encoder.encodePartition(partitionValue, scheme.bits)
207   }
208 
209   private case class XZ2Mapper(scheme: XZ2Scheme) extends SchemeMapper {
210     override def spec(b: PartitionSpec.Builder): PartitionSpec.Builder =
211       b.truncate(ZValueField.xz2(scheme.attribute).zValue, scheme.bits / 4)
212     override def toIceberg(partitionValue: String): String = partitionValue
213   }
214 
215   private case class HashMapper(scheme: HashScheme[_]) extends SchemeMapper {
216     override def spec(b: PartitionSpec.Builder): PartitionSpec.Builder = b.bucket(scheme.attribute, scheme.buckets)
217     override def toIceberg(key: String): String = key
218   }
219 
220   private case class IdentityMapper(scheme: PartitionScheme, lexicoder: TypeEncoder[_, String]) extends SchemeMapper {
221     override def spec(b: PartitionSpec.Builder): PartitionSpec.Builder = b.identity(scheme.attribute)
222     override def toIceberg(key: String): String = lexicoder.decode(key).toString
223   }
224 
225   private case class TruncateMapper(scheme: PartitionScheme, lexicoder: TypeEncoder[_, String], width: Int) extends SchemeMapper {
226     override def spec(b: PartitionSpec.Builder): PartitionSpec.Builder = b.truncate(scheme.attribute, width)
227     override def toIceberg(key: String): String = lexicoder.decode(key).toString
228   }
