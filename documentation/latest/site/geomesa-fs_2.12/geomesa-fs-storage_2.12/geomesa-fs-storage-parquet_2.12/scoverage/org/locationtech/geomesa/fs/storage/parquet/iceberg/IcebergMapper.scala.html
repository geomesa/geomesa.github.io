60   /**
61    * Convert geomesa file metadata to iceberg file metadata
62    *
63    * @param table iceberg table
64    * @param file file
65    * @return
66    */
67   def toDataFile(table: Table, file: StorageFile): DataFile = {
68     val uri = storage.context.root.resolve(file.file).toString
69     logger.whenDebugEnabled {
70       val inputFile = ParquetFileSystemReader.inputFile(storage.fs, new URI(uri))
71       val footer = WithClose(inputFile.newStream())(ParquetFileReader.readFooter(inputFile, ParquetReadOptions.builder().build(), _))
72       logger.debug(s"Parquet schema for ${file.file}: ${ParquetMetadata.toJSON(footer)}")
73     }
74     val inputFile = table.io().newInputFile(uri)
75     val metrics = ParquetUtil.fileMetrics(inputFile, metricsConfigs.computeIfAbsent(table.name(), _ => MetricsConfig.forTable(table)), null)
76     // TODO withSort(f.sort)
77     DataFiles.builder(table.spec())
78       .withPath(inputFile.location())
79       .withFormat(FileFormat.PARQUET)
80       .withFileSizeInBytes(inputFile.getLength)
81       .withMetrics(metrics)
82       .withPartitionValues(partitionValues(file.partition).asJava)
83       .withRecordCount(file.count)
84       .build()
85   }

87   /**
88    * Gets the iceberg partition values that correspond to a given partition
89    *
90    * @param partition partition
91    * @return
92    */
93   def partitionValues(partition: Partition): Seq[String] = {
94     mappers.map { m =>
95       val key = partition.values.find(_.name == m.scheme.name).getOrElse {
96         throw new IllegalArgumentException(
97           s"Could not find associated partition: ${m.scheme.name} out of ${partition.values.mkString(", ")}")
98       }
99       m.toIceberg(key.value)
100     }
101   }

106   /**
107    * Maps a partition scheme to iceberg
108    */
109   private trait SchemeMapper {
110 
111     /**
112      * The partition scheme being mapped
113      *
114      * @return
115      */
116     def scheme: PartitionScheme
117 
118     /**
119      * Creates the iceberg partition spec
120      *
121      * @param b spec builder
122      * @return
123      */
124     def spec(b: PartitionSpec.Builder): PartitionSpec.Builder
125 
126     /**
127      * Gets the iceberg partition value for a given geomesa partition value
128      *
129      * @param partitionValue geomesa partition value
130      * @return iceberg partition value
131      */
132     def toIceberg(partitionValue: String): String
133   }
134 
135   private object SchemeMapper {
136 
137     /**
138      * Maps a partition scheme to iceberg
139      *
140      * @param scheme the scheme to map
141      * @return a mapping, if the scheme is supported by iceberg
142      */
143     def apply(scheme: PartitionScheme, binding: Class[_]): Option[SchemeMapper] = scheme match {
144       case s: DateTimeScheme if s.step == 1 && s.unit == ChronoUnit.HOURS => Some(HourMapper(s))
145       case s: DateTimeScheme if s.step == 1 && s.unit == ChronoUnit.DAYS => Some(DayMapper(s))
146       case s: DateTimeScheme if s.step == 1 && s.unit == ChronoUnit.MONTHS => Some(MonthMapper(s))
147       case s: DateTimeScheme if s.step == 1 && s.unit == ChronoUnit.YEARS => Some(YearMapper(s))
148 
149       case s: Z2Scheme if s.bits % 4 == 0 => Some(Z2Mapper(s))
150       case s: XZ2Scheme if s.bits % 4 == 0 => Some(XZ2Mapper(s))
151 
152       case s: HashScheme[_] => Some(HashMapper(s))
153 
154       case s: AttributeScheme[_] if classOf[String].isAssignableFrom(binding) =>
155         s.bucketing match {
156           case None => Some(IdentityMapper(s, LexiTypeEncoders.stringEncoder()))
157           case Some(w: WidthBucketing) => Some(TruncateMapper(s, LexiTypeEncoders.stringEncoder(), w.max))
158         }
159 
160       case s: AttributeScheme[_] if classOf[Integer].isAssignableFrom(binding) =>
161         s.bucketing match {
162           case None => Some(IdentityMapper(s, LexiTypeEncoders.integerEncoder()))
163           case Some(i: IntegralBucketing[Int]) => Some(TruncateMapper(s, LexiTypeEncoders.integerEncoder(), i.divisor))
164         }
165 
166       case s: AttributeScheme[_] if classOf[java.lang.Long].isAssignableFrom(binding) =>
167         s.bucketing match {
168           case None => Some(IdentityMapper(s, LexiTypeEncoders.longEncoder()))
169           case Some(i: IntegralBucketing[Long]) => Some(TruncateMapper(s, LexiTypeEncoders.longEncoder(), i.divisor.toInt))
170         }
171 
172       case _ => None
173     }
174   }
