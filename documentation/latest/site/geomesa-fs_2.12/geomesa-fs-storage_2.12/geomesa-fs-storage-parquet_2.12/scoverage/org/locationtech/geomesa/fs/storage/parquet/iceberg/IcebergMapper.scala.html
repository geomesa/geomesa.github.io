63   /**
64    * Convert geomesa file metadata to iceberg file metadata
65    *
66    * @param table iceberg table
67    * @param file file
68    * @return
69    */
70   def toDataFile(table: Table, file: StorageFile): DataFile = {
71     val uri = storage.context.root.resolve(file.file).toString
72     logger.whenDebugEnabled {
73       val inputFile = ParquetFileSystemReader.inputFile(storage.fs, new URI(uri))
74       val footer = WithClose(inputFile.newStream())(ParquetFileReader.readFooter(inputFile, ParquetReadOptions.builder().build(), _))
75       logger.debug(s"Parquet schema for ${file.file}: ${ParquetMetadata.toJSON(footer)}")
76     }
77     val inputFile = table.io().newInputFile(uri)
78     val metrics = ParquetUtil.fileMetrics(inputFile, metricsConfigs.computeIfAbsent(table.name(), _ => MetricsConfig.forTable(table)), null)
79     val partitions = partitionValues(file.partition).asJava
80     // TODO withSort(f.sort)
81     DataFiles.builder(table.spec())
82       .withPath(inputFile.location())
83       .withFormat(FileFormat.PARQUET)
84       .withFileSizeInBytes(inputFile.getLength)
85       .withMetrics(metrics)
86       .withPartitionValues(partitions)
87       .withRecordCount(file.count)
88       .build()
89   }

91   /**
92    * Gets the iceberg partition values that correspond to a given partition
93    *
94    * @param partition partition
95    * @return
96    */
97   def partitionValues(partition: Partition): Seq[String] = {
98     mappers.map { m =>
99       val key = partition.values.find(_.name == m.scheme.name).getOrElse {
100         throw new IllegalArgumentException(
101           s"Could not find associated partition: ${m.scheme.name} out of ${partition.values.mkString(", ")}")
102       }
103       m.toIceberg(key.value)
104     }
105   }

108 object IcebergMapper {
109 
110   /**
111    * Maps a partition scheme to iceberg
112    */
113   private trait SchemeMapper {
114 
115     /**
116      * The partition scheme being mapped
117      *
118      * @return
119      */
120     def scheme: PartitionScheme
121 
122     /**
123      * Creates the iceberg partition spec
124      *
125      * @param b spec builder
126      * @return
127      */
128     def spec(b: PartitionSpec.Builder): PartitionSpec.Builder
129 
130     /**
131      * Gets the iceberg partition value for a given geomesa partition value
132      *
133      * @param partitionValue geomesa partition value
134      * @return iceberg partition value
135      */
136     def toIceberg(partitionValue: String): String
137   }
138 
139   private object SchemeMapper {
140 
141     /**
142      * Maps a partition scheme to iceberg
143      *
144      * @param scheme the scheme to map
145      * @return a mapping, if the scheme is supported by iceberg
146      */
147     def apply(scheme: PartitionScheme, binding: Class[_]): Option[SchemeMapper] = scheme match {
148       case s: DateTimeScheme if s.step == 1 && s.unit == ChronoUnit.HOURS => Some(HourMapper(s))
149       case s: DateTimeScheme if s.step == 1 && s.unit == ChronoUnit.DAYS => Some(DayMapper(s))
150       case s: DateTimeScheme if s.step == 1 && s.unit == ChronoUnit.MONTHS => Some(MonthMapper(s))
151       case s: DateTimeScheme if s.step == 1 && s.unit == ChronoUnit.YEARS => Some(YearMapper(s))
152 
153       case s: Z2Scheme if s.bits % 4 == 0 => Some(Z2Mapper(s))
154       case s: XZ2Scheme if s.bits % 4 == 0 => Some(XZ2Mapper(s))
155 
156       case s: HashScheme[_] => Some(HashMapper(s))
157 
158       case s: AttributeScheme[_] if classOf[String].isAssignableFrom(binding) =>
159         s.bucketing match {
160           case None => Some(IdentityMapper(s, LexiTypeEncoders.stringEncoder()))
161           case Some(w: WidthBucketing) => Some(TruncateMapper(s, LexiTypeEncoders.stringEncoder(), w.max))
162         }
163 
164       case s: AttributeScheme[_] if classOf[Integer].isAssignableFrom(binding) =>
165         s.bucketing match {
166           case None => Some(IdentityMapper(s, LexiTypeEncoders.integerEncoder()))
167           case Some(i: IntegralBucketing[Int]) => Some(TruncateMapper(s, LexiTypeEncoders.integerEncoder(), i.divisor))
168         }
169 
170       case s: AttributeScheme[_] if classOf[java.lang.Long].isAssignableFrom(binding) =>
171         s.bucketing match {
172           case None => Some(IdentityMapper(s, LexiTypeEncoders.longEncoder()))
173           case Some(i: IntegralBucketing[Long]) => Some(TruncateMapper(s, LexiTypeEncoders.longEncoder(), i.divisor.toInt))
174         }
175 
176       case _ => None
177     }
178   }
179 
180   private case class HourMapper(scheme: DateTimeScheme) extends SchemeMapper {
181     override def spec(b: PartitionSpec.Builder): PartitionSpec.Builder = b.hour(scheme.attribute)
182     override def toIceberg(key: String): String = LexiTypeEncoders.integerEncoder().decode(key).toString
183   }
184 
185   private case class DayMapper(scheme: DateTimeScheme) extends SchemeMapper {
186     override def spec(b: PartitionSpec.Builder): PartitionSpec.Builder = b.day(scheme.attribute)
187     // note: days are handled differently from other types, and expect an ISO_LOCAL_DATE formatted string
188     override def toIceberg(key: String): String = {
189       val days = LexiTypeEncoders.integerEncoder().decode(key)
190       DateTimeFormatter.ISO_LOCAL_DATE.format(LocalDate.EPOCH.plusDays(days.longValue()))
191     }
192   }
193 
194   private case class MonthMapper(scheme: DateTimeScheme) extends SchemeMapper {
195     override def spec(b: PartitionSpec.Builder): PartitionSpec.Builder = b.month(scheme.attribute)
196     override def toIceberg(key: String): String = LexiTypeEncoders.integerEncoder().decode(key).toString
197   }
198 
199   private case class YearMapper(scheme: DateTimeScheme) extends SchemeMapper {
200     override def spec(b: PartitionSpec.Builder): PartitionSpec.Builder = b.year(scheme.attribute)
201     override def toIceberg(key: String): String = LexiTypeEncoders.integerEncoder().decode(key).toString
202   }
203 
204   private case class Z2Mapper(scheme: Z2Scheme) extends SchemeMapper {
205     private val hexFormat = HexFormat.of()
206     private val width = scheme.bits / 4
207     override def spec(b: PartitionSpec.Builder): PartitionSpec.Builder =
208       b.truncate(ZValueField.z2(scheme.attribute).zValue, width)
209     override def toIceberg(partitionValue: String): String = hexFormat.toHexDigits(partitionValue.toLong, width)
210   }
211 
212   private case class XZ2Mapper(scheme: XZ2Scheme) extends SchemeMapper {
213     private val hexFormat = HexFormat.of()
214     private val width = scheme.bits / 4
215     override def spec(b: PartitionSpec.Builder): PartitionSpec.Builder =
216       b.truncate(ZValueField.xz2(scheme.attribute).zValue, width)
217     override def toIceberg(partitionValue: String): String = hexFormat.toHexDigits(partitionValue.toLong, width)
218   }
219 
220   private case class HashMapper(scheme: HashScheme[_]) extends SchemeMapper {
221     override def spec(b: PartitionSpec.Builder): PartitionSpec.Builder = b.bucket(scheme.attribute, scheme.buckets)
222     override def toIceberg(key: String): String = key
223   }
224 
225   private case class IdentityMapper(scheme: PartitionScheme, lexicoder: TypeEncoder[_, String]) extends SchemeMapper {
226     override def spec(b: PartitionSpec.Builder): PartitionSpec.Builder = b.identity(scheme.attribute)
227     override def toIceberg(key: String): String = lexicoder.decode(key).toString
228   }
229 
230   private case class TruncateMapper(scheme: PartitionScheme, lexicoder: TypeEncoder[_, String], width: Int) extends SchemeMapper {
231     override def spec(b: PartitionSpec.Builder): PartitionSpec.Builder = b.truncate(scheme.attribute, width)
232     override def toIceberg(key: String): String = lexicoder.decode(key).toString
233   }
