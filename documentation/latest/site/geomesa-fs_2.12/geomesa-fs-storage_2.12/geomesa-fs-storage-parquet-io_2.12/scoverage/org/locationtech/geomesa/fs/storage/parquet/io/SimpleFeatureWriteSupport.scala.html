29 class SimpleFeatureWriteSupport extends WriteSupport[SimpleFeature] {
30 
31   private var writer: SimpleFeatureWriteSupport.SimpleFeatureWriter = _
32   private var consumer: RecordConsumer = _
33   private var geoParquetObserver: GeoParquetObserver = _
34   private var baseMetadata: java.util.Map[String, String] = _
35 
36   override val getName: String = "SimpleFeatureWriteSupport"
37 
38   // called once
39   override def init(conf: Configuration): WriteContext = init(new HadoopParquetConfiguration(conf))
40 
41   override def init(conf: ParquetConfiguration): WriteContext = {
42     val schema = SimpleFeatureParquetSchema.write(conf).getOrElse {
43       throw new IllegalArgumentException("Could not extract SimpleFeatureType from write context")
44     }
45     init(schema)
46   }
47 
48   private def init(schema: SimpleFeatureParquetSchema): WriteContext = {
49     this.writer = new SimpleFeatureWriteSupport.SimpleFeatureWriter(schema)
50     this.geoParquetObserver = new GeoParquetObserver(schema)
51     this.baseMetadata = schema.metadata
52     new WriteContext(schema.schema, schema.metadata)
53   }
54 
55   // called per block
56   override def prepareForWrite(recordConsumer: RecordConsumer): Unit = consumer = recordConsumer
57 
58   // called per row
59   override def write(record: SimpleFeature): Unit = {
60     writer.write(consumer, record)
61     geoParquetObserver(record)
62   }
63 
64   // called once at the end
65   override def finalizeWrite(): FinalizedWriteContext = {
66     try {
67       val metadata = new java.util.HashMap[String, String]()
68       metadata.putAll(baseMetadata)
69       metadata.putAll(geoParquetObserver.metadata())
70       new FinalizedWriteContext(metadata)
71     } finally {
72       CloseWithLogging(geoParquetObserver)
73     }
74   }
75 }

81   private class SimpleFeatureWriter(schema: SimpleFeatureParquetSchema) {
82 
83     private val fids = new FidWriter(0) // ID is the 1st field
84     private val vis = if (schema.hasVisibilities) { new VisibilityWriter(1) } else { null } // vis is 2nd field
85 
86     private val attributes = {
87       var index = if (vis == null) { 1 } else { 2 }
88       Array.tabulate(schema.sft.getAttributeCount) { i =>
89         val writer = attribute(schema.sft.getDescriptor(i), index).asInstanceOf[AttributeWriter[AnyRef]]
90         index += writer.numFields
91         writer
92       }
93     }
94 
95     def write(consumer: RecordConsumer, value: SimpleFeature): Unit = {
96       consumer.startMessage()
97       fids.apply(consumer, value.getID)
98       if (vis != null) {
99         vis.apply(consumer, value.getUserData.get("geomesa.feature.visibility").asInstanceOf[String])
100       }
101       var i = 0
102       while (i < attributes.length) {
103         attributes(i).apply(consumer, value.getAttribute(i))
104         i += 1
105       }
106       consumer.endMessage()
107     }
108 
109     private def attribute(descriptor: AttributeDescriptor, index: Int): AttributeWriter[_] =
110       attribute(descriptor.getLocalName, index, ObjectType.selectType(descriptor))
111 
112     private def attribute(name: String, index: Int, bindings: Seq[ObjectType]): AttributeWriter[_] = {
113       lazy val safeName = alphaNumericSafeString(name)
114       bindings.head match {
115         case ObjectType.GEOMETRY => geometry(safeName, index, bindings.last, schema.bboxes.get(name))
116         case ObjectType.DATE     => new DateMicrosWriter(safeName, index)
117         case ObjectType.STRING   => new StringWriter(safeName, index)
118         case ObjectType.INT      => new IntegerWriter(safeName, index)
119         case ObjectType.LONG     => new LongWriter(safeName, index)
120         case ObjectType.FLOAT    => new FloatWriter(safeName, index)
121         case ObjectType.DOUBLE   => new DoubleWriter(safeName, index)
122         case ObjectType.BYTES    => new BytesWriter(safeName, index)
123         case ObjectType.LIST     => new ListWriter(safeName, index, attribute("element", 0, bindings.drop(1)))
124         case ObjectType.MAP      => new MapWriter(safeName, index, attribute("key", 0, bindings.slice(1, 2)), attribute("value", 1, bindings.slice(2, 3)))
125         case ObjectType.BOOLEAN  => new BooleanWriter(safeName, index)
126         case ObjectType.UUID     => new UuidWriter(safeName, index)
127         case _ => throw new IllegalArgumentException(s"Can't serialize field '$name' of type ${bindings.head}")
128       }
129     }
130 
131     // TODO support z/m
132     private def geometry(name: String, index: Int, binding: ObjectType, bbox: Option[String]): AttributeWriter[_] = {
133       if (schema.encodings.geometry == GeometryEncoding.GeoParquetWkb) {
134         new WkbWriter(name, index, bbox)
135       } else {
136         val native = schema.encodings.geometry == GeometryEncoding.GeoParquetNative
137         binding match {
138           case ObjectType.POINT                     => new PointWriter(name, index, bbox)
139           case ObjectType.LINESTRING      if native => new GeoParquetNativeLineStringWriter(name, index, bbox)
140           case ObjectType.LINESTRING                => new LineStringWriter(name, index, bbox)
141           case ObjectType.POLYGON         if native => new GeoParquetNativePolygonWriter(name, index, bbox)
142           case ObjectType.POLYGON                   => new PolygonWriter(name, index, bbox)
143           case ObjectType.MULTIPOINT      if native => new GeoParquetNativeMultiPointWriter(name, index, bbox)
144           case ObjectType.MULTIPOINT                => new MultiPointWriter(name, index, bbox)
145           case ObjectType.MULTILINESTRING if native => new GeoParquetNativeMultiLineStringWriter(name, index, bbox)
146           case ObjectType.MULTILINESTRING           => new MultiLineStringWriter(name, index, bbox)
147           case ObjectType.MULTIPOLYGON    if native => new GeoParquetNativeMultiPolygonWriter(name, index, bbox)
148           case ObjectType.MULTIPOLYGON              => new MultiPolygonWriter(name, index, bbox)
149           case ObjectType.GEOMETRY_COLLECTION       => new WkbWriter(name, index, bbox)
150           case ObjectType.GEOMETRY                  => new WkbWriter(name, index, bbox)
151           case _ => throw new IllegalArgumentException(s"Can't serialize field '$name' of type $binding")
152         }
153       }
154     }
155   }

234   private class ListWriter[T <: Any](name: String, index: Int, elements: AttributeWriter[T])
235       extends AttributeWriter[java.util.List[T]](name, index) {
236 
237     override def writeFields(consumer: RecordConsumer, value: java.util.List[T]): Unit = {
238       consumer.startGroup()
239       if (!value.isEmpty) {
240         consumer.startField("list", 0)
241         consumer.startGroup()
242         consumer.startField("element", 0)
243         val iter = value.iterator
244         while (iter.hasNext) {
245           val item = iter.next
246           if (item != null) {
247             elements.writeFields(consumer, item)
248           }
249         }
250         consumer.endField("element", 0)
251         consumer.endGroup()
252         consumer.endField("list", 0)
253       }
254       consumer.endGroup()
255     }

258   private class MapWriter[U <: Any, V <: Any](name: String, index: Int, keys: AttributeWriter[U], values: AttributeWriter[V])
259       extends AttributeWriter[java.util.Map[U, V]](name, index) {
260     override def writeFields(consumer: RecordConsumer, value: java.util.Map[U, V]): Unit = {
261       consumer.startGroup()
262       if (!value.isEmpty) {
263         consumer.startField("key_value", 0)
264         val iter = value.entrySet().iterator
265         while (iter.hasNext) {
266           val entry = iter.next()
267           consumer.startGroup()
268           keys(consumer, entry.getKey)
269           val v = entry.getValue
270           if (v != null) {
271             values(consumer, v)
272           }
273           consumer.endGroup()
274         }
275         consumer.endField("key_value", 0)
276       }
277       consumer.endGroup()
278     }
279   }

282     override def writeFields(consumer: RecordConsumer, value: UUID): Unit = {

287       consumer.addBinary(Binary.fromConstantByteBuffer(bb))

294   private abstract class GeometryWriter[T <: Geometry](name: String, index: Int, bbox: Option[String])
295       extends AttributeWriter[T](name, index, if (bbox.isEmpty) { 1 } else { 2 }) {
296 
297     /**
298      * Writes a value to the current record
299      *
300      * @param consumer the Parquet record consumer
301      * @param value value to write
302      */
303     override def apply(consumer: RecordConsumer, value: T): Unit = {
304       if (value != null) {
305         consumer.startField(name, index)
306         writeFields(consumer, value)
307         consumer.endField(name, index)
308         bbox.foreach { name =>
309           val bbox = value.getEnvelopeInternal
310           consumer.startField(name, index + 1)
311           consumer.startGroup()
312           consumer.startField(GeometrySchema.BoundingBoxField.XMin, 0)
313           consumer.addFloat(bbox.getMinX.toFloat)
314           consumer.endField(GeometrySchema.BoundingBoxField.XMin, 0)
315           consumer.startField(GeometrySchema.BoundingBoxField.YMin, 1)
316           consumer.addFloat(bbox.getMinY.toFloat)
317           consumer.endField(GeometrySchema.BoundingBoxField.YMin, 1)
318           consumer.startField(GeometrySchema.BoundingBoxField.XMax, 2)
319           consumer.addFloat(bbox.getMaxX.toFloat)
320           consumer.endField(GeometrySchema.BoundingBoxField.XMax, 2)
321           consumer.startField(GeometrySchema.BoundingBoxField.YMax, 3)
322           consumer.addFloat(bbox.getMaxY.toFloat)
323           consumer.endField(GeometrySchema.BoundingBoxField.YMax, 3)
324           consumer.endGroup()
325           consumer.endField(name, index + 1)
326         }
327       }
328     }
329   }
