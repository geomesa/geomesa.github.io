21 /**
22  * Mapping of fields to z-value fields
23  *
24  * @param fields fields
25  */
26 case class ZValues(fields: Seq[ZValueField]) {
27   private val fieldMap = fields.map(f => f.geometry -> f.zValue).toMap
28   def isEmpty: Boolean = fields.isEmpty
29   def nonEmpty: Boolean = fields.nonEmpty
30   def get(field: String): Option[String] = fieldMap.get(field)
31 }
32 
33 object ZValues {
34 
35   import scala.collection.JavaConverters._
36 
37   /**
38    * Gets the fields of this schema that have per-row z-values
39    *
40    * @param sft simple feature type
41    * @return
42    */
43   def apply(sft: SimpleFeatureType): ZValues = {
44     val bboxes = sft.getAttributeDescriptors.asScala.toSeq.flatMap { d =>
45       val binding = d.getType.getBinding
46       if (classOf[Geometry].isAssignableFrom(binding)) {
47         Some(ZValueField(d.getLocalName))
48       } else {
49         None
50       }
51     }
52     ZValues(bboxes)
53   }
54 
55   /**
56    * Holder for a z-value field, along with a reference back to the original geometry field
57    *
58    * @param geometry name of the original geometry field being covered
59    * @param zValue name of the z-value field
60    */
61   case class ZValueField(geometry: String, zValue: String)
62 
63   object ZValueField {
64 
65     val ZValueFieldPrefix = "__"
66     val ZValueFieldSuffix = "_z__"
67 
68     def apply(geometry: String, encoded: Boolean = false): ZValueField = {
69       val geom = if (encoded) { geometry } else { alphaNumericSafeString(geometry) }
70       val zValue = s"$ZValueFieldPrefix$geom$ZValueFieldSuffix"
71       ZValueField(geom, zValue)
72     }
73 
74     /**
75      * Creates a field name based on a z-value field
76      *
77      * @param field name of a potential z-value field
78      * @return
79      */
80     def fromFieldName(field: String): Option[ZValueField] = {
81       if (field.startsWith(ZValueFieldPrefix) && field.endsWith(ZValueFieldSuffix)) {
82         Some(ZValueField(field.substring(ZValueFieldPrefix.length, field.length - ZValueFieldSuffix.length), field))
83       } else {
84         None
85       }
86     }
87 
88     /**
89      * The parquet schema for a z-value field
90      *
91      * @param zValue field name
92      * @return
93      */
94     def schema(zValue: String, fieldIds: AtomicInteger): PrimitiveType =
95       Types.optional(PrimitiveTypeName.INT64).id(fieldIds.getAndIncrement()).named(zValue)
96 
97     /**
98      * The iceberg schema for a z-value field
99      *
100      * @param zValue field name
101      * @return
102      */
103     def icebergSchema(zValue: String, fieldIds: AtomicInteger): NestedField =
104       NestedField.optional(zValue).withId(fieldIds.getAndIncrement()).ofType(LongType.get()).build()
105   }
106 }
