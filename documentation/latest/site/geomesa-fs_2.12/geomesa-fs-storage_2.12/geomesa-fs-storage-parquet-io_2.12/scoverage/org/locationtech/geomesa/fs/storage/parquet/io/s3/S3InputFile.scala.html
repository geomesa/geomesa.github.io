24 /**
25  * Parquet input file implementation for reading from S3. Uses the aws-s3-analyticsaccelerator project to improve
26  * read times.
27  *
28  * @param fs object store impl
29  * @param path file path
30  */
31 class S3InputFile(fs: S3ObjectStore, path: URI) extends InputFile {
32 
33   private val key = S3ObjectStore.parseS3Path(path)
34 
35   override lazy val getLength: Long = fs.size(path)
36 
37   override def newStream(): SeekableInputStream =
38     new S3SeekableInputStream(fs.seekableInputStreamFactory.createStream(S3URI.of(key.bucket, key.key)))
39 }
