30 /**
31  * Parquet writer
32  *
33  * @param file file to write
34  * @param conf configuration, must include the feature type encoded according to `SimpleFeatureParquetSchema`
35  * @param observer any observers
36  */
37 class ParquetFileSystemWriter(
38     fs: ObjectStore,
39     conf: Map[String, String],
40     sft: SimpleFeatureType,
41     file: URI,
42     observer: FileSystemObserver = NoOpObserver
43   ) extends FileSystemWriter {
44 
45   import scala.collection.JavaConverters._
46 
47   private val parquetConf = new PlainParquetConfiguration(conf.asJava)
48   SimpleFeatureParquetSchema.setSft(parquetConf, sft)
49 
50   private val writer = ParquetFileSystemWriter.builder(fs, file, parquetConf).build()
51 
52   def getDataSize: Long = writer.getDataSize
53 
54   override def write(f: SimpleFeature): Unit = {
55     writer.write(f)
56     observer(f)
57   }
58 
59   override def flush(): Unit = observer.flush()
60 
61   override def close(): Unit = CloseQuietly(Seq(writer, observer)).foreach(e => throw e)
62 }

64 object ParquetFileSystemWriter extends LazyLogging {
65 
66   def builder(fs: ObjectStore, path: URI, conf: ParquetConfiguration): Builder = {
67     val codec = CompressionCodecName.fromConf(conf.get("parquet.compression", "ZSTD"))
68     logger.debug(s"Using Parquet Compression codec ${codec.name()}")
69 
70     val file = fs match {
71       case _: LocalObjectStore =>
72         val p = Path.of(path)
73         new LocalOutputFile(p) {
74           override def create(blockSize: Long): PositionOutputStream = {
75             Option(p.toFile.getParentFile).foreach(_.mkdirs())
76             super.create(blockSize)
77           }
78           override def createOrOverwrite(blockSize: Long): PositionOutputStream = {
79             Option(p.toFile.getParentFile).foreach(_.mkdirs())
80             super.createOrOverwrite(blockSize)
81           }
82         }
83 
84       case s3: S3ObjectStore => new S3OutputFile(s3, path)
85 
86       case _ => throw new UnsupportedOperationException(s"No file implementation for scheme ${fs.scheme}")
87     }
88 
89     new Builder(file)
90       .withConf(conf)
91       .withCompressionCodec(codec)
92       .withDictionaryEncoding(true)
93       .withDictionaryPageSize(ParquetWriter.DEFAULT_PAGE_SIZE)
94       .withMaxPaddingSize(ParquetWriter.MAX_PADDING_SIZE_DEFAULT)
95       .withPageSize(ParquetWriter.DEFAULT_PAGE_SIZE)
96       .withValidation(false)
97       .withWriteMode(ParquetFileWriter.Mode.OVERWRITE)
98       .withWriterVersion(ParquetProperties.WriterVersion.PARQUET_2_0)
99       .withRowGroupSize(8L*1024*1024)
100   }
101 
102   class Builder(file: OutputFile) extends ParquetWriter.Builder[SimpleFeature, Builder](file) {
103     override def self(): Builder = this
104     override protected def getWriteSupport(conf: Configuration): WriteSupport[SimpleFeature] =
105       new SimpleFeatureWriteSupport()
106     override protected def getWriteSupport(conf: ParquetConfiguration): WriteSupport[SimpleFeature] =
107       new SimpleFeatureWriteSupport()
108   }
