103   /**
104     * Extract the simple feature type from a parquet read context. The read context
105     * contains both file metadata and the provided read conf
106     *
107     * @param context parquet read context
108     * @return
109     */
110   def read(context: InitContext): Option[SimpleFeatureParquetSchema] = {
111     // copy in the file level metadata
112     val kvMeta = context.getKeyValueMetadata.asScala.collect { case (k, v) if !v.isEmpty => k -> v.iterator.next }
113     // noinspection ScalaDeprecation
114     val gmMeta = Seq(SftNameKey, SftSpecKey, SftReadSpecKey, GeometryEncodingKey).flatMap { key =>
115       val value =
116         try { context.getParquetConfiguration.get(key) } catch {
117           // noinspection ScalaDeprecation - use the older method for spark compatibility
118           case _: NoSuchMethodError => context.getConfiguration.get(key)
119         }
120       Option(value).map(key -> _)
121     }
122     // note: context meta goes second so that transforms override the file-level keys
123     read(context.getFileSchema, kvMeta.toMap ++ gmMeta.toMap)
124   }

135   /**
136    * Get a schema for writing. Encoding can be configured through `geomesa.parquet.geometries` and `geomesa.fs.visibilities`
137    *
138    * @param conf write configuration, including the sft spec
139    * @return
140    */
141   def write(conf: ParquetConfiguration): Option[SimpleFeatureParquetSchema] = {
142     for {
143       name <- Option(conf.get(SftNameKey))
144       spec <- Option(conf.get(SftSpecKey))
145     } yield {
146       val sft = SimpleFeatureTypes.createType(name, spec)
147       val geometries = Option(conf.get(GeometryEncodingKey)).map(GeometryEncoding.apply).getOrElse(GeometryEncoding.GeoParquetWkb)
148       // only include bboxes if they help with push-down filters
149       val bboxes =
150         if (Option(conf.get(BBoxEncodingKey)).forall(_.toBoolean)) { BoundingBoxes(sft, geometries) } else { BoundingBoxes(Seq.empty) }
151       val visibilities =
152         Option(sft.getUserData.get(VisibilityEncodingKey)).orElse(Option(conf.get(VisibilityEncodingKey))).forall(_.toString.toBoolean)
153       val metadata = Map(
154         SftNameKey -> name,
155         SftSpecKey -> spec,
156         GeometryEncodingKey -> geometries.toString,
157       ) ++ Option(conf.get(PartitionKey)).map(PartitionKey -> _)
158       val encodings = Encodings(geometries)
159       val messageType = schema(sft, None, encodings, bboxes, visibilities)
160       SimpleFeatureParquetSchema(sft, encodings, visibilities, bboxes, metadata.asJava, messageType)
161     }
162   }

177   /**
178    * Get the schema for a file being read
179    *
180    * @param fileSchema message type
181    * @param metadata file metadata
182    * @return
183    */
184   private def read(fileSchema: MessageType, metadata: Map[String, String]): Option[SimpleFeatureParquetSchema] = {
185     for {
186       name <- metadata.get(SftNameKey)
187       spec <- metadata.get(SftSpecKey)
188     } yield {
189       val sft = SimpleFeatureTypes.createType(name, spec)
190       val readSft = metadata.get(SftReadSpecKey).map(SimpleFeatureTypes.createType(name, _))
191       val geometries = metadata.get(GeometryEncodingKey).map(GeometryEncoding.apply).getOrElse {
192         throw new UnsupportedOperationException("GeoMesaV0/GeoMesaV1 encoding is no longer supported")
193       }
194       val encodings = Encodings(geometries)
195       val bboxes = BoundingBoxes(fileSchema.getFields.asScala.map(_.getName).flatMap(BoundingBoxField.fromBoundingBox).toSeq)
196       val visibilities = fileSchema.containsField(VisibilitiesField)
197       val schema = readSchema(fileSchema, sft, readSft, encodings, bboxes, visibilities)
198       SimpleFeatureParquetSchema(readSft.getOrElse(sft), encodings, visibilities, bboxes, metadata.asJava, schema)
199     }
200   }

202   /**
203    * Gets the projected schema used to read a (possibly transformed) sft
204    *
205    * @param fileSchema schema of the file being read
206    * @param sft feature type corresponding to the file being read
207    * @param readSft transformed feature type being read (may the same as the main sft)
208    * @param encodings type encoding
209    * @param bboxes fields with bounding boxes
210    * @param visibilities whether the schema encodes visibilities, or not
211    * @return
212    */
213   private def readSchema(
214       fileSchema: MessageType,
215       sft: SimpleFeatureType,
216       readSft: Option[SimpleFeatureType],
217       encodings: Encodings,
218       bboxes: BoundingBoxes,
219       visibilities: Boolean): MessageType = {
220     val consistentSchema = schema(sft, readSft, encodings, bboxes, visibilities) // current schema
221     if (fileSchema.getFieldName(0) == SimpleFeatureParquetSchema.FeatureIdField) {
222       consistentSchema
223     } else {
224       def getMappedField(name: String): Type = consistentSchema.getFields.get(consistentSchema.getFieldIndex(name))
225       // old files - attributes are first, then fid, vis, bboxes
226       val attributes = readSft.getOrElse(sft).getAttributeDescriptors.asScala.map { d =>
227         getMappedField(alphaNumericSafeString(d.getLocalName))
228       }
229       val id = getMappedField(FeatureIdField)
230       val vis = if (!visibilities) { None } else { Some(getMappedField(VisibilitiesField)) }
231       val boxes = sft.getAttributeDescriptors.asScala.flatMap { d =>
232         bboxes.get(d.getLocalName).map(getMappedField)
233       }
234       val fields = attributes ++ Seq(id) ++ vis ++ boxes
235       new MessageType(alphaNumericSafeString(sft.getTypeName), fields.asJava)
236     }
237   }

239   /**
240    * Get the message type for a simple feature type. We need the full sft in order to ensure field ids are
241    * consistent, but the schema may be reduced (e.g. on read) based on the filter sft
242    *
243    * @param sft simple feature type
244    * @param readSft optional feature type used to filter the fields in the schema
245    * @param encodings field type encoding
246    * @param bboxes include bounding boxes for each row
247    * @param visibilities include visibilities
248    * @return
249    */
250   private def schema(
251       sft: SimpleFeatureType,
252       readSft: Option[SimpleFeatureType],
253       encodings: Encodings,
254       bboxes: BoundingBoxes,
255       visibilities: Boolean): MessageType = {
256     // note: for iceberg compatibility, field ids need to start at one and increment (without gaps) across all top-level fields.
257     // All nested fields (structs, lists) get ids *after* all the top-level fields, once again incrementing without gaps
258     val fieldIds = new AtomicInteger(1)
259     val id =
260       Types.required(PrimitiveTypeName.BINARY)
261         .id(fieldIds.getAndIncrement())
262         .as(LogicalTypeAnnotation.stringType())
263         .named(FeatureIdField)
264     val vis = if (!visibilities) { None } else {
265       val f =
266         Types.optional(PrimitiveTypeName.BINARY)
267           .id(fieldIds.getAndIncrement())
268           .as(LogicalTypeAnnotation.stringType())
269           .named(VisibilitiesField)
270       Some(f)
271     }
272     val attributes = {
273       val builder = Map.newBuilder[String, Seq[Type]]
274       val nestedFieldIds = new AtomicInteger(fieldIds.get() + sft.getAttributeCount + bboxes.fields.size)
275       sft.getAttributeDescriptors.asScala.foreach { d =>
276         val name = alphaNumericSafeString(d.getLocalName)
277         val types =
278           Seq(buildType(name, ObjectType.selectType(d), encodings, fieldIds, nestedFieldIds)) ++
279             bboxes.get(d.getLocalName).map(bbox => BoundingBoxField.schema(bbox, fieldIds, nestedFieldIds))
280          builder += name -> types
281       }
282       builder.result()
283     }
284     val attributeFields = readSft.getOrElse(sft).getAttributeDescriptors.asScala.flatMap { d =>
285       attributes(alphaNumericSafeString(d.getLocalName))
286     }
287     // note: id field goes at the front of the record, then vis, then attributes and bounding boxes
288     val fields = Seq(id) ++ vis ++ attributeFields
289     val name = alphaNumericSafeString(sft.getTypeName)
290     new MessageType(name, fields.asJava)
291   }

293   private def icebergSchema(schema: SimpleFeatureParquetSchema): Schema = {
294     val fields = new java.util.ArrayList[NestedField](schema.sft.getAttributeCount + 3)
295     val aliases = new java.util.HashMap[String, Integer](schema.sft.getAttributeCount + 3)
296 
297     // see note in `schema` (above) about field ids
298     val fieldIds = new AtomicInteger(1)
299 
300     aliases.put(FeatureIdField, fieldIds.get())
301     fields.add(NestedField.required(fieldIds.getAndIncrement(), FeatureIdField, StringType.get()))
302 
303     if (schema.hasVisibilities) {
304       aliases.put(VisibilitiesField, fieldIds.get())
305       fields.add(NestedField.optional(fieldIds.getAndIncrement(), VisibilitiesField, StringType.get()))
306     }
307 
308     val nestedFieldIds = new AtomicInteger(fieldIds.get() + schema.sft.getAttributeCount + schema.bboxes.fields.size)
309     schema.sft.getAttributeDescriptors.asScala.foreach { d =>
310       val name = alphaNumericSafeString(d.getLocalName)
311       aliases.put(name, fieldIds.get())
312       fields.add(buildIcebergType(name, ObjectType.selectType(d), schema.encodings, fieldIds, nestedFieldIds))
313       schema.bboxes.get(d.getLocalName).foreach { bbox =>
314         aliases.put(bbox, fieldIds.get())
315         fields.add(BoundingBoxField.icebergSchema(bbox, fieldIds, nestedFieldIds))
316       }
317     }
318     new Schema(fields, aliases, java.util.Set.of[Integer](1))
319   }

321   /**
322    * Builds the schema type for an attribute
323    *
324    * @param name field name
325    * @param bindings object type
326    * @param encodings schema encodings
327    * @param repetition repetition
328    * @return
329    */
330   private def buildType(
331       name: String,
332       bindings: Seq[ObjectType],
333       encodings: Encodings,
334       fieldIds: AtomicInteger,
335       nestedFieldIds: AtomicInteger,
336       repetition: Repetition = Repetition.OPTIONAL): Type = {
337     val builder = bindings.head match {
338       case ObjectType.INT     => Types.primitive(PrimitiveTypeName.INT32, repetition).id(fieldIds.getAndIncrement())
339       case ObjectType.DOUBLE  => Types.primitive(PrimitiveTypeName.DOUBLE, repetition).id(fieldIds.getAndIncrement())
340       case ObjectType.LONG    => Types.primitive(PrimitiveTypeName.INT64, repetition).id(fieldIds.getAndIncrement())
341       case ObjectType.FLOAT   => Types.primitive(PrimitiveTypeName.FLOAT, repetition).id(fieldIds.getAndIncrement())
342       case ObjectType.BOOLEAN => Types.primitive(PrimitiveTypeName.BOOLEAN, repetition).id(fieldIds.getAndIncrement())
343       case ObjectType.BYTES   => Types.primitive(PrimitiveTypeName.BINARY, repetition).id(fieldIds.getAndIncrement())
344 
345       case ObjectType.STRING =>
346         Types.primitive(PrimitiveTypeName.BINARY, repetition)
347           .id(fieldIds.getAndIncrement())
348           .as(LogicalTypeAnnotation.stringType())
349 
350       case ObjectType.DATE =>
351         Types.primitive(PrimitiveTypeName.INT64, repetition)
352           .id(fieldIds.getAndIncrement())
353           .as(LogicalTypeAnnotation.timestampType(true, TimeUnit.MICROS))
354 
355       case ObjectType.UUID =>
356         Types.primitive(PrimitiveTypeName.FIXED_LEN_BYTE_ARRAY, repetition)
357           .id(fieldIds.getAndIncrement())
358           .length(16)
359           .as(LogicalTypeAnnotation.uuidType())
360 
361       case ObjectType.LIST =>
362         Types.optionalList()
363           .id(fieldIds.getAndIncrement())
364           .element(buildType("element", bindings.drop(1), encodings, nestedFieldIds, null /* should not be used*/, Repetition.REQUIRED))
365 
366       case ObjectType.MAP =>
367         Types.optionalMap()
368           .id(fieldIds.getAndIncrement())
369           .key(buildType("key", bindings.slice(1, 2), encodings, nestedFieldIds, null /* should not be used*/, Repetition.REQUIRED))
370           .value(buildType("value", bindings.slice(2, 3), encodings, nestedFieldIds, null /* should not be used*/))
371 
372       case ObjectType.GEOMETRY =>
373         encodings.geometry.schema(bindings(1), fieldIds, nestedFieldIds)
374 
375       case binding =>
376         throw new UnsupportedOperationException(s"No mapping defined for type $binding with encoding $encodings")
377     }
378     builder.named(name)
379   }

381   /**
382    * Builds the schema type for an attribute
383    *
384    * @param name field name
385    * @param bindings object type
386    * @param encodings schema encodings
387    * @return
388    */
389   private def buildIcebergType(
390       name: String,
391       bindings: Seq[ObjectType],
392       encodings: Encodings,
393       fieldIds: AtomicInteger,
394       nestedFieldIds: AtomicInteger): NestedField = {
395     val builder = NestedField.optional(name).withId(fieldIds.getAndIncrement())
396     val typed: NestedField.Builder = bindings.head match {
397       case ObjectType.INT     => builder.ofType(IntegerType.get())
398       case ObjectType.DOUBLE  => builder.ofType(DoubleType.get())
399       case ObjectType.LONG    => builder.ofType(LongType.get())
400       case ObjectType.FLOAT   => builder.ofType(FloatType.get())
401       case ObjectType.BOOLEAN => builder.ofType(BooleanType.get())
402       case ObjectType.BYTES   => builder.ofType(BinaryType.get())
403       case ObjectType.STRING  => builder.ofType(StringType.get())
404       case ObjectType.DATE    => builder.ofType(TimestampType.withoutZone())
405       case ObjectType.UUID    => builder.ofType(UUIDType.get())
406 
407       case ObjectType.LIST =>
408         val subType = buildIcebergType("", bindings.drop(1), encodings, nestedFieldIds, null /* should not be used*/)
409         builder.ofType(ListType.ofRequired(subType.fieldId(), subType.`type`()))
410 
411       case ObjectType.MAP =>
412         val keyType = buildIcebergType("", bindings.slice(1, 2), encodings, nestedFieldIds, null /* should not be used*/)
413         val valueType = buildIcebergType("", bindings.slice(2, 3), encodings, nestedFieldIds, null /* should not be used*/)
414         builder.ofType(MapType.ofRequired(keyType.fieldId(), valueType.fieldId(), keyType.`type`(), valueType.`type`()))
415 
416       case ObjectType.GEOMETRY =>
417         // not yet supported in spark or trino
418         // builder.ofType(GeometryType.crs84())
419         builder.ofType(BinaryType.get())
420 
421       case binding =>
422         throw new UnsupportedOperationException(s"No mapping defined for type $binding with encoding $encodings")
423     }
424     typed.build()
425   }
