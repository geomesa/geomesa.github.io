108   /**
109     * Extract the simple feature type from a parquet read context. The read context
110     * contains both file metadata and the provided read conf
111     *
112     * @param context parquet read context
113     * @return
114     */
115   def read(context: InitContext): Option[SimpleFeatureParquetSchema] = {
116     // copy in the file level metadata
117     val kvMeta = context.getKeyValueMetadata.asScala.collect { case (k, v) if !v.isEmpty => k -> v.iterator.next }
118     // noinspection ScalaDeprecation
119     val gmMeta = Seq(SftNameKey, SftSpecKey, SftReadSpecKey, GeometryEncodingKey).flatMap { key =>
120       val value =
121         try { context.getParquetConfiguration.get(key) } catch {
122           // noinspection ScalaDeprecation - use the older method for spark compatibility
123           case _: NoSuchMethodError => context.getConfiguration.get(key)
124         }
125       Option(value).map(key -> _)
126     }
127     // note: context meta goes second so that transforms override the file-level keys
128     read(context.getFileSchema, kvMeta.toMap ++ gmMeta.toMap)
129   }

140   /**
141    * Get a schema for writing. Encoding can be configured through `geomesa.parquet.geometries` and `geomesa.fs.visibilities`
142    *
143    * @param conf write configuration, including the sft spec
144    * @return
145    */
146   def write(conf: ParquetConfiguration): Option[SimpleFeatureParquetSchema] = {
147     for {
148       name <- Option(conf.get(SftNameKey))
149       spec <- Option(conf.get(SftSpecKey))
150     } yield {
151       val sft = SimpleFeatureTypes.createType(name, spec)
152       val geometries = Option(conf.get(GeometryEncodingKey)).map(GeometryEncoding.apply).getOrElse(GeometryEncoding.GeoParquetWkb)
153       val bboxes =
154         if (Option(conf.get(BBoxEncodingKey)).forall(_.toBoolean)) { BoundingBoxes(sft) } else { BoundingBoxes(Seq.empty) }
155       val zCols = if (Option(conf.get(ZValueColumKey)).forall(_.toBoolean)) { ZValues(sft) } else { ZValues(Seq.empty) }
156       val visibilities =
157         Option(sft.getUserData.get(VisibilityEncodingKey)).orElse(Option(conf.get(VisibilityEncodingKey))).forall(_.toString.toBoolean)
158       val metadata = Map(
159         SftNameKey -> name,
160         SftSpecKey -> spec,
161         GeometryEncodingKey -> geometries.toString,
162       ) ++ Option(conf.get(PartitionKey)).map(PartitionKey -> _)
163       val encodings = Encodings(geometries)
164       val messageType = schema(sft, encodings, bboxes, zCols, visibilities).toMessageType(sft)
165       SimpleFeatureParquetSchema(sft, encodings, visibilities, bboxes, zCols, metadata.asJava, messageType)
166     }
167   }

182   /**
183    * Get the schema for a file being read
184    *
185    * @param fileSchema message type
186    * @param metadata file metadata
187    * @return
188    */
189   private def read(fileSchema: MessageType, metadata: Map[String, String]): Option[SimpleFeatureParquetSchema] = {
190     for {
191       name <- metadata.get(SftNameKey)
192       spec <- metadata.get(SftSpecKey)
193     } yield {
194       val sft = SimpleFeatureTypes.createType(name, spec)
195       val readSft = metadata.get(SftReadSpecKey).map(SimpleFeatureTypes.createType(name, _))
196       val geometries = metadata.get(GeometryEncodingKey).map(GeometryEncoding.apply).getOrElse {
197         throw new UnsupportedOperationException("GeoMesaV0/GeoMesaV1 encoding is no longer supported")
198       }
199       val encodings = Encodings(geometries)
200       val bboxes = BoundingBoxes(fileSchema.getFields.asScala.map(_.getName).flatMap(BoundingBoxField.fromFieldName).toSeq)
201       val zCols = ZValues(fileSchema.getFields.asScala.map(_.getName).flatMap(ZValueField.fromFieldName).toSeq)
202       val visibilities = fileSchema.containsField(VisibilitiesField)
203       val schema = readSchema(fileSchema, sft, readSft, encodings, bboxes, zCols, visibilities)
204       SimpleFeatureParquetSchema(readSft.getOrElse(sft), encodings, visibilities, bboxes, zCols, metadata.asJava, schema)
205     }
206   }

230   /**
231    * Get the message type for a simple feature type. We need the full sft in order to ensure field ids are
232    * consistent, but the schema may be reduced (e.g. on read) based on the filter sft
233    *
234    * @param sft simple feature type
235    * @param encodings field type encoding
236    * @param bboxes include bounding boxes for each row
237    * @param visibilities include visibilities
238    * @return
239    */
240   private def schema(
241       sft: SimpleFeatureType,
242       encodings: Encodings,
243       bboxes: BoundingBoxes,
244       zValues: ZValues,
245       visibilities: Boolean): SchemaFields = {
246     // note: for iceberg compatibility, field ids need to start at one and increment (without gaps) across all top-level fields.
247     // All nested fields (structs, lists) get ids *after* all the top-level fields, once again incrementing without gaps
248     val fieldIds = new AtomicInteger(1)
249     val id =
250       Types.required(PrimitiveTypeName.BINARY)
251         .id(fieldIds.getAndIncrement())
252         .as(LogicalTypeAnnotation.stringType())
253         .named(FeatureIdField)
254     val vis = if (!visibilities) { None } else {
255       val f =
256         Types.optional(PrimitiveTypeName.BINARY)
257           .id(fieldIds.getAndIncrement())
258           .as(LogicalTypeAnnotation.stringType())
259           .named(VisibilitiesField)
260       Some(f)
261     }
262     val attributes = {
263       val builder = Map.newBuilder[String, Seq[Type]]
264       val nestedFieldIds = new AtomicInteger(fieldIds.get() + sft.getAttributeCount + bboxes.fields.size + zValues.fields.size)
265       sft.getAttributeDescriptors.asScala.foreach { d =>
266         val name = alphaNumericSafeString(d.getLocalName)
267         val types =
268           Seq(buildType(name, ObjectType.selectType(d), encodings, fieldIds, nestedFieldIds)) ++
269             bboxes.get(d.getLocalName).map(bbox => BoundingBoxField.schema(bbox, fieldIds, nestedFieldIds)) ++
270             zValues.get(d.getLocalName).map(zValue => ZValueField.schema(zValue, fieldIds))
271          builder += name -> types
272       }
273       builder.result()
274     }
275     SchemaFields(id, vis, attributes)
276   }

278   private def icebergSchema(schema: SimpleFeatureParquetSchema): Schema = {
279     val fields = new java.util.ArrayList[NestedField](schema.sft.getAttributeCount + 3)
280     val aliases = new java.util.HashMap[String, Integer](schema.sft.getAttributeCount + 3)
281 
282     // see note in `schema` (above) about field ids
283     val fieldIds = new AtomicInteger(1)
284 
285     aliases.put(FeatureIdField, fieldIds.get())
286     fields.add(NestedField.required(fieldIds.getAndIncrement(), FeatureIdField, StringType.get()))
287 
288     if (schema.hasVisibilities) {
289       aliases.put(VisibilitiesField, fieldIds.get())
290       fields.add(NestedField.optional(fieldIds.getAndIncrement(), VisibilitiesField, StringType.get()))
291     }
292 
293     val nestedFieldIds =
294       new AtomicInteger(fieldIds.get() + schema.sft.getAttributeCount + schema.bboxes.fields.size + schema.zValues.fields.size)
295     schema.sft.getAttributeDescriptors.asScala.foreach { d =>
296       val name = alphaNumericSafeString(d.getLocalName)
297       aliases.put(name, fieldIds.get())
298       fields.add(buildIcebergType(name, ObjectType.selectType(d), schema.encodings, fieldIds, nestedFieldIds))
299       schema.bboxes.get(d.getLocalName).foreach { bbox =>
300         aliases.put(bbox, fieldIds.get())
301         fields.add(BoundingBoxField.icebergSchema(bbox, fieldIds, nestedFieldIds))
302       }
303       schema.zValues.get(d.getLocalName).foreach { zValue =>
304         aliases.put(zValue, fieldIds.get())
305         fields.add(ZValueField.icebergSchema(zValue, fieldIds))
306       }
307     }
308     new Schema(fields, aliases, java.util.Set.of[Integer](1))
309   }

311   /**
312    * Builds the schema type for an attribute
313    *
314    * @param name field name
315    * @param bindings object type
316    * @param encodings schema encodings
317    * @param repetition repetition
318    * @return
319    */
320   private def buildType(
321       name: String,
322       bindings: Seq[ObjectType],
323       encodings: Encodings,
324       fieldIds: AtomicInteger,
325       nestedFieldIds: AtomicInteger,
326       repetition: Repetition = Repetition.OPTIONAL): Type = {
327     val builder = bindings.head match {
328       case ObjectType.INT     => Types.primitive(PrimitiveTypeName.INT32, repetition).id(fieldIds.getAndIncrement())
329       case ObjectType.DOUBLE  => Types.primitive(PrimitiveTypeName.DOUBLE, repetition).id(fieldIds.getAndIncrement())
330       case ObjectType.LONG    => Types.primitive(PrimitiveTypeName.INT64, repetition).id(fieldIds.getAndIncrement())
331       case ObjectType.FLOAT   => Types.primitive(PrimitiveTypeName.FLOAT, repetition).id(fieldIds.getAndIncrement())
332       case ObjectType.BOOLEAN => Types.primitive(PrimitiveTypeName.BOOLEAN, repetition).id(fieldIds.getAndIncrement())
333       case ObjectType.BYTES   => Types.primitive(PrimitiveTypeName.BINARY, repetition).id(fieldIds.getAndIncrement())
334 
335       case ObjectType.STRING =>
336         Types.primitive(PrimitiveTypeName.BINARY, repetition)
337           .id(fieldIds.getAndIncrement())
338           .as(LogicalTypeAnnotation.stringType())
339 
340       case ObjectType.DATE =>
341         Types.primitive(PrimitiveTypeName.INT64, repetition)
342           .id(fieldIds.getAndIncrement())
343           .as(LogicalTypeAnnotation.timestampType(true, TimeUnit.MICROS))
344 
345       case ObjectType.UUID =>
346         Types.primitive(PrimitiveTypeName.FIXED_LEN_BYTE_ARRAY, repetition)
347           .id(fieldIds.getAndIncrement())
348           .length(16)
349           .as(LogicalTypeAnnotation.uuidType())
350 
351       case ObjectType.LIST =>
352         Types.optionalList()
353           .id(fieldIds.getAndIncrement())
354           .element(buildType("element", bindings.drop(1), encodings, nestedFieldIds, null /* should not be used*/, Repetition.REQUIRED))
355 
356       case ObjectType.MAP =>
357         Types.optionalMap()
358           .id(fieldIds.getAndIncrement())
359           .key(buildType("key", bindings.slice(1, 2), encodings, nestedFieldIds, null /* should not be used*/, Repetition.REQUIRED))
360           .value(buildType("value", bindings.slice(2, 3), encodings, nestedFieldIds, null /* should not be used*/))
361 
362       case ObjectType.GEOMETRY =>
363         encodings.geometry.schema(bindings(1), fieldIds, nestedFieldIds)
364 
365       case binding =>
366         throw new UnsupportedOperationException(s"No mapping defined for type $binding with encoding $encodings")
367     }
368     builder.named(name)
369   }

371   /**
372    * Builds the schema type for an attribute
373    *
374    * @param name field name
375    * @param bindings object type
376    * @param encodings schema encodings
377    * @return
378    */
379   private def buildIcebergType(
380       name: String,
381       bindings: Seq[ObjectType],
382       encodings: Encodings,
383       fieldIds: AtomicInteger,
384       nestedFieldIds: AtomicInteger): NestedField = {
385     val builder = NestedField.optional(name).withId(fieldIds.getAndIncrement())
386     val typed: NestedField.Builder = bindings.head match {
387       case ObjectType.INT     => builder.ofType(IntegerType.get())
388       case ObjectType.DOUBLE  => builder.ofType(DoubleType.get())
389       case ObjectType.LONG    => builder.ofType(LongType.get())
390       case ObjectType.FLOAT   => builder.ofType(FloatType.get())
391       case ObjectType.BOOLEAN => builder.ofType(BooleanType.get())
392       case ObjectType.BYTES   => builder.ofType(BinaryType.get())
393       case ObjectType.STRING  => builder.ofType(StringType.get())
394       case ObjectType.DATE    => builder.ofType(TimestampType.withoutZone())
395       case ObjectType.UUID    => builder.ofType(UUIDType.get())
396 
397       case ObjectType.LIST =>
398         val subType = buildIcebergType("", bindings.drop(1), encodings, nestedFieldIds, null /* should not be used*/)
399         builder.ofType(ListType.ofRequired(subType.fieldId(), subType.`type`()))
400 
401       case ObjectType.MAP =>
402         val keyType = buildIcebergType("", bindings.slice(1, 2), encodings, nestedFieldIds, null /* should not be used*/)
403         val valueType = buildIcebergType("", bindings.slice(2, 3), encodings, nestedFieldIds, null /* should not be used*/)
404         builder.ofType(MapType.ofRequired(keyType.fieldId(), valueType.fieldId(), keyType.`type`(), valueType.`type`()))
405 
406       case ObjectType.GEOMETRY =>
407         // not yet supported in spark or trino
408         // builder.ofType(GeometryType.crs84())
409         builder.ofType(BinaryType.get())
410 
411       case binding =>
412         throw new UnsupportedOperationException(s"No mapping defined for type $binding with encoding $encodings")
413     }
414     typed.build()
415   }

417   private case class SchemaFields(fid: Type, vis: Option[Type], attributes: Map[String, Seq[Type]]) {
418     def toMessageType(sft: SimpleFeatureType, excludeBBoxes: Boolean = false, excludeZValues: Boolean = false): MessageType = {
419       // note: id field goes at the front of the record, then vis, then attributes and bounding boxes
420       val fields = Seq(fid) ++ vis ++ {
421         val all = sft.getAttributeDescriptors.asScala.flatMap { d =>
422           attributes(alphaNumericSafeString(d.getLocalName))
423         }
424         if (!excludeBBoxes && !excludeZValues) {
425           all
426         } else {
427           all.filterNot { f =>
428             (excludeBBoxes && BoundingBoxField.fromFieldName(f.getName).isDefined) ||
429               (excludeZValues && ZValueField.fromFieldName(f.getName).isDefined)
430           }
431         }
432       }
433       val name = alphaNumericSafeString(sft.getTypeName)
434       new MessageType(name, fields.asJava)
435     }
436   }
