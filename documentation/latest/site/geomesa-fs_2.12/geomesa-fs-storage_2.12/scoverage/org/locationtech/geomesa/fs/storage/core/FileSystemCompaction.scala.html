22 /** Stages complete replacement files before publishing a single validated Iceberg rewrite. */
23 class FileSystemCompaction(storage: FileSystemStorage, partitions: Seq[Partition], target: Option[Long]) {
24 
25   import FileSystemCompaction.{Output, Plan}
26 
27   import scala.collection.JavaConverters._
28 
29   def compact(): Unit = {
30     val plans = plan()
31     val outputs = plans.map(prepare)
32     commit(plans, outputs)
33   }
34 
35   private[core] def plan(): Seq[Plan] = {
36     storage.table.refresh()
37     require(storage.schema.schema.sameSchema(storage.table.schema()), "Reload storage after a table schema change")
38     require(storage.partitionSpecId == storage.table.spec().specId(), "Reload storage after a partition spec change")
39     Option(storage.table.currentSnapshot()).toSeq.flatMap { snapshot =>
40       val selected = partitions.toSet
41       val files = WithClose(storage.table.newScan().useSnapshot(snapshot.snapshotId()).planFiles()) { tasks =>
42         tasks.asScala.map(_.file().copy()).filter { file =>
43           require(storage.table.spec().compatibleWith(storage.table.specs().get(file.specId())),
44             "Compaction requires input files with a compatible partition spec")
45           selected.contains(storage.metadata.partition(file)) && !target.exists(storage.sizer.fileIsSized(file, _))
46         }.toList
47       }
48       files.groupBy(storage.metadata.partition).toSeq.map { case (partition, inputs) =>
49         Plan(snapshot.snapshotId(), storage.table.schema().schemaId(), storage.table.spec().specId(), partition, inputs)
50       }
51     }
52   }
53 
54   /** Writes all live rows without applying the caller's authorizations. No files are registered here. */
55   private[core] def prepare(plan: Plan): Output = {
56     validateSchema(plan.schemaId, plan.specId)
57     val paths = ArrayBuffer.empty[String]
58     val inputs = plan.files.map(_.location()).toSet
59     val readSchema = storage.schema.read(None, None, Set.empty).schema
60     val writeConf = storage.conf ++ ParquetFileSystemWriter.icebergBloomFilterConf(storage.table.properties()) ++
61       Map(SimpleFeatureSchema.PartitionKey -> plan.partition.toString)
62     val estimator = target.map(storage.sizer.estimator)
63     var writer: ParquetFileSystemWriter = null
64     var rows = 0L
65     var totalBytes = 0L
66     var remaining = estimator.fold(Long.MaxValue)(_.estimate(0L))
67 
68     def closeWriter(): Unit = if (writer != null) {
69       writer.close()
70       totalBytes += writer.size
71       writer = null
72     }
73 
74     try {
75       WithClose(new IcebergParquetScan(storage.table, readSchema, Expressions.alwaysTrue(), 1,
76           Some(inputs.contains), Some(plan.snapshotId))) { reader =>
77         reader.foreach { feature =>
78           if (writer == null) {
79             val path = storage.newFilePath()
80             paths += path
81             writer = ParquetFileSystemWriter(storage.schema, writeConf, storage.table.io(), path, NoOpObserver)
82           }
83           writer.write(feature)
84           rows += 1
85           remaining -= 1
86           if (remaining <= 0) {
87             closeWriter()
88             estimator.foreach(_.update(totalBytes, rows))
89             remaining = estimator.fold(Long.MaxValue)(_.estimate(0L))
90           }
91         }
92       }
93       closeWriter()
94       // Do not update table properties while files are merely staged.
95       Output(plan.partition, paths.toList, rows)
96     } catch {
97       case NonFatal(e) =>
98         CloseQuietly(Option(writer)).foreach(e.addSuppressed)
99         paths.foreach { path =>
100           try { storage.table.io().deleteFile(path) } catch { case NonFatal(cleanup) => e.addSuppressed(cleanup) }
101         }
102         throw e
103     }
104   }
105 
106   /**
107    * The compaction reader applies positional deletes, including deletion vectors,
108    * so replacement files contain only surviving rows.
109    * Iceberg removes deletion-vector entries for rewritten inputs during commit.
110    * Legacy positional-delete entries may remain until sequence-number cleanup removes them.
111    * Physical files remain available to retained snapshots until snapshot expiration
112    * can safely remove them.
113    */
114   private[core] def commit(plans: Seq[Plan], outputs: Seq[Output]): Unit = {
115     if (plans.isEmpty) return
116     require(plans.map(_.snapshotId).distinct.size == 1, "Compaction plans must share one snapshot")
117     require(outputs.map(_.partition).toSet == plans.map(_.partition).toSet && outputs.size == plans.size,
118       "Compaction results must cover every planned partition exactly once")
119     require(plans.map(_.schemaId).distinct.size == 1 && plans.map(_.specId).distinct.size == 1,
120       "Compaction plans must share one schema and partition spec")
121     validateSchema(plans.head.schemaId, plans.head.specId)
122     require(outputs.flatMap(_.paths).distinct.size == outputs.map(_.paths.size).sum,
123       "Compaction results contain duplicate output paths")
124     val spec = storage.table.specs().get(plans.head.specId)
125     val replacements = outputs.flatMap { output =>
126       val files = output.paths.map(storage.toDataFile(_, output.partition, spec))
127       require(output.rows >= 0 && files.map(_.recordCount()).sum == output.rows,
128         "Compaction result row counts do not match replacement files")
129       files
130     }
131     validateSchema(plans.head.schemaId, plans.head.specId)
132     val rewrite = storage.table.newRewrite().validateFromSnapshot(plans.head.snapshotId)
133     plans.flatMap(_.files).foreach(rewrite.deleteFile)
134     replacements.foreach(rewrite.addFile)
135     // Iceberg validates both missing source files and concurrent row-level deletes during commit.
136     // Leave staged files for orphan cleanup if commit fails: its final status can be unknown.
137     rewrite.commit()
138   }
139 
140   private def validateSchema(schemaId: Int, specId: Int): Unit = {
141     storage.table.refresh()
142     require(storage.table.schema().schemaId() == schemaId && storage.table.spec().specId() == specId,
143       "Table schema or partition spec changed during compaction")
144   }
145 }
