23 /**
24  * Parquet input file implementation for reading from S3. Uses the aws-s3-analyticsaccelerator project to improve
25  * read times.
26  *
27  * @param fs object store impl
28  * @param path file path
29  */
30 class S3InputFile(fs: S3ObjectStore, path: URI) extends InputFile {
31 
32   private val key = S3ObjectStore.parseS3Path(path)
33 
34   override lazy val getLength: Long = fs.size(path)
35 
36   override def newStream(): SeekableInputStream = new AcceleratedS3SeekableInputStream(fs.createStream(key.bucket, key.key))
37 }

54     override def read(buf: ByteBuffer): Int = {
55       val max = math.min(buf.remaining(), this.buf.length)
56       val count = s3.read(this.buf, 0, max)
57       if (count > 0) {

61     }

102       val allocate: IntFunction[ByteBuffer] = size => allocator.allocate(size)
