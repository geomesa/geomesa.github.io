25 /**
26  * A transaction against a file system storage instance. Writers created with a transaction do not modify the
27  * table - instead, all the data files and row-level deletes they produce are published in a single atomic commit
28  * when the transaction is committed, or discarded when it is rolled back.
29  *
30  * Reads never see uncommitted changes, including reads made by update writers in the same transaction. As a
31  * consequence, a row can only be modified once per transaction.
32  *
33  * A transaction is single-use - once it has been committed or rolled back, it can't be used again. Instances
34  * are thread-safe.
35  *
36  * Transactions are created through `FileSystemStorage.newTransaction`. Alternatively, `StorageTransaction.AutoCommit`
37  * may be used to have each writer commit its own changes when it is closed.
38  */
39 sealed trait StorageTransaction {
40 
41   /**
42    * Is this transaction still usable, i.e. it has not been committed or rolled back
43    *
44    * @return
45    */
46   def isOpen: Boolean
47 
48   /**
49    * Publishes all changes made in this transaction in a single commit. All writers must be closed first.
50    *
51    * If the commit fails, the transaction is left as-is, and may be re-tried or rolled back - unless the result
52    * of the commit is unknown, in which case the transaction is finished, and any files are left in place.
53    *
54    * @throws ConflictException if a concurrent operation modified the same rows or removed the files they are in
55    */
56   @throws(classOf[ConflictException])
57   def commit(): Unit
58 
59   /**
60    * Discards all changes made in this transaction, deleting any files that were written. Any writers that are
61    * still open will fail when they are closed.
62    *
63    * Has no effect if the transaction has already been committed or rolled back.
64    */
65   def rollback(): Unit
66 }

103     private var failure: Throwable = _

109     override def commit(): Unit = {
110       val toClose = synchronized {
111         doCommit()
112         val toClose = callbacks.values.toList
113         finish("committed")
114         toClose
115       }
116       // note: run outside of the lock, as the callbacks may be slow
117       toClose.foreach(CloseWithLogging(_))
118     }

120     private def doCommit(): Unit = {
121       checkOpen()
122       if (open > 0) {
123         throw new IllegalStateException(s"Cannot commit a transaction with $open open writer(s) - close all writers first")
124       } else if (failure != null) {
125         throw new IllegalStateException("Cannot commit a transaction after a failed write - it must be rolled back", failure)
126       }
127       if (dataFiles.nonEmpty || deletes.nonEmpty) {
128         val retries =
129           PropertyUtil.propertyAsInt(table.properties(), TableProperties.COMMIT_NUM_RETRIES, TableProperties.COMMIT_NUM_RETRIES_DEFAULT)
130         var attempt = 0
131         var done = false
132         while (!done) {
133           val puffins = ArrayBuffer.empty[String]
134           try {
135             val delta = table.newRowDelta()
136             dataFiles.foreach(delta.addRows)
137             if (deletes.nonEmpty) {
138               val result = writeDeletes(puffins)
139               result.deleteFiles().forEach(f => delta.addDeletes(f))
140               // remove any previous DVs whose deletes we merged into the newly written DVs - a data file can only
141               // be referenced by a single DV
142               result.rewrittenDeleteFiles().forEach(f => delta.removeDeletes(f))
143               // fail if any of the files we're deleting from, or the DVs we're replacing, have been removed.
144               // iceberg always fails if a DV has been added for the same data file, which is handled below
145               pinned().snapshot.foreach(s => delta.validateFromSnapshot(s.snapshotId()))
146               delta.validateDataFilesExist(result.referencedDataFiles()).validateDeletedFiles()
147             }
148             delta.commit()
149             done = true
150           } catch {
151             case e: CommitStateUnknownException =>
152               // we can't tell if the files are referenced by the table or not, so leave them for orphan cleanup
153               logger.warn(s"Leaving ${dataFiles.size} data file(s) in place as the result of the commit is unknown")
154               finish("committed with an unknown result")
155               throw e
156 
157             case NonFatal(e) =>
158               // the deletes are re-written on each attempt, as they have to be merged with the current table state
159               puffins.foreach(storage.deleteQuietly)
160               if (deletes.isEmpty || !isConflict(e)) {
161                 throw e
162               } else if (attempt >= retries) {
163                 throw new ConflictException(s"Transaction could not be committed after ${attempt + 1} attempt(s) due to concurrent modifications", e)
164               }
165               attempt += 1
166               rebase(e)
167           }
168         }
169       }
170     }
171 
172     override def rollback(): Unit = synchronized {
173       if (finished == null) {
174         dataFiles.foreach(f => storage.deleteQuietly(f.location()))
175         finish("rolled back")
176       }
177     }

180      * Commits the transaction, rolling it back if the commit fails (unless the result of the commit is unknown)

183       try { commit() } catch {
184         case e: CommitStateUnknownException => throw e
185         case NonFatal(e) => rollbackQuietly(e); throw e
186       }

269     /**
270      * Checks that a row can be deleted, without deleting it
271      *
272      * @param path data file path
273      * @param position row position in the file
274      * @throws ConflictException if the row can't be deleted
275      */
276     private[core] def validateDelete(path: String, position: Long): Unit = synchronized {
277       checkOpen()
278       validate(pinned(), path, position, null)
279       if (deletes.get(path).exists(_.positions.isDeleted(position))) {
280         throw new ConflictException(s"Row has already been modified in this transaction: $path:$position", null)
281       }
282     }

296     /**
297      * Deletes a row that has already been checked with `validateDelete`, without checking it again
298      *
299      * @param path data file path
300      * @param position row position in the file
301      */
302     private[core] def deleteValidated(path: String, position: Long): Unit = synchronized {
303       checkOpen()
304       val pending = deletes.getOrElseUpdate(path, {
305         val (specId, partition) = pinned().dataFile(path)
306         PendingDeletes(table.specs().get(specId), partition, Deletes.toPositionIndex(CloseableIterable.empty[java.lang.Long]()))
307       })
308       pending.positions.delete(position)
309     }

311     /**
312      * Validates that a row is live in the given table state
313      */
314     private def validate(view: View, path: String, position: Long, cause: Throwable): Unit = {
315       if (view.dataFile(path) == null) {
316         throw new ConflictException(s"Data file has been removed by another operation: $path", cause)
317       }
318       val previous = view.previousDeletes(path)
319       if (previous != null && previous.isDeleted(position)) {
320         throw new ConflictException(s"Row has been modified by another operation: $path:$position", cause)
321       }
322     }

324     /**
325      * Re-evaluates the pending deletes against the current table state, after a commit failed validation
326      *
327      * @param cause validation failure
328      */
329     private def rebase(cause: Throwable): Unit = {
330       val previous = pinned().snapshot.map(_.snapshotId())
331       val current = pin()
332       if (current.snapshot.map(_.snapshotId()) == previous) {
333         // the table hasn't changed, so the failure wasn't caused by a concurrent operation and will just happen again
334         throw cause
335       }
336       logger.debug(s"Rebasing transaction from snapshot ${previous.orNull} to ${current.snapshot.map(_.snapshotId()).orNull}")
337       deletes.foreach { case (path, pending) =>
338         pending.positions.forEach((position: Long) => validate(current, path, position, cause))
339       }
340       view = current
341     }

343     /**
344      * Writes out the pending deletes as deletion vectors, merged with any existing deletes for the same files
345      *
346      * @param paths any files created will be added to this buffer
347      * @return
348      */
349     private def writeDeletes(paths: ArrayBuffer[String]): DeleteWriteResult = {
350       val io = new FileIO() {
351         override def newInputFile(path: String): InputFile = table.io().newInputFile(path)
352         override def newOutputFile(path: String): OutputFile = { paths += path; table.io().newOutputFile(path) }
353         override def deleteFile(path: String): Unit = table.io().deleteFile(path)
354       }
355       val files = OutputFileFactory.builderFor(table, 0, 0L).format(FileFormat.PUFFIN).ioSupplier(() => io).build()
356       val writer = new BaseDVFileWriter(files, pinned())
357       deletes.foreach { case (path, pending) => writer.delete(path, pending.positions, pending.spec, pending.partition) }
358       // note: nothing is written until the writer is closed
359       writer.close()
360       writer.result()
361     }

402     /**
403      * View of the table at a given snapshot. Loads existing deletes for a given data file path, so they can be merged
404      * into a newly written DV - returns null if there are no existing deletes for the given path.
405      *
406      * @param snapshot snapshot, if the table has one
407      */
408     private class View(val snapshot: Option[Snapshot]) extends java.util.function.Function[String, PositionDeleteIndex] {
409 
410       // map of data-file-path -> spec id and partition. if partitioning is changed, old files might have different partition specs
411       private lazy val dataFilesByPath: CharSequenceMap[(Int, StructLike)] = {
412         val map = CharSequenceMap.create[(Int, StructLike)]()
413         snapshot.foreach { snapshot =>
414           val io = table.io()
415           val specs = table.specs()
416           snapshot.dataManifests(io).asScala.foreach { manifest =>
417             WithClose(ManifestFiles.read(manifest, io, specs)) { reader =>
418               WithClose(reader.iterator()) { iterator =>
419                 // manifest readers reuse record instances
420                 iterator.asScala.foreach(file => map.put(file.location(), (file.specId(), file.copy().partition())))
421               }
422             }
423           }
424         }
425         map
426       }
427 
428       // map of data-file-path -> set of delete files that reference it
429       private lazy val deletesByPath: CharSequenceMap[DeleteFileSet] = {
430         val map = CharSequenceMap.create[DeleteFileSet]()
431         snapshot.foreach { snapshot =>
432           val io = table.io()
433           val specs = table.specs()
434           snapshot.deleteManifests(io).asScala.foreach { manifest =>
435             WithClose(ManifestFiles.readDeleteManifest(manifest, io, specs)) { reader =>
436               reader.iterator().asScala.foreach { deleteFile =>
437                 val ref = deleteFile.referencedDataFile()
438                 if (ref != null) {
439                   // copy() to detach from the manifest reader, which reuses containers
440                   map.computeIfAbsent(ref, () => DeleteFileSet.create()).add(deleteFile.copy())
441                 }
442               }
443             }
444           }
445         }
446         map
447       }
448 
449       private lazy val loader: DeleteLoader = new BaseDeleteLoader(deleteFile => table.io().newInputFile(deleteFile))
450 
451       // the deletes are checked for each deleted row, and again when writing the new DVs, so hold on to them
452       private val loaded = scala.collection.mutable.Map.empty[String, PositionDeleteIndex]
453 
454       /**
455        * Gets the partition spec id and partition for a data file
456        *
457        * @param path data file path
458        * @return null if the data file is not part of the table
459        */
460       def dataFile(path: String): (Int, StructLike) = dataFilesByPath.get(path)
461 
462       /**
463        * Gets existing deletes for a data file
464        *
465        * @param path data file path
466        * @return null if there are no deletes for the file
467        */
468       def previousDeletes(path: String): PositionDeleteIndex = {
469         loaded.getOrElseUpdate(path, {
470           val set = deletesByPath.get(path)
471           if (set == null) { null } else { loader.loadPositionDeletes(set, path) }
472         })
473       }
474 
475       override def apply(path: String): PositionDeleteIndex = previousDeletes(path)
476     }
