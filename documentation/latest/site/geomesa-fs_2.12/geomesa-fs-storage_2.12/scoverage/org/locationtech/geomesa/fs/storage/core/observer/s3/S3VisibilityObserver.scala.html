24 /**
25  * Creates a tag containing the base64 encoded summary visibility for the observed file
26  *
27  * @param path file path
28  * @param fs s3 client
29  * @param tag tag name to use
30  */
31 class S3VisibilityObserver(path: String, fs: S3FileIO, tag: String) extends FileSystemObserver {
32 
33   private val visibilities = scala.collection.mutable.Set.empty[String]
34 
35   private val key = S3ObjectStore.parseS3Path(new URI(path))
36 
37   override def apply(feature: SimpleFeature): Unit = {
38     val vis = SecurityUtils.getVisibility(feature)
39     if (vis != null) {
40       visibilities.add(vis)
41     }
42   }
43 
44   override def flush(): Unit = {}
45 
46   override def close(): Unit = {
47     try { makeTagRequest(key.bucket, key.key) } catch {
48       case e: Exception => throw new IOException("Error tagging object", e)
49     }
50   }
51 
52   private def makeTagRequest(bucket: String, key: String): Unit = {
53     if (visibilities.nonEmpty) {
54       val vis = VisibilityUtils.normalize(visibilities.mkString("(", ")&(", ")"))
55       val visibility = Base64.getEncoder.encodeToString(vis.getBytes(StandardCharsets.UTF_8))
56       val request =
57         PutObjectTaggingRequest.builder()
58           .bucket(bucket)
59           .key(key)
60           .tagging(t => t.tagSet(Tag.builder().key(tag).value(visibility).build()))
61           .build()
62       fs.asyncClient().putObjectTagging(request)
63     }
64   }
65 }
