22   /**
23    * Parse a json path
24    *
25    * @param pathString path
26    * @param sft feature type
27    * @return descriptor indicated by the head of the path, and then the rest of the path pointing into the attribute
28    */
29   def parseJsonPath(pathString: String, sft: SimpleFeatureType): (AttributeDescriptor, JsonPath) = {
30     val path = try { JsonPathParser.parse(pathString) } catch {
31       case NonFatal(e) => throw new IllegalArgumentException(s"Could not evaluate as a JSON path: $pathString", e)
32     }
33     if (path.isEmpty) {
34       throw new IllegalArgumentException(s"Invalid JSON path - empty: $pathString")
35     }
36     val descriptor = path.head match {
37       case PathAttribute(name, _) =>
38         val descriptor = sft.getDescriptor(name)
39         if (descriptor == null) {
40           throw new IllegalArgumentException(s"Invalid JSON path - does not point at an attribute: $pathString")
41         } else if (!classOf[String].isAssignableFrom(descriptor.getType.getBinding)) {
42           throw new IllegalArgumentException(
43             s"Invalid JSON path - points at an invalid attribute of type ${descriptor.getType.getBinding.getSimpleName}: $pathString")
44         } else if (!descriptor.isJson()) {
45           throw new IllegalArgumentException(s"Invalid JSON path - points at a non-JSON attribute: $pathString")
46         }
47         descriptor
48 
49       case _ =>
50         throw new IllegalArgumentException(s"Invalid JSON path - first element must point at an attribute: $pathString")
51     }
52 
53     (descriptor, path.tail)
54   }
