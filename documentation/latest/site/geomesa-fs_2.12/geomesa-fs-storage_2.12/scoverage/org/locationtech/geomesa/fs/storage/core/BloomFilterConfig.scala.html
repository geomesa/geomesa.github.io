18 /**
19  * Parquet bloom filter configuration for a single attribute
20  *
21  * @param attribute attribute name
22  * @param ndv expected number of distinct values per row group, used to size the filter
23  * @param fpp false positive probability
24  */
25 case class BloomFilterConfig(attribute: String, ndv: Option[Long] = None, fpp: Option[Double] = None)

31   // types that are written as a single primitive parquet column, which bloom filters can be applied to
32   private val SupportedTypes = Set(
33     ObjectType.STRING, ObjectType.INT, ObjectType.LONG, ObjectType.FLOAT, ObjectType.DOUBLE, ObjectType.DATE,
34     ObjectType.UUID, ObjectType.BYTES,
35   )

37   /**
38    * Parse and validate bloom filter configurations. Attributes are separated with a comma (`,`), while options
39    * are separated with a colon (`:`), e.g. `name:ndv=10000:fpp=0.01,age`
40    *
41    * @param sft simple feature type
42    * @param spec bloom filter spec
43    * @return
44    */
45   def parse(sft: SimpleFeatureType, spec: String): Seq[BloomFilterConfig] = {
46     spec.split(",").toSeq.map(_.trim).filter(_.nonEmpty).map { entry =>
47       val parts = entry.split(":").map(_.trim)
48       val attribute = parts.head
49       val descriptor = Option(sft.getDescriptor(attribute)).getOrElse {
50         throw new IllegalArgumentException(s"Bloom filter attribute '$attribute' does not exist in schema ${sft.getTypeName}")
51       }
52       val bindings = ObjectType.selectType(descriptor)
53       if (!SupportedTypes.contains(bindings.head) || bindings.last == ObjectType.JSON) {
54         throw new IllegalArgumentException(
55           s"Bloom filters are not supported for attribute '$attribute' of type ${descriptor.getType.getBinding.getSimpleName}")
56       }
57       parts.tail.foldLeft(BloomFilterConfig(attribute)) { case (config, option) =>
58         option.split("=", 2).map(_.trim) match {
59           case Array("ndv", v) =>
60             val ndv = Try(v.toLong).toOption.filter(_ > 0).getOrElse {
61               throw new IllegalArgumentException(s"Invalid bloom filter ndv for attribute '$attribute': $v")
62             }
63             config.copy(ndv = Some(ndv))
64           case Array("fpp", v) =>
65             val fpp = Try(v.toDouble).toOption.filter(d => d > 0 && d < 1).getOrElse {
66               throw new IllegalArgumentException(s"Invalid bloom filter fpp for attribute '$attribute': $v")
67             }
68             config.copy(fpp = Some(fpp))
69           case _ =>
70             throw new IllegalArgumentException(s"Invalid bloom filter option for attribute '$attribute': $option")
71         }
72       }
73     }
74   }
75 
76   /**
77    * Iceberg table properties for the bloom filter configurations. These are honored by any iceberg writer,
78    * including GeoMesa (see `ParquetFileSystemWriter.icebergBloomFilterConf`)
79    *
80    * @param configs bloom filter configurations
81    * @return
82    */
83   def tableProperties(configs: Seq[BloomFilterConfig]): Map[String, String] = {
84     configs.flatMap { config =>
85       val column = ColumnName(config.attribute).column
86       Seq(s"$PARQUET_BLOOM_FILTER_COLUMN_ENABLED_PREFIX$column" -> "true") ++
87         config.ndv.map(n => s"$PARQUET_BLOOM_FILTER_COLUMN_NDV_PREFIX$column" -> n.toString) ++
88         config.fpp.map(f => s"$PARQUET_BLOOM_FILTER_COLUMN_FPP_PREFIX$column" -> f.toString)
89     }.toMap
90   }
