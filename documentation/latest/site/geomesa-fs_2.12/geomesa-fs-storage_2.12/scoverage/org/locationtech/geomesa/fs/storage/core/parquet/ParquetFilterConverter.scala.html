34   private def reduce(
35       sft: SimpleFeatureType
36     )(result: (Option[FilterPredicate], Option[Filter]),
37       name: String): (Option[FilterPredicate], Option[Filter]) = {
38     val (parquet, geotools) = result
39     val filter = geotools.orNull
40     if (filter == null) {
41       return result // no more filter to evaluate
42     }
43 
44     val bindings = ObjectType.selectType(sft.getDescriptor(name))
45     val col = ColumnName(name)
46 
47     val (predicate, remaining): (Option[FilterPredicate], Option[Filter]) = bindings.head match {
48       // note: non-points use repeated values, which aren't supported in parquet predicates
49       case ObjectType.GEOMETRY => spatial(sft, col, filter)
50       case ObjectType.DATE     => attribute(sft, col.attribute, filter, FilterApi.longColumn(col.column), toMicros)
51       case ObjectType.STRING   => attribute(sft, col.attribute, filter, FilterApi.binaryColumn(col.column), Binary.fromString)
52       case ObjectType.INT      => attribute(sft, col.attribute, filter, FilterApi.intColumn(col.column), identity[java.lang.Integer])
53       case ObjectType.LONG     => attribute(sft, col.attribute, filter, FilterApi.longColumn(col.column), identity[java.lang.Long])
54       case ObjectType.FLOAT    => attribute(sft, col.attribute, filter, FilterApi.floatColumn(col.column), identity[java.lang.Float])
55       case ObjectType.DOUBLE   => attribute(sft, col.attribute, filter, FilterApi.doubleColumn(col.column), identity[java.lang.Double])
56       case ObjectType.BOOLEAN  => boolean(sft, col.attribute, filter, FilterApi.booleanColumn(col.column))
57       case _ => (None, Some(filter))
58     }
59 
60     ((predicate.toSeq ++ parquet).reduceLeftOption(FilterApi.and), remaining)
61   }

63   private def spatial(sft: SimpleFeatureType, name: ColumnName, filter: Filter): (Option[FilterPredicate], Option[Filter]) = {
64     val (spatial, _) = FilterExtractingVisitor(filter, name.attribute, sft, SpatialFilterStrategy.spatialCheck)
65     val xyBounds = spatial.map(FilterHelper.extractGeometries(_, name.attribute)).flatMap { extracted =>
66       Some(extracted).filter(e => e.nonEmpty && !e.disjoint).map { e =>
67         e.values.map(GeometryUtils.bounds).reduce { (a, b) =>
68           (math.min(a._1, b._1), math.min(a._2, b._2), math.max(a._3, b._3), math.max(a._4, b._4))
69         }
70       }
71     }
72 
73     val isPoint = sft.getDescriptor(name.attribute).getType.getBinding == classOf[Point]
74     val predicate = xyBounds.map { case (xmin, ymin, xmax, ymax) =>
75       BoundingBoxField.filterParquet(name.column, xmin, ymin, xmax, ymax, isPoint)
76     }
77     // since we don't know what the actual file encoding is up front, we always have to evaluate the full predicate post-read
78     (predicate, Some(filter))
79   }

81   private def attribute[T : ClassTag, U <: Comparable[U]](
82       sft: SimpleFeatureType,
83       name: String,
84       filter: Filter,
85       col: Column[U] with SupportsLtGt,
86       conversion: T => U): (Option[FilterPredicate], Option[Filter]) = {
87     val (attribute, nonAttribute) = FilterExtractingVisitor(filter, name, sft)
88     val binding = implicitly[ClassTag[T]].runtimeClass.asInstanceOf[Class[T]]
89     val bounds = attribute.map(FilterHelper.extractAttributeBounds(_, name, binding))
90     val predicate = bounds.flatMap { extracted =>
91       Some(extracted).filter(e => e.nonEmpty && !e.disjoint && e.values.forall(_.isBounded)).map { e =>
92         val filters = e.values.map { bounds =>
93           if (bounds.isEquals) {
94             FilterApi.eq(col, conversion(bounds.lower.value.get))
95           } else {
96             val lower: Option[FilterPredicate] = bounds.lower.value.map { value =>
97               val converted = conversion(value)
98               if (bounds.lower.inclusive) { FilterApi.gtEq(col, converted) } else { FilterApi.gt(col, converted) }
99             }
100             val upper: Option[FilterPredicate] = bounds.upper.value.map { value =>
101               val converted = conversion(value)
102               if (bounds.upper.inclusive) { FilterApi.ltEq(col, converted) } else { FilterApi.lt(col, converted) }
103             }
104             (lower, upper) match {
105               case (Some(lo), Some(hi)) => FilterApi.and(lo, hi)
106               case (Some(f), None) => f
107               case (None, Some(f)) => f
108               case (None, None) => throw new IllegalStateException() // shouldn't happen due to checks above
109             }
110           }
111         }
112         merge(filters)
113       }
114     }
115     val remaining = if (bounds.exists(_.precise)) { nonAttribute } else { Some(filter) }
116     (predicate, remaining)
117   }

121       name: String,

133   }

137   /**
138     * Merge OR'd filters
139     *
140     * Detect and re-write "not equals" filters to handle null values
141     *
142     * FilterHelper methods make "foo != x" comes out as "foo < x OR foo > x"
143    *
144     * that won't return null values - not normally a concern since we don't index them in key-value dbs,
145     * but we want to handle that case here
146     *
147     * @param filters filters
148     * @return combined filter
149     */
150   private def merge[T <: Comparable[T]](filters: Seq[FilterPredicate]): FilterPredicate = {
151     lazy val values = filters.collect {
152       case f: Lt[T] => (f.getColumn.asInstanceOf[Column[T] with SupportsEqNotEq], f.getValue)
153       case f: Gt[T] => (f.getColumn.asInstanceOf[Column[T] with SupportsEqNotEq], f.getValue)
154     }.distinct
155 
156     if (filters.lengthCompare(2) == 0 &&
157         filters.exists(_.isInstanceOf[Lt[_]]) &&
158         filters.exists(_.isInstanceOf[Gt[_]]) &&
159         values.lengthCompare(1) == 0) {
160       FilterApi.notEq(values.head._1, values.head._2)
161     } else {
162       filters.reduce(FilterApi.or)
163     }
164   }
