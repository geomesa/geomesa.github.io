33   private def reduce(
34       sft: SimpleFeatureType
35     )(result: (Option[FilterPredicate], Option[Filter]),
36       name: String): (Option[FilterPredicate], Option[Filter]) = {
37     val (parquet, geotools) = result
38     val filter = geotools.orNull
39     if (filter == null) {
40       return result // no more filter to evaluate
41     }
42 
43     val bindings = ObjectType.selectType(sft.getDescriptor(name))
44     val col = ColumnName(name)
45 
46     val (predicate, remaining): (Option[FilterPredicate], Option[Filter]) = bindings.head match {
47       // note: non-points use repeated values, which aren't supported in parquet predicates
48       case ObjectType.GEOMETRY => spatial(sft, col, filter)
49       case ObjectType.DATE     => attribute(sft, col.attribute, filter, FilterApi.longColumn(col.column), toMicros)
50       case ObjectType.STRING   => attribute(sft, col.attribute, filter, FilterApi.binaryColumn(col.column), Binary.fromString)
51       case ObjectType.INT      => attribute(sft, col.attribute, filter, FilterApi.intColumn(col.column), identity[java.lang.Integer])
52       case ObjectType.LONG     => attribute(sft, col.attribute, filter, FilterApi.longColumn(col.column), identity[java.lang.Long])
53       case ObjectType.FLOAT    => attribute(sft, col.attribute, filter, FilterApi.floatColumn(col.column), identity[java.lang.Float])
54       case ObjectType.DOUBLE   => attribute(sft, col.attribute, filter, FilterApi.doubleColumn(col.column), identity[java.lang.Double])
55       case ObjectType.BOOLEAN  => boolean(sft, col.attribute, filter, FilterApi.booleanColumn(col.column))
56       case _ => (None, Some(filter))
57     }
58 
59     ((predicate.toSeq ++ parquet).reduceLeftOption(FilterApi.and), remaining)
60   }

62   private def spatial(sft: SimpleFeatureType, name: ColumnName, filter: Filter): (Option[FilterPredicate], Option[Filter]) = {
63     val (spatial, _) = FilterExtractingVisitor(filter, name.attribute, sft, SpatialFilterStrategy.spatialCheck)
64     val xyBounds = spatial.map(FilterHelper.extractGeometries(_, name.attribute)).flatMap { extracted =>
65       Some(extracted).filter(e => e.nonEmpty && !e.disjoint).map { e =>
66         e.values.map(GeometryUtils.bounds).reduce { (a, b) =>
67           (math.min(a._1, b._1), math.min(a._2, b._2), math.max(a._3, b._3), math.max(a._4, b._4))
68         }
69       }
70     }
71 
72     val predicate = xyBounds.map { case (xmin, ymin, xmax, ymax) =>
73       BoundingBoxField.filterParquet(name.column, xmin, ymin, xmax, ymax)
74     }
75     // since we don't know what the actual file encoding is up front, we always have to evaluate the full predicate post-read
76     (predicate, Some(filter))
77   }

79   private def attribute[T : ClassTag, U <: Comparable[U]](
80       sft: SimpleFeatureType,
81       name: String,
82       filter: Filter,
83       col: Column[U] with SupportsLtGt,
84       conversion: T => U): (Option[FilterPredicate], Option[Filter]) = {
85     val (attribute, nonAttribute) = FilterExtractingVisitor(filter, name, sft)
86     val binding = implicitly[ClassTag[T]].runtimeClass.asInstanceOf[Class[T]]
87     val bounds = attribute.map(FilterHelper.extractAttributeBounds(_, name, binding))
88     val predicate = bounds.flatMap { extracted =>
89       Some(extracted).filter(e => e.nonEmpty && !e.disjoint && e.values.forall(_.isBounded)).map { e =>
90         val filters = e.values.map { bounds =>
91           if (bounds.isEquals) {
92             FilterApi.eq(col, conversion(bounds.lower.value.get))
93           } else {
94             val lower: Option[FilterPredicate] = bounds.lower.value.map { value =>
95               val converted = conversion(value)
96               if (bounds.lower.inclusive) { FilterApi.gtEq(col, converted) } else { FilterApi.gt(col, converted) }
97             }
98             val upper: Option[FilterPredicate] = bounds.upper.value.map { value =>
99               val converted = conversion(value)
100               if (bounds.upper.inclusive) { FilterApi.ltEq(col, converted) } else { FilterApi.lt(col, converted) }
101             }
102             (lower, upper) match {
103               case (Some(lo), Some(hi)) => FilterApi.and(lo, hi)
104               case (Some(f), None) => f
105               case (None, Some(f)) => f
106               case (None, None) => throw new IllegalStateException() // shouldn't happen due to checks above
107             }
108           }
109         }
110         merge(filters)
111       }
112     }
113     val remaining = if (bounds.exists(_.precise)) { nonAttribute } else { Some(filter) }
114     (predicate, remaining)
115   }

119       name: String,
120       filter: Filter,
121       col: BooleanColumn): (Option[FilterPredicate], Option[Filter]) = {

128     }
129     val remaining = if (bounds.exists(_.precise)) { nonAttribute } else { Some(filter) }
130     (predicate, remaining)

135   /**
136     * Merge OR'd filters
137     *
138     * Detect and re-write "not equals" filters to handle null values
139     *
140     * FilterHelper methods make "foo != x" comes out as "foo < x OR foo > x"
141    *
142     * that won't return null values - not normally a concern since we don't index them in key-value dbs,
143     * but we want to handle that case here
144     *
145     * @param filters filters
146     * @return combined filter
147     */
148   private def merge[T <: Comparable[T]](filters: Seq[FilterPredicate]): FilterPredicate = {
149     lazy val values = filters.collect {
150       case f: Lt[T] => (f.getColumn.asInstanceOf[Column[T] with SupportsEqNotEq], f.getValue)
151       case f: Gt[T] => (f.getColumn.asInstanceOf[Column[T] with SupportsEqNotEq], f.getValue)
152     }.distinct
153 
154     if (filters.lengthCompare(2) == 0 &&
155         filters.exists(_.isInstanceOf[Lt[_]]) &&
156         filters.exists(_.isInstanceOf[Gt[_]]) &&
157         values.lengthCompare(1) == 0) {
158       FilterApi.notEq(values.head._1, values.head._2)
159     } else {
160       filters.reduce(FilterApi.or)
161     }
162   }
