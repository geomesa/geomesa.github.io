23 /**
24  * Attribute partitioning scheme, supporting optional bucketing
25  *
26  * @param attribute attribute being partitioned
27  * @param index index in the feature type of the attribute
28  * @param nullValue null value placeholder (type dependent)
29  * @param bucketing optional bucketing
30  */
31 abstract class AttributeScheme[T: ClassTag](val attribute: String, index: Int, nullValue: T, val bucketing: Option[Bucketing[T]])
32     extends PartitionScheme {
33 
34   override val name: String = {
35     val opts = new StringBuilder(s"${AttributeScheme.Name}:attribute=$attribute")
36     bucketing.foreach(b => opts.append(':').append(b.encoded))
37     opts.toString()
38   }
39 
40   override val column: String = ColumnName.encode(attribute)
41 
42   private val default = PartitionKey(name, toPartition(nullValue))
43 
44   override def getPartition(feature: SimpleFeature): PartitionKey = {
45     val value = feature.getAttribute(index)
46     if (value == null) {
47       return default
48     }
49     val partition = toPartition(value.asInstanceOf[T])
50     PartitionKey(name, partition)
51   }
52 
53   override def spec(b: PartitionSpec.Builder): PartitionSpec.Builder = bucketing match {
54     case None => b.identity(column)
55     case Some(bucket) => b.truncate(column, bucket.width)
56   }
57 
58   override def getPartition(partition: StructLike, i: Int): PartitionKey =
59     PartitionKey(name, partition.get(i, implicitly[ClassTag[T]].runtimeClass).toString)
60 
61   private def toPartition(value: T): String = {
62     bucketing match {
63       case None => value.toString
64       case Some(b) => b(value).toString
65     }
66   }
67 }

75   override def load(sft: SimpleFeatureType, scheme: String): Option[PartitionScheme] = {
76     val opts = SchemeOpts(scheme)
77     if (opts.name != Name) { None } else {
78       val attribute = opts.getSingle("attribute").orNull
79       require(attribute != null, s"Attribute scheme requires an attribute to be specified with 'attribute=<attribute>'")
80       val index = attributeIndex(sft, attribute)
81       val binding = sft.getDescriptor(index).getType.getBinding
82       require(AttributeIndexKey.encodable(binding), s"Invalid type binding '${binding.getName}' of attribute '$attribute'")
83 
84       val width = opts.getSingle("width").map(w => WidthBucketing(w.toInt))
85       val divisor = opts.getSingle("divisor").map(_.toInt)
86       val scale = opts.getSingle("scale").map(_.toInt)
87 
88       val isString = classOf[String].isAssignableFrom(binding)
89       val isWholeNumber = binding == classOf[Integer] || binding == classOf[java.lang.Long]
90       val isDecimalNumber = binding == classOf[java.lang.Float] || binding == classOf[java.lang.Double]
91 
92       if (width.isDefined && !isString) {
93         throw new IllegalArgumentException(
94           s"'width' option is only supported for String-type attributes, not ${binding.getSimpleName}")
95       } else if (divisor.isDefined && !isWholeNumber) {
96         throw new IllegalArgumentException(
97           s"'divisor' option is only supported for Integer and Long-type attributes, not ${binding.getSimpleName}")
98       } else if (scale.isDefined && !isDecimalNumber) {
99         throw new IllegalArgumentException(
100           s"'scale' option is only supported for Float and Double-type attributes, not ${binding.getSimpleName}")
101       }
102 
103       if (opts.getMulti("allow").nonEmpty) {
104         throw new IllegalArgumentException("`allow` option is no longer supported for attribute schemes")
105       } else if (opts.getSingle("default").isDefined) {
106         throw new IllegalArgumentException("`default` option is no longer supported for attribute schemes")
107       }
108 
109       if (isString) {
110         width match {
111           case None => Some(new StringScheme(attribute, index))
112           case Some(w) => Some(new TruncatedStringScheme(attribute, index, w))
113         }
114       } else if (binding == classOf[Integer]) {
115         val bucketing = divisor.map(IntegralBucketing.apply[Int])
116         Some(new IntScheme(attribute, index, bucketing))
117       } else if (binding == classOf[java.lang.Long]) {
118         val bucketing = divisor.map(d => IntegralBucketing(d.toLong))
119         Some(new LongScheme(attribute, index, bucketing))
120       } else if (binding == classOf[java.lang.Float]) {
121         val bucketing = scale.map(FractionalBucketing.apply[Float])
122         Some(new FloatScheme(attribute, index, bucketing))
123       } else if (binding == classOf[java.lang.Double]) {
124         val bucketing = scale.map(FractionalBucketing.apply[Double])
125         Some(new DoubleScheme(attribute, index, bucketing))
126       } else {
127         throw new IllegalArgumentException(
128           s"Attribute scheme is not supported for type ${binding.getSimpleName} - " +
129             s"supported types are String Integer, Long, Float, and Double")
130       }
131     }

134   /**
135    * Bucketing abstraction
136    *
137    * @tparam T attribute type
138    */
139   sealed trait Bucketing[T] extends (T => T) {
140 
141     /**
142      * Encoded option, used to identify this bucketing
143      *
144      * @return
145      */
146     def encoded: String
147 
148     /**
149      * Truncation width
150      *
151      * @return
152      */
153     def width: Int
154   }
155 
156   private case class WidthBucketing(width: Int) extends Bucketing[String] {
157     override def apply(value: String): String = value.slice(0, width)
158     override def encoded: String = s"width=$width"
159   }
160 
161   private case class IntegralBucketing[T: Integral](divisor: T) extends Bucketing[T] {
162     import Integral.Implicits.infixIntegralOps
163     override def apply(value: T): T = value - (((value % divisor) + divisor) % divisor)
164     override def encoded: String = s"divisor=$divisor"
165     // TODO longs not supported by the java api
166     override def width: Int = implicitly[Integral[T]].toInt(divisor)
167   }
168 
169   // scale here refers to the number of digits to the right of the decimal place that are kept
170   private case class FractionalBucketing[T: Fractional](scale: Int) extends Bucketing[T] {
171     import Fractional.Implicits.infixFractionalOps
172     private val fractional = implicitly[Fractional[T]]
173     val scaleT: T = fractional.fromInt(math.pow(10, scale).toInt)
174     override def apply(value: T): T = fractional.fromInt(Math.floor((value * scaleT).toDouble).toInt) / scaleT
175     override def encoded: String = s"scale=$scale"
176     // TODO only decimal types support truncate but not double/float
177     override def width: Int = scale
178   }

233 
234     override def getCoveringFilter(partition: PartitionKey): Filter = {
235       val value = decoder(partition.value)
236       val attr = ff.property(attribute)
237       divisor match {
238         case None => ff.equals(attr, ff.literal(value))
239         case Some(d) =>
240           val lower = ff.literal(value)
241           val upper = ff.literal(value + d.divisor)
242           ff.and(ff.greaterOrEqual(attr, lower), ff.less(attr, upper))
243       }
244     }
245 
246     override def getCoveringExpression(partition: PartitionKey): Expression = divisor match {
247       case None => Expressions.equal[T](column, decoder(partition.value))
