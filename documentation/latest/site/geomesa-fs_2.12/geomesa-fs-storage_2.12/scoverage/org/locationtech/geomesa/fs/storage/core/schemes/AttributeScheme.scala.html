25 /**
26  * Attribute partitioning scheme, supporting optional bucketing
27  *
28  * @param attribute attribute being partitioned
29  * @param index index in the feature type of the attribute
30  * @param nullValue null value placeholder (type dependent)
31  * @param bucketing optional bucketing
32  */
33 abstract class AttributeScheme[T: ClassTag](val attribute: String, index: Int, nullValue: T, val bucketing: Option[Bucketing[T]])
34     extends PartitionScheme {
35 
36   override val name: String = {
37     val opts = new StringBuilder(s"${AttributeScheme.Name}:attribute=$attribute")
38     bucketing.foreach(b => opts.append(':').append(b.encoded))
39     opts.toString()
40   }
41 
42   override val column: String = ColumnName.encode(attribute)
43 
44   private val default = PartitionKey(name, toPartition(nullValue))
45 
46   override def getPartition(feature: SimpleFeature): PartitionKey = {
47     val value = feature.getAttribute(index)
48     if (value == null) {
49       return default
50     }
51     val partition = toPartition(value.asInstanceOf[T])
52     PartitionKey(name, partition)
53   }
54 
55   override def spec(b: PartitionSpec.Builder): PartitionSpec.Builder = bucketing match {
56     case None => b.identity(column)
57     case Some(bucket) => b.truncate(column, bucket.width)
58   }
59 
60   override def getPartition(partition: StructLike, i: Int): PartitionKey =
61     PartitionKey(name, partition.get(i, implicitly[ClassTag[T]].runtimeClass).toString)
62 
63   private def toPartition(value: T): String = {
64     bucketing match {
65       case None => value.toString
66       case Some(b) => b(value).toString
67     }
68   }
69 }

79   override def load(sft: SimpleFeatureType, scheme: String): Option[PartitionScheme] = {
80     val opts = SchemeOpts(scheme)
81     if (opts.name != Name) { None } else {
82       val attribute = opts.getSingle("attribute").orNull
83       require(attribute != null, s"Attribute scheme requires an attribute to be specified with 'attribute=<attribute>'")
84       val index = attributeIndex(sft, attribute)
85       val binding = sft.getDescriptor(index).getType.getBinding
86 
87       val width = opts.getSingle("width").map(w => WidthBucketing(w.toInt))
88       val divisor = opts.getSingle("divisor").map(_.toInt)
89       val scale = opts.getSingle("scale").map(_.toInt)
90 
91       if (opts.getMulti("allow").nonEmpty) {
92         throw new IllegalArgumentException("`allow` option is no longer supported for attribute schemes")
93       } else if (opts.getSingle("default").isDefined) {
94         throw new IllegalArgumentException("`default` option is no longer supported for attribute schemes")
95       }
96 
97       val isString = classOf[String].isAssignableFrom(binding)
98       val isWholeNumber = binding == classOf[Integer] || binding == classOf[java.lang.Long]
99       val isDecimalNumber = binding == classOf[java.lang.Float] || binding == classOf[java.lang.Double]
100 
101       val scheme = if (isString) {
102         width match {
103           case None => new StringScheme(attribute, index)
104           case Some(w) => new TruncatedStringScheme(attribute, index, w)
105         }
106       } else if (binding == classOf[Integer]) {
107         new IntScheme(attribute, index, divisor.map(IntegralBucketing.apply[Int]))
108       } else if (binding == classOf[java.lang.Long]) {
109         new LongScheme(attribute, index, divisor.map(d => IntegralBucketing(d.toLong)))
110       } else if (binding == classOf[java.lang.Float]) {
111         new FloatScheme(attribute, index, scale.map(FractionalBucketing.apply[Float]))
112       } else if (binding == classOf[java.lang.Double]) {
113         new DoubleScheme(attribute, index, scale.map(FractionalBucketing.apply[Double]))
114       } else {
115         throw new IllegalArgumentException(
116           s"Attribute scheme is not supported for type ${binding.getSimpleName} - " +
117             s"supported types are String Integer, Long, Float, and Double")
118       }
119 
120       if (width.isDefined && !isString) {
121         throw new IllegalArgumentException(
122           s"'width' option is only supported for String-type attributes, not ${binding.getSimpleName}")
123       } else if (divisor.isDefined && !isWholeNumber) {
124         throw new IllegalArgumentException(
125           s"'divisor' option is only supported for Integer and Long-type attributes, not ${binding.getSimpleName}")
126       } else if (scale.isDefined && !isDecimalNumber) {
127         throw new IllegalArgumentException(
128           s"'scale' option is only supported for Float and Double-type attributes, not ${binding.getSimpleName}")
129       }
130 
131       Some(scheme)
132     }
133   }

172   sealed trait Bucketing[T] extends (T => T) {

184      * @return

189   private case class WidthBucketing(width: Int) extends Bucketing[String] {
190     override def apply(value: String): String = value.slice(0, width)
191     override def encoded: String = s"width=$width"
192   }
193 
194   private case class IntegralBucketing[T: Integral](divisor: T) extends Bucketing[T] {
195     import Integral.Implicits.infixIntegralOps
196     override def apply(value: T): T = value - (((value % divisor) + divisor) % divisor)
197     override def encoded: String = s"divisor=$divisor"
198     // TODO longs not supported by the java api
199     override def width: Int = implicitly[Integral[T]].toInt(divisor)
200   }
201 
202   // scale here refers to the number of digits to the right of the decimal place that are kept
203   private case class FractionalBucketing[T: Fractional](scale: Int) extends Bucketing[T] {
204     import Fractional.Implicits.infixFractionalOps
205     private val fractional = implicitly[Fractional[T]]
206     val scaleT: T = fractional.fromInt(math.pow(10, scale).toInt)
207     override def apply(value: T): T = fractional.fromInt(Math.floor((value * scaleT).toDouble).toInt) / scaleT
208     override def encoded: String = s"scale=$scale"
209     // TODO only decimal types support truncate but not double/float
210     override def width: Int = scale
211   }

239       val escapes = Seq("""\""" -> """\\""", """%""" -> """\%""", """_""" -> """\_""")

241         escapes.foldLeft(partition.value) { case (s, (literal, replacement)) => s.replaceAllLiterally(literal, replacement) }

266 
267     override def getCoveringFilter(partition: PartitionKey): Filter = {
268       val value = decoder(partition.value)
269       val attr = ff.property(attribute)
270       divisor match {
271         case None => ff.equals(attr, ff.literal(value))
272         case Some(d) =>
273           val lower = ff.literal(value)
274           val upper = ff.literal(value + d.divisor)
275           ff.and(ff.greaterOrEqual(attr, lower), ff.less(attr, upper))
276       }
277     }

280       case None => Expressions.equal[T](column, decoder(partition.value))
