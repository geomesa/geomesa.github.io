33   def apply(sft: SimpleFeatureType, filter: Filter): ReadFilter = {
34     if (filter == Filter.INCLUDE) {
35       ReadFilter(Expressions.alwaysTrue(), None, Set.empty)
36     } else if (filter == Filter.EXCLUDE) {
37       ReadFilter(Expressions.alwaysFalse(), None, Set.empty)
38     } else {
39       val names = FilterHelper.propertyNames(filter).map(ColumnName.apply)
40       names.foldLeft(ReadFilter(Expressions.alwaysTrue(), Some(filter), names.map(_.column).toSet))(reduce(sft))
41     }
42   }

44   private def reduce(sft: SimpleFeatureType)(result: ReadFilter, name: ColumnName): ReadFilter = {
45     val filter = result.remainder.orNull
46     if (filter == null) {
47       return result // no more filter to evaluate
48     }
49     val bindings = ObjectType.selectType(sft.getDescriptor(name.attribute))
50     val predicate = bindings.head match {
51       // note: non-points use repeated values, which aren't supported in parquet predicates
52       case ObjectType.GEOMETRY => spatial(sft, name, filter)
53       case ObjectType.DATE     => attribute[Date](sft, name, filter, Some(dateToMicros))
54       case ObjectType.STRING   => attribute[String](sft, name, filter)
55       case ObjectType.INT      => attribute[Integer](sft, name, filter)
56       case ObjectType.LONG     => attribute[java.lang.Long](sft, name, filter)
57       case ObjectType.FLOAT    => attribute[java.lang.Float](sft, name, filter)
58       case ObjectType.DOUBLE   => attribute[java.lang.Double](sft, name, filter)
59       case ObjectType.BOOLEAN  => attribute[java.lang.Boolean](sft, name, filter)
60       case _ => ReadFilter(Expressions.alwaysTrue(), result.remainder, Set.empty)
61     }
62     ReadFilter(Expressions.and(predicate.expression, result.expression), predicate.remainder, predicate.columns ++ result.columns)
63   }

65   private def spatial(sft: SimpleFeatureType, name: ColumnName, filter: Filter): ReadFilter = {
66     val (spatial, nonSpatial) = FilterExtractingVisitor(filter, name.attribute, sft, SpatialFilterStrategy.spatialCheck)
67     val bounds = spatial.map(FilterHelper.extractGeometries(_, name.attribute))
68     val xyBounds = bounds.toSeq.flatMap { extracted =>
69       Seq(extracted).filter(e => e.nonEmpty && !e.disjoint).flatMap { e =>
70         e.values.map(GeometryUtils.bounds)
71       }
72     }
73 
74     // filter against the bbox field
75     val predicate = xyBounds.map { case (xmin, ymin, xmax, ymax) =>
76       BoundingBoxField.filterIceberg(name.column, xmin, ymin, xmax, ymax)
77     }
78 
79     val remaining = if (bounds.exists(_.precise)) { nonSpatial } else { Some(filter) }
80     val geomCol = if (bounds.exists(b => !b.precise)) { Some(name.column) } else { None }
81     val bboxCol = if (predicate.nonEmpty) { Some(BoundingBoxField.groupName(name.column)) } else { None }
82     ReadFilter(predicate.reduce(Expressions.or), remaining, (geomCol ++ bboxCol).toSet)
83   }

85   private def attribute[T : ClassTag](
86       sft: SimpleFeatureType,
87       name: ColumnName,
88       filter: Filter,
89       transform: Option[T => Any] = None): ReadFilter = {
90     val (attribute, nonAttribute) = FilterExtractingVisitor(filter, name.attribute, sft)
91     val binding = implicitly[ClassTag[T]].runtimeClass.asInstanceOf[Class[T]]
92     val bounds = attribute.map(FilterHelper.extractAttributeBounds(_, name.attribute, binding))
93     val predicate = bounds.flatMap { extracted =>
94       Some(extracted).filter(e => e.nonEmpty && !e.disjoint && e.values.forall(_.isBounded)).map { e =>
95         val values = transform match {
96           case None => e.values
97           case Some(t) =>
98             e.values.map { bounds =>
99               bounds.copy(bounds.lower.copy(bounds.lower.value.map(t.apply)), bounds.upper.copy(bounds.upper.value.map(t.apply)))
100             }
101         }
102         val filters = values.map { bounds =>
103           if (bounds.isEquals) {
104             Expressions.equal(name.column, bounds.lower.value.get)
105           } else {
106             val lower = bounds.lower.value.map { value =>
107               if (bounds.lower.inclusive) { Expressions.greaterThanOrEqual(name.column, value) } else { Expressions.greaterThan(name.column, value) }
108             }
109             val upper = bounds.upper.value.map { value =>
110               if (bounds.upper.inclusive) { Expressions.lessThanOrEqual(name.column, value) } else { Expressions.lessThan(name.column, value) }
111             }
112             (lower, upper) match {
113               case (Some(lo), Some(hi)) => Expressions.and(lo, hi)
114               case (Some(f), None) => f
115               case (None, Some(f)) => f
116               case (None, None) => throw new IllegalStateException() // shouldn't happen due to checks above
117             }
118           }
119         }
120         merge(filters)
121       }
122     }
123     val remaining = if (bounds.exists(_.precise)) { nonAttribute } else { Some(filter) }
124     val cols = if (predicate.isDefined || bounds.exists(b => !b.precise)) { Set(name.column) } else { Set.empty[String] }
125     ReadFilter(predicate.getOrElse(Expressions.alwaysTrue()), remaining, cols)
126   }

128   /**
129    * Merge OR'd filters
130    *
131    * Detect and re-write "not equals" filters to handle null values
132    *
133    * FilterHelper methods make "foo != x" comes out as "foo < x OR foo > x"
134    *
135    * that won't return null values - not normally a concern since we don't index them in key-value dbs,
136    * but we want to handle that case here
137    *
138    * @param filters filters
139    * @return combined filter
140    */
141   private def merge(filters: Seq[Expression]): Expression = {
142     lazy val values = filters.collect { case f: UnboundPredicate[_] => (f.ref().name(), f.literal().value()) }.distinct
143     if (filters.lengthCompare(2) == 0 &&
144       filters.exists(_.op() == Operation.LT) &&
145       filters.exists(_.op == Operation.GT) &&
146       values.lengthCompare(1) == 0) {
147       Expressions.notEqual(values.head._1, values.head._2)
148     } else {
149       filters.reduce(Expressions.or)
150     }
151   }
