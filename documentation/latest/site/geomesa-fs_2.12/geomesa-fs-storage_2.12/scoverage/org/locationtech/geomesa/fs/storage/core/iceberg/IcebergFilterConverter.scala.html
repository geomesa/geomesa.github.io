38   /**
39    * Returns an iceberg expression and a residual GeoTools filter that isn't captured by the expression (if any)
40    *
41    * @param schema table schema
42    * @param filter geotools filter
43    * @return
44    */
45   def apply(schema: SimpleFeatureIcebergSchema, schemes: Seq[PartitionScheme], filter: Filter): ReadFilter = {
46     if (filter == Filter.INCLUDE) {
47       ReadFilter.clientSide(None)
48     } else if (filter == Filter.EXCLUDE) {
49       ReadFilter.exclude()
50     } else {
51       val fid = if (FilterHelper.hasIdFilter(filter)) { Seq(SimpleFeatureSchema.FeatureIdField) } else { Seq.empty }
52       val names = (fid ++ FilterHelper.propertyNames(filter)).map(ColumnName.apply)
53       names.foldLeft(ReadFilter(Expressions.alwaysTrue(), Some(filter), Set.empty))(reduce(schema, schemes))
54     }
55   }

57   private def reduce(schema: SimpleFeatureIcebergSchema, schemes: Seq[PartitionScheme])(result: ReadFilter, name: ColumnName): ReadFilter = {
58     val filter = result.remainder.orNull
59     if (filter == null) {
60       return result // no more filter to evaluate
61     }
62     val predicate =
63       if (name.column == SimpleFeatureSchema.FeatureIdField) {
64         fid(result)
65       } else {
66         val descriptor = schema.sft.getDescriptor(name.attribute)
67         if (descriptor != null) {
68           val bindings = ObjectType.selectType(schema.sft.getDescriptor(name.attribute))
69           bindings.head match {
70             case ObjectType.GEOMETRY => spatial(schema.sft, schemes, name, filter)
71             case ObjectType.DATE     => attribute[Date](schema.sft, name, filter, Some(dateToMicros))
72             case ObjectType.STRING   => attribute[String](schema.sft, name, filter)
73             case ObjectType.INT      => attribute[Integer](schema.sft, name, filter)
74             case ObjectType.LONG     => attribute[java.lang.Long](schema.sft, name, filter)
75             case ObjectType.FLOAT    => attribute[java.lang.Float](schema.sft, name, filter)
76             case ObjectType.DOUBLE   => attribute[java.lang.Double](schema.sft, name, filter)
77             case ObjectType.BOOLEAN  => attribute[java.lang.Boolean](schema.sft, name, filter)
78             case _                   => ReadFilter.clientSide(result.remainder)
79           }
80         } else if (name.attribute.startsWith("$")) {
81           jsonPath(schema, name.attribute, filter)
82         } else {
83           throw new IllegalArgumentException(s"Unknown attribute: ${name.attribute}")
84         }
85       }
86     ReadFilter(Expressions.and(predicate.expression, result.expression), predicate.remainder, predicate.columns ++ result.columns)
87   }

89   private def fid(result: ReadFilter): ReadFilter = {
90     val filter = result.remainder.orNull // not null already checked at the call sight
91     val fidCol = Set(SimpleFeatureSchema.FeatureIdField)
92     val (idFilters, notIds) = IdExtractingVisitor(filter)
93     val ids = idFilters.fold(Set.empty[String])(IdFilterStrategy.intersectIdFilters)
94     if (ids.isEmpty) {
95       logger.warn(s"Detected an ID filter, but could not extract it: ${ECQL.toCQL(filter)}")
96       ReadFilter.clientSide(result.remainder)
97     } else {
98       ReadFilter(ids.map(Expressions.equal(SimpleFeatureSchema.FeatureIdField, _)).reduce(Expressions.or), notIds, fidCol)
99     }
100   }

102   private def spatial(sft: SimpleFeatureType, schemes: Seq[PartitionScheme], name: ColumnName, filter: Filter): ReadFilter = {
103     val (spatial, nonSpatial) = FilterExtractingVisitor(filter, name.attribute, sft, SpatialFilterStrategy.spatialCheck)
104     if (spatial.isEmpty) {
105       // couldn't extract anything, fall back to client-side evaluation
106       return ReadFilter.clientSide(filter)
107     }
108 
109     val bounds = FilterHelper.extractGeometries(spatial.get, name.attribute)
110     if (bounds.disjoint) {
111       return ReadFilter.exclude()
112     }
113     val xyBounds = bounds.values.map(GeometryUtils.bounds)
114     if (xyBounds.isEmpty) {
115       // couldn't extract anything, fall back to client-side evaluation
116       return ReadFilter.clientSide(filter)
117     }
118 
119     // row/group level filter against the bbox field
120     val bboxPredicate = {
121       val isPoint = sft.getDescriptor(name.attribute).getType.getBinding == classOf[Point]
122       val predicates = xyBounds.map { case (xmin, ymin, xmax, ymax) =>
123         BoundingBoxField.filterIceberg(name.column, xmin, ymin, xmax, ymax, isPoint)
124       }
125       predicates.reduce(Expressions.or)
126     }
127     val bboxCols = ExpressionVisitors.visit(bboxPredicate, ReferenceVisitor)
128 
129     // partition level filter for partition pruning
130     // assertion - we don't need to include the z col in our read schema as this expression will be removed during manifest scanning
131     val spatialScheme = schemes.collectFirst { case s: SpatialScheme if s.attribute == name.attribute => s }
132     val partitionPredicate = spatialScheme.fold[Expression](Expressions.alwaysTrue())(_.getCoveringExpression(xyBounds))
133 
134     val remaining = if (bounds.precise) { nonSpatial } else { Some(filter) }
135 
136     ReadFilter(Expressions.and(bboxPredicate, partitionPredicate), remaining, bboxCols)
137   }

140       sft: SimpleFeatureType,
141       name: ColumnName,
142       filter: Filter,

154   /**
155    * Builds an iceberg predicate for a scalar attribute or nested field reference
156    *
157    * @param filter the full filter being reduced (used as the client-side residual when we can't fully push down)
158    * @param attributePart the portion of the filter that references our attribute
159    * @param nonAttributePart the portion of the filter that doesn't reference our attribute
160    * @param attribute the attribute name used in the geotools filter
161    * @param column the iceberg reference name (dotted for nested struct fields), including the encoded column
162    * @param binding the java type binding of the referenced field
163    * @param transform optional transform to convert extracted values into the iceberg storage representation
164    * @tparam T value type
165    * @return
166    */
167   private def predicate[T](
168       filter: Filter,
169       attributePart: Filter,
170       nonAttributePart: Option[Filter],
171       attribute: String,
172       column: String,
173       binding: Class[T],
174       transform: Option[T => Any]): ReadFilter = {
175     val bounds = FilterHelper.extractAttributeBounds(attributePart, attribute, binding)
176     if (bounds.disjoint) {
177       return ReadFilter.exclude()
178     } else if (bounds.isEmpty || bounds.exists(b => !b.isBounded)) {
179       // TODO detect the case where bounds were extracted, but are the wrong type (i.e. MyIntVar = 'alice') and return exclude
180       // couldn't extract anything, fall back to client-side evaluation
181       return ReadFilter.clientSide(filter)
182     }
183     val values = transform match {
184       case None => bounds.values
185       case Some(t) =>
186         bounds.values.map { bounds =>
187           bounds.copy(bounds.lower.copy(bounds.lower.value.map(t.apply)), bounds.upper.copy(bounds.upper.value.map(t.apply)))
188         }
189     }
190     val filters = values.map { bounds =>
191       if (bounds.isEquals) {
192         Expressions.equal(column, bounds.lower.value.get)
193       } else {
194         val lower = bounds.lower.value.map { value =>
195           if (bounds.lower.inclusive) { Expressions.greaterThanOrEqual(column, value) } else { Expressions.greaterThan(column, value) }
196         }
197         val upper = bounds.upper.value.map { value =>
198           if (bounds.upper.inclusive) { Expressions.lessThanOrEqual(column, value) } else { Expressions.lessThan(column, value) }
199         }
200         (lower, upper) match {
201           case (Some(lo), Some(hi)) => Expressions.and(lo, hi)
202           case (Some(f), None) => f
203           case (None, Some(f)) => f
204           case (None, None) => throw new IllegalStateException() // shouldn't happen due to checks above
205         }
206       }
207     }
208     val result = merge(filters)
209     val remaining = if (bounds.precise) { nonAttributePart } else { Some(filter) }
210     ReadFilter(result, remaining, Set(column))
211   }

213   private def jsonPath(schema: SimpleFeatureIcebergSchema, pathString: String, filter: Filter): ReadFilter = {
214     val (descriptor, path) = parseJsonPath(pathString, schema.sft)
215     if (descriptor.getJsonSchema().isEmpty) {
216       // not a structural type - the field is stored as an opaque variant, so we can't push down against it
217       return ReadFilter.clientSide(filter)
218     }
219     val topLevelColumn = ColumnName.encode(descriptor.getLocalName)
220     val field = schema.schema.findField(topLevelColumn)
221     if (field == null || path.function.isDefined) {
222       // field shouldn't ever be null, but guard as a sanity check
223       // path functions (.min(), .length(), etc) can't be evaluated as an iceberg predicate
224       return ReadFilter.clientSide(filter)
225     }
226     // extract just the part of the filter that references our json path - no sft as it won't recognize the path as an attribute
227     val (attribute, nonAttribute) = FilterExtractingVisitor(filter, pathString, null: SimpleFeatureType)
228     if (attribute.isEmpty) {
229       // couldn't extract anything, fall back to client-side evaluation
230       return ReadFilter.clientSide(filter)
231     }
232 
233     // navigate the remaining path elements through the nested struct type to find the leaf fields we're filtering on
234     navigate(topLevelColumn, field.`type`(), path) match {
235       // can't express the json path as an iceberg predicates, evaluate it client-side instead
236       case None => ReadFilter.clientSide(filter)
237       case Some(matchingFields) =>
238         if (matchingFields.isEmpty) {
239           // the path was evaluated but didn't match any fields
240           throw new IllegalArgumentException(
241             s"Invalid JSON path - does not match any elements of the structural type $topLevelColumn: $pathString")
242         }
243         val filters = matchingFields.map { case (column, leafType) =>
244           val binding = leafType.typeId() match {
245             case TypeID.STRING  => Some(classOf[String])
246             case TypeID.INTEGER => Some(classOf[Integer])
247             case TypeID.LONG    => Some(classOf[java.lang.Long])
248             case TypeID.FLOAT   => Some(classOf[java.lang.Float])
249             case TypeID.DOUBLE  => Some(classOf[java.lang.Double])
250             case TypeID.BOOLEAN => Some(classOf[java.lang.Boolean])
251             // unsupported leaf type (dates, times, uuids, decimals, binary, etc) - evaluate client-side
252             // TODO seems like we should be able to support at least dates here?
253             case _ => None
254           }
255           binding match {
256             case None => ReadFilter.clientSide(filter)
257             case Some(b) => predicate(filter, attribute.get, nonAttribute, pathString, column, b, None)
258           }
259         }
260         // if the path matches more than 1 leaf node, combine the filter expressions with ORs
261         filters.reduceLeft[ReadFilter] { case (left, right) =>
262           // if we have a client-side evaluation, then just do that
263           Seq(left, right).find(_.expression == Expressions.alwaysTrue()).getOrElse {
264             // predicate will always return either the full filter or the non-attribute part
265             val f = if (left.remainder.contains(filter) || right.remainder.contains(filter)) { Some(filter) } else { nonAttribute }
266             ReadFilter(Expressions.or(left.expression, right.expression), f, left.columns ++ right.columns)
267           }
268         }
269     }
270   }

272   /**
273    * Navigates a nested struct type following a json path, returning the nested field reference names and the
274    * leaf type if the path points at a scalar field, or None if the path can't be pushed down
275    *
276    * @param fieldPath dot-delimited path to the current field being evaluated
277    * @param fieldType the type of the field the path currently points at
278    * @param path the remaining path to navigate
279    * @return pairs of nested field names (matching the stored schema) and the leaf primitive type, or None if the path is not supported
280    */
281   private def navigate(fieldPath: String, fieldType: Type, path: JsonPath): Option[Seq[(String, Type)]] = {
282     if (path.isEmpty) {
283       if (fieldType.isPrimitiveType) {
284         return Some(Seq(fieldPath -> fieldType))
285       } else {
286         throw new UnsupportedOperationException(
287           s"Invalid JSON path - filters on non-leaf nodes are not supported: $$.$fieldPath")
288       }
289     }
290     path.head match {
291       case PathAttribute(name, _) if fieldType.isStructType =>
292         val nested = fieldType.asStructType().caseInsensitiveField(name)
293         if (nested == null) {
294           Some(Seq.empty) // path references a field that isn't in our schema
295         } else {
296           navigate(s"$fieldPath.${nested.name()}", nested.`type`(), path.tail)
297         }
298 
299       case PathAttributeWildCard if fieldType.isStructType =>
300         val tail = path.tail
301         val children = fieldType.asStructType().fields().asScala.map { nested =>
302           val nestedPath = s"$fieldPath.${nested.name()}"
303           if (tail.isEmpty) {
304             if (nested.`type`().isPrimitiveType) {
305               Some(Seq(nestedPath -> nested.`type`()))
306             } else {
307               Some(Seq.empty)
308             }
309           } else {
310             navigate(nestedPath, nested.`type`(), tail)
311           }
312         }
313         // if any children can't be evaluated, the whole expression can't be evaluated
314         if (children.exists(_.isEmpty)) { None } else { Some(children.flatMap(_.get).toSeq) }
315 
316       // valid predicate, but we can't evaluate it as an iceberg expression
317       case _: PathAttribute if fieldType.isMapType => None
318       case PathAttributeWildCard if fieldType.isMapType => None
319       case _: PathIndexRange if fieldType.isListType => None
320       case _: PathIndices if fieldType.isListType => None
321       case PathIndexWildCard if fieldType.isListType => None
322       case PathDeepScan => None
323       case _: PathFilter => None
324 
325       // anything else doesn't match the schema
326       case _ => Some(Seq.empty)
327     }
328   }

330   /**
331    * Merge OR'd filters
332    *
333    * Detect and re-write "not equals" filters to handle null values
334    *
335    * FilterHelper methods make "foo != x" comes out as "foo < x OR foo > x"
336    *
337    * that won't return null values - not normally a concern since we don't index them in key-value dbs,
338    * but we want to handle that case here
339    *
340    * @param filters filters
341    * @return combined filter
342    */
343   private def merge(filters: Seq[Expression]): Expression = {
344     lazy val values = filters.collect { case f: UnboundPredicate[_] => (f.ref().name(), f.literal().value()) }.distinct
345     if (filters.lengthCompare(2) == 0 &&
346       filters.exists(_.op() == Operation.LT) &&
347       filters.exists(_.op == Operation.GT) &&
348       values.lengthCompare(1) == 0) {
349       Expressions.notEqual(values.head._1, values.head._2)
350     } else {
351       filters.reduce(Expressions.or)
352     }
353   }

372   /**
373    * Visitor to extract column references (names) from an expression
374    */
375   private object ReferenceVisitor extends ExpressionVisitor[Set[String]] {
376     override def alwaysTrue: Set[String] = Set.empty
377     override def alwaysFalse: Set[String] = Set.empty
378     override def not(result: Set[String]): Set[String] = result
379     override def and(leftResult: Set[String], rightResult: Set[String]): Set[String] = leftResult ++ rightResult
380     override def or(leftResult: Set[String], rightResult: Set[String]): Set[String] = leftResult ++ rightResult
381     override def predicate[T](pred: UnboundPredicate[T]): Set[String] = unwrapTerm(pred.term())
382     override def predicate[T](pred: BoundPredicate[T]): Set[String] = unwrapTerm(pred.term())
383 
384     private def unwrapTerm(term: Term): Set[String] = term match {
385       case ref: Reference[_] => Set(ref.name())
386       case t: UnboundTransform[_, _] => Set(t.ref.name())
387       case t: BoundTransform[_, _] => Set(t.ref.name())
388       case _ => Set.empty
389     }
