38   def apply(sft: SimpleFeatureType, schemes: Seq[PartitionScheme], filter: Filter): ReadFilter = {
39     if (filter == Filter.INCLUDE) {
40       ReadFilter(Expressions.alwaysTrue(), None, Set.empty)
41     } else if (filter == Filter.EXCLUDE) {
42       ReadFilter(Expressions.alwaysFalse(), None, Set.empty)
43     } else {
44       val fid = if (FilterHelper.hasIdFilter(filter)) { Seq(SimpleFeatureSchema.FeatureIdField) } else { Seq.empty }
45       val names = (fid ++ FilterHelper.propertyNames(filter)).map(ColumnName.apply)
46       names.foldLeft(ReadFilter(Expressions.alwaysTrue(), Some(filter), Set.empty))(reduce(sft, schemes))
47     }
48   }

50   private def reduce(sft: SimpleFeatureType, schemes: Seq[PartitionScheme])(result: ReadFilter, name: ColumnName): ReadFilter = {
51     val filter = result.remainder.orNull
52     if (filter == null) {
53       return result // no more filter to evaluate
54     }
55     val predicate =
56       if (name.column == SimpleFeatureSchema.FeatureIdField) {
57         fid(result)
58       } else {
59         val bindings = ObjectType.selectType(sft.getDescriptor(name.attribute))
60         bindings.head match {
61           // note: non-points use repeated values, which aren't supported in parquet predicates
62           case ObjectType.GEOMETRY => spatial(sft, schemes, name, filter)
63           case ObjectType.DATE     => attribute[Date](sft, name, filter, Some(dateToMicros))
64           case ObjectType.STRING   => attribute[String](sft, name, filter)
65           case ObjectType.INT      => attribute[Integer](sft, name, filter)
66           case ObjectType.LONG     => attribute[java.lang.Long](sft, name, filter)
67           case ObjectType.FLOAT    => attribute[java.lang.Float](sft, name, filter)
68           case ObjectType.DOUBLE   => attribute[java.lang.Double](sft, name, filter)
69           case ObjectType.BOOLEAN  => attribute[java.lang.Boolean](sft, name, filter)
70           case _ => ReadFilter(Expressions.alwaysTrue(), result.remainder, Set(name.column))
71         }
72       }
73     ReadFilter(Expressions.and(predicate.expression, result.expression), predicate.remainder, predicate.columns ++ result.columns)
74   }

76   private def fid(result: ReadFilter): ReadFilter = {
77     val filter = result.remainder.orNull // not null already checked at the call sight
78     val fidCol = Set(SimpleFeatureSchema.FeatureIdField)
79     val (idFilters, notIds) = IdExtractingVisitor(filter)
80     val ids = idFilters.fold(Set.empty[String])(IdFilterStrategy.intersectIdFilters)
81     if (ids.isEmpty) {
82       logger.warn(s"Detected an ID filter, but could not extract it: ${ECQL.toCQL(filter)}")
83       ReadFilter(Expressions.alwaysTrue(), result.remainder, fidCol)
84     } else {
85       ReadFilter(ids.map(Expressions.equal(SimpleFeatureSchema.FeatureIdField, _)).reduce(Expressions.or), notIds, fidCol)
86     }
87   }

89   private def spatial(sft: SimpleFeatureType, schemes: Seq[PartitionScheme], name: ColumnName, filter: Filter): ReadFilter = {
90     val (spatial, nonSpatial) = FilterExtractingVisitor(filter, name.attribute, sft, SpatialFilterStrategy.spatialCheck)
91     if (spatial.isEmpty) {
92       return ReadFilter(Expressions.alwaysTrue(), Some(filter), Set.empty)
93     }
94 
95     val bounds = FilterHelper.extractGeometries(spatial.get, name.attribute)
96     if (bounds.disjoint) {
97       return ReadFilter(Expressions.alwaysFalse(), None, Set.empty)
98     }
99     val xyBounds = bounds.values.map(GeometryUtils.bounds)
100     if (xyBounds.isEmpty) {
101       // couldn't extract anything, all evaluation will be client-side against the raw filter
102       return ReadFilter(Expressions.alwaysTrue(), Some(filter), Set(name.column))
103     }
104 
105     // row/group level filter against the bbox field
106     val bboxPredicate = {
107       val isPoint = sft.getDescriptor(name.attribute).getType.getBinding == classOf[Point]
108       val predicates = xyBounds.map { case (xmin, ymin, xmax, ymax) =>
109         BoundingBoxField.filterIceberg(name.column, xmin, ymin, xmax, ymax, isPoint)
110       }
111       predicates.reduce(Expressions.or)
112     }
113     val bboxCols = ExpressionVisitors.visit(bboxPredicate, ReferenceVisitor)
114 
115     // partition level filter for partition pruning
116     // assertion - we don't need to include the z col in our read schema as this expression will be removed during manifest scanning
117     val spatialScheme = schemes.collectFirst { case s: SpatialScheme if s.attribute == name.attribute => s }
118     val partitionPredicate = spatialScheme.fold[Expression](Expressions.alwaysTrue())(_.getCoveringExpression(xyBounds))
119 
120     val (remaining, geomCol) = if (bounds.precise) { (nonSpatial, None) } else { (Some(filter), Some(name.column)) }
121     val filterCols = bboxCols ++ geomCol
122 
123     ReadFilter(Expressions.and(bboxPredicate, partitionPredicate), remaining, filterCols)
124   }

126   private def attribute[T : ClassTag](
127       sft: SimpleFeatureType,
128       name: ColumnName,
129       filter: Filter,
130       transform: Option[T => Any] = None): ReadFilter = {
131     val (attribute, nonAttribute) = FilterExtractingVisitor(filter, name.attribute, sft)
132     if (attribute.isEmpty) {
133       return ReadFilter(Expressions.alwaysTrue(), Some(filter), Set.empty)
134     }
135 
136     val binding = implicitly[ClassTag[T]].runtimeClass.asInstanceOf[Class[T]]
137     val bounds = FilterHelper.extractAttributeBounds(attribute.get, name.attribute, binding)
138     if (bounds.disjoint) {
139       return ReadFilter(Expressions.alwaysFalse(), None, Set.empty)
140     } else if (bounds.isEmpty || bounds.exists(b => !b.isBounded)) {
141       // couldn't extract anything, all evaluation will be client-side against the raw filter
142       return ReadFilter(Expressions.alwaysTrue(), Some(filter), Set(name.column))
143     }
144     val values = transform match {
145       case None => bounds.values
146       case Some(t) =>
147         bounds.values.map { bounds =>
148           bounds.copy(bounds.lower.copy(bounds.lower.value.map(t.apply)), bounds.upper.copy(bounds.upper.value.map(t.apply)))
149         }
150     }
151     val filters = values.map { bounds =>
152       if (bounds.isEquals) {
153         Expressions.equal(name.column, bounds.lower.value.get)
154       } else {
155         val lower = bounds.lower.value.map { value =>
156           if (bounds.lower.inclusive) { Expressions.greaterThanOrEqual(name.column, value) } else { Expressions.greaterThan(name.column, value) }
157         }
158         val upper = bounds.upper.value.map { value =>
159           if (bounds.upper.inclusive) { Expressions.lessThanOrEqual(name.column, value) } else { Expressions.lessThan(name.column, value) }
160         }
161         (lower, upper) match {
162           case (Some(lo), Some(hi)) => Expressions.and(lo, hi)
163           case (Some(f), None) => f
164           case (None, Some(f)) => f
165           case (None, None) => throw new IllegalStateException() // shouldn't happen due to checks above
166         }
167       }
168     }
169     val predicate = merge(filters)
170     val remaining = if (bounds.precise) { nonAttribute } else { Some(filter) }
171     ReadFilter(predicate, remaining, Set(name.column))
172   }

174   /**
175    * Merge OR'd filters
176    *
177    * Detect and re-write "not equals" filters to handle null values
178    *
179    * FilterHelper methods make "foo != x" comes out as "foo < x OR foo > x"
180    *
181    * that won't return null values - not normally a concern since we don't index them in key-value dbs,
182    * but we want to handle that case here
183    *
184    * @param filters filters
185    * @return combined filter
186    */
187   private def merge(filters: Seq[Expression]): Expression = {
188     lazy val values = filters.collect { case f: UnboundPredicate[_] => (f.ref().name(), f.literal().value()) }.distinct
189     if (filters.lengthCompare(2) == 0 &&
190       filters.exists(_.op() == Operation.LT) &&
191       filters.exists(_.op == Operation.GT) &&
192       values.lengthCompare(1) == 0) {
193       Expressions.notEqual(values.head._1, values.head._2)
194     } else {
195       filters.reduce(Expressions.or)
196     }
197   }

213   private object ReferenceVisitor extends ExpressionVisitor[Set[String]] {
214 
215     override def alwaysTrue: Set[String] = Set.empty
216 
217     override def alwaysFalse: Set[String] = Set.empty
218 
219     override def not(result: Set[String]): Set[String] = result
220 
221     override def and(leftResult: Set[String], rightResult: Set[String]): Set[String] = leftResult ++ rightResult
222 
223     override def or(leftResult: Set[String], rightResult: Set[String]): Set[String] = leftResult ++ rightResult
224 
225     override def predicate[T](pred: UnboundPredicate[T]): Set[String] = unwrapTerm(pred.term())
226 
227     override def predicate[T](pred: BoundPredicate[T]): Set[String] = unwrapTerm(pred.term())
228 
229     private def unwrapTerm(term: Term): Set[String] = term match {
230       case ref: Reference[_] => Set(ref.name())
231       case t: UnboundTransform[_, _] => Set(t.ref.name())
232       case t: BoundTransform[_, _] => Set(t.ref.name())
233       case _ => Set.empty
234     }
