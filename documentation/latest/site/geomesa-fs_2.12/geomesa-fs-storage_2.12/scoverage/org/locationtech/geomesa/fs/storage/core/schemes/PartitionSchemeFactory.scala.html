41 object PartitionSchemeFactory {
42 
43   import scala.collection.JavaConverters._
44 
45   lazy private val factories =
46     Seq(AttributeScheme, DateTimeScheme, HashScheme, XZ2Scheme, Z2Scheme)
47 
48   /**
49     * Create a partition scheme instance via available factories
50     *
51     * @param sft simple feature type
52     * @param scheme scheme options
53     * @return
54     */
55   def load(sft: SimpleFeatureType, scheme: String): PartitionScheme = {
56     factories.toStream.flatMap(_.load(sft, scheme)).headOption.getOrElse {
57       throw new IllegalArgumentException(s"No partition scheme factory implementation exists for name " +
58         s"'$scheme'. Available factories: ${factories.map(_.getClass.getName).mkString(", ")}")
59     }
60   }
61 
62 
63   /**
64    * Create a partition scheme instance via available factories
65    *
66    * @param schema simple feature schema
67    * @param spec iceberg partition spec
68    * @return
69    */
70   def load(schema: SimpleFeatureIcebergSchema, spec: PartitionSpec): Seq[PartitionScheme] = {
71     spec.fields().asScala.toSeq.map { field =>
72       factories.toStream.flatMap(_.load(schema, spec.schema(), field)).headOption.getOrElse {
73         throw new IllegalArgumentException(s"No partition scheme factory implementation exists for field " +
74           s"'$field'. Available factories: ${factories.map(_.getClass.getName).mkString(", ")}")
75       }
76     }
77   }
78 
79   /**
80    * Base class that implements spec visitor and returns None for everything. PartitionFactories
81    * can override the transforms that they are interested in when loading partition specs
82    */
83   private[schemes] class BaseSpecVisitor extends PartitionSpecVisitor[Option[PartitionScheme]]() {
84     override def identity(sourceName: String, sourceId: Int): Option[PartitionScheme] = None
85     override def bucket(sourceName: String, sourceId: Int, numBuckets: Int): Option[PartitionScheme] = None
86     override def truncate(sourceName: String, sourceId: Int, width: Int): Option[PartitionScheme] = None
87     override def year(sourceName: String, sourceId: Int): Option[PartitionScheme] = None
88     override def month(sourceName: String, sourceId: Int): Option[PartitionScheme] = None
89     override def day(sourceName: String, sourceId: Int): Option[PartitionScheme] = None
90     override def hour(sourceName: String, sourceId: Int): Option[PartitionScheme] = None
91     override def alwaysNull(fieldId: Int, sourceName: String, sourceId: Int): Option[PartitionScheme] = None
92     override def unknown(fieldId: Int, sourceName: String, sourceId: Int, transform: String): Option[PartitionScheme] = None
93   }
