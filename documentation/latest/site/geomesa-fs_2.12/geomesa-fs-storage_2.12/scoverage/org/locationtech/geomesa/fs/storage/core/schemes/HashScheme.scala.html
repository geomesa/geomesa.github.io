21 /**
22  * Hash (bucket) scheme
23  *
24  * @param attribute attribute name
25  * @param index index of the attribute in the feature type
26  * @param buckets number of buckets values are hashed into
27  * @param hasher type-specific hasher
28  */
29 case class HashScheme[T](attribute: String, index: Int, buckets: Int, hasher: MurmurHashFunction.Hashing[T])
30     extends PartitionScheme with EnumeratedScheme {
31 
32   import FilterHelper.ff
33 
34   override val name: String = s"${HashScheme.Name}:attribute=$attribute:buckets=$buckets"
35 
36   private val format = s"%0${(buckets - 1).toString.length}d"
37   private val default = PartitionKey(name, format.format(0))
38 
39   override def getPartition(feature: SimpleFeature): PartitionKey = {
40     val value = feature.getAttribute(index)
41     if (value == null) {
42       default
43     } else {
44       PartitionKey(name, format.format((hasher(value.asInstanceOf[T]) & Int.MaxValue) % buckets))
45     }
46   }
47 
48   override def getCoveringFilter(partition: PartitionKey): Filter = {
49     val fn = BucketHashFunction.Name.getFunctionName
50     ff.equals(ff.function(fn, ff.property(attribute), ff.literal(buckets)), ff.literal(partition.value.toInt))
51   }
52 
53   override def getCoveringExpression(partition: PartitionKey): Expression =
54     Expressions.equal(Expressions.bucket[Integer](ColumnName.encode(attribute), buckets), Integer.valueOf(partition.value))
55 
56   override def spec(b: PartitionSpec.Builder): PartitionSpec.Builder = b.bucket(ColumnName.encode(attribute), buckets)
57 
58   override def getPartition(partition: StructLike, i: Int): PartitionKey =
59     PartitionKey(name, format.format(partition.get(i, classOf[Integer])))
60 
61   override def partitions: Seq[PartitionKey] = Seq.tabulate(buckets)(i => PartitionKey(name, format.format(i)))
62 }
63 
64 object HashScheme extends PartitionSchemeFactory {
65 
66   val Name = "hash"
67 
68   override def load(sft: SimpleFeatureType, scheme: String): Option[PartitionScheme] = {
69     val opts = SchemeOpts(scheme)
70     if (opts.name != Name) { None } else {
71       val attribute = opts.getSingle("attribute").orNull
72       require(attribute != null, s"Hash scheme requires an attribute to be specified with 'attribute=<attribute>'")
73       val index = attributeIndex(sft, attribute)
74       val binding = sft.getDescriptor(index).getType.getBinding
75       val hasher = MurmurHashFunction.Hashers.find(_.binding.isAssignableFrom(binding)).orNull
76       require(hasher != null, s"Invalid type binding '${binding.getName}' of attribute '$attribute'")
77       val buckets = opts.getSingle("buckets").orNull
78       require(buckets != null, s"Hash scheme requires a number of buckets to be specified with 'buckets=<n>'")
79       Some(new HashScheme(attribute, index, buckets.toInt, hasher))
80     }
81   }
82 }
