24 /**
25  * Hash (bucket) scheme
26  *
27  * @param attribute attribute name
28  * @param index index of the attribute in the feature type
29  * @param buckets number of buckets values are hashed into
30  * @param hasher type-specific hasher
31  */
32 case class HashScheme[T](attribute: String, index: Int, buckets: Int, hasher: MurmurHashFunction.Hashing[T])
33     extends PartitionScheme with EnumeratedScheme {
34 
35   import FilterHelper.ff
36 
37   override val name: String = s"${HashScheme.Name}:attribute=$attribute:buckets=$buckets"
38 
39   override val column: String = ColumnName.encode(attribute)
40 
41   private val format = s"%0${(buckets - 1).toString.length}d"
42   private val default = PartitionKey(name, format.format(0))
43 
44   override def getPartition(feature: SimpleFeature): PartitionKey = {
45     val value = feature.getAttribute(index)
46     if (value == null) {
47       default
48     } else {
49       PartitionKey(name, format.format((hasher(value.asInstanceOf[T]) & Int.MaxValue) % buckets))
50     }
51   }
52 
53   override def getCoveringFilter(partition: PartitionKey): Filter = {
54     val fn = BucketHashFunction.Name.getFunctionName
55     ff.equals(ff.function(fn, ff.property(attribute), ff.literal(buckets)), ff.literal(partition.value.toInt))
56   }
57 
58   override def getCoveringExpression(partition: PartitionKey): Expression =
59     Expressions.equal(Expressions.bucket[Integer](column, buckets), Integer.valueOf(partition.value))
60 
61   override def spec(b: PartitionSpec.Builder): PartitionSpec.Builder = b.bucket(column, buckets)
62 
63   override def getPartition(partition: StructLike, i: Int): PartitionKey =
64     PartitionKey(name, format.format(partition.get(i, classOf[Integer])))
65 
66   override def partitions: Seq[PartitionKey] = Seq.tabulate(buckets)(i => PartitionKey(name, format.format(i)))
67 }

69 object HashScheme extends PartitionSchemeFactory {
70 
71   import scala.collection.JavaConverters._
72 
73   val Name = "hash"
74 
75   override def load(sft: SimpleFeatureType, scheme: String): Option[PartitionScheme] = {
76     val opts = SchemeOpts(scheme)
77     if (opts.name != Name) { None } else {
78       val attribute = opts.getSingle("attribute").orNull
79       require(attribute != null, s"Hash scheme requires an attribute to be specified with 'attribute=<attribute>'")
80       val index = attributeIndex(sft, attribute)
81       val binding = sft.getDescriptor(index).getType.getBinding
82       val hasher = MurmurHashFunction.Hashers.find(_.binding.isAssignableFrom(binding)).orNull
83       require(hasher != null, s"Invalid type binding '${binding.getName}' of attribute '$attribute'")
84       val buckets = opts.getSingle("buckets").orNull
85       require(buckets != null, s"Hash scheme requires a number of buckets to be specified with 'buckets=<n>'")
86       Some(new HashScheme(attribute, index, buckets.toInt, hasher))
87     }
88   }
89 
90   override def load(schema: SimpleFeatureIcebergSchema, spec: Schema, field: PartitionField): Option[PartitionScheme] = {
91     PartitionSpecVisitor.visit(spec, field, new BaseSpecVisitor() {
92       override def bucket(sourceName: String, sourceId: Int, numBuckets: Int): Option[PartitionScheme] = {
93         schema.sft.getAttributeDescriptors.asScala.find(d => sourceName == ColumnName.encode(d.getLocalName)).flatMap { d =>
94           val binding = d.getType.getBinding
95           MurmurHashFunction.Hashers.find(_.binding.isAssignableFrom(binding)).map { hasher =>
96             val attribute = d.getLocalName
97             new HashScheme(attribute, schema.sft.indexOf(attribute), numBuckets, hasher)
98           }
99         }
100       }
101     })
102   }
