21 /**
22  * Hash (bucket) scheme
23  *
24  * @param attribute attribute name
25  * @param index index of the attribute in the feature type
26  * @param buckets number of buckets values are hashed into
27  * @param hasher type-specific hasher
28  */
29 case class HashScheme[T](attribute: String, index: Int, buckets: Int, hasher: MurmurHashFunction.Hashing[T])
30     extends PartitionScheme with EnumeratedScheme {
31 
32   import FilterHelper.ff
33 
34   override val name: String = s"${HashScheme.Name}:attribute=$attribute:buckets=$buckets"
35 
36   override val column: String = ColumnName.encode(attribute)
37 
38   private val format = s"%0${(buckets - 1).toString.length}d"
39   private val default = PartitionKey(name, format.format(0))
40 
41   override def getPartition(feature: SimpleFeature): PartitionKey = {
42     val value = feature.getAttribute(index)
43     if (value == null) {
44       default
45     } else {
46       PartitionKey(name, format.format((hasher(value.asInstanceOf[T]) & Int.MaxValue) % buckets))
47     }
48   }
49 
50   override def getCoveringFilter(partition: PartitionKey): Filter = {
51     val fn = BucketHashFunction.Name.getFunctionName
52     ff.equals(ff.function(fn, ff.property(attribute), ff.literal(buckets)), ff.literal(partition.value.toInt))
53   }
54 
55   override def getCoveringExpression(partition: PartitionKey): Expression =
56     Expressions.equal(Expressions.bucket[Integer](column, buckets), Integer.valueOf(partition.value))
57 
58   override def spec(b: PartitionSpec.Builder): PartitionSpec.Builder = b.bucket(column, buckets)
59 
60   override def getPartition(partition: StructLike, i: Int): PartitionKey =
61     PartitionKey(name, format.format(partition.get(i, classOf[Integer])))
62 
63   override def partitions: Seq[PartitionKey] = Seq.tabulate(buckets)(i => PartitionKey(name, format.format(i)))
64 }
65 
66 object HashScheme extends PartitionSchemeFactory {
67 
68   val Name = "hash"
69 
70   override def load(sft: SimpleFeatureType, scheme: String): Option[PartitionScheme] = {
71     val opts = SchemeOpts(scheme)
72     if (opts.name != Name) { None } else {
73       val attribute = opts.getSingle("attribute").orNull
74       require(attribute != null, s"Hash scheme requires an attribute to be specified with 'attribute=<attribute>'")
75       val index = attributeIndex(sft, attribute)
76       val binding = sft.getDescriptor(index).getType.getBinding
77       val hasher = MurmurHashFunction.Hashers.find(_.binding.isAssignableFrom(binding)).orNull
78       require(hasher != null, s"Invalid type binding '${binding.getName}' of attribute '$attribute'")
79       val buckets = opts.getSingle("buckets").orNull
80       require(buckets != null, s"Hash scheme requires a number of buckets to be specified with 'buckets=<n>'")
81       Some(new HashScheme(attribute, index, buckets.toInt, hasher))
82     }
83   }
84 }
