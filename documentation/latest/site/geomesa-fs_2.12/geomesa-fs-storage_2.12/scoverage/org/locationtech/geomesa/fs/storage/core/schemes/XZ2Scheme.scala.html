23 /**
24  * XZ2 spatial scheme.
25  *
26  * This scheme uses a fixed high-resolution curve for indexing, then truncates the value to create partition groups,
27  * in order to align with the iceberg 'truncate' transforms.
28  *
29  * Index values are (max) 7-digit hex-encoded longs, as we're only using 25 bits for the high-resolution curve,
30  * we can fit it in 7 digits.
31  *
32  * @param attribute name of the attribute being partitioned
33  * @param index index in the sft of the attribute being partitioned
34  * @param bits number of bits of resolution used for partitioning
35  */
36 case class XZ2Scheme(attribute: String, index: Int, bits: Int) extends SpatialScheme {
37 
38   import FilterHelper.ff
39 
40   require(bits % 4 == 0, s"Bit precision must be a multiple of 4, but received $bits")
41 
42   private val xz2 = XZ2SFC
43 
44   // partition level derived from bits parameter
45   // each level adds 2 bits (4 quadrants)
46   private val partitionLevel = (bits / 2).toShort
47   // number of hex digits used to represent our z value - bits = (xz2.g - partitionLevel) * 2, then divide by 4 to get hex
48   override protected val digits = xz2.hexDigits - ((xz2.g - partitionLevel) / 2)
49 
50   override val name: String = s"${XZ2Scheme.name}:attribute=$attribute:bits=$bits"
51 
52   override def getPartition(feature: SimpleFeature): PartitionKey = {
53     val envelope = feature.getAttribute(index).asInstanceOf[Geometry].getEnvelopeInternal
54     PartitionKey(name, xz2.hexEncode(envelope.getMinX, envelope.getMinY, envelope.getMaxX, envelope.getMaxY).take(digits))
55   }
56 
57   override def getCoveringFilter(partition: PartitionKey): Filter = {
58     // TODO maybe we can improve this with *some* kind of bbox?
59     val hexPrefix = partition.value
60     val lower = hexPrefix.padTo(xz2.hexDigits, '0')
61     val upper = hexPrefix.padTo(xz2.hexDigits, 'f')
62     ff.between(ff.function(XZ2Function.FunctionName.getName), ff.literal(lower), ff.literal(upper))
63   }
64 
65   override def getCoveringExpression(partition: PartitionKey): Expression =
66     Expressions.equal[String](Expressions.truncate[String](ZValueField.xz2FieldName(ColumnName.encode(attribute)), digits), partition.value)
67 
68   override def spec(b: PartitionSpec.Builder): PartitionSpec.Builder =
69     b.truncate(ZValueField.xz2FieldName(ColumnName.encode(attribute)), digits)
70 
71   override def getPartition(partition: StructLike, i: Int): PartitionKey = PartitionKey(name, partition.get(i, classOf[String]))
72 }
73 
74 object XZ2Scheme extends SpatialPartitionSchemeFactory[Geometry]("xz2") {
75   override def buildPartitionScheme(bits: Int, geom: String, geomIndex: Int): PartitionScheme =
76     XZ2Scheme(geom, geomIndex, bits)
