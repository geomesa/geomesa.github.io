27 /**
28  * XZ2 spatial scheme.
29  *
30  * This scheme uses a fixed high-resolution curve for indexing, then truncates the value to create partition groups,
31  * in order to align with the iceberg 'truncate' transforms.
32  *
33  * Index values are (max) 7-digit hex-encoded longs, as we're only using 25 bits for the high-resolution curve,
34  * we can fit it in 7 digits.
35  *
36  * @param attribute name of the attribute being partitioned
37  * @param index index in the sft of the attribute being partitioned
38  * @param bits number of bits of resolution used for partitioning
39  */
40 case class XZ2Scheme(attribute: String, index: Int, bits: Int) extends SpatialScheme {
41 
42   import FilterHelper.ff
43 
44   require(bits % 4 == 0, s"Bit precision must be a multiple of 4, but received $bits")
45 
46   private val xz2 = XZ2SFC
47 
48   // partition level derived from bits parameter
49   // each level adds 2 bits (4 quadrants)
50   private val partitionLevel = (bits / 2).toShort
51   // number of hex digits used to represent our z value - bits = (xz2.g - partitionLevel) * 2, then divide by 4 to get hex
52   override protected val digits: Int = xz2.hexDigits - ((xz2.g - partitionLevel) / 2)
53 
54   override val name: String = s"${XZ2Scheme.name}:attribute=$attribute:bits=$bits"
55 
56   override val column: String = ZValueField.xz2FieldName(ColumnName.encode(attribute))
57 
58   override def getPartition(feature: SimpleFeature): PartitionKey = {
59     val envelope = feature.getAttribute(index).asInstanceOf[Geometry].getEnvelopeInternal
60     PartitionKey(name, xz2.hexEncode(envelope.getMinX, envelope.getMinY, envelope.getMaxX, envelope.getMaxY).take(digits))
61   }
62 
63   override def getCoveringFilter(partition: PartitionKey): Filter = {
64     // TODO maybe we can improve this with *some* kind of bbox?
65     val hexPrefix = partition.value
66     val lower = hexPrefix.padTo(xz2.hexDigits, '0')
67     val upper = hexPrefix.padTo(xz2.hexDigits, 'f')
68     ff.between(ff.function(XZ2Function.FunctionName.getName), ff.literal(lower), ff.literal(upper))
69   }
70 
71   override def getPartition(partition: StructLike, i: Int): PartitionKey = PartitionKey(name, partition.get(i, classOf[String]))
72 
73   override protected def hexRanges(bounds: Seq[(Double, Double, Double, Double)]): Seq[(String, String)] = {
74     val max = QueryProperties.ScanRangesTarget.toInt
75     xz2.ranges(bounds, max).map(r => (xz2.hexEncode(r.lower), xz2.hexEncode(r.upper)))
76   }
77 }

79 object XZ2Scheme extends SpatialPartitionSchemeFactory[Geometry]("xz2") {
80 
81   import scala.collection.JavaConverters._
82 
83   override def buildPartitionScheme(bits: Int, geom: String, geomIndex: Int): PartitionScheme =
84     XZ2Scheme(geom, geomIndex, bits)
85 
86   override def load(schema: SimpleFeatureIcebergSchema, spec: Schema, field: PartitionField): Option[PartitionScheme] = {
87     PartitionSpecVisitor.visit(spec, field, new BaseSpecVisitor() {
88       override def truncate(sourceName: String, sourceId: Int, width: Int): Option[PartitionScheme] = {
89         if (sourceName.endsWith(ZValueField.XZ2ValueFieldSuffix)) {
90           schema.sft.getAttributeDescriptors.asScala.collectFirst {
91             case d: GeometryDescriptor if ZValueField.xz2FieldName(ColumnName.encode(d.getLocalName)) == sourceName =>
92               // there's 1 extra hex digit compared to number of bits to account for bit-shifting
93               buildPartitionScheme((width - 1) * 4, d.getLocalName, schema.sft.indexOf(d.getLocalName))
94           }
95         } else {
96           None
97         }
98       }
99     })
100   }
