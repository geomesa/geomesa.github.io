23 /**
24  * XZ2 spatial scheme.
25  *
26  * This scheme uses a fixed high-resolution curve for indexing, then truncates the value to create partition groups,
27  * in order to align with the iceberg 'truncate' transforms.
28  *
29  * Index values are (max) 7-digit hex-encoded longs, as we're only using 25 bits for the high-resolution curve,
30  * we can fit it in 7 digits.
31  *
32  * @param attribute name of the attribute being partitioned
33  * @param index index in the sft of the attribute being partitioned
34  * @param bits number of bits of resolution used for partitioning
35  */
36 case class XZ2Scheme(attribute: String, index: Int, bits: Int) extends SpatialScheme {
37 
38   import FilterHelper.ff
39 
40   require(bits % 4 == 0, s"Bit precision must be a multiple of 4, but received $bits")
41 
42   private val xz2 = XZ2SFC
43 
44   // partition level derived from bits parameter
45   // each level adds 2 bits (4 quadrants)
46   private val partitionLevel = (bits / 2).toShort
47   // number of hex digits used to represent our z value - bits = (xz2.g - partitionLevel) * 2, then divide by 4 to get hex
48   override protected val digits: Int = xz2.hexDigits - ((xz2.g - partitionLevel) / 2)
49 
50   override val name: String = s"${XZ2Scheme.name}:attribute=$attribute:bits=$bits"
51 
52   override val column: String = ZValueField.xz2FieldName(ColumnName.encode(attribute))
53 
54   override def getPartition(feature: SimpleFeature): PartitionKey = {
55     val envelope = feature.getAttribute(index).asInstanceOf[Geometry].getEnvelopeInternal
56     PartitionKey(name, xz2.hexEncode(envelope.getMinX, envelope.getMinY, envelope.getMaxX, envelope.getMaxY).take(digits))
57   }
58 
59   override def getCoveringFilter(partition: PartitionKey): Filter = {
60     // TODO maybe we can improve this with *some* kind of bbox?
61     val hexPrefix = partition.value
62     val lower = hexPrefix.padTo(xz2.hexDigits, '0')
63     val upper = hexPrefix.padTo(xz2.hexDigits, 'f')
64     ff.between(ff.function(XZ2Function.FunctionName.getName), ff.literal(lower), ff.literal(upper))
65   }
66 
67   override def spec(b: PartitionSpec.Builder): PartitionSpec.Builder =
68     b.truncate(ZValueField.xz2FieldName(ColumnName.encode(attribute)), digits)
69 
70   override def getPartition(partition: StructLike, i: Int): PartitionKey = PartitionKey(name, partition.get(i, classOf[String]))
71 
72   override protected def hexRanges(bounds: Seq[(Double, Double, Double, Double)]): Seq[(String, String)] = {
73     val max = QueryProperties.ScanRangesTarget.toInt
74     xz2.ranges(bounds, max).map(r => (xz2.hexEncode(r.lower), xz2.hexEncode(r.upper)))
75   }
76 }
77 
78 object XZ2Scheme extends SpatialPartitionSchemeFactory[Geometry]("xz2") {
79   override def buildPartitionScheme(bits: Int, geom: String, geomIndex: Int): PartitionScheme =
80     XZ2Scheme(geom, geomIndex, bits)
