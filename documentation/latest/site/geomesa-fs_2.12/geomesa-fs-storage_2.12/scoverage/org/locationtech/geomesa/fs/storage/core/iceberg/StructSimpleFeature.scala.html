125   def apply(schema: SimpleFeatureIcebergSchema): StructSimpleFeature = {
126     var i = 0
127     val accessors = Array.ofDim[ColumnAccessor](schema.sft.getAttributeCount)
128     val cols = schema.schema.columns().asScala
129     val hasId = cols.headOption.exists(_.name() == SimpleFeatureSchema.FeatureIdField)
130     while (i < accessors.length) {
131       val descriptor = schema.sft.getDescriptor(i)
132       val col = ColumnName.encode(descriptor.getLocalName)
133       val offset = cols.indexWhere(_.name() == col)
134       accessors(i) = Converter(descriptor, col, schema) match {
135         case None => new DirectAccessor(offset)
136         case Some(c) => new ConverterAccessor(offset, c)
137       }
138       i += 1
139     }
140     val filePathAccessor = schema.schema.accessorForField(MetadataColumns.FILE_PATH.fieldId())
141     val rowPosAccessor = schema.schema.accessorForField(MetadataColumns.ROW_POSITION.fieldId())
142 
143     new StructSimpleFeature(schema.sft, hasId, accessors, filePathAccessor, rowPosAccessor)
144   }

163   private object Converter {
164     def apply(descriptor: AttributeDescriptor, col: String, schema: SimpleFeatureIcebergSchema): Option[Converter] = {
165       val types = ObjectType.selectType(descriptor)
166       if (types.head == ObjectType.GEOMETRY) {
167         val encoding = descriptor.getUserData.get(SimpleFeatureIcebergSchema.GeometryEncodingKey) match {
168           case e: String => GeometryEncoding(e)
169           case _ => GeometryEncoding.GeoParquetWkb
170         }
171         if (encoding != GeoParquetWkb) {
172           throw new UnsupportedOperationException(encoding.toString)
173         }
174         Some(FromWkbConverter)
175       } else if (types.last == ObjectType.JSON && descriptor.getJsonSchema().isDefined) {
176         Some(new StructJsonConverter(schema.schema.findType(col)))
177       } else if (types.head == ObjectType.LIST) {
178         primitive(types.last).map(new ListConverter(_))
179       } else if (types.head == ObjectType.MAP) {
180         val keyConverter = primitive(types(1))
181         val valueConverter = primitive(types(2))
182         (keyConverter, valueConverter) match {
183           case (None, None) => None
184           case (Some(k), None) => Some(new MapKeyConverter(k))
185           case (None, Some(v)) => Some(new MapValueConverter(v))
186           case (Some(k), Some(v)) => Some(new MapConverter(k, v))
187         }
188       } else {
189         primitive(types.head)
190       }
191     }
192 
193     private def primitive(types: ObjectType): Option[Converter] = types match {
194       case ObjectType.DATE  => Some(FromDateConverter)
195       case ObjectType.BYTES => Some(FromBytesConverter)
196       case _ => None
197     }
198   }

262   /**
263    * Converter for a structural-JSON attribute. Walks the materialized iceberg value (struct/list/map/leaf)
264    * against its iceberg type and rebuilds a compact JSON string, so the value round-trips as a JSON attribute.
265    */
266   private class StructJsonConverter(t: org.apache.iceberg.types.Type) extends Converter {
267     override def apply(value: AnyRef): AnyRef = StructuralJson.compact(toJson(t, value))
268 
269     private def toJson(t: org.apache.iceberg.types.Type, value: AnyRef): JsonElement = {
270       if (value == null) { JsonNull.INSTANCE } else {
271         t.typeId() match {
272           case TypeID.STRUCT =>
273             val obj = new JsonObject()
274             val fields = t.asStructType().fields()
275             val row = value.asInstanceOf[StructLike]
276             var i = 0
277             while (i < fields.size()) {
278               val field = fields.get(i)
279               val fieldValue = row.get(i, classOf[AnyRef])
280               if (fieldValue != null) {
281                 obj.add(field.name(), toJson(field.`type`(), fieldValue))
282               }
283               i += 1
284             }
285             obj
286 
287           case TypeID.LIST =>
288             val array = new JsonArray()
289             val elementType = t.asListType().elementType()
290             value.asInstanceOf[java.util.List[AnyRef]].forEach(v => array.add(toJson(elementType, v)))
291             array
292 
293           case TypeID.MAP =>
294             val obj = new JsonObject()
295             val valueType = t.asMapType().valueType()
296             value.asInstanceOf[java.util.Map[AnyRef, AnyRef]].forEach { (k, v) =>
297               obj.add(String.valueOf(k), toJson(valueType, v))
298             }
299             obj
300 
301           case TypeID.BOOLEAN => new JsonPrimitive(value.asInstanceOf[java.lang.Boolean])
302           case TypeID.INTEGER => new JsonPrimitive(value.asInstanceOf[java.lang.Integer])
303           case TypeID.LONG    => new JsonPrimitive(value.asInstanceOf[java.lang.Long])
304           case TypeID.FLOAT   => new JsonPrimitive(value.asInstanceOf[java.lang.Float])
305           case TypeID.DOUBLE  => new JsonPrimitive(value.asInstanceOf[java.lang.Double])
306           case TypeID.STRING  => new JsonPrimitive(value.toString)
307           case TypeID.DECIMAL => new JsonPrimitive(value.asInstanceOf[java.math.BigDecimal])
308           case TypeID.UUID    => new JsonPrimitive(value.asInstanceOf[UUID].toString)
309           case TypeID.DATE    => new JsonPrimitive(StructuralJson.dateToJson(value.asInstanceOf[LocalDate]))
310           case TypeID.TIME    => new JsonPrimitive(StructuralJson.timeToJson(value.asInstanceOf[LocalTime]))
311 
312           case TypeID.TIMESTAMP =>
313             if (t.asInstanceOf[TimestampType].shouldAdjustToUTC()) {
314               new JsonPrimitive(StructuralJson.timestampToJson(value.asInstanceOf[OffsetDateTime]))
315             } else {
316               new JsonPrimitive(StructuralJson.timestampToJson(value.asInstanceOf[LocalDateTime]))
317             }
318 
319           case TypeID.FIXED | TypeID.BINARY =>
320             new JsonPrimitive(StructuralJson.bytesToJson(value.asInstanceOf[ByteBuffer]))
321 
322           case id =>
323             throw new UnsupportedOperationException(s"No structural JSON mapping defined for iceberg type: $id")
324         }
325       }
326     }
327   }
