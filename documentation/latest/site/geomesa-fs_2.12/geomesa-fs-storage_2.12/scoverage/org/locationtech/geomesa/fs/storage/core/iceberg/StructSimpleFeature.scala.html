30 /**
31  * A simple feature implementation that wraps an iceberg record
32  *
33  * @param sft simple feature type
34  * @param fields converters for the raw record fields
35  */
36 class StructSimpleFeature(
37     sft: SimpleFeatureType,
38     hasIds: Boolean,
39     fields: Array[ColumnAccessor],
40     filePathAccessor: Accessor[StructLike],
41     rowPosAccessor: Accessor[StructLike],
42   ) extends AbstractMutableSimpleFeature(sft) {
43 
44   private var row: StructLike = _
45   private var userData: java.util.Map[AnyRef, AnyRef] = _
46 
47   private val values = Array.ofDim[AnyRef](fields.length)
48   private val visCol = if (hasIds) { 1 } else { 0 }
49 
50   def setRow(row: StructLike): StructSimpleFeature = {
51     this.row = row
52     this.id = null
53     this.userData = null
54     var i = 0
55     while (i < values.length) {

120   def apply(schema: SimpleFeatureIcebergSchema): StructSimpleFeature = {
121     var i = 0
122     val accessors = Array.ofDim[ColumnAccessor](schema.sft.getAttributeCount)
123     val cols = schema.schema.columns().asScala
124     val hasId = cols.headOption.exists(_.name() == SimpleFeatureSchema.FeatureIdField)
125     while (i < accessors.length) {
126       val descriptor = schema.sft.getDescriptor(i)
127       val converter = Converter(descriptor)
128       val col = ColumnName.encode(descriptor.getLocalName)
129       val offset = cols.indexWhere(_.name() == col)
130       accessors(i) = converter match {
131         case None => new DirectAccessor(offset)
132         case Some(c) => new ConverterAccessor(offset, c)
133       }
134       i += 1
135     }
136     val filePathAccessor = schema.schema.accessorForField(MetadataColumns.FILE_PATH.fieldId())
137     val rowPosAccessor = schema.schema.accessorForField(MetadataColumns.ROW_POSITION.fieldId())
138 
139     new StructSimpleFeature(schema.sft, hasId, accessors, filePathAccessor, rowPosAccessor)
140   }

142   private sealed trait ColumnAccessor extends (StructLike => AnyRef) {
143     def apply(row: StructLike): AnyRef
144   }
145 
146   private class DirectAccessor(i: Int) extends ColumnAccessor {
147     override def apply(row: StructLike): AnyRef = row.get(i, classOf[AnyRef])
148   }
149 
150   private class ConverterAccessor(i: Int, converter: Converter) extends ColumnAccessor {
151     override def apply(row: StructLike): AnyRef = {
152       val value = row.get(i, classOf[AnyRef])
153       if (value == null) { null } else { converter(value) }
154     }
155   }
156 
157   private sealed trait Converter extends (AnyRef => AnyRef)
158 
159   private object Converter {
160     def apply(descriptor: AttributeDescriptor): Option[Converter] = {
161       val types = ObjectType.selectType(descriptor)
162       if (types.head == ObjectType.GEOMETRY) {
163         val encoding = descriptor.getUserData.get(SimpleFeatureIcebergSchema.GeometryEncodingKey) match {
164           case e: String => GeometryEncoding(e)
165           case _ => GeometryEncoding.GeoParquetWkb
166         }
167         encoding match {
168           case GeoParquetWkb => Some(FromWkbConverter)
169           case _ => throw new UnsupportedOperationException(encoding.toString)
170         }
171       } else if (types.head == ObjectType.LIST) {
172         primitive(types.last).map(new ListConverter(_))
173       } else if (types.head == ObjectType.MAP) {
174         val keyConverter = primitive(types(1))
175         val valueConverter = primitive(types(2))
176         (keyConverter, valueConverter) match {
177           case (None, None) => None
178           case (Some(k), None) => Some(new MapKeyConverter(k))
179           case (None, Some(v)) => Some(new MapValueConverter(v))
180           case (Some(k), Some(v)) => Some(new MapConverter(k, v))
181         }
182       } else {
183         primitive(types.head)
184       }
185     }
186 
187     private def primitive(types: ObjectType): Option[Converter] = types match {
188       case ObjectType.DATE  => Some(FromDateConverter)
189       case ObjectType.BYTES => Some(FromBytesConverter)
190       case _ => None
191     }
192   }
193 
194   private object FromDateConverter extends Converter {
195     override def apply(value: AnyRef): AnyRef = Date.from(value.asInstanceOf[OffsetDateTime].toInstant)
196   }
197 
198   private object FromBytesConverter extends Converter {
199     override def apply(value: AnyRef): AnyRef = {
200       val buffer = value.asInstanceOf[ByteBuffer]
201       val pos = buffer.position()
202       val buf = Array.ofDim[Byte](buffer.remaining())
203       buffer.get(buf, 0, buf.length)
204       buffer.position(pos)
205       buf
206     }
207   }
208 
209   private class ListConverter(subtype: Converter) extends Converter {
210     override def apply(value: AnyRef): AnyRef = {
211       val list = value.asInstanceOf[java.util.List[AnyRef]]
212       val result = new java.util.ArrayList[AnyRef](list.size())
213       list.forEach(v => result.add(subtype(v)))
214       result
215     }
216   }
217 
218   private class MapConverter(keyType: Converter, valueType: Converter) extends Converter {
219     override def apply(value: AnyRef): AnyRef = {
220       val map = value.asInstanceOf[java.util.Map[AnyRef, AnyRef]]
221       val result = new java.util.HashMap[AnyRef, AnyRef](map.size())
222       map.forEach { case (k, v) => result.put(keyType(k), valueType(v)) }
223       result
224     }
225   }
226 
227   private class MapKeyConverter(keyType: Converter) extends Converter {
228     override def apply(value: AnyRef): AnyRef = {
229       val map = value.asInstanceOf[java.util.Map[AnyRef, AnyRef]]
230       val result = new java.util.HashMap[AnyRef, AnyRef](map.size())
231       map.forEach { case (k, v) => result.put(keyType(k), v) }
232       result
233     }
234   }
235 
236   private class MapValueConverter(valueType: Converter) extends Converter {
237     override def apply(value: AnyRef): AnyRef = {
238       val map = value.asInstanceOf[java.util.Map[AnyRef, AnyRef]]
239       val result = new java.util.HashMap[AnyRef, AnyRef](map.size())
240       map.forEach { case (k, v) => result.put(k, valueType(v)) }
241       result
242     }
243   }
244 
245   private object FromWkbConverter extends Converter {
246     override def apply(value: AnyRef): AnyRef = {
247       val buffer = value.asInstanceOf[ByteBuffer]
248       val pos = buffer.position()
249       val buf = Array.ofDim[Byte](buffer.remaining())
250       buffer.get(buf, 0, buf.length)
251       buffer.position(pos)
252       WKBUtils.read(buf)
253     }
254   }
