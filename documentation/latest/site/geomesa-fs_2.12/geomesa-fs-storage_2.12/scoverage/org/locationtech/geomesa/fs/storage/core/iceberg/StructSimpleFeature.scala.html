128   def apply(schema: SimpleFeatureIcebergSchema): StructSimpleFeature = {
129     var i = 0
130     val accessors = Array.ofDim[ColumnAccessor](schema.sft.getAttributeCount)
131     val cols = schema.schema.columns().asScala
132     val hasId = cols.headOption.exists(_.name() == SimpleFeatureSchema.FeatureIdField)
133     while (i < accessors.length) {
134       val descriptor = schema.sft.getDescriptor(i)
135       accessors(i) = if (descriptor.getLocalName.startsWith("$")) {
136         val path = JsonPathParser.parse(descriptor.getLocalName)
137         val col = ColumnName.encode(path.head.asInstanceOf[PathAttribute].name)
138         val offset = cols.indexWhere(_.name() == col)
139         val typed = cols(offset).`type`()
140         if (typed.isStructType) {
141           // note: we might have to extract only a subset of the struct if some fields were added for filtering
142           new ConverterAccessor(offset, new StructJsonPathConverter(typed, path.tail))
143         } else if (typed.isVariantType || typed.typeId() == TypeID.STRING) {
144           new VariantPathAccessor(offset, path.tail) // note: also works with string types
145         } else {
146           throw new IllegalArgumentException(s"Unexpected column for json path expression: ${cols(offset)}")
147         }
148       } else {
149         val col = ColumnName.encode(descriptor.getLocalName)
150         val offset = cols.indexWhere(_.name() == col)
151         Converter(descriptor, col, schema) match {
152           case None => new DirectAccessor(offset)
153           case Some(c) => new ConverterAccessor(offset, c)
154         }
155       }
156       i += 1
157     }
158     val filePathAccessor = schema.schema.accessorForField(MetadataColumns.FILE_PATH.fieldId())
159     val rowPosAccessor = schema.schema.accessorForField(MetadataColumns.ROW_POSITION.fieldId())
160 
161     new StructSimpleFeature(schema.sft, hasId, accessors, filePathAccessor, rowPosAccessor)
162   }

186   private object Converter {
187     def apply(descriptor: AttributeDescriptor, col: String, schema: SimpleFeatureIcebergSchema): Option[Converter] = {
188       val types = ObjectType.selectType(descriptor)
189       if (types.head == ObjectType.GEOMETRY) {
190         val encoding = descriptor.getEncoding.fold[GeometryEncoding](GeoParquetWkb)(GeometryEncoding.apply)
191         if (encoding != GeoParquetWkb) {
192           throw new UnsupportedOperationException(encoding.toString)
193         }
194         Some(FromWkbConverter)
195       } else if (types.last == ObjectType.JSON && descriptor.getJsonSchema().isDefined) {
196         Some(new StructJsonConverter(schema.schema.findType(col)))
197       } else if (types.head == ObjectType.LIST) {
198         primitive(types.last).map(new ListConverter(_))
199       } else if (types.head == ObjectType.MAP) {
200         val keyConverter = primitive(types(1))
201         val valueConverter = primitive(types(2))
202         (keyConverter, valueConverter) match {
203           case (None, None) => None
204           case (Some(k), None) => Some(new MapKeyConverter(k))
205           case (None, Some(v)) => Some(new MapValueConverter(v))
206           case (Some(k), Some(v)) => Some(new MapConverter(k, v))
207         }
208       } else {
209         primitive(types.head)
210       }
211     }

214       case ObjectType.DATE  => Some(FromDateConverter)
215       case ObjectType.BYTES => Some(FromBytesConverter)

282   /**
283    * Converter for a structural-JSON attribute. Walks the materialized iceberg value (struct/list/map/leaf)
284    * against its iceberg type and rebuilds a compact JSON string, so the value round-trips as a JSON attribute.
285    */
286   private class StructJsonConverter(t: org.apache.iceberg.types.Type) extends Converter {
287 
288     override def apply(value: AnyRef): AnyRef = StructuralJson.compact(toJson(t, value))
289 
290     protected def toJson(t: org.apache.iceberg.types.Type, value: AnyRef): JsonElement = {
291       if (value == null) {
292         return JsonNull.INSTANCE
293       }
294       if (t.typeId() == TypeID.STRUCT) {
295         val obj = new JsonObject()
296         val fields = t.asStructType().fields()
297         val row = value.asInstanceOf[StructLike]
298         var i = 0
299         while (i < fields.size()) {
300           val field = fields.get(i)
301             val fieldValue = row.get(i, classOf[AnyRef])
302             if (fieldValue != null) {
303               obj.add(field.name(), toJson(field.`type`(), fieldValue))
304             }
305           i += 1
306         }
307         if (obj.isEmpty) { JsonNull.INSTANCE } else { obj }
308       } else if (t.typeId() == TypeID.MAP) {
309         val obj = new JsonObject()
310         val valueType = t.asMapType().valueType()
311         value.asInstanceOf[java.util.Map[AnyRef, AnyRef]].forEach { (k, v) =>
312           val key = String.valueOf(k)
313           obj.add(key, toJson(valueType, v))
314         }
315         obj
316       } else if (t.typeId() == TypeID.LIST) {
317         val array = new JsonArray()
318         val elementType = t.asListType().elementType()
319         value.asInstanceOf[java.util.List[AnyRef]].forEach(v => array.add(toJson(elementType, v)))
320         array
321       } else {
322         t.typeId() match {
323           case TypeID.BOOLEAN => new JsonPrimitive(value.asInstanceOf[java.lang.Boolean])
324           case TypeID.INTEGER => new JsonPrimitive(value.asInstanceOf[java.lang.Integer])
325           case TypeID.LONG    => new JsonPrimitive(value.asInstanceOf[java.lang.Long])
326           case TypeID.FLOAT   => new JsonPrimitive(value.asInstanceOf[java.lang.Float])
327           case TypeID.DOUBLE  => new JsonPrimitive(value.asInstanceOf[java.lang.Double])
328           case TypeID.STRING  => new JsonPrimitive(value.toString)
329           case TypeID.DECIMAL => new JsonPrimitive(value.asInstanceOf[java.math.BigDecimal])
330           case TypeID.UUID    => new JsonPrimitive(value.asInstanceOf[UUID].toString)
331           case TypeID.DATE    => new JsonPrimitive(StructuralJson.dateToJson(value.asInstanceOf[LocalDate]))
332           case TypeID.TIME    => new JsonPrimitive(StructuralJson.timeToJson(value.asInstanceOf[LocalTime]))
333 
334           case TypeID.TIMESTAMP =>
335             if (t.asInstanceOf[TimestampType].shouldAdjustToUTC()) {
336               new JsonPrimitive(StructuralJson.timestampToJson(value.asInstanceOf[OffsetDateTime]))
337             } else {
338               new JsonPrimitive(StructuralJson.timestampToJson(value.asInstanceOf[LocalDateTime]))
339             }
340 
341           case TypeID.FIXED | TypeID.BINARY =>
342             new JsonPrimitive(StructuralJson.bytesToJson(value.asInstanceOf[ByteBuffer]))
343 
344           case id =>
345             throw new UnsupportedOperationException(s"No structural JSON mapping defined for iceberg type: $id")
346         }
347       }
348     }
349   }

351   /**
352    * Converter for a structural-JSON attribute that only returns values matching the json path
353    */
354   private class StructJsonPathConverter(t: org.apache.iceberg.types.Type, path: JsonPath) extends StructJsonConverter(t) {
355 
356     private val simplePath =
357       Option(path).filter(p => p.function.isEmpty && p.elements.forall(_.isInstanceOf[PathAttribute]))
358         .map(_.elements.map(_.asInstanceOf[PathAttribute].name))
359 
360     override def apply(value: AnyRef): AnyRef = {
361       simplePath match {
362         case None => JsonPathPropertyAccessor.evaluateJsonPath(super.apply(value).asInstanceOf[String], path)
363         case Some(p) => StructuralJson.compact(toJson(t, value, p))
364       }
365     }
366 
367     @tailrec
368     private def toJson(t: org.apache.iceberg.types.Type, value: AnyRef, path: Seq[String]): JsonElement = {
369       if (path.isEmpty) {
370         toJson(t, value)
371       } else if (value == null) {
372         JsonNull.INSTANCE
373       } else if (t.typeId() == TypeID.STRUCT) {
374         val fields = t.asStructType().fields()
375         val i = fields.asScala.indexWhere(_.name() == path.head)
376         if (i == -1) {
377           JsonNull.INSTANCE
378         } else {
379           toJson(fields.get(i).`type`(), value.asInstanceOf[StructLike].get(i, classOf[AnyRef]), path.tail)
380         }
381       } else if (t.typeId() == TypeID.MAP) {
382         toJson(t.asMapType().valueType(), value.asInstanceOf[java.util.Map[AnyRef, AnyRef]].get(path.head), path.tail)
383       } else {
384         JsonNull.INSTANCE
385       }
386     }
