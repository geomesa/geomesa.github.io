128   def apply(schema: SimpleFeatureIcebergSchema): StructSimpleFeature = {
129     var i = 0
130     val accessors = Array.ofDim[ColumnAccessor](schema.sft.getAttributeCount)
131     val cols = schema.schema.columns().asScala
132     val hasId = cols.headOption.exists(_.name() == SimpleFeatureSchema.FeatureIdField)
133     while (i < accessors.length) {
134       val descriptor = schema.sft.getDescriptor(i)
135       accessors(i) = if (descriptor.getLocalName.startsWith("$")) {
136         val path = JsonPathParser.parse(descriptor.getLocalName)
137         val col = ColumnName.encode(path.head.asInstanceOf[PathAttribute].name)
138         val offset = cols.indexWhere(_.name() == col)
139         val typed = cols(offset).`type`()
140         if (typed.isStructType) {
141           // note: we might have to extract only a subset of the struct if some fields were added for filtering
142           new ConverterAccessor(offset, new StructJsonPathConverter(typed, path.tail))
143         } else if (typed.isVariantType || typed.typeId() == TypeID.STRING) {
144           new VariantPathAccessor(offset, path.tail) // note: also works with string types
145         } else {
146           throw new IllegalArgumentException(s"Unexpected column for json path expression: ${cols(offset)}")
147         }
148       } else {
149         val col = ColumnName.encode(descriptor.getLocalName)
150         val offset = cols.indexWhere(_.name() == col)
151         Converter(descriptor, col, schema) match {
152           case None => new DirectAccessor(offset)
153           case Some(c) => new ConverterAccessor(offset, c)
154         }
155       }
156       i += 1
157     }
158     val filePathAccessor = schema.schema.accessorForField(MetadataColumns.FILE_PATH.fieldId())
159     val rowPosAccessor = schema.schema.accessorForField(MetadataColumns.ROW_POSITION.fieldId())
160 
161     new StructSimpleFeature(schema.sft, hasId, accessors, filePathAccessor, rowPosAccessor)
162   }

186   private object Converter {
187     def apply(descriptor: AttributeDescriptor, col: String, schema: SimpleFeatureIcebergSchema): Option[Converter] = {
188       val types = ObjectType.selectType(descriptor)
189       if (types.head == ObjectType.GEOMETRY) {
190         val encoding = descriptor.getUserData.get(SimpleFeatureIcebergSchema.GeometryEncodingKey) match {
191           case e: String => GeometryEncoding(e)
192           case _ => GeometryEncoding.GeoParquetWkb
193         }
194         if (encoding != GeoParquetWkb) {
195           throw new UnsupportedOperationException(encoding.toString)
196         }
197         Some(FromWkbConverter)
198       } else if (types.last == ObjectType.JSON && descriptor.getJsonSchema().isDefined) {
199         Some(new StructJsonConverter(schema.schema.findType(col)))
200       } else if (types.head == ObjectType.LIST) {
201         primitive(types.last).map(new ListConverter(_))
202       } else if (types.head == ObjectType.MAP) {
203         val keyConverter = primitive(types(1))
204         val valueConverter = primitive(types(2))
205         (keyConverter, valueConverter) match {
206           case (None, None) => None
207           case (Some(k), None) => Some(new MapKeyConverter(k))
208           case (None, Some(v)) => Some(new MapValueConverter(v))
209           case (Some(k), Some(v)) => Some(new MapConverter(k, v))
210         }
211       } else {
212         primitive(types.head)
213       }
214     }
215 
216     private def primitive(types: ObjectType): Option[Converter] = types match {
217       case ObjectType.DATE  => Some(FromDateConverter)
218       case ObjectType.BYTES => Some(FromBytesConverter)
219       case _ => None
220     }
221   }

285   /**
286    * Converter for a structural-JSON attribute. Walks the materialized iceberg value (struct/list/map/leaf)
287    * against its iceberg type and rebuilds a compact JSON string, so the value round-trips as a JSON attribute.
288    */
289   private class StructJsonConverter(t: org.apache.iceberg.types.Type) extends Converter {
290 
291     override def apply(value: AnyRef): AnyRef = StructuralJson.compact(toJson(t, value))
292 
293     protected def toJson(t: org.apache.iceberg.types.Type, value: AnyRef): JsonElement = {
294       if (value == null) {
295         return JsonNull.INSTANCE
296       }
297       if (t.typeId() == TypeID.STRUCT) {
298         val obj = new JsonObject()
299         val fields = t.asStructType().fields()
300         val row = value.asInstanceOf[StructLike]
301         var i = 0
302         while (i < fields.size()) {
303           val field = fields.get(i)
304             val fieldValue = row.get(i, classOf[AnyRef])
305             if (fieldValue != null) {
306               obj.add(field.name(), toJson(field.`type`(), fieldValue))
307             }
308           i += 1
309         }
310         if (obj.isEmpty) { JsonNull.INSTANCE } else { obj }
311       } else if (t.typeId() == TypeID.MAP) {
312         val obj = new JsonObject()
313         val valueType = t.asMapType().valueType()
314         value.asInstanceOf[java.util.Map[AnyRef, AnyRef]].forEach { (k, v) =>
315           val key = String.valueOf(k)
316           obj.add(key, toJson(valueType, v))
317         }
318         obj
319       } else if (t.typeId() == TypeID.LIST) {
320         val array = new JsonArray()
321         val elementType = t.asListType().elementType()
322         value.asInstanceOf[java.util.List[AnyRef]].forEach(v => array.add(toJson(elementType, v)))
323         array
324       } else {
325         t.typeId() match {
326           case TypeID.BOOLEAN => new JsonPrimitive(value.asInstanceOf[java.lang.Boolean])
327           case TypeID.INTEGER => new JsonPrimitive(value.asInstanceOf[java.lang.Integer])
328           case TypeID.LONG    => new JsonPrimitive(value.asInstanceOf[java.lang.Long])
329           case TypeID.FLOAT   => new JsonPrimitive(value.asInstanceOf[java.lang.Float])
330           case TypeID.DOUBLE  => new JsonPrimitive(value.asInstanceOf[java.lang.Double])
331           case TypeID.STRING  => new JsonPrimitive(value.toString)
332           case TypeID.DECIMAL => new JsonPrimitive(value.asInstanceOf[java.math.BigDecimal])
333           case TypeID.UUID    => new JsonPrimitive(value.asInstanceOf[UUID].toString)
334           case TypeID.DATE    => new JsonPrimitive(StructuralJson.dateToJson(value.asInstanceOf[LocalDate]))
335           case TypeID.TIME    => new JsonPrimitive(StructuralJson.timeToJson(value.asInstanceOf[LocalTime]))
336 
337           case TypeID.TIMESTAMP =>
338             if (t.asInstanceOf[TimestampType].shouldAdjustToUTC()) {
339               new JsonPrimitive(StructuralJson.timestampToJson(value.asInstanceOf[OffsetDateTime]))
340             } else {
341               new JsonPrimitive(StructuralJson.timestampToJson(value.asInstanceOf[LocalDateTime]))
342             }
343 
344           case TypeID.FIXED | TypeID.BINARY =>
345             new JsonPrimitive(StructuralJson.bytesToJson(value.asInstanceOf[ByteBuffer]))
346 
347           case id =>
348             throw new UnsupportedOperationException(s"No structural JSON mapping defined for iceberg type: $id")
349         }
350       }
351     }
352   }

354   /**
355    * Converter for a structural-JSON attribute that only returns values matching the json path
356    */
357   private class StructJsonPathConverter(t: org.apache.iceberg.types.Type, path: JsonPath) extends StructJsonConverter(t) {
358 
359     private val simplePath =
360       Option(path).filter(p => p.function.isEmpty && p.elements.forall(_.isInstanceOf[PathAttribute]))
361         .map(_.elements.map(_.asInstanceOf[PathAttribute].name))
362 
363     override def apply(value: AnyRef): AnyRef = {
364       simplePath match {
365         case None => JsonPathPropertyAccessor.evaluateJsonPath(super.apply(value).asInstanceOf[String], path)
366         case Some(p) => StructuralJson.compact(toJson(t, value, p))
367       }
368     }
369 
370     @tailrec
371     private def toJson(t: org.apache.iceberg.types.Type, value: AnyRef, path: Seq[String]): JsonElement = {
372       if (path.isEmpty) {
373         toJson(t, value)
374       } else if (value == null) {
375         JsonNull.INSTANCE
376       } else if (t.typeId() == TypeID.STRUCT) {
377         val fields = t.asStructType().fields()
378         val i = fields.asScala.indexWhere(_.name() == path.head)
379         if (i == -1) {
380           JsonNull.INSTANCE
381         } else {
382           toJson(fields.get(i).`type`(), value.asInstanceOf[StructLike].get(i, classOf[AnyRef]), path.tail)
383         }
384       } else if (t.typeId() == TypeID.MAP) {
385         toJson(t.asMapType().valueType(), value.asInstanceOf[java.util.Map[AnyRef, AnyRef]].get(path.head), path.tail)
386       } else {
387         JsonNull.INSTANCE
388       }
389     }
