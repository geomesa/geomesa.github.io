28 /**
29  * A simple feature implementation that wraps an iceberg record
30  *
31  * @param sft simple feature type
32  * @param fields converters for the raw record fields
33  */
34 class StructSimpleFeature(
35     sft: SimpleFeatureType,
36     hasIds: Boolean,
37     fields: Array[ColumnAccessor],
38     filePathAccessor: Accessor[StructLike],
39     rowPosAccessor: Accessor[StructLike],
40   ) extends AbstractMutableSimpleFeature(sft) {
41 
42   private var row: StructLike = _
43   private var userData: java.util.Map[AnyRef, AnyRef] = _
44 
45   private val values = Array.ofDim[AnyRef](fields.length)
46   private val visCol = if (hasIds) { 1 } else { 0 }
47 
48   def setRow(row: StructLike): StructSimpleFeature = {
49     this.row = row
50     this.id = null
51     this.userData = null
52     var i = 0
53     while (i < values.length) {
54       values(i) = null
55       i += 1
56     }
57     this
58   }
59 
60   /**
61    * Gets the full file path that this feature came from. Note - this requires the path to be selected on read, otherwise
62    * the behavior of this method is not defined
63    *
64    * @return
65    */
66   def getFilePath: String = filePathAccessor.get(row).asInstanceOf[String]
67 
68   /**
69    * Gets the row number in the file that this feature came from. Note - this requires the row to be selected on read, otherwise
70    * the behavior of this method is not defined
71    *
72    * @return
73    */
74   def getRowPosition: Long = rowPosAccessor.get(row).asInstanceOf[java.lang.Long]
75 
76   override def getIdentifier: FeatureId = { getID; super.getIdentifier }
77 
78   override def getID: String = {
79     if (id == null) {
80       if (hasIds) {
81         id = row.get(0, classOf[String])
82       } else {
83         id = java.lang.Long.toString(StructSimpleFeature.IdCounter.getAndIncrement())
84       }
85     }
86     id
87   }
88 
89   override def setAttributeNoConvert(index: Int, value: AnyRef): Unit = values(index) = value
90 
91   override def getAttribute(index: Int): AnyRef = {
92     var cached = values(index)
93     if (cached == null) {
94       cached = fields(index).apply(row)
95       values(index) = cached
96     }
97     cached
98   }
99 
100   override def getUserData: java.util.Map[AnyRef, AnyRef] = {
101     if (userData == null) {
102       userData = new java.util.HashMap(1)
103       val visibility = row.get(visCol, classOf[String])
104       if (visibility != null) {
105         userData.put(SecurityUtils.FEATURE_VISIBILITY, visibility)
106       }
107     }
108     userData
109   }
110 }

118   def apply(schema: SimpleFeatureIcebergSchema): StructSimpleFeature = {
119     var i = 0
120     val accessors = Array.ofDim[ColumnAccessor](schema.sft.getAttributeCount)
121     val cols = schema.schema.columns().asScala
122     val hasId = cols.headOption.exists(_.name() == SimpleFeatureSchema.FeatureIdField)
123     while (i < accessors.length) {
124       val descriptor = schema.sft.getDescriptor(i)
125       val types = ObjectType.selectType(descriptor)
126       val converter = Converter(types, schema)
127       val col = ColumnName.encode(descriptor.getLocalName)
128       val offset = cols.indexWhere(_.name() == col)
129       accessors(i) = converter match {
130         case None => new DirectAccessor(offset)
131         case Some(c) => new ConverterAccessor(offset, c)
132       }
133       i += 1
134     }
135     val filePathAccessor = schema.schema.accessorForField(MetadataColumns.FILE_PATH.fieldId())
136     val rowPosAccessor = schema.schema.accessorForField(MetadataColumns.ROW_POSITION.fieldId())
137 
138     new StructSimpleFeature(schema.sft, hasId, accessors, filePathAccessor, rowPosAccessor)
139   }

141   private sealed trait ColumnAccessor extends (StructLike => AnyRef) {
142     def apply(row: StructLike): AnyRef
143   }
144 
145   private class DirectAccessor(i: Int) extends ColumnAccessor {
146     override def apply(row: StructLike): AnyRef = row.get(i, classOf[AnyRef])
147   }
148 
149   private class ConverterAccessor(i: Int, converter: Converter) extends ColumnAccessor {
150     override def apply(row: StructLike): AnyRef = {
151       val value = row.get(i, classOf[AnyRef])
152       if (value == null) { null } else { converter(value) }
153     }
154   }
155 
156   private sealed trait Converter extends (AnyRef => AnyRef)
157 
158   private object Converter {
159     def apply(types: Seq[ObjectType], schema: SimpleFeatureIcebergSchema): Option[Converter] = types.head match {
160       case ObjectType.GEOMETRY if schema.geometries == GeoParquetWkb => Some(FromWkbConverter)
161       case ObjectType.GEOMETRY if schema.geometries == GeoParquetNative => throw new UnsupportedOperationException()
162       case ObjectType.GEOMETRY => throw new UnsupportedOperationException("An implementation is missing")
163       case ObjectType.DATE     => Some(FromDateConverter)
164       case ObjectType.BYTES    => Some(FromBytesConverter)
165       case ObjectType.LIST     => ListConverter(types.last, schema)
166       case ObjectType.MAP      => MapConverter(types(1), types(2), schema)
167       case _ => None
168     }
169   }
170 
171   private object FromDateConverter extends Converter {
172     override def apply(value: AnyRef): AnyRef = Date.from(value.asInstanceOf[OffsetDateTime].toInstant)
173   }
174 
175   private object FromBytesConverter extends Converter {
176     override def apply(value: AnyRef): AnyRef = {
177       val buffer = value.asInstanceOf[ByteBuffer]
178       val pos = buffer.position()
179       val buf = Array.ofDim[Byte](buffer.remaining())
180       buffer.get(buf, 0, buf.length)
181       buffer.position(pos)
182       buf
183     }
184   }
185 
186   private object ListConverter extends Converter {
187     def apply(subtype: ObjectType, schema: SimpleFeatureIcebergSchema): Option[Converter] =
188       Converter(Seq(subtype), schema).map(new ListConverter(_))
189 
190     override def apply(value: AnyRef): AnyRef = java.util.List.copyOf(value.asInstanceOf[java.util.List[AnyRef]])
191   }
192 
193   private class ListConverter(subtype: Converter) extends Converter {
194     override def apply(value: AnyRef): AnyRef = {
195       val list = value.asInstanceOf[java.util.List[AnyRef]]
196       val result = new java.util.ArrayList[AnyRef](list.size())
197       list.forEach(v => result.add(subtype(v)))
198       result
199     }
200   }
201 
202   private object MapConverter extends Converter {
203     def apply(keyType: ObjectType, valueType: ObjectType, schema: SimpleFeatureIcebergSchema): Option[Converter] = {
204       val keyConverter = Converter(Seq(keyType), schema)
205       val valueConverter = Converter(Seq(valueType), schema)
206       (keyConverter, valueConverter) match {
207         case (None, None) => None
208         case (Some(k), None) => Some(new MapKeyConverter(k))
209         case (None, Some(v)) => Some(new MapValueConverter(v))
210         case (Some(k), Some(v)) => Some(new MapConverter(k, v))
211       }
212     }
213 
214     override def apply(value: AnyRef): AnyRef =
215       java.util.Map.copyOf[AnyRef, AnyRef](value.asInstanceOf[java.util.Map[AnyRef, AnyRef]])
216   }
217 
218   private class MapConverter(keyType: Converter, valueType: Converter) extends Converter {
219     override def apply(value: AnyRef): AnyRef = {

222       map.forEach { case (k, v) => result.put(keyType(k), valueType(v)) }
223       result
224     }
225   }

243   }
