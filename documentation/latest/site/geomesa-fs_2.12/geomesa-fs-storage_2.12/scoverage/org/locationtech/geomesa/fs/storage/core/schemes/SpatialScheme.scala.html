23   /**
24    * Number of digits in the hex-encoded space-filling curve values
25    *
26    * @return
27    */
28   protected def digits: Int
29 
30   /**
31    * Gets the full (not truncated) hex-encoded ranges for a query
32    *
33    * @param bounds query bounds, in the form (xmin, ymin, xmax, ymax)
34    * @return
35    */
36   protected def hexRanges(bounds: Seq[(Double, Double, Double, Double)]): Seq[(String, String)]
37 
38   /**
39    * Gets an expression that will match the specified bounds
40    *
41    * @param bounds spatial bounds, in the form (xmin, ymin, xmax, ymax). Multiple bounds are logically OR'd together
42    * @return
43    */
44   def getCoveringExpression(bounds: Seq[(Double, Double, Double, Double)]): Expression = {
45     // truncate and merge any overlapping ranges
46     val allRanges = hexRanges(bounds).map(PartitionRange.apply).distinct.sortBy(_.lower)
47 
48     val inValues = new java.util.ArrayList[String]()
49     var expression: Expression = Expressions.alwaysFalse()
50 
51     def addPartitionRange(r: PartitionRange): Unit = {
52       if (r.lower == r.upper) {
53         inValues.add(r.lower)
54       } else {
55         // TODO will this work with schema migration?
56         val gte = Expressions.greaterThanOrEqual(Expressions.truncate[String](column, digits), r.lower)
57         val lte = Expressions.lessThanOrEqual(Expressions.truncate[String](column, digits), r.upper)
58         expression = Expressions.or(expression, Expressions.and(gte, lte))
59       }
60     }
61 
62     val last = allRanges.reduce { (left, right) =>
63       left.merge(right) match {
64         case None => addPartitionRange(left); right
65         case Some(merged) => merged
66       }
67     }
68     addPartitionRange(last)
69 
70     if (!inValues.isEmpty) {
71       expression = Expressions.or(expression, Expressions.in(Expressions.truncate[String](column, digits), inValues))
72     }
73 
74     expression
75   }
76 
77   override def getCoveringExpression(partition: PartitionKey): Expression =
78     Expressions.equal[String](Expressions.truncate[String](column, digits), partition.value)
79 
80   override def partitions: Seq[PartitionKey] = {
81     val keys = (1 to digits).foldLeft(Seq("")) { (accumulated, _) =>
82       for {
83         str <- accumulated
84         char <- SpatialScheme.HexChars
85       } yield {
86         str + char
87       }
88     }
89     keys.map(PartitionKey(name, _))
90   }
91 
92   override def spec(b: PartitionSpec.Builder): PartitionSpec.Builder = b.truncate(column, digits)
93 
94   /**
95    * Ranged bounds, used for filtering on partitions
96    *
97    * @param lower lower bound, inclusive
98    * @param upper upper bound, exclusive
99    */
100   private case class PartitionRange(lower: String, upper: String) {
101 
102     /**
103      * Attempt to merge two bounds. Only overlapping bounds will result in a successful merge. Trying to merge
104      * bounds from a different partition scheme is a logical error.
105      *
106      * @param other bounds to merge
107      * @return
108      */
109     def merge(other: PartitionRange): Option[PartitionRange] = {
110       if (lower <= other.lower) {
111         if (upper >= other.upper) {
112           Some(this)
113         } else if (upper >= other.lower) {
114           Some(PartitionRange(lower, other.upper))
115         } else {
116           None
117         }
118       } else if (lower > other.upper) {
119         None
120       } else if (upper >= other.upper) {
121         Some(PartitionRange(other.lower, upper))
122       } else {
123         Some(other)
124       }
125     }
126   }
127 
128   private object PartitionRange {
129     def apply(bounds: (String, String)): PartitionRange = PartitionRange(bounds._1.take(digits), bounds._2.take(digits))
130   }

144       val opts = SchemeOpts(scheme)

148         val geom = opts.getSingle("attribute").orElse(Option(sft.getGeomField)).orNull
149         require(geom != null, s"Spatial schemes requires an attribute to be specified with 'attribute=<attribute>'")
150         val index = attributeIndex(sft, geom, Some(implicitly[ClassTag[T]].runtimeClass))
151         Some(buildPartitionScheme(resolution, geom, index))
152       }
153 
154       if (opts.name == this.name) {
155         val bits = opts.getSingle("bits").map(_.toShort).getOrElse {
156           throw new IllegalArgumentException(s"Spatial schemes requires a resolution to be specified with 'bits=<resolution>'")
157         }
158         build(bits)
159       } else if (matcher.matches()) {
160         build(matcher.group(1).toShort)
161       } else {
162         None
163       }
164     }
