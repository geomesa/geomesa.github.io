22   /**
23    * Number of digits in the hex-encoded space-filling curve values
24    *
25    * @return
26    */
27   protected def digits: Int
28 
29   /**
30    * Gets the full (not truncated) hex-encoded ranges for a query
31    *
32    * @param bounds query bounds, in the form (xmin, ymin, xmax, ymax)
33    * @return
34    */
35   protected def hexRanges(bounds: Seq[(Double, Double, Double, Double)]): Seq[(String, String)]
36 
37   /**
38    * Gets an expression that will match the specified bounds
39    *
40    * @param bounds spatial bounds, in the form (xmin, ymin, xmax, ymax). Multiple bounds are logically OR'd together
41    * @return
42    */
43   def getCoveringExpression(bounds: Seq[(Double, Double, Double, Double)]): Expression = {
44     // truncate and merge any overlapping ranges
45     val allRanges = hexRanges(bounds).map(PartitionRange.apply).distinct.sortBy(_.lower)
46 
47     val inValues = new java.util.ArrayList[String]()
48     var expression: Expression = Expressions.alwaysFalse()
49 
50     def addPartitionRange(r: PartitionRange): Unit = {
51       if (r.lower == r.upper) {
52         inValues.add(r.lower)
53       } else {
54         val gte = Expressions.greaterThanOrEqual(Expressions.truncate[String](column, digits), r.lower)
55         val lte = Expressions.lessThanOrEqual(Expressions.truncate[String](column, digits), r.upper)
56         expression = Expressions.or(expression, Expressions.and(gte, lte))
57       }
58     }
59 
60     val last = allRanges.reduce { (left, right) =>
61       left.merge(right) match {
62         case None => addPartitionRange(left); right
63         case Some(merged) => merged
64       }
65     }
66     addPartitionRange(last)
67 
68     if (!inValues.isEmpty) {
69       expression = Expressions.or(expression, Expressions.in(Expressions.truncate[String](column, digits), inValues))
70     }
71 
72     expression
73   }
74 
75   override def getCoveringExpression(partition: PartitionKey): Expression =
76     Expressions.equal[String](Expressions.truncate[String](column, digits), partition.value)
77 
78   override def partitions: Seq[PartitionKey] = {
79     val keys = (1 to digits).foldLeft(Seq("")) { (accumulated, _) =>
80       for {
81         str <- accumulated
82         char <- SpatialScheme.HexChars
83       } yield {
84         str + char
85       }
86     }
87     keys.map(PartitionKey(name, _))
88   }
89 
90   /**
91    * Ranged bounds, used for filtering on partitions
92    *
93    * @param lower lower bound, inclusive
94    * @param upper upper bound, exclusive
95    */
96   private case class PartitionRange(lower: String, upper: String) {
97 
98     /**
99      * Attempt to merge two bounds. Only overlapping bounds will result in a successful merge. Trying to merge
100      * bounds from a different partition scheme is a logical error.
101      *
102      * @param other bounds to merge
103      * @return
104      */
105     def merge(other: PartitionRange): Option[PartitionRange] = {
106       if (lower <= other.lower) {
107         if (upper >= other.upper) {
108           Some(this)
109         } else if (upper >= other.lower) {
110           Some(PartitionRange(lower, other.upper))
111         } else {
112           None
113         }
114       } else if (lower > other.upper) {
115         None
116       } else if (upper >= other.upper) {
117         Some(PartitionRange(other.lower, upper))
118       } else {
119         Some(other)
120       }
121     }
122   }
123 
124   private object PartitionRange {
125     def apply(bounds: (String, String)): PartitionRange = PartitionRange(bounds._1.take(digits), bounds._2.take(digits))
126   }

153         }
154         build(bits)
155       } else if (matcher.matches()) {
156         build(matcher.group(1).toShort)
157       } else {
158         None
159       }
160     }
