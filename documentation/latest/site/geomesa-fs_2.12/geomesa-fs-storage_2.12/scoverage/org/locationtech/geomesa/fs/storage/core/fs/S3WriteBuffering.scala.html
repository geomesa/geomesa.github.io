40   /**
41    * Disk write buffering
42    *
43    * @param bufferSize buffer size, in bytes
44    * @param dir tmp dir used for buffering
45    */
46   class DiskBuffering(bufferSize: Int, dir: File) extends S3WriteBuffering {
47 
48     override def write(fs: S3ObjectStore, bucket: String, key: String, overwrite: Boolean): OutputStream =
49       new DiskOutputStream(fs, bucket, key, overwrite)
50 
51     private class DiskOutputStream(
52         fs: S3ObjectStore,
53         bucket: String,
54         key: String,
55         overwrite: Boolean,
56         uid: String = UUID.randomUUID().toString.substring(0, 8),
57         private var i: Int = 0
58       ) extends ChunkedOutputStream(fs, bucket, key, bufferSize, overwrite) {
59       override protected def newChunk(): Chunk = {
60         val path = s"${bucket.replaceAll("[^a-zA-Z0-9_-]", "_")}/${key.replaceAll("[^a-zA-Z0-9_-]", "_")}/part_${uid}_$i"
61         val file = new File(dir, path)
62         file.getParentFile.mkdirs()
63         i += 1
64         new DiskChunk(file)
65       }
66     }
67 
68     private class DiskChunk(file: File) extends Chunk(new BufferedOutputStream(new FileOutputStream(file))) {
69       override def read: AsyncRequestBody = AsyncRequestBody.fromFile(file)
70       override def cleanup(): Unit = file.delete()
71     }
72   }

84     private class MemoryOutputStream(fs: S3ObjectStore, bucket: String, key: String, overwrite: Boolean)
85         extends ChunkedOutputStream(fs, bucket, key, bufferSize, overwrite) {
86       override protected def newChunk(): Chunk = new MemoryChunk(ByteBuffer.allocateDirect(bufferSize))

89     private class MemoryChunk(buf: ByteBuffer) extends Chunk(new ByteBufferOutputStream(buf)) {
90       override def read: AsyncRequestBody = {
91         buf.rewind()
92         AsyncRequestBody.fromRemainingByteBuffersUnsafe(buf)
93       }
94       override def cleanup(): Unit = {}
95     }

97     private class ByteBufferOutputStream(buf: ByteBuffer) extends OutputStream {
98       override def write(data: Int): Unit = buf.put(data.toByte)
99       override def write(data: Array[Byte]): Unit = buf.put(data)
100       override def write(data: Array[Byte], off: Int, len: Int): Unit = buf.put(data, off, len)
101       override def flush(): Unit = {}
102       override def close(): Unit = {}
103     }
104   }

106   private abstract class ChunkedOutputStream(fs: S3ObjectStore, bucket: String, key: String, bufferSize: Int, overwrite: Boolean)
107       extends OutputStream with LazyLogging {
108 
109     private var currentChunk = newChunk()
110     private var currentChunkSize = 0
111     private var uploadId: String = _
112     private val uploads = ArrayBuffer.empty[CompletableFuture[CompletedPart]]
113 
114     override def write(data: Int): Unit = {
115       currentChunk.os.write(data)
116       currentChunkSize += 1
117       if (currentChunkSize >= bufferSize) {
118         closeChunk()
119       }
120     }
121 
122     override def write(data: Array[Byte]): Unit = {
123       currentChunk.os.write(data)
124       currentChunkSize += data.length
125       if (currentChunkSize >= bufferSize) {
126         closeChunk()
127       }
128     }
129 
130     override def write(data: Array[Byte], off: Int, len: Int): Unit = {
131       currentChunk.os.write(data, off, len)
132       currentChunkSize += len
133       if (currentChunkSize >= bufferSize) {
134         closeChunk()
135       }
136     }
137 
138     override def flush(): Unit = currentChunk.os.flush()
139 
140     override def close(): Unit = closeChunk(true)
141 
142     private def closeChunk(done: Boolean = false): Unit = {
143       currentChunk.os.close()
144       if (done && uploadId == null) {
145         // didn't hit our multipart threshold, just use a regular put object
146         logger.debug(s"Starting regular s3 upload to $bucket/$key")
147         fs.putObject(bucket, key, currentChunk.read, overwrite)
148         logger.debug(s"Completed regular s3 upload to $bucket/$key")
149       } else {
150         if (uploadId == null) {
151           logger.debug(s"Starting multipart upload to $bucket/$key")
152           uploadId = fs.createMultipartUpload(bucket, key)
153           logger.debug(s"Initiated multipart upload with id $uploadId to $bucket/$key")
154         }
155         val chunk = currentChunk
156         val future = retry(chunk, uploads.size + 1, 2)
157         uploads += future
158         future.whenCompleteAsync((_, _) => chunk.cleanup())
159         logger.debug(s"Initiated part ${uploads.size} for upload with id $uploadId to $bucket/$key")
160         if (done) {
161           val parts = uploads.map(_.join()).toSeq
162           fs.completeMultipartUpload(bucket, key, uploadId, parts, overwrite)
163           logger.debug(s"Completed multipart upload with id $uploadId consisting of ${uploads.size} parts to $bucket/$key")
164         } else {
165           currentChunk = newChunk()
166           currentChunkSize = 0
167         }
168       }
169     }
170 
171     private def retry(currentChunk: Chunk, partNumber: Int, retries: Int): CompletableFuture[CompletedPart] = {
172       fs.uploadPart(bucket, key, uploadId, partNumber, currentChunk.read).exceptionallyCompose { e =>
173         if (retries > 0) {
174           logger.warn(s"Error with multipart upload, retrying $retries more times:", e)
175           retry(currentChunk, partNumber, retries - 1)
176         } else {
177           logger.error("Error with multipart upload:", e)
178           throw e
179         }
180       }
181     }
182 
183     protected def newChunk(): Chunk
184   }
185 
186   private abstract class Chunk(val os: OutputStream) {
187     def read: AsyncRequestBody
188     def cleanup(): Unit
189   }
