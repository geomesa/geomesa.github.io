20 /**
21  * A catalog for creating, listing and retrieving storage instances
22  */
23 trait StorageCatalog extends Closeable {
24 
25   /**
26    * Configuration properties
27    *
28    * @return
29    */
30   def conf: Map[String, String]
31 
32   /**
33    * Get the feature types known by this factory
34    *
35    * @return
36    */
37   def getTypeNames: Seq[String]
38 
39   /**
40    * Load an existing storage instance by name
41    *
42    * @param typeName feature type name
43    * @return
44    */
45   def load(typeName: String): FileSystemStorage
46 
47   /**
48    * Create a storage instance using the provided options
49    *
50    * @param sft simple feature type
51    * @param partitions storage partitions
52    * @param targetFileSize target file size, in bytes
53    * @param bloomFilters parquet bloom filters to write
54    * @return
55    */
56   def create(
57       sft: SimpleFeatureType,
58       partitions: Seq[String],
59       targetFileSize: Option[Long] = None,
60       bloomFilters: Seq[BloomFilterConfig] = Seq.empty): FileSystemStorage
61 
62   /**
63    * Update an existing storage instance
64    *
65    * @param typeName type to update
66    */
67   def update(typeName: String): CatalogUpdate
68 
69   /**
70    * Delete a storage instance
71    *
72    * @param typeName feature type name
73    * @param purge if true, all underlying data will be deleted
74    */
75   def delete(typeName: String, purge: Boolean): Unit
76 }

78 object StorageCatalog extends LazyLogging {
79 
80   val NamespaceConfigKey = "namespace"
81 
82   /**
83    * Create a new storage catalog instance
84    *
85    * @param conf config
86    * @return
87    */
88   def apply(conf: Map[String, String]): StorageCatalog = new IcebergCatalog(conf)
89 
90   /**
91    * Update API. All operations are metadata only, in that they do not update any existing data files. Existing files
92    * will still be accessible using the updated table, but they will not benefit from a new partition scheme, they will only
93    * return null values for new columns, etc.
94    */
95   trait CatalogUpdate {
96 
97     /**
98      * Updates to columns
99      *
100      * @return
101      */
102     def columns(): CatalogUpdate.ColumnUpdate
103 
104     /**
105      * Updates to indices and partitioning
106      *
107      * @return
108      */
109     def indices(): CatalogUpdate.IndexUpdate
110 
111     /**
112      * Updates to metadata values
113      *
114      * @return
115      */
116     def metadata(): CatalogUpdate.MetadataUpdate
117   }
118 
119   object CatalogUpdate {
120 
121     trait ColumnUpdate {
122 
123       /**
124        * Add a new column
125        *
126        * @param attribute attribute to add
127        */
128       def add(attribute: AttributeDescriptor): Unit
129 
130       /**
131        * Update an existing column, by widening the type. Note: not all updates are supported. For structural types,
132        * nested columns can be added, removed, or updated based solely on the column name.
133        *
134        * @param update updated attribute
135        */
136       def update(update: AttributeDescriptor): Unit
137 
138       /**
139        * Rename an existing column
140        *
141        * @param attribute current attribute name
142        * @param newName updated attribute name
143        */
144       def rename(attribute: String, newName: String): Unit
145 
146       /**
147        * Remove a column
148        *
149        * @param attribute attribute name to remove
150        */
151       def remove(attribute: String): Unit
152 
153       /**
154        * Change the position of an attribute
155        *
156        * @param attribute attribute to move
157        * @param position new position, in the range [0, attributes.length - 1]. other attributes will be shifted left or right
158        *                 as necessary, depending on the original and the destination position
159        */
160       def reorder(attribute: String, position: Int): Unit
161     }
162 
163     trait IndexUpdate {
164 
165       /**
166        * Overwrite the current partition spec
167        *
168        * @param partitions new partitions
169        */
170       def setPartitions(partitions: Seq[String]): Unit
171 
172       /**
173        * Add bloom filters to columns
174        *
175        * @param filters filters
176        */
177       def addBloomFilters(filters: Seq[BloomFilterConfig]): Unit
178 
179       /**
180        * Remove bloom filters from columns
181        *
182        * @param attributes columns to remove the filters from
183        */
184       def removeBloomFilters(attributes: Seq[String]): Unit
185     }
186 
187     trait MetadataUpdate {
188 
189       /**

191        *
192        * @param size target size, in bytes, or None to clear an existing target size
193        */
194       def setTargetFileSize(size: Option[Long]): Unit
195     }
196   }
