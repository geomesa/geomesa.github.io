24 /**
25  * Multi partition actions
26  *
27  * @param storage file system storage instance
28  * @param maxOpenPartitions max open partition writers
29  */
30 abstract class MultiPartitionAction[T <: Closeable with Flushable, U](storage: FileSystemStorage, maxOpenPartitions: Int)
31     extends (SimpleFeature => Unit) with Closeable with Flushable with LazyLogging {
32 
33   private val perPartition = new ConcurrentHashMap[U, ActionHolder[T]]()
34   private val oldest = MultiPartitionAction.WriterHolder.oldest[T, U]
35 
36   private lazy val logId = getClass.getSimpleName
37 
38   protected def createAction(partition: U): T
39   protected def apply(action: T, feature: SimpleFeature): Unit
40   protected def grouping(feature: SimpleFeature): U
41 
42   override def apply(feature: SimpleFeature): Unit = {
43     val partition = grouping(feature)
44     apply(perPartition.compute(partition, ComputeForAction).action, feature)
45     if (perPartition.size() > maxOpenPartitions) {
46       val oldest = perPartition.reduceEntries(maxOpenPartitions / 2, this.oldest)
47       if (perPartition.remove(oldest.getKey, oldest.getValue)) {
48         logger.debug(
49           s"Closing $logId for partition ${oldest.getKey} (last accessed ${oldest.getValue.age}ms ago) due to hitting max " +
50             s"open threshold $maxOpenPartitions")
51         CloseWithLogging(oldest.getValue)
52       }
53     }
54   }
55 
56   override def flush(): Unit = {
57     var ex: Throwable = null
58     perPartition.entrySet().forEach { entry =>
59       try { perPartition.computeIfPresent(entry.getKey, ComputeForFlush) } catch {
60         case NonFatal(e) =>  if (ex == null) { ex = e } else { ex.addSuppressed(e) }
61       }
62     }
63     if (ex != null) {
64       throw ex
65     }
66   }
67 
68   override def close(): Unit = {
69     var ex: Throwable = null
70     perPartition.forEach { (p, writer) =>
71       logger.debug(s"Closing $logId for partition $p")
72       CloseQuietly(writer).foreach(e => if (ex == null) { ex = e } else { ex.addSuppressed(e) })
73     }
74     if (ex != null) {
75       throw ex
76     }
77   }
78 
79   private object ComputeForAction extends BiFunction[U, ActionHolder[T], ActionHolder[T]] {
80     override def apply(p: U, current: ActionHolder[T]): ActionHolder[T] = {
81       if (current == null) {
82         logger.debug(s"Opening $logId for partition $p")
83         new ActionHolder(createAction(p))
84       } else {
85         current.updateAccessTime() // update access time before returning so it's not expired
86         current
87       }
88     }
89   }
90 
91   private object ComputeForFlush extends BiFunction[U, ActionHolder[T], ActionHolder[T]] {
92     override def apply(p: U, current: ActionHolder[T]): ActionHolder[T] = {
93       if (current != null) {
94         logger.debug(s"Flushing $logId for partition $p")
95         current.flush()
96       }
97       current
98     }
99   }
100 }

102 object MultiPartitionAction extends LazyLogging {
103 
104   private class ActionHolder[T <: Closeable with Flushable](val action: T) extends Closeable with Flushable {
105 
106     @volatile
107     private var lastAccess: Long = System.currentTimeMillis()
108     private val closed: AtomicBoolean = new AtomicBoolean(false)
109 
110     def age: Long = System.currentTimeMillis() - lastAccess
111     def updateAccessTime(): Unit = lastAccess = System.currentTimeMillis()
112 
113     override def flush(): Unit = action.flush()
114 
115     override def close(): Unit = {
116       if (closed.compareAndSet(false, true)) {
117         action.close()
118       }
119     }
120   }
121 
122   private object WriterHolder {
123     def oldest[T <: Closeable with Flushable, U]: BiFunction[Entry[U, ActionHolder[T]], Entry[U, ActionHolder[T]], Entry[U, ActionHolder[T]]] =
124       (left, right) => if (right.getValue.age < left.getValue.age) { right } else { left }
125   }
