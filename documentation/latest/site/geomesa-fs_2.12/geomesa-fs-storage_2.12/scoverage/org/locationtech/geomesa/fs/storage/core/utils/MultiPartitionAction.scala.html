24 /**
25  * Multi partition actions
26  *
27  * @param storage file system storage instance
28  * @param maxOpenPartitions max open partition writers
29  */
30 abstract class MultiPartitionAction[T <: Closeable with Flushable](storage: FileSystemStorage, maxOpenPartitions: Int)
31     extends (SimpleFeature => Unit) with Closeable with Flushable with LazyLogging {
32 
33   private val perPartition = new ConcurrentHashMap[Partition, ActionHolder[T]]()
34   private val oldest = MultiPartitionAction.WriterHolder.oldest[T]
35 
36   private lazy val logId = getClass.getSimpleName
37 
38   protected def createAction(partition: Partition): T
39   protected def apply(action: T, feature: SimpleFeature): Unit
40 
41   override def apply(feature: SimpleFeature): Unit = {
42     val partition = Partition(storage.schemes.map(_.getPartition(feature)))
43     apply(perPartition.compute(partition, ComputeForAction).action, feature)
44     if (perPartition.size() > maxOpenPartitions) {
45       val oldest = perPartition.reduceEntries(maxOpenPartitions / 2, this.oldest)
46       if (perPartition.remove(oldest.getKey, oldest.getValue)) {
47         logger.debug(
48           s"Closing $logId for partition ${oldest.getKey} (last accessed ${oldest.getValue.age}ms ago) due to hitting max " +
49             s"open threshold $maxOpenPartitions")
50         CloseWithLogging(oldest.getValue)
51       }
52     }
53   }
54 
55   override def flush(): Unit = {
56     var ex: Throwable = null
57     perPartition.entrySet().forEach { entry =>
58       try { perPartition.computeIfPresent(entry.getKey, ComputeForFlush) } catch {
59         case NonFatal(e) =>  if (ex == null) { ex = e } else { ex.addSuppressed(e) }
60       }
61     }
62     if (ex != null) {
63       throw ex
64     }
65   }
66 
67   override def close(): Unit = {
68     var ex: Throwable = null
69     perPartition.forEach { (p, writer) =>
70       logger.debug(s"Closing $logId for partition $p")
71       CloseQuietly(writer).foreach(e => if (ex == null) { ex = e } else { ex.addSuppressed(e) })
72     }
73     if (ex != null) {
74       throw ex
75     }
76   }
77 
78   private object ComputeForAction extends BiFunction[Partition, ActionHolder[T], ActionHolder[T]] {
79     override def apply(p: Partition, current: ActionHolder[T]): ActionHolder[T] = {
80       if (current == null) {
81         logger.debug(s"Opening $logId for partition $p")
82         new ActionHolder(createAction(p))
83       } else {
84         current.updateAccessTime() // update access time before returning so it's not expired
85         current
86       }
87     }
88   }
89 
90   private object ComputeForFlush extends BiFunction[Partition, ActionHolder[T], ActionHolder[T]] {
91     override def apply(p: Partition, current: ActionHolder[T]): ActionHolder[T] = {
92       if (current != null) {
93         logger.debug(s"Flushing $logId for partition $p")
94         current.flush()
95       }
96       current
97     }
98   }
99 }

101 object MultiPartitionAction extends LazyLogging {
102 
103   private class ActionHolder[T <: Closeable with Flushable](val action: T) extends Closeable with Flushable {
104 
105     @volatile
106     private var lastAccess: Long = System.currentTimeMillis()
107     private val closed: AtomicBoolean = new AtomicBoolean(false)
108 
109     def age: Long = System.currentTimeMillis() - lastAccess
110     def updateAccessTime(): Unit = lastAccess = System.currentTimeMillis()
111 
112     override def flush(): Unit = action.flush()
113 
114     override def close(): Unit = {
115       if (closed.compareAndSet(false, true)) {
116         action.close()
117       }
118     }
119   }
120 
121   private object WriterHolder {
122     def oldest[T <: Closeable with Flushable]: BiFunction[Entry[Partition, ActionHolder[T]], Entry[Partition, ActionHolder[T]], Entry[Partition, ActionHolder[T]]] =
123       (left, right) => if (right.getValue.age < left.getValue.age) { right } else { left }
124   }
