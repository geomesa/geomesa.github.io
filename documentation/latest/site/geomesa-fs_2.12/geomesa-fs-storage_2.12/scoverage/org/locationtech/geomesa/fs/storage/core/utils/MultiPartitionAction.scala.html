24 /**
25  * Multi partition actions
26  *
27  * @param storage file system storage instance
28  * @param maxOpenPartitions max open partition writers
29  */
30 abstract class MultiPartitionAction[T <: Closeable with Flushable](storage: FileSystemStorage, maxOpenPartitions: Int)
31     extends (SimpleFeature => Unit) with Closeable with Flushable with LazyLogging {
32 
33   @volatile
34   private var closed = false
35 
36   private val perPartition = new ConcurrentHashMap[Partition, ActionHolder[T]]()
37   private val oldest = MultiPartitionAction.WriterHolder.oldest[T]
38 
39   private lazy val logId = getClass.getSimpleName
40 
41   protected def createAction(partition: Partition): T
42   protected def apply(action: T, feature: SimpleFeature): Unit
43 
44   override def apply(feature: SimpleFeature): Unit = {
45     val partition = Partition(storage.schemes.map(_.getPartition(feature)))
46     apply(perPartition.compute(partition, ComputeForAction).action, feature)
47     if (perPartition.size() > maxOpenPartitions) {
48       val oldest = perPartition.reduceEntries(maxOpenPartitions / 2, this.oldest)
49       if (perPartition.remove(oldest.getKey, oldest.getValue)) {
50         logger.debug(
51           s"Closing $logId for partition ${oldest.getKey} (last accessed ${oldest.getValue.age}ms ago) due to hitting max " +
52             s"open threshold $maxOpenPartitions")
53         CloseWithLogging(oldest.getValue)
54       }
55     }
56   }
57 
58   override def flush(): Unit = {
59     var ex: Throwable = null
60     perPartition.entrySet().forEach { entry =>
61       try { perPartition.computeIfPresent(entry.getKey, ComputeForFlush) } catch {
62         case NonFatal(e) =>  if (ex == null) { ex = e } else { ex.addSuppressed(e) }
63       }
64     }
65     if (ex != null) {
66       throw ex
67     }
68   }
69 
70   override def close(): Unit = {
71     closed = true
72     var ex: Throwable = null
73     perPartition.forEach { (p, writer) =>
74       logger.debug(s"Closing $logId for partition $p")
75       CloseQuietly(writer).foreach(e => if (ex == null) { ex = e } else { ex.addSuppressed(e) })
76     }
77     if (ex != null) {
78       throw ex
79     }
80   }
81 
82   private object ComputeForAction extends BiFunction[Partition, ActionHolder[T], ActionHolder[T]] {
83     override def apply(p: Partition, current: ActionHolder[T]): ActionHolder[T] = {
84       if (current == null) {
85         logger.debug(s"Opening $logId for partition $p")
86         new ActionHolder(createAction(p))
87       } else {
88         current.updateAccessTime() // update access time before returning so it's not expired
89         current
90       }
91     }
92   }
93 
94   private object ComputeForFlush extends BiFunction[Partition, ActionHolder[T], ActionHolder[T]] {
95     override def apply(p: Partition, current: ActionHolder[T]): ActionHolder[T] = {
96       if (current != null) {
97         logger.debug(s"Flushing $logId for partition $p")
98         current.flush()
99       }
100       current
101     }
102   }
103 }
104 
105 object MultiPartitionAction extends LazyLogging {
106 
107   private class ActionHolder[T <: Closeable with Flushable](val action: T) extends Closeable with Flushable {
108 
109     @volatile
110     private var lastAccess: Long = System.currentTimeMillis()
111     private val closed: AtomicBoolean = new AtomicBoolean(false)
112 
113     def age: Long = System.currentTimeMillis() - lastAccess
114     def updateAccessTime(): Unit = lastAccess = System.currentTimeMillis()
115 
116     override def flush(): Unit = action.flush()
117 
118     override def close(): Unit = {
119       if (closed.compareAndSet(false, true)) {
120         action.close()
121       }
122     }
123   }
124 
125   private object WriterHolder {
126     def oldest[T <: Closeable with Flushable]: BiFunction[Entry[Partition, ActionHolder[T]], Entry[Partition, ActionHolder[T]], Entry[Partition, ActionHolder[T]]] =
127       (left, right) => if (right.getValue.age < left.getValue.age) { right } else { left }
128   }
