36 /**
37  * Update writer implementation
38  *
39  * @param storage file system storage instance
40  * @param reader features to update
41  * @param writer writer for updated features
42  * @param deleter deleter for removed/modified features
43  */
44 class IcebergUpdateWriter(
45     storage: FileSystemStorage,
46     reader: CloseableIterator[SimpleFeature],
47     writer: MultiPartitionWriter,
48     deleter: MultiPartitionDeleter,
49   ) extends FileSystemUpdateWriter {
50 
51   // feature returned from reader
52   private var original: SimpleFeature = _
53 
54   // feature that caller will modify
55   private var live: FastSettableFeature = _
56 
57   override def hasNext: Boolean = reader.hasNext
58 
59   override def next(): FastSettableFeature = {
60     original = reader.next()
61     live = ScalaSimpleFeature.copy(storage.sft, original) // this copies user data as well
62     // set the use provided FID hint - allows user to update fid if desired,
63     // but if not we'll use the existing one
64     live.getUserData.put(Hints.USE_PROVIDED_FID, java.lang.Boolean.TRUE)
65     live
66   }
67 
68   override def write(): Unit = {
69     if (original == null) {
70       throw new IllegalStateException("next() must be called before write()")
71     }
72     // update the feature id based on hints before we compare for changes
73     live = GeoMesaFeatureWriter.featureWithFid(live)
74     // only write if feature has actually changed...
75     // comparison of feature ID and attributes - doesn't consider concrete class used
76     if (!ScalaSimpleFeature.equalIdAndAttributes(live, original) ||
77         SecurityUtils.getVisibility(live) != SecurityUtils.getVisibility(original)) {
78       deleter(original)
79       writer(live)
80     }
81     original = null
82     live = null
83   }
84 
85   override def remove(): Unit = {
86     if (original == null) {
87       throw new IllegalStateException("next() must be called before remove()")
88     }
89     deleter(original)
90     original = null
91     live = null
92   }
93 
94   override def flush(): Unit = FlushQuietly.raise(writer, deleter)
95 
96   override def close(): Unit = {
97     CloseWithLogging.raise(Seq(reader, writer, deleter))
98     if (!deleter.files.isEmpty) {
99       val delta = storage.table.newRowDelta()
100       // guard against concurrent commits that touch the same delete files
101       Option(storage.table.currentSnapshot()).foreach(s => delta.validateFromSnapshot(s.snapshotId()))
102       deleter.files.forEach(f => delta.addDeletes(f))
103       // remove any previous DVs whose deletes we merged into the newly written DVs, otherwise
104       // the commit will fail validation with multiple live DVs referencing the same data file
105       deleter.rewritten.forEach(f => delta.removeDeletes(f))
106       delta.commit()
107     }
108   }
109 }

113   /**
114    * Create a new update writer
115    *
116    * @param storage storage
117    * @param filter filter for features to update
118    * @param readThreads number of threads using to read features to update
119    * @param maxOpenPartitions max open writer partitions
120    * @return
121    */
122   def apply(storage: FileSystemStorage, filter: Filter, readThreads: Int, maxOpenPartitions: Int): IcebergUpdateWriter = {
123     // note: writer and deleter are Closeable, but they don't need to be closed unless they initialize at least one partition,
124     // so it's safe to not clean them up if there's an exception in this method
125     val writer = new MultiPartitionWriter(storage, maxOpenPartitions)
126     val deleter = new MultiPartitionDeleter(storage, maxOpenPartitions)
127     // note: this class expects the reader to return RecordSimpleFeatures, but this isn't enforced through the call signature
128     val reader = storage.getReader(new Query(storage.sft.getTypeName, filter), readThreads, forUpdate = true)
129     new IcebergUpdateWriter(storage, reader, writer, deleter)
130   }
131 
132   /**
133    * Multi-partition deleter
134    *
135    * @param storage file system storage instance
136    * @param maxOpenPartitions max open partition writers
137    */
138   private class MultiPartitionDeleter(storage: FileSystemStorage, maxOpenPartitions: Int)
139       extends MultiPartitionAction[DeleteWriter](storage, maxOpenPartitions) {
140 
141     // newly written delete files (DVs) to add on commit
142     val files: java.util.Set[DeleteFile] = Collections.newSetFromMap(new ConcurrentHashMap[DeleteFile, java.lang.Boolean]())
143     // previous DVs whose deletes we merged into the new DVs - these must be removed on commit
144     val rewritten: java.util.Set[DeleteFile] = Collections.newSetFromMap(new ConcurrentHashMap[DeleteFile, java.lang.Boolean]())
145 
146     private val closed = new AtomicBoolean(false)
147     private val fileSupplier: Supplier[OutputFile] = () => storage.table.io().newOutputFile(storage.newFilePath("puffin"))
148     // loads existing deletes for a data file so they can be merged into the new DV - a data file
149     // can only be referenced by a single DV, so we have to rewrite any existing one
150     private val writer = new BaseDVFileWriter(fileSupplier, new PreviousDeleteLoader(storage))
151 
152     override protected def createAction(partition: Partition): DeleteWriter =
153       new DeleteWriter(writer, storage.table.spec(), storage.metadata.partition(partition))
154     override protected def apply(action: DeleteWriter, feature: SimpleFeature): Unit =
155       action.apply(feature.asInstanceOf[StructSimpleFeature])
156 
157     override def close(): Unit = {
158       if (closed.compareAndSet(false, true)) {
159         try { super.close() } finally  {
160           CloseQuietly.raise(writer)
161         }
162         val result = writer.result()
163         files.addAll(result.deleteFiles())
164         rewritten.addAll(result.rewrittenDeleteFiles())
165       }
166     }
167   }
168 
169   /**
170    * Loads existing deletes (deletion vectors) for a given data file path, so they can be merged into a
171    * newly written DV. Returns null if there are no existing deletes for the given path.
172    *
173    * @param storage file system storage instance
174    */
175   private class PreviousDeleteLoader(storage: FileSystemStorage) extends Function[String, PositionDeleteIndex] {
176 
177     import scala.collection.JavaConverters._
178 
179     // map of data-file-path -> set of delete files that reference it, from the current table state
180     private lazy val deletesByPath: CharSequenceMap[DeleteFileSet] = {
181       val map = CharSequenceMap.create[DeleteFileSet]()
182       val snapshot = storage.table.currentSnapshot()
183       if (snapshot != null) {
184         val io = storage.table.io()
185         val specs = storage.table.specs()
186         snapshot.deleteManifests(io).asScala.foreach { manifest =>
187           WithClose(ManifestFiles.readDeleteManifest(manifest, io, specs)) { reader =>
188             reader.iterator().asScala.foreach { deleteFile =>
189               val ref = deleteFile.referencedDataFile()
190               if (ref != null) {
191                 // copy() to detach from the manifest reader, which reuses containers
192                 map.computeIfAbsent(ref, () => DeleteFileSet.create()).add(deleteFile.copy())
193               }
194             }
195           }
196         }
197       }
198       map
199     }
200 
201     private lazy val loader: DeleteLoader =
202       new BaseDeleteLoader(deleteFile => storage.table.io().newInputFile(deleteFile))
203 
204     override def apply(path: String): PositionDeleteIndex = {
205       val set = deletesByPath.get(path)
206       if (set == null) { null } else {
207         loader.loadPositionDeletes(set, path)
208       }
209     }
210   }

225     override def flush(): Unit = {}
226     override def close(): Unit = {}
227   }
