35 /**
36  * Update writer implementation
37  *
38  * @param storage file system storage instance
39  * @param reader features to update
40  * @param writer writer for updated features
41  * @param deleter deleter for removed/modified features
42  */
43 class IcebergUpdateWriter(
44     storage: FileSystemStorage,
45     reader: CloseableIterator[SimpleFeature],
46     writer: MultiPartitionWriter,
47     deleter: MultiPartitionDeleter,
48   ) extends FileSystemUpdateWriter {
49 
50   // feature returned from reader
51   private var original: SimpleFeature = _
52 
53   // feature that caller will modify
54   private var live: FastSettableFeature = _
55 
56   override def hasNext: Boolean = reader.hasNext
57 
58   override def next(): FastSettableFeature = {
59     original = reader.next()
60     live = ScalaSimpleFeature.copy(storage.sft, original) // this copies user data as well
61     // set the use provided FID hint - allows user to update fid if desired,
62     // but if not we'll use the existing one
63     live.getUserData.put(Hints.USE_PROVIDED_FID, java.lang.Boolean.TRUE)
64     live
65   }
66 
67   override def write(): Unit = {
68     if (original == null) {
69       throw new IllegalStateException("next() must be called before write()")
70     }
71     // update the feature id based on hints before we compare for changes
72     live = GeoMesaFeatureWriter.featureWithFid(live)
73     // only write if feature has actually changed...
74     // comparison of feature ID and attributes - doesn't consider concrete class used
75     if (!ScalaSimpleFeature.equalIdAndAttributes(live, original) ||
76         SecurityUtils.getVisibility(live) != SecurityUtils.getVisibility(original)) {
77       deleter(original)
78       writer(live)
79     }
80     original = null
81     live = null
82   }
83 
84   override def remove(): Unit = {
85     if (original == null) {
86       throw new IllegalStateException("next() must be called before remove()")
87     }
88     deleter(original)
89     original = null
90     live = null
91   }
92 
93   override def flush(): Unit = FlushQuietly.raise(writer, deleter)
94 
95   override def close(): Unit = {
96     CloseWithLogging.raise(Seq(reader, writer, deleter))
97     if (!deleter.files.isEmpty) {
98       val delta = storage.table.newRowDelta()
99       deleter.files.forEach(f => delta.addDeletes(f))
100       delta.commit()
101     }
102   }
103 }

105 object IcebergUpdateWriter {
106 
107   /**
108    * Create a new update writer
109    *
110    * @param storage storage
111    * @param filter filter for features to update
112    * @param readThreads number of threads using to read features to update
113    * @param maxOpenPartitions max open writer partitions
114    * @return
115    */
116   def apply(storage: FileSystemStorage, filter: Filter, readThreads: Int, maxOpenPartitions: Int): IcebergUpdateWriter = {
117     // note: writer and deleter are Closeable, but they don't need to be closed unless they initialize at least one partition,
118     // so it's safe to not clean them up if there's an exception in this method
119     val writer = new MultiPartitionWriter(storage, maxOpenPartitions)
120     val deleter = new MultiPartitionDeleter(storage, maxOpenPartitions)
121     // note: this class expects the reader to return RecordSimpleFeatures, but this isn't enforced through the call signature
122     val reader = storage.getReader(new Query(storage.sft.getTypeName, filter), readThreads, forUpdate = true)
123     new IcebergUpdateWriter(storage, reader, writer, deleter)
124   }
125 
126   /**
127    * Multi-partition deleter
128    *
129    * @param storage file system storage instance
130    * @param maxOpenPartitions max open partition writers
131    */
132   private class MultiPartitionDeleter(storage: FileSystemStorage, maxOpenPartitions: Int)
133       extends MultiPartitionAction[DeleteWriter](storage, maxOpenPartitions) {
134 
135     val files: java.util.Set[DeleteFile] = Collections.newSetFromMap(new ConcurrentHashMap[DeleteFile, java.lang.Boolean]())
136 
137     override protected def createAction(partition: Partition): DeleteWriter = new DeleteWriter(storage, partition, files)
138     override protected def apply(action: DeleteWriter, feature: SimpleFeature): Unit =
139       action.apply(feature.asInstanceOf[StructSimpleFeature])
140   }
141 
142   /**
143    * Single partition deleter
144    *
145    * @param storage file system storage instance
146    * @param partition partition to delete records out of
147    * @param files set for returning any delete files created by this deleter
148    */
149   private class DeleteWriter(storage: FileSystemStorage, partition: Partition, files: java.util.Set[DeleteFile])
150       extends (StructSimpleFeature => Unit) with Closeable with Flushable {
151 
152     private val closed = new AtomicBoolean(false)
153     private val deletes = Seq.newBuilder[(String, Long)]
154     private val path = storage.context.root.resolve(FileSystemStorage.newFilePath(storage.sft.getTypeName, "x-"))
155     private val file = storage.table.io().newOutputFile(path.toString)
156 
157     private val writer =
158       Parquet.writeDeletes(file)
159         .withSpec(storage.table.spec())
160         .withPartition(storage.metadata.partition(partition))
161         .createWriterFunc((schema: Schema, `type`: MessageType) => GenericParquetWriter.create(schema, `type`))
162         .buildPositionWriter[Unit]()
163 
164     override def apply(feature: StructSimpleFeature): Unit = deletes += (feature.getFilePath -> feature.getRowPosition)
165 
166     override def flush(): Unit = {}
167 
168     override def close(): Unit = {
169       if (closed.compareAndSet(false, true)) {
170         try {
171           val struct = PositionDelete.create[Unit]()
172           // note: deletes need to be sorted by file+row
173           deletes.result().sorted.foreach { case (path, pos) =>
174             writer.write(struct.set(path, pos))
175           }
176         } finally{
177           CloseWithLogging(writer)
178         }
179         files.add(writer.toDeleteFile)
180       }
181     }
182   }
