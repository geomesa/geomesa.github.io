28 /**
29  * Update writer implementation. Updated features are written to new data files, and the original rows are deleted
30  * using deletion vectors. All changes are tracked in a transaction, and are not visible until it is committed.
31  *
32  * @param storage file system storage instance
33  * @param transaction transaction tracking the changes
34  * @param reader features to update
35  * @param writer writer for updated features
36  */
37 class IcebergUpdateWriter(
38     storage: FileSystemStorage,
39     transaction: TableTransaction,
40     reader: CloseableIterator[SimpleFeature],
41     writer: MultiPartitionWriter,
42   ) extends FileSystemUpdateWriter {
43 
44   private val closed = new AtomicBoolean(false)
45 
46   // feature returned from reader
47   private var original: SimpleFeature = _
48 
49   // feature that caller will modify
50   private var live: FastSettableFeature = _
51 
52   transaction.opened()
53 
54   override def hasNext: Boolean = reader.hasNext
55 
56   override def next(): FastSettableFeature = {
57     original = reader.next()
58     live = ScalaSimpleFeature.copy(storage.sft, original) // this copies user data as well
59     // set the use provided FID hint - allows user to update fid if desired,
60     // but if not we'll use the existing one
61     live.getUserData.put(Hints.USE_PROVIDED_FID, java.lang.Boolean.TRUE)
62     live
63   }
64 
65   override def write(): Unit = {
66     if (original == null) {
67       throw new IllegalStateException("next() must be called before write()")
68     }
69     // update the feature id based on hints before we compare for changes
70     live = GeoMesaFeatureWriter.featureWithFid(live)
71     // only write if feature has actually changed...
72     // comparison of feature ID and attributes - doesn't consider concrete class used
73     if (!ScalaSimpleFeature.equalIdAndAttributes(live, original) ||
74         SecurityUtils.getVisibility(live) != SecurityUtils.getVisibility(original)) {
75       val row = original.asInstanceOf[StructSimpleFeature]
76       val path = row.getFilePath
77       val position = row.getRowPosition
78       // check that the original can be deleted before writing the update, so that we don't end up with both.
79       // write before deleting to ensure the feature is valid
80       transaction.validateDelete(path, position)
81       writer.write(live)
82       transaction.deleteValidated(path, position)
83     }
84     original = null
85     live = null
86   }
87 
88   override def remove(): Unit = {
89     if (original == null) {
90       throw new IllegalStateException("next() must be called before remove()")
91     }
92     val row = original.asInstanceOf[StructSimpleFeature]
93     transaction.delete(row.getFilePath, row.getRowPosition)
94     original = null
95     live = null
96   }
97 
98   override def flush(): Unit = writer.flush()
99 
100   override def close(): Unit = {
101     if (closed.compareAndSet(false, true)) {
102       try { CloseWithLogging.raise(Seq(reader, writer)) } catch {
103         case NonFatal(e) => transaction.failed(e); throw e
104       } finally {
105         transaction.closed()
106       }
107     }
108   }
109 }

111 object IcebergUpdateWriter {
112 
113   /**
114    * Create a new update writer
115    *
116    * @param storage storage
117    * @param transaction transaction
118    * @param filter filter for features to update
119    * @param readThreads number of threads using to read features to update
120    * @param maxOpenPartitions max open writer partitions
121    * @return
122    */
123   def apply(
124       storage: FileSystemStorage,
125       transaction: TableTransaction,
126       filter: Filter,
127       readThreads: Int,
128       maxOpenPartitions: Int): IcebergUpdateWriter = {
129     val writer = new MultiPartitionWriter(storage, transaction, maxOpenPartitions)
130     try {
131       // read from the same snapshot that deletes are validated against - if the table is empty, there's nothing to update
132       // note: this class expects the reader to return StructSimpleFeatures, but this isn't enforced through the call signature
133       val reader = transaction.snapshotId match {
134         case None => CloseableIterator.empty[SimpleFeature]
135         case id => storage.getReader(new Query(storage.sft.getTypeName, filter), readThreads, forUpdate = true, snapshotId = id)
136       }
137       try { new IcebergUpdateWriter(storage, transaction, reader, writer) } catch {
138         case NonFatal(e) => CloseWithLogging(reader); throw e
139       }
140     } catch {
141       case NonFatal(e) => CloseWithLogging(writer); throw e
142     }
143   }
