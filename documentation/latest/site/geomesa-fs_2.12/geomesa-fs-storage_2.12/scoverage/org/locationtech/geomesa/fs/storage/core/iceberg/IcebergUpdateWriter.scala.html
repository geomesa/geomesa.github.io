36 /**
37  * Update writer implementation
38  *
39  * @param storage file system storage instance
40  * @param reader features to update
41  * @param writer writer for updated features
42  * @param deleter deleter for removed/modified features
43  */
44 class IcebergUpdateWriter(
45     storage: FileSystemStorage,
46     reader: CloseableIterator[SimpleFeature],
47     writer: MultiPartitionWriter,
48     deleter: MultiPartitionDeleter,
49   ) extends FileSystemUpdateWriter {
50 
51   private val closed = new AtomicBoolean(false)
52 
53   // feature returned from reader
54   private var original: SimpleFeature = _
55 
56   // feature that caller will modify
57   private var live: FastSettableFeature = _
58 
59   override def hasNext: Boolean = reader.hasNext
60 
61   override def next(): FastSettableFeature = {
62     original = reader.next()
63     live = ScalaSimpleFeature.copy(storage.sft, original) // this copies user data as well
64     // set the use provided FID hint - allows user to update fid if desired,
65     // but if not we'll use the existing one
66     live.getUserData.put(Hints.USE_PROVIDED_FID, java.lang.Boolean.TRUE)
67     live
68   }
69 
70   override def write(): Unit = {
71     if (original == null) {
72       throw new IllegalStateException("next() must be called before write()")
73     }
74     // update the feature id based on hints before we compare for changes
75     live = GeoMesaFeatureWriter.featureWithFid(live)
76     // only write if feature has actually changed...
77     // comparison of feature ID and attributes - doesn't consider concrete class used
78     if (!ScalaSimpleFeature.equalIdAndAttributes(live, original) ||
79         SecurityUtils.getVisibility(live) != SecurityUtils.getVisibility(original)) {
80       // write first to ensure feature is valid
81       writer(live)
82       deleter(original)
83     }
84     original = null
85     live = null
86   }
87 
88   override def remove(): Unit = {
89     if (original == null) {
90       throw new IllegalStateException("next() must be called before remove()")
91     }
92     deleter(original)
93     original = null
94     live = null
95   }
96 
97   override def flush(): Unit = FlushQuietly.raise(writer, deleter)
98 
99   override def close(): Unit = {
100     if (closed.compareAndSet(false, true)) {
101       CloseWithLogging.raise(Seq(reader, writer, deleter))
102       if (!deleter.files.isEmpty) {
103         val delta = storage.table.newRowDelta()
104         // guard against concurrent commits that touch the same delete files
105         Option(storage.table.currentSnapshot()).foreach(s => delta.validateFromSnapshot(s.snapshotId()))
106         deleter.files.forEach(f => delta.addDeletes(f))
107         // remove any previous DVs whose deletes we merged into the newly written DVs, otherwise
108         // the commit will fail validation with multiple live DVs referencing the same data file
109         deleter.rewritten.forEach(f => delta.removeDeletes(f))
110         delta.commit()
111       }
112     }
113   }
114 }

116 object IcebergUpdateWriter {

118   /**
119    * Create a new update writer
120    *
121    * @param storage storage
122    * @param filter filter for features to update
123    * @param readThreads number of threads using to read features to update
124    * @param maxOpenPartitions max open writer partitions
125    * @return
126    */
127   def apply(storage: FileSystemStorage, filter: Filter, readThreads: Int, maxOpenPartitions: Int): IcebergUpdateWriter = {
128     // note: writer and deleter are Closeable, but they don't need to be closed unless they initialize at least one partition,
129     // so it's safe to not clean them up if there's an exception in this method
130     val writer = new MultiPartitionWriter(storage, maxOpenPartitions)
131     val deleter = new MultiPartitionDeleter(storage, maxOpenPartitions)
132     // note: this class expects the reader to return RecordSimpleFeatures, but this isn't enforced through the call signature
133     val reader = storage.getReader(new Query(storage.sft.getTypeName, filter), readThreads, forUpdate = true)
134     new IcebergUpdateWriter(storage, reader, writer, deleter)
135   }
136 
137   /**
138    * Multi-partition deleter
139    *
140    * @param storage file system storage instance
141    * @param maxOpenPartitions max open partition writers
142    */
143   private class MultiPartitionDeleter(storage: FileSystemStorage, maxOpenPartitions: Int)
144       extends MultiPartitionAction[DeleteWriter, (Int, StructLike)](storage, maxOpenPartitions) {
145 
146     // newly written delete files (DVs) to add on commit
147     val files: java.util.Set[DeleteFile] = Collections.newSetFromMap(new ConcurrentHashMap[DeleteFile, java.lang.Boolean]())
148     // previous DVs whose deletes we merged into the new DVs - these must be removed on commit
149     val rewritten: java.util.Set[DeleteFile] = Collections.newSetFromMap(new ConcurrentHashMap[DeleteFile, java.lang.Boolean]())
150 
151     private val closed = new AtomicBoolean(false)
152     private val fileSupplier: Supplier[OutputFile] = () => storage.table.io().newOutputFile(storage.newFilePath("puffin"))
153     // loads existing deletes for a data file so they can be merged into the new DV - a data file
154     // can only be referenced by a single DV, so we have to rewrite any existing one
155     private val writer = new BaseDVFileWriter(fileSupplier, new PreviousDeleteLoader(storage))
156     // loads partition specs from data files - if partitioning is changed, old files might have different partition schemes
157     private val dataFileLoader = new ManifestLoader(storage)
158 
159     override protected def createAction(partition: (Int, StructLike)): DeleteWriter = {
160       val (specId, partitionStruct) = partition
161       new DeleteWriter(writer, storage.table.specs().get(specId), partitionStruct)
162     }
163 
164     override protected def apply(action: DeleteWriter, feature: SimpleFeature): Unit =
165       action.apply(feature.asInstanceOf[StructSimpleFeature])
166     override protected def grouping(feature: SimpleFeature): (Int, StructLike) =
167       dataFileLoader(feature.asInstanceOf[StructSimpleFeature].getFilePath)
168 
169     override def close(): Unit = {
170       if (closed.compareAndSet(false, true)) {
171         try { super.close() } finally  {
172           CloseQuietly.raise(writer)
173         }
174         val result = writer.result()
175         files.addAll(result.deleteFiles())
176         rewritten.addAll(result.rewrittenDeleteFiles())
177       }
178     }
179   }
180 
181   /**
182    * Loads existing deletes (deletion vectors) for a given data file path, so they can be merged into a
183    * newly written DV. Returns null if there are no existing deletes for the given path.
184    *
185    * @param storage file system storage instance
186    */
187   private class PreviousDeleteLoader(storage: FileSystemStorage) extends Function[String, PositionDeleteIndex] {
188 
189     import scala.collection.JavaConverters._
190 
191     // map of data-file-path -> set of delete files that reference it, from the current table state
192     private lazy val deletesByPath: CharSequenceMap[DeleteFileSet] = {
193       val map = CharSequenceMap.create[DeleteFileSet]()
194       val snapshot = storage.table.currentSnapshot()
195       if (snapshot != null) {
196         val io = storage.table.io()
197         val specs = storage.table.specs()
198         snapshot.deleteManifests(io).asScala.foreach { manifest =>
199           WithClose(ManifestFiles.readDeleteManifest(manifest, io, specs)) { reader =>
200             reader.iterator().asScala.foreach { deleteFile =>
201               val ref = deleteFile.referencedDataFile()
202               if (ref != null) {
203                 // copy() to detach from the manifest reader, which reuses containers
204                 map.computeIfAbsent(ref, () => DeleteFileSet.create()).add(deleteFile.copy())
205               }
206             }
207           }
208         }
209       }
210       map
211     }
212 
213     private lazy val loader: DeleteLoader =
214       new BaseDeleteLoader(deleteFile => storage.table.io().newInputFile(deleteFile))
215 
216     override def apply(path: String): PositionDeleteIndex = {
217       val set = deletesByPath.get(path)
218       if (set == null) { null } else {
219         loader.loadPositionDeletes(set, path)
220       }
221     }
222   }
223 
224   private class ManifestLoader(storage: FileSystemStorage) extends Function[String, (Int, StructLike)] {
225 
226     import scala.collection.JavaConverters._
227 
228     private lazy val dataFilesByPath: CharSequenceMap[(Int, StructLike)] = {
229       val map = CharSequenceMap.create[(Int, StructLike)]()
230       val snapshot = storage.table.currentSnapshot()
231       if (snapshot != null) {
232         val io = storage.table.io()
233         val specs = storage.table.specs()
234         snapshot.dataManifests(io).asScala.foreach { manifest =>
235           WithClose(ManifestFiles.read(manifest, io, specs)) { reader =>
236             WithClose(reader.iterator()) { iterator =>
237               // manifest readers reuse record instances
238               iterator.asScala.foreach(file => map.put(file.location(), (file.specId(), file.copy().partition())))
239             }
240           }
241         }
242       }
243       map
244     }
245 
246     override def apply(path: String): (Int, StructLike) = dataFilesByPath.get(path)
247   }
