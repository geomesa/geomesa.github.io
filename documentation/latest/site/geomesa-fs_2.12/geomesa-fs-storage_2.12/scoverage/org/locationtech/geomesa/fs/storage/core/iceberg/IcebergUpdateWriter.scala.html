36 /**
37  * Update writer implementation
38  *
39  * @param storage file system storage instance
40  * @param reader features to update
41  * @param writer writer for updated features
42  * @param deleter deleter for removed/modified features
43  */
44 class IcebergUpdateWriter(
45     storage: FileSystemStorage,
46     reader: CloseableIterator[SimpleFeature],
47     writer: MultiPartitionWriter,
48     deleter: MultiPartitionDeleter,
49   ) extends FileSystemUpdateWriter {
50 
51   // feature returned from reader
52   private var original: SimpleFeature = _
53 
54   // feature that caller will modify
55   private var live: FastSettableFeature = _
56 
57   override def hasNext: Boolean = reader.hasNext
58 
59   override def next(): FastSettableFeature = {
60     original = reader.next()
61     live = ScalaSimpleFeature.copy(storage.sft, original) // this copies user data as well
62     // set the use provided FID hint - allows user to update fid if desired,
63     // but if not we'll use the existing one
64     live.getUserData.put(Hints.USE_PROVIDED_FID, java.lang.Boolean.TRUE)
65     live
66   }
67 
68   override def write(): Unit = {
69     if (original == null) {
70       throw new IllegalStateException("next() must be called before write()")
71     }
72     // update the feature id based on hints before we compare for changes
73     live = GeoMesaFeatureWriter.featureWithFid(live)
74     // only write if feature has actually changed...
75     // comparison of feature ID and attributes - doesn't consider concrete class used
76     if (!ScalaSimpleFeature.equalIdAndAttributes(live, original) ||
77         SecurityUtils.getVisibility(live) != SecurityUtils.getVisibility(original)) {
78       // write first to ensure feature is valid
79       writer(live)
80       deleter(original)
81     }
82     original = null
83     live = null
84   }
85 
86   override def remove(): Unit = {
87     if (original == null) {
88       throw new IllegalStateException("next() must be called before remove()")
89     }
90     deleter(original)
91     original = null
92     live = null
93   }
94 
95   override def flush(): Unit = FlushQuietly.raise(writer, deleter)
96 
97   override def close(): Unit = {
98     CloseWithLogging.raise(Seq(reader, writer, deleter))
99     if (!deleter.files.isEmpty) {
100       val delta = storage.table.newRowDelta()
101       // guard against concurrent commits that touch the same delete files
102       Option(storage.table.currentSnapshot()).foreach(s => delta.validateFromSnapshot(s.snapshotId()))
103       deleter.files.forEach(f => delta.addDeletes(f))
104       // remove any previous DVs whose deletes we merged into the newly written DVs, otherwise
105       // the commit will fail validation with multiple live DVs referencing the same data file
106       deleter.rewritten.forEach(f => delta.removeDeletes(f))
107       delta.commit()
108     }
109   }
110 }

114   /**
115    * Create a new update writer
116    *
117    * @param storage storage
118    * @param filter filter for features to update
119    * @param readThreads number of threads using to read features to update
120    * @param maxOpenPartitions max open writer partitions
121    * @return
122    */
123   def apply(storage: FileSystemStorage, filter: Filter, readThreads: Int, maxOpenPartitions: Int): IcebergUpdateWriter = {
124     // note: writer and deleter are Closeable, but they don't need to be closed unless they initialize at least one partition,
125     // so it's safe to not clean them up if there's an exception in this method
126     val writer = new MultiPartitionWriter(storage, maxOpenPartitions)
127     val deleter = new MultiPartitionDeleter(storage, maxOpenPartitions)
128     // note: this class expects the reader to return RecordSimpleFeatures, but this isn't enforced through the call signature
129     val reader = storage.getReader(new Query(storage.sft.getTypeName, filter), readThreads, forUpdate = true)
130     new IcebergUpdateWriter(storage, reader, writer, deleter)
131   }
132 
133   /**
134    * Multi-partition deleter
135    *
136    * @param storage file system storage instance
137    * @param maxOpenPartitions max open partition writers
138    */
139   private class MultiPartitionDeleter(storage: FileSystemStorage, maxOpenPartitions: Int)
140       extends MultiPartitionAction[DeleteWriter](storage, maxOpenPartitions) {
141 
142     // newly written delete files (DVs) to add on commit
143     val files: java.util.Set[DeleteFile] = Collections.newSetFromMap(new ConcurrentHashMap[DeleteFile, java.lang.Boolean]())
144     // previous DVs whose deletes we merged into the new DVs - these must be removed on commit
145     val rewritten: java.util.Set[DeleteFile] = Collections.newSetFromMap(new ConcurrentHashMap[DeleteFile, java.lang.Boolean]())
146 
147     private val closed = new AtomicBoolean(false)
148     private val fileSupplier: Supplier[OutputFile] = () => storage.table.io().newOutputFile(storage.newFilePath("puffin"))
149     // loads existing deletes for a data file so they can be merged into the new DV - a data file
150     // can only be referenced by a single DV, so we have to rewrite any existing one
151     private val writer = new BaseDVFileWriter(fileSupplier, new PreviousDeleteLoader(storage))
152 
153     override protected def createAction(partition: Partition): DeleteWriter =
154       new DeleteWriter(writer, storage.table.spec(), storage.metadata.partition(partition))
155     override protected def apply(action: DeleteWriter, feature: SimpleFeature): Unit =
156       action.apply(feature.asInstanceOf[StructSimpleFeature])
157 
158     override def close(): Unit = {
159       if (closed.compareAndSet(false, true)) {
160         try { super.close() } finally  {
161           CloseQuietly.raise(writer)
162         }
163         val result = writer.result()
164         files.addAll(result.deleteFiles())
165         rewritten.addAll(result.rewrittenDeleteFiles())
166       }
167     }
168   }
169 
170   /**
171    * Loads existing deletes (deletion vectors) for a given data file path, so they can be merged into a
172    * newly written DV. Returns null if there are no existing deletes for the given path.
173    *
174    * @param storage file system storage instance
175    */
176   private class PreviousDeleteLoader(storage: FileSystemStorage) extends Function[String, PositionDeleteIndex] {
177 
178     import scala.collection.JavaConverters._
179 
180     // map of data-file-path -> set of delete files that reference it, from the current table state
181     private lazy val deletesByPath: CharSequenceMap[DeleteFileSet] = {
182       val map = CharSequenceMap.create[DeleteFileSet]()
183       val snapshot = storage.table.currentSnapshot()
184       if (snapshot != null) {
185         val io = storage.table.io()
186         val specs = storage.table.specs()
187         snapshot.deleteManifests(io).asScala.foreach { manifest =>
188           WithClose(ManifestFiles.readDeleteManifest(manifest, io, specs)) { reader =>
189             reader.iterator().asScala.foreach { deleteFile =>
190               val ref = deleteFile.referencedDataFile()
191               if (ref != null) {
192                 // copy() to detach from the manifest reader, which reuses containers
193                 map.computeIfAbsent(ref, () => DeleteFileSet.create()).add(deleteFile.copy())
194               }
195             }
196           }
197         }
198       }
199       map
200     }
201 
202     private lazy val loader: DeleteLoader =
203       new BaseDeleteLoader(deleteFile => storage.table.io().newInputFile(deleteFile))
204 
205     override def apply(path: String): PositionDeleteIndex = {
206       val set = deletesByPath.get(path)
207       if (set == null) { null } else {
208         loader.loadPositionDeletes(set, path)
209       }
210     }
211   }
212

226     override def flush(): Unit = {}
227     override def close(): Unit = {}
228   }
