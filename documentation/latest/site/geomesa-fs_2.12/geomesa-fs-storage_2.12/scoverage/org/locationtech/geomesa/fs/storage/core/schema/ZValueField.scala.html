27   /**

29    *

35   /**
36    * Gets the column name for the bbox group field
37    *
38    * @param geom geometry column name
39    * @return

41   def z2FieldName(geom: String): String = s"${SimpleFeatureSchema.InternalFieldDelimiter}$geom$Z2ValueFieldSuffix"
42 
43   /**
44    * The parquet schema for a z-value field
45    *
46    * @param geom geometry column name
47    * @param geometryType geom type
48    * @return
49    */
50   def parquetSchema(geom: String, geometryType: ObjectType, fieldIds: AtomicInteger): PrimitiveType =
51     Types.optional(PrimitiveTypeName.BINARY).id(fieldIds.getAndIncrement()).as(LogicalTypeAnnotation.stringType()).named(name(geom, geometryType))
52 
53   /**
54    * The iceberg schema for a z-value field
55    *
56    * @param geom geometry column name
57    * @param geometryType geom type
58    * @return
59    */
60   def icebergSchema(geom: String, geometryType: ObjectType, fieldIds: AtomicInteger): NestedField =
61     NestedField.optional(name(geom, geometryType)).withId(fieldIds.getAndIncrement()).ofType(StringType.get()).build()
62 
63   private def name(geom: String, geometryType: ObjectType): String = geometryType match {
64     case ObjectType.POINT => z2FieldName(geom)
65     case _ => xz2FieldName(geom)
66   }
67 }
