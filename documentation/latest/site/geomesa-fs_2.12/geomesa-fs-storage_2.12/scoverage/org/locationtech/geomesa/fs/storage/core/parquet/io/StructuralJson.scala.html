19 /**
20  * Shared helpers for reading/writing structural JSON attributes. Leaf value formats (dates, times,
21  * timestamps, binary, uuids) mirror [[VariantJsonWriter]] so that structural and variant JSON round-trip
22  * consistently across the parquet and iceberg backends.
23  */
24 object StructuralJson {

26   private val encoder = Base64.getEncoder
27   private val decoder = Base64.getDecoder

29   /**

31    */
32   def compact(element: JsonElement): String = element.toString

38   def bytesToJson(bb: ByteBuffer): String = {
39     val dup = bb.duplicate()

49   /**
50    * Encodes a decimal as the two's-complement, sign-extended, fixed-length byte array used by
51    * parquet FIXED_LEN_BYTE_ARRAY decimals. Mirrors iceberg's DecimalUtil.toReusedFixLengthBytes.
52    *
53    * @param value decimal value
54    * @param scale schema scale - the value is rescaled to match, as required by the fixed encoding
55    * @param length fixed byte length from the schema
56    */
57   def decimalToFixedBytes(value: java.math.BigDecimal, scale: Int, length: Int): Array[Byte] = {
58     val scaled = value.setScale(scale)
59     val unscaled = scaled.unscaledValue().toByteArray
60     if (unscaled.length == length) { unscaled } else {
61       val buf = Array.ofDim[Byte](length)
62       val fill = if (scaled.signum() < 0) { 0xFF.toByte } else { 0x00.toByte }
63       val offset = length - unscaled.length
64       var i = 0
65       while (i < length) {
66         buf(i) = if (i < offset) { fill } else { unscaled(i - offset) }
67         i += 1
68       }
69       buf
70     }
71   }

75   def uuidToBytes(uuid: UUID): Array[Byte] = {
76     val bb = ByteBuffer.wrap(Array.ofDim[Byte](16))
77     bb.putLong(uuid.getMostSignificantBits)
78     bb.putLong(uuid.getLeastSignificantBits)
79     bb.array()
80   }

96   // ---- timestamp (in the given unit, adjusted to utc or not) ----
97 
98   def timestampToJson(value: Long, unit: TimeUnit, utc: Boolean): String = {
99     val offset = toInstant(value, unit).atOffset(ZoneOffset.UTC)
100     if (utc) {
101       offset.format(DateTimeFormatter.ISO_OFFSET_DATE_TIME)
102     } else {
103       offset.toLocalDateTime.format(DateTimeFormatter.ISO_LOCAL_DATE_TIME)
104     }
105   }
106   def timestampToJson(value: OffsetDateTime): String = value.format(DateTimeFormatter.ISO_OFFSET_DATE_TIME)
107   def timestampToJson(value: LocalDateTime): String = value.format(DateTimeFormatter.ISO_LOCAL_DATE_TIME)
108 
109   def jsonToTimestamp(value: String, unit: TimeUnit, utc: Boolean): Long = {
110     val instant =
111       if (utc) {
112         OffsetDateTime.parse(value, DateTimeFormatter.ISO_OFFSET_DATE_TIME).toInstant
113       } else {
114         LocalDateTime.parse(value, DateTimeFormatter.ISO_LOCAL_DATE_TIME).toInstant(ZoneOffset.UTC)
115       }
116     fromInstant(instant, unit)
117   }
118 
119   private def nanosPerUnit(unit: TimeUnit): Long = unit match {
120     case TimeUnit.MILLIS => 1000000L
121     case TimeUnit.MICROS => 1000L
122     case TimeUnit.NANOS  => 1L
123   }
124 
125   private def toInstant(value: Long, unit: TimeUnit): Instant = unit match {
126     case TimeUnit.MILLIS => Instant.ofEpochMilli(value)
127     case TimeUnit.MICROS => Instant.ofEpochSecond(value / 1000000L, (value % 1000000L) * 1000L)
128     case TimeUnit.NANOS  => Instant.ofEpochSecond(value / 1000000000L, value % 1000000000L)
129   }
130 
131   private def fromInstant(instant: Instant, unit: TimeUnit): Long = unit match {
132     case TimeUnit.MILLIS => instant.toEpochMilli
133     case TimeUnit.MICROS => instant.getEpochSecond * 1000000L + instant.getNano / 1000L
134     case TimeUnit.NANOS  => instant.getEpochSecond * 1000000000L + instant.getNano
135   }
