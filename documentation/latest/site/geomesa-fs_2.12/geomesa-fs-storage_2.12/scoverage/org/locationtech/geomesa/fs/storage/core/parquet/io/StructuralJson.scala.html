19 /**
20  * Shared helpers for reading/writing structural JSON attributes. Leaf value formats (dates, times,
21  * timestamps, binary, uuids) mirror [[VariantJsonWriter]] so that structural and variant JSON round-trip
22  * consistently across the parquet and iceberg backends.
23  */
24 object StructuralJson {
25 
26   private val encoder = Base64.getEncoder
27   private val decoder = Base64.getDecoder
28 
29   /**
30    * Serializes a gson element to a compact JSON string
31    */
32   def compact(element: JsonElement): AnyRef = {
33     if (element.isJsonNull) {
34       null
35     } else if (element.isJsonPrimitive) {
36       val primitive = element.getAsJsonPrimitive
37       if (primitive.isString) {
38         primitive.getAsString
39       } else if (primitive.isNumber) {
40         primitive.getAsNumber
41       } else if (primitive.isBoolean) {
42         Boolean.box(primitive.getAsBoolean)
43       } else {
44         primitive.toString
45       }
46     } else {
47       element.toString
48     }
49   }

51   // ---- binary ----
52 
53   def bytesToJson(bytes: Array[Byte]): String = encoder.encodeToString(bytes)
54 
55   def bytesToJson(bb: ByteBuffer): String = {
56     val dup = bb.duplicate()
57     val bytes = Array.ofDim[Byte](dup.remaining())
58     dup.get(bytes)
59     encoder.encodeToString(bytes)
60   }

62   def jsonToBytes(value: String): Array[Byte] = decoder.decode(value)

66   /**
67    * Encodes a decimal as the two's-complement, sign-extended, fixed-length byte array used by
68    * parquet FIXED_LEN_BYTE_ARRAY decimals. Mirrors iceberg's DecimalUtil.toReusedFixLengthBytes.
69    *
70    * @param value decimal value
71    * @param scale schema scale - the value is rescaled to match, as required by the fixed encoding
72    * @param length fixed byte length from the schema
73    */
74   def decimalToFixedBytes(value: java.math.BigDecimal, scale: Int, length: Int): Array[Byte] = {
75     val scaled = value.setScale(scale)
76     val unscaled = scaled.unscaledValue().toByteArray
77     if (unscaled.length == length) { unscaled } else {
78       val buf = Array.ofDim[Byte](length)
79       val fill = if (scaled.signum() < 0) { 0xFF.toByte } else { 0x00.toByte }
80       val offset = length - unscaled.length
81       var i = 0
82       while (i < length) {
83         buf(i) = if (i < offset) { fill } else { unscaled(i - offset) }
84         i += 1
85       }
86       buf
87     }
88   }

92   def uuidToBytes(uuid: UUID): Array[Byte] = {
93     val bb = ByteBuffer.wrap(Array.ofDim[Byte](16))
94     bb.putLong(uuid.getMostSignificantBits)
95     bb.putLong(uuid.getLeastSignificantBits)
96     bb.array()
97   }

101   def dateToJson(epochDay: Long): String = LocalDate.ofEpochDay(epochDay).format(DateTimeFormatter.ISO_LOCAL_DATE)
102   def dateToJson(date: LocalDate): String = date.format(DateTimeFormatter.ISO_LOCAL_DATE)
103   def jsonToEpochDay(value: String): Int = LocalDate.parse(value, DateTimeFormatter.ISO_LOCAL_DATE).toEpochDay.toInt

105   // ---- time (of day, in the given unit) ----
106 
107   def timeToJson(value: Long, unit: TimeUnit): String =
108     LocalTime.ofNanoOfDay(value * nanosPerUnit(unit)).format(DateTimeFormatter.ISO_LOCAL_TIME)
109   def timeToJson(time: LocalTime): String = time.format(DateTimeFormatter.ISO_LOCAL_TIME)
110   def jsonToTime(value: String, unit: TimeUnit): Long =
111     LocalTime.parse(value, DateTimeFormatter.ISO_LOCAL_TIME).toNanoOfDay / nanosPerUnit(unit)

113   // ---- timestamp (in the given unit, adjusted to utc or not) ----
114 
115   def timestampToJson(value: Long, unit: TimeUnit, utc: Boolean): String = {
116     val offset = toInstant(value, unit).atOffset(ZoneOffset.UTC)
117     if (utc) {
118       offset.format(DateTimeFormatter.ISO_OFFSET_DATE_TIME)
119     } else {
120       offset.toLocalDateTime.format(DateTimeFormatter.ISO_LOCAL_DATE_TIME)
121     }
122   }
123   def timestampToJson(value: OffsetDateTime): String = value.format(DateTimeFormatter.ISO_OFFSET_DATE_TIME)
124   def timestampToJson(value: LocalDateTime): String = value.format(DateTimeFormatter.ISO_LOCAL_DATE_TIME)
125 
126   def jsonToTimestamp(value: String, unit: TimeUnit, utc: Boolean): Long = {
127     val instant =
128       if (utc) {
129         OffsetDateTime.parse(value, DateTimeFormatter.ISO_OFFSET_DATE_TIME).toInstant
130       } else {
131         LocalDateTime.parse(value, DateTimeFormatter.ISO_LOCAL_DATE_TIME).toInstant(ZoneOffset.UTC)
132       }
133     fromInstant(instant, unit)
134   }
135 
136   private def nanosPerUnit(unit: TimeUnit): Long = unit match {
137     case TimeUnit.MILLIS => 1000000L
138     case TimeUnit.MICROS => 1000L
139     case TimeUnit.NANOS  => 1L
140   }
141 
142   private def toInstant(value: Long, unit: TimeUnit): Instant = unit match {
143     case TimeUnit.MILLIS => Instant.ofEpochMilli(value)
144     case TimeUnit.MICROS => Instant.ofEpochSecond(value / 1000000L, (value % 1000000L) * 1000L)
145     case TimeUnit.NANOS  => Instant.ofEpochSecond(value / 1000000000L, value % 1000000000L)
146   }
147 
148   private def fromInstant(instant: Instant, unit: TimeUnit): Long = unit match {
149     case TimeUnit.MILLIS => instant.toEpochMilli
150     case TimeUnit.MICROS => instant.getEpochSecond * 1000000L + instant.getNano / 1000L
151     case TimeUnit.NANOS  => instant.getEpochSecond * 1000000000L + instant.getNano
152   }
