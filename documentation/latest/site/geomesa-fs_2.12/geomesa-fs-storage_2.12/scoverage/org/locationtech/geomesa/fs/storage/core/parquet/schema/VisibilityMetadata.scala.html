17 /** Complete file-wide AND/OR rollups of the expressions written to the visibility column.
18   * Invalid expressions disable both proofs; they must never be dropped from a summary.
19   * Exceeding a bound discards the entire summary; a partial summary must never be published.
20   */
21 class VisibilityMetadata(fieldId: Int) {
22   import VisibilityMetadata._
23 
24   private val expressions = mutable.LinkedHashSet.empty[String]
25   private var rows = 0L
26   private var bytes = 0
27   private var valid = fieldId >= 0
28   private var unrestricted = false
29 
30   def observe(expression: String): Unit = {
31     rows += 1L
32     if (expression == null || expression.isEmpty) unrestricted = true
33     if (valid && expression != null && expression.nonEmpty && expressions.add(expression)) {
34       // Avoid allocating UTF-8 bytes when the character count already exceeds the limit.
35       bytes += (if (expression.length > MaxBytes) MaxBytes + 1 else expression.getBytes(UTF_8).length)
36       if (expressions.size > MaxExpressions || bytes > MaxBytes) {
37         discard()
38       }
39     }
40   }
41 
42   def metadata: Option[String] = {
43     if (!valid) { None } else {
44       try {
45         // Validate and normalize only at file close, after exact-string deduplication. Every
46         // operand is parenthesized so compound AND/OR expressions retain their meaning.
47         val operands = expressions.iterator.map(expression => s"($expression)").toSeq
48         val combined = operands.mkString("&")
49         if (!withinDepthLimit(combined)) {
50           return None
51         }
52         val allOf = VisibilityUtils.normalize(combined)
53         // Validate all operands above even when an unrestricted row makes the OR rollup true.
54         val anyOf = if (unrestricted) { "" } else { VisibilityUtils.normalize(operands.mkString("|")) }
55         val json = new JsonObject()
56         json.addProperty("version", 1)
57         json.addProperty("fieldId", fieldId)
58         json.addProperty("scope", "file")
59         json.addProperty("rowCount", rows)
60         json.addProperty("allOf", allOf)
61         json.addProperty("anyOf", anyOf)
62         val encoded = json.toString
63         if (encoded.getBytes(UTF_8).length > MaxBytes) None else Some(encoded)
64       } catch {
65         case _: IllegalArgumentException => None
66         case _: StackOverflowError => None // optional metadata must not fail file finalization
67       }
68     }
69   }
70 
71   private def discard(): Unit = {
72     valid = false
73     expressions.clear()
74   }
75 }
76 
77 object VisibilityMetadata {
78   val MetadataKey = "geomesa.visibility.rollup"
79   // Conservative application limits, not Parquet format limits: bound the number of operands
80   // and the memory used for optional visibility summaries. MaxBytes caps both accumulated
81   // UTF-8 expression bytes and the final JSON footer value, including rollups and escaping.
82   val MaxExpressions = 1024
83   val MaxBytes = 64 * 1024
84   val MaxDepth = 128
85 
86   /** Bound recursive parsing without counting parentheses inside quoted authorization labels. */
87   private def withinDepthLimit(expression: String): Boolean = {
88     var depth = 0
89     var quoted = false
90     var escaped = false
91     var i = 0
92     while (i < expression.length) {
93       val c = expression.charAt(i)
94       if (quoted) {
95         if (escaped) escaped = false
96         else if (c == '\\') escaped = true
97         else if (c == '"') quoted = false
98       } else if (c == '"') {
99         quoted = true
100       } else if (c == '(') {
101         depth += 1
102         if (depth > MaxDepth) return false
103       } else if (c == ')') {
104         depth -= 1
105         if (depth < 0) return false
106       }
107       i += 1
108     }
109     depth == 0 && !quoted
110   }
111 }
