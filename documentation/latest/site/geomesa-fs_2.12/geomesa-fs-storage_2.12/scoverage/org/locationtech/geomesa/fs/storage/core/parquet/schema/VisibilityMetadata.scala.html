17 /** Complete file-wide AND/OR rollups of the expressions written to the visibility column.
18   * Invalid expressions disable both proofs; they must never be dropped from a summary.
19   * Exceeding a bound discards the entire summary; a partial summary must never be published.
20   */
21 class VisibilityMetadata(fieldId: Int) {
22   import VisibilityMetadata._
23 
24   private val expressions = mutable.LinkedHashSet.empty[String]
25   private var rows = 0L
26   private var bytes = 0
27   private var valid = fieldId >= 0
28   private var unrestricted = false
29 
30   def observe(expression: String): Unit = {
31     rows += 1L
32     if (expression == null || expression.isEmpty) unrestricted = true
33     if (valid && expression != null && expression.nonEmpty && expressions.add(expression)) {
34       // Avoid allocating UTF-8 bytes when the character count already exceeds the limit.
35       bytes += (if (expression.length > MaxBytes) MaxBytes + 1 else expression.getBytes(UTF_8).length)
36       if (expressions.size > MaxExpressions || bytes > MaxBytes) {
37         discard()
38       }
39     }
40   }
41 
42   def metadata: Option[String] = {
43     if (!valid) { None } else {
44       try {
45         // Validate and normalize only at file close, after exact-string deduplication. Every
46         // operand is parenthesized so compound AND/OR expressions retain their meaning.
47         val operands = expressions.iterator.map(expression => s"($expression)").toSeq
48         val combined = operands.mkString("&")
49         if (!withinDepthLimit(combined)) return None
50         val allOf = AccessExpression.of(combined, true).getExpression
51         // Validate all operands above even when an unrestricted row makes the OR rollup true.
52         val anyOf = if (unrestricted) "" else AccessExpression.of(operands.mkString("|"), true).getExpression
53         val json = new JsonObject()
54         json.addProperty("version", 1)
55         json.addProperty("fieldId", fieldId)
56         json.addProperty("scope", "file")
57         json.addProperty("rowCount", rows)
58         json.addProperty("allOf", allOf)
59         json.addProperty("anyOf", anyOf)
60         val encoded = json.toString
61         if (encoded.getBytes(UTF_8).length > MaxBytes) None else Some(encoded)
62       } catch {
63         case _: IllegalAccessExpressionException => None
64         case _: StackOverflowError => None // optional metadata must not fail file finalization
65       }
66     }
67   }
68 
69   private def discard(): Unit = {
70     valid = false
71     expressions.clear()
72   }
73 }
74 
75 object VisibilityMetadata {
76   val MetadataKey = "geomesa.visibility.rollup"
77   // Conservative application limits, not Parquet format limits: bound the number of operands
78   // and the memory used for optional visibility summaries. MaxBytes caps both accumulated
79   // UTF-8 expression bytes and the final JSON footer value, including rollups and escaping.
80   val MaxExpressions = 1024
81   val MaxBytes = 64 * 1024
82   val MaxDepth = 128
83 
84   /** Bound recursive parsing without counting parentheses inside quoted authorization labels. */
85   def withinDepthLimit(expression: String): Boolean = {
86     var depth = 0
87     var quoted = false
88     var escaped = false
89     var i = 0
90     while (i < expression.length) {
91       val c = expression.charAt(i)
92       if (quoted) {
93         if (escaped) escaped = false
94         else if (c == '\\') escaped = true
95         else if (c == '"') quoted = false
96       } else if (c == '"') {
97         quoted = true
98       } else if (c == '(') {
99         depth += 1
100         if (depth > MaxDepth) return false
101       } else if (c == ')') {
102         depth -= 1
103         if (depth < 0) return false
104       }
105       i += 1
106     }
107     depth == 0 && !quoted
108   }
109 }
