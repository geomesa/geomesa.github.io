135 
136     val configured = QueryRunner.configureQuery(sft, query)
137     val filter = Option(configured.getFilter).getOrElse(Filter.INCLUDE)
138     val icebergFilter = IcebergFilterConverter(sft, schemes, filter)
139     val visFilter = VisibilityUtils.visible(authProvider)
140     val transform = configured.getHints.getTransform
141     val includeFids = configured.getHints.isIncludeFid
142     val sort = configured.getHints.getSortFields
143     val max = configured.getHints.getMaxFeatures
144     val readSchema = schema.read(transform.map(_._1), icebergFilter.columns, includeFids, forUpdate)
145 
146     logger.debug(s"Running query '${query.getTypeName}' ${ECQL.toCQL(filter)}")
147     logger.debug(s"  Original filter: ${ECQL.toCQL(query.getFilter)}")
148     logger.debug(s"  Push-down filter: ${icebergFilter.expression}")
149     logger.debug(s"  Client-side filter: ${icebergFilter.remainder.fold("none")(ECQL.toCQL)}")
150     logger.debug(s"  Transforms: ${transform.fold("none") { case (t, _) => if (t.isEmpty) { "empty" } else { t }}}, with${if (includeFids) { "" } else { "out" }} FIDs")
151     logger.debug(s"  Read schema: ${readSchema.schema}")
152     logger.debug(s"  Sort: ${sort.fold("none") { fields => fields.map { case (f, rev) => s"$f ${if (rev) "descending" else ""}"}.mkString(", ")}}")
153     logger.debug(s"  Max features: ${max.getOrElse("none")}")
154 
155     val remainingFilter = icebergFilter.remainder.map(FastFilterFactory.optimize(readSchema.sft, _))
156     val transformer = transform.flatMap { case (tdefs, tsft) =>
157       val transforms = Transforms(readSchema.sft, tdefs).toArray
158       if (tsft == readSchema.sft && transforms.forall(_.isInstanceOf[PropertyTransform])) {
159         // simple case where transform is handled by the iceberg scan
160         None
161       } else {
162         // need to evaluate transform expressions
163         Some(new TransformSimpleFeature(tsft, transforms))
164       }
165     }
166 
167     val scan = new IcebergParquetScan(table, readSchema, icebergFilter.expression, threads)
168     try {
169       val visible = scan.filter(visFilter.apply)
170       val filtered = remainingFilter.fold(visible)(f => visible.filter(f.evaluate))
171       val transformed = transformer.fold(filtered)(t => filtered.map(t.setFeature))
172       // note - have to copy the features since sorting will not just be sequential access
173       val sorted = sort.fold(transformed)(new SortingSimpleFeatureIterator(transformed.map(ScalaSimpleFeature.copy), _))
174       val limited = max.fold(sorted)(m => sorted.take(m))
175       limited
176     } catch {
177       case NonFatal(e) => CloseWithLogging(scan); throw e
178     }
179   }

399   /**
400    * Writes files up to a given size, then starts a new file
401    *
402    * @param writers iterator of files to write
403    * @param estimator target file size estimator
404    */
405   // noinspection ScalaWeakerAccess
406   class ChunkedFileSystemWriter(writers: Iterator[FileSystemWriter], estimator: UpdatingFileSizeEstimator)
407       extends FileSystemWriter {
408 
409     private var totalCount = 0L // total number of features written across all chunks
410     private var totalBytes = 0L // sum size of all finished chunks
411     private var remaining = estimator.estimate(0L)
412     private var writer: FileSystemWriter = _
413 
414     override def write(feature: SimpleFeature): Unit = {
415       if (writer == null) {
416         writer = writers.next()
417       }
418       writer.write(feature)
419       totalCount += 1
420       remaining -= 1
421       if (remaining == 0) {
422         val dataSize = writer.size
423         if (estimator.done(dataSize)) {
424           writer.close()
425           totalBytes += writer.size // re-calculate now that writer is closed, so we get the final, accurate size
426           writer = null
427           // adjust our estimate to account for the actual bytes written
428           estimator.update(totalBytes, totalCount)
429           remaining = estimator.estimate(0L)
430         } else {
431           remaining = math.max(100L, estimator.estimate(dataSize))
432         }
433       }
434     }
435 
436     override def size: Long = totalBytes + Option(writer).fold(0L)(_.size)
437 
438     override def flush(): Unit = if (writer != null) { writer.flush() }
439 
440     override def close(): Unit = {
441       if (writer != null) {
442         writer.close()
443       }
444       estimator.close()
445     }
446   }

480    *
481    * @param typeName simple feature type name
482    * @return

484   def newFilePath(typeName: String, ext: String = "parquet"): String = {
485     val filename = s"${ColumnName.encode(typeName).take(20)}_${UUID.randomUUID().toString.replaceAllLiterally("-", "")}.$ext"
486     // partitioning logic taken from Apache Iceberg: https://iceberg.apache.org/docs/nightly/aws/#object-store-file-layout
487     val hash = {
488       val bytes = filename.getBytes(StandardCharsets.UTF_8)
489       val hash = MurmurHash3.hash32x86(bytes, 0, bytes.length, 0)
490       // Integer#toBinaryString excludes leading zeros, which we want to preserve
491       Integer.toBinaryString(hash | Integer.MIN_VALUE)
492     }
493     s"${hash.substring(0, 4)}/${hash.substring(4, 8)}/${hash.substring(8, 12)}/${hash.substring(12, 20)}/$filename"
494   }
