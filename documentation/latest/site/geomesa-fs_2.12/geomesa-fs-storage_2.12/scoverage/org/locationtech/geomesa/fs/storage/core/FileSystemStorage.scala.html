135    * @param query query
136    * @param threads read threads
137    * @param forUpdate include row and file data required for positional deletes
138    * @return
139    */
140   private[core] def getReader(query: Query, threads: Int, forUpdate: Boolean): CloseableIterator[SimpleFeature] = {
141     val configured = QueryRunner.configureQuery(sft, query)
142     val filter = Option(configured.getFilter).getOrElse(Filter.INCLUDE)
143     val icebergFilter = IcebergFilterConverter(schema, schemes, filter)
144     val visFilter = VisibilityUtils.visible(authProvider) // TODO can optimize out on a per-file basis based on metadata
145     val transform = configured.getHints.getTransform
146     val includeFids = configured.getHints.isIncludeFid
147     val sort = configured.getHints.getSortFields
148     val max = configured.getHints.getMaxFeatures
149     val readSchema = schema.read(transform, icebergFilter.remainder, icebergFilter.columns, includeFids, forUpdate)
150 
151     logger.debug(s"Running query '${query.getTypeName}' ${ECQL.toCQL(filter)}")
152     logger.debug(s"  Original filter: ${ECQL.toCQL(query.getFilter)}")
153     logger.debug(s"  Push-down filter: ${icebergFilter.expression}")
154     logger.debug(s"  Client-side filter: ${icebergFilter.remainder.fold("none")(ECQL.toCQL)}")
155     logger.debug(s"  Transforms: ${transform.fold("none") { case (t, _) => if (t.isEmpty) { "empty" } else { t }}}, with${if (includeFids) { "" } else { "out" }} FIDs")
156     logger.debug(s"  Read schema: ${readSchema.schema.schema}")
157     logger.debug(s"  Sort: ${sort.fold("none") { fields => fields.map { case (f, rev) => s"$f ${if (rev) "descending" else ""}"}.mkString(", ")}}")
158     logger.debug(s"  Max features: ${max.getOrElse("none")}")
159 
160     val remainingFilter = icebergFilter.remainder.map(FastFilterFactory.optimize(readSchema.schema.sft, _))
161     val transformer = readSchema.transforms.map { case (tsft, transforms) => new TransformSimpleFeature(tsft, transforms) }
162 
163     val scan = new IcebergParquetScan(table, readSchema.schema, icebergFilter.expression, threads)
164     try {
165       val visible = scan.filter(visFilter.apply)
166       val filtered = remainingFilter.fold(visible)(f => visible.filter(f.evaluate))
167       val transformed = transformer.fold(filtered)(t => filtered.map(t.setFeature))
168       // note - have to copy the features since sorting will not just be sequential access
169       val sorted = sort.fold(transformed)(new SortingSimpleFeatureIterator(transformed.map(ScalaSimpleFeature.copy), _))
170       val limited = max.fold(sorted)(m => sorted.take(m))
171       limited
172     } catch {
173       case NonFatal(e) => CloseWithLogging(scan); throw e
174     }
175   }

212   /**
213    * Read data files with visibilities. For optimized cases, operate directly on files that are wholly visible to the user.
214    * For other cases, we have to read the file and evaluate row-by-row.
215    *
216    * @param filter filter
217    * @param threads read threads
218    * @param ops operations to run
219    */
220   private def fileOps[T](filter: Filter, threads: Int, ops: FileOps[T]): T = {
221     val remainingFiles = ArrayBuffer.empty[String]
222     val visFieldId = schema.schema.findField(SimpleFeatureSchema.VisibilitiesField).fieldId()
223     val visFilter = VisibilityUtils.check(authProvider)
224 
225     val query = QueryRunner.configureQuery(sft, new Query(sft.getTypeName, filter, ops.readAttributes: _*))
226 
227     metadata.files().includeFileStats().forFilter(query.getFilter).scan().foreach { task =>
228       val file = task.file()
229       val nullValueCount = file.nullValueCounts().get(visFieldId)
230       if (nullValueCount != null && nullValueCount == file.valueCounts().get(visFieldId)) {
231         // all nulls
232         if (!ops.fileOp(file, task.deletes().asScala.toSeq)) {
233           remainingFiles += file.location()
234         }
235       } else {
236         val lowerBound = StringParser.apply(file.lowerBounds().get(visFieldId))
237         if (lowerBound != null && lowerBound == StringParser.apply(file.upperBounds().get(visFieldId))) {
238           // only one vis marking, we can evaluate it here
239           if (visFilter.apply(lowerBound)) {
240             if (!ops.fileOp(file, task.deletes().asScala.toSeq)) {
241               remainingFiles += file.location()
242             }
243           }
244         } else {
245           // TODO can read vis metadata from the footer and potentially apply the file-level op
246           remainingFiles += file.location()
247         }
248       }
249     }
250 
251     if (remainingFiles.nonEmpty) {
252       val icebergFilter = IcebergFilterConverter(schema, schemes, query.getFilter)
253       val transform = query.getHints.getTransform
254       val readSchema = schema.read(transform, icebergFilter.remainder, icebergFilter.columns, includeFids = false)
255       val fileFilter: Option[String => Boolean] = Some(remainingFiles.contains)
256       WithClose(new IcebergParquetScan(table, readSchema.schema, icebergFilter.expression, threads, fileFilter)) { scan =>
257         val features = icebergFilter.remainder.fold[CloseableIterator[SimpleFeature]](scan)(f => scan.filter(f.evaluate))
258         features.foreach { f =>
259           if (visFilter.apply(SecurityUtils.getVisibility(f))) {
260             ops.featureOp(f)
261           }
262         }
263       }
264     }
265 
266     ops.result
267   }

446   /**
447    * Writes files up to a given size, then starts a new file
448    *
449    * @param writers iterator of files to write
450    * @param estimator target file size estimator
451    */
452   // noinspection ScalaWeakerAccess
453   class ChunkedFileSystemWriter(writers: Iterator[FileSystemWriter], estimator: UpdatingFileSizeEstimator)
454       extends FileSystemWriter {
455 
456     private var totalCount = 0L // total number of features written across all chunks
457     private var totalBytes = 0L // sum size of all finished chunks
458     private var remaining = estimator.estimate(0L)
459     private var writer: FileSystemWriter = _
460 
461     override def write(feature: SimpleFeature): Unit = {
462       if (writer == null) {
463         writer = writers.next()
464       }
465       writer.write(feature)
466       totalCount += 1
467       remaining -= 1
468       if (remaining == 0) {
469         val dataSize = writer.size
470         if (estimator.done(dataSize)) {
471           writer.close()
472           totalBytes += writer.size // re-calculate now that writer is closed, so we get the final, accurate size
473           writer = null
474           // adjust our estimate to account for the actual bytes written
475           estimator.update(totalBytes, totalCount)
476           remaining = estimator.estimate(0L)
477         } else {
478           remaining = math.max(100L, estimator.estimate(dataSize))
479         }
480       }
481     }
482 
483     override def size: Long = totalBytes + Option(writer).fold(0L)(_.size)
484 
485     override def flush(): Unit = if (writer != null) { writer.flush() }
486 
487     override def close(): Unit = {
488       if (writer != null) {
489         writer.close()
490       }
491       estimator.close()
492     }
493   }

526    * Get the path for a new data file, using Iceberg FileContent semantics
527    *
528    * @param typeName simple feature type name
529    * @return
530    */
531   def newFilePath(typeName: String, ext: String = "parquet"): String = {
532     val filename = s"${ColumnName.encode(typeName).take(20)}_${UUID.randomUUID().toString.replaceAllLiterally("-", "")}.$ext"
533     // partitioning logic taken from Apache Iceberg: https://iceberg.apache.org/docs/nightly/aws/#object-store-file-layout
534     val hash = {
535       val bytes = filename.getBytes(StandardCharsets.UTF_8)
536       val hash = MurmurHash3.hash32x86(bytes, 0, bytes.length, 0)
537       // Integer#toBinaryString excludes leading zeros, which we want to preserve
538       Integer.toBinaryString(hash | Integer.MIN_VALUE)
539     }
540     s"${hash.substring(0, 4)}/${hash.substring(4, 8)}/${hash.substring(8, 12)}/${hash.substring(12, 20)}/$filename"
541   }
