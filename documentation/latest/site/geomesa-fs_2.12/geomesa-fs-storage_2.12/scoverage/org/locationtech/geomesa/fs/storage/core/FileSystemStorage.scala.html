125    * @param forUpdate include row and file data required for positional deletes
126    * @return

128   private[core] def getReader(query: Query, threads: Int, forUpdate: Boolean): CloseableIterator[SimpleFeature] = {
129     import org.locationtech.geomesa.index.conf.QueryHints.RichHints
130 
131     val configured = QueryRunner.configureQuery(sft, query)
132     val filter = Option(configured.getFilter).getOrElse(Filter.INCLUDE)
133     val icebergFilter = IcebergFilterConverter(sft, schemes, filter)
134     val visFilter = VisibilityUtils.visible(authProvider)
135     val transform = configured.getHints.getTransform
136     val includeFids = configured.getHints.isIncludeFid
137     val sort = configured.getHints.getSortFields
138     val max = configured.getHints.getMaxFeatures
139     val readSchema = schema.read(transform.map(_._1), icebergFilter.columns, includeFids, forUpdate)
140 
141     logger.debug(s"Running query '${query.getTypeName}' ${ECQL.toCQL(filter)}")
142     logger.debug(s"  Original filter: ${ECQL.toCQL(query.getFilter)}")
143     logger.debug(s"  Push-down filter: ${icebergFilter.expression}")
144     logger.debug(s"  Client-side filter: ${icebergFilter.remainder.fold("none")(ECQL.toCQL)}")
145     logger.debug(s"  Transforms: ${transform.fold("none") { case (t, _) => if (t.isEmpty) { "empty" } else { t }}}, with${if (includeFids) { "" } else { "out" }} FIDs")
146     logger.debug(s"  Read schema: ${readSchema.schema}")
147     logger.debug(s"  Sort: ${sort.fold("none") { fields => fields.map { case (f, rev) => s"$f ${if (rev) "descending" else ""}"}.mkString(", ")}}")
148     logger.debug(s"  Max features: ${max.getOrElse("none")}")
149 
150     val remainingFilter = icebergFilter.remainder.map(FastFilterFactory.optimize(readSchema.sft, _))
151     val transformer = transform.flatMap { case (tdefs, tsft) =>
152       val transforms = Transforms(readSchema.sft, tdefs).toArray
153       if (tsft == readSchema.sft && transforms.forall(_.isInstanceOf[PropertyTransform])) {
154         // simple case where transform is handled by the iceberg scan
155         None
156       } else {
157         // need to evaluate transform expressions
158         Some(new TransformSimpleFeature(tsft, transforms))
159       }
160     }
161 
162     val scan = new IcebergParquetScan(table, readSchema, icebergFilter.expression, threads)
163     try {
164       val visible = scan.filter(visFilter.apply)
165       val filtered = remainingFilter.fold(visible)(f => visible.filter(f.evaluate))
166       val transformed = transformer.fold(filtered)(t => filtered.map(t.setFeature))
167       // note - have to copy the features since sorting will not just be sequential access
168       val sorted = sort.fold(transformed)(new SortingSimpleFeatureIterator(transformed.map(ScalaSimpleFeature.copy), _))
169       val limited = max.fold(sorted)(m => sorted.take(m))
170       limited
171     } catch {
172       case NonFatal(e) => CloseWithLogging(scan); throw e
173     }
174   }

392   /**
393    * Writes files up to a given size, then starts a new file
394    *
395    * @param writers iterator of files to write
396    * @param estimator target file size estimator
397    */
398   // noinspection ScalaWeakerAccess
399   class ChunkedFileSystemWriter(writers: Iterator[FileSystemWriter], estimator: UpdatingFileSizeEstimator)
400       extends FileSystemWriter {
401 
402     private var totalCount = 0L // total number of features written across all chunks
403     private var totalBytes = 0L // sum size of all finished chunks
404     private var remaining = estimator.estimate(0L)
405     private var writer: FileSystemWriter = _
406 
407     override def write(feature: SimpleFeature): Unit = {
408       if (writer == null) {
409         writer = writers.next()
410       }
411       writer.write(feature)
412       totalCount += 1
413       remaining -= 1
414       if (remaining == 0) {
415         val dataSize = writer.size
416         if (estimator.done(dataSize)) {
417           writer.close()
418           totalBytes += writer.size // re-calculate now that writer is closed, so we get the final, accurate size
419           writer = null
420           // adjust our estimate to account for the actual bytes written
421           estimator.update(totalBytes, totalCount)
422           remaining = estimator.estimate(0L)
423         } else {
424           remaining = math.max(100L, estimator.estimate(dataSize))
425         }
426       }
427     }
428 
429     override def size: Long = totalBytes + Option(writer).fold(0L)(_.size)
430 
431     override def flush(): Unit = if (writer != null) { writer.flush() }
432 
433     override def close(): Unit = {
434       if (writer != null) {
435         writer.close()
436       }
437       estimator.close()
438     }
439   }

472    * Get the path for a new data file, using Iceberg FileContent semantics
473    *
474    * @param typeName simple feature type name
475    * @return
476    */
477   def newFilePath(typeName: String, prefix: String = ""): String = {
478     val filename =
479       s"${ColumnName.encode(typeName).take(20)}_$prefix${UUID.randomUUID().toString.replaceAllLiterally("-", "")}.parquet"
480     // partitioning logic taken from Apache Iceberg: https://iceberg.apache.org/docs/nightly/aws/#object-store-file-layout
481     val hash = {
482       val bytes = filename.getBytes(StandardCharsets.UTF_8)
483       val hash = MurmurHash3.hash32x86(bytes, 0, bytes.length, 0)
484       // Integer#toBinaryString excludes leading zeros, which we want to preserve
485       Integer.toBinaryString(hash | Integer.MIN_VALUE)
486     }
487     s"${hash.substring(0, 4)}/${hash.substring(4, 8)}/${hash.substring(8, 12)}/${hash.substring(12, 20)}/$filename"
488   }
