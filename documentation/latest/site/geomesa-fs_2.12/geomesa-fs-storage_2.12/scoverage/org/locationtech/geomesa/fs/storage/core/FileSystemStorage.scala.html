149     logger.debug(s"  Transforms: ${transform.fold("none") { case (t, _) => if (t.isEmpty) { "empty" } else { t }}}, with${if (includeFids) { "" } else { "out" }} FIDs")

163       val sorted = sort.fold(transformed)(new SortingSimpleFeatureIterator(transformed.map(ScalaSimpleFeature.copy), _))

389   /**
390    * Writes files up to a given size, then starts a new file
391    *
392    * @param writers iterator of files to write
393    * @param estimator target file size estimator
394    */
395   // noinspection ScalaWeakerAccess
396   class ChunkedFileSystemWriter(writers: Iterator[FileSystemWriter], estimator: UpdatingFileSizeEstimator)
397       extends FileSystemWriter {
398 
399     private var totalCount = 0L // total number of features written across all chunks
400     private var totalBytes = 0L // sum size of all finished chunks
401     private var remaining = estimator.estimate(0L)
402     private var writer: FileSystemWriter = _
403 
404     override def write(feature: SimpleFeature): Unit = {
405       if (writer == null) {
406         writer = writers.next()
407       }
408       writer.write(feature)
409       totalCount += 1
410       remaining -= 1
411       if (remaining == 0) {
412         val dataSize = writer.size
413         if (estimator.done(dataSize)) {
414           writer.close()
415           totalBytes += writer.size // re-calculate now that writer is closed, so we get the final, accurate size
416           writer = null
417           // adjust our estimate to account for the actual bytes written
418           estimator.update(totalBytes, totalCount)
419           remaining = estimator.estimate(0L)
420         } else {
421           remaining = math.max(100L, estimator.estimate(dataSize))
422         }
423       }
424     }
425 
426     override def size: Long = totalBytes + Option(writer).fold(0L)(_.size)
427 
428     override def flush(): Unit = if (writer != null) { writer.flush() }
429 
430     override def close(): Unit = {
431       if (writer != null) {
432         writer.close()
433       }
434       estimator.close()
435     }
436   }

469    * Get the path for a new data file, using Iceberg FileContent semantics
470    *
471    * @param typeName simple feature type name
472    * @return
473    */
474   def newFilePath(typeName: String, ext: String = "parquet"): String = {
475     val filename = s"${ColumnName.encode(typeName).take(20)}_${UUID.randomUUID().toString.replaceAllLiterally("-", "")}.$ext"
476     // partitioning logic taken from Apache Iceberg: https://iceberg.apache.org/docs/nightly/aws/#object-store-file-layout
477     val hash = {
478       val bytes = filename.getBytes(StandardCharsets.UTF_8)
479       val hash = MurmurHash3.hash32x86(bytes, 0, bytes.length, 0)
480       // Integer#toBinaryString excludes leading zeros, which we want to preserve
481       Integer.toBinaryString(hash | Integer.MIN_VALUE)
482     }
483     s"${hash.substring(0, 4)}/${hash.substring(4, 8)}/${hash.substring(8, 12)}/${hash.substring(12, 20)}/$filename"
484   }
