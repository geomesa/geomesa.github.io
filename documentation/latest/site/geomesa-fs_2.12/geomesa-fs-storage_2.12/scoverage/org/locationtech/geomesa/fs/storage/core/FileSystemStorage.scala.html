136    * @param forUpdate include row and file data required for positional deletes
137    * @return
138    */
139   private[core] def getReader(query: Query, threads: Int, forUpdate: Boolean): CloseableIterator[SimpleFeature] = {
140     val configured = QueryRunner.configureQuery(sft, query)
141     val filter = Option(configured.getFilter).getOrElse(Filter.INCLUDE)
142     val icebergFilter = IcebergFilterConverter(schema, schemes, filter)
143     val visFilter = VisibilityUtils.visible(authProvider)
144     val transform = configured.getHints.getTransform
145     val includeFids = configured.getHints.isIncludeFid
146     val sort = configured.getHints.getSortFields
147     val max = configured.getHints.getMaxFeatures
148     val readSchema = schema.read(transform, icebergFilter.remainder, icebergFilter.columns, includeFids, forUpdate)
149 
150     logger.debug(s"Running query '${query.getTypeName}' ${ECQL.toCQL(filter)}")
151     logger.debug(s"  Original filter: ${ECQL.toCQL(query.getFilter)}")
152     logger.debug(s"  Push-down filter: ${icebergFilter.expression}")
153     logger.debug(s"  Client-side filter: ${icebergFilter.remainder.fold("none")(ECQL.toCQL)}")
154     logger.debug(s"  Transforms: ${transform.fold("none") { case (t, _) => if (t.isEmpty) { "empty" } else { t }}}, with${if (includeFids) { "" } else { "out" }} FIDs")
155     logger.debug(s"  Read schema: ${readSchema.schema.schema}")
156     logger.debug(s"  Sort: ${sort.fold("none") { fields => fields.map { case (f, rev) => s"$f ${if (rev) "descending" else ""}"}.mkString(", ")}}")
157     logger.debug(s"  Max features: ${max.getOrElse("none")}")
158 
159     val remainingFilter = icebergFilter.remainder.map(FastFilterFactory.optimize(readSchema.schema.sft, _))
160     val transformer = readSchema.transforms.map { case (tsft, transforms) => new TransformSimpleFeature(tsft, transforms) }
161 
162     val scan = new IcebergParquetScan(table, readSchema.schema, icebergFilter.expression, threads)
163     try {
164       val visible = scan.filter(visFilter.apply)
165       val filtered = remainingFilter.fold(visible)(f => visible.filter(f.evaluate))
166       val transformed = transformer.fold(filtered)(t => filtered.map(t.setFeature))
167       // note - have to copy the features since sorting will not just be sequential access
168       val sorted = sort.fold(transformed)(new SortingSimpleFeatureIterator(transformed.map(ScalaSimpleFeature.copy), _))
169       val limited = max.fold(sorted)(m => sorted.take(m))
170       limited
171     } catch {
172       case NonFatal(e) => CloseWithLogging(scan); throw e
173     }
174   }

211   /**
212    * Read data files with visibilities. For optimized cases, operate directly on files that are wholly visible to the user.
213    * For other cases, we have to read the file and evaluate row-by-row.
214    *
215    * @param filter filter
216    * @param threads read threads
217    * @param ops operations to run
218    */
219   private def fileOps[T](filter: Filter, threads: Int, ops: FileOps[T]): T = {
220     val remainingFiles = ArrayBuffer.empty[String]
221     val visFieldId = schema.schema.findField(SimpleFeatureSchema.VisibilitiesField).fieldId()
222     val visFilter = VisibilityUtils.check(authProvider)
223 
224     val query = QueryRunner.configureQuery(sft, new Query(sft.getTypeName, filter, ops.readAttributes: _*))
225 
226     metadata.files().includeFileStats().forFilter(query.getFilter).scan().foreach { task =>
227       val file = task.file()
228       val nullValueCount = file.nullValueCounts().get(visFieldId)
229       if (nullValueCount != null && nullValueCount == file.valueCounts().get(visFieldId)) {
230         // all nulls
231         if (!ops.fileOp(file, task.deletes().asScala.toSeq)) {
232           remainingFiles += file.location()
233         }
234       } else {
235         val lowerBound = StringParser.apply(file.lowerBounds().get(visFieldId))
236         if (lowerBound != null && lowerBound == StringParser.apply(file.upperBounds().get(visFieldId))) {
237           // only one vis marking, we can evaluate it here
238           if (visFilter.apply(lowerBound)) {
239             if (!ops.fileOp(file, task.deletes().asScala.toSeq)) {
240               remainingFiles += file.location()
241             }
242           }
243         } else {
244           remainingFiles += file.location()
245         }
246       }
247     }
248 
249     if (remainingFiles.nonEmpty) {
250       val icebergFilter = IcebergFilterConverter(schema, schemes, query.getFilter)
251       val transform = query.getHints.getTransform
252       val readSchema = schema.read(transform, icebergFilter.remainder, icebergFilter.columns, includeFids = false)
253       val fileFilter: Option[String => Boolean] = Some(remainingFiles.contains)
254       WithClose(new IcebergParquetScan(table, readSchema.schema, icebergFilter.expression, threads, fileFilter)) { scan =>
255         val features = icebergFilter.remainder.fold[CloseableIterator[SimpleFeature]](scan)(f => scan.filter(f.evaluate))
256         features.foreach { f =>
257           if (visFilter.apply(SecurityUtils.getVisibility(f))) {
258             ops.featureOp(f)
259           }
260         }
261       }
262     }
263 
264     ops.result
265   }

269    * but a given feature can only be appended/modified in a single thread, otherwise the behavior is undefined.

275     val conf = this.conf ++ Map(SimpleFeatureSchema.PartitionKey -> partition.toString)

282       }
283       ParquetFileSystemWriter(schema, conf, table.io(), path, observer)
284     }
285 
286     sizer.targetSize match {
287       case None => newWriter()
288       case Some(s) => new ChunkedFileSystemWriter(Iterator.continually(newWriter()), sizer.estimator(s))

442   /**
443    * Writes files up to a given size, then starts a new file
444    *
445    * @param writers iterator of files to write
446    * @param estimator target file size estimator
447    */
448   // noinspection ScalaWeakerAccess
449   class ChunkedFileSystemWriter(writers: Iterator[FileSystemWriter], estimator: UpdatingFileSizeEstimator)
450       extends FileSystemWriter {
451 
452     private var totalCount = 0L // total number of features written across all chunks
453     private var totalBytes = 0L // sum size of all finished chunks
454     private var remaining = estimator.estimate(0L)
455     private var writer: FileSystemWriter = _
456 
457     override def write(feature: SimpleFeature): Unit = {
458       if (writer == null) {
459         writer = writers.next()
460       }
461       writer.write(feature)
462       totalCount += 1
463       remaining -= 1
464       if (remaining == 0) {
465         val dataSize = writer.size
466         if (estimator.done(dataSize)) {
467           writer.close()
468           totalBytes += writer.size // re-calculate now that writer is closed, so we get the final, accurate size
469           writer = null
470           // adjust our estimate to account for the actual bytes written
471           estimator.update(totalBytes, totalCount)
472           remaining = estimator.estimate(0L)
473         } else {
474           remaining = math.max(100L, estimator.estimate(dataSize))
475         }
476       }
477     }
478 
479     override def size: Long = totalBytes + Option(writer).fold(0L)(_.size)
480 
481     override def flush(): Unit = if (writer != null) { writer.flush() }
482 
483     override def close(): Unit = {
484       if (writer != null) {
485         writer.close()
486       }
487       estimator.close()
488     }
489   }

521   /**
522    * Get the path for a new data file, using Iceberg FileContent semantics
523    *
524    * @param typeName simple feature type name
525    * @return
526    */
527   def newFilePath(typeName: String, ext: String = "parquet"): String = {
528     val filename = s"${ColumnName.encode(typeName).take(20)}_${UUID.randomUUID().toString.replaceAllLiterally("-", "")}.$ext"
529     // partitioning logic taken from Apache Iceberg: https://iceberg.apache.org/docs/nightly/aws/#object-store-file-layout
530     val hash = {
531       val bytes = filename.getBytes(StandardCharsets.UTF_8)
532       val hash = MurmurHash3.hash32x86(bytes, 0, bytes.length, 0)
533       // Integer#toBinaryString excludes leading zeros, which we want to preserve
534       Integer.toBinaryString(hash | Integer.MIN_VALUE)
535     }
536     s"${hash.substring(0, 4)}/${hash.substring(4, 8)}/${hash.substring(8, 12)}/${hash.substring(12, 20)}/$filename"
537   }
