397   /**
398    * Writes files up to a given size, then starts a new file
399    *
400    * @param writers iterator of files to write
401    * @param estimator target file size estimator
402    */
403   // noinspection ScalaWeakerAccess
404   class ChunkedFileSystemWriter(writers: Iterator[FileSystemWriter], estimator: UpdatingFileSizeEstimator)
405       extends FileSystemWriter {
406 
407     private var totalCount = 0L // total number of features written across all chunks
408     private var totalBytes = 0L // sum size of all finished chunks
409     private var remaining = estimator.estimate(0L)
410     private var writer: FileSystemWriter = _
411 
412     override def write(feature: SimpleFeature): Unit = {
413       if (writer == null) {
414         writer = writers.next()
415       }
416       writer.write(feature)
417       totalCount += 1
418       remaining -= 1
419       if (remaining == 0) {
420         val dataSize = writer.size
421         if (estimator.done(dataSize)) {
422           writer.close()
423           totalBytes += writer.size // re-calculate now that writer is closed, so we get the final, accurate size
424           writer = null
425           // adjust our estimate to account for the actual bytes written
426           estimator.update(totalBytes, totalCount)
427           remaining = estimator.estimate(0L)
428         } else {
429           remaining = math.max(100L, estimator.estimate(dataSize))
430         }
431       }
432     }
433 
434     override def size: Long = totalBytes + Option(writer).fold(0L)(_.size)
435 
436     override def flush(): Unit = if (writer != null) { writer.flush() }
437 
438     override def close(): Unit = {
439       if (writer != null) {
440         writer.close()
441       }
442       estimator.close()
443     }
444   }

478    *
479    * @param typeName simple feature type name
480    * @return

482   def newFilePath(typeName: String, ext: String = "parquet"): String = {
483     val filename = s"${ColumnName.encode(typeName).take(20)}_${UUID.randomUUID().toString.replaceAllLiterally("-", "")}.$ext"
484     // partitioning logic taken from Apache Iceberg: https://iceberg.apache.org/docs/nightly/aws/#object-store-file-layout
485     val hash = {
486       val bytes = filename.getBytes(StandardCharsets.UTF_8)
487       val hash = MurmurHash3.hash32x86(bytes, 0, bytes.length, 0)
488       // Integer#toBinaryString excludes leading zeros, which we want to preserve
489       Integer.toBinaryString(hash | Integer.MIN_VALUE)
490     }
491     s"${hash.substring(0, 4)}/${hash.substring(4, 8)}/${hash.substring(8, 12)}/${hash.substring(12, 20)}/$filename"
492   }
