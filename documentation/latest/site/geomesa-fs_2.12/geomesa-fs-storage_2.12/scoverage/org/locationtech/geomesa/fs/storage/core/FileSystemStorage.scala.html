136    * Gets a reader
137    *
138    * @param query query
139    * @param threads read threads
140    * @param forUpdate include row and file data required for positional deletes
141    * @param snapshotId snapshot to read, instead of the current table state
142    * @return
143    */
144   private[core] def getReader(
145       query: Query,
146       threads: Int,
147       forUpdate: Boolean,
148       snapshotId: Option[Long] = None): CloseableIterator[SimpleFeature] = {
149     val configured = QueryRunner.configureQuery(sft, query)
150     val filter = Option(configured.getFilter).getOrElse(Filter.INCLUDE)
151     val icebergFilter = IcebergFilterConverter(schema, schemes, filter)
152     val visFilter = VisibilityUtils.visible(authProvider) // TODO can optimize out on a per-file basis based on metadata
153     val transform = configured.getHints.getTransform
154     val includeFids = configured.getHints.isIncludeFid
155     val sort = configured.getHints.getSortFields
156     val max = configured.getHints.getMaxFeatures
157     val readSchema = schema.read(transform, icebergFilter.remainder, icebergFilter.columns, includeFids, forUpdate)
158 
159     logger.debug(s"Running query '${query.getTypeName}' ${ECQL.toCQL(filter)}")
160     logger.debug(s"  Original filter: ${ECQL.toCQL(query.getFilter)}")
161     logger.debug(s"  Push-down filter: ${icebergFilter.expression}")
162     logger.debug(s"  Client-side filter: ${icebergFilter.remainder.fold("none")(ECQL.toCQL)}")
163     logger.debug(s"  Transforms: ${transform.fold("none") { case (t, _) => if (t.isEmpty) { "empty" } else { t }}}, with${if (includeFids) { "" } else { "out" }} FIDs")
164     logger.debug(s"  Read schema: ${readSchema.schema.schema}")
165     logger.debug(s"  Sort: ${sort.fold("none") { fields => fields.map { case (f, rev) => s"$f ${if (rev) "descending" else ""}"}.mkString(", ")}}")
166     logger.debug(s"  Max features: ${max.getOrElse("none")}")
167 
168     val remainingFilter = icebergFilter.remainder.map(FastFilterFactory.optimize(readSchema.schema.sft, _))
169     val transformer = readSchema.transforms.map { case (tsft, transforms) => new TransformSimpleFeature(tsft, transforms) }
170 
171     val scan = new IcebergParquetScan(table, readSchema.schema, icebergFilter.expression, threads, snapshotId = snapshotId)
172     try {
173       val visible = scan.filter(visFilter.apply)
174       val filtered = remainingFilter.fold(visible)(f => visible.filter(f.evaluate))
175       val transformed = transformer.fold(filtered)(t => filtered.map(t.setFeature))
176       // note - have to copy the features since sorting will not just be sequential access
177       val sorted = sort.fold(transformed)(new SortingSimpleFeatureIterator(transformed.map(ScalaSimpleFeature.copy), _))
178       val limited = max.fold(sorted)(m => sorted.take(m))
179       limited
180     } catch {
181       case NonFatal(e) => CloseWithLogging(scan); throw e
182     }
183   }

220   /**
221    * Read data files with visibilities. For optimized cases, operate directly on files that are wholly visible to the user.
222    * For other cases, we have to read the file and evaluate row-by-row.
223    *
224    * @param filter filter
225    * @param threads read threads
226    * @param ops operations to run
227    */
228   private def fileOps[T](filter: Filter, threads: Int, ops: FileOps[T]): T = {
229     val remainingFiles = ArrayBuffer.empty[String]
230     val visFieldId = schema.schema.findField(SimpleFeatureSchema.VisibilitiesField).fieldId()
231     val visFilter = VisibilityUtils.check(authProvider)
232 
233     val query = QueryRunner.configureQuery(sft, new Query(sft.getTypeName, filter, ops.readAttributes: _*))
234 
235     metadata.files().includeFileStats().forFilter(query.getFilter).scan().foreach { task =>
236       val file = task.file()
237       val nullValueCount = file.nullValueCounts().get(visFieldId)
238       if (nullValueCount != null && nullValueCount == file.valueCounts().get(visFieldId)) {
239         // all nulls
240         if (!ops.fileOp(file, task.deletes().asScala.toSeq)) {
241           remainingFiles += file.location()
242         }
243       } else {
244         val lowerBound = StringParser.apply(file.lowerBounds().get(visFieldId))
245         if (lowerBound != null && lowerBound == StringParser.apply(file.upperBounds().get(visFieldId))) {
246           // only one vis marking, we can evaluate it here
247           if (visFilter.apply(lowerBound)) {
248             if (!ops.fileOp(file, task.deletes().asScala.toSeq)) {
249               remainingFiles += file.location()
250             }
251           }
252         } else {
253           // TODO can read vis metadata from the footer and potentially apply the file-level op
254           remainingFiles += file.location()
255         }
256       }
257     }
258 
259     if (remainingFiles.nonEmpty) {
260       val icebergFilter = IcebergFilterConverter(schema, schemes, query.getFilter)
261       val transform = query.getHints.getTransform
262       val readSchema = schema.read(transform, icebergFilter.remainder, icebergFilter.columns, includeFids = false)
263       val fileFilter: Option[String => Boolean] = Some(remainingFiles.contains)
264       WithClose(new IcebergParquetScan(table, readSchema.schema, icebergFilter.expression, threads, fileFilter)) { scan =>
265         val features = icebergFilter.remainder.fold[CloseableIterator[SimpleFeature]](scan)(f => scan.filter(f.evaluate))
266         features.foreach { f =>
267           if (visFilter.apply(SecurityUtils.getVisibility(f))) {
268             ops.featureOp(f)
269           }
270         }
271       }
272     }
273 
274     ops.result
275   }

359   /**
360    * Creates a writer for a partition that adds its files to a transaction
361    *
362    * @param partition partition
363    * @param transaction transaction
364    * @return
365    */
366   private def newWriter(partition: Partition, transaction: TableTransaction): FileSystemWriter = {
367     val conf = this.conf ++ Map(SimpleFeatureSchema.PartitionKey -> partition.toString)
368     transaction.opened()
369 
370     def newFileWriter(): FileSystemWriter = {
371       val path = newFilePath()
372       val observer = if (observers.isEmpty) { NoOpObserver } else { CompositeObserver.merge(observers.map(_.apply(path))) }
373       new DataFileWriter(ParquetFileSystemWriter(schema, conf, table.io(), path, observer), path, partition, transaction)
374     }
375 
376     val writer = try {
377       sizer.targetSize match {
378         case None => newFileWriter()
379         case Some(s) =>
380           // note: the file size estimate is persisted in the table metadata, so it's only updated after the data is
381           // committed, and only with the most recent estimate from the transaction
382           new ChunkedFileSystemWriter(
383             Iterator.continually(newFileWriter()), sizer.estimator(s), transaction.afterCommit(FileSizeCallbackKey, _))
384       }
385     } catch {
386       case NonFatal(e) => transaction.closed(); throw e
387     }
388     new TrackedWriter(writer, transaction)
389   }

430     private def register(id: Option[String], files: Map[Partition, Seq[URI]]): Seq[DataFile] = {
431       id.foreach { id =>
432         require(id != null && id.trim.nonEmpty, "Registration id must not be empty")
433         require(!id.startsWith(GeoMesaPropertyPrefix), s"Registration id must not start with '$GeoMesaPropertyPrefix': $id")
434       }
435 
436       // copy files into the table location, without adding them to the table
437       val dataFiles = files.toSeq.flatMap { case (partition, paths) =>
438         paths.map { path =>
439           val destination = newFilePath()
440           logger.debug(s"Copying $path to $destination")
441           val uri = URI.create(destination)
442           WithClose(ObjectStore(uri.getScheme, conf))(_.copy(path, uri))
443           toDataFile(destination, partition)
444         }
445       }
446       if (dataFiles.nonEmpty) {
447         try {
448           // the id is committed atomically with the files
449           val tx = id.map(_ -> table.newTransaction())
450           val append = tx.fold(table.newAppend()) { case (_, tx) => tx.newAppend() }
451           dataFiles.foreach(append.appendFile)
452           append.commit()
453           tx.foreach { case (id, tx) =>
454             // note: snapshot needs to be evaluated after committing the file append
455             tx.updateProperties().set(id, tx.table().currentSnapshot().snapshotId().toString).commit()
456             tx.commitTransaction()
457           }
458         } catch {
459           case NonFatal(e) =>
460             // CleanableFailure indicates that cleanup is safe,
461             // otherwise we can't be certain if the files are referenced by the table or not, so leave them for orphan cleanup
462             if (e.isInstanceOf[CleanableFailure]) {
463               dataFiles.foreach(f => deleteQuietly(f.location()))
464             }
465             throw e
466         }
467       }
468       dataFiles
469     }

477     private def byPartition(files: Seq[URI]): Map[Partition, Seq[URI]] = {
478       val partitioned = scala.collection.mutable.Map.empty[String, ArrayBuffer[URI]]
479       WithClose(ObjectStore(files.head.getScheme, conf)) { fs =>
480         files.foreach { file =>
481           WithClose(ParquetFileReader.open(ParquetFileSystemReader.inputFile(fs, file))) { reader =>

485             }
486             partitioned.getOrElseUpdate(partition, ArrayBuffer.empty) += file
487           }
488         }
489       }
490 
491       partitioned.map { case (k, v) => Partition(k) -> v.toSeq }.toMap
492     }

542    *
543    * @param delegate file writer
544    * @param path file path

548   private class DataFileWriter(
549       delegate: FileSystemWriter,
550       path: String,
551       partition: Partition,
552       transaction: TableTransaction
553     ) extends FileSystemWriter with VisibilityChecker {
554 
555     private var empty = true
556     private var closed = false
557 
558     // note: explicit to avoid a conflict with the implicit class of the same name in this package
559     private val requireVis =
560       new org.locationtech.geomesa.utils.geotools.RichSimpleFeatureType.RichSimpleFeatureType(schema.sft).isVisibilityRequired
561 
562     override def write(feature: SimpleFeature): Unit = {
563       if (requireVis) {
564         // check up front, so that an invalid feature is rejected without failing the transaction
565         requireVisibilities(feature)
566       }
567       try { delegate.write(feature) } catch {
568         // the file may be missing features or have a partial write, so the transaction can't be committed
569         case NonFatal(e) => transaction.failed(e); throw e
570       }
571       empty = false
572     }
573 
574     override def size: Long = delegate.size
575
