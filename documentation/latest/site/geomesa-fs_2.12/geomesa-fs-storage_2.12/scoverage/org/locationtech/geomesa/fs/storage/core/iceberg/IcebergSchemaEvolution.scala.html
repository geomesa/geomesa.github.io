46     override def add(attribute: AttributeDescriptor): Unit = {
47       WithClose(catalog.loadTable(tableId)) { table =>
48         val tx = table.newTransaction()
49         colsFor(attribute).foldLeft(tx.updateSchema())((update, col) => update.addColumn(col.name(), col.`type`())).commit()

56     /**
57      * Update an attribute. The only supported type changes are widening (int -> long, etc) and modifying structural types.
58      * For structural types, changes are detected based on the field names - fields with the same name will be treated as updates,
59      * while fields with different names will be treated as new fields (and missing field names will be dropped).
60      *
61      * @param update updated attribute
62      */
63     override def update(update: AttributeDescriptor): Unit = {
64       WithClose(catalog.loadTable(tableId)) { table =>
65         val schema = SimpleFeatureIcebergSchema(table)
66         val existing = schema.sft.getDescriptor(update.getLocalName)
67         if (existing == null) {
68           throw new IllegalArgumentException(
69             s"Attribute '${update.getLocalName}' does not exist in the schema: ${schema.sft.getTypeName} " +
70               SimpleFeatureTypes.encodeType(schema.sft))
71         }
72 
73         val tx = table.newTransaction()
74 
75         if (classOf[Geometry].isAssignableFrom(existing.getType.getBinding)) {
76           // we don't need to make any actual column changes, just validate that the op is allowed
77           if (existing.getType.getBinding == classOf[Point] || update.getType.getBinding != classOf[Geometry]) {
78             throw new UnsupportedOperationException(
79               s"Attribute '${existing.getLocalName}' can not be changed from ${existing.getType.getBinding} " +
80                 s"to ${update.getType.getBinding.getSimpleName}: the only supported geometry change is widening non-point types " +
81                 s"to mixed Geometry")
82           }
83         } else {
84           // assertion: only geometry types create more than 1 column, so we only need to deal with .head
85           val column = IcebergSchemaEvolution.colsFor(update).head
86           val updateSchema = tx.updateSchema()
87           walkColumnUpdates(null, column, schema.schema.findField(column.name()), updateSchema)
88           updateSchema.commit()
89         }
90 
91         val (columnSpecProp, columnSpec) = attributeProperty(update)
92         tx.updateProperties().set(columnSpecProp, columnSpec).commit()
93         tx.commitTransaction()
94       }
95     }

97     private def walkColumnUpdates(parent: String, update: NestedField, existing: NestedField, op: UpdateSchema): Unit = {
98       if (existing == null) {
99         op.addColumn(parent, update.name(), update.`type`())
100       } else if (update.`type`() != existing.`type`()) {
101         val path = Option(parent).fold(update.name())(p => s"$p.${update.name()}")
102         if (update.`type`().isStructType && existing.`type`().isStructType) {
103           val updateStruct = update.`type`().asStructType()
104           val existingStruct = existing.`type`().asStructType()
105           updateStruct.fields().asScala.foreach { field =>
106             walkColumnUpdates(path, field, existingStruct.field(field.name()), op)
107           }
108           existingStruct.fields().asScala.foreach { field =>
109             if (updateStruct.field(field.name()) == null) {
110               op.deleteColumn(s"$path.${field.name()}")
111             }
112           }
113         } else if (update.`type`().isPrimitiveType && existing.`type`().isPrimitiveType) {
114           op.updateColumn(path, update.`type`().asPrimitiveType())
115         } else {
116           throw new UnsupportedOperationException(
117             s"Attribute '$path' can't be changed from ${existing.`type`()} to ${update.`type`()}")
118         }
119       }
120     }

122     override def rename(attribute: String, newName: String): Unit = {
123       WithClose(catalog.loadTable(tableId)) { table =>
124         val schema = SimpleFeatureIcebergSchema(table)
125         val descriptor = schema.sft.getDescriptor(attribute)
126         if (descriptor == null) {
127           throw new IllegalArgumentException(
128             s"Attribute '$attribute' does not exist in the schema: ${schema.sft.getTypeName} " +
129               SimpleFeatureTypes.encodeType(schema.sft))
130         }
131         val renamedDescriptor = SimpleFeatureTypes.renameDescriptor(descriptor, newName)
132         val oldCols = colsFor(descriptor).map(_.name())
133         val newCols = colsFor(renamedDescriptor).map(_.name())
134         val oldBloomProps = bloomFilterKeys(attribute).map(k => k -> Option(table.properties().get(k)))
135         val propsToAdd = {
136           val (columnSpecProp, columnSpec) = attributeProperty(renamedDescriptor)
137           val blooms = bloomFilterKeys(newName).zip(oldBloomProps).flatMap { case (newKey, (_, value)) => value.map(newKey -> _) }
138           val dtgUserData = {
139             val key = s"${IcebergCatalog.UserDataPrefix}${Configs.DefaultDtgField}"
140             if (table.properties().get(key) == attribute) { Map(key -> newName) } else { Map.empty }
141           }
142           Map(columnSpecProp -> columnSpec) ++ blooms ++ dtgUserData
143         }
144         val propsToRemove = oldBloomProps.map(_._1) :+ IcebergCatalog.columnSpecProperty(oldCols.head)
145         val tx = table.newTransaction()
146         val updateSchema = tx.updateSchema()
147         oldCols.zip(newCols).foreach { case (old, newName) => updateSchema.renameColumn(old, newName) }
148         updateSchema.commit()
149         val updateProperties = tx.updateProperties()
150         propsToRemove.foreach(updateProperties.remove)
151         propsToAdd.foreach { case (k, v) => updateProperties.set(k, v) }
152         updateProperties.commit()
153         tx.commitTransaction()
154       }
155     }

157     override def remove(attribute: String): Unit = {
158       WithClose(catalog.loadTable(tableId)) { table =>
159         val schema = SimpleFeatureIcebergSchema(table)
160         val colsToRemove = colsFor(schema.sft.getDescriptor(attribute)).map(_.name())
161         val propsToRemove = bloomFilterKeys(attribute) :+ IcebergCatalog.columnSpecProperty(colsToRemove.head)
162         val tx = table.newTransaction()
163         colsToRemove.foldLeft(tx.updateSchema())((update, col) => update.deleteColumn(col)).commit()
164         propsToRemove.foldLeft(tx.updateProperties())((update, prop) => update.remove(prop)).commit()
165         tx.commitTransaction()
166       }
167     }

169     override def reorder(attribute: String, position: Int): Unit = {
170       WithClose(catalog.loadTable(tableId)) { table =>
171         val sft = SimpleFeatureIcebergSchema(table).sft
172         val descriptor = sft.getDescriptor(attribute)
173         if (descriptor == null) {
174           throw new IllegalArgumentException(
175             s"Attribute '$attribute' does not exist in the schema: ${sft.getTypeName} " +
176               SimpleFeatureTypes.encodeType(sft))
177         } else if (position < 0 || position > sft.getAttributeCount - 1) {
178           throw new IllegalArgumentException(
179             s"Invalid reorder position, not in [0,${sft.getAttributeCount - 1}]: $position")
180         } else if (position == sft.indexOf(attribute)) {
181           logger.warn(s"Ignoring no-op move of attribute $attribute to $position")
182           return
183         }
184         val toMove = colsFor(descriptor)
185         if (position == sft.getAttributeCount - 1) {
186           val location = table.schema().columns().asScala.last.name()
187           toMove.reverse.foldLeft(table.updateSchema())((update, col) => update.moveAfter(col.name(), location)).commit()
188         } else {
189           val location = ColumnName.encode(sft.getAttributeDescriptors.asScala.filterNot(_ == descriptor)(position).getLocalName)
190           toMove.foldLeft(table.updateSchema())((update, col) => update.moveBefore(col.name(), location)).commit()
191         }
192       }
193     }

198     override def setPartitions(partitions: Seq[String]): Unit = {
199       WithClose(catalog.loadTable(tableId)) { table =>
200         val schema = SimpleFeatureIcebergSchema(table)
201         val currentSpec = table.spec().fields().asScala.map(f => (f.sourceId(), f.transform(), f.name()))
202         val updatedSpec = {
203           val schemes = partitions.map(PartitionSchemeFactory.load(schema.sft, _)).sortBy(_.name)
204           val fields = schemes.foldLeft(PartitionSpec.builderFor(schema.schema))((b, m) => m.spec(b)).build().fields().asScala
205           fields.map(f => (f.sourceId(), f.transform(), f.name()))
206         }
207 
208         val updateSpec = table.updateSpec()
209         currentSpec.filterNot(updatedSpec.contains).foreach { case (_, _, name) => updateSpec.removeField(name) }
210         updatedSpec.filterNot(currentSpec.contains).foreach { case (sourceId, transform, name) =>
211           updateSpec.addField(name, Expressions.transform(schema.schema.findColumnName(sourceId), transform))
212         }
213         updateSpec.commit()
214       }
215     }
