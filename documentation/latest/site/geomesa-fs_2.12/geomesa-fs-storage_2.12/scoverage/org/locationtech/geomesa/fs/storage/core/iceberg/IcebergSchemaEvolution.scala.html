59     /**
60      * Update an attribute. The only supported type changes are widening (int -> long, etc) and modifying structural types.
61      * For structural types, changes are detected based on the field names - fields with the same name will be treated as updates,
62      * while fields with different names will be treated as new fields (and missing field names will be dropped).
63      *
64      * @param update updated attribute
65      */
66     override def update(update: AttributeDescriptor): Unit = {
67       WithClose(catalog.loadTable(tableId)) { table =>
68         val schema = SimpleFeatureIcebergSchema(table)
69         val existing = schema.sft.getDescriptor(update.getLocalName)
70         if (existing == null) {
71           throw new IllegalArgumentException(
72             s"Attribute '${update.getLocalName}' does not exist in the schema: ${schema.sft.getTypeName} " +
73               SimpleFeatureTypes.encodeType(schema.sft))
74         }
75 
76         val existingIsGeom = classOf[Geometry].isAssignableFrom(existing.getType.getBinding)
77         val updateIsGeom = classOf[Geometry].isAssignableFrom(update.getType.getBinding)
78 
79         val tx = table.newTransaction()
80 
81         if (existingIsGeom || updateIsGeom) {
82           // we don't need to make any actual column changes, just validate that the op is allowed
83           if (!existingIsGeom || !updateIsGeom) {
84             throw new UnsupportedOperationException(
85               s"Attribute '${existing.getLocalName}' can not be changed from ${existing.getType.getBinding.getSimpleName} " +
86                 s"to ${update.getType.getBinding.getSimpleName}: geometry attributes can't be converted to or from " +
87                 "non-geometry types")
88           } else if (existing.getType.getBinding != update.getType.getBinding &&
89               (existing.getType.getBinding == classOf[Point] || update.getType.getBinding != classOf[Geometry])) {
90             throw new UnsupportedOperationException(
91               s"Attribute '${existing.getLocalName}' can not be changed from ${existing.getType.getBinding.getSimpleName} " +
92                 s"to ${update.getType.getBinding.getSimpleName}: the only supported geometry change is widening non-point " +
93                 "types to mixed Geometry")
94           }
95           // the stored data is always wkb, so the encoding can't be changed
96           update.getEncoding.foreach { encoding =>
97             if (GeometryEncoding(encoding) != GeometryEncoding.GeoParquetWkb) {
98               throw new UnsupportedOperationException(
99                 s"Attribute '${existing.getLocalName}' can not be changed to encoding '$encoding': " +
100                   "only WKB encoding is supported for geometry types")
101             }
102           }
103         } else {
104           // assertion: only geometry types create more than 1 column, so we only need to deal with .head
105           val column = IcebergSchemaEvolution.colsFor(update).head
106           val updateSchema = tx.updateSchema()
107           val existingColumn = schema.schema.findField(column.name())
108           val path = FieldPath(existing.getLocalName, column.name())
109           checkNullabilityUpdates(path, column.isOptional, existingColumn.isOptional, updateSchema)
110           walkTypeUpdates(path, column.`type`(), existingColumn.`type`(), updateSchema)
111           updateSchema.commit()
112         }
113 
114         val (columnSpecProp, columnSpec) = attributeProperty(update)
115         tx.updateProperties().set(columnSpecProp, columnSpec).commit()
116         tx.commitTransaction()
117       }
118     }

120     private def checkNullabilityUpdates(path: FieldPath, updateOptional: Boolean, existingOptional: Boolean, op: UpdateSchema): Unit = {
121       if (updateOptional && !existingOptional) {
122         op.makeColumnOptional(path.column)
123       } else if (!updateOptional && existingOptional) {
124         // existing data may contain nulls, so this isn't safe
125         throw path.error("can't be changed from optional to required")
126       }
127     }

129     // note: nested types are compared structurally, as the field ids of the update will not match the existing ids
130     // noinspection NoTailRecursionAnnotation
131     private def walkTypeUpdates(path: FieldPath, update: Type, existing: Type, op: UpdateSchema): Unit = {
132       if (update.isStructType && existing.isStructType) {
133         val updateStruct = update.asStructType()
134         val existingStruct = existing.asStructType()
135         updateStruct.fields().asScala.foreach { field =>
136           walkStructFields(path, field, existingStruct.field(field.name()), op)
137         }
138         existingStruct.fields().asScala.foreach { field =>
139           if (updateStruct.field(field.name()) == null) {
140             op.deleteColumn(s"${path.column}.${field.name()}")
141           }
142         }
143       } else if (update.isListType && existing.isListType) {
144         val updateList = update.asListType()
145         val existingList = existing.asListType()
146         val element = path.element
147         checkNullabilityUpdates(element, updateList.isElementOptional, existingList.isElementOptional, op)
148         walkTypeUpdates(element, updateList.elementType(), existingList.elementType(), op)
149       } else if (update.isMapType && existing.isMapType) {
150         val updateMap = update.asMapType()
151         val existingMap = existing.asMapType()
152         if (!updateMap.keyType().isPrimitiveType || !existingMap.keyType().isPrimitiveType) {
153           // note: geomesa and avro maps always have primitive keys, so this shouldn't come up in practice
154           throw path.error("can't be updated, as only maps with primitive keys are supported")
155         } else if (updateMap.keyType() != existingMap.keyType()) {
156           throw path.error(s"can't change map keys from ${existingMap.keyType()} to ${updateMap.keyType()}")
157         }
158         val value = path.value
159         checkNullabilityUpdates(value, updateMap.isValueOptional, existingMap.isValueOptional, op)
160         walkTypeUpdates(value, updateMap.valueType(), existingMap.valueType(), op)
161       } else if (update.isPrimitiveType && existing.isPrimitiveType) {
162         if (update != existing) {
163           if (!TypeUtil.isPromotionAllowed(existing, update.asPrimitiveType())) {
164             throw path.error(s"can't be changed from $existing to $update: the only supported primitive type changes " +
165               "are widening int to long and float to double")
166           }
167           op.updateColumn(path.column, update.asPrimitiveType())
168         }
169       } else if (update != existing) {
170         throw path.error(s"can't be changed from $existing to $update")
171       }
172     }

174     private def walkStructFields(parent: FieldPath, update: NestedField, existing: NestedField, op: UpdateSchema): Unit = {
175       if (existing == null) {
176         op.addColumn(parent.column, update.name(), update.`type`())
177       } else {
178         val path = parent.field(update.name())
179         checkNullabilityUpdates(path, update.isOptional, existing.isOptional, op)
180         walkTypeUpdates(path, update.`type`(), existing.`type`(), op)
181       }
182     }

184     override def rename(attribute: String, newName: String): Unit = {
185       WithClose(catalog.loadTable(tableId)) { table =>
186         val schema = SimpleFeatureIcebergSchema(table)
187         val descriptor = schema.sft.getDescriptor(attribute)
188         if (descriptor == null) {
189           throw new IllegalArgumentException(
190             s"Attribute '$attribute' does not exist in the schema: ${schema.sft.getTypeName} " +
191               SimpleFeatureTypes.encodeType(schema.sft))
192         }
193         val renamedDescriptor = SimpleFeatureTypes.renameDescriptor(descriptor, newName)
194         val oldCols = colsFor(descriptor).map(_.name())
195         val newCols = colsFor(renamedDescriptor).map(_.name())
196         val oldBloomProps = bloomFilterKeys(attribute).map(k => k -> Option(table.properties().get(k)))
197         val propsToAdd = {
198           val (columnSpecProp, columnSpec) = attributeProperty(renamedDescriptor)
199           val blooms = bloomFilterKeys(newName).zip(oldBloomProps).flatMap { case (newKey, (_, value)) => value.map(newKey -> _) }
200           val dtgUserData = {
201             val key = s"${IcebergCatalog.UserDataPrefix}${Configs.DefaultDtgField}"
202             if (table.properties().get(key) == attribute) { Map(key -> newName) } else { Map.empty }
203           }
204           Map(columnSpecProp -> columnSpec) ++ blooms ++ dtgUserData
205         }
206         val propsToRemove = oldBloomProps.map(_._1) :+ IcebergCatalog.columnSpecProperty(oldCols.head)
207         val tx = table.newTransaction()
208         val updateSchema = tx.updateSchema()
209         oldCols.zip(newCols).foreach { case (old, newName) => updateSchema.renameColumn(old, newName) }
210         updateSchema.commit()
211         val updateProperties = tx.updateProperties()
212         propsToRemove.foreach(updateProperties.remove)
213         propsToAdd.foreach { case (k, v) => updateProperties.set(k, v) }
214         updateProperties.commit()
215         tx.commitTransaction()
216       }
217     }

219     override def remove(attribute: String): Unit = {
220       WithClose(catalog.loadTable(tableId)) { table =>
221         val schema = SimpleFeatureIcebergSchema(table)
222         val colsToRemove = colsFor(schema.sft.getDescriptor(attribute)).map(_.name())
223         val propsToRemove = bloomFilterKeys(attribute) :+ IcebergCatalog.columnSpecProperty(colsToRemove.head)
224         val tx = table.newTransaction()
225         colsToRemove.foldLeft(tx.updateSchema())((update, col) => update.deleteColumn(col)).commit()
226         propsToRemove.foldLeft(tx.updateProperties())((update, prop) => update.remove(prop)).commit()
227         tx.commitTransaction()
228       }

231     override def reorder(attribute: String, position: Int): Unit = {
232       WithClose(catalog.loadTable(tableId)) { table =>
233         val sft = SimpleFeatureIcebergSchema(table).sft
234         val descriptor = sft.getDescriptor(attribute)
235         if (descriptor == null) {
236           throw new IllegalArgumentException(
237             s"Attribute '$attribute' does not exist in the schema: ${sft.getTypeName} " +
238               SimpleFeatureTypes.encodeType(sft))
239         } else if (position < 0 || position > sft.getAttributeCount - 1) {
240           throw new IllegalArgumentException(
241             s"Invalid reorder position, not in [0,${sft.getAttributeCount - 1}]: $position")
242         } else if (position == sft.indexOf(attribute)) {
243           logger.warn(s"Ignoring no-op move of attribute $attribute to $position")
244           return
245         }
246         val toMove = colsFor(descriptor)
247         if (position == sft.getAttributeCount - 1) {
248           val location = table.schema().columns().asScala.last.name()
249           toMove.reverse.foldLeft(table.updateSchema())((update, col) => update.moveAfter(col.name(), location)).commit()
250         } else {
251           val location = ColumnName.encode(sft.getAttributeDescriptors.asScala.filterNot(_ == descriptor)(position).getLocalName)
252           toMove.foldLeft(table.updateSchema())((update, col) => update.moveBefore(col.name(), location)).commit()
253         }
254       }
255     }

257     /**
258      * Path to a (possibly nested) field
259      *
260      * @param attribute top-level attribute name, used for error messages
261      * @param column iceberg column path, used for schema updates
262      * @param display user-facing description of the nested field, used for error messages
263      */
264     private case class FieldPath(attribute: String, column: String, display: Option[String] = None) {
265       def field(name: String): FieldPath = FieldPath(attribute, s"$column.$name", Some(display.fold(name)(d => s"$d.$name")))
266       def element: FieldPath = FieldPath(attribute, s"$column.element", Some(s"${display.getOrElse("")}[element]"))
267       def value: FieldPath = FieldPath(attribute, s"$column.value", Some(s"${display.getOrElse("")}[value]"))
268       def error(msg: String): UnsupportedOperationException = {
269         val location = display.fold("")(d => s" (nested field '$d')")
270         new UnsupportedOperationException(s"Attribute '$attribute'$location $msg")
271       }
272     }

285     override def setPartitions(partitions: Seq[String]): Unit = {
286       WithClose(catalog.loadTable(tableId)) { table =>
287         val schema = SimpleFeatureIcebergSchema(table)
288         val currentSpec = table.spec().fields().asScala.map(f => (f.sourceId(), f.transform(), f.name()))
289         val updatedSpec = {
290           val schemes = partitions.map(PartitionSchemeFactory.load(schema.sft, _)).sortBy(_.name)
291           val fields = schemes.foldLeft(PartitionSpec.builderFor(schema.schema))((b, m) => m.spec(b)).build().fields().asScala
292           fields.map(f => (f.sourceId(), f.transform(), f.name()))
293         }
294 
295         val updateSpec = table.updateSpec()
296         currentSpec.filterNot(updatedSpec.contains).foreach { case (_, _, name) => updateSpec.removeField(name) }
297         updatedSpec.filterNot(currentSpec.contains).foreach { case (sourceId, transform, name) =>
298           updateSpec.addField(name, Expressions.transform(schema.schema.findColumnName(sourceId), transform))
299         }
300         updateSpec.commit()
301       }
302     }
