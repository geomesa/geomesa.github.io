33 /**
34  * Schema evolution for iceberg tables.
35  *
36  * Each change is validated when it is requested, by running it (along with any previously staged changes) in an iceberg
37  * transaction that is then discarded, without modifying the table. Any validation errors are thrown when the change is
38  * requested, and the invalid change is not staged. Nothing is persisted until `apply` is invoked, at which point all the
39  * staged changes are re-run in a single transaction and committed atomically. The table is only held open for the
40  * duration of each call, so an update can be safely abandoned without being applied.
41  *
42  * An update can only be applied once, even if the apply fails - to re-try a failed apply, a new update must be created.
43  *
44  * The one exception is a table rename, which iceberg does not support as part of a transaction - it is executed
45  * immediately after the transaction is committed.
46  *
47  * @param catalog catalog
48  * @param tableId tableId
49  */
50 class IcebergSchemaEvolution(catalog: Catalog, tableId: TableIdentifier) extends CatalogUpdate with LazyLogging {

66   override def apply(): Unit = {
67     if (applied) {
68       throw new IllegalStateException("Update has already been applied")
69     }
70     applied = true
71     withTransaction(_.commitTransaction())
72     // the rename can't be done as part of the transaction
73     renameTo.foreach(catalog.renameTable(tableId, _))
74   }

76   /**
77    * Stage a change, after validating it against the current table plus any previously staged changes
78    *
79    * @param change change to stage - takes a transaction, including any previously staged changes
80    * @return
81    */
82   private def stage(change: Transaction => Unit): CatalogUpdate = {
83     // the transaction is discarded without being committed, it's only used for validation
84     withTransaction(change)
85     changes += change
86     this
87   }

95     WithClose(catalog.loadTable(tableId)) { table =>

122     /**
123      * Update an attribute. The only supported type changes are widening (int -> long, etc) and modifying structural types.
124      * For structural types, changes are detected based on the field names - fields with the same name will be treated as updates,
125      * while fields with different names will be treated as new fields (and missing field names will be dropped).
126      *
127      * @param update updated attribute
128      */
129     override def update(update: AttributeDescriptor): CatalogUpdate = {
130       stage { tx =>
131         val schema = SimpleFeatureIcebergSchema(tx.table())
132         val existing = descriptor(schema, update.getLocalName)
133 
134         val existingIsGeom = classOf[Geometry].isAssignableFrom(existing.getType.getBinding)
135         val updateIsGeom = classOf[Geometry].isAssignableFrom(update.getType.getBinding)
136 
137         if (existingIsGeom || updateIsGeom) {
138           // we don't need to make any actual column changes, just validate that the op is allowed
139           if (!existingIsGeom || !updateIsGeom) {
140             throw new UnsupportedOperationException(
141               s"Attribute '${existing.getLocalName}' can not be changed from ${existing.getType.getBinding.getSimpleName} " +
142                 s"to ${update.getType.getBinding.getSimpleName}: geometry attributes can't be converted to or from " +
143                 "non-geometry types")
144           } else if (existing.getType.getBinding != update.getType.getBinding &&
145               (existing.getType.getBinding == classOf[Point] || update.getType.getBinding != classOf[Geometry])) {
146             throw new UnsupportedOperationException(
147               s"Attribute '${existing.getLocalName}' can not be changed from ${existing.getType.getBinding.getSimpleName} " +
148                 s"to ${update.getType.getBinding.getSimpleName}: the only supported geometry change is widening non-point " +
149                 "types to mixed Geometry")
150           }
151           // the stored data is always wkb, so the encoding can't be changed
152           update.getEncoding.foreach { encoding =>
153             if (GeometryEncoding(encoding) != GeometryEncoding.GeoParquetWkb) {
154               throw new UnsupportedOperationException(
155                 s"Attribute '${existing.getLocalName}' can not be changed to encoding '$encoding': " +
156                   "only WKB encoding is supported for geometry types")
157             }
158           }
159         } else {
160           // assertion: only geometry types create more than 1 column, so we only need to deal with .head
161           val column = IcebergSchemaEvolution.colsFor(update).head
162           val updateSchema = tx.updateSchema()
163           val existingColumn = schema.schema.findField(column.name())
164           val path = FieldPath(existing.getLocalName, column.name())
165           checkNullabilityUpdates(path, column.isOptional, existingColumn.isOptional, updateSchema)
166           walkTypeUpdates(path, column.`type`(), existingColumn.`type`(), updateSchema)
167           updateSchema.commit()
168         }
169 
170         val (columnSpecProp, columnSpec) = attributeProperty(update)
171         tx.updateProperties().set(columnSpecProp, columnSpec).commit()
172       }
173     }

175     private def checkNullabilityUpdates(path: FieldPath, updateOptional: Boolean, existingOptional: Boolean, op: UpdateSchema): Unit = {
176       if (updateOptional && !existingOptional) {
177         op.makeColumnOptional(path.column)
178       } else if (!updateOptional && existingOptional) {
179         // existing data may contain nulls, so this isn't safe
180         throw path.error("can't be changed from optional to required")
181       }
182     }

184     // note: nested types are compared structurally, as the field ids of the update will not match the existing ids
185     // noinspection NoTailRecursionAnnotation
186     private def walkTypeUpdates(path: FieldPath, update: Type, existing: Type, op: UpdateSchema): Unit = {
187       if (update.isStructType && existing.isStructType) {
188         val updateStruct = update.asStructType()
189         val existingStruct = existing.asStructType()
190         updateStruct.fields().asScala.foreach { field =>
191           walkStructFields(path, field, existingStruct.field(field.name()), op)
192         }
193         existingStruct.fields().asScala.foreach { field =>
194           if (updateStruct.field(field.name()) == null) {
195             op.deleteColumn(s"${path.column}.${field.name()}")
196           }
197         }
198       } else if (update.isListType && existing.isListType) {
199         val updateList = update.asListType()
200         val existingList = existing.asListType()
201         val element = path.element
202         checkNullabilityUpdates(element, updateList.isElementOptional, existingList.isElementOptional, op)
203         walkTypeUpdates(element, updateList.elementType(), existingList.elementType(), op)
204       } else if (update.isMapType && existing.isMapType) {
205         val updateMap = update.asMapType()
206         val existingMap = existing.asMapType()
207         if (!updateMap.keyType().isPrimitiveType || !existingMap.keyType().isPrimitiveType) {
208           // note: geomesa and avro maps always have primitive keys, so this shouldn't come up in practice
209           throw path.error("can't be updated, as only maps with primitive keys are supported")
210         } else if (updateMap.keyType() != existingMap.keyType()) {
211           throw path.error(s"can't change map keys from ${existingMap.keyType()} to ${updateMap.keyType()}")
212         }
213         val value = path.value
214         checkNullabilityUpdates(value, updateMap.isValueOptional, existingMap.isValueOptional, op)
215         walkTypeUpdates(value, updateMap.valueType(), existingMap.valueType(), op)
216       } else if (update.isPrimitiveType && existing.isPrimitiveType) {
217         if (update != existing) {
218           if (!TypeUtil.isPromotionAllowed(existing, update.asPrimitiveType())) {
219             throw path.error(s"can't be changed from $existing to $update: the only supported primitive type changes " +
220               "are widening int to long and float to double")
221           }
222           op.updateColumn(path.column, update.asPrimitiveType())
223         }
224       } else if (update != existing) {
225         throw path.error(s"can't be changed from $existing to $update")
226       }
227     }

229     private def walkStructFields(parent: FieldPath, update: NestedField, existing: NestedField, op: UpdateSchema): Unit = {
230       if (existing == null) {
231         op.addColumn(parent.column, update.name(), update.`type`())
232       } else {
233         val path = parent.field(update.name())
234         checkNullabilityUpdates(path, update.isOptional, existing.isOptional, op)
235         walkTypeUpdates(path, update.`type`(), existing.`type`(), op)
236       }
237     }

239     override def rename(attribute: String, newName: String): CatalogUpdate = {
240       stage { tx =>
241         val table = tx.table()
242         val schema = SimpleFeatureIcebergSchema(table)
243         val existing = descriptor(schema, attribute)
244         val renamedDescriptor = SimpleFeatureTypes.renameDescriptor(existing, newName)
245         val oldCols = colsFor(existing).map(_.name())
246         val newCols = colsFor(renamedDescriptor).map(_.name())
247         val oldBloomProps = bloomFilterKeys(attribute).map(k => k -> Option(table.properties().get(k)))
248         val propsToAdd = {
249           val (columnSpecProp, columnSpec) = attributeProperty(renamedDescriptor)
250           val blooms = bloomFilterKeys(newName).zip(oldBloomProps).flatMap { case (newKey, (_, value)) => value.map(newKey -> _) }
251           val dtgUserData = {
252             val key = s"${IcebergCatalog.UserDataPrefix}${Configs.DefaultDtgField}"
253             if (table.properties().get(key) == attribute) { Map(key -> newName) } else { Map.empty }
254           }
255           Map(columnSpecProp -> columnSpec) ++ blooms ++ dtgUserData
256         }
257         val propsToRemove = oldBloomProps.map(_._1) :+ IcebergCatalog.columnSpecProperty(oldCols.head)
258         val updateSchema = tx.updateSchema()
259         oldCols.zip(newCols).foreach { case (old, newName) => updateSchema.renameColumn(old, newName) }
260         updateSchema.commit()
261         val updateProperties = tx.updateProperties()
262         propsToRemove.foreach(updateProperties.remove)
263         propsToAdd.foreach { case (k, v) => updateProperties.set(k, v) }
264         updateProperties.commit()
265       }
266     }

268     override def remove(attribute: String): CatalogUpdate = {
269       stage { tx =>
270         val schema = SimpleFeatureIcebergSchema(tx.table())
271         val colsToRemove = colsFor(descriptor(schema, attribute)).map(_.name())
272         val propsToRemove = bloomFilterKeys(attribute) :+ IcebergCatalog.columnSpecProperty(colsToRemove.head)
273         colsToRemove.foldLeft(tx.updateSchema())((update, col) => update.deleteColumn(col)).commit()
274         propsToRemove.foldLeft(tx.updateProperties())((update, prop) => update.remove(prop)).commit()
275       }

278     override def reorder(attribute: String, position: Int): CatalogUpdate = {
279       var warn = true // only warn once, as staged changes are run multiple times
280       stage { tx =>
281         val table = tx.table()
282         val schema = SimpleFeatureIcebergSchema(table)
283         val sft = schema.sft
284         val existing = descriptor(schema, attribute)
285         if (position < 0 || position > sft.getAttributeCount - 1) {
286           throw new IllegalArgumentException(
287             s"Invalid reorder position, not in [0,${sft.getAttributeCount - 1}]: $position")
288         } else if (position == sft.indexOf(attribute)) {
289           if (warn) {
290             logger.warn(s"Ignoring no-op move of attribute $attribute to $position")
291             warn = false
292           }
293         } else {
294           val toMove = colsFor(existing)
295           if (position == sft.getAttributeCount - 1) {
296             val location = table.schema().columns().asScala.last.name()
297             toMove.reverse.foldLeft(tx.updateSchema())((update, col) => update.moveAfter(col.name(), location)).commit()
298           } else {
299             val location = ColumnName.encode(sft.getAttributeDescriptors.asScala.filterNot(_ == existing)(position).getLocalName)
300             toMove.foldLeft(tx.updateSchema())((update, col) => update.moveBefore(col.name(), location)).commit()
301           }
302         }
303       }
304     }

307      * Path to a (possibly nested) field
308      *
309      * @param attribute top-level attribute name, used for error messages
310      * @param column iceberg column path, used for schema updates
311      * @param display user-facing description of the nested field, used for error messages
312      */
313     private case class FieldPath(attribute: String, column: String, display: Option[String] = None) {
314       def field(name: String): FieldPath = FieldPath(attribute, s"$column.$name", Some(display.fold(name)(d => s"$d.$name")))
315       def element: FieldPath = FieldPath(attribute, s"$column.element", Some(s"${display.getOrElse("")}[element]"))
316       def value: FieldPath = FieldPath(attribute, s"$column.value", Some(s"${display.getOrElse("")}[value]"))
317       def error(msg: String): UnsupportedOperationException = {
318         val location = display.fold("")(d => s" (nested field '$d')")
319         new UnsupportedOperationException(s"Attribute '$attribute'$location $msg")
320       }
321     }

327       val newTableId = TableIdentifier.of(tableId.namespace(), ColumnName.encode(newName))

339     override def setPartitions(partitions: Seq[String]): CatalogUpdate = {
340       stage { tx =>
341         val table = tx.table()
342         val schema = SimpleFeatureIcebergSchema(table)
343         val currentSpec = table.spec().fields().asScala.map(f => (f.sourceId(), f.transform(), f.name()))
344         val updatedSpec = {
345           val schemes = partitions.map(PartitionSchemeFactory.load(schema.sft, _)).sortBy(_.name)
346           val fields = schemes.foldLeft(PartitionSpec.builderFor(schema.schema))((b, m) => m.spec(b)).build().fields().asScala
347           fields.map(f => (f.sourceId(), f.transform(), f.name()))
348         }
349 
350         val updateSpec = tx.updateSpec()
351         currentSpec.filterNot(updatedSpec.contains).foreach { case (_, _, name) => updateSpec.removeField(name) }
352         updatedSpec.filterNot(currentSpec.contains).foreach { case (sourceId, transform, name) =>
353           updateSpec.addField(name, Expressions.transform(schema.schema.findColumnName(sourceId), transform))
354         }
355         updateSpec.commit()
356       }
357     }
