22 case class Z2Scheme(attribute: String, index: Int, bits: Int) extends SpatialScheme {
23 
24   import org.locationtech.geomesa.filter.{andFilters, ff}
25   import org.locationtech.geomesa.utils.geotools.CRS_EPSG_4326
26 
27   require(bits % 4 == 0, s"Bit precision must be a multiple of 4, but received $bits")
28 
29   private val z2 = Z2SFC
30 
31   // number of hex digits used to represent our z value
32   override protected val digits: Int = bits / 4
33 
34   // in getCoveringFilter, expand the bbox by half a cell to account for round-trip errors
35   // invert returns the center of the cell, so we need to expand to the edges
36   // cell size at 31-bit precision: lon = 360.0 / 2^31, lat = 180.0 / 2^31
37   private val xDelta = ((z2.lon.max - z2.lon.min) / (1L << z2.precision)) / 2.0
38   private val yDelta = ((z2.lat.max - z2.lat.min) / (1L << z2.precision)) / 2.0
39 
40   override val name: String = s"${Z2Scheme.name}:attribute=$attribute:bits=$bits"
41 
42   override val column: String = ZValueField.z2FieldName(ColumnName.encode(attribute))
43 
44   override def getPartition(feature: SimpleFeature): PartitionKey = {
45     val pt = feature.getAttribute(index).asInstanceOf[Point]
46     PartitionKey(name, z2.hexEncode(pt.getX, pt.getY).take(digits))
47   }
48 
49   override def getCoveringFilter(partition: PartitionKey): Filter = {
50     val lower = z2.hexDecode(partition.value.padTo(16, '0'))
51     val upper = z2.hexDecode(partition.value.padTo(16, 'f'))
52     val (xmin, ymin) = z2.invert(lower) match { case (xmin, ymin) => (xmin - xDelta, ymin - yDelta) }
53     val (xmax, ymax) = z2.invert(upper) match { case (xmax, ymax) => (xmax + xDelta, ymax + yDelta) }
54 
55     val bbox = ff.bbox(ff.property(attribute), new ReferencedEnvelope(xmin, xmax, ymin, ymax, CRS_EPSG_4326))
56     // account for borders between z-cells (make upper bounds exclusive except on the upper-right edge)
57     val xExclusive = if (xmax == z2.lon.max) { None } else {
58       Some(ff.less(ff.function("getX", ff.property(attribute)), ff.literal(xmax)))
59     }
60     val yExclusive = if (ymax == z2.lat.max) { None } else {
61       Some(ff.less(ff.function("getY", ff.property(attribute)), ff.literal(ymax)))
62     }
63     andFilters(Seq(bbox) ++ xExclusive ++ yExclusive)
64   }
65 
66   override def spec(b: PartitionSpec.Builder): PartitionSpec.Builder =
67     b.truncate(ZValueField.z2FieldName(ColumnName.encode(attribute)), digits)
68 
69   override def getPartition(partition: StructLike, i: Int): PartitionKey = PartitionKey(name, partition.get(i, classOf[String]))
70 
71   override protected def hexRanges(bounds: Seq[(Double, Double, Double, Double)]): Seq[(String, String)] = {
72     val max = QueryProperties.ScanRangesTarget.toInt
73     z2.ranges(bounds, maxRanges = max).map(r => (z2.hexEncode(r.lower), z2.hexEncode(r.upper)))
74   }
75 }
