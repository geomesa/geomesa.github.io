26 case class Z2Scheme(attribute: String, index: Int, bits: Int) extends SpatialScheme {
27 
28   import org.locationtech.geomesa.filter.{andFilters, ff}
29   import org.locationtech.geomesa.utils.geotools.CRS_EPSG_4326
30 
31   require(bits % 4 == 0, s"Bit precision must be a multiple of 4, but received $bits")
32 
33   private val z2 = Z2SFC
34 
35   // number of hex digits used to represent our z value
36   override protected val digits: Int = bits / 4
37 
38   // in getCoveringFilter, expand the bbox by half a cell to account for round-trip errors
39   // invert returns the center of the cell, so we need to expand to the edges
40   // cell size at 31-bit precision: lon = 360.0 / 2^31, lat = 180.0 / 2^31
41   private val xDelta = ((z2.lon.max - z2.lon.min) / (1L << z2.precision)) / 2.0
42   private val yDelta = ((z2.lat.max - z2.lat.min) / (1L << z2.precision)) / 2.0
43 
44   override val name: String = s"${Z2Scheme.name}:attribute=$attribute:bits=$bits"
45 
46   override val column: String = ZValueField.z2FieldName(ColumnName.encode(attribute))
47 
48   override def getPartition(feature: SimpleFeature): PartitionKey = {
49     val pt = feature.getAttribute(index).asInstanceOf[Point]
50     PartitionKey(name, z2.hexEncode(pt.getX, pt.getY).take(digits))
51   }
52 
53   override def getCoveringFilter(partition: PartitionKey): Filter = {
54     val lower = z2.hexDecode(partition.value.padTo(16, '0'))
55     val upper = z2.hexDecode(partition.value.padTo(16, 'f'))
56     val (xmin, ymin) = z2.invert(lower) match { case (xmin, ymin) => (xmin - xDelta, ymin - yDelta) }
57     val (xmax, ymax) = z2.invert(upper) match { case (xmax, ymax) => (xmax + xDelta, ymax + yDelta) }
58 
59     val bbox = ff.bbox(ff.property(attribute), new ReferencedEnvelope(xmin, xmax, ymin, ymax, CRS_EPSG_4326))
60     // account for borders between z-cells (make upper bounds exclusive except on the upper-right edge)
61     val xExclusive = if (xmax == z2.lon.max) { None } else {
62       Some(ff.less(ff.function("getX", ff.property(attribute)), ff.literal(xmax)))
63     }
64     val yExclusive = if (ymax == z2.lat.max) { None } else {
65       Some(ff.less(ff.function("getY", ff.property(attribute)), ff.literal(ymax)))
66     }
67     andFilters(Seq(bbox) ++ xExclusive ++ yExclusive)
68   }
69 
70   override def getPartition(partition: StructLike, i: Int): PartitionKey = PartitionKey(name, partition.get(i, classOf[String]))
71 
72   override protected def hexRanges(bounds: Seq[(Double, Double, Double, Double)]): Seq[(String, String)] = {
73     val max = QueryProperties.ScanRangesTarget.toInt
74     z2.ranges(bounds, maxRanges = max).map(r => (z2.hexEncode(r.lower), z2.hexEncode(r.upper)))
75   }
76 }
