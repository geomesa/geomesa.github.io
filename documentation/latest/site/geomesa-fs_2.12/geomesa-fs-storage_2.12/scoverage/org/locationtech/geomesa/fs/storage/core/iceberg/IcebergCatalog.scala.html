53   // avoid repeatedly loading tables when getting type names
54   private val typeNameCache = Caffeine.newBuilder().expireAfterWrite(expiry, TimeUnit.MILLISECONDS).build(
55     new CacheLoader[TableIdentifier, Option[String]]() {
56       override def load(id: TableIdentifier): Option[String] = {
57         val table = catalog.loadTable(id)
58         try { Option(table.properties().get("geomesa.sft.name")) } finally {
59           CloseWithLogging(Option(table).collect { case c: Closeable => c })
60         }

65   override def getTypeNames: Seq[String] = {
66     if (catalog.namespaceExists(namespace)) {
67       catalog.listTables(namespace).asScala.toSeq.flatMap(id => typeNameCache.get(id))
68     } else {
69       Seq.empty
70     }

73   override def load(typeName: String): FileSystemStorage = {
74     val table = catalog.loadTable(tableId(typeName))
75     val ns = conf.get(StorageCatalog.NamespaceConfigKey)
76     val sft = {
77       val typeName = table.properties().get("geomesa.sft.name")
78       val attributes = table.schema().columns().asScala.flatMap(deriveDescriptor)
79       if (attributes.isEmpty) {
80         // back compatibility check
81         SimpleFeatureTypes.createType(ns.fold(typeName)(n => s"$n:$typeName"), table.properties().get("geomesa.sft.spec"))
82       } else {
83         val b = new SimpleFeatureTypeBuilder()
84         ns.foreach(b.setNamespaceURI)
85         b.setName(typeName)
86         b.addAll(attributes.asJava)
87         attributes.find(d => d.getUserData.get(AttributeOptions.OptDefault) == "true" && d.isInstanceOf[GeometryDescriptor]).foreach { d =>
88           b.setDefaultGeometry(d.getLocalName)
89         }
90         val sft = b.buildFeatureType()
91         table.properties().asScala.foreach { case (k, v) =>
92           if (k.startsWith(UserDataPrefix)) {
93             sft.getUserData.put(k.substring(UserDataPrefix.length), v)
94           }
95         }
96         sft
97       }
98     }
99     val schema = SimpleFeatureIcebergSchema(sft, conf)
100     val schemes = PartitionSchemeFactory.load(schema, table.spec())
101     FileSystemStorage(table, schemes, schema, conf)
102   }
103 
104   override def create(sft: SimpleFeatureType, partitions: Seq[String], targetFileSize: Option[Long] = None): FileSystemStorage = {
105     val schema = SimpleFeatureIcebergSchema(namespaced(sft, conf.get(StorageCatalog.NamespaceConfigKey)), conf)
106     if (schema.geometries != GeoParquetWkb) {
107       // TODO supports native geometry encoding
108       throw new UnsupportedOperationException(s"Only WKB geometry encoding is supported: ${schema.geometries}")
109     }
110     // load the partition scheme first in case it fails
111     val schemes = partitions.map(PartitionSchemeFactory.load(sft, _)).sortBy(_.name)
112     val tableProps = {
113       val typeName = Map("geomesa.sft.name" -> sft.getTypeName)
114       val userData = {
115         val prefixes = sft.getUserDataPrefixes
116         sft.getUserData.asScala.collect {
117           case (k, v) if v != null && prefixes.exists(k.toString.startsWith) => s"$UserDataPrefix$k" -> v.toString
118         }
119       }
120       val size = targetFileSize.map(s => s"${Metadata.PropertyPrefix}${Metadata.TargetFileSize}" -> s.toString).toMap
121       // for back-compatibility
122       val spec = Map("geomesa.sft.spec" -> SimpleFeatureTypes.encodeType(sft, includeUserData = true))
123       // file format v3 lets us use native geometries - but it's not yet supported in spark or trino
124       // val format = Map(TableProperties.FORMAT_VERSION -> "3")
125       typeName ++ userData ++ size ++ spec
126     }
127 
128     val spec = schemes.foldLeft(PartitionSpec.builderFor(schema.schema))((b, m) => m.spec(b)).build()
129     catalog.ensureNamespace(namespace)
130     val table = catalog.createTable(tableId(sft.getTypeName), schema.schema, spec, null, tableProps.asJava)
131     FileSystemStorage(table, schemes, schema, conf)
132   }
133 
134   override def close(): Unit = catalog.close()
135 
136   private def tableId(typeName: String): TableIdentifier = TableIdentifier.of(namespace, ColumnName.encode(typeName))
137 
138   private def deriveDescriptor(f: NestedField): Option[AttributeDescriptor] = {
139     if (f.name().startsWith(InternalFieldDelimiter) && f.name().endsWith(InternalFieldDelimiter)) { None } else {
140       Option(f.doc()).flatMap { d =>
141         try { Some(SimpleFeatureTypes.createDescriptor(d)) } catch {
142           case NonFatal(e) => logger.warn(s"Error parsing column doc as descriptor: $d", e); None
143         }
144       }
145     }
146   }
147 }

155   private implicit class RichConf(val conf: Map[String, String]) extends AnyVal {
156     def required(k: String): String =
157       conf.getOrElse(k, throw new IllegalArgumentException(s"Iceberg catalog requires configuration `$k` to be specified"))
158   }
159 
160   private implicit class RichCatalog(val catalog: Catalog) extends AnyVal {
161 
162     def ensureNamespace(namespace: Namespace): Unit = {
163       sn.foreach { supportsNamespace =>
164         if (!supportsNamespace.namespaceExists(namespace)) {
165           supportsNamespace.createNamespace(namespace)
166         }
167       }
168     }
169 
170     def namespaceExists(namespace: Namespace): Boolean = sn.forall(_.namespaceExists(namespace))
171 
172     def close(): Unit = CloseWithLogging(Option(catalog).collect { case c: Closeable => c })
173 
174     private def sn: Option[SupportsNamespaces] = Option(catalog).collect { case sn: SupportsNamespaces => sn }
175   }
176 
177   private def createCatalog(conf: Map[String, String]): Catalog = {
178     // add some defaults, to reduce boilerplate
179     val defaults = Map(
180       "io-impl" -> "org.apache.iceberg.aws.s3.S3FileIO",
181       "file-format" -> "PARQUET",
182     )
183     val props = defaults ++ conf
184     CatalogUtil.buildIcebergCatalog("geomesa", props.asJava, null)
185   }
