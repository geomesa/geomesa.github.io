33 class IcebergCatalog(config: Map[String, String]) extends StorageCatalog with LazyLogging {
34 
35   import IcebergCatalog.{RichCatalog, RichConf}
36 
37   import scala.collection.JavaConverters._
38 
39   val conf: Map[String, String] = S3ObjectStore.s3Configs(config)
40 
41   private val expiry = TableBasedMetadata.Expiry.toDuration.get.toMillis
42 
43   private val namespace = Namespace.of(ColumnName.encode(conf.required("iceberg.namespace")))
44   private val catalog = IcebergCatalog.createCatalog(conf)
45 
46   // avoid repeatedly loading tables when getting type names
47   private val typeNameCache = Caffeine.newBuilder().expireAfterWrite(expiry, TimeUnit.MILLISECONDS).build(
48     new CacheLoader[TableIdentifier, Option[String]]() {
49       override def load(id: TableIdentifier): Option[String] = {
50         val table = catalog.loadTable(id)
51         try { Option(table.properties().get("geomesa.sft.name")) } finally {
52           CloseWithLogging(Option(table).collect { case c: Closeable => c })
53         }
54       }
55     }
56   )
57 
58   override def getTypeNames: Seq[String] = {
59     if (catalog.namespaceExists(namespace)) {
60       catalog.listTables(namespace).asScala.toSeq.flatMap(id => typeNameCache.get(id))
61     } else {
62       Seq.empty
63     }
64   }
65 
66   override def load(typeName: String): FileSystemStorage = load(catalog.loadTable(tableId(typeName)))
67 
68   private def load(table: Table): FileSystemStorage = {
69     val ns = conf.get(StorageCatalog.NamespaceConfigKey)
70     val schema = SimpleFeatureIcebergSchema(table, ns)
71     val schemes = PartitionSchemeFactory.load(schema, table.spec())
72     FileSystemStorage(table, schemes, schema, conf)
73   }
74 
75   override def create(sft: SimpleFeatureType, partitions: Seq[String], targetFileSize: Option[Long] = None): FileSystemStorage = {
76     val geoms = conf.get(GeometryEncodingKey).map(GeometryEncoding.apply).getOrElse(GeometryEncoding.GeoParquetWkb)
77     val schema = SimpleFeatureIcebergSchema.create(sft, geoms)
78     // load the partition scheme first in case it fails
79     val schemes = partitions.map(PartitionSchemeFactory.load(sft, _)).sortBy(_.name)
80     val tableProps = IcebergCatalog.sftTableProperties(sft) ++ {
81       val size = targetFileSize.map(s => s"${Metadata.PropertyPrefix}${Metadata.TargetFileSize}" -> s.toString).toMap
82       // file format v3 lets us use variant encoding
83        val format =
84          Map(TableProperties.FORMAT_VERSION -> "3", TableProperties.DELETE_MODE -> RowLevelOperationMode.MERGE_ON_READ.modeName())
85       size ++ format
86     }
87     val spec = schemes.foldLeft(PartitionSpec.builderFor(schema))((b, m) => m.spec(b)).build()
88     catalog.ensureNamespace(namespace)
89     val table = catalog.createTable(tableId(sft.getTypeName), schema, spec, null, tableProps.asJava)
90     // re-load to pick up the correct schema field ids
91     load(table)
92   }
93 
94   override def close(): Unit = catalog.close()
95 
96   private def tableId(typeName: String): TableIdentifier = TableIdentifier.of(namespace, ColumnName.encode(typeName))
97 }

99 object IcebergCatalog {

101   import org.locationtech.geomesa.utils.geotools.RichSimpleFeatureType.RichSimpleFeatureType
102

104 
105   val UserDataPrefix = "geomesa.userdata."
106 
107   /**
108    * Prefix for properties describing a single column.
109    */
110   val ColumnPrefix = "geomesa.col."
111 
112   /**
113    * Property carrying the full attribute spec for one column, written for every column and read
114    * back by `SimpleFeatureIcebergSchema.apply`.
115    *
116    * @param column storage column name, as encoded by `ColumnName`
117    * @return property key
118    */
119   def columnSpecProperty(column: String): String = s"$ColumnPrefix$column.spec"
120 
121   /**
122    * Table-level properties needed to reconstruct the feature type
123    *
124    * @param sft simple feature type
125    * @return
126    */
127   def sftTableProperties(sft: SimpleFeatureType): Map[String, String] = {
128     val typeName = Map("geomesa.sft.name" -> sft.getTypeName)
129     val userData = {
130       val prefixes = sft.getUserDataPrefixes
131       sft.getUserData.asScala.collect {
132         case (k, v) if v != null && prefixes.exists(k.toString.startsWith) => s"$UserDataPrefix$k" -> v.toString
133       }
134     }
135     val columnSpecs = SimpleFeatureIcebergSchema.columnSpecs(sft).map {
136       case (column, spec) => columnSpecProperty(column) -> spec
137     }
138     typeName ++ userData ++ columnSpecs
139   }
140 
141   private implicit class RichConf(val conf: Map[String, String]) extends AnyVal {
142     def required(k: String): String =
143       conf.getOrElse(k, throw new IllegalArgumentException(s"Iceberg catalog requires configuration `$k` to be specified"))
144   }
145 
146   private implicit class RichCatalog(val catalog: Catalog) extends AnyVal {
147 
148     def ensureNamespace(namespace: Namespace): Unit = {
149       sn.foreach { supportsNamespace =>
150         if (!supportsNamespace.namespaceExists(namespace)) {
151           supportsNamespace.createNamespace(namespace)
152         }
153       }
154     }
155 
156     def namespaceExists(namespace: Namespace): Boolean = sn.forall(_.namespaceExists(namespace))
157 
158     def close(): Unit = CloseWithLogging(Option(catalog).collect { case c: Closeable => c })
159 
160     private def sn: Option[SupportsNamespaces] = Option(catalog).collect { case sn: SupportsNamespaces => sn }
161   }
162 
163   private def createCatalog(conf: Map[String, String]): Catalog = {
164     // add some defaults, to reduce boilerplate
165     val defaults = Map(
166       "io-impl" -> "org.apache.iceberg.aws.s3.S3FileIO",
167       "file-format" -> "PARQUET",
168     )
169     val props = defaults ++ conf
170     CatalogUtil.buildIcebergCatalog("geomesa", props.asJava, null)
171   }
