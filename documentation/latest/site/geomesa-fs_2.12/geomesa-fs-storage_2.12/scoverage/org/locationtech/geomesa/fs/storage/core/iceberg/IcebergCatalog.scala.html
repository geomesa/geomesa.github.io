31 class IcebergCatalog(val context: FileSystemContext) extends StorageCatalog {
32 
33   import IcebergCatalog.{RichCatalog, RichConf}
34 
35   import scala.collection.JavaConverters._
36 
37   private val expiry = TableBasedMetadata.Expiry.toDuration.get.toMillis
38 
39   private val namespace = Namespace.of(ColumnName.encode(context.conf.required("iceberg.namespace")))
40   private val catalog = IcebergCatalog.createCatalog(context)
41 
42   // avoid repeatedly loading tables when getting type names
43   private val typeNameCache = Caffeine.newBuilder().expireAfterWrite(expiry, TimeUnit.MILLISECONDS).build(
44     new CacheLoader[TableIdentifier, Option[String]]() {
45       override def load(id: TableIdentifier): Option[String] = {
46         val table = catalog.loadTable(id)
47         try { Option(table.properties().get("geomesa.sft.name")) } finally {
48           CloseWithLogging(Option(table).collect { case c: Closeable => c })
49         }
50       }
51     }
52   )
53 
54   override def getTypeNames: Seq[String] = {
55     if (catalog.namespaceExists(namespace)) {
56       catalog.listTables(namespace).asScala.toSeq.flatMap(id => typeNameCache.get(id))
57     } else {
58       Seq.empty
59     }
60   }
61 
62   override def load(typeName: String): FileSystemStorage = {
63     val table = catalog.loadTable(tableId(typeName))
64     val sft =
65       namespaced(
66         SimpleFeatureTypes.createType(table.properties().get("geomesa.sft.name"), table.properties().get("geomesa.sft.spec")),
67         context.namespace)
68     // TODO get this from the table itself to allow for scheme migration
69     val schemes = table.properties().get("geomesa.partition.spec").split(",").map(PartitionSchemeFactory.load(sft, _))
70     val schema = SimpleFeatureIcebergSchema(sft, context.conf)
71     FileSystemStorage(context, table, schemes, schema)
72   }
73 
74   override def create(sft: SimpleFeatureType, partitions: Seq[String], targetFileSize: Option[Long] = None): FileSystemStorage = {
75     val schema = SimpleFeatureIcebergSchema(namespaced(sft, context.namespace), context.conf)
76     if (schema.geometries != GeoParquetWkb) {
77       // TODO supports native geometry encoding
78       throw new UnsupportedOperationException(s"Only WKB geometry encoding is supported: ${schema.geometries}")
79     }
80     // load the partition scheme first in case it fails
81     val schemes = partitions.map(PartitionSchemeFactory.load(sft, _)).sortBy(_.name)
82     val tableProps = Map(
83       "geomesa.sft.name" -> sft.getTypeName,
84       "geomesa.sft.spec" -> SimpleFeatureTypes.encodeType(sft, includeUserData = true),
85       "geomesa.partition.spec" -> schemes.map(_.name).mkString(","),
86       // file format v3 lets us use native geometries - but it's not yet supported in spark or trino
87       // TableProperties.FORMAT_VERSION -> "3"
88     ) ++ targetFileSize.map(s => s"${Metadata.PropertyPrefix}${Metadata.TargetFileSize}" -> s.toString).toMap
89 
90     val spec = schemes.foldLeft(PartitionSpec.builderFor(schema.schema))((b, m) => m.spec(b)).build()
91     catalog.ensureNamespace(namespace)
92     val table = catalog.createTable(tableId(sft.getTypeName), schema.schema, spec, null, tableProps.asJava)
93     FileSystemStorage(context, table, schemes, schema)
94   }
95 
96   override def close(): Unit = catalog.close()
97 
98   private def tableId(typeName: String): TableIdentifier = TableIdentifier.of(namespace, ColumnName.encode(typeName))
99 }

101 object IcebergCatalog {

103   import scala.collection.JavaConverters._
104 
105   private implicit class RichConf(val conf: Map[String, String]) extends AnyVal {
106     def required(k: String): String =
107       conf.getOrElse(k, throw new IllegalArgumentException(s"Iceberg catalog requires configuration `$k` to be specified"))
108   }
109 
110   private implicit class RichCatalog(val catalog: Catalog) extends AnyVal {
111 
112     def ensureNamespace(namespace: Namespace): Unit = {
113       sn.foreach { supportsNamespace =>
114         if (!supportsNamespace.namespaceExists(namespace)) {
115           supportsNamespace.createNamespace(namespace)
116         }
117       }
118     }
119 
120     def namespaceExists(namespace: Namespace): Boolean = sn.forall(_.namespaceExists(namespace))
121 
122     def close(): Unit = CloseWithLogging(Option(catalog).collect { case c: Closeable => c })
123 
124     private def sn: Option[SupportsNamespaces] = Option(catalog).collect { case sn: SupportsNamespaces => sn }
125   }
126 
127   private def createCatalog(context: FileSystemContext): Catalog = {
128     // add some defaults, to reduce boilerplate
129     val defaults = Map(
130       "io-impl" -> "org.apache.iceberg.aws.s3.S3FileIO",
131       "file-format" -> "PARQUET",
132       "warehouse" -> context.root.resolve("metadata/").toString
133     )
134     val props = defaults ++ context.conf
135     CatalogUtil.buildIcebergCatalog("geomesa", props.asJava, null)
136   }
