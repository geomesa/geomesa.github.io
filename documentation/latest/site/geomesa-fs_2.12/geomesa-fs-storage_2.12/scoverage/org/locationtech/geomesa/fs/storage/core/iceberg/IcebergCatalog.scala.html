33

36   import scala.collection.JavaConverters._

38   val conf: Map[String, String] = S3ObjectStore.s3Configs(config)
39 
40   private val expiry = TableBasedMetadata.Expiry.toDuration.get.toMillis
41 
42   private val namespace = Namespace.of(ColumnName.encode(conf.required("iceberg.namespace")))
43   private val catalog = IcebergCatalog.createCatalog(conf)
44 
45   // avoid repeatedly loading tables when getting type names
46   private val typeNameCache = Caffeine.newBuilder().expireAfterWrite(expiry, TimeUnit.MILLISECONDS).build(
47     new CacheLoader[TableIdentifier, Option[String]]() {
48       override def load(id: TableIdentifier): Option[String] = {
49         val table = catalog.loadTable(id)
50         try { Option(table.properties().get("geomesa.sft.name")) } finally {
51           CloseWithLogging(Option(table).collect { case c: Closeable => c })
52         }
53       }
54     }
55   )
56 
57   override def getTypeNames: Seq[String] = {
58     if (catalog.namespaceExists(namespace)) {
59       catalog.listTables(namespace).asScala.toSeq.flatMap(id => typeNameCache.get(id))
60     } else {
61       Seq.empty
62     }
63   }
64 
65   override def load(typeName: String): FileSystemStorage = load(catalog.loadTable(tableId(typeName)))
66 
67   private def load(table: Table): FileSystemStorage = {
68     val ns = conf.get(StorageCatalog.NamespaceConfigKey)
69     val schema = SimpleFeatureIcebergSchema(table, ns)
70     val schemes = PartitionSchemeFactory.load(schema, table.spec())
71     FileSystemStorage(table, schemes, schema, conf ++ ParquetFileSystemWriter.icebergBloomFilterConf(table.properties()))
72   }
73 
74   override def create(
75       sft: SimpleFeatureType,
76       partitions: Seq[String],
77       targetFileSize: Option[Long] = None,
78       bloomFilters: Seq[BloomFilterConfig] = Seq.empty): FileSystemStorage = {
79     val schema = SimpleFeatureIcebergSchema.create(sft)
80     // load the partition scheme first in case it fails
81     val schemes = partitions.map(PartitionSchemeFactory.load(sft, _)).sortBy(_.name)
82     val tableProps = IcebergCatalog.sftTableProperties(sft) ++ {
83       val size = targetFileSize.map(s => s"${Metadata.PropertyPrefix}${Metadata.TargetFileSize}" -> s.toString).toMap
84       // file format v3 lets us use variant encoding
85        val format =
86          Map(TableProperties.FORMAT_VERSION -> "3", TableProperties.DELETE_MODE -> RowLevelOperationMode.MERGE_ON_READ.modeName())
87       // persisted as the standard iceberg properties, so that other iceberg writers (e.g. compaction) honor them too
88       size ++ format ++ BloomFilterConfig.tableProperties(bloomFilters)
89     }
90     val spec = schemes.foldLeft(PartitionSpec.builderFor(schema))((b, m) => m.spec(b)).build()
91     catalog.ensureNamespace(namespace)
92     val table = catalog.createTable(tableId(sft.getTypeName), schema, spec, null, tableProps.asJava)
93     // re-load to pick up the correct schema field ids
94     load(table)
95   }
96 
97   override def close(): Unit = catalog.close()
98 
99   private def tableId(typeName: String): TableIdentifier = TableIdentifier.of(namespace, ColumnName.encode(typeName))
100 }

102 object IcebergCatalog {
103 
104   import org.locationtech.geomesa.utils.geotools.RichSimpleFeatureType.RichSimpleFeatureType

106   import scala.collection.JavaConverters._
107 
108   val UserDataPrefix = "geomesa.userdata."
109 
110   /**
111    * Prefix for properties describing a single column.
112    */
113   val ColumnPrefix = "geomesa.col."
114 
115   /**
116    * Property carrying the full attribute spec for one column, written for every column and read
117    * back by `SimpleFeatureIcebergSchema.apply`.
118    *
119    * @param column storage column name, as encoded by `ColumnName`
120    * @return property key
121    */
122   def columnSpecProperty(column: String): String = s"$ColumnPrefix$column.spec"
123 
124   /**
125    * Table-level properties needed to reconstruct the feature type
126    *
127    * @param sft simple feature type
128    * @return
129    */
130   def sftTableProperties(sft: SimpleFeatureType): Map[String, String] = {
131     val typeName = Map("geomesa.sft.name" -> sft.getTypeName)
132     val userData = {
133       val prefixes = sft.getUserDataPrefixes
134       sft.getUserData.asScala.collect {
135         case (k, v) if v != null && prefixes.exists(k.toString.startsWith) => s"$UserDataPrefix$k" -> v.toString
136       }
137     }
138     val columnSpecs = SimpleFeatureIcebergSchema.columnSpecs(sft).map {
139       case (column, spec) => columnSpecProperty(column) -> spec
140     }
141     typeName ++ userData ++ columnSpecs
142   }
143 
144   private implicit class RichConf(val conf: Map[String, String]) extends AnyVal {
145     def required(k: String): String =
146       conf.getOrElse(k, throw new IllegalArgumentException(s"Iceberg catalog requires configuration `$k` to be specified"))
147   }
148 
149   private implicit class RichCatalog(val catalog: Catalog) extends AnyVal {
150 
151     def ensureNamespace(namespace: Namespace): Unit = {
152       sn.foreach { supportsNamespace =>
153         if (!supportsNamespace.namespaceExists(namespace)) {
154           supportsNamespace.createNamespace(namespace)
155         }
156       }
157     }
158 
159     def namespaceExists(namespace: Namespace): Boolean = sn.forall(_.namespaceExists(namespace))
160 
161     def close(): Unit = CloseWithLogging(Option(catalog).collect { case c: Closeable => c })
162 
163     private def sn: Option[SupportsNamespaces] = Option(catalog).collect { case sn: SupportsNamespaces => sn }
164   }
165 
166   private def createCatalog(conf: Map[String, String]): Catalog = {
167     // add some defaults, to reduce boilerplate
168     val defaults = Map(
169       "io-impl" -> "org.apache.iceberg.aws.s3.S3FileIO",
170       "file-format" -> "PARQUET",
171     )
172     val props = defaults ++ conf
173     CatalogUtil.buildIcebergCatalog("geomesa", props.asJava, null)
174   }
