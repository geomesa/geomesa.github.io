37 class IcebergCatalog(config: Map[String, String]) extends StorageCatalog with LazyLogging {
38 
39   import IcebergCatalog.{RichCatalog, RichConf}

41   import scala.collection.JavaConverters._

43   val conf: Map[String, String] = S3ObjectStore.s3Configs(config)
44 
45   private val expiry = TableBasedMetadata.Expiry.toDuration.get.toMillis
46 
47   private val namespace = Namespace.of(ColumnName.encode(conf.required("iceberg.namespace")))
48   private val catalog = IcebergCatalog.createCatalog(conf)
49 
50   // avoid repeatedly loading tables when getting type names
51   private val typeNameCache = Caffeine.newBuilder().expireAfterWrite(expiry, TimeUnit.MILLISECONDS).build(
52     new CacheLoader[TableIdentifier, Option[String]]() {
53       override def load(id: TableIdentifier): Option[String] = {
54         WithClose(catalog.loadTable(id)) { table =>
55           Option(table.properties().get(IcebergCatalog.TypeNameKey))
56         }
57       }
58     }
59   )
60 
61   override def getTypeNames: Seq[String] = {
62     if (catalog.namespaceExists(namespace)) {
63       catalog.listTables(namespace).asScala.toSeq.flatMap(id => typeNameCache.get(id))
64     } else {
65       Seq.empty
66     }
67   }
68 
69   override def load(typeName: String): FileSystemStorage = load(catalog.loadTable(tableId(typeName)))
70 
71   private def load(table: Table): FileSystemStorage = {
72     val ns = conf.get(StorageCatalog.NamespaceConfigKey)
73     val schema = SimpleFeatureIcebergSchema(table, ns)
74     val schemes = PartitionSchemeFactory.load(schema, table.spec())
75     FileSystemStorage(table, schemes, schema, conf ++ ParquetFileSystemWriter.icebergBloomFilterConf(table.properties()))
76   }
77 
78   override def create(
79       sft: SimpleFeatureType,
80       partitions: Seq[String],
81       targetFileSize: Option[Long] = None,
82       bloomFilters: Seq[BloomFilterConfig] = Seq.empty): FileSystemStorage = {
83     val schema = SimpleFeatureIcebergSchema.create(sft)
84     // load the partition scheme first in case it fails
85     val schemes = partitions.map(PartitionSchemeFactory.load(sft, _)).sortBy(_.name)
86     val tableProps = IcebergCatalog.sftTableProperties(sft) ++ {
87       val size = targetFileSize.map(s => s"${Metadata.PropertyPrefix}${Metadata.TargetFileSize}" -> s.toString).toMap
88       // file format v3 lets us use variant encoding
89       val format = Map(TableProperties.FORMAT_VERSION -> "3")
90       val deleteMode = Map(TableProperties.DELETE_MODE -> RowLevelOperationMode.MERGE_ON_READ.modeName())
91       // persisted as the standard iceberg properties, so that other iceberg writers (e.g. compaction) honor them too
92       val blooms = BloomFilterConfig.tableProperties(bloomFilters)
93       size ++ format ++ deleteMode ++ blooms
94     }
95     val spec = schemes.foldLeft(PartitionSpec.builderFor(schema))((b, m) => m.spec(b)).build()
96     catalog.ensureNamespace(namespace)
97     val table = catalog.createTable(tableId(sft.getTypeName), schema, spec, null, tableProps.asJava)
98     // re-load to pick up the correct schema field ids
99     load(table)
100   }
101 
102   override def update(typeName: String): CatalogUpdate = new IcebergSchemaEvolution(catalog, tableId(typeName))
103 
104   override def delete(typeName: String, purge: Boolean): Unit = {
105     val table = tableId(typeName)
106     catalog.dropTable(table, purge)
107     typeNameCache.invalidate(table)
108   }
109 
110   override def close(): Unit = catalog.close()
111 
112   private def tableId(typeName: String): TableIdentifier = TableIdentifier.of(namespace, ColumnName.encode(typeName))
113 }

121   val TypeNameKey = s"${GeoMesaPropertyPrefix}sft.name"
122 
123   val UserDataPrefix = s"${GeoMesaPropertyPrefix}userdata."
124 
125   /**
126    * Prefix for properties describing a single column.
127    */
128   val ColumnPrefix = s"${GeoMesaPropertyPrefix}col."
129 
130   /**
131    * Property carrying the full attribute spec for one column, written for every column and read
132    * back by `SimpleFeatureIcebergSchema.apply`.
133    *
134    * @param column storage column name, as encoded by `ColumnName`
135    * @return property key
136    */
137   def columnSpecProperty(column: String): String = s"$ColumnPrefix$column.spec"
138 
139   /**
140    * Table-level properties needed to reconstruct the feature type
141    *
142    * @param sft simple feature type
143    * @return
144    */
145   def sftTableProperties(sft: SimpleFeatureType): Map[String, String] = {
146     val typeName = Map(TypeNameKey -> sft.getTypeName)
147     val userData = {
148       val prefixes = sft.getUserDataPrefixes
149       sft.getUserData.asScala.collect {
150         case (k, v) if v != null && prefixes.exists(k.toString.startsWith) => s"$UserDataPrefix$k" -> v.toString
151       }
152     }
153     val columnSpecs = sft.getAttributeDescriptors.asScala.map(sftAttributeProperty(sft, _)).toMap
154     typeName ++ userData ++ columnSpecs
155   }
156 
157   /**
158    * Property for a single attribute
159    *
160    * @param sft simple feature type
161    * @param d attribute
162    * @return
163    */
164   def sftAttributeProperty(sft: SimpleFeatureType, d: AttributeDescriptor): (String, String) =
165     columnSpecProperty(ColumnName.encode(d.getLocalName)) -> SimpleFeatureTypes.encodeDescriptor(sft, d)
166 
167   private implicit class RichConf(val conf: Map[String, String]) extends AnyVal {
168     def required(k: String): String =
169       conf.getOrElse(k, throw new IllegalArgumentException(s"Iceberg catalog requires configuration `$k` to be specified"))
170   }
171 
172   private implicit class RichCatalog(val catalog: Catalog) extends AnyVal {
173 
174     def ensureNamespace(namespace: Namespace): Unit = {
175       sn.foreach { supportsNamespace =>
176         if (!supportsNamespace.namespaceExists(namespace)) {
177           supportsNamespace.createNamespace(namespace)
178         }
179       }
180     }
181 
182     def namespaceExists(namespace: Namespace): Boolean = sn.forall(_.namespaceExists(namespace))
183 
184     def close(): Unit = CloseWithLogging(Option(catalog).collect { case c: Closeable => c })
185 
186     private def sn: Option[SupportsNamespaces] = Option(catalog).collect { case sn: SupportsNamespaces => sn }
187   }
188 
189   private def createCatalog(conf: Map[String, String]): Catalog = {
190     // add some defaults, to reduce boilerplate
191     val defaults = Map(
192       "io-impl" -> "org.apache.iceberg.aws.s3.S3FileIO",
193       "file-format" -> "PARQUET",
194     )
195     val props = defaults ++ conf
196     CatalogUtil.buildIcebergCatalog("geomesa", props.asJava, null)
197   }
