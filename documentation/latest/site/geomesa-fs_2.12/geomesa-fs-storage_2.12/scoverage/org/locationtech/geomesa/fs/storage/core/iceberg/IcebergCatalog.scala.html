32 class IcebergCatalog(config: Map[String, String]) extends StorageCatalog {
33 
34   import IcebergCatalog.{RichCatalog, RichConf}
35 
36   import scala.collection.JavaConverters._
37 
38   val conf: Map[String, String] = S3ObjectStore.s3Configs(config)
39 
40   private val expiry = TableBasedMetadata.Expiry.toDuration.get.toMillis
41 
42   private val namespace = Namespace.of(ColumnName.encode(conf.required("iceberg.namespace")))
43   private val catalog = IcebergCatalog.createCatalog(conf)
44 
45   // avoid repeatedly loading tables when getting type names
46   private val typeNameCache = Caffeine.newBuilder().expireAfterWrite(expiry, TimeUnit.MILLISECONDS).build(
47     new CacheLoader[TableIdentifier, Option[String]]() {
48       override def load(id: TableIdentifier): Option[String] = {
49         val table = catalog.loadTable(id)
50         try { Option(table.properties().get("geomesa.sft.name")) } finally {
51           CloseWithLogging(Option(table).collect { case c: Closeable => c })
52         }
53       }
54     }
55   )
56 
57   override def getTypeNames: Seq[String] = {
58     if (catalog.namespaceExists(namespace)) {
59       catalog.listTables(namespace).asScala.toSeq.flatMap(id => typeNameCache.get(id))
60     } else {
61       Seq.empty
62     }
63   }
64 
65   override def load(typeName: String): FileSystemStorage = {
66     val table = catalog.loadTable(tableId(typeName))
67     val sft =
68       namespaced(
69         SimpleFeatureTypes.createType(table.properties().get("geomesa.sft.name"), table.properties().get("geomesa.sft.spec")),
70         conf.get(StorageCatalog.NamespaceConfigKey))
71     // TODO get this from the table itself to allow for scheme migration
72     val schemes = table.properties().get("geomesa.partition.spec").split(",").map(PartitionSchemeFactory.load(sft, _))
73     val schema = SimpleFeatureIcebergSchema(sft, conf)
74     FileSystemStorage(table, schemes, schema, conf)
75   }
76 
77   override def create(sft: SimpleFeatureType, partitions: Seq[String], targetFileSize: Option[Long] = None): FileSystemStorage = {
78     val schema = SimpleFeatureIcebergSchema(namespaced(sft, conf.get(StorageCatalog.NamespaceConfigKey)), conf)
79     if (schema.geometries != GeoParquetWkb) {
80       // TODO supports native geometry encoding
81       throw new UnsupportedOperationException(s"Only WKB geometry encoding is supported: ${schema.geometries}")
82     }
83     // load the partition scheme first in case it fails
84     val schemes = partitions.map(PartitionSchemeFactory.load(sft, _)).sortBy(_.name)
85     val tableProps = Map(
86       "geomesa.sft.name" -> sft.getTypeName,
87       "geomesa.sft.spec" -> SimpleFeatureTypes.encodeType(sft, includeUserData = true),
88       "geomesa.partition.spec" -> schemes.map(_.name).mkString(","),
89       // file format v3 lets us use native geometries - but it's not yet supported in spark or trino
90       // TableProperties.FORMAT_VERSION -> "3"
91     ) ++ targetFileSize.map(s => s"${Metadata.PropertyPrefix}${Metadata.TargetFileSize}" -> s.toString).toMap
92 
93     val spec = schemes.foldLeft(PartitionSpec.builderFor(schema.schema))((b, m) => m.spec(b)).build()
94     catalog.ensureNamespace(namespace)
95     val table = catalog.createTable(tableId(sft.getTypeName), schema.schema, spec, null, tableProps.asJava)
96     FileSystemStorage(table, schemes, schema, conf)
97   }
98 
99   override def close(): Unit = catalog.close()
100 
101   private def tableId(typeName: String): TableIdentifier = TableIdentifier.of(namespace, ColumnName.encode(typeName))
102 }

104 object IcebergCatalog {
105 
106   import scala.collection.JavaConverters._
107 
108   private implicit class RichConf(val conf: Map[String, String]) extends AnyVal {
109     def required(k: String): String =
110       conf.getOrElse(k, throw new IllegalArgumentException(s"Iceberg catalog requires configuration `$k` to be specified"))
111   }
112 
113   private implicit class RichCatalog(val catalog: Catalog) extends AnyVal {
114 
115     def ensureNamespace(namespace: Namespace): Unit = {
116       sn.foreach { supportsNamespace =>
117         if (!supportsNamespace.namespaceExists(namespace)) {
118           supportsNamespace.createNamespace(namespace)
119         }
120       }
121     }
122 
123     def namespaceExists(namespace: Namespace): Boolean = sn.forall(_.namespaceExists(namespace))
124 
125     def close(): Unit = CloseWithLogging(Option(catalog).collect { case c: Closeable => c })
126 
127     private def sn: Option[SupportsNamespaces] = Option(catalog).collect { case sn: SupportsNamespaces => sn }
128   }
129 
130   private def createCatalog(conf: Map[String, String]): Catalog = {
131     // add some defaults, to reduce boilerplate
132     val defaults = Map(
133       "io-impl" -> "org.apache.iceberg.aws.s3.S3FileIO",
134       "file-format" -> "PARQUET",
135     )
136     val props = defaults ++ conf
137     CatalogUtil.buildIcebergCatalog("geomesa", props.asJava, null)
138   }
