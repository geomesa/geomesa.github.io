23 /**
24  * Parse a value out of an iceberg column stat
25  */
26 trait FileBoundsParser[T] {
27 
28   def apply(buffer: ByteBuffer): T
29 
30   protected def parse[U](fieldType: Type, buffer: ByteBuffer): U = Conversions.fromByteBuffer[U](fieldType, buffer)
31 }
32 
33 object FileBoundsParser {
34 
35   import org.locationtech.geomesa.utils.geotools.RichAttributeDescriptors.RichAttributeDescriptor
36 
37   /**
38    * Create a parser for a given attribute
39    *
40    * @param descriptor attribute descriptor
41    * @param fieldType iceberg field type
42    * @tparam T binding of the attribute
43    * @return
44    */
45   def apply[T](descriptor: AttributeDescriptor, fieldType: Type): Option[FileBoundsParser[T]] = {
46     val types = ObjectType.selectType(descriptor)
47     val parser = types.head match {
48       case ObjectType.STRING if descriptor.getJsonSchema().isEmpty => Some(StringParser)
49 
50       case ObjectType.DATE  => Some(DateParser)
51       case ObjectType.BYTES => Some(BytesParser)
52 
53       case ObjectType.GEOMETRY =>
54         val encoding = descriptor.getEncoding.fold[GeometryEncoding](GeoParquetWkb)(GeometryEncoding.apply)
55         if (encoding != GeometryEncoding.GeoParquetWkb) {
56           throw new UnsupportedOperationException(encoding.toString)
57         }
58         Some(WkbParser)
59 
60       case ObjectType.LIST | ObjectType.MAP | ObjectType.STRING => None
61 
62       case _ => Some(new GenericParser(fieldType))
63     }
64 
65     parser.asInstanceOf[Option[FileBoundsParser[T]]]
66   }
67 
68   private object WkbParser extends FileBoundsParser[Geometry] {
69     override def apply(buffer: ByteBuffer): Geometry = {
70       val parsed = parse[ByteBuffer](Types.BinaryType.get(), buffer)
71       if (parsed == null) { null } else {
72         val buf = Array.ofDim[Byte](parsed.remaining())
73         parsed.get(buf, 0, buf.length)
74         WKBUtils.read(buf)
75       }
76     }
77   }
78 
79   object StringParser extends FileBoundsParser[String] {
80     override def apply(buffer: ByteBuffer): String = {
81       val charseq = parse[CharSequence](Types.StringType.get(), buffer)
82       if (charseq == null) { null } else { charseq.toString }
83     }
84   }
85 
86   private object DateParser extends FileBoundsParser[Date] {
87     override def apply(buffer: ByteBuffer): Date = {
88       val micros = parse[java.lang.Long](Types.LongType.get(), buffer)
89       if (micros == null) { null } else { Date.from(Instant.ofEpochMilli(micros / 1000L)) }
90     }
91   }
92 
93   private object BytesParser extends FileBoundsParser[Array[Byte]] {
94     override def apply(buffer: ByteBuffer): Array[Byte] = {
95       val parsed = parse[ByteBuffer](Types.BinaryType.get(), buffer)
96       val buf = Array.ofDim[Byte](parsed.remaining())
97       parsed.get(buf, 0, buf.length)
98       buf
99     }
100   }
101 
102   private class GenericParser(fieldType: Type) extends FileBoundsParser[AnyRef] {
103     override def apply(buffer: ByteBuffer): AnyRef = parse(fieldType, buffer)
104   }
