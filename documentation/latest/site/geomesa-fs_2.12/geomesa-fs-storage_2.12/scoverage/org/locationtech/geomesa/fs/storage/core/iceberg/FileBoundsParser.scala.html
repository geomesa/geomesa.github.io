22 /**
23  * Parse a value out of an iceberg column stat
24  */
25 trait FileBoundsParser[T] {
26 
27   def apply(buffer: ByteBuffer): T
28 
29   protected def parse[U](fieldType: Type, buffer: ByteBuffer): U = Conversions.fromByteBuffer[U](fieldType, buffer)
30 }
31 
32 object FileBoundsParser {
33 
34   import org.locationtech.geomesa.utils.geotools.RichAttributeDescriptors.RichAttributeDescriptor
35 
36   /**
37    * Create a parser for a given attribute
38    *
39    * @param descriptor attribute descriptor
40    * @param fieldType iceberg field type
41    * @tparam T binding of the attribute
42    * @return
43    */
44   def apply[T](descriptor: AttributeDescriptor, fieldType: Type): Option[FileBoundsParser[T]] = {
45     val types = ObjectType.selectType(descriptor)
46     val parser = types.head match {
47       case ObjectType.STRING if descriptor.getJsonSchema().isEmpty => Some(StringParser)
48 
49       case ObjectType.DATE  => Some(DateParser)
50       case ObjectType.BYTES => Some(BytesParser)
51 
52       case ObjectType.GEOMETRY =>
53         val encoding = descriptor.getUserData.get(SimpleFeatureIcebergSchema.GeometryEncodingKey) match {
54           case e: String => GeometryEncoding(e)
55           case _ => GeometryEncoding.GeoParquetWkb
56         }
57         if (encoding != GeometryEncoding.GeoParquetWkb) {
58           throw new UnsupportedOperationException(encoding.toString)
59         }
60         Some(WkbParser)
61 
62       case ObjectType.LIST | ObjectType.MAP | ObjectType.STRING => None
63 
64       case _ => Some(new GenericParser(fieldType))
65     }
66 
67     parser.asInstanceOf[Option[FileBoundsParser[T]]]
68   }
69 
70   private object WkbParser extends FileBoundsParser[Geometry] {
71     override def apply(buffer: ByteBuffer): Geometry = {
72       val parsed = parse[ByteBuffer](Types.BinaryType.get(), buffer)
73       if (parsed == null) { null } else {
74         val buf = Array.ofDim[Byte](parsed.remaining())
75         parsed.get(buf, 0, buf.length)
76         WKBUtils.read(buf)
77       }
78     }
79   }
80 
81   object StringParser extends FileBoundsParser[String] {
82     override def apply(buffer: ByteBuffer): String = {
83       val charseq = parse[CharSequence](Types.StringType.get(), buffer)
84       if (charseq == null) { null } else { charseq.toString }
85     }
86   }
87 
88   private object DateParser extends FileBoundsParser[Date] {
89     override def apply(buffer: ByteBuffer): Date = {
90       val micros = parse[java.lang.Long](Types.LongType.get(), buffer)
91       if (micros == null) { null } else { Date.from(Instant.ofEpochMilli(micros / 1000L)) }
92     }
93   }
94 
95   private object BytesParser extends FileBoundsParser[Array[Byte]] {
96     override def apply(buffer: ByteBuffer): Array[Byte] = {
97       val parsed = parse[ByteBuffer](Types.BinaryType.get(), buffer)
98       val buf = Array.ofDim[Byte](parsed.remaining())
99       parsed.get(buf, 0, buf.length)
100       buf
101     }
102   }
103 
104   private class GenericParser(fieldType: Type) extends FileBoundsParser[AnyRef] {
105     override def apply(buffer: ByteBuffer): AnyRef = parse(fieldType, buffer)
106   }
