31 class ParquetFileSystemReader(
32     fs: ObjectStore,
33     conf: Map[String, String],
34     sft: SimpleFeatureType,
35     parquetFilter: FilterCompat.Filter,
36     gtFilter: Option[org.geotools.api.filter.Filter],
37     visFilter: SimpleFeature => Boolean,
38     transform: Option[(String, SimpleFeatureType)]
39   ) extends FileSystemPathReader with LazyLogging {
40 
41   import scala.collection.JavaConverters._
42 
43   private val gtf = gtFilter.orNull
44 
45   private val transformFeature: SimpleFeature => SimpleFeature = transform match {
46     case None => null
47     case Some((tdefs, tsft)) =>
48       val definitions = Transforms(sft, tdefs).toArray
49       f => new TransformSimpleFeature(tsft, definitions, f)
50   }
51 
52   private val parquetConf = new PlainParquetConfiguration(conf.asJava)
53   SimpleFeatureParquetSchema.setSft(parquetConf, sft)
54 
55   override def read(file: URI): CloseableIterator[SimpleFeature] = {
56     // TODO we can examine file bounds and simplify/eliminate the gtFilter
57     logger.debug(s"Opening reader for path $file")
58     new ParquetFileIterator(file)
59   }
60 
61   private class ParquetFileIterator(path: URI) extends CloseableIterator[SimpleFeature] {
62 
63     private val reader: ParquetReader[SimpleFeature] =
64       ParquetFileSystemReader.builder(fs, path).withConf(parquetConf).withFilter(parquetFilter).build()
65 
66     private var staged: SimpleFeature = _
67 
68     override def close(): Unit = {
69       logger.debug(s"Closing reader for path $path")
70       reader.close()
71     }
72 
73     override def next(): SimpleFeature = {
74       val res = staged
75       staged = null
76       res
77     }
78 
79     @tailrec
80     override final def hasNext: Boolean = {
81       if (staged != null) { true } else {
82         val read = try { reader.read() } catch {
83           case NonFatal(e) => logger.error(s"Error reading file '$path'", e); null
84         }
85         if (read == null) {
86           false
87         } else if (visFilter(read) && (gtf == null || gtf.evaluate(read))) {
88           staged = if (transformFeature == null) { read } else { transformFeature(read) }
89           true
90         } else {
91           hasNext
92         }
93       }
94     }
95   }
96 }

100   /**
101    * Create a new configurable reader
102    *
103    * @param fs object store
104    * @param path file to read
105    * @return
106    */
107   def builder(fs: ObjectStore, path: URI): Builder = new Builder(inputFile(fs, path))
108 
109   /**
110    * Get an input file compatible with the parquet api
111    *
112    * @param fs object store
113    * @param path file path
114    * @return
115    */
116   def inputFile(fs: ObjectStore, path: URI): InputFile = fs match {
117     case _: LocalObjectStore => new LocalInputFile(Path.of(path))
118     case s3: S3ObjectStore   => new S3InputFile(s3, path)
119     case _ => throw new UnsupportedOperationException(s"No file implementation for scheme ${fs.scheme}")
120   }
121 
122   class Builder(file: InputFile) extends ParquetReader.Builder[SimpleFeature](file) {
123     override protected def getReadSupport: ReadSupport[SimpleFeature] = new SimpleFeatureReadSupport()
124   }
