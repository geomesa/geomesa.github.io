32 class ParquetFileSystemReader(
33     fs: ObjectStore,
34     context: FileSystemContext,
35     sft: SimpleFeatureType,
36     parquetFilter: FilterCompat.Filter,
37     gtFilter: Option[org.geotools.api.filter.Filter],
38     visFilter: SimpleFeature => Boolean,
39     transform: Option[(String, SimpleFeatureType)]
40   ) extends FileSystemPathReader with LazyLogging {
41 
42   import scala.collection.JavaConverters._
43 
44   private val gtf = gtFilter.orNull
45 
46   private val transformFeature: SimpleFeature => SimpleFeature = transform match {
47     case None => null
48     case Some((tdefs, tsft)) =>
49       val definitions = Transforms(sft, tdefs).toArray
50       f => new TransformSimpleFeature(tsft, definitions, f)
51   }
52 
53   private val conf = new PlainParquetConfiguration(context.conf.asJava)
54   SimpleFeatureParquetSchema.setSft(conf, sft)
55 
56   override def root: URI = context.root
57 
58   override def read(file: URI): CloseableIterator[SimpleFeature] = {
59     // TODO we can examine file bounds and simplify/eliminate the gtFilter
60     logger.debug(s"Opening reader for path $file")
61     new ParquetFileIterator(file)
62   }
63 
64   private class ParquetFileIterator(path: URI) extends CloseableIterator[SimpleFeature] {
65 
66     private val reader: ParquetReader[SimpleFeature] =
67       ParquetFileSystemReader.builder(fs, path).withConf(conf).withFilter(parquetFilter).build()
68 
69     private var staged: SimpleFeature = _
70 
71     override def close(): Unit = {
72       logger.debug(s"Closing reader for path $path")
73       reader.close()
74     }
75 
76     override def next(): SimpleFeature = {
77       val res = staged
78       staged = null
79       res
80     }
81 
82     @tailrec
83     override final def hasNext: Boolean = {
84       if (staged != null) { true } else {
85         val read = try { reader.read() } catch {
86           case NonFatal(e) => logger.error(s"Error reading file '$path'", e); null
87         }
88         if (read == null) {
89           false
90         } else if (visFilter(read) && (gtf == null || gtf.evaluate(read))) {
91           staged = if (transformFeature == null) { read } else { transformFeature(read) }
92           true
93         } else {
94           hasNext
95         }
96       }
97     }
98   }
99 }

101 object ParquetFileSystemReader {
102 
103   /**
104    * Create a new configurable reader
105    *
106    * @param fs object store
107    * @param path file to read
108    * @return
109    */
110   def builder(fs: ObjectStore, path: URI): Builder = new Builder(inputFile(fs, path))
111 
112   /**
113    * Get an input file compatible with the parquet api
114    *
115    * @param fs object store
116    * @param path file path
117    * @return
118    */
119   def inputFile(fs: ObjectStore, path: URI): InputFile = fs match {
120     case _: LocalObjectStore => new LocalInputFile(Path.of(path))
121     case s3: S3ObjectStore   => new S3InputFile(s3, path)
122     case _ => throw new UnsupportedOperationException(s"No file implementation for scheme ${fs.scheme}")
123   }
124 
125   class Builder(file: InputFile) extends ParquetReader.Builder[SimpleFeature](file) {
126     override protected def getReadSupport: ReadSupport[SimpleFeature] = new SimpleFeatureReadSupport()
127   }
