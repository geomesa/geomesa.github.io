22 /**
23  * Abstraction around object storage for data files
24  */
25 trait ObjectStore extends Closeable {
26 
27   /**
28    * Default URI scheme
29    *
30    * @return
31    */
32   def scheme: String
33 
34   /**
35    * Does an object exist at the given path
36    *
37    * @param path file path
38    * @return
39    */
40   def exists(path: URI): Boolean
41 
42   /**
43    * Get the size of the object stored at this path, in bytes. If the file does not exist, 0L will be returned
44    *
45    * @param path file path
46    * @return size in bytes
47    */
48   def size(path: URI): Long
49 
50   /**
51    * Get the last modified date of the object stored at this path, as millis since the unix epoch. If the file does not exist,
52    * an empty option will be returned
53    *
54    * @param path file path
55    * @return last modified date in millis since epoch
56    */
57   def modified(path: URI): Option[Long]

59   /**
60    * Write to the given path. If a file already exists at the path, an empty option will be returned. Note that this

62    *
63    * @param path file path
64    * @return output stream for writing to the file
65    */
66   def create(path: URI): Option[OutputStream]
67 
68   /**
69    * Write to the given path. If a file already exists at the path, it will be overwritten
70    *
71    * @param path file path
72    * @return output stream for writing to the file
73    */
74   def overwrite(path: URI): OutputStream
75 
76   /**
77    * Reads the file at the given path. If the file does not exist, an empty option will be returned
78    *
79    * @param path file path
80    * @return input stream for reading to the file
81    */
82   def read(path: URI): Option[InputStream]
83 
84   /**
85    * Reads an archive file (tar, zip or jar) into its constituent files

87    * @param path file path

89    */
90   def read(path: URI, format: ArchiveFormat): CloseableIterator[NamedInputStream]
91 
92   /**
93    * List any files directly under the given directory path

95    * @param path directory path

97    */
98   def list(path: URI): CloseableIterator[URI]

100   /**
101    * Copy a file
102    *
103    * @param from from
104    * @param to to
105    */
106   def copy(from: URI, to: URI): Unit
107 
108   /**
109    * Delete a file
110    *
111    * @param path file path
112    */
113   def delete(path: URI): Unit
114 
115   /**
116    * Utility method to return the "name" of a file
117    *
118    * @param path path
119    * @return
120    */
121   def filename(path: URI): String = filename(path.toString)
122 
123   @tailrec
124   private def filename(path: String): String = {
125     val i = path.lastIndexOf('/')
126     if (i == -1) {
127       path
128     } else if (i < path.length - 1) {
129       path.substring(i + 1)
130     } else {
131       filename(path.substring(0, path.length - 1))
132     }
133   }
134 }

138   def apply(scheme: String, conf: Map[String, String]): ObjectStore = {
139     scheme.toLowerCase(Locale.US) match {
140       case "file" => LocalObjectStore
141       case "s3" | "s3a" => S3ObjectStore(conf)
142       case _ => throw new UnsupportedOperationException(s"No object store implemented for scheme: $scheme")
143     }
144   }
145 
146   object ArchiveFormat extends Enumeration {
147 
148     type ArchiveFormat = Value
149     val Tar, Zip = Value
150 
151     /**
152      * Gets the archive format of a file, if any.
153      *
154      * Currently, this is just based on the file name
155      *
156      * @param path file path
157      * @return
158      */
159     def apply(path: URI): Option[ArchiveFormat] = {
160       PathUtils.getUncompressedExtension(path.toString).toLowerCase(Locale.US) match {
161         case TAR => Some(ArchiveFormat.Tar)
162         case ZIP | JAR => Some(ArchiveFormat.Zip)
163         case _ => None
164       }
165     }
166   }
