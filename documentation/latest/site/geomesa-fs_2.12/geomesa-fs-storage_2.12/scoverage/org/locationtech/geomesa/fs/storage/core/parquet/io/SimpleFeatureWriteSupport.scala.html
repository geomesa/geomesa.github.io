47   // called once
48   override def init(conf: Configuration): WriteContext = init(new HadoopParquetConfiguration(conf))

50   override def init(conf: ParquetConfiguration): WriteContext = {
51     val schema = SimpleFeatureParquetSchema.apply(conf).getOrElse {
52       throw new IllegalArgumentException("Could not extract SimpleFeatureType from write context")
53     }
54     init(schema)

57   private def init(schema: SimpleFeatureParquetSchema): WriteContext = {
58     this.writer = new SimpleFeatureWriteSupport.SimpleFeatureWriter(schema)
59     this.geoParquetObserver = new GeoParquetObserver(schema)
60     this.baseMetadata = schema.metadata
61     new WriteContext(schema.messageType, schema.metadata)
62   }
63 
64   // called per block
65   override def prepareForWrite(recordConsumer: RecordConsumer): Unit = consumer = recordConsumer
66 
67   // called per row
68   override def write(record: SimpleFeature): Unit = {
69     writer.write(consumer, record)
70     geoParquetObserver(record)
71   }
72 
73   // called once at the end
74   override def finalizeWrite(): FinalizedWriteContext = {
75     try {
76       val metadata = new java.util.HashMap[String, String]()
77       metadata.putAll(baseMetadata)
78       metadata.putAll(geoParquetObserver.metadata())
79       new FinalizedWriteContext(metadata)
80     } finally {
81       CloseWithLogging(geoParquetObserver)
82     }
83   }

89 
90     private val fids = new FidWriter(0) // ID is the 1st field
91     private val vis = new VisibilityWriter(1) // vis is 2nd field
92 
93     private val attributes = {
94       var index = 2
95       Array.tabulate(schema.sft.getAttributeCount) { i =>
96         val writer = attribute(schema.sft.getDescriptor(i), index).asInstanceOf[AttributeWriter[AnyRef]]
97         index += writer.numFields
98         writer
99       }
100     }
101 
102     def write(consumer: RecordConsumer, value: SimpleFeature): Unit = {
103       consumer.startMessage()
104       fids.apply(consumer, value.getID)
105       if (vis != null) {
106         vis.apply(consumer, value.getUserData.get("geomesa.feature.visibility").asInstanceOf[String])
107       }
108       var i = 0
109       while (i < attributes.length) {
110         attributes(i).apply(consumer, value.getAttribute(i))
111         i += 1
112       }
113       consumer.endMessage()
114     }

116     private def attribute(descriptor: AttributeDescriptor, index: Int): AttributeWriter[_] = {
117       val bindings = ObjectType.selectType(descriptor)
118       val col = ColumnName(descriptor.getLocalName)
119       if (bindings.head == ObjectType.GEOMETRY) {
120         geometry(col, index, bindings.last)
121       } else if (bindings.last == ObjectType.JSON) {
122         val groupType = schema.messageType.getType(index).asGroupType()
123         if (descriptor.getJsonSchema().isDefined) {
124           new StructuralJsonWriter(col.column, index, groupType)
125         } else {
126           new VariantWriter(col.column, index, groupType)
127         }
128       } else {
129         attribute(col.column, index, bindings)
130       }
131     }
132 
133     private def attribute(name: String, index: Int, bindings: Seq[ObjectType]): AttributeWriter[_] = {
134       bindings.head match {
135         case ObjectType.DATE     => new DateMicrosWriter(name, index)
136         case ObjectType.STRING   => new StringWriter(name, index)
137         case ObjectType.INT      => new IntegerWriter(name, index)
138         case ObjectType.LONG     => new LongWriter(name, index)
139         case ObjectType.FLOAT    => new FloatWriter(name, index)
140         case ObjectType.DOUBLE   => new DoubleWriter(name, index)
141         case ObjectType.BYTES    => new BytesWriter(name, index)
142         case ObjectType.LIST     => new ListWriter(name, index, attribute("element", 0, bindings.drop(1)))
143         case ObjectType.MAP      => new MapWriter(name, index, attribute("key", 0, bindings.slice(1, 2)), attribute("value", 1, bindings.slice(2, 3)))
144         case ObjectType.BOOLEAN  => new BooleanWriter(name, index)
145         case ObjectType.UUID     => new UuidWriter(name, index)
146         case _ => throw new IllegalArgumentException(s"Can't serialize field '$name' of type ${bindings.head}")
147       }
148     }
149 
150     // TODO support z/m
151     private def geometry(col: ColumnName, index: Int, binding: ObjectType): AttributeWriter[_] = {
152       if (schema.geometries == GeometryEncoding.GeoParquetWkb) {
153         if (binding == ObjectType.POINT) {
154           new WkbPointWriter(col.column, index)
155         } else {
156           new WkbWriter(col.column, index)
157         }
158       } else {
159         binding match {
160           case ObjectType.POINT               => new PointWriter(col.column, index)
161           case ObjectType.LINESTRING          => new NativeLineStringWriter(col.column, index)
162           case ObjectType.POLYGON             => new NativePolygonWriter(col.column, index)
163           case ObjectType.MULTIPOINT          => new NativeMultiPointWriter(col.column, index)
164           case ObjectType.MULTILINESTRING     => new NativeMultiLineStringWriter(col.column, index)
165           case ObjectType.MULTIPOLYGON        => new NativeMultiPolygonWriter(col.column, index)
166           case ObjectType.GEOMETRY_COLLECTION => new WkbWriter(col.column, index)
167           case ObjectType.GEOMETRY            => new WkbWriter(col.column, index)
168           case _ => throw new IllegalArgumentException(s"Can't serialize field '${col.attribute}' of type $binding")
169         }
170       }
171     }
172   }

242    * Writes a JSON string structurally, according to the parquet schema generated from the attribute's avro schema.
243    * The schema uses standard 3-level lists and string-keyed maps (see SimpleFeatureParquetSchema.buildStructuralType).

245   private class StructuralJsonWriter(name: String, index: Int, schema: GroupType)
246       extends AttributeWriter[String](name, index) {
247 
248     override def apply(consumer: RecordConsumer, value: String): Unit = {
249       if (value != null) {
250         val element = JsonParser.parseString(value)
251         // need to guard against json null here before calling startField
252         if (!element.isJsonNull) {
253           consumer.startField(name, index)
254           StructuralJsonWriter.writeValue(consumer, schema, element)
255           consumer.endField(name, index)
256         }
257       }
258     }

261     override def writeFields(consumer: RecordConsumer, value: String): Unit = {
262       val element = JsonParser.parseString(value)
263       if (!element.isJsonNull) {

269   private object StructuralJsonWriter {
270 
271     private def writeValue(consumer: RecordConsumer, tpe: Type, element: JsonElement): Unit = {
272       if (tpe.isPrimitive) {
273         writePrimitive(consumer, tpe.asPrimitiveType(), element.getAsJsonPrimitive)
274       } else {
275         val group = tpe.asGroupType()
276         group.getLogicalTypeAnnotation match {
277           case _: ListLogicalTypeAnnotation => writeList(consumer, group, element.getAsJsonArray)
278           case _: MapLogicalTypeAnnotation  => writeMap(consumer, group, element.getAsJsonObject)
279           case _                            => writeRecord(consumer, group, element.getAsJsonObject)
280         }
281       }
282     }
283 
284     private def writeRecord(consumer: RecordConsumer, group: GroupType, obj: JsonObject): Unit = {
285       consumer.startGroup()
286       var i = 0
287       val fields = group.getFields
288       while (i < fields.size()) {
289         val field = fields.get(i)
290         val fieldName = field.getName
291         val value = obj.get(fieldName)
292         if (value == null || value.isJsonNull) {
293           if (field.isRepetition(Type.Repetition.REQUIRED)) {
294             throw new IllegalArgumentException(s"JSON is missing required field '$fieldName'")
295           }
296         } else {
297           consumer.startField(fieldName, i)
298           writeValue(consumer, field, value)
299           consumer.endField(fieldName, i)
300         }
301         i += 1
302       }
303       consumer.endGroup()
304     }
305 
306     // standard 3-level list: <name> (LIST) { repeated group list { <element>; } }
307     private def writeList(consumer: RecordConsumer, group: GroupType, array: JsonArray): Unit = {
308       consumer.startGroup()
309       if (array.size() > 0) {
310         val list = group.getType(0).asGroupType() // repeated 'list' group
311         val listName = group.getFieldName(0)
312         val element = list.getType(0)
313         val elementName = list.getFieldName(0)
314         consumer.startField(listName, 0)
315         val iter = array.iterator()
316         while (iter.hasNext) {
317           val item = iter.next()
318           consumer.startGroup()
319           if (!item.isJsonNull) {
320             consumer.startField(elementName, 0)
321             writeValue(consumer, element, item)
322             consumer.endField(elementName, 0)
323           }
324           consumer.endGroup()
325         }
326         consumer.endField(listName, 0)
327       }
328       consumer.endGroup()
329     }
330 
331     // string-keyed map: <name> (MAP) { repeated group key_value { required key; value; } }
332     private def writeMap(consumer: RecordConsumer, group: GroupType, obj: JsonObject): Unit = {
333       consumer.startGroup()
334       if (obj.size() > 0) {
335         val keyValue = group.getType(0).asGroupType()
336         val keyValueName = group.getFieldName(0)
337         val keyType = keyValue.getType(0)
338         val valueType = keyValue.getType(1)
339         consumer.startField(keyValueName, 0)
340         val iter = obj.entrySet().iterator()
341         while (iter.hasNext) {
342           val entry = iter.next()
343           consumer.startGroup()
344           consumer.startField("key", 0)
345           writeValue(consumer, keyType, new JsonPrimitive(entry.getKey))
346           consumer.endField("key", 0)
347           val v = entry.getValue
348           if (v != null && !v.isJsonNull) {
349             consumer.startField("value", 1)
350             writeValue(consumer, valueType, v)
351             consumer.endField("value", 1)
352           }
353           consumer.endGroup()
354         }
355         consumer.endField(keyValueName, 0)
356       }
357       consumer.endGroup()
358     }
359 
360     private def writePrimitive(consumer: RecordConsumer, tpe: PrimitiveType, value: JsonPrimitive): Unit = {
361       val logical = Option(tpe.getLogicalTypeAnnotation)
362       tpe.getPrimitiveTypeName match {
363         case PrimitiveTypeName.BOOLEAN => consumer.addBoolean(value.getAsBoolean)
364         case PrimitiveTypeName.FLOAT   => consumer.addFloat(value.getAsFloat)
365         case PrimitiveTypeName.DOUBLE  => consumer.addDouble(value.getAsDouble)
366 
367         case PrimitiveTypeName.INT32 =>
368           logical match {
369             case Some(_: DateLogicalTypeAnnotation) =>
370               consumer.addInteger(StructuralJson.jsonToEpochDay(value.getAsString))
371             case Some(t: TimeLogicalTypeAnnotation) =>
372               consumer.addInteger(StructuralJson.jsonToTime(value.getAsString, t.getUnit).toInt)
373             case Some(d: DecimalLogicalTypeAnnotation) =>
374               consumer.addInteger(value.getAsBigDecimal.setScale(d.getScale).unscaledValue().intValueExact())
375             case _ =>
376               consumer.addInteger(value.getAsInt)
377           }
378 
379         case PrimitiveTypeName.INT64 =>
380           logical match {
381             case Some(t: TimestampLogicalTypeAnnotation) =>
382               consumer.addLong(StructuralJson.jsonToTimestamp(value.getAsString, t.getUnit, t.isAdjustedToUTC))
383             case Some(t: TimeLogicalTypeAnnotation) =>
384               consumer.addLong(StructuralJson.jsonToTime(value.getAsString, t.getUnit))
385             case Some(d: DecimalLogicalTypeAnnotation) =>
386               consumer.addLong(value.getAsBigDecimal.setScale(d.getScale).unscaledValue().longValueExact())
387             case _ =>
388               consumer.addLong(value.getAsLong)
389           }
390 
391         case PrimitiveTypeName.BINARY =>
392           logical match {
393             case Some(_: StringLogicalTypeAnnotation) =>
394               consumer.addBinary(Binary.fromString(value.getAsString))
395             case Some(_: EnumLogicalTypeAnnotation) =>
396               consumer.addBinary(Binary.fromString(value.getAsString))
397             case _ =>
398               consumer.addBinary(Binary.fromConstantByteArray(StructuralJson.jsonToBytes(value.getAsString)))
399           }
400 
401         case PrimitiveTypeName.FIXED_LEN_BYTE_ARRAY =>
402           logical match {
403             case Some(_: UUIDLogicalTypeAnnotation) =>
404               consumer.addBinary(Binary.fromConstantByteArray(StructuralJson.uuidToBytes(UUID.fromString(value.getAsString))))
405             case Some(d: DecimalLogicalTypeAnnotation) =>
406               val bytes = StructuralJson.decimalToFixedBytes(value.getAsBigDecimal, d.getScale, tpe.getTypeLength)
407               consumer.addBinary(Binary.fromConstantByteArray(bytes))
408             case _ =>
409               consumer.addBinary(Binary.fromConstantByteArray(StructuralJson.jsonToBytes(value.getAsString)))
410           }
411 
412         case PrimitiveTypeName.INT96 =>
413           consumer.addBinary(Binary.fromConstantByteArray(StructuralJson.jsonToBytes(value.getAsString)))
414       }
415     }
416   }

488   private abstract class GeometryWriter[T <: Geometry](name: String, index: Int) extends AttributeWriter[T](name, index, 3) {
489 
490     private val bboxCol = BoundingBoxField.groupName(name)
491 
492     protected def zCol: String
493     protected def z(geom: T): String
494 
495     /**
496      * Writes a value to the current record
497      *
498      * @param consumer the Parquet record consumer
499      * @param value value to write
500      */
501     override def apply(consumer: RecordConsumer, value: T): Unit = {
502       if (value != null) {
503         consumer.startField(name, index)
504         writeFields(consumer, value)
505         consumer.endField(name, index)
506         writeBbox(consumer, value)
507         writeZVal(consumer, value)
508       }
509     }
510 
511     private def writeBbox(consumer: RecordConsumer, value: T): Unit = {
512       val bbox = value.getEnvelopeInternal
513       consumer.startField(bboxCol, index + 1)
514       consumer.startGroup()
515       consumer.startField(BoundingBoxField.XMin, 0)
516       consumer.addFloat(bbox.getMinX.toFloat)
517       consumer.endField(BoundingBoxField.XMin, 0)
518       consumer.startField(BoundingBoxField.YMin, 1)
519       consumer.addFloat(bbox.getMinY.toFloat)
520       consumer.endField(BoundingBoxField.YMin, 1)
521       consumer.startField(BoundingBoxField.XMax, 2)
522       consumer.addFloat(bbox.getMaxX.toFloat)
523       consumer.endField(BoundingBoxField.XMax, 2)
524       consumer.startField(BoundingBoxField.YMax, 3)
525       consumer.addFloat(bbox.getMaxY.toFloat)
526       consumer.endField(BoundingBoxField.YMax, 3)
527       consumer.endGroup()
528       consumer.endField(bboxCol, index + 1)
529     }
530 
531     private def writeZVal(consumer: RecordConsumer, value: T): Unit = {
532       consumer.startField(zCol, index + 2)
533       consumer.addBinary(Binary.fromString(z(value)))
534       consumer.endField(zCol, index + 2)
535     }
536   }
