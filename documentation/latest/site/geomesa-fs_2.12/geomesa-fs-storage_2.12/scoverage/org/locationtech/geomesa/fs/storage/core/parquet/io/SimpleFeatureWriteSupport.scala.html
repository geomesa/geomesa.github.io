82   private class SimpleFeatureWriter(schema: SimpleFeatureParquetSchema) {
83 
84     private val fids = new FidWriter(0) // ID is the 1st field
85     private val vis = new VisibilityWriter(1) // vis is 2nd field
86 
87     private val attributes = {
88       var index = 2
89       Array.tabulate(schema.sft.getAttributeCount) { i =>
90         val writer = attribute(schema.sft.getDescriptor(i), index).asInstanceOf[AttributeWriter[AnyRef]]
91         index += writer.numFields
92         writer
93       }
94     }
95 
96     def write(consumer: RecordConsumer, value: SimpleFeature): Unit = {
97       consumer.startMessage()
98       fids.apply(consumer, value.getID)
99       if (vis != null) {
100         vis.apply(consumer, value.getUserData.get("geomesa.feature.visibility").asInstanceOf[String])
101       }
102       var i = 0
103       while (i < attributes.length) {
104         attributes(i).apply(consumer, value.getAttribute(i))
105         i += 1
106       }
107       consumer.endMessage()
108     }
109 
110     private def attribute(descriptor: AttributeDescriptor, index: Int): AttributeWriter[_] =
111       attribute(descriptor.getLocalName, index, ObjectType.selectType(descriptor))
112 
113     private def attribute(name: String, index: Int, bindings: Seq[ObjectType]): AttributeWriter[_] = {
114       val col = ColumnName(name)
115       bindings.head match {
116         case ObjectType.GEOMETRY => geometry(col, index, bindings.last)
117         case ObjectType.DATE     => new DateMicrosWriter(col.column, index)
118         case ObjectType.STRING   => new StringWriter(col.column, index)
119         case ObjectType.INT      => new IntegerWriter(col.column, index)
120         case ObjectType.LONG     => new LongWriter(col.column, index)
121         case ObjectType.FLOAT    => new FloatWriter(col.column, index)
122         case ObjectType.DOUBLE   => new DoubleWriter(col.column, index)
123         case ObjectType.BYTES    => new BytesWriter(col.column, index)
124         case ObjectType.LIST     => new ListWriter(col.column, index, attribute("element", 0, bindings.drop(1)))
125         case ObjectType.MAP      => new MapWriter(col.column, index, attribute("key", 0, bindings.slice(1, 2)), attribute("value", 1, bindings.slice(2, 3)))
126         case ObjectType.BOOLEAN  => new BooleanWriter(col.column, index)
127         case ObjectType.UUID     => new UuidWriter(col.column, index)
128         case _ => throw new IllegalArgumentException(s"Can't serialize field '$name' of type ${bindings.head}")
129       }
130     }
131 
132     // TODO support z/m
133     private def geometry(col: ColumnName, index: Int, binding: ObjectType): AttributeWriter[_] = {
134       if (schema.geometries == GeometryEncoding.GeoParquetWkb) {
135         if (binding == ObjectType.POINT) {
136           new WkbPointWriter(col.column, index)
137         } else {
138           new WkbWriter(col.column, index)
139         }
140       } else {
141         binding match {
142           case ObjectType.POINT               => new PointWriter(col.column, index)
143           case ObjectType.LINESTRING          => new NativeLineStringWriter(col.column, index)
144           case ObjectType.POLYGON             => new NativePolygonWriter(col.column, index)
145           case ObjectType.MULTIPOINT          => new NativeMultiPointWriter(col.column, index)
146           case ObjectType.MULTILINESTRING     => new NativeMultiLineStringWriter(col.column, index)
147           case ObjectType.MULTIPOLYGON        => new NativeMultiPolygonWriter(col.column, index)
148           case ObjectType.GEOMETRY_COLLECTION => new WkbWriter(col.column, index)
149           case ObjectType.GEOMETRY            => new WkbWriter(col.column, index)
150           case _ => throw new IllegalArgumentException(s"Can't serialize field '${col.attribute}' of type $binding")
151         }
152       }
153     }
154   }

259         while (iter.hasNext) {

261           consumer.startGroup()

288   private abstract class GeometryWriter[T <: Geometry](name: String, index: Int) extends AttributeWriter[T](name, index, 3) {
289 
290     private val bboxCol = BoundingBoxField.groupName(name)
291 
292     protected def zCol: String
293     protected def z(geom: T): String
294 
295     /**
296      * Writes a value to the current record
297      *
298      * @param consumer the Parquet record consumer
299      * @param value value to write
300      */
301     override def apply(consumer: RecordConsumer, value: T): Unit = {
302       if (value != null) {
303         consumer.startField(name, index)
304         writeFields(consumer, value)
305         consumer.endField(name, index)
306         writeBbox(consumer, value)
307         writeZVal(consumer, value)
308       }
309     }
310 
311     private def writeBbox(consumer: RecordConsumer, value: T): Unit = {
312       val bbox = value.getEnvelopeInternal
313       consumer.startField(bboxCol, index + 1)
314       consumer.startGroup()
315       consumer.startField(BoundingBoxField.XMin, 0)
316       consumer.addFloat(bbox.getMinX.toFloat)
317       consumer.endField(BoundingBoxField.XMin, 0)
318       consumer.startField(BoundingBoxField.YMin, 1)
319       consumer.addFloat(bbox.getMinY.toFloat)
320       consumer.endField(BoundingBoxField.YMin, 1)
321       consumer.startField(BoundingBoxField.XMax, 2)
322       consumer.addFloat(bbox.getMaxX.toFloat)
323       consumer.endField(BoundingBoxField.XMax, 2)
324       consumer.startField(BoundingBoxField.YMax, 3)
325       consumer.addFloat(bbox.getMaxY.toFloat)
326       consumer.endField(BoundingBoxField.YMax, 3)
327       consumer.endGroup()
328       consumer.endField(bboxCol, index + 1)
329     }
330 
331     private def writeZVal(consumer: RecordConsumer, value: T): Unit = {
332       consumer.startField(zCol, index + 2)
333       consumer.addBinary(Binary.fromString(z(value)))
334       consumer.endField(zCol, index + 2)
335     }
