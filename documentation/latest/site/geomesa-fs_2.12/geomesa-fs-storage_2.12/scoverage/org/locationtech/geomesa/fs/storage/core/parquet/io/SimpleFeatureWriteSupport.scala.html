40 class SimpleFeatureWriteSupport extends WriteSupport[SimpleFeature] {
41 
42   private var writer: SimpleFeatureWriteSupport.SimpleFeatureWriter = _
43   private var consumer: RecordConsumer = _
44   private var geoParquetObserver: GeoParquetObserver = _
45   private var baseMetadata: java.util.Map[String, String] = _
46   private var visibilityMetadata: VisibilityMetadata = _
47 
48   override val getName: String = "SimpleFeatureWriteSupport"
49 
50   // called once
51   override def init(conf: Configuration): WriteContext = init(new HadoopParquetConfiguration(conf))
52 
53   override def init(conf: ParquetConfiguration): WriteContext = {
54     val schema = SimpleFeatureParquetSchema.apply(conf).getOrElse {
55       throw new IllegalArgumentException("Could not extract SimpleFeatureType from write context")
56     }
57     init(schema)
58   }
59 
60   private def init(schema: SimpleFeatureParquetSchema): WriteContext = {
61     this.writer = new SimpleFeatureWriteSupport.SimpleFeatureWriter(schema)
62     this.geoParquetObserver = new GeoParquetObserver(schema)
63     this.baseMetadata = new java.util.HashMap[String, String](schema.metadata)
64     this.baseMetadata.remove(VisibilityMetadata.MetadataKey)
65     val visibility = schema.messageType.getFields.get(schema.messageType.getFieldIndex(SimpleFeatureSchema.VisibilitiesField))
66     this.visibilityMetadata = new VisibilityMetadata(Option(visibility.getId).map(_.intValue()).getOrElse(-1))
67     new WriteContext(schema.messageType, baseMetadata)
68   }
69 
70   // called per block
71   override def prepareForWrite(recordConsumer: RecordConsumer): Unit = consumer = recordConsumer
72 
73   // called per row
74   override def write(record: SimpleFeature): Unit = {
75     writer.write(consumer, record)
76     visibilityMetadata.observe(record.getUserData.get("geomesa.feature.visibility").asInstanceOf[String])
77     geoParquetObserver(record)
78   }
79 
80   // called once at the end
81   override def finalizeWrite(): FinalizedWriteContext = {
82     try {
83       val metadata = new java.util.HashMap[String, String]()
84       metadata.putAll(baseMetadata)
85       metadata.putAll(geoParquetObserver.metadata())
86       // Never retain a summary supplied in base metadata from an earlier file.
87       metadata.remove(VisibilityMetadata.MetadataKey)
88       visibilityMetadata.metadata.foreach(metadata.put(VisibilityMetadata.MetadataKey, _))
89       new FinalizedWriteContext(metadata)
90     } finally {
91       CloseWithLogging(geoParquetObserver)
92     }
93   }
94 }

98   private class SimpleFeatureWriter(schema: SimpleFeatureParquetSchema) {
99 
100     private val fids = new FidWriter(0) // ID is the 1st field
101     private val vis = new VisibilityWriter(1) // vis is 2nd field
102 
103     private val attributes = {
104       var index = 2
105       Array.tabulate(schema.sft.getAttributeCount) { i =>
106         val writer = attribute(schema.sft.getDescriptor(i), index).asInstanceOf[AttributeWriter[AnyRef]]
107         index += writer.numFields
108         writer
109       }
110     }
111 
112     def write(consumer: RecordConsumer, value: SimpleFeature): Unit = {
113       consumer.startMessage()
114       fids.apply(consumer, value.getID)
115       if (vis != null) {
116         vis.apply(consumer, value.getUserData.get("geomesa.feature.visibility").asInstanceOf[String])
117       }
118       var i = 0
119       while (i < attributes.length) {
120         attributes(i).apply(consumer, value.getAttribute(i))
121         i += 1
122       }
123       consumer.endMessage()
124     }
125 
126     private def attribute(descriptor: AttributeDescriptor, index: Int): AttributeWriter[_] = {
127       val bindings = ObjectType.selectType(descriptor)
128       val col = ColumnName(descriptor.getLocalName)
129       if (bindings.head == ObjectType.GEOMETRY) {
130         geometry(col, index, bindings.last, descriptor.getEncoding.fold[GeometryEncoding](GeoParquetWkb)(GeometryEncoding.apply))
131       } else if (bindings.last == ObjectType.JSON) {
132         val groupType = schema.messageType.getType(index).asGroupType()
133         if (descriptor.getJsonSchema().isDefined) {
134           new StructuralJsonWriter(col.column, index, groupType)
135         } else {
136           new VariantWriter(col.column, index, groupType)
137         }
138       } else {
139         attribute(col.column, index, bindings)
140       }
141     }
142 
143     private def attribute(name: String, index: Int, bindings: Seq[ObjectType]): AttributeWriter[_] = {
144       bindings.head match {
145         case ObjectType.DATE     => new DateMicrosWriter(name, index)
146         case ObjectType.STRING   => new StringWriter(name, index)
147         case ObjectType.INT      => new IntegerWriter(name, index)
148         case ObjectType.LONG     => new LongWriter(name, index)
149         case ObjectType.FLOAT    => new FloatWriter(name, index)
150         case ObjectType.DOUBLE   => new DoubleWriter(name, index)
151         case ObjectType.BYTES    => new BytesWriter(name, index)
152         case ObjectType.LIST     => new ListWriter(name, index, attribute("element", 0, bindings.drop(1)))
153         case ObjectType.MAP      => new MapWriter(name, index, attribute("key", 0, bindings.slice(1, 2)), attribute("value", 1, bindings.slice(2, 3)))
154         case ObjectType.BOOLEAN  => new BooleanWriter(name, index)
155         case ObjectType.UUID     => new UuidWriter(name, index)
156         case _ => throw new IllegalArgumentException(s"Can't serialize field '$name' of type ${bindings.head}")
157       }
158     }
159 
160     // TODO support z/m
161     private def geometry(col: ColumnName, index: Int, binding: ObjectType, encoding: GeometryEncoding): AttributeWriter[_] = {
162       if (encoding == GeometryEncoding.GeoParquetWkb) {
163         if (binding == ObjectType.POINT) {
164           new WkbPointWriter(col.column, index)
165         } else {
166           new WkbWriter(col.column, index)
167         }
168       } else if (encoding == GeometryEncoding.GeoParquetNative) {
169         binding match {
170           case ObjectType.POINT               => new PointWriter(col.column, index)
171           case ObjectType.LINESTRING          => new NativeLineStringWriter(col.column, index)
172           case ObjectType.POLYGON             => new NativePolygonWriter(col.column, index)
173           case ObjectType.MULTIPOINT          => new NativeMultiPointWriter(col.column, index)
174           case ObjectType.MULTILINESTRING     => new NativeMultiLineStringWriter(col.column, index)
175           case ObjectType.MULTIPOLYGON        => new NativeMultiPolygonWriter(col.column, index)
176           case ObjectType.GEOMETRY_COLLECTION => new WkbWriter(col.column, index)
177           case ObjectType.GEOMETRY            => new WkbWriter(col.column, index)
178           case _ => throw new IllegalArgumentException(s"Can't serialize field '${col.attribute}' of type $binding")
179         }
180       } else {
181         throw new UnsupportedOperationException(s"Can't write geometries encoded with '$encoding'")
182       }
183     }
184   }

253   /**
254    * Writes a JSON string structurally, according to the parquet schema generated from the attribute's avro schema.
255    * The schema uses standard 3-level lists and string-keyed maps (see SimpleFeatureParquetSchema.buildStructuralType).
256    */
257   private class StructuralJsonWriter(name: String, index: Int, schema: GroupType)
258       extends AttributeWriter[String](name, index) {
259 
260     override def apply(consumer: RecordConsumer, value: String): Unit = {
261       if (value != null) {
262         val element = JsonParser.parseString(value)
263         // need to guard against json null here before calling startField
264         if (!element.isJsonNull) {
265           consumer.startField(name, index)
266           StructuralJsonWriter.writeValue(consumer, schema, element)
267           consumer.endField(name, index)
268         }
269       }
270     }
271 
272     // note: not used due to override of apply, but included for completeness
273     override def writeFields(consumer: RecordConsumer, value: String): Unit = {
274       val element = JsonParser.parseString(value)
275       if (!element.isJsonNull) {
276         StructuralJsonWriter.writeValue(consumer, schema, element)
277       }
278     }
279   }

281   private object StructuralJsonWriter {
282 
283     private def writeValue(consumer: RecordConsumer, tpe: Type, element: JsonElement): Unit = {
284       if (tpe.isPrimitive) {
285         writePrimitive(consumer, tpe.asPrimitiveType(), element.getAsJsonPrimitive)
286       } else {
287         val group = tpe.asGroupType()
288         group.getLogicalTypeAnnotation match {
289           case _: ListLogicalTypeAnnotation => writeList(consumer, group, element.getAsJsonArray)
290           case _: MapLogicalTypeAnnotation  => writeMap(consumer, group, element.getAsJsonObject)
291           case _                            => writeRecord(consumer, group, element.getAsJsonObject)
292         }
293       }
294     }
295 
296     private def writeRecord(consumer: RecordConsumer, group: GroupType, obj: JsonObject): Unit = {
297       consumer.startGroup()
298       var i = 0
299       val fields = group.getFields
300       while (i < fields.size()) {
301         val field = fields.get(i)
302         val fieldName = field.getName
303         val value = obj.get(fieldName)
304         if (value == null || value.isJsonNull) {
305           if (field.isRepetition(Type.Repetition.REQUIRED)) {
306             throw new IllegalArgumentException(s"JSON is missing required field '$fieldName'")
307           }
308         } else {
309           consumer.startField(fieldName, i)
310           writeValue(consumer, field, value)
311           consumer.endField(fieldName, i)
312         }
313         i += 1
314       }
315       consumer.endGroup()
316     }
317 
318     // standard 3-level list: <name> (LIST) { repeated group list { <element>; } }
319     private def writeList(consumer: RecordConsumer, group: GroupType, array: JsonArray): Unit = {
320       consumer.startGroup()
321       if (array.size() > 0) {
322         val list = group.getType(0).asGroupType() // repeated 'list' group
323         val listName = group.getFieldName(0)
324         val element = list.getType(0)
325         val elementName = list.getFieldName(0)
326         consumer.startField(listName, 0)
327         val iter = array.iterator()
328         while (iter.hasNext) {
329           val item = iter.next()
330           consumer.startGroup()
331           if (!item.isJsonNull) {
332             consumer.startField(elementName, 0)
333             writeValue(consumer, element, item)
334             consumer.endField(elementName, 0)
335           }
336           consumer.endGroup()
337         }
338         consumer.endField(listName, 0)
339       }
340       consumer.endGroup()
341     }
342 
343     // string-keyed map: <name> (MAP) { repeated group key_value { required key; value; } }
344     private def writeMap(consumer: RecordConsumer, group: GroupType, obj: JsonObject): Unit = {
345       consumer.startGroup()
346       if (obj.size() > 0) {
347         val keyValue = group.getType(0).asGroupType()
348         val keyValueName = group.getFieldName(0)
349         val keyType = keyValue.getType(0)
350         val valueType = keyValue.getType(1)
351         consumer.startField(keyValueName, 0)
352         val iter = obj.entrySet().iterator()
353         while (iter.hasNext) {
354           val entry = iter.next()
355           consumer.startGroup()
356           consumer.startField("key", 0)
357           writeValue(consumer, keyType, new JsonPrimitive(entry.getKey))
358           consumer.endField("key", 0)
359           val v = entry.getValue
360           if (v != null && !v.isJsonNull) {
361             consumer.startField("value", 1)
362             writeValue(consumer, valueType, v)
363             consumer.endField("value", 1)
364           }
365           consumer.endGroup()
366         }
367         consumer.endField(keyValueName, 0)
368       }
369       consumer.endGroup()
370     }
371 
372     private def writePrimitive(consumer: RecordConsumer, tpe: PrimitiveType, value: JsonPrimitive): Unit = {
373       val logical = Option(tpe.getLogicalTypeAnnotation)
374       tpe.getPrimitiveTypeName match {
375         case PrimitiveTypeName.BOOLEAN => consumer.addBoolean(value.getAsBoolean)
376         case PrimitiveTypeName.FLOAT   => consumer.addFloat(value.getAsFloat)
377         case PrimitiveTypeName.DOUBLE  => consumer.addDouble(value.getAsDouble)
378 
379         case PrimitiveTypeName.INT32 =>
380           logical match {
381             case Some(_: DateLogicalTypeAnnotation) =>
382               consumer.addInteger(StructuralJson.jsonToEpochDay(value.getAsString))
383             case Some(t: TimeLogicalTypeAnnotation) =>
384               consumer.addInteger(StructuralJson.jsonToTime(value.getAsString, t.getUnit).toInt)
385             case Some(d: DecimalLogicalTypeAnnotation) =>
386               consumer.addInteger(value.getAsBigDecimal.setScale(d.getScale).unscaledValue().intValueExact())
387             case _ =>
388               consumer.addInteger(value.getAsInt)
389           }
390 
391         case PrimitiveTypeName.INT64 =>
392           logical match {
393             case Some(t: TimestampLogicalTypeAnnotation) =>
394               consumer.addLong(StructuralJson.jsonToTimestamp(value.getAsString, t.getUnit, t.isAdjustedToUTC))
395             case Some(t: TimeLogicalTypeAnnotation) =>
396               consumer.addLong(StructuralJson.jsonToTime(value.getAsString, t.getUnit))
397             case Some(d: DecimalLogicalTypeAnnotation) =>
398               consumer.addLong(value.getAsBigDecimal.setScale(d.getScale).unscaledValue().longValueExact())
399             case _ =>
400               consumer.addLong(value.getAsLong)
401           }
402 
403         case PrimitiveTypeName.BINARY =>
404           logical match {
405             case Some(_: StringLogicalTypeAnnotation) =>
406               consumer.addBinary(Binary.fromString(value.getAsString))
407             case Some(_: EnumLogicalTypeAnnotation) =>
408               consumer.addBinary(Binary.fromString(value.getAsString))
409             case _ =>
410               consumer.addBinary(Binary.fromConstantByteArray(StructuralJson.jsonToBytes(value.getAsString)))
411           }
412 
413         case PrimitiveTypeName.FIXED_LEN_BYTE_ARRAY =>
414           logical match {
415             case Some(_: UUIDLogicalTypeAnnotation) =>
416               consumer.addBinary(Binary.fromConstantByteArray(StructuralJson.uuidToBytes(UUID.fromString(value.getAsString))))
417             case Some(d: DecimalLogicalTypeAnnotation) =>
418               val bytes = StructuralJson.decimalToFixedBytes(value.getAsBigDecimal, d.getScale, tpe.getTypeLength)
419               consumer.addBinary(Binary.fromConstantByteArray(bytes))
420             case _ =>
421               consumer.addBinary(Binary.fromConstantByteArray(StructuralJson.jsonToBytes(value.getAsString)))
422           }
423 
424         case PrimitiveTypeName.INT96 =>
425           consumer.addBinary(Binary.fromConstantByteArray(StructuralJson.jsonToBytes(value.getAsString)))
426       }
427     }
428   }

443     override def writeFields(consumer: RecordConsumer, value: java.util.List[T]): Unit = {
444       consumer.startGroup()
445       if (!value.isEmpty) {
446         consumer.startField("list", 0)
447         consumer.startGroup()
448         consumer.startField("element", 0)
449         val iter = value.iterator
450         while (iter.hasNext) {
451           val item = iter.next
452           if (item != null) {
453             elements.writeFields(consumer, item)
454           }
455         }
456         consumer.endField("element", 0)
457         consumer.endGroup()
458         consumer.endField("list", 0)
459       }
460       consumer.endGroup()
461     }

473           consumer.startGroup()
474           keys(consumer, entry.getKey)

477             values(consumer, v)

481         consumer.endField("key_value", 0)

500   private abstract class GeometryWriter[T <: Geometry](name: String, index: Int) extends AttributeWriter[T](name, index, 3) {
501 
502     private val bboxCol = BoundingBoxField.groupName(name)
503 
504     protected def zCol: String
505     protected def z(geom: T): String
506 
507     /**
508      * Writes a value to the current record
509      *
510      * @param consumer the Parquet record consumer
511      * @param value value to write
512      */
513     override def apply(consumer: RecordConsumer, value: T): Unit = {
514       if (value != null) {
515         consumer.startField(name, index)
516         writeFields(consumer, value)
517         consumer.endField(name, index)
518         writeBbox(consumer, value)
519         writeZVal(consumer, value)
520       }
521     }
522 
523     private def writeBbox(consumer: RecordConsumer, value: T): Unit = {
524       val bbox = value.getEnvelopeInternal
525       consumer.startField(bboxCol, index + 1)
526       consumer.startGroup()
527       consumer.startField(BoundingBoxField.XMin, 0)
528       consumer.addFloat(bbox.getMinX.toFloat)
529       consumer.endField(BoundingBoxField.XMin, 0)
530       consumer.startField(BoundingBoxField.YMin, 1)
531       consumer.addFloat(bbox.getMinY.toFloat)
532       consumer.endField(BoundingBoxField.YMin, 1)
533       consumer.startField(BoundingBoxField.XMax, 2)
534       consumer.addFloat(bbox.getMaxX.toFloat)
535       consumer.endField(BoundingBoxField.XMax, 2)
536       consumer.startField(BoundingBoxField.YMax, 3)
537       consumer.addFloat(bbox.getMaxY.toFloat)
538       consumer.endField(BoundingBoxField.YMax, 3)
539       consumer.endGroup()
540       consumer.endField(bboxCol, index + 1)
541     }
542 
543     private def writeZVal(consumer: RecordConsumer, value: T): Unit = {
544       consumer.startField(zCol, index + 2)
545       consumer.addBinary(Binary.fromString(z(value)))
546       consumer.endField(zCol, index + 2)
547     }
548   }
