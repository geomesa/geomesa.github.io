54     this.writer = new SimpleFeatureWriteSupport.SimpleFeatureWriter(schema)

84   private class SimpleFeatureWriter(schema: SimpleFeatureParquetSchema) {
85 
86     private val fids = new FidWriter(0) // ID is the 1st field
87     private val vis = new VisibilityWriter(1) // vis is 2nd field
88 
89     private val attributes = {
90       var index = 2
91       Array.tabulate(schema.sft.getAttributeCount) { i =>
92         val writer = attribute(schema.sft.getDescriptor(i), index).asInstanceOf[AttributeWriter[AnyRef]]
93         index += writer.numFields
94         writer
95       }
96     }
97 
98     def write(consumer: RecordConsumer, value: SimpleFeature): Unit = {
99       consumer.startMessage()
100       fids.apply(consumer, value.getID)
101       if (vis != null) {
102         vis.apply(consumer, value.getUserData.get("geomesa.feature.visibility").asInstanceOf[String])
103       }
104       var i = 0
105       while (i < attributes.length) {
106         attributes(i).apply(consumer, value.getAttribute(i))
107         i += 1
108       }
109       consumer.endMessage()
110     }
111 
112     private def attribute(descriptor: AttributeDescriptor, index: Int): AttributeWriter[_] =
113       attribute(descriptor.getLocalName, index, ObjectType.selectType(descriptor))
114 
115     private def attribute(name: String, index: Int, bindings: Seq[ObjectType]): AttributeWriter[_] = {
116       val col = ColumnName(name)
117       bindings.head match {
118         case ObjectType.GEOMETRY => geometry(col, index, bindings.last)
119         case ObjectType.DATE     => new DateMicrosWriter(col.column, index)
120         case ObjectType.STRING   => string(col, index, bindings.last)
121         case ObjectType.INT      => new IntegerWriter(col.column, index)
122         case ObjectType.LONG     => new LongWriter(col.column, index)
123         case ObjectType.FLOAT    => new FloatWriter(col.column, index)
124         case ObjectType.DOUBLE   => new DoubleWriter(col.column, index)
125         case ObjectType.BYTES    => new BytesWriter(col.column, index)
126         case ObjectType.LIST     => new ListWriter(col.column, index, attribute("element", 0, bindings.drop(1)))
127         case ObjectType.MAP      => new MapWriter(col.column, index, attribute("key", 0, bindings.slice(1, 2)), attribute("value", 1, bindings.slice(2, 3)))
128         case ObjectType.BOOLEAN  => new BooleanWriter(col.column, index)
129         case ObjectType.UUID     => new UuidWriter(col.column, index)
130         case _ => throw new IllegalArgumentException(s"Can't serialize field '$name' of type ${bindings.head}")
131       }
132     }
133 
134     private def string(col: ColumnName, index: Int, binding: ObjectType): AttributeWriter[_] = {
135       if (binding == ObjectType.JSON) {
136         new VariantWriter(col.column, index, schema.messageType.getType(index).asGroupType())
137       } else {
138         new StringWriter(col.column, index)
139       }
140     }
141 
142     // TODO support z/m
143     private def geometry(col: ColumnName, index: Int, binding: ObjectType): AttributeWriter[_] = {
144       if (schema.geometries == GeometryEncoding.GeoParquetWkb) {
145         if (binding == ObjectType.POINT) {
146           new WkbPointWriter(col.column, index)
147         } else {
148           new WkbWriter(col.column, index)
149         }
150       } else {
151         binding match {
152           case ObjectType.POINT               => new PointWriter(col.column, index)
153           case ObjectType.LINESTRING          => new NativeLineStringWriter(col.column, index)
154           case ObjectType.POLYGON             => new NativePolygonWriter(col.column, index)
155           case ObjectType.MULTIPOINT          => new NativeMultiPointWriter(col.column, index)
156           case ObjectType.MULTILINESTRING     => new NativeMultiLineStringWriter(col.column, index)
157           case ObjectType.MULTIPOLYGON        => new NativeMultiPolygonWriter(col.column, index)
158           case ObjectType.GEOMETRY_COLLECTION => new WkbWriter(col.column, index)
159           case ObjectType.GEOMETRY            => new WkbWriter(col.column, index)
160           case _ => throw new IllegalArgumentException(s"Can't serialize field '${col.attribute}' of type $binding")
161         }
162       }
163     }
164   }

303   private abstract class GeometryWriter[T <: Geometry](name: String, index: Int) extends AttributeWriter[T](name, index, 3) {
304 
305     private val bboxCol = BoundingBoxField.groupName(name)
306 
307     protected def zCol: String
308     protected def z(geom: T): String
309 
310     /**
311      * Writes a value to the current record
312      *
313      * @param consumer the Parquet record consumer
314      * @param value value to write
315      */
316     override def apply(consumer: RecordConsumer, value: T): Unit = {
317       if (value != null) {
318         consumer.startField(name, index)
319         writeFields(consumer, value)
320         consumer.endField(name, index)
321         writeBbox(consumer, value)
322         writeZVal(consumer, value)
323       }
324     }
325 
326     private def writeBbox(consumer: RecordConsumer, value: T): Unit = {
327       val bbox = value.getEnvelopeInternal
328       consumer.startField(bboxCol, index + 1)
329       consumer.startGroup()
330       consumer.startField(BoundingBoxField.XMin, 0)
331       consumer.addFloat(bbox.getMinX.toFloat)
332       consumer.endField(BoundingBoxField.XMin, 0)
333       consumer.startField(BoundingBoxField.YMin, 1)
334       consumer.addFloat(bbox.getMinY.toFloat)
335       consumer.endField(BoundingBoxField.YMin, 1)
336       consumer.startField(BoundingBoxField.XMax, 2)
337       consumer.addFloat(bbox.getMaxX.toFloat)
338       consumer.endField(BoundingBoxField.XMax, 2)
339       consumer.startField(BoundingBoxField.YMax, 3)
340       consumer.addFloat(bbox.getMaxY.toFloat)
341       consumer.endField(BoundingBoxField.YMax, 3)
342       consumer.endGroup()
343       consumer.endField(bboxCol, index + 1)
344     }
345 
346     private def writeZVal(consumer: RecordConsumer, value: T): Unit = {
347       consumer.startField(zCol, index + 2)
348       consumer.addBinary(Binary.fromString(z(value)))
349       consumer.endField(zCol, index + 2)
350     }
