26   case class SchemeOpts(name: String, opts: Map[String, String], multiOpts: Map[String, Seq[String]]) {
27     def getSingle(k: String): Option[String] = {
28       if (multiOpts.contains(k)) {
29         throw new IllegalArgumentException(s"Expected a single value for '$k' but got: ${multiOpts(k).mkString(", ")}")
30       }
31       opts.get(k)
32     }
33 
34     def getMulti(k: String): Seq[String] = multiOpts.get(k).orElse(opts.get(k).map(v => Seq(v))).getOrElse(Seq.empty)
35   }

37   object SchemeOpts {
38     def apply(scheme: String): SchemeOpts = {
39       val parts = scheme.split(":")
40       val name = parts.head.toLowerCase(Locale.US)
41       val opts = parts.drop(1).map(_.split("=", 2) match { case Array(k, v) => k.toLowerCase(Locale.US) -> v }).groupBy(_._1)
42       val singleOpts = opts.collect { case (k, Array(v)) => k -> v._2 }
43       val multiOpts = opts.collect { case (k, v) if v.lengthCompare(1) > 0 => k -> v.map(_._2).toSeq }
44       SchemeOpts(name, singleOpts, multiOpts)
45     }
46   }
