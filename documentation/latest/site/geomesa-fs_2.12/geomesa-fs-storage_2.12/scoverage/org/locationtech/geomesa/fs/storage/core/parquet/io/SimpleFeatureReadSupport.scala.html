73     * Zip x and y values into coordinates
74     *
75     * @param x x values
76     * @param y corresponding y values
77     * @return
78     */
79   def zip(x: Array[Double], y: Array[Double]): Array[Coordinate] = {
80     val result = Array.ofDim[Coordinate](x.length)
81     var i = 0
82     while (i < result.length) {
83       result(i) = new Coordinate(x(i), y(i))
84       i += 1
85     }
86     result

97     * Group converter that can create simple features
98     */
99   class SimpleFeatureGroupConverter(schema: SimpleFeatureParquetSchema)
100       extends GroupConverter with ValueMaterializer[SimpleFeature] {
101 
102     private val idConverter = new StringConverter()
103 
104     private val visConverter = new StringConverter() {
105       override def materialize(): String = {
106         super.materialize() match {
107           case null => null
108           case s: String => s.intern()
109         }
110       }
111     }
112 
113     private val attributes = Array.ofDim[ValueMaterializer[_ <: AnyRef]](schema.sft.getAttributeCount)
114 
115     private val converters = {
116       val builder = Array.newBuilder[ValueMaterializer[_ <: AnyRef]]
117       builder += idConverter
118       builder += visConverter
119       var i = 0
120       var fieldOffset = 2 // 0 is fid, 1 is vis
121       while (i < schema.sft.getAttributeCount) {
122         val descriptor = schema.sft.getDescriptor(i)
123         val types = ObjectType.selectType(descriptor)
124         val materializer =
125           if (types.head == ObjectType.GEOMETRY) {
126             geometry(types.last, descriptor.getEncoding.fold[GeometryEncoding](GeoParquetWkb)(GeometryEncoding.apply))
127           } else if (types.last == ObjectType.JSON) {
128             val groupType = schema.messageType.getType(fieldOffset).asGroupType()
129             if (descriptor.getJsonSchema().isDefined) {
130               new StructuralJsonConverter(groupType)
131             } else {
132               new VariantConverter(groupType)
133             }
134           } else {
135             attribute(types)
136           }
137         builder += materializer
138         attributes(i) = materializer
139         // note: zValues are excluded from our read schema, they're only used for partitioning
140         // note: bboxes have to be present for filtering, but we don't do anything with them on read
141         if (types.head == ObjectType.GEOMETRY && fieldOffset + 1 < schema.messageType.getFieldCount &&
142             schema.messageType.getFields.get(fieldOffset + 1).getName.startsWith(SimpleFeatureSchema.InternalFieldDelimiter)) {
143           builder += new BoundingBoxConverter()
144           fieldOffset += 1
145         }
146         i += 1
147         fieldOffset += 1
148       }
149       builder.result()
150     }
151 
152     override def reset(): Unit = start()
153 
154     override def materialize(): SimpleFeature = {
155       val id = idConverter.materialize()
156       val vis = visConverter.materialize()
157       val values = Array.tabulate[AnyRef](schema.sft.getAttributeCount)(i => attributes(i).materialize())
158       val userData = if (vis == null) { null } else {
159         val map = new java.util.HashMap[AnyRef, AnyRef](1)
160         map.put("geomesa.feature.visibility", vis)
161         map
162       }
163       new ScalaSimpleFeature(schema.sft, id, values, userData)
164     }
165 
166     override def start(): Unit = converters.foreach(_.reset())
167 
168     override def getConverter(fieldIndex: Int): ValueMaterializer[_ <: AnyRef] = converters(fieldIndex)
169 
170     override def end(): Unit = {}
171 
172     def fieldCount: Int = converters.length
173 
174     private def attribute(bindings: Seq[ObjectType]): ValueMaterializer[_ <: AnyRef] = {
175       bindings.head match {
176         case ObjectType.DATE     => new DateMicrosConverter()
177         case ObjectType.STRING   => new StringConverter()
178         case ObjectType.INT      => new IntConverter()
179         case ObjectType.DOUBLE   => new DoubleConverter()
180         case ObjectType.LONG     => new LongConverter()
181         case ObjectType.FLOAT    => new FloatConverter()
182         case ObjectType.BOOLEAN  => new BooleanConverter()
183         case ObjectType.BYTES    => new BytesConverter()
184         case ObjectType.LIST     => new ListConverter(attribute(bindings.drop(1)))
185         case ObjectType.MAP      => new MapConverter(attribute(bindings.slice(1, 2)), attribute(bindings.slice(2, 3)))
186         case ObjectType.UUID     => new UuidConverter()
187         case _ => throw new IllegalArgumentException(s"Can't deserialize field of type ${bindings.head}")
188       }
189     }
190 
191     private def geometry(binding: ObjectType, encoding: GeometryEncoding): ValueMaterializer[_ <: Geometry] = {
192       if (encoding == GeometryEncoding.GeoParquetWkb) {
193         new WkbConverter()
194       } else if (encoding == GeometryEncoding.GeoParquetNative) {
195         binding match {
196           case ObjectType.POINT           => new PointConverter()
197           case ObjectType.LINESTRING      => new GeoParquetNativeLineStringConverter()
198           case ObjectType.POLYGON         => new GeoParquetNativePolygonConverter()
199           case ObjectType.MULTIPOINT      => new GeoParquetNativeMultiPointConverter()
200           case ObjectType.MULTILINESTRING => new GeoParquetNativeMultiLineStringConverter()
201           case ObjectType.MULTIPOLYGON    => new GeoParquetNativeMultiPolygonConverter()
202           case _                          => new WkbConverter()
203         }
204       } else {
205         throw new UnsupportedOperationException(s"Can't read geometries encoded with '$encoding'")
206       }
207     }
208   }

425   private class JsonRecordConverter(schema: GroupType) extends GroupConverter with JsonElementConverter {
426     private val count = schema.getFieldCount
427     private val names = Array.tabulate(count)(i => schema.getFieldName(i))
428     private val converters = Array.tabulate(count)(i => jsonConverter(schema.getType(i)))
429     private var obj: JsonObject = _
430     private var set = false
431 
432     override def getConverter(fieldIndex: Int): Converter = converters(fieldIndex)
433     override def start(): Unit = {
434       set = true
435       obj = new JsonObject()
436       converters.foreach(_.reset())
437     }
438     override def end(): Unit = {
439       var i = 0
440       while (i < count) {
441         if (converters(i).isSet) {
442           obj.add(names(i), converters(i).element())
443         }
444         i += 1
445       }
446     }
447     override def reset(): Unit = { set = false; obj = null }
448     override def isSet: Boolean = set
449     override def element(): JsonElement = obj
450   }

500   private class PrimitiveJsonConverter(schema: PrimitiveType) extends PrimitiveConverter with JsonElementConverter {
501     private val annotation = Option(schema.getLogicalTypeAnnotation)
502     private var value: JsonElement = _
503 
504     override def isSet: Boolean = value != null
505     override def reset(): Unit = value = null
506     override def element(): JsonElement = value
507 
508     override def addBoolean(v: Boolean): Unit = value = new JsonPrimitive(Boolean.box(v))
509     override def addFloat(v: Float): Unit = value = new JsonPrimitive(Float.box(v))
510     override def addDouble(v: Double): Unit = value = new JsonPrimitive(Double.box(v))
511 
512     override def addInt(v: Int): Unit = value = annotation match {
513       case Some(_: DateLogicalTypeAnnotation) => new JsonPrimitive(StructuralJson.dateToJson(v.toLong))
514       case Some(t: TimeLogicalTypeAnnotation) => new JsonPrimitive(StructuralJson.timeToJson(v.toLong, t.getUnit))
515       case Some(d: DecimalLogicalTypeAnnotation) => new JsonPrimitive(BigDecimal(BigInt(v), d.getScale).bigDecimal)
516       case _ => new JsonPrimitive(Int.box(v))
517     }
518 
519     override def addLong(v: Long): Unit = value = annotation match {
520       case Some(t: TimestampLogicalTypeAnnotation) => new JsonPrimitive(StructuralJson.timestampToJson(v, t.getUnit, t.isAdjustedToUTC))
521       case Some(t: TimeLogicalTypeAnnotation) => new JsonPrimitive(StructuralJson.timeToJson(v, t.getUnit))
522       case Some(d: DecimalLogicalTypeAnnotation) => new JsonPrimitive(BigDecimal(BigInt(v), d.getScale).bigDecimal)
523       case _ => new JsonPrimitive(Long.box(v))
524     }
525 
526     override def addBinary(v: Binary): Unit = value = annotation match {
527       case Some(_: StringLogicalTypeAnnotation) => new JsonPrimitive(v.toStringUsingUTF8)
528       case Some(_: EnumLogicalTypeAnnotation) => new JsonPrimitive(v.toStringUsingUTF8)
529       case Some(d: DecimalLogicalTypeAnnotation) => new JsonPrimitive(BigDecimal(BigInt(v.getBytes), d.getScale).bigDecimal)
530       case Some(_: UUIDLogicalTypeAnnotation) =>
531         val bb = v.toByteBuffer
532         new JsonPrimitive(new UUID(bb.getLong, bb.getLong).toString)
533       case _ => new JsonPrimitive(StructuralJson.bytesToJson(v.getBytes))
534     }
535   }
