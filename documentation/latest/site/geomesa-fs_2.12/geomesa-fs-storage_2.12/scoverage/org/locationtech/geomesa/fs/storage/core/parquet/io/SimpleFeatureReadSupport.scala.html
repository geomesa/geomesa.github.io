111     private val converters = {
112       val builder = Array.newBuilder[ValueMaterializer[_ <: AnyRef]]
113       builder += idConverter
114       builder += visConverter
115       var i = 0
116       var offset = 2 // 0 is fid, 1 is vis
117       while (i < schema.sft.getAttributeCount) {
118         val types = ObjectType.selectType(schema.sft.getDescriptor(i))
119         val materializer = attribute(types, offset)
120         builder += materializer
121         attributes(i) = materializer
122         // note: zValues are excluded from our read schema, they're only used for partitioning
123         // note: bboxes have to be present for filtering, but we don't do anything with them on read
124         if (types.head == ObjectType.GEOMETRY && offset + 1 < schema.messageType.getFieldCount &&
125             schema.messageType.getFields.get(offset + 1).getName.startsWith(SimpleFeatureSchema.InternalFieldDelimiter)) {
126           builder += new BoundingBoxConverter()
127           offset += 1
128         }
129         i += 1
130         offset += 1
131       }
132       builder.result()
133     }

137     override def materialize(): SimpleFeature = {
138       val id = idConverter.materialize()
139       val vis = visConverter.materialize()
140       val values = Array.tabulate[AnyRef](schema.sft.getAttributeCount)(i => attributes(i).materialize())
141       val userData = if (vis == null) { null } else {
142         val map = new java.util.HashMap[AnyRef, AnyRef](1)
143         map.put("geomesa.feature.visibility", vis)

145       }

158       bindings.head match {

162         case ObjectType.INT      => new IntConverter()
163         case ObjectType.DOUBLE   => new DoubleConverter()
164         case ObjectType.LONG     => new LongConverter()

167         case ObjectType.BYTES    => new BytesConverter()

172       }
173     }
