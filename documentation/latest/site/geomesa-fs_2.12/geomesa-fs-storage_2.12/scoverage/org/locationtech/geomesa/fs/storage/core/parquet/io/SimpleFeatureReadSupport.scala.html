94       extends GroupConverter with ValueMaterializer[SimpleFeature] {
95 
96     private val idConverter = new StringConverter()
97 
98     private val visConverter = new StringConverter() {
99       override def materialize(): String = {
100         super.materialize() match {
101           case null => null
102           case s: String => s.intern()
103         }
104       }
105     }

107     private val attributes = Array.ofDim[ValueMaterializer[_ <: AnyRef]](schema.sft.getAttributeCount)
108 
109     private val converters = {
110       val builder = Array.newBuilder[ValueMaterializer[_ <: AnyRef]]
111       builder += idConverter
112       builder += visConverter
113       var i = 0
114       var offset = 2 // 0 is fid, 1 is vis
115       while (i < schema.sft.getAttributeCount) {
116         val types = ObjectType.selectType(schema.sft.getDescriptor(i))
117         val materializer = attribute(types)
118         builder += materializer
119         attributes(i) = materializer
120         // note: zValues are excluded from our read schema, they're only used for partitioning
121         // note: bboxes have to be present for filtering, but we don't do anything with them on read
122         if (types.head == ObjectType.GEOMETRY && offset + 1 < schema.messageType.getFieldCount &&
123             schema.messageType.getFields.get(offset + 1).getName.startsWith(SimpleFeatureSchema.InternalFieldDelimiter)) {
124           builder += new BoundingBoxConverter()
125           offset += 1
126         }
127         i += 1
128         offset += 1
129       }
130       builder.result()
131     }
132 
133     override def reset(): Unit = start()
134 
135     override def materialize(): SimpleFeature = {
136       val id = idConverter.materialize()
137       val vis = visConverter.materialize()
138       val values = Array.tabulate[AnyRef](schema.sft.getAttributeCount)(i => attributes(i).materialize())
139       val userData = if (vis == null) { null } else {
140         val map = new java.util.HashMap[AnyRef, AnyRef](1)
141         map.put("geomesa.feature.visibility", vis)
142         map
143       }
144       new ScalaSimpleFeature(schema.sft, id, values, userData)
145     }
146 
147     override def start(): Unit = converters.foreach(_.reset())
148 
149     override def getConverter(fieldIndex: Int): ValueMaterializer[_ <: AnyRef] = converters(fieldIndex)
150 
151     override def end(): Unit = {}
152 
153     def fieldCount: Int = converters.length

155     private def attribute(bindings: Seq[ObjectType]): ValueMaterializer[_ <: AnyRef] = {
156       bindings.head match {
157         case ObjectType.GEOMETRY => geometry(bindings.last)
158         case ObjectType.DATE     => new DateMicrosConverter()
159         case ObjectType.STRING   => new StringConverter()
160         case ObjectType.INT      => new IntConverter()
161         case ObjectType.DOUBLE   => new DoubleConverter()
162         case ObjectType.LONG     => new LongConverter()
163         case ObjectType.FLOAT    => new FloatConverter()
164         case ObjectType.BOOLEAN  => new BooleanConverter()
165         case ObjectType.BYTES    => new BytesConverter()
166         case ObjectType.LIST     => new ListConverter(attribute(bindings.drop(1)))
167         case ObjectType.MAP      => new MapConverter(attribute(bindings.slice(1, 2)), attribute(bindings.slice(2, 3)))
168         case ObjectType.UUID     => new UuidConverter()
169         case _ => throw new IllegalArgumentException(s"Can't deserialize field of type ${bindings.head}")
170       }
171     }
172 
173     private def geometry(binding: ObjectType): ValueMaterializer[_ <: Geometry] = {
174       if (schema.geometries == GeometryEncoding.GeoParquetWkb) {
175         new WkbConverter()
176       } else if (schema.geometries == GeometryEncoding.GeoParquetNative) {
177         binding match {
178           case ObjectType.POINT           => new PointConverter()
179           case ObjectType.LINESTRING      => new GeoParquetNativeLineStringConverter()
180           case ObjectType.POLYGON         => new GeoParquetNativePolygonConverter()
181           case ObjectType.MULTIPOINT      => new GeoParquetNativeMultiPointConverter()
182           case ObjectType.MULTILINESTRING => new GeoParquetNativeMultiLineStringConverter()
183           case ObjectType.MULTIPOLYGON    => new GeoParquetNativeMultiPolygonConverter()
184           case _                          => new WkbConverter()
185         }
186       } else {
187         throw new UnsupportedOperationException(s"Can't read geometries encoded with '${schema.geometries}'")
188       }
189     }
