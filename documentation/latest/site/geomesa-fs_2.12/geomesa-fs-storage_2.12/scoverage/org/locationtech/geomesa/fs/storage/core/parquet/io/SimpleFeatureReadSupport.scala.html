74     * @param x x values
75     * @param y corresponding y values
76     * @return
77     */
78   def zip(x: Array[Double], y: Array[Double]): Array[Coordinate] = {
79     val result = Array.ofDim[Coordinate](x.length)
80     var i = 0
81     while (i < result.length) {
82       result(i) = new Coordinate(x(i), y(i))
83       i += 1
84     }
85     result

98   class SimpleFeatureGroupConverter(schema: SimpleFeatureParquetSchema)
99       extends GroupConverter with ValueMaterializer[SimpleFeature] {
100 
101     private val idConverter = new StringConverter()
102 
103     private val visConverter = new StringConverter() {
104       override def materialize(): String = {
105         super.materialize() match {
106           case null => null
107           case s: String => s.intern()
108         }
109       }
110     }
111 
112     private val attributes = Array.ofDim[ValueMaterializer[_ <: AnyRef]](schema.sft.getAttributeCount)
113 
114     private val converters = {
115       val builder = Array.newBuilder[ValueMaterializer[_ <: AnyRef]]
116       builder += idConverter
117       builder += visConverter
118       var i = 0
119       var fieldOffset = 2 // 0 is fid, 1 is vis
120       while (i < schema.sft.getAttributeCount) {
121         val descriptor = schema.sft.getDescriptor(i)
122         val types = ObjectType.selectType(descriptor)
123         val materializer =
124           if (types.head == ObjectType.GEOMETRY) {
125             geometry(types.last)
126           } else if (types.last == ObjectType.JSON) {
127             val groupType = schema.messageType.getType(fieldOffset).asGroupType()
128             if (descriptor.getJsonSchema().isDefined) {
129               new StructuralJsonConverter(groupType)
130             } else {
131               new VariantConverter(groupType)
132             }
133           } else {
134             attribute(types)
135           }
136         builder += materializer
137         attributes(i) = materializer
138         // note: zValues are excluded from our read schema, they're only used for partitioning
139         // note: bboxes have to be present for filtering, but we don't do anything with them on read
140         if (types.head == ObjectType.GEOMETRY && fieldOffset + 1 < schema.messageType.getFieldCount &&
141             schema.messageType.getFields.get(fieldOffset + 1).getName.startsWith(SimpleFeatureSchema.InternalFieldDelimiter)) {
142           builder += new BoundingBoxConverter()
143           fieldOffset += 1
144         }
145         i += 1
146         fieldOffset += 1
147       }
148       builder.result()
149     }
150 
151     override def reset(): Unit = start()
152 
153     override def materialize(): SimpleFeature = {
154       val id = idConverter.materialize()
155       val vis = visConverter.materialize()
156       val values = Array.tabulate[AnyRef](schema.sft.getAttributeCount)(i => attributes(i).materialize())
157       val userData = if (vis == null) { null } else {
158         val map = new java.util.HashMap[AnyRef, AnyRef](1)
159         map.put("geomesa.feature.visibility", vis)
160         map
161       }
162       new ScalaSimpleFeature(schema.sft, id, values, userData)
163     }
164 
165     override def start(): Unit = converters.foreach(_.reset())
166 
167     override def getConverter(fieldIndex: Int): ValueMaterializer[_ <: AnyRef] = converters(fieldIndex)
168 
169     override def end(): Unit = {}
170 
171     def fieldCount: Int = converters.length
172 
173     private def attribute(bindings: Seq[ObjectType]): ValueMaterializer[_ <: AnyRef] = {
174       bindings.head match {
175         case ObjectType.DATE     => new DateMicrosConverter()
176         case ObjectType.STRING   => new StringConverter()
177         case ObjectType.INT      => new IntConverter()
178         case ObjectType.DOUBLE   => new DoubleConverter()
179         case ObjectType.LONG     => new LongConverter()
180         case ObjectType.FLOAT    => new FloatConverter()
181         case ObjectType.BOOLEAN  => new BooleanConverter()
182         case ObjectType.BYTES    => new BytesConverter()
183         case ObjectType.LIST     => new ListConverter(attribute(bindings.drop(1)))
184         case ObjectType.MAP      => new MapConverter(attribute(bindings.slice(1, 2)), attribute(bindings.slice(2, 3)))
185         case ObjectType.UUID     => new UuidConverter()
186         case _ => throw new IllegalArgumentException(s"Can't deserialize field of type ${bindings.head}")
187       }
188     }
189 
190     private def geometry(binding: ObjectType): ValueMaterializer[_ <: Geometry] = {
191       if (schema.geometries == GeometryEncoding.GeoParquetWkb) {
192         new WkbConverter()
193       } else if (schema.geometries == GeometryEncoding.GeoParquetNative) {
194         binding match {
195           case ObjectType.POINT           => new PointConverter()
196           case ObjectType.LINESTRING      => new GeoParquetNativeLineStringConverter()
197           case ObjectType.POLYGON         => new GeoParquetNativePolygonConverter()
198           case ObjectType.MULTIPOINT      => new GeoParquetNativeMultiPointConverter()
199           case ObjectType.MULTILINESTRING => new GeoParquetNativeMultiLineStringConverter()
200           case ObjectType.MULTIPOLYGON    => new GeoParquetNativeMultiPolygonConverter()
201           case _                          => new WkbConverter()
202         }
203       } else {
204         throw new UnsupportedOperationException(s"Can't read geometries encoded with '${schema.geometries}'")
205       }
206     }
207   }

424   private class JsonRecordConverter(schema: GroupType) extends GroupConverter with JsonElementConverter {
425     private val count = schema.getFieldCount
426     private val names = Array.tabulate(count)(i => schema.getFieldName(i))
427     private val converters = Array.tabulate(count)(i => jsonConverter(schema.getType(i)))
428     private var obj: JsonObject = _
429     private var set = false
430 
431     override def getConverter(fieldIndex: Int): Converter = converters(fieldIndex)
432     override def start(): Unit = {
433       set = true
434       obj = new JsonObject()
435       converters.foreach(_.reset())
436     }
437     override def end(): Unit = {
438       var i = 0
439       while (i < count) {
440         if (converters(i).isSet) {
441           obj.add(names(i), converters(i).element())
442         }
443         i += 1
444       }
445     }
446     override def reset(): Unit = { set = false; obj = null }
447     override def isSet: Boolean = set
448     override def element(): JsonElement = obj
449   }

499   private class PrimitiveJsonConverter(schema: PrimitiveType) extends PrimitiveConverter with JsonElementConverter {
500     private val annotation = Option(schema.getLogicalTypeAnnotation)
501     private var value: JsonElement = _
502 
503     override def isSet: Boolean = value != null
504     override def reset(): Unit = value = null
505     override def element(): JsonElement = value
506 
507     override def addBoolean(v: Boolean): Unit = value = new JsonPrimitive(Boolean.box(v))
508     override def addFloat(v: Float): Unit = value = new JsonPrimitive(Float.box(v))
509     override def addDouble(v: Double): Unit = value = new JsonPrimitive(Double.box(v))
510 
511     override def addInt(v: Int): Unit = value = annotation match {
512       case Some(_: DateLogicalTypeAnnotation) => new JsonPrimitive(StructuralJson.dateToJson(v.toLong))
513       case Some(t: TimeLogicalTypeAnnotation) => new JsonPrimitive(StructuralJson.timeToJson(v.toLong, t.getUnit))
514       case Some(d: DecimalLogicalTypeAnnotation) => new JsonPrimitive(BigDecimal(BigInt(v), d.getScale).bigDecimal)
515       case _ => new JsonPrimitive(Int.box(v))
516     }
517 
518     override def addLong(v: Long): Unit = value = annotation match {
519       case Some(t: TimestampLogicalTypeAnnotation) => new JsonPrimitive(StructuralJson.timestampToJson(v, t.getUnit, t.isAdjustedToUTC))
520       case Some(t: TimeLogicalTypeAnnotation) => new JsonPrimitive(StructuralJson.timeToJson(v, t.getUnit))
521       case Some(d: DecimalLogicalTypeAnnotation) => new JsonPrimitive(BigDecimal(BigInt(v), d.getScale).bigDecimal)
522       case _ => new JsonPrimitive(Long.box(v))
523     }
524 
525     override def addBinary(v: Binary): Unit = value = annotation match {
526       case Some(_: StringLogicalTypeAnnotation) => new JsonPrimitive(v.toStringUsingUTF8)
527       case Some(_: EnumLogicalTypeAnnotation) => new JsonPrimitive(v.toStringUsingUTF8)
528       case Some(d: DecimalLogicalTypeAnnotation) => new JsonPrimitive(BigDecimal(BigInt(v.getBytes), d.getScale).bigDecimal)
529       case Some(_: UUIDLogicalTypeAnnotation) =>
530         val bb = v.toByteBuffer
531         new JsonPrimitive(new UUID(bb.getLong, bb.getLong).toString)
532       case _ => new JsonPrimitive(StructuralJson.bytesToJson(v.getBytes))
533     }
534   }
