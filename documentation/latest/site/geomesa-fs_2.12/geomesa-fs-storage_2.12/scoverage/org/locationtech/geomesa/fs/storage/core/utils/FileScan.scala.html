19 /**
20  * Fluent builder for scanning files from an iceberg table
21  *
22  * @param tableScan underlying scan
23  */
24 class FileScan(protected val tableScan: TableScan) {
25 
26   import scala.collection.JavaConverters._
27 
28   /**
29    * Return the selected files
30    *
31    * @return
32    */
33   def scan(): Seq[FileScanTask] = WithClose(tableScan.planFiles())(_.asScala.toList)
34 }
35 
36 object FileScan {
37 
38   type FluentScan = FileScan with RetrieveMetadata[FilterScan[FileScan]] with FilterScan[RetrieveMetadata[FileScan]]
39 
40   /**
41    * A step in the scan builder that allows for selecting metadata (e.g. column-level stats) to be retrieved
42    */
43   trait RetrieveMetadata[T <: FileScan] extends FileScan {
44     protected def tableScan: TableScan
45     protected def metadataStep(scan: TableScan): T
46 
47     /**
48      * Include column stats in the returned files
49      *
50      * @return
51      */
52     def includeFileStats(): T = metadataStep(tableScan.includeColumnStats())
53   }
54 
55   /**
56    * A step in the scan builder that allows for filtering the returned files based on a CQL filter or partition
57    */
58   trait FilterScan[T <: FileScan] extends FileScan {
59 
60     protected def schema: SimpleFeatureIcebergSchema
61     protected def schemes: Seq[PartitionScheme]
62     protected def filterStep(scan: TableScan): T
63 
64     /**
65      * Filter results based on CQL
66      *
67      * @param filter filter
68      * @return
69      */
70     def forFilter(filter: Filter): FileScan =
71       filterStep(tableScan.filter(IcebergFilterConverter(schema, schemes, filter).expression))
72 
73     /**
74      * Filter results based on a partition
75      *
76      * @param partition partition
77      * @return
78      */
79     def forPartition(partition: Partition): T = {
80       val filters = schemes.zip(partition.values).map { case (s, p) => s.getCoveringExpression(p) }
81       filterStep(tableScan.filter(filters.reduce(Expressions.and)))
82     }
83   }
84 
85   /**
86    * Create a new fluent file scan builder
87    *
88    * @param table table
89    * @param schema table schema
90    * @param schemes partition schemes
91    * @return
92    */
93   def apply(table: Table, schema: SimpleFeatureIcebergSchema, schemes: Seq[PartitionScheme]): FluentScan =
94     new InitialScan(table.newScan().caseSensitive(false), schema, schemes)
95 
96   // fluent class implementations to enforce scan building steps
97 
98   private class InitialScan(scan: TableScan, protected val schema: SimpleFeatureIcebergSchema, protected val schemes: Seq[PartitionScheme])
99       extends FileScan(scan)
100         with RetrieveMetadata[FilterScan[FileScan]]
101         with FilterScan[RetrieveMetadata[FileScan]] {
102     override protected def metadataStep(scan: TableScan): FilterableScan = new FilterableScan(scan, schema, schemes)
103     override protected def filterStep(scan: TableScan): OptionalMetadataScan = new OptionalMetadataScan(scan)
104   }
105 
106   private class FilterableScan(scan: TableScan, protected val schema: SimpleFeatureIcebergSchema, protected val schemes: Seq[PartitionScheme])
107       extends FileScan(scan) with FilterScan[FileScan] {
108     override protected def filterStep(scan: TableScan): FileScan = new FileScan(scan)
109   }
110 
111   private class OptionalMetadataScan(scan: TableScan) extends FileScan(scan) with RetrieveMetadata[FileScan] {
112     override protected def metadataStep(scan: TableScan): FileScan = new FileScan(scan)
113   }
114 }
