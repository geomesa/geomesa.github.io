19 /**
20  * Fluent builder for scanning files from an iceberg table
21  *
22  * @param tableScan underlying scan
23  */
24 class FileScan(protected val tableScan: TableScan) {
25 
26   import scala.collection.JavaConverters._
27 
28   /**
29    * Return the selected files
30    *
31    * @return
32    */
33   def scan(): Seq[DataFile] = WithClose(tableScan.planFiles())(_.asScala.map(_.file()).toSeq)
34 }
35 
36 object FileScan {
37 
38   type FluentScan = FileScan with RetrieveMetadata[FilterScan[FileScan]] with FilterScan[RetrieveMetadata[FileScan]]
39 
40   /**
41    * A step in the scan builder that allows for selecting metadata (e.g. column-level stats) to be retrieved
42    */
43   trait RetrieveMetadata[T <: FileScan] extends FileScan {
44     protected def tableScan: TableScan
45     protected def metadataStep(scan: TableScan): T
46 
47     /**
48      * Include column stats in the returned files
49      *
50      * @return
51      */
52     def includeFileStats(): T = metadataStep(tableScan.includeColumnStats())
53   }
54 
55   /**
56    * A step in the scan builder that allows for filtering the returned files based on a CQL filter or partition
57    */
58   trait FilterScan[T <: FileScan] extends FileScan {
59 
60     protected def tableScan: TableScan
61     protected def schema: SimpleFeatureIcebergSchema
62     protected def schemes: Seq[PartitionScheme]
63     protected def filterStep(scan: TableScan): T
64 
65     /**
66      * Filter results based on CQL
67      *
68      * @param filter filter
69      * @return
70      */
71     def forFilter(filter: Filter): FileScan =
72       filterStep(tableScan.filter(IcebergFilterConverter(schema, schemes, filter).expression))
73 
74     /**
75      * Filter results based on a partition
76      *
77      * @param partition partition
78      * @return
79      */
80     def forPartition(partition: Partition): T = {
81       val filters = schemes.zip(partition.values).map { case (s, p) => s.getCoveringExpression(p) }
82       filterStep(tableScan.filter(filters.reduce(Expressions.and)))
83     }
84   }
85 
86   /**
87    * Create a new fluent file scan builder
88    *
89    * @param table table
90    * @param schema table schema
91    * @param schemes partition schemes
92    * @return
93    */
94   def apply(table: Table, schema: SimpleFeatureIcebergSchema, schemes: Seq[PartitionScheme]): FluentScan =
95     new InitialScan(table.newScan().caseSensitive(false), schema, schemes)
96 
97   // fluent class implementations to enforce scan building steps
98 
99   private class InitialScan(scan: TableScan, protected val schema: SimpleFeatureIcebergSchema, protected val schemes: Seq[PartitionScheme])
100       extends FileScan(scan)
101         with RetrieveMetadata[FilterScan[FileScan]]
102         with FilterScan[RetrieveMetadata[FileScan]] {
103     override protected def metadataStep(scan: TableScan): FilterableScan = new FilterableScan(scan, schema, schemes)
104     override protected def filterStep(scan: TableScan): OptionalMetadataScan = new OptionalMetadataScan(scan)
105   }
106 
107   private class FilterableScan(scan: TableScan, protected val schema: SimpleFeatureIcebergSchema, protected val schemes: Seq[PartitionScheme])
108       extends FileScan(scan) with FilterScan[FileScan] {
109     override protected def filterStep(scan: TableScan): FileScan = new FileScan(scan)
110   }
111 
112   private class OptionalMetadataScan(scan: TableScan) extends FileScan(scan) with RetrieveMetadata[FileScan] {
113     override protected def metadataStep(scan: TableScan): FileScan = new FileScan(scan)
114   }
115 }
