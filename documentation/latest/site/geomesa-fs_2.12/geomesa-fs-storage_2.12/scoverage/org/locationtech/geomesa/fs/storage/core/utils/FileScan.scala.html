20 /**
21  * Fluent builder for scanning files from an iceberg table
22  *
23  * @param tableScan underlying scan
24  */
25 class FileScan(protected val tableScan: TableScan) {
26 
27   import scala.collection.JavaConverters._
28 
29   /**
30    * Return the selected files
31    *
32    * @return
33    */
34   def scan(): Seq[DataFile] = WithClose(tableScan.planFiles())(_.asScala.map(_.file()).toSeq)
35 }
36 
37 object FileScan {
38 
39   type FluentScan = FileScan with RetrieveMetadata[FilterScan[FileScan]] with FilterScan[RetrieveMetadata[FileScan]]
40 
41   /**
42    * A step in the scan builder that allows for selecting metadata (e.g. column-level stats) to be retrieved
43    */
44   trait RetrieveMetadata[T <: FileScan] extends FileScan {
45     protected def tableScan: TableScan
46     protected def metadataStep(scan: TableScan): T
47 
48     /**
49      * Include column stats in the returned files
50      *
51      * @return
52      */
53     def includeFileStats(): T = metadataStep(tableScan.includeColumnStats())
54   }
55 
56   /**
57    * A step in the scan builder that allows for filtering the returned files based on a CQL filter or partition
58    */
59   trait FilterScan[T <: FileScan] extends FileScan {
60 
61     protected def tableScan: TableScan
62     protected def sft: SimpleFeatureType
63     protected def schemes: Seq[PartitionScheme]
64     protected def filterStep(scan: TableScan): T
65 
66     /**
67      * Filter results based on CQL
68      *
69      * @param filter filter
70      * @return
71      */
72     def forFilter(filter: Filter): FileScan =
73       filterStep(tableScan.filter(IcebergFilterConverter(sft, schemes, filter).expression))
74 
75     /**
76      * Filter results based on a partition
77      *
78      * @param partition partition
79      * @return
80      */
81     def forPartition(partition: Partition): T = {
82       val filters = schemes.zip(partition.values).map { case (s, p) => s.getCoveringExpression(p) }
83       filterStep(tableScan.filter(filters.reduce(Expressions.and)))
84     }
85   }
86 
87   /**
88    * Create a new fluent file scan builder
89    *
90    * @param table table
91    * @param sft simple feature type
92    * @param schemes partition schemes
93    * @return
94    */
95   def apply(table: Table, sft: SimpleFeatureType, schemes: Seq[PartitionScheme]): FluentScan =
96     new InitialScan(table.newScan().caseSensitive(false), sft, schemes)
97 
98   // fluent class implementations to enforce scan building steps
99 
100   private class InitialScan(scan: TableScan, protected val sft: SimpleFeatureType, protected val schemes: Seq[PartitionScheme])
101       extends FileScan(scan)
102         with RetrieveMetadata[FilterScan[FileScan]]
103         with FilterScan[RetrieveMetadata[FileScan]] {
104     override protected def metadataStep(scan: TableScan): FilterableScan = new FilterableScan(scan, sft, schemes)
105     override protected def filterStep(scan: TableScan): OptionalMetadataScan = new OptionalMetadataScan(scan)
106   }
107 
108   private class FilterableScan(scan: TableScan, protected val sft: SimpleFeatureType, protected val schemes: Seq[PartitionScheme])
109       extends FileScan(scan) with FilterScan[FileScan] {
110     override protected def filterStep(scan: TableScan): FileScan = new FileScan(scan)
111   }
112 
113   private class OptionalMetadataScan(scan: TableScan) extends FileScan(scan) with RetrieveMetadata[FileScan] {
114     override protected def metadataStep(scan: TableScan): FileScan = new FileScan(scan)
115   }
116 }
