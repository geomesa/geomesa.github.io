20 /**
21  * Fluent builder for scanning files from an iceberg table
22  *
23  * @param tableScan underlying scan
24  */
25 class FileScan(protected val tableScan: TableScan) {
26 
27   import scala.collection.JavaConverters._
28 
29   /**
30    * Return the selected files
31    *
32    * @return
33    */
34   def scan(): Seq[DataFile] = WithClose(tableScan.planFiles())(_.asScala.map(_.file()).toSeq)
35 }
36 
37 object FileScan {
38 
39   type FluentScan =
40     FileScan with RetrieveMetadata[PartitionScan[FileScan] with FilterScan[FileScan]]
41       with PartitionScan[RetrieveMetadata[FileScan]] with FilterScan[RetrieveMetadata[FileScan]]
42 
43   /**
44    * A step in the scan builder that allows for selecting metadata (e.g. column-level stats) to be retrieved
45    */
46   trait RetrieveMetadata[T <: FileScan] extends FileScan {
47     protected def tableScan: TableScan
48     protected def metadataStep(scan: TableScan): T
49 
50     def includeFileStats(): T = metadataStep(tableScan.includeColumnStats())
51   }
52 
53   /**
54    * A step in the scan builder that allows for filtering the returned files based on partition
55    */
56   trait PartitionScan[T <: FileScan] extends FileScan {
57 
58     protected def tableScan: TableScan
59     protected def schemes: Seq[PartitionScheme]
60     protected def filterStep(scan: TableScan): T
61 
62     def forPartition(partition: Partition): T = {
63       val filters = schemes.zip(partition.values).map { case (s, p) => s.getCoveringExpression(p) }
64       filterStep(tableScan.filter(filters.reduce(Expressions.and)))
65     }
66   }
67 
68   /**
69    * A step in the scan builder that allows for filtering the returned files based on a CQL filter
70    */
71   trait FilterScan[T <: FileScan] extends FileScan {
72 
73     protected def tableScan: TableScan
74     protected def sft: SimpleFeatureType
75     protected def filterStep(scan: TableScan): T
76 
77     def forFilter(filter: Filter): FileScan = filterStep(tableScan.filter(IcebergFilterConverter(sft, filter).expression))
78   }
79 
80   /**
81    * Create a new fluent file scan builder
82    *
83    * @param table table
84    * @param sft simple feature type
85    * @param schemes partition schemes
86    * @return
87    */
88   def apply(table: Table, sft: SimpleFeatureType, schemes: Seq[PartitionScheme]): FluentScan =
89     new InitialScan(table.newScan().caseSensitive(false), sft, schemes)
90 
91   // fluent class implementations to enforce scan building steps
92 
93   private class InitialScan(scan: TableScan, protected val sft: SimpleFeatureType, protected val schemes: Seq[PartitionScheme])
94       extends FileScan(scan)
95         with RetrieveMetadata[PartitionScan[FileScan] with FilterScan[FileScan]]
96         with PartitionScan[RetrieveMetadata[FileScan]]
97         with FilterScan[RetrieveMetadata[FileScan]] {
98     override protected def metadataStep(scan: TableScan): FilterableScan = new FilterableScan(scan, sft, schemes)
99     override protected def filterStep(scan: TableScan): OptionalMetadataScan = new OptionalMetadataScan(scan)
100   }
101 
102   private class FilterableScan(scan: TableScan, protected val sft: SimpleFeatureType, protected val schemes: Seq[PartitionScheme])
103       extends FileScan(scan) with PartitionScan[FileScan] with FilterScan[FileScan] {
104     override protected def filterStep(scan: TableScan): FileScan = new FileScan(scan)
105   }
106 
107   private class OptionalMetadataScan(scan: TableScan) extends FileScan(scan) with RetrieveMetadata[FileScan] {
108     override protected def metadataStep(scan: TableScan): FileScan = new FileScan(scan)
109   }
