20 /**
21  * Scheme for partitioning features into various named partitions (e.g. buckets) on disk, for
22  * faster querying. Partition schemes do not have any persistent state, they only know how to map features
23  * and filters to partition names.
24  *
25  * Note that generally partitioning is implemented to align with iceberg's partitioning logic, so
26  * that partitions in iceberg have a 1:1 mapping with partitions in geomesa (the reverse is not true,
27  * as geomesa supports more partition schemes than iceberg, e.g. spatial partitions)
28  */
29 trait PartitionScheme {
30 
31   /**
32    * Name of this partition scheme
33    *
34    * @return
35    */
36   def name: String
37 
38   /**
39    * Name of the attribute used for partitioning
40    *
41    * @return
42    */
43   def attribute: String
44 
45   /**
46    * Name of the column used for expressions
47    *
48    * @return
49    */
50   def column: String
51 
52   /**
53     * Return the partition in which a SimpleFeature should be stored
54     *
55     * @param feature simple feature
56     * @return partition name
57     */
58   def getPartition(feature: SimpleFeature): PartitionKey
59 
60   /**
61    * Gets the geomesa partition value for a given iceberg partition value
62    *
63    * @param partition iceberg partition struct
64    * @param i offest into the partition struct
65    * @return geomesa partition value
66    */
67   def getPartition(partition: StructLike, i: Int): PartitionKey
68 
69   /**
70    * Get a filter that will cover a partitions, i.e. the filter will return all features
71    * in the given partition and none from other partitions
72    *
73    * @param partition partition to cover
74    * @return filter
75    */
76   def getCoveringFilter(partition: PartitionKey): Filter
77 
78   /**
79    * Get a filter that will cover a partitions, i.e. the filter will return all features
80    * in the given partition and none from other partitions
81    *
82    * @param partition partition to cover
83    * @return filter
84    */
85   def getCoveringExpression(partition: PartitionKey): Expression
86 
87   /**
88    * Creates the iceberg partition spec
89    *
90    * @param b spec builder
91    * @return
92    */
93   def spec(b: PartitionSpec.Builder): PartitionSpec.Builder
94 }

96 object PartitionScheme {
97 
98   /**
99    * Enumerate all possible partitions for a given date. Note that not all partition schemes support enumeration
100    *
101    * @param schemes partition schemes
102    * @param date date
103    * @return all possible partitions for the given date
104    * @throws UnsupportedOperationException if the partition scheme can't be enumerated
105    */
106   def enumerate(schemes: Seq[PartitionScheme], date: ZonedDateTime): Seq[Partition] = {
107     val keys = schemes.map {
108       case s: TemporalScheme => Seq(s.partition(date).partition)
109       case s: EnumeratedScheme => s.partitions
110       case s => throw new UnsupportedOperationException(s"The partition scheme '${s.name}' does not support enumerating partitions")
111     }
112     keys.foldLeft(Seq(Partition.None)) { case (partitions, keys) =>
113       for { partition <- partitions; key <- keys } yield {
114         Partition(partition.values :+ key)
115       }
116     }
117   }
118 
119   /**
120    * A scheme with a fixed set of partitions
121    */
122   trait EnumeratedScheme extends PartitionScheme {
123 
124     /**
125      * Get all potential partitions for this scheme
126      *
127      * @return
128      */
129     def partitions: Seq[PartitionKey]
130   }
131 
132   /**
133    * A schemed with a fixed set of partitions per time period
134    */
135   trait TemporalScheme extends PartitionScheme {
136 
137     /**
138      * Gets the partitions for this scheme for the given time
139      *
140      * @param time date to get partitions for
141      * @return
142      */
143     def partition(time: ZonedDateTime): TemporalPartition
144   }
145 
146   /**
147    * A partition, with the associated start and end period
148    *
149    * @param partition partition
150    * @param start start of partitions (inclusive)
151    * @param end end of partitions (exclusive)
152    */
153   case class TemporalPartition(partition: PartitionKey, start: ZonedDateTime, end: ZonedDateTime)
