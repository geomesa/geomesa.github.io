17 object GeometrySchema {
18 
19   val GeometryColumnX = "x"
20   val GeometryColumnY = "y"

22   /**
23    * Enumeration of supported geometry encodings
24    */
25   sealed trait GeometryEncoding {
26 
27     /**
28      * Create a builder for a parquet geometry field
29      *
30      * @param binding geometry type
31      * @return
32      */
33     def schema(binding: ObjectType): Types.Builder[_, _ <: Type]
34   }
35 
36   object GeometryEncoding {
37 
38     def apply(name: String): GeometryEncoding = {
39       if (Seq("wkb", GeoParquetWkb.toString).exists(_.equalsIgnoreCase(name))) {
40         GeoParquetWkb
41       } else if (Seq("native", GeoParquetNative.toString).exists(_.equalsIgnoreCase(name))) {
42         GeoParquetNative
43       } else if (name.equalsIgnoreCase("GeoMesaV0") || name.equalsIgnoreCase("GeoMesaV1")) {
44         throw new UnsupportedOperationException(s"Geometry encoding '$name' is not longer supported")
45       } else {
46         throw new IllegalArgumentException(s"No geometry encoding defined for '$name'")
47       }
48     }
49 
50     case object GeoParquetNative extends GeometryEncoding {
51       override def schema(binding: ObjectType): Types.Builder[_, _ <: Type] = {
52         binding match {
53           case ObjectType.POINT =>
54             Types.optionalGroup()
55               .required(PrimitiveTypeName.DOUBLE).named(GeometryColumnX)
56               .required(PrimitiveTypeName.DOUBLE).named(GeometryColumnY)
57 
58           case ObjectType.LINESTRING =>
59             // TODO do we need multiple levels of nesting field ids???
60             Types.optionalList()
61               .requiredGroupElement() // the coordinate
62               .required(PrimitiveTypeName.DOUBLE).named(GeometryColumnX)
63               .required(PrimitiveTypeName.DOUBLE).named(GeometryColumnY)
64 
65           case ObjectType.MULTIPOINT =>
66             Types.optionalList
67               .requiredGroupElement() // the coordinate
68               .required(PrimitiveTypeName.DOUBLE).named(GeometryColumnX)
69               .required(PrimitiveTypeName.DOUBLE).named(GeometryColumnY)
70 
71           case ObjectType.POLYGON =>
72             Types.optionalList
73               .requiredListElement() // the coordinates of one ring
74               .requiredGroupElement() // the coordinate
75               .required(PrimitiveTypeName.DOUBLE).named(GeometryColumnX)
76               .required(PrimitiveTypeName.DOUBLE).named(GeometryColumnY)
77 
78           case ObjectType.MULTILINESTRING =>
79             Types.optionalList
80               .requiredListElement() // the coordinates of one line string
81               .requiredGroupElement() // the coordinate
82               .required(PrimitiveTypeName.DOUBLE).named(GeometryColumnX)
83               .required(PrimitiveTypeName.DOUBLE).named(GeometryColumnY)
84 
85           case ObjectType.MULTIPOLYGON =>
86             Types.optionalList
87               .requiredListElement() // the rings of the MultiPolygon
88               .requiredListElement() // the coordinates of one ring
89               .requiredGroupElement() // the coordinate
90               .required(PrimitiveTypeName.DOUBLE).named(GeometryColumnX)
91               .required(PrimitiveTypeName.DOUBLE).named(GeometryColumnY)
92 
93           case ObjectType.GEOMETRY_COLLECTION =>
94             Types.primitive(PrimitiveTypeName.BINARY, Repetition.OPTIONAL)
95 
96           case ObjectType.GEOMETRY =>
97             Types.primitive(PrimitiveTypeName.BINARY, Repetition.OPTIONAL)
98         }
99       }
100     }
101 
102     case object GeoParquetWkb extends GeometryEncoding {
103       override def schema(binding: ObjectType): Types.Builder[_, _ <: Type] =
104         Types.primitive(PrimitiveTypeName.BINARY, Repetition.OPTIONAL)
105     }
106   }
