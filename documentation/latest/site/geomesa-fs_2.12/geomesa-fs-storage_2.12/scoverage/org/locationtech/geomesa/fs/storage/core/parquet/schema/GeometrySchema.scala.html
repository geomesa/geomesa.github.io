17 object GeometrySchema {
18 
19   val GeometryColumnX = "x"
20   val GeometryColumnY = "y"

22   /**
23    * Enumeration of supported geometry encodings
24    */
25   sealed trait GeometryEncoding {
26 
27     /**
28      * Create a builder for a parquet geometry field
29      *
30      * @param binding geometry type
31      * @return
32      */
33     def schema(binding: ObjectType): Types.Builder[_, _ <: Type]
34   }
35 
36   object GeometryEncoding {
37 
38     def apply(name: String): GeometryEncoding = {
39       Seq(GeoParquetNative, GeoParquetWkb).find(_.toString.equalsIgnoreCase(name)).getOrElse {
40         if (name.equalsIgnoreCase("GeoMesaV0") || name.equalsIgnoreCase("GeoMesaV1")) {
41           throw new UnsupportedOperationException(s"Geometry encoding '$name' is not longer supported")
42         }
43         throw new IllegalArgumentException(s"No geometry encoding defined for '$name'")
44       }
45     }
46 
47     case object GeoParquetNative extends GeometryEncoding {
48       override def schema(binding: ObjectType): Types.Builder[_, _ <: Type] = {
49         binding match {
50           case ObjectType.POINT =>
51             Types.optionalGroup()
52               .required(PrimitiveTypeName.DOUBLE).named(GeometryColumnX)
53               .required(PrimitiveTypeName.DOUBLE).named(GeometryColumnY)
54 
55           case ObjectType.LINESTRING =>
56             // TODO do we need multiple levels of nesting field ids???
57             Types.optionalList()
58               .requiredGroupElement() // the coordinate
59               .required(PrimitiveTypeName.DOUBLE).named(GeometryColumnX)
60               .required(PrimitiveTypeName.DOUBLE).named(GeometryColumnY)
61 
62           case ObjectType.MULTIPOINT =>
63             Types.optionalList
64               .requiredGroupElement() // the coordinate
65               .required(PrimitiveTypeName.DOUBLE).named(GeometryColumnX)
66               .required(PrimitiveTypeName.DOUBLE).named(GeometryColumnY)
67 
68           case ObjectType.POLYGON =>
69             Types.optionalList
70               .requiredListElement() // the coordinates of one ring
71               .requiredGroupElement() // the coordinate
72               .required(PrimitiveTypeName.DOUBLE).named(GeometryColumnX)
73               .required(PrimitiveTypeName.DOUBLE).named(GeometryColumnY)
74 
75           case ObjectType.MULTILINESTRING =>
76             Types.optionalList
77               .requiredListElement() // the coordinates of one line string
78               .requiredGroupElement() // the coordinate
79               .required(PrimitiveTypeName.DOUBLE).named(GeometryColumnX)
80               .required(PrimitiveTypeName.DOUBLE).named(GeometryColumnY)
81 
82           case ObjectType.MULTIPOLYGON =>
83             Types.optionalList
84               .requiredListElement() // the rings of the MultiPolygon
85               .requiredListElement() // the coordinates of one ring
86               .requiredGroupElement() // the coordinate
87               .required(PrimitiveTypeName.DOUBLE).named(GeometryColumnX)
88               .required(PrimitiveTypeName.DOUBLE).named(GeometryColumnY)
89 
90           case ObjectType.GEOMETRY_COLLECTION =>
91             Types.primitive(PrimitiveTypeName.BINARY, Repetition.OPTIONAL)
92 
93           case ObjectType.GEOMETRY =>
94             Types.primitive(PrimitiveTypeName.BINARY, Repetition.OPTIONAL)
95         }
96       }
97     }
98 
99     case object GeoParquetWkb extends GeometryEncoding {
100       override def schema(binding: ObjectType): Types.Builder[_, _ <: Type] =
101         Types.primitive(PrimitiveTypeName.BINARY, Repetition.OPTIONAL)
102     }
103   }
