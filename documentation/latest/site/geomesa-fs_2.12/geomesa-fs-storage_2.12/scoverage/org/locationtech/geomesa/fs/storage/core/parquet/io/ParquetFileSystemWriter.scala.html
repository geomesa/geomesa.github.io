34 /**
35  * Parquet writer
36  *
37  * @param sft simple feature type
38  * @param conf configuration
39  * @param output file to write
40  * @param observer any observers
41  */
42 class ParquetFileSystemWriter(
43     sft: SimpleFeatureType,
44     conf: Map[String, String],
45     output: FileOutput,
46     observer: FileSystemObserver
47   ) extends FileSystemWriter {
48 
49   import scala.collection.JavaConverters._
50 
51   // TODO consolidate this on FileIO instead of ObjectStore
52   def this(sft: SimpleFeatureType, conf: Map[String, String], fs: ObjectStore, file: URI, observer: FileSystemObserver) =
53     this(sft, conf, ObjectStoreOutput(fs, file), observer)
54 
55   def this(sft: SimpleFeatureType, conf: Map[String, String], fs: ObjectStore, file: URI) =
56     this(sft, conf, fs, file, NoOpObserver)
57 
58   def this(sft: SimpleFeatureType, conf: Map[String, String], io: FileIO, file: URI, observer: FileSystemObserver) =
59     this(sft, conf, IcebergOutput(io, file), observer)
60 
61   def this(sft: SimpleFeatureType, conf: Map[String, String], io: FileIO, file: URI) =
62     this(sft, conf, io, file, NoOpObserver)
63 
64   private val parquetConf = new PlainParquetConfiguration(conf.asJava)
65   SimpleFeatureParquetSchema.setSft(parquetConf, sft)
66 
67   private val writer = ParquetFileSystemWriter.builder(output.file, parquetConf).build()
68 
69   @volatile
70   private var closed = false
71 
72   override def size: Long = if (closed) { output.size }  else { writer.getDataSize }
73 
74   override def write(f: SimpleFeature): Unit = {
75     writer.write(f)
76     observer(f)
77   }
78 
79   override def flush(): Unit = observer.flush()
80 
81   override def close(): Unit = {
82     closed = true
83     CloseQuietly(Seq(writer, observer)).foreach(e => throw e)
84   }
85 }

87 object ParquetFileSystemWriter extends LazyLogging {
88 
89   /**
90    * Create a new configurable writer
91    *
92    * @param file file to write
93    * @param conf write configuration
94    * @return
95    */
96   def builder(file: OutputFile, conf: ParquetConfiguration): Builder = {
97     val version = WriterVersion.fromString(conf.get("parquet.writer.version", WriterVersion.PARQUET_2_0.name()))
98     val codec = CompressionCodecName.fromConf(conf.get("parquet.compression", "ZSTD"))
99     logger.debug(s"Using Parquet file version $version with compression ${codec.name()}")
100     new Builder(file)
101       .withConf(conf)
102       .withCompressionCodec(codec)
103       .withDictionaryEncoding(true)
104       .withDictionaryPageSize(ParquetWriter.DEFAULT_PAGE_SIZE)
105       .withMaxPaddingSize(ParquetWriter.MAX_PADDING_SIZE_DEFAULT)
106       .withPageSize(ParquetWriter.DEFAULT_PAGE_SIZE)
107       .withValidation(false)
108       .withWriteMode(ParquetFileWriter.Mode.OVERWRITE)
109       .withWriterVersion(version)
110       .withRowGroupSize(8L*1024*1024)
111   }
112 
113   private sealed trait FileOutput {
114 
115     /**
116      * The output file
117      *
118      * @return
119      */
120     def file: OutputFile
121 
122     /**
123      * Gets the size of the file
124      *
125      * @return
126      */
127     def size: Long
128   }
129 
130   private case class ObjectStoreOutput(fs: ObjectStore, path: URI) extends FileOutput {
131 
132     override val file: OutputFile = fs match {
133       case _: LocalObjectStore => new LocalOutputFileWithParent(Path.of(path))
134       case s3: S3ObjectStore => new S3OutputFile(s3, path)
135       case _ => throw new UnsupportedOperationException(s"No file implementation for scheme ${fs.scheme}")
136     }
137 
138     override def size: Long = fs.size(path)
139   }
140 
141   private case class IcebergOutput(io: FileIO, path: URI) extends FileOutput {
142     override val file: IcebergOutputFile = new IcebergOutputFile(io.newOutputFile(path.toString))
143     override def size: Long = file.original.toInputFile.getLength
144   }
145 
146   class Builder(file: OutputFile) extends ParquetWriter.Builder[SimpleFeature, Builder](file) {
147     override def self(): Builder = this
148     override protected def getWriteSupport(conf: Configuration): WriteSupport[SimpleFeature] =
149       new SimpleFeatureWriteSupport()
150     override protected def getWriteSupport(conf: ParquetConfiguration): WriteSupport[SimpleFeature] =
151       new SimpleFeatureWriteSupport()
152   }
153 
154   private class LocalOutputFileWithParent(file: Path) extends LocalOutputFile(file) {
155     override def create(blockSize: Long): PositionOutputStream = {
156       Option(file.toFile.getParentFile).foreach(_.mkdirs())
157       super.create(blockSize)
158     }
159     override def createOrOverwrite(blockSize: Long): PositionOutputStream = {
160       Option(file.toFile.getParentFile).foreach(_.mkdirs())
161       super.createOrOverwrite(blockSize)
162     }
163   }
