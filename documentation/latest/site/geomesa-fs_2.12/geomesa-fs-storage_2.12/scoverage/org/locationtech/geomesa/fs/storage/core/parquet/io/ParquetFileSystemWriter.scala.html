156   /**
157    * Applies any parquet writer properties in the configuration to the writer. `ParquetWriter.Builder` only reads these
158    * through its own methods (unlike `ParquetOutputFormat`, which parses them out of the configuration), so they have to be
159    * passed through explicitly. Options use the standard parquet keys, as defined in `ParquetOutputFormat`. Options
160    * that can be set per column are applied to a single column by appending `#<column path>`,
161    * e.g. `parquet.bloom.filter.enabled#name`. Only options that are present are applied, so parquet's defaults are
162    * otherwise unchanged.
163    *
164    * Compression, writer version and row group size are handled separately, as they have geomesa-specific defaults.
165    *
166    * @param builder writer builder
167    * @param conf write configuration
168    * @return the builder
169    */
170   private[io] def configureProperties(builder: Builder, conf: ParquetConfiguration): Builder = {
171     import ParquetOutputFormat._
172 
173     def option[T](key: String, value: String)(parse: String => Option[T])(apply: T => Unit): Unit = {
174       parse(value.trim) match {
175         case Some(v) => apply(v)
176         case None => logger.warn(s"Ignoring invalid value for $key: '$value'")
177       }
178     }
179     def bool(v: String): Option[Boolean] = v.toLowerCase(Locale.US) match {
180       case "true" => Some(true)
181       case "false" => Some(false)
182       case _ => None
183     }
184     def int(v: String): Option[Int] = Try(v.toInt).toOption.filter(_ > 0)
185     def long(v: String): Option[Long] = Try(v.toLong).toOption.filter(_ > 0)
186     def fpp(v: String): Option[Double] = Try(v.toDouble).toOption.filter(d => d > 0 && d < 1)
187     def bytes(v: String): Option[Int] = Suffixes.Memory.bytes(v).toOption.filter(b => b >= 0 && b <= Int.MaxValue).map(_.toInt)
188     def codec(v: String): Option[CompressionCodecName] = Try(CompressionCodecName.fromConf(v)).toOption
189     def level(v: String): Option[Integer] = Try(Int.box(v.toInt)).toOption
190 
191     conf.iterator().asScala.foreach { entry =>
192       val key = entry.getKey
193       val value = entry.getValue
194       key.split("#", 2) match {
195         case Array(PAGE_SIZE)                         => option(key, value)(bytes)(builder.withPageSize)
196         case Array(DICTIONARY_PAGE_SIZE)              => option(key, value)(bytes)(builder.withDictionaryPageSize)
197         case Array(ENABLE_DICTIONARY)                 => option(key, value)(bool)(builder.withDictionaryEncoding(_))
198         case Array(ENABLE_DICTIONARY, col)            => option(key, value)(bool)(builder.withDictionaryEncoding(col, _))
199         case Array(ENABLE_BYTE_STREAM_SPLIT)          => option(key, value)(bool)(builder.withByteStreamSplitEncoding(_))
200         case Array(ENABLE_BYTE_STREAM_SPLIT, col)     => option(key, value)(bool)(builder.withByteStreamSplitEncoding(col, _))
201         case Array(VALIDATION)                        => option(key, value)(bool)(builder.withValidation)
202         case Array(MAX_PADDING_BYTES)                 => option(key, value)(bytes)(builder.withMaxPaddingSize)
203         case Array(MIN_ROW_COUNT_FOR_PAGE_SIZE_CHECK) => option(key, value)(int)(builder.withMinRowCountForPageSizeCheck)
204         case Array(MAX_ROW_COUNT_FOR_PAGE_SIZE_CHECK) => option(key, value)(int)(builder.withMaxRowCountForPageSizeCheck)
205         case Array(COLUMN_INDEX_TRUNCATE_LENGTH)      => option(key, value)(int)(builder.withColumnIndexTruncateLength)
206         case Array(STATISTICS_TRUNCATE_LENGTH)        => option(key, value)(int)(builder.withStatisticsTruncateLength)
207         case Array(BLOCK_ROW_COUNT_LIMIT)             => option(key, value)(int)(builder.withRowGroupRowCountLimit)
208         case Array(PAGE_ROW_COUNT_LIMIT)              => option(key, value)(int)(builder.withPageRowCountLimit)
209         case Array(PAGE_WRITE_CHECKSUM_ENABLED)       => option(key, value)(bool)(builder.withPageWriteChecksumEnabled)
210         case Array(STATISTICS_ENABLED)                => option(key, value)(bool)(builder.withStatisticsEnabled(_))
211         case Array(STATISTICS_ENABLED, col)           => option(key, value)(bool)(builder.withStatisticsEnabled(col, _))
212         case Array(SIZE_STATISTICS_ENABLED)           => option(key, value)(bool)(builder.withSizeStatisticsEnabled(_))
213         case Array(SIZE_STATISTICS_ENABLED, col)      => option(key, value)(bool)(builder.withSizeStatisticsEnabled(col, _))
214         case Array(COMPRESSION, col)                  => option(key, value)(codec)(builder.withCompressionCodec(col, _))
215         case Array(COLUMN_COMPRESSION_LEVEL_PREFIX, col) => option(key, value)(level)(builder.withCompressionLevel(col, _))
216         case Array(BLOOM_FILTER_ENABLED)              => option(key, value)(bool)(builder.withBloomFilterEnabled(_))
217         case Array(BLOOM_FILTER_ENABLED, col)         => option(key, value)(bool)(builder.withBloomFilterEnabled(col, _))
218         case Array(BLOOM_FILTER_EXPECTED_NDV, col)    => option(key, value)(long)(builder.withBloomFilterNDV(col, _))
219         case Array(BLOOM_FILTER_FPP, col)             => option(key, value)(fpp)(builder.withBloomFilterFPP(col, _))
220         case Array(BLOOM_FILTER_CANDIDATES_NUMBER, col) => option(key, value)(int)(builder.withBloomFilterCandidateNumber(col, _))
221         case Array(BLOOM_FILTER_MAX_BYTES)            => option(key, value)(bytes)(builder.withMaxBloomFilterBytes)
222         case Array(ADAPTIVE_BLOOM_FILTER_ENABLED)     => option(key, value)(bool)(builder.withAdaptiveBloomFilterEnabled)
223         case _ => // not a writer property (or handled elsewhere)
224       }
225     }
226     builder
227   }

229   /**
230    * Translates the parquet bloom filter properties of an iceberg table (as set by e.g. Trino's
231    * `parquet_bloom_filter_columns`, and honored by other iceberg writers such as compaction) into the
232    * equivalent parquet write options
233    *
234    * @param properties iceberg table properties
235    * @return parquet write options
236    */
237   def icebergBloomFilterConf(properties: java.util.Map[String, String]): Map[String, String] = {
238     import ParquetOutputFormat._
239     import TableProperties._
240 
241     val prefixes = Seq(
242       PARQUET_BLOOM_FILTER_COLUMN_ENABLED_PREFIX -> BLOOM_FILTER_ENABLED,
243       PARQUET_BLOOM_FILTER_COLUMN_NDV_PREFIX     -> BLOOM_FILTER_EXPECTED_NDV,
244       PARQUET_BLOOM_FILTER_COLUMN_FPP_PREFIX     -> BLOOM_FILTER_FPP,
245     )
246     properties.asScala.toMap.flatMap {
247       case (PARQUET_BLOOM_FILTER_MAX_BYTES, v) => Some(BLOOM_FILTER_MAX_BYTES -> v)
248       case (k, v) =>
249         prefixes.collectFirst { case (prefix, key) if k.startsWith(prefix) && k.length > prefix.length =>
250           s"$key#${k.substring(prefix.length)}" -> v
251         }
252     }
253   }

255   /**
256    * The configured row group size, or parquet's default (`ParquetWriter.DEFAULT_BLOCK_SIZE`) if absent or invalid
257    *
258    * @param conf write configuration
259    * @return row group size, in bytes
260    */
261   private[io] def rowGroupSize(conf: ParquetConfiguration): Long = {
262     Option(conf.get(ParquetRowGroupSizeOpt)).map(_.trim).filter(_.nonEmpty) match {
263       case None => ParquetWriter.DEFAULT_BLOCK_SIZE
264       case Some(size) =>
265         Suffixes.Memory.bytes(size).getOrElse {
266           logger.warn(s"Invalid $ParquetRowGroupSizeOpt '$size', using the default of ${ParquetWriter.DEFAULT_BLOCK_SIZE} bytes")
267           ParquetWriter.DEFAULT_BLOCK_SIZE
268         }
269     }
