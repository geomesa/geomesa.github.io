33 /**
34  * Parquet writer
35  *
36  * @param sft simple feature type
37  * @param conf configuration
38  * @param output file to write
39  * @param observer any observers
40  */
41 class ParquetFileSystemWriter(
42     sft: SimpleFeatureType,
43     conf: Map[String, String],
44     output: FileOutput,
45     observer: FileSystemObserver
46   ) extends FileSystemWriter {
47 
48   import scala.collection.JavaConverters._
49 
50   // TODO consolidate this on FileIO instead of ObjectStore
51   def this(sft: SimpleFeatureType, conf: Map[String, String], fs: ObjectStore, file: String, observer: FileSystemObserver) =
52     this(sft, conf, ObjectStoreOutput(fs, URI.create(file)), observer)
53 
54   def this(sft: SimpleFeatureType, conf: Map[String, String], fs: ObjectStore, file: String) =
55     this(sft, conf, fs, file, NoOpObserver)
56 
57   def this(sft: SimpleFeatureType, conf: Map[String, String], io: FileIO, file: String, observer: FileSystemObserver) =
58     this(sft, conf, IcebergOutput(io, file), observer)
59 
60   def this(sft: SimpleFeatureType, conf: Map[String, String], io: FileIO, file: String) =
61     this(sft, conf, io, file, NoOpObserver)
62 
63   private val parquetConf = new PlainParquetConfiguration(conf.asJava)
64   SimpleFeatureParquetSchema.setSft(parquetConf, sft)
65 
66   private val writer = ParquetFileSystemWriter.builder(output.file, parquetConf).build()
67 
68   @volatile
69   private var closed = false
70 
71   override def size: Long = if (closed) { output.size }  else { writer.getDataSize }
72 
73   override def write(f: SimpleFeature): Unit = {
74     writer.write(f)
75     observer(f)
76   }
77 
78   override def flush(): Unit = observer.flush()
79 
80   override def close(): Unit = {
81     closed = true
82     CloseQuietly(Seq(writer, observer)).foreach(e => throw e)
83   }
84 }

86 object ParquetFileSystemWriter extends LazyLogging {
87 
88   /**
89    * Create a new configurable writer
90    *
91    * @param file file to write
92    * @param conf write configuration
93    * @return
94    */
95   def builder(file: OutputFile, conf: ParquetConfiguration): Builder = {
96     val version = WriterVersion.fromString(conf.get("parquet.writer.version", WriterVersion.PARQUET_2_0.name()))
97     val codec = CompressionCodecName.fromConf(conf.get("parquet.compression", "ZSTD"))
98     logger.debug(s"Using Parquet file version $version with compression ${codec.name()}")
99     new Builder(file)
100       .withConf(conf)
101       .withCompressionCodec(codec)
102       .withDictionaryEncoding(true)
103       .withDictionaryPageSize(ParquetWriter.DEFAULT_PAGE_SIZE)
104       .withMaxPaddingSize(ParquetWriter.MAX_PADDING_SIZE_DEFAULT)
105       .withPageSize(ParquetWriter.DEFAULT_PAGE_SIZE)
106       .withValidation(false)
107       .withWriteMode(ParquetFileWriter.Mode.OVERWRITE)
108       .withWriterVersion(version)
109       .withRowGroupSize(8L*1024*1024)
110   }
111 
112   private sealed trait FileOutput {
113 
114     /**
115      * The output file
116      *
117      * @return
118      */
119     def file: OutputFile
120 
121     /**
122      * Gets the size of the file
123      *
124      * @return
125      */
126     def size: Long
127   }
128 
129   private case class ObjectStoreOutput(fs: ObjectStore, path: URI) extends FileOutput {
130 
131     override val file: OutputFile = fs match {
132       case _: LocalObjectStore => new LocalOutputFileWithParent(Path.of(path))
133       case s3: S3ObjectStore => new S3OutputFile(s3, path)
134       case _ => throw new UnsupportedOperationException(s"No file implementation for scheme ${fs.scheme}")
135     }
136 
137     override def size: Long = fs.size(path)
138   }
139 
140   private case class IcebergOutput(io: FileIO, path: String) extends FileOutput {
141     override val file: IcebergOutputFile = new IcebergOutputFile(io.newOutputFile(path))
142     override def size: Long = file.original.toInputFile.getLength
143   }
144 
145   class Builder(file: OutputFile) extends ParquetWriter.Builder[SimpleFeature, Builder](file) {
146     override def self(): Builder = this
147     override protected def getWriteSupport(conf: Configuration): WriteSupport[SimpleFeature] =
148       new SimpleFeatureWriteSupport()
149     override protected def getWriteSupport(conf: ParquetConfiguration): WriteSupport[SimpleFeature] =
150       new SimpleFeatureWriteSupport()
151   }
152 
153   private class LocalOutputFileWithParent(file: Path) extends LocalOutputFile(file) {
154     override def create(blockSize: Long): PositionOutputStream = {
155       Option(file.toFile.getParentFile).foreach(_.mkdirs())
156       super.create(blockSize)
157     }
158     override def createOrOverwrite(blockSize: Long): PositionOutputStream = {
159       Option(file.toFile.getParentFile).foreach(_.mkdirs())
160       super.createOrOverwrite(blockSize)
161     }
162   }
