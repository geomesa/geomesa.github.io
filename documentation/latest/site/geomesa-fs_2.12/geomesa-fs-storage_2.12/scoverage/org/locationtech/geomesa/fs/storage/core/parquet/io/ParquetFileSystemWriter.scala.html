33 /**
34  * Parquet writer
35  *
36  * @param sft simple feature type
37  * @param conf configuration
38  * @param output file to write
39  * @param observer any observers
40  */
41 class ParquetFileSystemWriter(
42     sft: SimpleFeatureType,
43     conf: Map[String, String],
44     output: FileOutput,
45     observer: FileSystemObserver
46   ) extends FileSystemWriter {
47 
48   import scala.collection.JavaConverters._
49 
50   // TODO consolidate this on FileIO instead of ObjectStore
51   def this(sft: SimpleFeatureType, conf: Map[String, String], fs: ObjectStore, file: URI, observer: FileSystemObserver) =
52     this(sft, conf, ObjectStoreOutput(fs, file), observer)
53 
54   def this(sft: SimpleFeatureType, conf: Map[String, String], fs: ObjectStore, file: URI) =
55     this(sft, conf, fs, file, NoOpObserver)
56 
57   def this(sft: SimpleFeatureType, conf: Map[String, String], io: FileIO, file: URI, observer: FileSystemObserver) =
58     this(sft, conf, IcebergOutput(io, file), observer)
59 
60   def this(sft: SimpleFeatureType, conf: Map[String, String], io: FileIO, file: URI) =
61     this(sft, conf, io, file, NoOpObserver)
62 
63   private val parquetConf = new PlainParquetConfiguration(conf.asJava)
64   SimpleFeatureParquetSchema.setSft(parquetConf, sft)
65 
66   private val writer = ParquetFileSystemWriter.builder(output.file, parquetConf).build()
67 
68   @volatile
69   private var closed = false
70 
71   override def size: Long = if (closed) { output.size }  else { writer.getDataSize }
72 
73   override def write(f: SimpleFeature): Unit = {
74     writer.write(f)
75     observer(f)
76   }
77 
78   override def flush(): Unit = observer.flush()
79 
80   override def close(): Unit = {
81     closed = true
82     CloseQuietly(Seq(writer, observer)).foreach(e => throw e)
83   }
84 }

86 object ParquetFileSystemWriter extends LazyLogging {
87 
88   /**
89    * Create a new configurable writer
90    *
91    * @param file file to write
92    * @param conf write configuration
93    * @return
94    */
95   def builder(file: OutputFile, conf: ParquetConfiguration): Builder = {
96     val codec = CompressionCodecName.fromConf(conf.get("parquet.compression", "ZSTD"))
97     logger.debug(s"Using Parquet Compression codec ${codec.name()}")
98     new Builder(file)
99       .withConf(conf)
100       .withCompressionCodec(codec)
101       .withDictionaryEncoding(true)
102       .withDictionaryPageSize(ParquetWriter.DEFAULT_PAGE_SIZE)
103       .withMaxPaddingSize(ParquetWriter.MAX_PADDING_SIZE_DEFAULT)
104       .withPageSize(ParquetWriter.DEFAULT_PAGE_SIZE)
105       .withValidation(false)
106       .withWriteMode(ParquetFileWriter.Mode.OVERWRITE)
107       .withWriterVersion(ParquetProperties.WriterVersion.PARQUET_2_0)
108       .withRowGroupSize(8L*1024*1024)
109   }
110 
111   private sealed trait FileOutput {
112 
113     /**
114      * The output file
115      *
116      * @return
117      */
118     def file: OutputFile
119 
120     /**
121      * Gets the size of the file
122      *
123      * @return
124      */
125     def size: Long
126   }
127 
128   private case class ObjectStoreOutput(fs: ObjectStore, path: URI) extends FileOutput {
129 
130     override val file: OutputFile = fs match {
131       case _: LocalObjectStore => new LocalOutputFileWithParent(Path.of(path))
132       case s3: S3ObjectStore => new S3OutputFile(s3, path)
133       case _ => throw new UnsupportedOperationException(s"No file implementation for scheme ${fs.scheme}")
134     }
135 
136     override def size: Long = fs.size(path)
137   }
138 
139   private case class IcebergOutput(io: FileIO, path: URI) extends FileOutput {
140     override val file: IcebergOutputFile = new IcebergOutputFile(io.newOutputFile(path.toString))
141     override def size: Long = file.original.toInputFile.getLength
142   }
143 
144   class Builder(file: OutputFile) extends ParquetWriter.Builder[SimpleFeature, Builder](file) {
145     override def self(): Builder = this
146     override protected def getWriteSupport(conf: Configuration): WriteSupport[SimpleFeature] =
147       new SimpleFeatureWriteSupport()
148     override protected def getWriteSupport(conf: ParquetConfiguration): WriteSupport[SimpleFeature] =
149       new SimpleFeatureWriteSupport()
150   }
151 
152   private class LocalOutputFileWithParent(file: Path) extends LocalOutputFile(file) {
153     override def create(blockSize: Long): PositionOutputStream = {
154       Option(file.toFile.getParentFile).foreach(_.mkdirs())
155       super.create(blockSize)
156     }
157     override def createOrOverwrite(blockSize: Long): PositionOutputStream = {
158       Option(file.toFile.getParentFile).foreach(_.mkdirs())
159       super.createOrOverwrite(blockSize)
160     }
161   }
