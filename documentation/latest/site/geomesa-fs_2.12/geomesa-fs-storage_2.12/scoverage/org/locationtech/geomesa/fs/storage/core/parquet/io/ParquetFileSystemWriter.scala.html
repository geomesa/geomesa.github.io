37 /**
38  * Parquet writer
39  *
40  * @param conf configuration
41  * @param output file to write
42  * @param observer any observers
43  */
44 class ParquetFileSystemWriter private (conf: ParquetConfiguration, output: FileOutput, observer: FileSystemObserver)
45     extends FileSystemWriter {
46 
47   @volatile
48   private var closed = false
49 
50   private val writer = ParquetFileSystemWriter.builder(output.file, conf).build()
51 
52   override def size: Long = if (closed) { output.size }  else { writer.getDataSize }
53 
54   override def write(f: SimpleFeature): Unit = {
55     writer.write(f)
56     observer(f)
57   }
58 
59   override def flush(): Unit = observer.flush()
60 
61   override def close(): Unit = {
62     closed = true
63     CloseQuietly(Seq(writer, observer)).foreach(e => throw e)
64   }
65 }

67 object ParquetFileSystemWriter extends LazyLogging {
68 
69   import scala.collection.JavaConverters._
70 
71   /**
72    * Primary iceberg constructor - when writing to iceberg, needs to use this constructor in order to ensure field ids align
73    *
74    * @param schema iceberg schema
75    * @param conf conf
76    * @param io file io
77    * @param file file to write
78    * @param observer observer
79    */
80   def apply(
81       schema: SimpleFeatureIcebergSchema,
82       conf: Map[String, String],
83       io: FileIO,
84       file: String,
85       observer: FileSystemObserver): ParquetFileSystemWriter = {
86     // stamp the written parquet files with the table's iceberg field ids (by name) so reads resolve by id
87     val nameMapping = Map(SimpleFeatureSchema.IcebergNameMappingKey -> NameMappingParser.toJson(MappingUtil.create(schema.schema)))
88     val sft = SimpleFeatureParquetSchema.sftConf(schema.sft)
89     apply(conf ++ nameMapping ++ sft, IcebergOutput(io, file), observer)
90   }
91 
92   /**
93    * Secondary constructor - for non-iceberg exports. Note: *will not* align with Iceberg field ids
94    *
95    * TODO consolidate this on FileIO instead of ObjectStore
96    *
97    * @param sft simple feature type
98    * @param conf configuration options
99    * @param fs object store
100    * @param file output file path
101    * @return
102    */
103   def apply(sft: SimpleFeatureType, conf: Map[String, String], fs: ObjectStore, file: String): ParquetFileSystemWriter = {
104     val sftConf = SimpleFeatureParquetSchema.sftConf(sft)
105     apply(conf ++ sftConf, ObjectStoreOutput(fs, URI.create(file)), NoOpObserver)
106   }
107 
108   /**
109    * Constructor - requires the sft to be encoded in the conf
110    *
111    * @param conf configuration options
112    * @param output file output
113    * @param observer observer
114    * @return
115    */
116   private def apply(conf: Map[String, String], output: FileOutput, observer: FileSystemObserver): ParquetFileSystemWriter = {
117     // system properties provide defaults, which the storage configuration overrides
118     val sysProps = Seq(ParquetCompressionOpt, ParquetRowGroupSizeOpt).flatMap(k => Option(System.getProperty(k)).map(k -> _)).toMap
119     val parquetConf = new PlainParquetConfiguration((sysProps ++ conf).asJava)
120     new ParquetFileSystemWriter(parquetConf, output, observer)
121   }
122 
123   /**
124    * Create a new configurable writer
125    *
126    * @param file file to write
127    * @param conf write configuration
128    * @return
129    */
130   private def builder(file: OutputFile, conf: ParquetConfiguration): Builder = {
131     val version = WriterVersion.fromString(conf.get("parquet.writer.version", WriterVersion.PARQUET_2_0.name()))
132     val codec = CompressionCodecName.fromConf(conf.get("parquet.compression", "ZSTD"))
133     val rowGroupSize = this.rowGroupSize(conf)
134     logger.debug(s"Using Parquet file version $version with compression ${codec.name()} and row group size $rowGroupSize")
135 
136     new Builder(file)
137       .withConf(conf)
138       .withCompressionCodec(codec)
139       .withWriteMode(ParquetFileWriter.Mode.OVERWRITE)
140       .withWriterVersion(version)
141       .withRowGroupSize(rowGroupSize)
142   }
143 
144   /**
145    * The configured row group size, or parquet's default (`ParquetWriter.DEFAULT_BLOCK_SIZE`) if absent or invalid
146    *
147    * @param conf write configuration
148    * @return row group size, in bytes
149    */
150   private[io] def rowGroupSize(conf: ParquetConfiguration): Long = {
151     Option(conf.get(ParquetRowGroupSizeOpt)).map(_.trim).filter(_.nonEmpty) match {
152       case None => ParquetWriter.DEFAULT_BLOCK_SIZE
153       case Some(size) =>
154         Suffixes.Memory.bytes(size).getOrElse {
155           logger.warn(s"Invalid $ParquetRowGroupSizeOpt '$size', using the default of ${ParquetWriter.DEFAULT_BLOCK_SIZE} bytes")
156           ParquetWriter.DEFAULT_BLOCK_SIZE
157         }
158     }
159   }

178   private case class ObjectStoreOutput(fs: ObjectStore, path: URI) extends FileOutput {
179 
180     override val file: OutputFile = fs match {
181       case _: LocalObjectStore => new LocalOutputFileWithParent(Path.of(path))
182       case s3: S3ObjectStore => new S3OutputFile(s3, path)
183       case _ => throw new UnsupportedOperationException(s"No file implementation for scheme ${fs.scheme}")
184     }
185 
186     override def size: Long = fs.size(path)
187   }
188 
189   private case class IcebergOutput(io: FileIO, path: String) extends FileOutput {
190     override val file: IcebergOutputFile = new IcebergOutputFile(io.newOutputFile(path))
191     override def size: Long = file.original.toInputFile.getLength
192   }
193 
194   class Builder(file: OutputFile) extends ParquetWriter.Builder[SimpleFeature, Builder](file) {
195     override def self(): Builder = this
196     override protected def getWriteSupport(conf: Configuration): WriteSupport[SimpleFeature] =
197       new SimpleFeatureWriteSupport()
198     override protected def getWriteSupport(conf: ParquetConfiguration): WriteSupport[SimpleFeature] =
199       new SimpleFeatureWriteSupport()
200   }
201 
202   private class LocalOutputFileWithParent(file: Path) extends LocalOutputFile(file) {
203     override def create(blockSize: Long): PositionOutputStream = {
204       Option(file.toFile.getParentFile).foreach(_.mkdirs())
205       super.create(blockSize)
206     }
207     override def createOrOverwrite(blockSize: Long): PositionOutputStream = {
208       Option(file.toFile.getParentFile).foreach(_.mkdirs())
209       super.createOrOverwrite(blockSize)
210     }
211   }
