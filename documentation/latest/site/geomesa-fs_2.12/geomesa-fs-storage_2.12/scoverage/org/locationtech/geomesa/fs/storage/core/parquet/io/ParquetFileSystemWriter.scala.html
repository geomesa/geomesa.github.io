36 /**
37  * Parquet writer
38  *
39  * @param conf configuration
40  * @param output file to write
41  * @param observer any observers
42  */
43 class ParquetFileSystemWriter private (conf: ParquetConfiguration, output: FileOutput, observer: FileSystemObserver)
44     extends FileSystemWriter {
45 
46   @volatile
47   private var closed = false
48 
49   private val writer = ParquetFileSystemWriter.builder(output.file, conf).build()
50 
51   override def size: Long = if (closed) { output.size }  else { writer.getDataSize }
52 
53   override def write(f: SimpleFeature): Unit = {
54     writer.write(f)
55     observer(f)
56   }
57 
58   override def flush(): Unit = observer.flush()
59 
60   override def close(): Unit = {
61     closed = true
62     CloseQuietly(Seq(writer, observer)).foreach(e => throw e)
63   }
64 }

66 object ParquetFileSystemWriter extends LazyLogging {
67 
68   import scala.collection.JavaConverters._
69 
70   /**
71    * Primary iceberg constructor - when writing to iceberg, needs to use this constructor in order to ensure field ids align
72    *
73    * @param schema iceberg schema
74    * @param conf conf
75    * @param io file io
76    * @param file file to write
77    * @param observer observer
78    */
79   def apply(
80       schema: SimpleFeatureIcebergSchema,
81       conf: Map[String, String],
82       io: FileIO,
83       file: String,
84       observer: FileSystemObserver): ParquetFileSystemWriter = {
85     // stamp the written parquet files with the table's iceberg field ids (by name) so reads resolve by id
86     val nameMapping = Map(SimpleFeatureSchema.IcebergNameMappingKey -> NameMappingParser.toJson(MappingUtil.create(schema.schema)))
87     val sft = SimpleFeatureParquetSchema.sftConf(schema.sft)
88     apply(conf ++ nameMapping ++ sft, IcebergOutput(io, file), observer)
89   }
90 
91   /**
92    * Secondary constructor - for non-iceberg exports. Note: *will not* align with Iceberg field ids
93    *
94    * TODO consolidate this on FileIO instead of ObjectStore
95    *
96    * @param sft simple feature type
97    * @param conf configuration options
98    * @param fs object store
99    * @param file output file path
100    * @return
101    */
102   def apply(sft: SimpleFeatureType, conf: Map[String, String], fs: ObjectStore, file: String): ParquetFileSystemWriter = {
103     val sftConf = SimpleFeatureParquetSchema.sftConf(sft)
104     apply(conf ++ sftConf, ObjectStoreOutput(fs, URI.create(file)), NoOpObserver)
105   }
106 
107   /**
108    * Constructor - requires the sft to be encoded in the conf
109    *
110    * @param conf configuration options
111    * @param output file output
112    * @param observer observer
113    * @return
114    */
115   private def apply(conf: Map[String, String], output: FileOutput, observer: FileSystemObserver): ParquetFileSystemWriter = {
116     val compression = Option(System.getProperty(ParquetCompressionOpt)).map(ParquetCompressionOpt -> _).toMap
117     val parquetConf = new PlainParquetConfiguration((compression ++ conf).asJava)
118     new ParquetFileSystemWriter(parquetConf, output, observer)
119   }
120 
121   /**
122    * Create a new configurable writer
123    *
124    * @param file file to write
125    * @param conf write configuration
126    * @return
127    */
128   private def builder(file: OutputFile, conf: ParquetConfiguration): Builder = {
129     val version = WriterVersion.fromString(conf.get("parquet.writer.version", WriterVersion.PARQUET_2_0.name()))
130     val codec = CompressionCodecName.fromConf(conf.get("parquet.compression", "ZSTD"))
131     logger.debug(s"Using Parquet file version $version with compression ${codec.name()}")
132 
133     new Builder(file)
134       .withConf(conf)
135       .withCompressionCodec(codec)
136       .withDictionaryEncoding(true)
137       .withDictionaryPageSize(ParquetWriter.DEFAULT_PAGE_SIZE)
138       .withMaxPaddingSize(ParquetWriter.MAX_PADDING_SIZE_DEFAULT)
139       .withPageSize(ParquetWriter.DEFAULT_PAGE_SIZE)
140       .withValidation(false)
141       .withWriteMode(ParquetFileWriter.Mode.OVERWRITE)
142       .withWriterVersion(version)
143       .withRowGroupSize(8L*1024*1024)
144   }

163   private case class ObjectStoreOutput(fs: ObjectStore, path: URI) extends FileOutput {
164 
165     override val file: OutputFile = fs match {
166       case _: LocalObjectStore => new LocalOutputFileWithParent(Path.of(path))
167       case s3: S3ObjectStore => new S3OutputFile(s3, path)
168       case _ => throw new UnsupportedOperationException(s"No file implementation for scheme ${fs.scheme}")
169     }
170 
171     override def size: Long = fs.size(path)
172   }
173 
174   private case class IcebergOutput(io: FileIO, path: String) extends FileOutput {
175     override val file: IcebergOutputFile = new IcebergOutputFile(io.newOutputFile(path))
176     override def size: Long = file.original.toInputFile.getLength
177   }
178 
179   class Builder(file: OutputFile) extends ParquetWriter.Builder[SimpleFeature, Builder](file) {
180     override def self(): Builder = this
181     override protected def getWriteSupport(conf: Configuration): WriteSupport[SimpleFeature] =
182       new SimpleFeatureWriteSupport()
183     override protected def getWriteSupport(conf: ParquetConfiguration): WriteSupport[SimpleFeature] =
184       new SimpleFeatureWriteSupport()
185   }
186 
187   private class LocalOutputFileWithParent(file: Path) extends LocalOutputFile(file) {
188     override def create(blockSize: Long): PositionOutputStream = {
189       Option(file.toFile.getParentFile).foreach(_.mkdirs())
190       super.create(blockSize)
191     }
192     override def createOrOverwrite(blockSize: Long): PositionOutputStream = {
193       Option(file.toFile.getParentFile).foreach(_.mkdirs())
194       super.createOrOverwrite(blockSize)
195     }
196   }
