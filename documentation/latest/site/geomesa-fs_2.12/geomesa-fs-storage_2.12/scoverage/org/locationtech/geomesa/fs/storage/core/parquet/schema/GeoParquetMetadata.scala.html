79     /**
80      * Gets the geometry object type based on the GeoParquet column types
81      *
82      * @param colTypes column types from GeoParquet metadata
83      * @return
84      */
85     def toObjectType(colTypes: Seq[GeoParquetColumnType]): ObjectType = {
86       if (colTypes.lengthCompare(1) != 0) {
87         ObjectType.GEOMETRY // more than 1 geometry type
88       } else if (colTypes.head == GeometryCollection || colTypes.head == `GeometryCollection Z` ) {
89         ObjectType.GEOMETRY_COLLECTION // this is the one objectType whose name doesn't align with the columnType
90       } else {
91         ObjectType.withName(colTypes.head.toString.replace(" Z", "").toUpperCase(Locale.US))
92       }
93     }
94   }

96   /**
97    * Generate json describing the GeoParquet file, conforming to the 1.1.0 GeoParquet schema
98    *
99    * @param metadata metadata
100    * @return
101    */
102   def toJson(metadata: GeoParquetMetadata): String = {
103     val model = new JsonObject()
104     model.addProperty("version", "1.1.0")
105     model.addProperty("primary_column", metadata.primaryGeometry)
106     val cols = model.addObject("columns")
107     metadata.geometries.foreach { column =>
108       val metadata = cols.addObject(column.name)
109       metadata.addProperty("encoding", column.encoding.toString)
110       val types = metadata.addArray("geometry_types", 1)
111       column.types.foreach(t => types.add(t.toString))
112       if (!column.bounds.isNull) {
113         val bbox = metadata.addArray("bbox", 4)
114         bbox.add(column.bounds.getMinX)
115         bbox.add(column.bounds.getMinY)
116         bbox.add(column.bounds.getMaxX)
117         bbox.add(column.bounds.getMaxY)
118       }
119       column.covering.foreach { c =>
120         val bbox = metadata.addObject("covering").addObject("bbox")
121         Seq("xmin", "ymin", "xmax", "ymax").foreach { name =>
122           val bound = bbox.addArray(name, 2)
123           bound.add(c)
124           bound.add(name)
125         }
126       }
127     }
128 
129     gson.toJson(model)
130   }

132   /**
133    * Parse a GeoParquet json into a model object
134    *
135    * @param json json string
136    * @return
137    */
138   def fromJson(json: String): GeoParquetMetadata = {
139     val obj = gson.fromJson(json, classOf[JsonObject])
140     val primary = obj.get("primary_column").getAsString
141     val cols = obj.getAsJsonObject("columns").entrySet().asScala.map { entry =>
142       val name = entry.getKey
143       val obj = entry.getValue.getAsJsonObject
144       // required entries
145       val encoding = GeoParquetColumnEncoding.withName(obj.get("encoding").getAsString)
146       val types = obj.getAsJsonArray("geometry_types").asList().asScala.map(t => GeoParquetColumnType.withName(t.getAsString))
147       // optional entries
148       // note: all covering fields *must* refer to the same column, so we only pull out one
149       val covering =
150         Option(obj.getAsJsonObject("covering")).map(_.getAsJsonObject("bbox").getAsJsonArray("xmin").get(0).getAsString)
151       val bounds = new Envelope()
152       Option(obj.getAsJsonArray("bbox")).foreach { bbox =>
153         bounds.expandToInclude(bbox.get(0).getAsDouble, bbox.get(1).getAsDouble)
154         bounds.expandToInclude(bbox.get(2).getAsDouble, bbox.get(3).getAsDouble)
155       }
156       // TODO crs, orientation, edges, epoch
157       ColumnMetadata(name, encoding, types.toSeq, covering, bounds)
158     }
159     GeoParquetMetadata(primary, cols.toSeq)
160   }

180   /**
181    * Observer class that generates GeoParquet metadata
182    *
183    * @param schema simple feature schema
184    */
185   class GeoParquetObserver(schema: SimpleFeatureParquetSchema) extends FileSystemObserver {
186 
187     import org.locationtech.geomesa.utils.geotools.RichAttributeDescriptors.RichAttributeDescriptor
188 
189     import scala.collection.JavaConverters._
190 
191     private val columns =
192       schema.sft.getAttributeDescriptors.asScala.toSeq.collect { case d: GeometryDescriptor =>
193         val col = ColumnName(d.getLocalName)
194         val binding = d.getType.getBinding
195         val encoding = d.getEncoding.map(GeometryEncoding.apply) match {
196           // for non-wkb encoding schemes, we still use WKB for mixed-type geometries
197           case Some(GeoParquetNative) if binding != classOf[Geometry] && binding != classOf[GeometryCollection] =>
198             GeoParquetColumnEncoding.withName(binding.getSimpleName.toLowerCase(Locale.US))
199           case _ => GeoParquetColumnEncoding.WKB
200         }
201         // TODO add z for 3d points
202         val types = if (binding == classOf[Geometry]) { Seq.empty } else { Seq(GeoParquetColumnType.withName(binding.getSimpleName)) }
203         val covering = Some(BoundingBoxField.groupName(col.column)).filter(schema.messageType.containsField)
204         schema.sft.indexOf(col.attribute) -> ColumnMetadata(col.column, encoding, types, covering)
205       }
206 
207     /**
208      * Gets the file metadata for geoparquet
209      *
210      * @return
211      */
212     def metadata(): java.util.Map[String, String] = {
213       if (columns.isEmpty) { Collections.emptyMap() } else {
214         val cols = columns.map(_._2)
215         val primary = {
216           // there's a chance that the primary geom wasn't encoded in the file, but other geoms were
217           val expected = ColumnName(schema.sft.getGeometryDescriptor.getLocalName).column
218           if (cols.exists(_.name == expected)) { expected } else { cols.head.name }
219         }
220         Collections.singletonMap(GeoParquetMetadataKey, GeoParquetMetadata(primary, cols).toJson())
221       }
222     }
223 
224     override def apply(feature: SimpleFeature): Unit = {
225       columns.foreach { case (i, meta) =>
226         val geom = feature.getAttribute(i).asInstanceOf[Geometry]
227         if (geom != null) {
228           meta.bounds.expandToInclude(geom.getEnvelopeInternal)
229         }
230       }
231     }
232 
233     override def flush(): Unit = {}
234     override def close(): Unit = {}
235   }
