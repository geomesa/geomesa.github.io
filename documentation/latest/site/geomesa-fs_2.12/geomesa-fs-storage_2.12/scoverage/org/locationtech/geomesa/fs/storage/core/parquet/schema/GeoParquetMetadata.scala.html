78     /**
79      * Gets the geometry object type based on the GeoParquet column types
80      *
81      * @param colTypes column types from GeoParquet metadata
82      * @return
83      */
84     def toObjectType(colTypes: Seq[GeoParquetColumnType]): ObjectType = {
85       if (colTypes.lengthCompare(1) != 0) {
86         ObjectType.GEOMETRY // more than 1 geometry type
87       } else if (colTypes.head == GeometryCollection || colTypes.head == `GeometryCollection Z` ) {
88         ObjectType.GEOMETRY_COLLECTION // this is the one objectType whose name doesn't align with the columnType
89       } else {
90         ObjectType.withName(colTypes.head.toString.replace(" Z", "").toUpperCase(Locale.US))
91       }
92     }
93   }

95   /**
96    * Generate json describing the GeoParquet file, conforming to the 1.1.0 GeoParquet schema
97    *
98    * @param metadata metadata
99    * @return
100    */
101   def toJson(metadata: GeoParquetMetadata): String = {
102     val model = new JsonObject()
103     model.addProperty("version", "1.1.0")
104     model.addProperty("primary_column", metadata.primaryGeometry)
105     val cols = model.addObject("columns")
106     metadata.geometries.foreach { column =>
107       val metadata = cols.addObject(column.name)
108       metadata.addProperty("encoding", column.encoding.toString)
109       val types = metadata.addArray("geometry_types", 1)
110       column.types.foreach(t => types.add(t.toString))
111       if (!column.bounds.isNull) {
112         val bbox = metadata.addArray("bbox", 4)
113         bbox.add(column.bounds.getMinX)
114         bbox.add(column.bounds.getMinY)
115         bbox.add(column.bounds.getMaxX)
116         bbox.add(column.bounds.getMaxY)
117       }
118       column.covering.foreach { c =>
119         val bbox = metadata.addObject("covering").addObject("bbox")
120         Seq("xmin", "ymin", "xmax", "ymax").foreach { name =>
121           val bound = bbox.addArray(name, 2)
122           bound.add(c)
123           bound.add(name)
124         }
125       }
126     }
127 
128     gson.toJson(model)
129   }

131   /**
132    * Parse a GeoParquet json into a model object
133    *
134    * @param json json string
135    * @return
136    */
137   def fromJson(json: String): GeoParquetMetadata = {
138     val obj = gson.fromJson(json, classOf[JsonObject])
139     val primary = obj.get("primary_column").getAsString
140     val cols = obj.getAsJsonObject("columns").entrySet().asScala.map { entry =>
141       val name = entry.getKey
142       val obj = entry.getValue.getAsJsonObject
143       // required entries
144       val encoding = GeoParquetColumnEncoding.withName(obj.get("encoding").getAsString)
145       val types = obj.getAsJsonArray("geometry_types").asList().asScala.map(t => GeoParquetColumnType.withName(t.getAsString))
146       // optional entries
147       // note: all covering fields *must* refer to the same column, so we only pull out one
148       val covering =
149         Option(obj.getAsJsonObject("covering")).map(_.getAsJsonObject("bbox").getAsJsonArray("xmin").get(0).getAsString)
150       val bounds = new Envelope()
151       Option(obj.getAsJsonArray("bbox")).foreach { bbox =>
152         bounds.expandToInclude(bbox.get(0).getAsDouble, bbox.get(1).getAsDouble)
153         bounds.expandToInclude(bbox.get(2).getAsDouble, bbox.get(3).getAsDouble)
154       }
155       // TODO crs, orientation, edges, epoch
156       ColumnMetadata(name, encoding, types.toSeq, covering, bounds)
157     }
158     GeoParquetMetadata(primary, cols.toSeq)
159   }

179   /**
180    * Observer class that generates GeoParquet metadata
181    *
182    * @param schema simple feature schema
183    */
184   class GeoParquetObserver(schema: SimpleFeatureParquetSchema) extends FileSystemObserver {
185 
186     import scala.collection.JavaConverters._
187 
188     private val columns =
189       schema.sft.getAttributeDescriptors.asScala.toSeq.collect { case d: GeometryDescriptor =>
190         val col = ColumnName(d.getLocalName)
191         val binding = d.getType.getBinding
192         val encoding = {
193           // for non-wkb encoding schemes, we still use WKB for mixed-type geometries
194           if (schema.geometries == GeometryEncoding.GeoParquetWkb ||
195             binding == classOf[Geometry] || binding == classOf[GeometryCollection]) {
196             GeoParquetColumnEncoding.WKB
197           } else {
198             GeoParquetColumnEncoding.withName(binding.getSimpleName.toLowerCase(Locale.US))
199           }
200         }
201         // TODO add z for 3d points
202         val types = if (binding == classOf[Geometry]) { Seq.empty } else { Seq(GeoParquetColumnType.withName(binding.getSimpleName)) }
203         val covering = Some(BoundingBoxField.groupName(col.column)).filter(schema.messageType.containsField)
204         schema.sft.indexOf(col.attribute) -> ColumnMetadata(col.column, encoding, types, covering)
205       }
206 
207     /**
208      * Gets the file metadata for geoparquet
209      *
210      * @return
211      */
212     def metadata(): java.util.Map[String, String] = {
213       if (columns.isEmpty) { Collections.emptyMap() } else {
214         val cols = columns.map(_._2)
215         val primary = {
216           // there's a chance that the primary geom wasn't encoded in the file, but other geoms were
217           val expected = ColumnName(schema.sft.getGeometryDescriptor.getLocalName).column
218           if (cols.exists(_.name == expected)) { expected } else { cols.head.name }
219         }
220         Collections.singletonMap(GeoParquetMetadataKey, GeoParquetMetadata(primary, cols).toJson())
221       }
222     }
223 
224     override def apply(feature: SimpleFeature): Unit = {
225       columns.foreach { case (i, meta) =>
226         val geom = feature.getAttribute(i).asInstanceOf[Geometry]
227         if (geom != null) {
228           meta.bounds.expandToInclude(geom.getEnvelopeInternal)
229         }
230       }
231     }
232 
233     override def flush(): Unit = {}
234     override def close(): Unit = {}
235   }
