88     object PartitionSerializer extends JsonSerializer[Partition] with JsonDeserializer[Partition] {
89 
90       override def serialize(src: Partition, typeOfSrc: Type, context: JsonSerializationContext): JsonElement = {
91         val array = new JsonArray(src.values.size)
92         src.values.foreach { value =>
93           array.add(context.serialize(value))
94         }
95         array

98       override def deserialize(json: JsonElement, typeOfT: Type, context: JsonDeserializationContext): Partition = {
99         val array = json.getAsJsonArray
100         val values = Seq.newBuilder[PartitionKey]
101         var i = 0
102         while (i < array.size()) {
103           values += context.deserialize(array.get(i), classOf[PartitionKey])
104           i += 1
105         }
106         Partition(values.result())
107       }
108     }

173     def setScheme(names: String): Unit = sft.getUserData.put(SchemeKey, names)
174     def removeScheme(): Option[Seq[String]] = {
175       remove(SchemeKey).map { scheme =>
176         // back compatible check for old json-serialized schemes
177         if (scheme.trim.startsWith("{")) {
178           try {
179             def result(name: String, options: Map[String, String]): Seq[String] = {
180               val opts = options.map { case (k, v) => s"$k=$v" }.mkString(":")
181               name.split(",").toSeq.map(n => s"$n:$opts")
182             }
183             val source = ConfigSource.fromConfig(ConfigFactory.parseString(scheme))
184             source.load[SchemeOptions] match {
185               case Right(o) => result(o.scheme, o.options)
186               case Left(_) =>
187                 val n = source.loadOrThrow[NamedOptions]
188                 result(n.name, n.options)
189             }
190           } catch {
191             case NonFatal(e) => throw new RuntimeException(s"Could not parse legacy scheme options: $scheme", e)
192           }
193         } else {
194           scheme.split(",").toSeq
195         }
196       }
197     }
198 
199     def setTargetFileSize(size: String): Unit = {
200       // validate input
201       Memory.bytes(size).failed.foreach(e => throw new IllegalArgumentException("Invalid file size", e))
202       sft.getUserData.put(FileSizeKey, size)
203     }
204     def removeTargetFileSize(): Option[Long] = {
205       remove(FileSizeKey).map { s =>
206         Memory.bytes(s) match {
207           case Success(b) => b
208           case Failure(e) => throw new IllegalArgumentException("Invalid file size", e)
209         }
210       }
211     }

217     }
