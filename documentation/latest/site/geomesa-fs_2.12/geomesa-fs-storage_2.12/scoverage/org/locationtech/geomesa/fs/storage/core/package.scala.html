97      * Json serializer for partitions

99     object PartitionSerializer extends JsonSerializer[Partition] with JsonDeserializer[Partition] {
100 
101       override def serialize(src: Partition, typeOfSrc: Type, context: JsonSerializationContext): JsonElement = {
102         val array = new JsonArray(src.values.size)
103         src.values.foreach { value =>
104           array.add(context.serialize(value))
105         }
106         array
107       }
108 
109       override def deserialize(json: JsonElement, typeOfT: Type, context: JsonDeserializationContext): Partition = {
110         val array = json.getAsJsonArray
111         val values = Seq.newBuilder[PartitionKey]
112         var i = 0
113         while (i < array.size()) {
114           values += context.deserialize(array.get(i), classOf[PartitionKey])
115           i += 1
116         }
117         Partition(values.result())
118       }
119     }

184     def setScheme(names: String): Unit = sft.getUserData.put(SchemeKey, names)
185     def removeScheme(): Option[Seq[String]] = {
186       remove(SchemeKey).map { scheme =>
187         // back compatible check for old json-serialized schemes
188         if (scheme.trim.startsWith("{")) {
189           try {
190             def result(name: String, options: Map[String, String]): Seq[String] = {
191               val opts = options.map { case (k, v) => s"$k=$v" }.mkString(":")
192               name.split(",").toSeq.map(n => s"$n:$opts")
193             }
194             val source = ConfigSource.fromConfig(ConfigFactory.parseString(scheme))
195             source.load[SchemeOptions] match {
196               case Right(o) => result(o.scheme, o.options)
197               case Left(_) =>
198                 val n = source.loadOrThrow[NamedOptions]
199                 result(n.name, n.options)
200             }
201           } catch {
202             case NonFatal(e) => throw new RuntimeException(s"Could not parse legacy scheme options: $scheme", e)
203           }
204         } else {
205           scheme.split(",").toSeq
206         }
207       }
208     }
209 
210     def setTargetFileSize(size: String): Unit = {
211       // validate input
212       Memory.bytes(size).failed.foreach(e => throw new IllegalArgumentException("Invalid file size", e))
213       sft.getUserData.put(FileSizeKey, size)
214     }
215     def removeTargetFileSize(): Option[Long] = {
216       remove(FileSizeKey).map { s =>
217         Memory.bytes(s) match {
218           case Success(b) => b
219           case Failure(e) => throw new IllegalArgumentException("Invalid file size", e)
220         }
221       }
222     }
223 
224     def setObservers(names: Seq[String]): Unit = sft.getUserData.put(ObserversKey, names.mkString(","))
225     def getObservers: Seq[String] = {
226       val obs = sft.getUserData.get(ObserversKey).asInstanceOf[String]
227       if (obs == null || obs.isEmpty) { Seq.empty } else { obs.split(",") }
228     }
229 
230     private def remove(key: String): Option[String] = Option(sft.getUserData.remove(key).asInstanceOf[String])
231   }
