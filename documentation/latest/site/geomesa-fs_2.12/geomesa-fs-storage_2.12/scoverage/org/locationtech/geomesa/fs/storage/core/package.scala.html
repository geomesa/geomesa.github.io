75      * Helper method to ensure the root is property formatted

77      * @param root root path
78      * @param conf configuration
79      * @param namespace namespace
80      * @return

82     def create(root: URI, conf: Map[String, String], namespace: Option[String]): FileSystemContext = {
83       val validatedRoot = {
84         val rootWithScheme = if (root.getScheme != null && root.getScheme.nonEmpty) { root } else {
85           new URI("file", root.getAuthority, root.getPath, root.getQuery, root.getFragment)
86         }
87         val rootWithSlash = if (rootWithScheme.toString.endsWith("/")) { rootWithScheme } else { new URI(rootWithScheme.toString + "/") }
88         rootWithSlash
89       }
90       FileSystemContext(validatedRoot, S3ObjectStore.s3Configs(conf), namespace)
91     }

145     object PartitionSerializer extends JsonSerializer[Partition] with JsonDeserializer[Partition] {
146 
147       override def serialize(src: Partition, typeOfSrc: Type, context: JsonSerializationContext): JsonElement = {
148         val array = new JsonArray(src.values.size)
149         src.values.foreach { value =>
150           array.add(context.serialize(value))
151         }
152         array
153       }

155       override def deserialize(json: JsonElement, typeOfT: Type, context: JsonDeserializationContext): Partition = {
156         val array = json.getAsJsonArray
157         val values = Seq.newBuilder[PartitionKey]
158         var i = 0
159         while (i < array.size()) {
160           values += context.deserialize(array.get(i), classOf[PartitionKey])
161           i += 1
162         }
163         Partition(values.result())
164       }
165     }

238     def removeScheme(): Option[Seq[String]] = {
239       remove(SchemeKey).map { scheme =>
240         // back compatible check for old json-serialized schemes
241         if (scheme.trim.startsWith("{")) {
242           try {
243             def result(name: String, options: Map[String, String]): Seq[String] = {
244               val opts = options.map { case (k, v) => s"$k=$v" }.mkString(":")
245               name.split(",").toSeq.map(n => s"$n:$opts")
246             }
247             val source = ConfigSource.fromConfig(ConfigFactory.parseString(scheme))
248             source.load[SchemeOptions] match {
249               case Right(o) => result(o.scheme, o.options)
250               case Left(_) =>
251                 val n = source.loadOrThrow[NamedOptions]
252                 result(n.name, n.options)
253             }
254           } catch {
255             case NonFatal(e) => throw new RuntimeException(s"Could not parse legacy scheme options: $scheme", e)
256           }
257         } else {
258           scheme.split(",").toSeq
259         }
260       }
261     }
