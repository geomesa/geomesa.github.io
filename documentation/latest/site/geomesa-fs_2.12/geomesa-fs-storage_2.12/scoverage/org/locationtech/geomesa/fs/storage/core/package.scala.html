105         val array = json.getAsJsonArray
106         val values = Seq.newBuilder[PartitionKey]

109           values += context.deserialize(array.get(i), classOf[PartitionKey])
110           i += 1
111         }
112         Partition(values.result())
113       }

180     def setScheme(names: String): Unit = sft.getUserData.put(SchemeKey, names)
181     def removeScheme(): Option[Seq[String]] = {
182       remove(SchemeKey).map { scheme =>
183         // back compatible check for old json-serialized schemes
184         if (scheme.trim.startsWith("{")) {
185           try {
186             def result(name: String, options: Map[String, String]): Seq[String] = {
187               val opts = options.map { case (k, v) => s"$k=$v" }.mkString(":")
188               name.split(",").toSeq.map(n => s"$n:$opts")
189             }
190             val source = ConfigSource.fromConfig(ConfigFactory.parseString(scheme))
191             source.load[SchemeOptions] match {
192               case Right(o) => result(o.scheme, o.options)
193               case Left(_) =>
194                 val n = source.loadOrThrow[NamedOptions]
195                 result(n.name, n.options)
196             }
197           } catch {
198             case NonFatal(e) => throw new RuntimeException(s"Could not parse legacy scheme options: $scheme", e)
199           }
200         } else {
201           scheme.split(",").toSeq
202         }
203       }
204     }

207       // validate input

209       sft.getUserData.put(FileSizeKey, size)

211     def removeTargetFileSize(): Option[Long] = {
