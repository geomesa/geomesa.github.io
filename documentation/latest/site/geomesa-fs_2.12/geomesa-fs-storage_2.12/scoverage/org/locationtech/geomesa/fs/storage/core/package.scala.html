98     /**
99      * Json serializer for partitions
100      */
101     object PartitionSerializer extends JsonSerializer[Partition] with JsonDeserializer[Partition] {
102 
103       override def serialize(src: Partition, typeOfSrc: Type, context: JsonSerializationContext): JsonElement = {
104         val array = new JsonArray(src.values.size)
105         src.values.foreach { value =>
106           array.add(context.serialize(value))
107         }
108         array
109       }
110 
111       override def deserialize(json: JsonElement, typeOfT: Type, context: JsonDeserializationContext): Partition = {
112         val array = json.getAsJsonArray
113         val values = Seq.newBuilder[PartitionKey]
114         var i = 0
115         while (i < array.size()) {
116           values += context.deserialize(array.get(i), classOf[PartitionKey])
117           i += 1
118         }
119         Partition(values.result())
120       }
121     }
122   }

190     def setScheme(names: String): Unit = sft.getUserData.put(SchemeKey, names)
191     def removeScheme(): Option[Seq[String]] = {
192       remove(SchemeKey).map { scheme =>
193         // back compatible check for old json-serialized schemes
194         if (scheme.trim.startsWith("{")) {
195           try {
196             def result(name: String, options: Map[String, String]): Seq[String] = {
197               val opts = options.map { case (k, v) => s"$k=$v" }.mkString(":")
198               name.split(",").toSeq.map(n => s"$n:$opts")
199             }
200             val source = ConfigSource.fromConfig(ConfigFactory.parseString(scheme))
201             source.load[SchemeOptions] match {
202               case Right(o) => result(o.scheme, o.options)
203               case Left(_) =>
204                 val n = source.loadOrThrow[NamedOptions]
205                 result(n.name, n.options)
206             }
207           } catch {
208             case NonFatal(e) => throw new RuntimeException(s"Could not parse legacy scheme options: $scheme", e)
209           }
210         } else {
211           scheme.split(",").toSeq
212         }
213       }
214     }
215 
216     def setTargetFileSize(size: String): Unit = {
217       // validate input
218       Memory.bytes(size).failed.foreach(e => throw new IllegalArgumentException("Invalid file size", e))
219       sft.getUserData.put(FileSizeKey, size)
220     }
221     def removeTargetFileSize(): Option[Long] = {
222       remove(FileSizeKey).map { s =>
223         Memory.bytes(s) match {
224           case Success(b) => b
225           case Failure(e) => throw new IllegalArgumentException("Invalid file size", e)
226         }
227       }
228     }

234     }
