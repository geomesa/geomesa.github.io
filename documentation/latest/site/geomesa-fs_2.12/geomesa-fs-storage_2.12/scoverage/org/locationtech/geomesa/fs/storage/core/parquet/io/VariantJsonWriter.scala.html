21 /**
22  * Utility object to serialize [[Variant]] values to JSON using Gson.
23  */
24 object VariantJsonWriter {
25 
26   /**
27    * Converts a Variant to a compact JSON string.
28    *
29    * @param variant the variant to serialize
30    * @return compact JSON string
31    */
32   def toJson(variant: Variant): String = {
33     val sw = new StringWriter()
34     WithClose(new JsonWriter(sw)) { writer =>
35       toJson(writer, variant)
36     }
37     sw.toString
38   }
39 
40   /**
41    * Writes a Variant to a JsonWriter.
42    */
43   def toJson(writer: JsonWriter, variant: Variant): Unit = {
44     val variantType = variant.getType
45 
46     variantType match {
47       case Type.NULL =>
48         writer.nullValue()
49 
50       case Type.BOOLEAN =>
51         writer.value(variant.getBoolean)
52 
53       case Type.BYTE | Type.SHORT | Type.INT | Type.LONG =>
54         writer.value(variant.getLong)
55 
56       case Type.FLOAT =>
57         val f = variant.getFloat
58         if (f.isNaN || f.isInfinite) {
59           writer.nullValue()  // JSON doesn't support NaN/Infinity
60         } else {
61           writer.value(f)
62         }
63 
64       case Type.DOUBLE =>
65         val d = variant.getDouble
66         if (d.isNaN || d.isInfinite) {
67           writer.nullValue()  // JSON doesn't support NaN/Infinity
68         } else {
69           writer.value(d)
70         }
71 
72       case Type.DECIMAL4 | Type.DECIMAL8 | Type.DECIMAL16 =>
73         writer.value(variant.getDecimal)
74 
75       case Type.STRING =>
76         writer.value(variant.getString)
77 
78       case Type.BINARY =>
79         val binary = variant.getBinary
80         val bytes = new Array[Byte](binary.remaining())
81         binary.duplicate().get(bytes)
82         val base64 = Base64.getEncoder.encodeToString(bytes)
83         writer.value(base64)
84 
85       case Type.DATE =>
86         // convert days since epoch to ISO date string
87         val daysSinceEpoch = variant.getInt
88         val date = LocalDate.ofEpochDay(daysSinceEpoch.toLong)
89         writer.value(date.format(DateTimeFormatter.ISO_LOCAL_DATE))
90 
91       case Type.TIME =>
92         // convert microseconds since midnight to ISO time string
93         val microsSinceMidnight = variant.getLong
94         val time = LocalTime.ofNanoOfDay(microsSinceMidnight * 1000L)
95         writer.value(time.format(DateTimeFormatter.ISO_LOCAL_TIME))
96 
97       case Type.TIMESTAMP_TZ | Type.TIMESTAMP_NANOS_TZ =>
98         val micros = variant.getLong
99         val instant = if (variantType == Type.TIMESTAMP_TZ) {
100           Instant.ofEpochSecond(micros / 1000000L, (micros % 1000000L) * 1000L)
101         } else {
102           Instant.ofEpochSecond(micros / 1000000000L, micros % 1000000000L)
103         }
104         writer.value(instant.atOffset(ZoneOffset.UTC).format(DateTimeFormatter.ISO_OFFSET_DATE_TIME))
105 
106       case Type.TIMESTAMP_NTZ | Type.TIMESTAMP_NANOS_NTZ =>
107         val micros = variant.getLong
108         val instant = if (variantType == Type.TIMESTAMP_NTZ) {
109           Instant.ofEpochSecond(micros / 1000000L, (micros % 1000000L) * 1000L)
110         } else {
111           Instant.ofEpochSecond(micros / 1000000000L, micros % 1000000000L)
112         }
113         writer.value(instant.atOffset(ZoneOffset.UTC).format(DateTimeFormatter.ISO_LOCAL_DATE_TIME))
114 
115       case Type.ARRAY =>
116         writer.beginArray()
117         val size = variant.numArrayElements()
118         var i = 0
119         while (i < size) {
120           toJson(writer, variant.getElementAtIndex(i))
121           i += 1
122         }
123         writer.endArray()
124 
125       case Type.OBJECT =>
126         writer.beginObject()
127         val size = variant.numObjectElements()
128         var i = 0
129         while (i < size) {
130           val field = variant.getFieldAtIndex(i)
131           writer.name(field.key)
132           toJson(writer, field.value)
133           i += 1
134         }
135         writer.endObject()
136 
137       case _ =>
138         throw new IllegalArgumentException(s"Unsupported variant type: $variantType")
139     }
140   }
