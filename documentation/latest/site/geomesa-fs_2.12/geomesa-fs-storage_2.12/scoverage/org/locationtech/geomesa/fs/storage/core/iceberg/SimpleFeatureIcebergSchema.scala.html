41   /**
42    * Gets the schema needed for reading a file
43    *
44    * @param transform query transform definition
45    * @param filtered columns that have filters against them
46    * @param includeFids include __fid__ column for accurate feature IDs
47    * @param includeRowPositions include _file and _pos columns necessary for handling updates/deletes
48    * @return
49    */
50   def read(
51       transform: Option[String],
52       filtered: Set[String],
53       includeFids: Boolean = true,
54       includeRowPositions: Boolean = false): SimpleFeatureIcebergSchema = {
55     val baseCols = if (includeFids) { Seq(FeatureIdField, VisibilitiesField) } else { Seq(VisibilitiesField) }
56     val (readCols, readSft) = transform match {
57       case None =>
58         val featureCols = baseCols ++ sft.getAttributeDescriptors.asScala.map(d => ColumnName.encode(d.getLocalName))
59         val readCols = featureCols ++ filtered.filterNot(featureCols.contains)
60         (readCols, sft)
61       case Some(defs) =>
62         val fromTransform = Transforms(sft, defs).flatMap {
63           case t: PropertyTransform => Seq(sft.getDescriptor(t.i).getLocalName)
64           case t: RenameTransform => Seq(sft.getDescriptor(t.i).getLocalName)
65           case t: ExpressionTransform => FilterHelper.propertyNames(t.expression, sft)
66           case t => throw new UnsupportedOperationException(s"An implementation is missing: ${t.getClass}")
67         }
68         val attributes = fromTransform.distinct
69         val featureCols = baseCols ++ attributes.map(ColumnName.encode)
70         val readCols = featureCols ++ filtered.filterNot(featureCols.contains)
71         val readSft = {
72           val sftBuilder = new SimpleFeatureTypeBuilder()
73           attributes.foreach(name => sftBuilder.add(sft.getDescriptor(name)))
74           sftBuilder.setName(sft.getName)
75           sftBuilder.buildFeatureType()
76         }
77         readSft.getUserData.putAll(sft.getUserData)
78         (readCols, readSft)
79     }
80     val readSchema = {
81       // here we handle the case where  we're only reading some nested fields out of a given top-level field,
82       // so that we're not reading things we don't have to
83       val parsedCols = new java.util.LinkedHashMap[String, Seq[String]]()
84       readCols.foreach { col =>
85         val sep = col.indexOf('.')
86         if (sep == -1) { parsedCols.put(col, Seq.empty) } else {
87           parsedCols.compute(col.substring(0, sep),
88             (_, children) => (Option(children).getOrElse(Seq.empty) :+ col.substring(sep + 1)).distinct)
89         }
90       }
91       val projection = parsedCols.asScala.toSeq.map { case (name, children) =>
92         val field = schema.findField(name)
93         if (field == null) {
94           throw new IllegalStateException(s"Unexpected projection: $name")
95         }
96         if (children.isEmpty || children.size == field.`type`().asStructType().fields().size()) { field } else {
97           val subfields = field.`type`().asStructType().fields().asScala.filter(f => children.contains(f.name()))
98           Types.NestedField.optional(field.fieldId(), field.name(), Types.StructType.of(subfields.asJava))
99         }
100       }
101       val withRows =
102         if (includeRowPositions) { projection ++ Seq(MetadataColumns.FILE_PATH, MetadataColumns.ROW_POSITION) } else { projection }
103       val ids = projection.collectFirst { case f if f.name() == FeatureIdField => Int.box(f.fieldId()) }
104       new Schema(withRows.asJava, schema.getAliases, ids.toSet.asJava)
105     }
106     new SimpleFeatureIcebergSchema(readSft, readSchema)
107   }
108 }

119   def apply(table: Table, namespace: Option[String] = None): SimpleFeatureIcebergSchema = {
120     val sft = {
121       val typeName = table.properties().get("geomesa.sft.name")
122       val attributes = table.schema().columns().asScala.flatMap(deriveDescriptor(_, table.properties()))
123       val b = new SimpleFeatureTypeBuilder()
124       b.setNamespaceURI(namespace.orNull) // important to set this null if not defined so it doesn't default to gml namespace
125       b.setName(typeName)
126       b.addAll(attributes.asJava)
127       attributes.find(d => d.getUserData.get(AttributeOptions.OptDefault) == "true" && d.isInstanceOf[GeometryDescriptor]).foreach { d =>
128         b.setDefaultGeometry(d.getLocalName)
129       }
130       val sft = b.buildFeatureType()
131       table.properties().asScala.foreach { case (k, v) =>
132         if (k.startsWith(IcebergCatalog.UserDataPrefix)) {
133           sft.getUserData.put(k.substring(IcebergCatalog.UserDataPrefix.length), v)
134         }
135       }
136       sft
137     }
138     new SimpleFeatureIcebergSchema(sft, table.schema())
139   }

141   private def deriveDescriptor(f: NestedField, properties: java.util.Map[String, String]): Option[AttributeDescriptor] = {
142     if (f.name().startsWith(InternalFieldDelimiter) && f.name().endsWith(InternalFieldDelimiter)) { None } else {
143       // the spec is a table property - not a column doc
144       val key = IcebergCatalog.columnSpecProperty(f.name())
145       Option(properties.get(key)) match {
146         case None => logger.warn(s"No attribute spec for column '${f.name()}' (expected property '$key')"); None
147         case Some(d) =>
148           try { Some(SimpleFeatureTypes.createDescriptor(d)) } catch {
149             case NonFatal(e) => logger.warn(s"Error parsing column spec as descriptor: $d", e); None
150           }
151       }
152     }

155   /**
156    * Get a schema based on a simple feature type. Encoding can be configured through `geomesa.parquet.geometries`.
157    *
158    * Note: this should only be called for creating a new schema, for an existing table the field ids will not be correct
159    *
160    * @param sft simple feature type
161    * @param geometries geometry encoding
162    * @return
163    */
164   def create(sft: SimpleFeatureType, geometries: GeometryEncoding): Schema = {
165     val builder = Seq.newBuilder[NestedField]
166     // note: we have to use unique field ids, but iceberg will throw them out and regenerate them when creating a table
167     val fieldIds = new AtomicInteger(1)
168     builder += NestedField.required(FeatureIdField).withId(fieldIds.getAndIncrement()).ofType(StringType.get()).build()
169     builder += buildField(VisibilitiesField, fieldIds.getAndIncrement(), null, StringType.get())
170     sft.getAttributeDescriptors.asScala.foreach { d =>
171       val name = ColumnName(d.getLocalName)
172       val objectType = ObjectType.selectType(d)
173       // no doc is written: an attribute spec does not fit in a column comment once it carries a
174       // structural type definition, so every attribute's spec is a table property - see columnSpecs
175       if (objectType.head == ObjectType.GEOMETRY) {
176         // TODO supports native geometry encoding
177         require(geometries == GeometryEncoding.GeoParquetWkb, "Only WKB encoding is supported for Geometry types")
178         // not yet supported in spark or trino: GeometryType.crs84()
179         builder += buildField(name.column, fieldIds.getAndIncrement(), null, BinaryType.get())
180         builder += BoundingBoxField.icebergSchema(name.column, fieldIds)
181         builder += ZValueField.icebergSchema(name.column, objectType(1), fieldIds)
182       } else if (objectType.last == ObjectType.JSON) {
183         builder +=
184           buildField(name.column, fieldIds.getAndIncrement(), null,
185             d.getJsonSchema().fold[Type](VariantType.get())(buildStructuralType(_, () => fieldIds.getAndIncrement())))
186       } else {
187         builder += buildField(name.column, fieldIds.getAndIncrement(), null, getType(objectType, fieldIds))
188       }
189     }
190 
191     new Schema(builder.result().asJava,  java.util.Set.of[Integer](1))
192   }

222   /**
223    * Builds the schema type for an attribute
224    *
225    * @param bindings object type
226    * @param fieldIds ids for nested fields (as needed)
227    * @return
228    */
229   private def getType(bindings: Seq[ObjectType], fieldIds: AtomicInteger): Type = {
230     bindings.head match {
231       case ObjectType.INT     => IntegerType.get()
232       case ObjectType.DOUBLE  => DoubleType.get()
233       case ObjectType.LONG    => LongType.get()
234       case ObjectType.FLOAT   => FloatType.get()
235       case ObjectType.BOOLEAN => BooleanType.get()
236       case ObjectType.BYTES   => BinaryType.get()
237       case ObjectType.DATE    => TimestampType.withZone()
238       case ObjectType.UUID    => UUIDType.get()
239       case ObjectType.STRING  => StringType.get()
240 
241       case ObjectType.LIST =>
242         val subType = buildField("", fieldIds.getAndIncrement(), null, getType(bindings.drop(1), fieldIds))
243         ListType.ofRequired(subType.fieldId(), subType.`type`())
244 
245       case ObjectType.MAP =>
246         val keyType = buildField("", fieldIds.getAndIncrement(), null, getType(bindings.slice(1, 2), fieldIds))
247         val valueType = buildField("", fieldIds.getAndIncrement(), null, getType(bindings.slice(2, 3), fieldIds))
248         MapType.ofRequired(keyType.fieldId(), valueType.fieldId(), keyType.`type`(), valueType.`type`())
249 
250       case binding =>
251         throw new UnsupportedOperationException(s"No mapping defined for type: $binding")
252     }
253   }

255   private[core] def buildStructuralType(schemaDef: String, fieldIds: TypeUtil.NextID): Type = {
256     val avroSchema = new org.apache.avro.Schema.Parser().parse(schemaDef)
257     // wrap in a record so that we handle non-record schemas (arrays, maps), convert, then extract the single field.
258     // this mirrors the parquet conversion - a record schema just becomes a struct-typed field.
259     val wrapped =
260       org.apache.avro.SchemaBuilder.record("wrapper").namespace("tmp")
261         .fields().name("value").`type`(avroSchema).noDefault().endRecord()
262     val typed = AvroSchemaUtil.convert(wrapped).asStructType().fields().get(0).`type`()
263     // top-level scalars are not supported - just use a regular attribute in that case
264     if (typed.isPrimitiveType) {
265       throw new IllegalArgumentException(
266         s"Structural JSON schema must be a record, array, or map, but was a scalar type: $typed")
267     }
268     // re-stamp the field ids so that they don't overlap our other fields
269     TypeUtil.assignFreshIds(typed, fieldIds)
270   }
