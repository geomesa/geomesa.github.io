40   /**
41    * Gets the schema needed for reading a file
42    *
43    * @param transform query transform definition
44    * @param filtered columns that have filters against them
45    * @param includeFids include __fid__ column for accurate feature IDs
46    * @param includeRowPositions include _file and _pos columns necessary for handling updates/deletes
47    * @return
48    */
49   def read(
50       transform: Option[String],
51       filtered: Set[String],
52       includeFids: Boolean = true,
53       includeRowPositions: Boolean = false): SimpleFeatureIcebergSchema = {
54     val baseCols = if (includeFids) { Seq(FeatureIdField, VisibilitiesField) } else { Seq(VisibilitiesField) }
55     val (readCols, readSft) = transform match {
56       case None =>
57         val featureCols = baseCols ++ sft.getAttributeDescriptors.asScala.map(d => ColumnName.encode(d.getLocalName))
58         val readCols = featureCols ++ filtered.filterNot(featureCols.contains)
59         (readCols, sft)
60       case Some(defs) =>
61         val fromTransform = Transforms(sft, defs).flatMap {
62           case t: PropertyTransform => Seq(sft.getDescriptor(t.i).getLocalName)
63           case t: RenameTransform => Seq(sft.getDescriptor(t.i).getLocalName)
64           case t: ExpressionTransform => FilterHelper.propertyNames(t.expression, sft)
65           case t => throw new UnsupportedOperationException(s"An implementation is missing: ${t.getClass}")
66         }
67         val attributes = fromTransform.distinct
68         val featureCols = baseCols ++ attributes.map(ColumnName.encode)
69         val readCols = featureCols ++ filtered.filterNot(featureCols.contains)
70         val readSft = {
71           val sftBuilder = new SimpleFeatureTypeBuilder()
72           attributes.foreach(name => sftBuilder.add(sft.getDescriptor(name)))
73           sftBuilder.setName(sft.getName)
74           sftBuilder.buildFeatureType()
75         }
76         readSft.getUserData.putAll(sft.getUserData)
77         (readCols, readSft)
78     }
79     val readSchema = {
80       // here we handle the case where  we're only reading some nested fields out of a given top-level field,
81       // so that we're not reading things we don't have to
82       val parsedCols = new java.util.LinkedHashMap[String, Seq[String]]()
83       readCols.foreach { col =>
84         val sep = col.indexOf('.')
85         if (sep == -1) { parsedCols.put(col, Seq.empty) } else {
86           parsedCols.compute(col.substring(0, sep),
87             (_, children) => (Option(children).getOrElse(Seq.empty) :+ col.substring(sep + 1)).distinct)
88         }
89       }
90       val projection = parsedCols.asScala.toSeq.map { case (name, children) =>
91         val field = schema.findField(name)
92         if (field == null) {
93           throw new IllegalStateException(s"Unexpected projection: $name")
94         }
95         if (children.isEmpty || children.size == field.`type`().asStructType().fields().size()) { field } else {
96           val subfields = field.`type`().asStructType().fields().asScala.filter(f => children.contains(f.name()))
97           Types.NestedField.optional(field.fieldId(), field.name(), Types.StructType.of(subfields.asJava))
98         }
99       }
100       val withRows =
101         if (includeRowPositions) { projection ++ Seq(MetadataColumns.FILE_PATH, MetadataColumns.ROW_POSITION) } else { projection }
102       val ids = projection.collectFirst { case f if f.name() == FeatureIdField => Int.box(f.fieldId()) }
103       new Schema(withRows.asJava, schema.getAliases, ids.toSet.asJava)
104     }
105     new SimpleFeatureIcebergSchema(readSft, readSchema)
106   }
107 }

117   def apply(table: Table, namespace: Option[String] = None): SimpleFeatureIcebergSchema = {
118     val sft = {
119       val typeName = table.properties().get("geomesa.sft.name")
120       val attributes = table.schema().columns().asScala.flatMap(deriveDescriptor)
121       if (attributes.isEmpty) {
122         // back compatibility check
123         SimpleFeatureTypes.createType(namespace.fold(typeName)(n => s"$n:$typeName"), table.properties().get("geomesa.sft.spec"))
124       } else {
125         val b = new SimpleFeatureTypeBuilder()
126         b.setNamespaceURI(namespace.orNull) // important to set this null if not defined so it doesn't default to gml namespace
127         b.setName(typeName)
128         b.addAll(attributes.asJava)
129         attributes.find(d => d.getUserData.get(AttributeOptions.OptDefault) == "true" && d.isInstanceOf[GeometryDescriptor]).foreach { d =>
130           b.setDefaultGeometry(d.getLocalName)
131         }
132         val sft = b.buildFeatureType()
133         table.properties().asScala.foreach { case (k, v) =>
134           if (k.startsWith(IcebergCatalog.UserDataPrefix)) {
135             sft.getUserData.put(k.substring(IcebergCatalog.UserDataPrefix.length), v)
136           }
137         }
138         sft
139       }
140     }
141     new SimpleFeatureIcebergSchema(sft, table.schema())
142   }

154   /**
155    * Get a schema based on a simple feature type. Encoding can be configured through `geomesa.parquet.geometries`.
156    *
157    * Note: this should only be called for creating a new schema, for an existing table the field ids will not be correct
158    *
159    * @param sft simple feature type
160    * @param geometries geometry encoding
161    * @return
162    */
163   def create(sft: SimpleFeatureType, geometries: GeometryEncoding): Schema = {
164     val builder = Seq.newBuilder[NestedField]
165     // note: we have to use unique field ids, but iceberg will throw them out and regenerate them when creating a table
166     val fieldIds = new AtomicInteger(1)
167     builder += NestedField.required(FeatureIdField).withId(fieldIds.getAndIncrement()).ofType(StringType.get()).build()
168     builder += buildField(VisibilitiesField, fieldIds.getAndIncrement(), null, StringType.get())
169     sft.getAttributeDescriptors.asScala.foreach { d =>
170       val name = ColumnName(d.getLocalName)
171       val objectType = ObjectType.selectType(d)
172       val doc = SimpleFeatureTypes.encodeDescriptor(sft, d)
173       if (objectType.head == ObjectType.STRING) {
174         val typed = if (objectType.last == ObjectType.JSON) { VariantType.get() } else { StringType.get() }
175         builder += buildField(name.column, fieldIds.getAndIncrement(), doc, typed)
176       } else if (objectType.head == ObjectType.GEOMETRY) {
177         // TODO supports native geometry encoding
178         require(geometries == GeometryEncoding.GeoParquetWkb, "Only WKB encoding is supported for Geometry types")
179         val geomDoc = {
180           // note: geotools AttributeTypeBuilder shares the user data map - reparse instead so we don't change the original
181           val descriptor = SimpleFeatureTypes.createDescriptor(doc)
182           descriptor.getUserData.put(GeometryEncodingKey, geometries.toString)
183           SimpleFeatureTypes.encodeDescriptor(sft, d)
184         }
185         // not yet supported in spark or trino: GeometryType.crs84()
186         builder += buildField(name.column, fieldIds.getAndIncrement(), geomDoc, BinaryType.get())
187         builder += BoundingBoxField.icebergSchema(name.column, fieldIds)
188         builder += ZValueField.icebergSchema(name.column, objectType(1), fieldIds)
189       } else {
190         builder += buildField(name.column, fieldIds.getAndIncrement(), doc, getType(objectType, fieldIds))
191       }
192     }
193 
194     new Schema(builder.result().asJava,  java.util.Set.of[Integer](1))
195   }

209   /**
210    * Builds the schema type for an attribute
211    *
212    * @param bindings object type
213    * @param fieldIds ids for nested fields (as needed)
214    * @return
215    */
216   private def getType(bindings: Seq[ObjectType], fieldIds: AtomicInteger): Type = {
217     bindings.head match {
218       case ObjectType.INT     => IntegerType.get()
219       case ObjectType.DOUBLE  => DoubleType.get()
220       case ObjectType.LONG    => LongType.get()
221       case ObjectType.FLOAT   => FloatType.get()
222       case ObjectType.BOOLEAN => BooleanType.get()
223       case ObjectType.BYTES   => BinaryType.get()
224       case ObjectType.DATE    => TimestampType.withZone()
225       case ObjectType.UUID    => UUIDType.get()
226       case ObjectType.STRING  => StringType.get()
227 
228       case ObjectType.LIST =>
229         val subType = buildField("", fieldIds.getAndIncrement(), null, getType(bindings.drop(1), fieldIds))
230         ListType.ofRequired(subType.fieldId(), subType.`type`())
231 
232       case ObjectType.MAP =>
233         val keyType = buildField("", fieldIds.getAndIncrement(), null, getType(bindings.slice(1, 2), fieldIds))
234         val valueType = buildField("", fieldIds.getAndIncrement(), null, getType(bindings.slice(2, 3), fieldIds))
235         MapType.ofRequired(keyType.fieldId(), valueType.fieldId(), keyType.`type`(), valueType.`type`())
236 
237       case binding =>
238         throw new UnsupportedOperationException(s"No mapping defined for type: $binding")
239     }
