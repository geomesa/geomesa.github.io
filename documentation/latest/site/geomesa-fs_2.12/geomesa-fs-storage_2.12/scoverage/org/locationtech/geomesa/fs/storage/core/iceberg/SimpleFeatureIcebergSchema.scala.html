41   /**
42    * Gets the schema needed for reading a file
43    *
44    * @param transform query transform definition
45    * @param filtered columns that have filters against them
46    * @param includeFids include __fid__ column for accurate feature IDs
47    * @param includeRowPositions include _file and _pos columns necessary for handling updates/deletes
48    * @return
49    */
50   def read(
51       transform: Option[String],
52       filtered: Set[String],
53       includeFids: Boolean = true,
54       includeRowPositions: Boolean = false): SimpleFeatureIcebergSchema = {
55     val baseCols = if (includeFids) { Seq(FeatureIdField, VisibilitiesField) } else { Seq(VisibilitiesField) }
56     val (readCols, readSft) = transform match {
57       case None =>
58         val featureCols = baseCols ++ sft.getAttributeDescriptors.asScala.map(d => ColumnName.encode(d.getLocalName))
59         val readCols = featureCols ++ filtered.filterNot(featureCols.contains)
60         (readCols, sft)
61       case Some(defs) =>
62         val fromTransform = Transforms(sft, defs).flatMap {
63           case t: PropertyTransform => Seq(sft.getDescriptor(t.i).getLocalName)
64           case t: RenameTransform => Seq(sft.getDescriptor(t.i).getLocalName)
65           case t: ExpressionTransform => FilterHelper.propertyNames(t.expression, sft)
66           case t => throw new UnsupportedOperationException(s"An implementation is missing: ${t.getClass}")
67         }
68         val attributes = fromTransform.distinct
69         val featureCols = baseCols ++ attributes.map(ColumnName.encode)
70         val readCols = featureCols ++ filtered.filterNot(featureCols.contains)
71         val readSft = {
72           val sftBuilder = new SimpleFeatureTypeBuilder()
73           attributes.foreach(name => sftBuilder.add(sft.getDescriptor(name)))
74           sftBuilder.setName(sft.getName)
75           sftBuilder.buildFeatureType()
76         }
77         readSft.getUserData.putAll(sft.getUserData)
78         (readCols, readSft)
79     }
80     val readSchema = {
81       // here we handle the case where  we're only reading some nested fields out of a given top-level field,
82       // so that we're not reading things we don't have to
83       val parsedCols = new java.util.LinkedHashMap[String, Seq[String]]()
84       readCols.foreach { col =>
85         val sep = col.indexOf('.')
86         if (sep == -1) { parsedCols.put(col, Seq.empty) } else {
87           parsedCols.compute(col.substring(0, sep),
88             (_, children) => (Option(children).getOrElse(Seq.empty) :+ col.substring(sep + 1)).distinct)
89         }
90       }
91       val projection = parsedCols.asScala.toSeq.map { case (name, children) =>
92         val field = schema.findField(name)
93         if (field == null) {
94           throw new IllegalStateException(s"Unexpected projection: $name")
95         }
96         if (children.isEmpty || children.size == field.`type`().asStructType().fields().size()) { field } else {
97           val subfields = field.`type`().asStructType().fields().asScala.filter(f => children.contains(f.name()))
98           Types.NestedField.optional(field.fieldId(), field.name(), Types.StructType.of(subfields.asJava))
99         }
100       }
101       val withRows =
102         if (includeRowPositions) { projection ++ Seq(MetadataColumns.FILE_PATH, MetadataColumns.ROW_POSITION) } else { projection }
103       val ids = projection.collectFirst { case f if f.name() == FeatureIdField => Int.box(f.fieldId()) }
104       new Schema(withRows.asJava, schema.getAliases, ids.toSet.asJava)
105     }
106     new SimpleFeatureIcebergSchema(readSft, readSchema)
107   }
108 }

119   def apply(table: Table, namespace: Option[String] = None): SimpleFeatureIcebergSchema = {
120     val sft = {
121       val typeName = table.properties().get("geomesa.sft.name")
122       val attributes = table.schema().columns().asScala.flatMap(deriveDescriptor)
123       if (attributes.isEmpty) {
124         // back compatibility check
125         SimpleFeatureTypes.createType(namespace.fold(typeName)(n => s"$n:$typeName"), table.properties().get("geomesa.sft.spec"))
126       } else {
127         val b = new SimpleFeatureTypeBuilder()
128         b.setNamespaceURI(namespace.orNull) // important to set this null if not defined so it doesn't default to gml namespace
129         b.setName(typeName)
130         b.addAll(attributes.asJava)
131         attributes.find(d => d.getUserData.get(AttributeOptions.OptDefault) == "true" && d.isInstanceOf[GeometryDescriptor]).foreach { d =>
132           b.setDefaultGeometry(d.getLocalName)
133         }
134         val sft = b.buildFeatureType()
135         table.properties().asScala.foreach { case (k, v) =>
136           if (k.startsWith(IcebergCatalog.UserDataPrefix)) {
137             sft.getUserData.put(k.substring(IcebergCatalog.UserDataPrefix.length), v)
138           }
139         }
140         sft
141       }
142     }
143     new SimpleFeatureIcebergSchema(sft, table.schema())
144   }

156   /**
157    * Get a schema based on a simple feature type. Encoding can be configured through `geomesa.parquet.geometries`.
158    *
159    * Note: this should only be called for creating a new schema, for an existing table the field ids will not be correct
160    *
161    * @param sft simple feature type
162    * @param geometries geometry encoding
163    * @return
164    */
165   def create(sft: SimpleFeatureType, geometries: GeometryEncoding): Schema = {
166     val builder = Seq.newBuilder[NestedField]
167     // note: we have to use unique field ids, but iceberg will throw them out and regenerate them when creating a table
168     val fieldIds = new AtomicInteger(1)
169     builder += NestedField.required(FeatureIdField).withId(fieldIds.getAndIncrement()).ofType(StringType.get()).build()
170     builder += buildField(VisibilitiesField, fieldIds.getAndIncrement(), null, StringType.get())
171     sft.getAttributeDescriptors.asScala.foreach { d =>
172       val name = ColumnName(d.getLocalName)
173       val objectType = ObjectType.selectType(d)
174       val doc = SimpleFeatureTypes.encodeDescriptor(sft, d)
175       if (objectType.head == ObjectType.GEOMETRY) {
176         // TODO supports native geometry encoding
177         require(geometries == GeometryEncoding.GeoParquetWkb, "Only WKB encoding is supported for Geometry types")
178         val geomDoc = {
179           // note: geotools AttributeTypeBuilder shares the user data map - reparse instead so we don't change the original
180           val descriptor = SimpleFeatureTypes.createDescriptor(doc)
181           descriptor.getUserData.put(GeometryEncodingKey, geometries.toString)
182           SimpleFeatureTypes.encodeDescriptor(sft, d)
183         }
184         // not yet supported in spark or trino: GeometryType.crs84()
185         builder += buildField(name.column, fieldIds.getAndIncrement(), geomDoc, BinaryType.get())
186         builder += BoundingBoxField.icebergSchema(name.column, fieldIds)
187         builder += ZValueField.icebergSchema(name.column, objectType(1), fieldIds)
188       } else if (objectType.last == ObjectType.JSON) {
189         builder +=
190           buildField(name.column, fieldIds.getAndIncrement(), doc,
191             d.getJsonSchema().fold[Type](VariantType.get())(buildStructuralType(_, () => fieldIds.getAndIncrement())))
192       } else {
193         builder += buildField(name.column, fieldIds.getAndIncrement(), doc, getType(objectType, fieldIds))
194       }
195     }
196 
197     new Schema(builder.result().asJava,  java.util.Set.of[Integer](1))
198   }

212   /**
213    * Builds the schema type for an attribute
214    *
215    * @param bindings object type
216    * @param fieldIds ids for nested fields (as needed)
217    * @return
218    */
219   private def getType(bindings: Seq[ObjectType], fieldIds: AtomicInteger): Type = {
220     bindings.head match {
221       case ObjectType.INT     => IntegerType.get()
222       case ObjectType.DOUBLE  => DoubleType.get()
223       case ObjectType.LONG    => LongType.get()
224       case ObjectType.FLOAT   => FloatType.get()
225       case ObjectType.BOOLEAN => BooleanType.get()
226       case ObjectType.BYTES   => BinaryType.get()
227       case ObjectType.DATE    => TimestampType.withZone()
228       case ObjectType.UUID    => UUIDType.get()
229       case ObjectType.STRING  => StringType.get()
230 
231       case ObjectType.LIST =>
232         val subType = buildField("", fieldIds.getAndIncrement(), null, getType(bindings.drop(1), fieldIds))
233         ListType.ofRequired(subType.fieldId(), subType.`type`())
234 
235       case ObjectType.MAP =>
236         val keyType = buildField("", fieldIds.getAndIncrement(), null, getType(bindings.slice(1, 2), fieldIds))
237         val valueType = buildField("", fieldIds.getAndIncrement(), null, getType(bindings.slice(2, 3), fieldIds))
238         MapType.ofRequired(keyType.fieldId(), valueType.fieldId(), keyType.`type`(), valueType.`type`())
239 
240       case binding =>
241         throw new UnsupportedOperationException(s"No mapping defined for type: $binding")
242     }
243   }

245   private[core] def buildStructuralType(schemaDef: String, fieldIds: TypeUtil.NextID): Type = {
246     val avroSchema = new org.apache.avro.Schema.Parser().parse(schemaDef)
247     // wrap in a record so that we handle non-record schemas (arrays, maps), convert, then extract the single field.
248     // this mirrors the parquet conversion - a record schema just becomes a struct-typed field.
249     val wrapped =
250       org.apache.avro.SchemaBuilder.record("wrapper").namespace("tmp")
251         .fields().name("value").`type`(avroSchema).noDefault().endRecord()
252     val typed = AvroSchemaUtil.convert(wrapped).asStructType().fields().get(0).`type`()
253     // top-level scalars are not supported - just use a regular attribute in that case
254     if (typed.isPrimitiveType) {
255       throw new IllegalArgumentException(
256         s"Structural JSON schema must be a record, array, or map, but was a scalar type: $typed")
257     }
258     // re-stamp the field ids so that they don't overlap our other fields
259     TypeUtil.assignFreshIds(typed, fieldIds)
260   }
