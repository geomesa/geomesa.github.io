46   /**
47    * Gets the schema needed for reading a file
48    *
49    * @param transform query transform definition
50    * @param filtered columns that have filters against them
51    * @param includeFids include __fid__ column for accurate feature IDs
52    * @param includeRowPositions include _file and _pos columns necessary for handling updates/deletes
53    * @return
54    */
55   def read(
56       transform: Option[String],
57       filtered: Set[String],
58       includeFids: Boolean = true,
59       includeRowPositions: Boolean = false): SimpleFeatureIcebergSchema = {
60     val baseCols = if (includeFids) { Seq(FeatureIdField, VisibilitiesField) } else { Seq(VisibilitiesField) }
61     val (readCols, readSft) = transform match {
62       case None =>
63         val featureCols = baseCols ++ sft.getAttributeDescriptors.asScala.map(d => ColumnName.encode(d.getLocalName))
64         val readCols = featureCols ++ filtered.filterNot(featureCols.contains)
65         (readCols, sft)
66       case Some(defs) =>
67         val fromTransform = Transforms(sft, defs).flatMap {
68           case t: PropertyTransform => Seq(sft.getDescriptor(t.i).getLocalName)
69           case t: RenameTransform => Seq(sft.getDescriptor(t.i).getLocalName)
70           case t: ExpressionTransform => FilterHelper.propertyNames(t.expression, sft)
71           case t => throw new UnsupportedOperationException(s"An implementation is missing: ${t.getClass}")
72         }
73         val attributes = fromTransform.distinct
74         val featureCols = baseCols ++ attributes.map(ColumnName.encode)
75         val readCols = featureCols ++ filtered.filterNot(featureCols.contains)
76         val readSft = {
77           val sftBuilder = new SimpleFeatureTypeBuilder()
78           attributes.foreach(name => sftBuilder.add(sft.getDescriptor(name)))
79           sftBuilder.setName(sft.getName)
80           sftBuilder.buildFeatureType()
81         }
82         readSft.getUserData.putAll(sft.getUserData)
83         (readCols, readSft)
84     }
85     val schema = fields.toSchema(readCols, includeRowPositions)
86     new SimpleFeatureIcebergSchema(readSft, geometries, metadata, schema, fields)
87   }
88 }

113   private def schema(sft: SimpleFeatureType, geometries: GeometryEncoding): IcebergFields = {
114     // note: for iceberg compatibility, field ids need to start at one and increment (without gaps) across all top-level fields.
115     // All nested fields (structs, lists) get ids *after* all the top-level fields, once again incrementing without gaps
116     val fieldIds = new AtomicInteger(1)
117     val nestedFieldIds = {
118       val extraGeomCount = 2 * sft.getAttributeDescriptors.asScala.count {
119         case d if classOf[Geometry].isAssignableFrom(d.getType.getBinding) => true
120         case _ => false
121       }
122       new AtomicInteger(3 + sft.getAttributeCount + extraGeomCount)
123     }
124 
125     val builder = Seq.newBuilder[NestedField]
126     builder += NestedField.required(fieldIds.getAndIncrement(), FeatureIdField, StringType.get())
127     builder += NestedField.optional(fieldIds.getAndIncrement(), VisibilitiesField, StringType.get())
128 
129     sft.getAttributeDescriptors.asScala.foreach { d =>
130       val name = ColumnName(d.getLocalName)
131       val objectType = ObjectType.selectType(d)
132       val doc = SimpleFeatureTypes.encodeDescriptor(sft, d)
133       builder += buildField(name.column, objectType, doc, geometries, fieldIds, nestedFieldIds)
134       if (objectType.head == ObjectType.GEOMETRY) {
135         builder += BoundingBoxField.icebergSchema(name.column, fieldIds, nestedFieldIds)
136         builder += ZValueField.icebergSchema(name.column, objectType(1), fieldIds)
137       }
138     }
139 
140     val fields = builder.result()
141     val aliases = fields.map(f => f.name() -> Int.box(f.fieldId())).toMap
142 
143     IcebergFields(fields, aliases)
144   }

146   /**
147    * Builds the schema type for an attribute
148    *
149    * @param name field name
150    * @param bindings object type
151    * @param geometries geometry type encodings
152    * @return
153    */
154   private def buildField(
155       name: String,
156       bindings: Seq[ObjectType],
157       doc: String,
158       geometries: GeometryEncoding,
159       fieldIds: AtomicInteger,
160       nestedFieldIds: AtomicInteger): NestedField = {
161     val builder = NestedField.optional(name).withId(fieldIds.getAndIncrement()).withDoc(doc)
162     val typed: NestedField.Builder = bindings.head match {
163       case ObjectType.INT     => builder.ofType(IntegerType.get())
164       case ObjectType.DOUBLE  => builder.ofType(DoubleType.get())
165       case ObjectType.LONG    => builder.ofType(LongType.get())
166       case ObjectType.FLOAT   => builder.ofType(FloatType.get())
167       case ObjectType.BOOLEAN => builder.ofType(BooleanType.get())
168       case ObjectType.BYTES   => builder.ofType(BinaryType.get())
169       case ObjectType.STRING  => builder.ofType(StringType.get())
170       case ObjectType.DATE    => builder.ofType(TimestampType.withZone())
171       case ObjectType.UUID    => builder.ofType(UUIDType.get())
172 
173       case ObjectType.LIST =>
174         val subType = buildField("", bindings.drop(1), null, geometries, nestedFieldIds, null /* should not be used*/)
175         builder.ofType(ListType.ofRequired(subType.fieldId(), subType.`type`()))
176 
177       case ObjectType.MAP =>
178         val keyType = buildField("", bindings.slice(1, 2), null, geometries, nestedFieldIds, null /* should not be used*/)
179         val valueType = buildField("", bindings.slice(2, 3), null, geometries, nestedFieldIds, null /* should not be used*/)
180         builder.ofType(MapType.ofRequired(keyType.fieldId(), valueType.fieldId(), keyType.`type`(), valueType.`type`()))
181 
182       case ObjectType.GEOMETRY =>
183         // not yet supported in spark or trino
184         // builder.ofType(GeometryType.crs84())
185         builder.ofType(BinaryType.get())
186 
187       case binding =>
188         throw new UnsupportedOperationException(s"No mapping defined for type: $binding")
189     }
190     typed.build()
191   }

193   private case class IcebergFields(fields: Seq[NestedField], aliases: Map[String, Integer]) {
194     def toSchema: Schema = new Schema(fields.asJava, aliases.asJava, java.util.Set.of[Integer](1))
195     def toSchema(cols: Seq[String], includeRowPositions: Boolean): Schema = {
196       val parsedCols = new java.util.LinkedHashMap[String, Seq[String]]()
197       cols.foreach { col =>
198         val sep = col.indexOf('.')
199         if (sep == -1) { parsedCols.put(col, Seq.empty) } else {
200           parsedCols.compute(col.substring(0, sep),
201             (_, children) => (Option(children).getOrElse(Seq.empty) :+ col.substring(sep + 1)).distinct)
202         }
203       }
204       val projection = parsedCols.asScala.toSeq.map { case (name, children) =>
205         val field = fields.find(_.name() == name).getOrElse(throw new IllegalStateException(s"Unexpected projection: $name"))
206         if (children.isEmpty || children.size == field.`type`().asStructType().fields().size()) { field } else {
207           val subfields = field.`type`().asStructType().fields().asScala.filter(f => children.contains(f.name()))
208           Types.NestedField.optional(field.fieldId(), field.name(), Types.StructType.of(subfields.asJava))
209         }
210       }
211       val withRows =
212         if (includeRowPositions) { projection ++ Seq(MetadataColumns.FILE_PATH, MetadataColumns.ROW_POSITION) } else { projection }
213       val ids = projection.collectFirst { case f if f.name() == FeatureIdField => Int.box(f.fieldId()) }
214       new Schema(withRows.asJava, aliases.asJava, ids.toSet.asJava)
215     }
216   }
