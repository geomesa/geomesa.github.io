46   /**
47    * Gets the schema needed for reading a file
48    *
49    * @param transform query transform definition
50    * @param filtered columns that have filters against them
51    * @param includeFids include __fid__ column for accurate feature IDs
52    * @param includeRowPositions include _file and _pos columns necessary for handling updates/deletes
53    * @return
54    */
55   def read(
56       transform: Option[String],
57       filtered: Set[String],
58       includeFids: Boolean = true,
59       includeRowPositions: Boolean = false): SimpleFeatureIcebergSchema = {
60     val baseCols = if (includeFids) { Seq(FeatureIdField, VisibilitiesField) } else { Seq(VisibilitiesField) }
61     val (readCols, readSft) = transform match {
62       case None =>
63         val featureCols = baseCols ++ sft.getAttributeDescriptors.asScala.map(d => ColumnName.encode(d.getLocalName))
64         val readCols = featureCols ++ filtered.filterNot(featureCols.contains)
65         (readCols, sft)
66       case Some(defs) =>
67         val fromTransform = Transforms(sft, defs).flatMap {
68           case t: PropertyTransform => Seq(sft.getDescriptor(t.i).getLocalName)
69           case t: RenameTransform => Seq(sft.getDescriptor(t.i).getLocalName)
70           case t: ExpressionTransform => FilterHelper.propertyNames(t.expression, sft)
71           case t => throw new UnsupportedOperationException(s"An implementation is missing: ${t.getClass}")
72         }
73         val attributes = fromTransform.distinct
74         val featureCols = baseCols ++ attributes.map(ColumnName.encode)
75         val readCols = featureCols ++ filtered.filterNot(featureCols.contains)
76         val readSft = {
77           val sftBuilder = new SimpleFeatureTypeBuilder()
78           attributes.foreach(name => sftBuilder.add(sft.getDescriptor(name)))
79           sftBuilder.setName(sft.getName)
80           sftBuilder.buildFeatureType()
81         }
82         readSft.getUserData.putAll(sft.getUserData)
83         (readCols, readSft)
84     }
85     val schema = fields.toSchema(readCols, includeRowPositions)
86     new SimpleFeatureIcebergSchema(readSft, geometries, metadata, schema, fields)
87   }
88 }

113   private def schema(sft: SimpleFeatureType, geometries: GeometryEncoding): IcebergFields = {
114     // note: for iceberg compatibility, field ids need to start at one and increment (without gaps) across all top-level fields.
115     // All nested fields (structs, lists) get ids *after* all the top-level fields, once again incrementing without gaps
116     val fieldIds = new AtomicInteger(1)
117     val nestedFieldIds = {
118       val extraGeomCount = 2 * sft.getAttributeDescriptors.asScala.count {
119         case d if classOf[Geometry].isAssignableFrom(d.getType.getBinding) => true
120         case _ => false
121       }
122       new AtomicInteger(3 + sft.getAttributeCount + extraGeomCount)
123     }
124 
125     val builder = Seq.newBuilder[NestedField]
126     builder += NestedField.required(fieldIds.getAndIncrement(), FeatureIdField, StringType.get())
127     builder += NestedField.optional(fieldIds.getAndIncrement(), VisibilitiesField, StringType.get())
128 
129     sft.getAttributeDescriptors.asScala.foreach { d =>
130       val name = ColumnName(d.getLocalName)
131       val objectType = ObjectType.selectType(d)
132       builder += buildField(name.column, objectType, geometries, fieldIds, nestedFieldIds)
133       if (objectType.head == ObjectType.GEOMETRY) {
134         builder += BoundingBoxField.icebergSchema(name.column, fieldIds, nestedFieldIds)
135         builder += ZValueField.icebergSchema(name.column, objectType(1), fieldIds)
136       }
137     }
138 
139     val fields = builder.result()
140     val aliases = fields.map(f => f.name() -> Int.box(f.fieldId())).toMap
141 
142     IcebergFields(fields, aliases)
143   }

145   /**
146    * Builds the schema type for an attribute
147    *
148    * @param name field name
149    * @param bindings object type
150    * @param geometries geometry type encodings
151    * @return
152    */
153   private def buildField(
154       name: String,
155       bindings: Seq[ObjectType],
156       geometries: GeometryEncoding,
157       fieldIds: AtomicInteger,
158       nestedFieldIds: AtomicInteger): NestedField = {
159     val builder = NestedField.optional(name).withId(fieldIds.getAndIncrement())
160     val typed: NestedField.Builder = bindings.head match {
161       case ObjectType.INT     => builder.ofType(IntegerType.get())
162       case ObjectType.DOUBLE  => builder.ofType(DoubleType.get())
163       case ObjectType.LONG    => builder.ofType(LongType.get())
164       case ObjectType.FLOAT   => builder.ofType(FloatType.get())
165       case ObjectType.BOOLEAN => builder.ofType(BooleanType.get())
166       case ObjectType.BYTES   => builder.ofType(BinaryType.get())
167       case ObjectType.STRING  => builder.ofType(StringType.get())
168       case ObjectType.DATE    => builder.ofType(TimestampType.withZone())
169       case ObjectType.UUID    => builder.ofType(UUIDType.get())
170 
171       case ObjectType.LIST =>
172         val subType = buildField("", bindings.drop(1), geometries, nestedFieldIds, null /* should not be used*/)
173         builder.ofType(ListType.ofRequired(subType.fieldId(), subType.`type`()))
174 
175       case ObjectType.MAP =>
176         val keyType = buildField("", bindings.slice(1, 2), geometries, nestedFieldIds, null /* should not be used*/)
177         val valueType = buildField("", bindings.slice(2, 3), geometries, nestedFieldIds, null /* should not be used*/)
178         builder.ofType(MapType.ofRequired(keyType.fieldId(), valueType.fieldId(), keyType.`type`(), valueType.`type`()))
179 
180       case ObjectType.GEOMETRY =>
181         // not yet supported in spark or trino
182         // builder.ofType(GeometryType.crs84())
183         builder.ofType(BinaryType.get())
184 
185       case binding =>
186         throw new UnsupportedOperationException(s"No mapping defined for type: $binding")
187     }
188     typed.build()
189   }

191   private case class IcebergFields(fields: Seq[NestedField], aliases: Map[String, Integer]) {
192     def toSchema: Schema = new Schema(fields.asJava, aliases.asJava, java.util.Set.of[Integer](1))
193     def toSchema(cols: Seq[String], includeRowPositions: Boolean): Schema = {
194       val parsedCols = new java.util.LinkedHashMap[String, Seq[String]]()
195       cols.foreach { col =>
196         val sep = col.indexOf('.')
197         if (sep == -1) { parsedCols.put(col, Seq.empty) } else {
198           parsedCols.compute(col.substring(0, sep),
199             (_, children) => (Option(children).getOrElse(Seq.empty) :+ col.substring(sep + 1)).distinct)
200         }
201       }
202       val projection = parsedCols.asScala.toSeq.map { case (name, children) =>
203         val field = fields.find(_.name() == name).getOrElse(throw new IllegalStateException(s"Unexpected projection: $name"))
204         if (children.isEmpty || children.size == field.`type`().asStructType().fields().size()) { field } else {
205           val subfields = field.`type`().asStructType().fields().asScala.filter(f => children.contains(f.name()))
206           Types.NestedField.optional(field.fieldId(), field.name(), Types.StructType.of(subfields.asJava))
207         }
208       }
209       val withRows =
210         if (includeRowPositions) { projection ++ Seq(MetadataColumns.FILE_PATH, MetadataColumns.ROW_POSITION) } else { projection }
211       val ids = projection.collectFirst { case f if f.name() == FeatureIdField => Int.box(f.fieldId()) }
212       new Schema(withRows.asJava, aliases.asJava, ids.toSet.asJava)
213     }
214   }
