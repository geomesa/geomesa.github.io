49   /**
50    * Gets the schema needed for reading a file
51    *
52    * @param transform query transform definition
53    * @param postReadFilter filter that will be applied to the results after reading from iceberg
54    * @param filterCols columns that have iceberg filters against them
55    * @param includeFids include __fid__ column for accurate feature IDs
56    * @param includeRowPositions include _file and _pos columns necessary for handling updates/deletes
57    * @return
58    */
59   def read(
60       transform: Option[(String, SimpleFeatureType)],
61       postReadFilter: Option[Filter],
62       filterCols: Set[String],
63       includeFids: Boolean = true,
64       includeRowPositions: Boolean = false): ReadSchema = {
65     // map of top-level columns to nested fields that we're reading - an empty value means read all nested fields
66     val readCols = new java.util.LinkedHashMap[String, Seq[String]]()
67     // descriptors for our read sft
68     val readDescriptors = ArrayBuffer.empty[AttributeDescriptor]
69     // secondary transform we have to apply after running the client-side filter
70     var postReadTransform: Option[(SimpleFeatureType, Array[Transform])] = None
71 
72     // add a dot-delimited field to read
73     def addReadPath(field: String): Unit = {
74       val i = field.indexOf('.')
75       if (i == -1) {
76         val existing = readCols.get(field)
77         if (existing == null || existing.nonEmpty) {
78           readCols.put(field, Seq.empty)
79         }
80       } else {
81         val topLevelCol = field.substring(0, i)
82         val existing = readCols.get(topLevelCol)
83         if (existing == null) {
84           readCols.put(topLevelCol, Seq(field.substring(i + 1)))
85         } else if (existing.nonEmpty) {
86           readCols.put(topLevelCol, (existing ++ Seq(field.substring(i + 1))).distinct)
87         }
88       }
89     }
90 
91     if (includeFids || postReadFilter.exists(FilterHelper.hasIdFilter)) {
92       addReadPath(FeatureIdField)
93     }
94     addReadPath(VisibilitiesField)
95 
96     transform match {
97       case None =>
98         sft.getAttributeDescriptors.asScala.foreach { descriptor =>
99           addReadPath(ColumnName.encode(descriptor.getLocalName))
100           readDescriptors += descriptor
101         }
102 
103       case Some((defs, tsft)) =>
104         val postReadTransforms = ArrayBuffer.empty[Transform]
105         var requiresRetype = false
106 
107         Transforms(sft, defs).foreach {
108           case t: PropertyTransform =>
109             val descriptor = sft.getDescriptor(t.i)
110             readDescriptors += descriptor
111             postReadTransforms += t.copy(i = readDescriptors.size - 1)
112             addReadPath(ColumnName.encode(descriptor.getLocalName))
113 
114           case t: RenameTransform =>
115             requiresRetype = true // to rename the sft attribute
116             val descriptor = sft.getDescriptor(t.i)
117             readDescriptors += descriptor
118             postReadTransforms += PropertyTransform(t.name, t.binding, readDescriptors.size - 1)
119             addReadPath(ColumnName.encode(descriptor.getLocalName))
120 
121           case t: ExpressionTransform =>
122             t.expression match {
123               case p: PropertyName if p.getPropertyName.startsWith("$") =>
124                 try {
125                   val (descriptor, path) = parseJsonPath(p.getPropertyName, sft)
126                   val td = new AttributeTypeBuilder().binding(classOf[AnyRef]).nillable(true).buildDescriptor(p.getPropertyName)
127                   readDescriptors += td
128                   // can just read the property directly as the read path accounts for it
129                   postReadTransforms += PropertyTransform(t.name, t.binding, readDescriptors.size - 1)
130                   val topLevelCol = ColumnName.encode(descriptor.getLocalName)
131                   if (descriptor.getJsonSchema().isEmpty) {
132                     addReadPath(topLevelCol)
133                   } else {
134                     val nested = path.elements.takeWhile(_.isInstanceOf[PathAttribute]).map(_.asInstanceOf[PathAttribute].name)
135                     addReadPath((Seq(topLevelCol) ++ nested).mkString("."))
136                   }
137                 } catch {
138                   case NonFatal(e) =>
139                     logger.warn("Error parsing json-path and evaluating read columns:", e)
140                     requiresRetype = true
141                     FilterHelper.propertyNames(t.expression, sft).map(sft.getDescriptor).foreach { descriptor =>
142                       readDescriptors += descriptor
143                       addReadPath(ColumnName.encode(descriptor.getLocalName))
144                     }
145                     postReadTransforms += t
146                 }
147 
148               case _ =>
149                 requiresRetype = true
150                 FilterHelper.propertyNames(t.expression, sft).map(sft.getDescriptor).foreach { descriptor =>
151                   readDescriptors += descriptor
152                   addReadPath(ColumnName.encode(descriptor.getLocalName))
153                 }
154                 postReadTransforms += t
155             }
156 
157           case t =>
158             throw new UnsupportedOperationException(s"An implementation is missing: ${t.getClass}")
159         }
160 
161         postReadFilter.foreach { filter =>
162           FilterHelper.propertyNames(filter, sft).foreach { p =>
163             val descriptor = if (p.startsWith("$")) { parseJsonPath(p, sft)._1 } else { sft.getDescriptor(p) }
164             if (!readDescriptors.exists(_.getLocalName == descriptor.getLocalName)) {
165               requiresRetype = true
166               readDescriptors += descriptor
167               addReadPath(ColumnName.encode(descriptor.getLocalName))
168             }
169           }
170         }
171 
172         if (requiresRetype) {
173           postReadTransform = Some(tsft -> postReadTransforms.toArray)
174         }
175     }
176 
177     filterCols.foreach(addReadPath)
178 
179     val readSchema = {
180       val fields = readCols.asScala.toSeq.map { case (name, children) =>
181         val field = schema.findField(name)
182         if (field == null) {
183           throw new IllegalArgumentException(s"Unexpected projection: $name")
184         }
185         if (children.isEmpty) {
186           field
187         } else {
188           val subFields = children.map { dot =>
189             val path = s"$name.$dot"
190             val subField = schema.findField(path)
191             if (subField == null) {
192               throw new IllegalArgumentException(s"Unexpected projection: $path")
193             }
194             dot -> subField
195           }
196           buildProjectedType(field, subFields)
197         }
198       }
199       val projection =
200         if (includeRowPositions) { fields ++ Seq(MetadataColumns.FILE_PATH, MetadataColumns.ROW_POSITION) } else { fields }
201       val ids = fields.collectFirst { case f if f.name() == FeatureIdField => Int.box(f.fieldId()) }
202       new Schema(projection.asJava, schema.getAliases, ids.toSet.asJava)
203     }
204 
205     val readSft = {
206       val readSftBuilder = new SimpleFeatureTypeBuilder()
207       readSftBuilder.setName(sft.getName)
208       readDescriptors.foreach(readSftBuilder.add)
209       val readSft = readSftBuilder.buildFeatureType()
210       readSft.getUserData.putAll(sft.getUserData)
211       readSft
212     }
213 
214     ReadSchema(new SimpleFeatureIcebergSchema(readSft, readSchema), postReadTransform)
215   }

217   /**
218    * Build a projected field that only contains the subfields that are passed in
219    *
220    * @param field base field
221    * @param subFields subfields to keep, along with the path to the field in dot-notation, may be multiple levels deep
222    * @return
223    */
224   private def buildProjectedType(field: NestedField, subFields: Seq[(String, NestedField)]): NestedField = {
225     if (subFields.isEmpty) {
226       return field
227     }
228     // group the requested paths by their first path segment
229     val grouped = subFields.groupBy { case (path, _) =>
230       val i = path.indexOf('.')
231       if (i == -1) { path } else { path.substring(0, i) }
232     }
233     // walk the original struct fields so we preserve their order, then keep only the requested children
234     val projected = field.`type`().asStructType().fields().asScala.flatMap { child =>
235       grouped.get(child.name()).map { group =>
236         // if the child was requested in full (a leaf path), keep its entire subtree
237         if (group.exists { case (path, _) => path.indexOf('.') == -1 }) {
238           child
239         } else {
240           val remaining = group.map { case (path, sub) => path.substring(path.indexOf('.') + 1) -> sub }
241           buildProjectedType(child, remaining)
242         }
243       }
244     }
245     // preserve the field id, name, doc and optionality, just narrowing the struct type
246     NestedField.from(field).ofType(StructType.of(projected.asJava)).build()
247   }

295   /**
296    * Get a schema based on a simple feature type. Encoding can be configured through `geomesa.parquet.geometries`.
297    *
298    * Note: this should only be called for creating a new schema, for an existing table the field ids will not be correct
299    *
300    * @param sft simple feature type
301    * @param geometries geometry encoding
302    * @return
303    */
304   def create(sft: SimpleFeatureType, geometries: GeometryEncoding): Schema = {
305     val builder = Seq.newBuilder[NestedField]
306     // note: we have to use unique field ids, but iceberg will throw them out and regenerate them when creating a table
307     val fieldIds = new AtomicInteger(1)
308     builder += NestedField.required(FeatureIdField).withId(fieldIds.getAndIncrement()).ofType(StringType.get()).build()
309     builder += buildField(VisibilitiesField, fieldIds.getAndIncrement(), null, StringType.get())
310     sft.getAttributeDescriptors.asScala.foreach { d =>
311       val name = ColumnName(d.getLocalName)
312       val objectType = ObjectType.selectType(d)
313       // no doc is written: an attribute spec does not fit in a column comment once it carries a
314       // structural type definition, so every attribute's spec is a table property - see columnSpecs
315       if (objectType.head == ObjectType.GEOMETRY) {
316         // TODO supports native geometry encoding
317         require(geometries == GeometryEncoding.GeoParquetWkb, "Only WKB encoding is supported for Geometry types")
318         // not yet supported in spark or trino: GeometryType.crs84()
319         builder += buildField(name.column, fieldIds.getAndIncrement(), null, BinaryType.get())
320         builder += BoundingBoxField.icebergSchema(name.column, fieldIds)
321         builder += ZValueField.icebergSchema(name.column, objectType(1), fieldIds)
322       } else if (objectType.last == ObjectType.JSON) {
323         builder +=
324           buildField(name.column, fieldIds.getAndIncrement(), null,
325             d.getJsonSchema().fold[Type](VariantType.get())(buildStructuralType(_, () => fieldIds.getAndIncrement())))
326       } else {
327         builder += buildField(name.column, fieldIds.getAndIncrement(), null, getType(objectType, fieldIds))
328       }
329     }
330 
331     new Schema(builder.result().asJava,  java.util.Set.of[Integer](1))
332   }

372       case ObjectType.DOUBLE  => DoubleType.get()
373       case ObjectType.LONG    => LongType.get()
374       case ObjectType.FLOAT   => FloatType.get()

381       case ObjectType.LIST =>

385       case ObjectType.MAP =>

387         val valueType = buildField("", fieldIds.getAndIncrement(), null, getType(bindings.slice(2, 3), fieldIds))
388         MapType.ofRequired(keyType.fieldId(), valueType.fieldId(), keyType.`type`(), valueType.`type`())
389 
390       case binding =>
391         throw new UnsupportedOperationException(s"No mapping defined for type: $binding")
392     }

395   private[core] def buildStructuralType(schemaDef: String, fieldIds: TypeUtil.NextID): Type = {
396     val avroSchema = new org.apache.avro.Schema.Parser().parse(schemaDef)
397     // wrap in a record so that we handle non-record schemas (arrays, maps), convert, then extract the single field.
398     // this mirrors the parquet conversion - a record schema just becomes a struct-typed field.
399     val wrapped =
400       org.apache.avro.SchemaBuilder.record("wrapper").namespace("tmp")
401         .fields().name("value").`type`(avroSchema).noDefault().endRecord()
402     val typed = AvroSchemaUtil.convert(wrapped).asStructType().fields().get(0).`type`()
403     // top-level scalars are not supported - just use a regular attribute in that case
404     if (typed.isPrimitiveType) {
405       throw new IllegalArgumentException(
406         s"Structural JSON schema must be a record, array, or map, but was a scalar type: $typed")
407     }
408     // re-stamp the field ids so that they don't overlap our other fields
409     TypeUtil.assignFreshIds(typed, fieldIds)
410   }
