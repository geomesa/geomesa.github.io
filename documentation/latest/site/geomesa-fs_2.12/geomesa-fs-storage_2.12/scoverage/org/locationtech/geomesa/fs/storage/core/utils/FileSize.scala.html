21 /**
22  * Utility for tracking target file sizes
23  *
24  * @param table table, not cleaned up, must be closed externally
25  */
26 class FileSize(table: Table) {
27 
28   private val fileSizeError = FileSize.FileSizeErrorThreshold.toFloat.get
29 
30   private var averageBytesPerFeature = Metadata.get(table, FileSize.BytesPerFeature) match {
31     case Some(b) => b.toFloat
32     case None    => table.schema().columns().size() * 1.6f // 1.6 taken from some sample data estimates...
33   }
34 
35   /**
36    * Gets the target file size
37    *
38    * @return
39    */
40   def targetSize: Option[Long] = Metadata.get(table, Metadata.TargetFileSize).map(_.toLong)
41 
42   /**
43    * Check if a file is already the desired size
44    *
45    * @param path file path
46    * @param target target file size
47    * @return true if the file is appropriately sized
48    */
49   def fileIsSized(path: URI, target: Long): Boolean = isSized(table.io().newInputFile(path.toString).getLength, target)
50 
51   /**
52    * Check if a file is already the desired size
53    *
54    * @param file file
55    * @param target target file size
56    * @return true if the file is appropriately sized
57    */
58   def fileIsSized(file: DataFile, target: Long): Boolean = isSized(file.fileSizeInBytes(), target)
59 
60   /**
61    * Compare target with actual size, allowing for some margin of error
62    *
63    * @param size file size
64    * @param target target size
65    * @return
66    */
67   private def isSized(size: Long, target: Long): Boolean = math.abs((size.toDouble / target) - 1d) <= fileSizeError
68 
69   /**
70    * Gets a file size estimator for this storage instance
71    *
72    * @param size target file size
73    * @return
74    */
75   def estimator(size: Long): UpdatingFileSizeEstimator =
76     FileSize.this.synchronized {
77       new UpdatingFileSizeEstimator(size, this)
78     }
79 
80   private def updateFileSize(bytesPerFeature: Float): Unit = {
81     if (Metadata.get(table, FileSize.UseDynamicSizing).forall(_.toBoolean)) {
82       synchronized {
83         if (math.abs((bytesPerFeature / averageBytesPerFeature) - 1f) > fileSizeError) {
84           averageBytesPerFeature = bytesPerFeature
85           Metadata.set(table, FileSize.BytesPerFeature, java.lang.Float.toString(bytesPerFeature))
86         }
87       }
88     }
89   }
90 }
91 
92 object FileSize {
93 
94   val BytesPerFeature  = "bytes-per-feature"
95   val UseDynamicSizing = "use-dynamic-sizing"
96 
97   val FileSizeErrorThreshold: SystemProperty = SystemProperty("geomesa.fs.size.threshold", "0.05")
98 
99   class UpdatingFileSizeEstimator(target: Long, instance: FileSize)
100       extends FileSizeEstimator(target, instance.fileSizeError, instance.averageBytesPerFeature) with Closeable {
101     override def close(): Unit = getBytesPerFeature.foreach(instance.updateFileSize)
102   }
103 }
