28 /**
29  * Reads parquet files based on an iceberg table scan
30  *
31  * @param table table to scan
32  * @param schema read schema
33  * @param filter filter
34  * @param threads number of threads used to execute
35  * @param fileFilter optional filter for restricting the files that are scanned
36  */
37 class IcebergParquetScan(
38     table: Table,
39     schema: SimpleFeatureIcebergSchema,
40     filter: Expression,
41     threads: Int,
42     fileFilter: Option[String => Boolean] = None
43   ) extends CloseableIterator[SimpleFeature] with LazyLogging {
44 
45   import scala.collection.JavaConverters._
46 
47   private val queue = new LinkedBlockingQueue[StructLike](10000)
48   private val closed = new AtomicBoolean(false)
49 
50   private val row = StructSimpleFeature(schema)
51 
52   // WithClose here shuts down the executor service, but already submitted tasks will still be executed
53   // note: we have to be explicit, as starting with Java 19 ExecutorService extends AutoCloseable which means two implicits match
54   private implicit val exIsCloseable: ExecutorServiceIsCloseable = IsCloseable.executorServiceIsCloseable
55   private val ex =
56     WithClose(new CachedThreadPool(threads)) { pool =>
57       // note: exclude z2 cols even if there's no transform
58       val scan = table.newScan().caseSensitive(false).project(schema.schema).filter(filter)
59       try {
60         WithClose(scan.planTasks()) { plan =>
61           WithClose(plan.iterator()) { tasks =>
62             var taskCount = 0
63             var fileCount = 0
64             logger.debug("Submitting tasks")
65             tasks.forEachRemaining { task =>
66               logger.trace(s"Submitting task with ${task.filesCount()} file(s)")
67               pool.submit(new TaskRunnable(task))
68               taskCount += 1
69               fileCount += task.filesCount()
70             }
71             logger.debug(s"Submitted $taskCount tasks (scanning $fileCount total files) using $threads threads")
72           }
73         }
74       } finally {
75         CloseWithLogging(Option(scan).collect { case c: Closeable => c })
76       }
77       pool
78     }
79 
80   private var current: StructLike = _
81 
82   override def hasNext: Boolean = {
83     if (current != null) {
84       return true
85     }
86     current = queue.poll()
87     if (current != null) {
88       return true
89     }
90 
91     if (closed.get()) {
92       false
93     } else if (ex.isTerminated) {
94       current = queue.poll()
95       current != null
96     } else {
97       while (current == null && !ex.isTerminated) {
98         current = queue.poll(100, TimeUnit.MILLISECONDS)
99       }
100       current != null
101     }
102   }
103 
104   override def next(): SimpleFeature = {
105     if (hasNext) {
106       row.setRow(current)
107       current = null
108       row
109     } else {
110       Iterator.empty.next
111     }
112   }
113 
114   override def close(): Unit = {
115     if (closed.compareAndSet(false, true)) {
116       ex.shutdownNow()
117       ex.awaitTermination(1, TimeUnit.SECONDS)
118     }
119   }
120 
121   private def readFile(task: FileScanTask, projection: Schema): CloseableIterator[StructLike] = {
122     val inputFile = table.io().newInputFile(task.file())
123     if (fileFilter.exists(_.apply(inputFile.location()) == false)) {
124       logger.debug(s"Skipping file ${inputFile.location()} [${task.start()}:${task.length()}] due to file filter")
125       CloseableIterator.empty[StructLike]
126     } else {
127       logger.debug(s"Reading file ${inputFile.location()} [${task.start()}:${task.length()}] with filter: ${task.residual()}")
128       // we have to pass in the file path as a constant
129       val idToConstant = java.util.Map.of[Integer, Any](MetadataColumns.FILE_PATH_COLUMN_ID, inputFile.location())
130       // note: can't reuse containers b/c we put everything onto the queue - reading row-by-row with reused containers is much slower
131       val reader =
132         Parquet.read(inputFile)
133           .project(projection)
134           .split(task.start(), task.length())
135           .caseSensitive(false)
136           .filter(task.residual())
137           .createReaderFunc(fileSchema => GenericParquetReaders.buildReader(projection, fileSchema, idToConstant))
138           .build[Record]()
139       val iter = reader.iterator()
140       CloseableIterator(iter.asScala, CloseWithLogging(Seq(iter, reader)))
141     }
142   }
143 
144   private class TaskRunnable(val task: CombinedScanTask) extends Runnable {
145     override def run(): Unit = {
146       try {
147         task.files().iterator().asScala.foreach { file =>
148           if (!closed.get()) {
149             val deleteFilter = if (file.deletes().isEmpty) { None } else {
150               val filter = new GenericDeleteFilter(table.io(), file, table.schema(), schema.schema)
151               require(!filter.hasEqDeletes,
152                 s"Only positional deletes are supported, but got equality deletes: ${file.deletes().asScala.map(_.location()).mkString(", ")}")
153               Some(filter).filter(_.hasPosDeletes)
154             }
155             val projection = deleteFilter.fold(schema.schema)(_.requiredSchema())
156             WithClose(readFile(file, projection)) { iter =>
157               val withDeletes = deleteFilter.fold(iter) { deletes =>
158                 val index = deletes.deletedRowPositions()
159                 val position = projection.accessorForField(MetadataColumns.ROW_POSITION.fieldId())
160                 iter.filterNot(r => index.isDeleted(position.get(r).asInstanceOf[java.lang.Long]))
161               }
162               val residual = file.residual()
163               val filtered = if (residual == null || residual == Expressions.alwaysTrue()) { withDeletes } else {
164                 val filter = new Evaluator(projection.asStruct(), residual, false)
165                 val wrapper = new InternalRecordWrapper(projection.asStruct())
166                 withDeletes.filter(r => filter.eval(wrapper.wrap(r)))
167               }
168               filtered.foreach(queue.put)
169             }
170           }
171         }
172       } catch {
173         case NonFatal(e) => logger.error("Error running scan task:", e)
174       }
175     }
176   }
