32  * @param table table to scan

35  * @param threads number of threads used to execute

63       snapshotId.foreach(scan.useSnapshot)

80                     }
81                     res
82                   }
83               }
84               if (filteredTasks.nonEmpty) {
85                 logger.trace(s"Submitting task with ${filteredTasks.size} file(s)")
86                 pool.submit(new TaskRunnable(filteredTasks))
87                 taskCount += 1
88                 fileCount += filteredTasks.size
89               }
90             }
91             logger.debug(s"Submitted $taskCount tasks (scanning $fileCount total files) using $threads threads")
92           }
93         }
94       } finally {
95         CloseWithLogging(Option(scan).collect { case c: Closeable => c })
96       }

98     }

144   private def readFile(task: FileScanTask, projection: Schema): CloseableIterator[StructLike] = {
145     logger.debug(s"Reading file ${task.file().location()} [${task.start()}:${task.length()}] with filter: ${task.residual()}")
146     val inputFile = table.io().newInputFile(task.file())
147     // we have to pass in the file path as a constant
148     val idToConstant = java.util.Map.of[Integer, Any](MetadataColumns.FILE_PATH_COLUMN_ID, inputFile.location())
149     // note: can't reuse containers b/c we put everything onto the queue - reading row-by-row with reused containers is much slower
150     val reader =
151       Parquet.read(inputFile)
152         .project(projection)
153         .split(task.start(), task.length())
154         .caseSensitive(false)
155         .filter(task.residual())
156         .createReaderFunc(fileSchema => GenericParquetReaders.buildReader(projection, fileSchema, idToConstant))
157         .build[Record]()
158     val iter = reader.iterator()
159     CloseableIterator(iter.asScala, CloseWithLogging(Seq(iter, reader)))
160   }

162   private def withResidualColumns(projection: Schema, residual: Expression): Schema = {
163     if (residual == null) {
164       return projection
165     }
166     val columns = projection.columns().asScala
167     val columnNames = columns.map(_.name()).toSet
168     val toAdd = ReferenceVisitor(residual).flatMap { reference =>
169       val column = reference.takeWhile(_ != '.')
170       if (columnNames.contains(column)) { None } else { Option(table.schema().findField(column)) }
171     }
172     if (toAdd.isEmpty) {
173       projection
174     } else {
175       new Schema((columns ++ toAdd).asJava, projection.getAliases, projection.identifierFieldIds())
176     }
177   }

179   private class TaskRunnable(tasks: Seq[FileScanTask]) extends Runnable {
180     override def run(): Unit = {
181       try {
182         tasks.foreach { task =>
183           if (!closed.get()) {
184             val deleteFilter = if (task.deletes().isEmpty) { None } else {
185               val filter = new GenericDeleteFilter(table.io(), task, table.schema(), schema.schema)
186               require(!filter.hasEqDeletes,
187                 s"Only positional deletes are supported, but got equality deletes: ${task.deletes().asScala.map(_.location()).mkString(", ")}")
188               Some(filter).filter(_.hasPosDeletes)
189             }
190             val residual = task.residual()
191             // we sometimes get residuals that aren't part of the filter, e.g. due to partition spec changes - add them here
192             val projection = withResidualColumns(deleteFilter.fold(schema.schema)(_.requiredSchema()), residual)
193             WithClose(readFile(task, projection)) { iter =>
194               val withDeletes = deleteFilter.fold(iter) { deletes =>
195                 val index = deletes.deletedRowPositions()
196                 val position = projection.accessorForField(MetadataColumns.ROW_POSITION.fieldId())
197                 iter.filterNot(r => index.isDeleted(position.get(r).asInstanceOf[java.lang.Long]))
198               }
199               val filtered = if (residual == null || residual == Expressions.alwaysTrue()) { withDeletes } else {
200                 val filter = new Evaluator(projection.asStruct(), residual, false)
201                 val wrapper = new InternalRecordWrapper(projection.asStruct())
202                 withDeletes.filter(r => filter.eval(wrapper.wrap(r)))
203               }
204               filtered.foreach(queue.put)
205             }
206           }
207         }
208       } catch {
209         case e: Throwable =>
210           // submit captures all throwables in its Future. Publish even fatal failures to the reader so
211           // a failed task cannot appear to be an empty input (and cause compaction to discard its rows).
212           failure.compareAndSet(null, e)
213           if (NonFatal(e)) {
214             logger.error("Error running scan task:", e)
215           } else {
216             throw e
217           }
218       }
219     }
220   }
