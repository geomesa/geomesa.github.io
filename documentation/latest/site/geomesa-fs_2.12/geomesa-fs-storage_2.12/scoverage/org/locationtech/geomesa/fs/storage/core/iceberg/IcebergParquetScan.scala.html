29 /**
30  * Reads parquet files based on an iceberg table scan
31  *
32  * @param table table to scan
33  * @param schema read schema
34  * @param filter filter
35  * @param threads number of threads used to execute
36  * @param fileFilter optional filter for restricting the files that are scanned
37  * @param snapshotId optional snapshot to pin the scan to
38  */
39 class IcebergParquetScan(
40     table: Table,
41     schema: SimpleFeatureIcebergSchema,
42     filter: Expression,
43     threads: Int,
44     fileFilter: Option[String => Boolean] = None,
45     snapshotId: Option[Long] = None

59   private val ex =
60     WithClose(new CachedThreadPool(threads)) { pool =>
61       // note: exclude z2 cols even if there's no transform
62       // note: scans are immutable, each refinement returns a new scan
63       val scan = snapshotId.foldLeft(table.newScan().caseSensitive(false).project(schema.schema).filter(filter)) {
64         case (scan, id) => scan.useSnapshot(id)
65       }
66       try {
67         WithClose(scan.planTasks()) { plan =>
68           WithClose(plan.iterator()) { tasks =>
69             var taskCount = 0
70             var fileCount = 0
71             logger.debug("Submitting tasks")
72             tasks.forEachRemaining { task =>
73               val fileTasks = task.files().asScala.toSeq
74               val filteredTasks = fileFilter match {
75                 case None => fileTasks
76                 case Some(filt) =>
77                   fileTasks.filter { fileTask =>
78                     val res = filt.apply(fileTask.file().location())
79                     if (!res) {
80                       logger.debug(
81                         s"Skipping file ${fileTask.file().location()} [${fileTask.start()}:${fileTask.length()}] due to file filter")
82                     }
83                     res
84                   }
85               }
86               if (filteredTasks.nonEmpty) {
87                 logger.trace(s"Submitting task with ${filteredTasks.size} file(s)")
88                 pool.submit(new TaskRunnable(filteredTasks))
89                 taskCount += 1
90                 fileCount += filteredTasks.size
91               }
92             }
93             logger.debug(s"Submitted $taskCount tasks (scanning $fileCount total files) using $threads threads")
94           }
95         }
96       } finally {
97         CloseWithLogging(Option(scan).collect { case c: Closeable => c })
98       }
99       pool
100     }

104   override def hasNext: Boolean = {
105     Option(failure.get()).foreach(e => throw new RuntimeException("Error reading Iceberg data", e))
106     if (current != null) {

114     if (closed.get()) {
115       false
116     } else if (ex.isTerminated) {

121       while (current == null && !ex.isTerminated) {
122         current = queue.poll(100, TimeUnit.MILLISECONDS)

146   private def readFile(task: FileScanTask, projection: Schema): CloseableIterator[StructLike] = {
147     logger.debug(s"Reading file ${task.file().location()} [${task.start()}:${task.length()}] with filter: ${task.residual()}")
148     val inputFile = table.io().newInputFile(task.file())
149     // we have to pass in the file path as a constant
150     val idToConstant = java.util.Map.of[Integer, Any](MetadataColumns.FILE_PATH_COLUMN_ID, inputFile.location())
151     // note: can't reuse containers b/c we put everything onto the queue - reading row-by-row with reused containers is much slower
152     val reader =
153       Parquet.read(inputFile)
154         .project(projection)
155         .split(task.start(), task.length())
156         .caseSensitive(false)
157         .filter(task.residual())
158         .createReaderFunc(fileSchema => GenericParquetReaders.buildReader(projection, fileSchema, idToConstant))
159         .build[Record]()
160     val iter = reader.iterator()
161     CloseableIterator(iter.asScala, CloseWithLogging(Seq(iter, reader)))
162   }

164   private def withResidualColumns(projection: Schema, residual: Expression): Schema = {
165     if (residual == null) {
166       return projection
167     }
168     val columns = projection.columns().asScala
169     val columnNames = columns.map(_.name()).toSet
170     val toAdd = ReferenceVisitor(residual).flatMap { reference =>
171       val column = reference.takeWhile(_ != '.')
172       if (columnNames.contains(column)) { None } else { Option(table.schema().findField(column)) }
173     }
174     if (toAdd.isEmpty) {
175       projection
176     } else {
177       new Schema((columns ++ toAdd).asJava, projection.getAliases, projection.identifierFieldIds())
178     }
179   }

181   private class TaskRunnable(tasks: Seq[FileScanTask]) extends Runnable {
182     override def run(): Unit = {
183       try {
184         tasks.foreach { task =>
185           if (!closed.get()) {
186             val deleteFilter = if (task.deletes().isEmpty) { None } else {
187               val filter = new GenericDeleteFilter(table.io(), task, table.schema(), schema.schema)
188               require(!filter.hasEqDeletes,
189                 s"Only positional deletes are supported, but got equality deletes: ${task.deletes().asScala.map(_.location()).mkString(", ")}")
190               Some(filter).filter(_.hasPosDeletes)
191             }
192             val residual = task.residual()
193             // we sometimes get residuals that aren't part of the filter, e.g. due to partition spec changes - add them here
194             val projection = withResidualColumns(deleteFilter.fold(schema.schema)(_.requiredSchema()), residual)
195             WithClose(readFile(task, projection)) { iter =>
196               val withDeletes = deleteFilter.fold(iter) { deletes =>
197                 val index = deletes.deletedRowPositions()
198                 val position = projection.accessorForField(MetadataColumns.ROW_POSITION.fieldId())
199                 iter.filterNot(r => index.isDeleted(position.get(r).asInstanceOf[java.lang.Long]))
200               }
201               val filtered = if (residual == null || residual == Expressions.alwaysTrue()) { withDeletes } else {
202                 val filter = new Evaluator(projection.asStruct(), residual, false)
203                 val wrapper = new InternalRecordWrapper(projection.asStruct())
204                 withDeletes.filter(r => filter.eval(wrapper.wrap(r)))
205               }
206               filtered.foreach(queue.put)
207             }
208           }
209         }
210       } catch {
211         case e: Throwable =>
212           // submit captures all throwables in its Future. Publish even fatal failures to the reader so
213           // a failed task cannot appear to be an empty input (and cause compaction to discard its rows).
214           failure.compareAndSet(null, e)
215           if (NonFatal(e)) {
216             logger.error("Error running scan task:", e)
217           } else {
218             throw e
219           }
220       }
221     }
222   }
