29 /**
30  * Reads parquet files based on an iceberg table scan
31  *
32  * @param scan scan
33  * @param threads number of threads used to execute
34  * @param fileFilter optional filter for restricting the files that are scanned
35  */
36 class IcebergParquetScan(scan: TableScan, threads: Int, fileFilter: Option[String => Boolean] = None)
37     extends CloseableIterator[Record] with LazyLogging {
38 
39   import scala.collection.JavaConverters._
40 
41   private val sharedQueue = new LinkedBlockingQueue[Record](2000000)
42   private val localQueue = new java.util.LinkedList[Record]()
43 
44   private val projection = scan.schema()
45   private lazy val deleteProjection =
46     if (projection.findField(MetadataColumns.ROW_POSITION.fieldId()) != null) { projection } else {
47       TypeUtil.join(projection, new Schema(MetadataColumns.ROW_POSITION))
48     }
49 
50   private val caseSensitive = scan.isCaseSensitive
51 
52   private val closed = new AtomicBoolean(false)
53 
54   private val ex = new CachedThreadPool(threads)
55   private val tasks = scan.planTasks()
56 
57   var i = 0
58   logger.debug("Submitting tasks")
59   tasks.forEach { task =>
60     logger.trace(s"Submitting task: $task")
61     ex.submit(new TaskRunnable(task))
62     i += 1
63   }
64   logger.debug(s"Submitted $i tasks, using $threads threads")
65   ex.shutdown()
66 
67   private var current: Record = _
68 
69   override def hasNext: Boolean = {
70     if (current != null) {
71       return true
72     }
73     current = localQueue.pollFirst()
74     if (current != null) {
75       return true
76     }
77 
78     while (!ex.isTerminated) {
79       current = sharedQueue.poll(100, TimeUnit.MILLISECONDS)
80       if (current != null) {
81         sharedQueue.drainTo(localQueue, 10000)
82         return true
83       }
84     }
85 
86     // last check - if ex.isTerminated, the queue should have whatever values are left
87     current = sharedQueue.poll()
88     if (current != null) {
89       sharedQueue.drainTo(localQueue, 10000)
90       true
91     } else {
92       false
93     }
94   }
95 
96   override def next(): Record = {
97     if (hasNext) {
98       val ret = current
99       current = null
100       ret
101     } else {
102       Iterator.empty.next
103     }
104   }
105 
106   override def close(): Unit = {
107     if (closed.compareAndSet(false, true)) {
108       try {
109         ex.shutdownNow()
110         ex.awaitTermination(2, TimeUnit.SECONDS)
111       } finally {
112         CloseWithLogging(Seq(tasks) ++ Option(scan).collect { case c: Closeable => c })
113       }
114     }
115   }
116 
117   private def readFile(task: FileScanTask, projection: Schema): CloseableIterator[Record] = {
118     val inputFile = scan.table().io().newInputFile(task.file())
119     if (fileFilter.exists(_.apply(inputFile.location()) == false)) {
120       logger.debug(s"Skipping file ${inputFile.location()} [${task.start()}:${task.length()}] due to file filter")
121       CloseableIterator.empty[Record]
122     } else {
123       logger.debug(s"Reading file ${inputFile.location()} [${task.start()}:${task.length()}]")
124       // we have to pass in the file path as a constant
125       val idToConstant = java.util.Map.of[Integer, Any](MetadataColumns.FILE_PATH_COLUMN_ID, inputFile.location())
126       val reader =
127         Parquet.read(inputFile)
128           .project(projection)
129           .split(task.start(), task.length())
130           .caseSensitive(caseSensitive)
131           .filter(task.residual())
132           // TODO implement ParquetValueReader directly instead of using records
133           .createReaderFunc(fileSchema => GenericParquetReaders.buildReader(projection, fileSchema, idToConstant))
134           .build[Record]()
135       val iter = reader.iterator()
136       CloseableIterator(iter.asScala, CloseWithLogging(Seq(iter, reader)))
137     }
138   }
139 
140   private def readDeleteFile(delete: DeleteFile, dataFilePath: String): PositionDeleteIndex = {
141     require(delete.content() == FileContent.POSITION_DELETES,
142       s"Only positional deletes are supported, but got: ${delete.content()}")
143     val deleteFileSchema = DeleteSchemaUtil.pathPosSchema()
144     val builder =
145       Parquet.read(scan.table().io().newInputFile(delete))
146         .project(deleteFileSchema)
147         .createReaderFunc(fileSchema => GenericParquetReaders.buildReader(deleteFileSchema, fileSchema))
148         .filter(Expressions.equal(MetadataColumns.DELETE_FILE_PATH.name(), dataFilePath))
149     WithClose(builder.build()) { deletes =>
150       Deletes.toPositionIndex(dataFilePath, deletes, delete)
151     }
152   }
153 
154   private class TaskRunnable(task: CombinedScanTask) extends Runnable {
155     override def run(): Unit = {
156       try {
157         task.files().iterator().asScala.foreach { file =>
158           if (!closed.get()) {
159             val deleteIndex = if (file.deletes().isEmpty) { None } else {
160               val deletes = file.deletes().asScala.map(readDeleteFile(_, file.file().location()))
161               Some(PositionDeleteIndexUtil.merge(deletes.asJava))
162             }
163             val projection = if (deleteIndex.isEmpty) { IcebergParquetScan.this.projection } else { deleteProjection }
164 
165             WithClose(readFile(file, projection)) { iter =>
166               lazy val wrapper = new InternalRecordWrapper(projection.asStruct())
167               val withDeletes = deleteIndex.fold(iter) { index =>
168                 val position = projection.accessorForField(MetadataColumns.ROW_POSITION.fieldId())
169                 iter.filterNot(r => index.isDeleted(position.get(wrapper.wrap(r)).asInstanceOf[java.lang.Long]))
170               }
171               val residual = file.residual()
172               val filtered = if (residual == null || residual == Expressions.alwaysTrue()) { withDeletes } else {
173                 val filter = new Evaluator(projection.asStruct(), residual, caseSensitive)
174                 withDeletes.filter(r => filter.eval(wrapper.wrap(r)))
175               }
176               filtered.foreach(sharedQueue.put)
177             }
178           }
179         }
180       } catch {
181         case NonFatal(e) => logger.error("Error running scan task:", e)
182       }
183     }
184   }
