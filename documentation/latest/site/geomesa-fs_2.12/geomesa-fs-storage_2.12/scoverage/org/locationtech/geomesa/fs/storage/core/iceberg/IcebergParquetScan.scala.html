31 /**
32  * Reads parquet files based on an iceberg table scan
33  *
34  * @param table table to scan
35  * @param schema read schema
36  * @param filter filter
37  * @param threads number of threads used to execute
38  * @param fileFilter optional filter for restricting the files that are scanned
39  */
40 class IcebergParquetScan(
41     table: Table,
42     schema: SimpleFeatureIcebergSchema,
43     filter: Expression,
44     threads: Int,
45     fileFilter: Option[String => Boolean] = None
46   ) extends CloseableIterator[SimpleFeature] with LazyLogging {
47 
48   import scala.collection.JavaConverters._

50   private val queue = new LinkedBlockingQueue[StructLike](10000)
51   private val closed = new AtomicBoolean(false)
52 
53   private lazy val deleteProjection =
54     if (schema.schema.findField(MetadataColumns.ROW_POSITION.fieldId()) != null) { schema.schema } else {
55       TypeUtil.join(schema.schema, new Schema(MetadataColumns.ROW_POSITION))
56     }
57 
58   private val row = StructSimpleFeature(schema)
59 
60   // WithClose here shuts down the executor service, but already submitted tasks will still be executed
61   // note: we have to be explicit, as starting with Java 19 ExecutorService extends AutoCloseable which means two implicits match
62   private implicit val exIsCloseable: ExecutorServiceIsCloseable = IsCloseable.executorServiceIsCloseable
63   private val ex =
64     WithClose(new CachedThreadPool(threads)) { pool =>
65       // note: exclude z2 cols even if there's no transform
66       val scan = table.newScan().caseSensitive(false).project(schema.schema).filter(filter)
67       try {
68         WithClose(scan.planTasks()) { plan =>
69           WithClose(plan.iterator()) { tasks =>
70             var taskCount = 0
71             var fileCount = 0
72             logger.debug("Submitting tasks")
73             tasks.forEachRemaining { task =>
74               logger.trace(s"Submitting task with ${task.filesCount()} file(s)")
75               pool.submit(new TaskRunnable(task))
76               taskCount += 1
77               fileCount += task.filesCount()
78             }
79             logger.debug(s"Submitted $taskCount tasks (scanning $fileCount total files) using $threads threads")
80           }
81         }
82       } finally {
83         CloseWithLogging(Option(scan).collect { case c: Closeable => c })
84       }
85       pool
86     }

90   override def hasNext: Boolean = {
91     if (current != null) {
92       return true
93     }
94     current = queue.poll()
95     if (current != null) {
96       return true
97     }
98 
99     if (closed.get()) {
100       false
101     } else if (ex.isTerminated) {
102       current = queue.poll()
103       current != null
104     } else {
105       while (current == null && !ex.isTerminated) {
106         current = queue.poll(100, TimeUnit.MILLISECONDS)
107       }
108       current != null
109     }
110   }

112   override def next(): SimpleFeature = {
113     if (hasNext) {
114       row.setRow(current)
115       current = null
116       row
117     } else {
118       Iterator.empty.next
119     }
120   }

122   override def close(): Unit = {
123     if (closed.compareAndSet(false, true)) {
124       ex.shutdownNow()
125       ex.awaitTermination(1, TimeUnit.SECONDS)
126     }
127   }

129   private def readFile(task: FileScanTask, projection: Schema): CloseableIterator[StructLike] = {
130     val inputFile = table.io().newInputFile(task.file())
131     if (fileFilter.exists(_.apply(inputFile.location()) == false)) {
132       logger.debug(s"Skipping file ${inputFile.location()} [${task.start()}:${task.length()}] due to file filter")
133       CloseableIterator.empty[StructLike]
134     } else {
135       logger.debug(s"Reading file ${inputFile.location()} [${task.start()}:${task.length()}] with filter: ${task.residual()}")
136       // we have to pass in the file path as a constant
137       val idToConstant = java.util.Map.of[Integer, Any](MetadataColumns.FILE_PATH_COLUMN_ID, inputFile.location())
138       // note: can't reuse containers b/c we put everything onto the queue - reading row-by-row with reused containers is much slower
139       val reader =
140         Parquet.read(inputFile)
141           .project(projection)
142           .split(task.start(), task.length())
143           .caseSensitive(false)
144           .filter(task.residual())
145           .createReaderFunc(fileSchema => GenericParquetReaders.buildReader(projection, fileSchema, idToConstant))
146           .build[Record]()
147       val iter = reader.iterator()
148       CloseableIterator(iter.asScala, CloseWithLogging(Seq(iter, reader)))
149     }
150   }
151 
152   private def readDeleteFile(delete: DeleteFile, dataFilePath: String): PositionDeleteIndex = {
153     require(delete.content() == FileContent.POSITION_DELETES,
154       s"Only positional deletes are supported, but got: ${delete.content()}")
155     logger.debug(s"Reading delete file ${delete.location()} [${delete.contentSizeInBytes()}]")
156     val deleteFileSchema = DeleteSchemaUtil.pathPosSchema()
157     val builder =
158       Parquet.read(table.io().newInputFile(delete))
159         .project(deleteFileSchema)
160         .createReaderFunc(fileSchema => GenericParquetReaders.buildReader(deleteFileSchema, fileSchema))
161         .filter(Expressions.equal(MetadataColumns.DELETE_FILE_PATH.name(), dataFilePath))
162     WithClose(builder.build()) { deletes =>
163       Deletes.toPositionIndex(dataFilePath, deletes, delete)
164     }
165   }
166 
167   private class TaskRunnable(val task: CombinedScanTask) extends Runnable {
168     override def run(): Unit = {
169       try {
170         task.files().iterator().asScala.foreach { file =>
171           if (!closed.get()) {
172             val deleteIndex = if (file.deletes().isEmpty) { None } else {
173               val deletes = file.deletes().asScala.map(readDeleteFile(_, file.file().location()))
174               Some(PositionDeleteIndexUtil.merge(deletes.asJava))
175             }
176             val projection = if (deleteIndex.isEmpty) { schema.schema } else { deleteProjection }
177 
178             WithClose(readFile(file, projection)) { iter =>
179               val withDeletes = deleteIndex.fold(iter) { index =>
180                 val position = projection.accessorForField(MetadataColumns.ROW_POSITION.fieldId())
181                 iter.filterNot(r => index.isDeleted(position.get(r).asInstanceOf[java.lang.Long]))
182               }
183               val residual = file.residual()
184               val filtered = if (residual == null || residual == Expressions.alwaysTrue()) { withDeletes } else {
185                 val filter = new Evaluator(projection.asStruct(), residual, false)
186                 val wrapper = new InternalRecordWrapper(projection.asStruct())
187                 withDeletes.filter(r => filter.eval(wrapper.wrap(r)))
188               }
189               filtered.foreach(queue.put)
190             }
191           }
192         }
193       } catch {
194         case NonFatal(e) => logger.error("Error running scan task:", e)
195       }
196     }
197   }
