28 /**
29  * Reads parquet files based on an iceberg table scan
30  *
31  * @param table table to scan
32  * @param schema read schema
33  * @param filter filter
34  * @param threads number of threads used to execute
35  * @param fileFilter optional filter for restricting the files that are scanned
36  * @param snapshotId optional snapshot to pin the scan to
37  */
38 class IcebergParquetScan(
39     table: Table,
40     schema: SimpleFeatureIcebergSchema,
41     filter: Expression,
42     threads: Int,
43     fileFilter: Option[String => Boolean] = None,
44     snapshotId: Option[Long] = None

58   private val ex =
59     WithClose(new CachedThreadPool(threads)) { pool =>
60       // note: exclude z2 cols even if there's no transform
61       val scan = table.newScan().caseSensitive(false).project(schema.schema).filter(filter)
62       snapshotId.foreach(scan.useSnapshot)
63       try {
64         WithClose(scan.planTasks()) { plan =>
65           WithClose(plan.iterator()) { tasks =>
66             var taskCount = 0
67             var fileCount = 0
68             logger.debug("Submitting tasks")
69             tasks.forEachRemaining { task =>
70               val fileTasks = task.files().asScala.toSeq
71               val filteredTasks = fileFilter match {
72                 case None => fileTasks
73                 case Some(filt) =>
74                   fileTasks.filter { fileTask =>
75                     val res = filt.apply(fileTask.file().location())
76                     if (!res) {
77                       logger.debug(
78                         s"Skipping file ${fileTask.file().location()} [${fileTask.start()}:${fileTask.length()}] due to file filter")
79                     }
80                     res
81                   }
82               }
83               if (filteredTasks.nonEmpty) {
84                 logger.trace(s"Submitting task with ${filteredTasks.size} file(s)")
85                 pool.submit(new TaskRunnable(filteredTasks))
86                 taskCount += 1
87                 fileCount += filteredTasks.size
88               }
89             }
90             logger.debug(s"Submitted $taskCount tasks (scanning $fileCount total files) using $threads threads")
91           }
92         }
93       } finally {
94         CloseWithLogging(Option(scan).collect { case c: Closeable => c })
95       }
96       pool
97     }

101   override def hasNext: Boolean = {
102     Option(failure.get()).foreach(e => throw new RuntimeException("Error reading Iceberg data", e))
103     if (current != null) {
104       return true
105     }
106     current = queue.poll()
107     if (current != null) {
108       return true
109     }
110 
111     if (closed.get()) {
112       false
113     } else if (ex.isTerminated) {
114       Option(failure.get()).foreach(e => throw new RuntimeException("Error reading Iceberg data", e))
115       current = queue.poll()
116       current != null
117     } else {
118       while (current == null && !ex.isTerminated) {
119         current = queue.poll(100, TimeUnit.MILLISECONDS)
120       }
121       Option(failure.get()).foreach(e => throw new RuntimeException("Error reading Iceberg data", e))
122       current != null
123     }
124   }

143   private def readFile(task: FileScanTask, projection: Schema): CloseableIterator[StructLike] = {
144     logger.debug(s"Reading file ${task.file().location()} [${task.start()}:${task.length()}] with filter: ${task.residual()}")
145     val inputFile = table.io().newInputFile(task.file())
146     // we have to pass in the file path as a constant
147     val idToConstant = java.util.Map.of[Integer, Any](MetadataColumns.FILE_PATH_COLUMN_ID, inputFile.location())
148     // note: can't reuse containers b/c we put everything onto the queue - reading row-by-row with reused containers is much slower
149     val reader =
150       Parquet.read(inputFile)
151         .project(projection)
152         .split(task.start(), task.length())
153         .caseSensitive(false)
154         .filter(task.residual())
155         .createReaderFunc(fileSchema => GenericParquetReaders.buildReader(projection, fileSchema, idToConstant))
156         .build[Record]()
157     val iter = reader.iterator()
158     CloseableIterator(iter.asScala, CloseWithLogging(Seq(iter, reader)))
159   }

161   private class TaskRunnable(tasks: Seq[FileScanTask]) extends Runnable {
162     override def run(): Unit = {
163       try {
164         tasks.foreach { task =>
165           if (!closed.get()) {
166             val deleteFilter = if (task.deletes().isEmpty) { None } else {
167               val filter = new GenericDeleteFilter(table.io(), task, table.schema(), schema.schema)
168               require(!filter.hasEqDeletes,
169                 s"Only positional deletes are supported, but got equality deletes: ${task.deletes().asScala.map(_.location()).mkString(", ")}")
170               Some(filter).filter(_.hasPosDeletes)
171             }
172             val projection = deleteFilter.fold(schema.schema)(_.requiredSchema())
173             WithClose(readFile(task, projection)) { iter =>
174               val withDeletes = deleteFilter.fold(iter) { deletes =>
175                 val index = deletes.deletedRowPositions()
176                 val position = projection.accessorForField(MetadataColumns.ROW_POSITION.fieldId())
177                 iter.filterNot(r => index.isDeleted(position.get(r).asInstanceOf[java.lang.Long]))
178               }
179               val residual = task.residual()
180               val filtered = if (residual == null || residual == Expressions.alwaysTrue()) { withDeletes } else {
181                 val filter = new Evaluator(projection.asStruct(), residual, false)
182                 val wrapper = new InternalRecordWrapper(projection.asStruct())
183                 withDeletes.filter(r => filter.eval(wrapper.wrap(r)))
184               }
185               filtered.foreach(queue.put)
186             }
187           }
188         }
189       } catch {
190         case e: Throwable =>
191           // submit captures all throwables in its Future. Publish even fatal failures to the reader so
192           // a failed task cannot appear to be an empty input (and cause compaction to discard its rows).
193           failure.compareAndSet(null, e)
194           if (NonFatal(e)) {
195             logger.error("Error running scan task:", e)
196           } else {
197             throw e
198           }
199       }
200     }
201   }
