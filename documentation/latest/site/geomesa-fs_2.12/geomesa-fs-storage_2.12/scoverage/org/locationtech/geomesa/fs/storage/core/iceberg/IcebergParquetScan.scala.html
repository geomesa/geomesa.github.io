28 /**
29  * Reads parquet files based on an iceberg table scan
30  *
31  * @param table table to scan
32  * @param schema read schema
33  * @param filter filter
34  * @param threads number of threads used to execute
35  * @param fileFilter optional filter for restricting the files that are scanned
36  */
37 class IcebergParquetScan(
38     table: Table,
39     schema: SimpleFeatureIcebergSchema,
40     filter: Expression,
41     threads: Int,
42     fileFilter: Option[String => Boolean] = None
43   ) extends CloseableIterator[SimpleFeature] with LazyLogging {

47   private val queue = new LinkedBlockingQueue[StructLike](10000)
48   private val closed = new AtomicBoolean(false)

52   // WithClose here shuts down the executor service, but already submitted tasks will still be executed
53   // note: we have to be explicit, as starting with Java 19 ExecutorService extends AutoCloseable which means two implicits match
54   private implicit val exIsCloseable: ExecutorServiceIsCloseable = IsCloseable.executorServiceIsCloseable
55   private val ex =
56     WithClose(new CachedThreadPool(threads)) { pool =>
57       // note: exclude z2 cols even if there's no transform
58       val scan = table.newScan().caseSensitive(false).project(schema.schema).filter(filter)
59       try {
60         WithClose(scan.planTasks()) { plan =>
61           WithClose(plan.iterator()) { tasks =>
62             var taskCount = 0
63             var fileCount = 0
64             logger.debug("Submitting tasks")
65             tasks.forEachRemaining { task =>
66               val fileTasks = task.files().asScala.toSeq
67               val filteredTasks = fileFilter match {
68                 case None => fileTasks
69                 case Some(filt) =>
70                   fileTasks.filter { fileTask =>
71                     val res = filt.apply(fileTask.file().location())
72                     if (!res) {
73                       logger.debug(
74                         s"Skipping file ${fileTask.file().location()} [${fileTask.start()}:${fileTask.length()}] due to file filter")
75                     }
76                     res
77                   }
78               }
79               if (filteredTasks.nonEmpty) {
80                 logger.trace(s"Submitting task with ${filteredTasks.size} file(s)")
81                 pool.submit(new TaskRunnable(filteredTasks))
82                 taskCount += 1
83                 fileCount += filteredTasks.size
84               }
85             }
86             logger.debug(s"Submitted $taskCount tasks (scanning $fileCount total files) using $threads threads")
87           }
88         }
89       } finally {
90         CloseWithLogging(Option(scan).collect { case c: Closeable => c })
91       }
92       pool
93     }

97   override def hasNext: Boolean = {
98     if (current != null) {
99       return true
100     }
101     current = queue.poll()
102     if (current != null) {
103       return true
104     }
105 
106     if (closed.get()) {
107       false
108     } else if (ex.isTerminated) {
109       current = queue.poll()
110       current != null
111     } else {
112       while (current == null && !ex.isTerminated) {
113         current = queue.poll(100, TimeUnit.MILLISECONDS)
114       }
115       current != null
116     }
117   }

119   override def next(): SimpleFeature = {
120     if (hasNext) {
121       row.setRow(current)
122       current = null
123       row
124     } else {
125       Iterator.empty.next
126     }

129   override def close(): Unit = {
130     if (closed.compareAndSet(false, true)) {
131       ex.shutdownNow()
132       ex.awaitTermination(1, TimeUnit.SECONDS)
133     }
134   }

136   private def readFile(task: FileScanTask, projection: Schema): CloseableIterator[StructLike] = {
137     logger.debug(s"Reading file ${task.file().location()} [${task.start()}:${task.length()}] with filter: ${task.residual()}")
138     val inputFile = table.io().newInputFile(task.file())
139     // we have to pass in the file path as a constant
140     val idToConstant = java.util.Map.of[Integer, Any](MetadataColumns.FILE_PATH_COLUMN_ID, inputFile.location())
141     // note: can't reuse containers b/c we put everything onto the queue - reading row-by-row with reused containers is much slower
142     val reader =
143       Parquet.read(inputFile)
144         .project(projection)
145         .split(task.start(), task.length())
146         .caseSensitive(false)
147         .filter(task.residual())
148         .createReaderFunc(fileSchema => GenericParquetReaders.buildReader(projection, fileSchema, idToConstant))
149         .build[Record]()
150     val iter = reader.iterator()
151     CloseableIterator(iter.asScala, CloseWithLogging(Seq(iter, reader)))
152   }
153 
154   private class TaskRunnable(tasks: Seq[FileScanTask]) extends Runnable {
155     override def run(): Unit = {
156       try {
157         tasks.foreach { task =>
158           if (!closed.get()) {
159             val deleteFilter = if (task.deletes().isEmpty) { None } else {
160               val filter = new GenericDeleteFilter(table.io(), task, table.schema(), schema.schema)
161               require(!filter.hasEqDeletes,
162                 s"Only positional deletes are supported, but got equality deletes: ${task.deletes().asScala.map(_.location()).mkString(", ")}")
163               Some(filter).filter(_.hasPosDeletes)
164             }
165             val projection = deleteFilter.fold(schema.schema)(_.requiredSchema())
166             WithClose(readFile(task, projection)) { iter =>
167               val withDeletes = deleteFilter.fold(iter) { deletes =>
168                 val index = deletes.deletedRowPositions()
169                 val position = projection.accessorForField(MetadataColumns.ROW_POSITION.fieldId())
170                 iter.filterNot(r => index.isDeleted(position.get(r).asInstanceOf[java.lang.Long]))
171               }
172               val residual = task.residual()
173               val filtered = if (residual == null || residual == Expressions.alwaysTrue()) { withDeletes } else {
174                 val filter = new Evaluator(projection.asStruct(), residual, false)
175                 val wrapper = new InternalRecordWrapper(projection.asStruct())
176                 withDeletes.filter(r => filter.eval(wrapper.wrap(r)))
177               }
178               filtered.foreach(queue.put)
179             }
180           }
181         }
182       } catch {
183         case NonFatal(e) => logger.error("Error running scan task:", e)
184       }
185     }
186   }
