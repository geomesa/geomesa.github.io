30 /**
31  * Reads parquet files based on an iceberg table scan
32  *
33  * @param table table to scan
34  * @param schema read schema
35  * @param filter filter
36  * @param threads number of threads used to execute
37  * @param fileFilter optional filter for restricting the files that are scanned
38  */
39 class IcebergParquetScan(
40     table: Table,
41     schema: SimpleFeatureIcebergSchema,
42     filter: Expression,
43     threads: Int,
44     fileFilter: Option[String => Boolean] = None
45   ) extends CloseableIterator[SimpleFeature] with LazyLogging {
46 
47   import scala.collection.JavaConverters._

49   private val queue = new LinkedBlockingQueue[StructLike](10000)
50   private val closed = new AtomicBoolean(false)
51 
52   private lazy val deleteProjection =
53     if (schema.schema.findField(MetadataColumns.ROW_POSITION.fieldId()) != null) { schema.schema } else {
54       TypeUtil.join(schema.schema, new Schema(MetadataColumns.ROW_POSITION))
55     }
56 
57   private val row = StructSimpleFeature(schema)
58 
59   // WithClose here shuts down the executor service, but already submitted tasks will still be executed
60   private val ex =
61     WithClose(new CachedThreadPool(threads)) { pool =>
62       // note: exclude z2 cols even if there's no transform
63       val scan = table.newScan().caseSensitive(false).project(schema.schema).filter(filter)
64       try {
65         WithClose(scan.planTasks()) { plan =>
66           WithClose(plan.iterator()) { tasks =>
67             var taskCount = 0
68             var fileCount = 0
69             logger.debug("Submitting tasks")
70             tasks.forEachRemaining { task =>
71               logger.trace(s"Submitting task with ${task.filesCount()} file(s)")
72               pool.submit(new TaskRunnable(task))
73               taskCount += 1
74               fileCount += task.filesCount()
75             }
76             logger.debug(s"Submitted $taskCount tasks (scanning $fileCount total files) using $threads threads")
77           }
78         }
79       } finally {
80         CloseWithLogging(Option(scan).collect { case c: Closeable => c })
81       }
82       pool
83     }

87   override def hasNext: Boolean = {
88     if (current != null) {
89       return true
90     }

94     }

97       false
98     } else if (ex.isTerminated) {

101     } else {
102       while (current == null && !ex.isTerminated) {
103         current = queue.poll(100, TimeUnit.MILLISECONDS)

109   override def next(): SimpleFeature = {
110     if (hasNext) {
111       row.setRow(current)
112       current = null
113       row
114     } else {
115       Iterator.empty.next
116     }
117   }

119   override def close(): Unit = {
120     if (closed.compareAndSet(false, true)) {
121       ex.shutdownNow()
122       ex.awaitTermination(1, TimeUnit.SECONDS)
123     }
124   }

126   private def readFile(task: FileScanTask, projection: Schema): CloseableIterator[StructLike] = {
127     val inputFile = table.io().newInputFile(task.file())
128     if (fileFilter.exists(_.apply(inputFile.location()) == false)) {
129       logger.debug(s"Skipping file ${inputFile.location()} [${task.start()}:${task.length()}] due to file filter")
130       CloseableIterator.empty[StructLike]
131     } else {
132       logger.debug(s"Reading file ${inputFile.location()} [${task.start()}:${task.length()}] with filter: ${task.residual()}")
133       // we have to pass in the file path as a constant
134       val idToConstant = java.util.Map.of[Integer, Any](MetadataColumns.FILE_PATH_COLUMN_ID, inputFile.location())
135       // note: can't reuse containers b/c we put everything onto the queue - reading row-by-row with reused containers is much slower
136       val reader =
137         Parquet.read(inputFile)
138           .project(projection)
139           .split(task.start(), task.length())
140           .caseSensitive(false)
141           .filter(task.residual())
142           .createReaderFunc(fileSchema => GenericParquetReaders.buildReader(projection, fileSchema, idToConstant))
143           .build[Record]()
144       val iter = reader.iterator()
145       CloseableIterator(iter.asScala, CloseWithLogging(Seq(iter, reader)))
146     }
147   }
148 
149   private def readDeleteFile(delete: DeleteFile, dataFilePath: String): PositionDeleteIndex = {
150     require(delete.content() == FileContent.POSITION_DELETES,
151       s"Only positional deletes are supported, but got: ${delete.content()}")
152     logger.debug(s"Reading delete file ${delete.location()} [${delete.contentSizeInBytes()}]")
153     val deleteFileSchema = DeleteSchemaUtil.pathPosSchema()
154     val builder =
155       Parquet.read(table.io().newInputFile(delete))
156         .project(deleteFileSchema)
157         .createReaderFunc(fileSchema => GenericParquetReaders.buildReader(deleteFileSchema, fileSchema))
158         .filter(Expressions.equal(MetadataColumns.DELETE_FILE_PATH.name(), dataFilePath))
159     WithClose(builder.build()) { deletes =>
160       Deletes.toPositionIndex(dataFilePath, deletes, delete)
161     }
162   }
163 
164   private class TaskRunnable(val task: CombinedScanTask) extends Runnable {
165     override def run(): Unit = {
166       try {
167         task.files().iterator().asScala.foreach { file =>
168           if (!closed.get()) {
169             val deleteIndex = if (file.deletes().isEmpty) { None } else {
170               val deletes = file.deletes().asScala.map(readDeleteFile(_, file.file().location()))
171               Some(PositionDeleteIndexUtil.merge(deletes.asJava))
172             }
173             val projection = if (deleteIndex.isEmpty) { schema.schema } else { deleteProjection }
174 
175             WithClose(readFile(file, projection)) { iter =>
176               val withDeletes = deleteIndex.fold(iter) { index =>
177                 val position = projection.accessorForField(MetadataColumns.ROW_POSITION.fieldId())
178                 iter.filterNot(r => index.isDeleted(position.get(r).asInstanceOf[java.lang.Long]))
179               }
180               val residual = file.residual()
181               val filtered = if (residual == null || residual == Expressions.alwaysTrue()) { withDeletes } else {
182                 val filter = new Evaluator(projection.asStruct(), residual, false)
183                 val wrapper = new InternalRecordWrapper(projection.asStruct())
184                 withDeletes.filter(r => filter.eval(wrapper.wrap(r)))
185               }
186               filtered.foreach(queue.put)
187             }
188           }
189         }
190       } catch {
191         case NonFatal(e) => logger.error("Error running scan task:", e)
192       }
193     }
194   }
