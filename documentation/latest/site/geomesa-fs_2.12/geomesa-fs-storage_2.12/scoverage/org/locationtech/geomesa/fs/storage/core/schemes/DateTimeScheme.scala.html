29 /**
30  * Date based partitioning
31  *
32  * @param attribute attribute being partitioned
33  * @param index index in the feature type of the attribute
34  * @param unit chrono unit used for partitioning
35  */
36 case class DateTimeScheme(attribute: String, index: Int, unit: ChronoUnit) extends PartitionScheme with TemporalScheme {
37 
38   import FilterHelper.ff
39 
40   override val name: String = s"${unit.name().toLowerCase(Locale.US)}:attribute=$attribute"
41 
42   override val column: String = ColumnName.encode(attribute)
43 
44   private val encoder = DateTimeScheme.encoder(unit)
45 
46   override def spec(b: PartitionSpec.Builder): PartitionSpec.Builder = unit match {
47     case ChronoUnit.HOURS  => b.hour(column)
48     case ChronoUnit.DAYS   => b.day(column)
49     case ChronoUnit.MONTHS => b.month(column)
50     case ChronoUnit.YEARS  => b.year(column)
51     case _ => throw new UnsupportedOperationException("An implementation is missing")
52   }
53 
54   override def getPartition(feature: SimpleFeature): PartitionKey = {
55     val instant = feature.getAttribute(index).asInstanceOf[Date].toInstant
56     val value = unit.between(DateTimeScheme.Epoch, ZonedDateTime.ofInstant(instant, ZoneOffset.UTC)).toInt
57     PartitionKey(name, encoder.encode(value))
58   }
59 
60   override def getCoveringFilter(partition: PartitionKey): Filter = {
61     val offset = encoder.decode(partition.value)
62     val start = DateTimeScheme.Epoch.plus(offset.longValue(), unit)
63     val end = ff.literal(DateParsing.format(start.plus(1, unit)))
64     ff.and(ff.greaterOrEqual(ff.property(attribute), ff.literal(DateParsing.format(start))), ff.less(ff.property(attribute), end))
65   }
66 
67   override def getCoveringExpression(partition: PartitionKey): Expression = {
68     val transform = unit match {
69       case ChronoUnit.HOURS  => Expressions.hour[Integer](column)
70       case ChronoUnit.DAYS   => Expressions.day[Integer](column)
71       case ChronoUnit.MONTHS => Expressions.month[Integer](column)
72       case ChronoUnit.YEARS  => Expressions.year[Integer](column)
73       case _ => throw new UnsupportedOperationException("An implementation is missing")
74     }
75     Expressions.equal[Integer](transform, encoder.decode(partition.value))
76   }
77 
78   override def getPartition(partition: StructLike, i: Int): PartitionKey = PartitionKey(name, encoder.getPartition(partition, i))
79 
80   override def partition(time: ZonedDateTime): TemporalPartition = {
81     val value = unit.between(DateTimeScheme.Epoch, time).toInt
82     val partition = PartitionKey(name, encoder.encode(value))
83     val start = DateTimeScheme.Epoch.plus(value, unit)
84     val end = start.plus(1, unit)
85     TemporalPartition(partition, start, end)
86   }
87 }

89 object DateTimeScheme extends PartitionSchemeFactory {

95   val Epoch: ZonedDateTime = ZonedDateTime.ofInstant(Instant.EPOCH, ZoneOffset.UTC)
96 
97   override def load(sft: SimpleFeatureType, scheme: String): Option[PartitionScheme] = {
98     val opts = SchemeOpts(scheme)
99     val unit = opts.name match {
100       case "year"  | "years"  | "yearly"  => Some(ChronoUnit.YEARS)
101       case "month" | "months" | "monthly" => Some(ChronoUnit.MONTHS)
102       case "day"   | "days"   | "daily"   => Some(ChronoUnit.DAYS)
103       case "hour"  | "hours"  | "hourly"  => Some(ChronoUnit.HOURS)
104       case _ => None
105     }
106     unit.map { u =>
107       val dtg = opts.getSingle("attribute").orElse(sft.getDtgField).orNull
108       require(dtg != null, s"Date scheme requires an attribute to be specified with 'attribute=<attribute>'")
109       val index = attributeIndex(sft, dtg, Some(classOf[Date]))
110       if (opts.getSingle("step").exists(_ != "1")) {
111         throw new IllegalArgumentException("`step` argument is no longer supported in date-time schemes")
112       }
113       DateTimeScheme(dtg, index, u)
114     }
115   }
116 
117   override def load(schema: SimpleFeatureIcebergSchema, spec: Schema, field: PartitionField): Option[PartitionScheme] = {
118     PartitionSpecVisitor.visit(spec, field, new BaseSpecVisitor() {
119       override def year(sourceName: String, sourceId: Int): Option[PartitionScheme] = load(schema, sourceName, ChronoUnit.YEARS)
120       override def month(sourceName: String, sourceId: Int): Option[PartitionScheme] = load(schema, sourceName, ChronoUnit.MONTHS)
121       override def day(sourceName: String, sourceId: Int): Option[PartitionScheme] = load(schema, sourceName, ChronoUnit.DAYS)
122       override def hour(sourceName: String, sourceId: Int): Option[PartitionScheme] = load(schema, sourceName, ChronoUnit.HOURS)
123     })
124   }
125 
126   private def load(schema: SimpleFeatureIcebergSchema, col: String, unit: ChronoUnit): Option[PartitionScheme] = {
127     schema.sft.getAttributeDescriptors.asScala.find(d => col == ColumnName.encode(d.getLocalName)).flatMap { d =>
128       if (classOf[Date].isAssignableFrom(d.getType.getBinding)) {
129         Some(DateTimeScheme(d.getLocalName, schema.sft.indexOf(d.getLocalName), unit))
130       } else {
131         None
132       }
133     }
134   }
135 
136   private def encoder(unit: ChronoUnit): PartitionEncoder = {
137     unit match {
138       case ChronoUnit.DAYS => DayEncoder
139       case _ => DefaultEncoder
140     }
141   }
142 
143   private sealed trait PartitionEncoder {
144     def encode(value: Int): String
145     def decode(value: String): Int
146     def getPartition(partition: StructLike, i: Int): String
147   }
148 
149   private object DefaultEncoder extends PartitionEncoder {
150     override def encode(value: Int): String = value.toString
151     override def decode(value: String): Int = value.toInt
152     override def getPartition(partition: StructLike, i: Int): String = partition.get(i, classOf[Integer]).toString
153   }
154 
155   // note: days are handled differently from other types, and expect an ISO_LOCAL_DATE formatted string
156   private object DayEncoder extends PartitionEncoder {
157     override def encode(value: Int): String = DateTimeFormatter.ISO_LOCAL_DATE.format(LocalDate.EPOCH.plusDays(value))
158     override def decode(value: String): Int = {
159       val date = DateParsing.parse(value, DateTimeFormatter.ISO_LOCAL_DATE)
160       ChronoUnit.DAYS.between(DateTimeScheme.Epoch, date).toInt
161     }
162     override def getPartition(partition: StructLike, i: Int): String = encode(partition.get(i, classOf[Integer]))
163   }
