26 /**
27  * Date based partitioning
28  *
29  * @param attribute attribute being partitioned
30  * @param index index in the feature type of the attribute
31  * @param unit chrono unit used for partitioning
32  */
33 case class DateTimeScheme(attribute: String, index: Int, unit: ChronoUnit) extends PartitionScheme with TemporalScheme {
34 
35   import FilterHelper.ff
36 
37   override val name: String = s"${unit.name().toLowerCase(Locale.US)}:attribute=$attribute"
38 
39   private val encoder = DateTimeScheme.encoder(unit)
40 
41   override def spec(b: PartitionSpec.Builder): PartitionSpec.Builder = unit match {
42     case ChronoUnit.HOURS  => b.hour(ColumnName.encode(attribute))
43     case ChronoUnit.DAYS   => b.day(ColumnName.encode(attribute))
44     case ChronoUnit.MONTHS => b.month(ColumnName.encode(attribute))
45     case ChronoUnit.YEARS  => b.year(ColumnName.encode(attribute))
46     case _ => throw new UnsupportedOperationException("An implementation is missing")
47   }
48 
49   override def getPartition(feature: SimpleFeature): PartitionKey = {
50     val instant = feature.getAttribute(index).asInstanceOf[Date].toInstant
51     val value = unit.between(DateTimeScheme.Epoch, ZonedDateTime.ofInstant(instant, ZoneOffset.UTC)).toInt
52     PartitionKey(name, encoder.encode(value))
53   }
54 
55   override def getCoveringFilter(partition: PartitionKey): Filter = {
56     val offset = encoder.decode(partition.value)
57     val start = DateTimeScheme.Epoch.plus(offset.longValue(), unit)
58     val end = ff.literal(DateParsing.format(start.plus(1, unit)))
59     ff.and(ff.greaterOrEqual(ff.property(attribute), ff.literal(DateParsing.format(start))), ff.less(ff.property(attribute), end))
60   }
61 
62   override def getCoveringExpression(partition: PartitionKey): Expression = {
63     val transform = unit match {
64       case ChronoUnit.HOURS  => Expressions.hour[Integer](ColumnName.encode(attribute))
65       case ChronoUnit.DAYS   => Expressions.day[Integer](ColumnName.encode(attribute))
66       case ChronoUnit.MONTHS => Expressions.month[Integer](ColumnName.encode(attribute))
67       case ChronoUnit.YEARS  => Expressions.year[Integer](ColumnName.encode(attribute))
68       case _ => throw new UnsupportedOperationException("An implementation is missing")
69     }
70     Expressions.equal[Integer](transform, encoder.decode(partition.value))
71   }
72 
73   override def getPartition(partition: StructLike, i: Int): PartitionKey = PartitionKey(name, encoder.getPartition(partition, i))
74 
75   override def partition(time: ZonedDateTime): TemporalPartition = {
76     val value = unit.between(DateTimeScheme.Epoch, time).toInt
77     val partition = PartitionKey(name, encoder.encode(value))
78     val start = DateTimeScheme.Epoch.plus(value, unit)
79     val end = start.plus(1, unit)
80     TemporalPartition(partition, start, end)
81   }
82 }

84 object DateTimeScheme extends PartitionSchemeFactory {
85 
86   import org.locationtech.geomesa.utils.geotools.RichSimpleFeatureType.RichSimpleFeatureType
87 
88   val Epoch: ZonedDateTime = ZonedDateTime.ofInstant(Instant.EPOCH, ZoneOffset.UTC)
89 
90   override def load(sft: SimpleFeatureType, scheme: String): Option[PartitionScheme] = {
91     val opts = SchemeOpts(scheme)
92     val unit = opts.name match {
93       case "year"  | "years"  | "yearly"  => Some(ChronoUnit.YEARS)
94       case "month" | "months" | "monthly" => Some(ChronoUnit.MONTHS)
95       case "day"   | "days"   | "daily"   => Some(ChronoUnit.DAYS)
96       case "hour"  | "hours"  | "hourly"  => Some(ChronoUnit.HOURS)
97       case _ => None
98     }
99     unit.map { u =>
100       val dtg = opts.getSingle("attribute").orElse(sft.getDtgField).orNull
101       require(dtg != null, s"Date scheme requires an attribute to be specified with 'attribute=<attribute>'")
102       val index = attributeIndex(sft, dtg, Some(classOf[Date]))
103       if (opts.getSingle("step").exists(_ != "1")) {
104         throw new IllegalArgumentException("`step` argument is no longer supported in date-time schemes")
105       }
106       DateTimeScheme(dtg, index, u)
107     }
108   }
109 
110   private def encoder(unit: ChronoUnit): PartitionEncoder = {
111     unit match {
112       case ChronoUnit.DAYS => DayEncoder
113       case _ => DefaultEncoder
114     }
115   }
116 
117   private sealed trait PartitionEncoder {
118     def encode(value: Int): String
119     def decode(value: String): Int
120     def getPartition(partition: StructLike, i: Int): String
121   }
122 
123   private object DefaultEncoder extends PartitionEncoder {
124     override def encode(value: Int): String = value.toString
125     override def decode(value: String): Int = value.toInt
126     override def getPartition(partition: StructLike, i: Int): String = partition.get(i, classOf[Integer]).toString
127   }
128 
129   // note: days are handled differently from other types, and expect an ISO_LOCAL_DATE formatted string
130   private object DayEncoder extends PartitionEncoder {
131     override def encode(value: Int): String = DateTimeFormatter.ISO_LOCAL_DATE.format(LocalDate.EPOCH.plusDays(value))
132     override def decode(value: String): Int = {
133       val date = DateParsing.parse(value, DateTimeFormatter.ISO_LOCAL_DATE)
134       ChronoUnit.DAYS.between(DateTimeScheme.Epoch, date).toInt
135     }
136     override def getPartition(partition: StructLike, i: Int): String = encode(partition.get(i, classOf[Integer]))
137   }
