26 /**
27  * Date based partitioning
28  *
29  * @param attribute attribute being partitioned
30  * @param index index in the feature type of the attribute
31  * @param unit chrono unit used for partitioning
32  */
33 case class DateTimeScheme(attribute: String, index: Int, unit: ChronoUnit) extends PartitionScheme with TemporalScheme {
34 
35   import FilterHelper.ff
36 
37   override val name: String = s"${unit.name().toLowerCase(Locale.US)}:attribute=$attribute"
38 
39   override val column: String = ColumnName.encode(attribute)
40 
41   private val encoder = DateTimeScheme.encoder(unit)
42 
43   override def spec(b: PartitionSpec.Builder): PartitionSpec.Builder = unit match {
44     case ChronoUnit.HOURS  => b.hour(column)
45     case ChronoUnit.DAYS   => b.day(column)
46     case ChronoUnit.MONTHS => b.month(column)
47     case ChronoUnit.YEARS  => b.year(column)
48     case _ => throw new UnsupportedOperationException("An implementation is missing")
49   }
50 
51   override def getPartition(feature: SimpleFeature): PartitionKey = {
52     val instant = feature.getAttribute(index).asInstanceOf[Date].toInstant
53     val value = unit.between(DateTimeScheme.Epoch, ZonedDateTime.ofInstant(instant, ZoneOffset.UTC)).toInt
54     PartitionKey(name, encoder.encode(value))
55   }
56 
57   override def getCoveringFilter(partition: PartitionKey): Filter = {
58     val offset = encoder.decode(partition.value)
59     val start = DateTimeScheme.Epoch.plus(offset.longValue(), unit)
60     val end = ff.literal(DateParsing.format(start.plus(1, unit)))
61     ff.and(ff.greaterOrEqual(ff.property(attribute), ff.literal(DateParsing.format(start))), ff.less(ff.property(attribute), end))
62   }
63 
64   override def getCoveringExpression(partition: PartitionKey): Expression = {
65     val transform = unit match {
66       case ChronoUnit.HOURS  => Expressions.hour[Integer](column)
67       case ChronoUnit.DAYS   => Expressions.day[Integer](column)
68       case ChronoUnit.MONTHS => Expressions.month[Integer](column)
69       case ChronoUnit.YEARS  => Expressions.year[Integer](column)
70       case _ => throw new UnsupportedOperationException("An implementation is missing")
71     }
72     Expressions.equal[Integer](transform, encoder.decode(partition.value))
73   }
74 
75   override def getPartition(partition: StructLike, i: Int): PartitionKey = PartitionKey(name, encoder.getPartition(partition, i))
76 
77   override def partition(time: ZonedDateTime): TemporalPartition = {
78     val value = unit.between(DateTimeScheme.Epoch, time).toInt
79     val partition = PartitionKey(name, encoder.encode(value))
80     val start = DateTimeScheme.Epoch.plus(value, unit)
81     val end = start.plus(1, unit)
82     TemporalPartition(partition, start, end)
83   }
84 }

86 object DateTimeScheme extends PartitionSchemeFactory {
87 
88   import org.locationtech.geomesa.utils.geotools.RichSimpleFeatureType.RichSimpleFeatureType
89 
90   val Epoch: ZonedDateTime = ZonedDateTime.ofInstant(Instant.EPOCH, ZoneOffset.UTC)
91 
92   override def load(sft: SimpleFeatureType, scheme: String): Option[PartitionScheme] = {
93     val opts = SchemeOpts(scheme)
94     val unit = opts.name match {
95       case "year"  | "years"  | "yearly"  => Some(ChronoUnit.YEARS)
96       case "month" | "months" | "monthly" => Some(ChronoUnit.MONTHS)
97       case "day"   | "days"   | "daily"   => Some(ChronoUnit.DAYS)
98       case "hour"  | "hours"  | "hourly"  => Some(ChronoUnit.HOURS)
99       case _ => None
100     }
101     unit.map { u =>
102       val dtg = opts.getSingle("attribute").orElse(sft.getDtgField).orNull
103       require(dtg != null, s"Date scheme requires an attribute to be specified with 'attribute=<attribute>'")
104       val index = attributeIndex(sft, dtg, Some(classOf[Date]))
105       if (opts.getSingle("step").exists(_ != "1")) {
106         throw new IllegalArgumentException("`step` argument is no longer supported in date-time schemes")
107       }
108       DateTimeScheme(dtg, index, u)
109     }
110   }
111 
112   private def encoder(unit: ChronoUnit): PartitionEncoder = {
113     unit match {
114       case ChronoUnit.DAYS => DayEncoder
115       case _ => DefaultEncoder
116     }
117   }
118 
119   private sealed trait PartitionEncoder {
120     def encode(value: Int): String
121     def decode(value: String): Int
122     def getPartition(partition: StructLike, i: Int): String
123   }
124 
125   private object DefaultEncoder extends PartitionEncoder {
126     override def encode(value: Int): String = value.toString
127     override def decode(value: String): Int = value.toInt
128     override def getPartition(partition: StructLike, i: Int): String = partition.get(i, classOf[Integer]).toString
129   }
130 
131   // note: days are handled differently from other types, and expect an ISO_LOCAL_DATE formatted string
132   private object DayEncoder extends PartitionEncoder {
133     override def encode(value: Int): String = DateTimeFormatter.ISO_LOCAL_DATE.format(LocalDate.EPOCH.plusDays(value))
134     override def decode(value: String): Int = {
135       val date = DateParsing.parse(value, DateTimeFormatter.ISO_LOCAL_DATE)
136       ChronoUnit.DAYS.between(DateTimeScheme.Epoch, date).toInt
137     }
138     override def getPartition(partition: StructLike, i: Int): String = encode(partition.get(i, classOf[Integer]))
139   }
