22 object BoundingBoxField {
23 
24   val BoundingBoxFieldSuffix = s"_bbox${SimpleFeatureSchema.InternalFieldDelimiter}"

26   val XMin = "xmin"
27   val YMin = "ymin"
28   val XMax = "xmax"
29   val YMax = "ymax"

31   /**
32    * Gets the column name for the bbox group field
33    *
34    * @param geom geometry column name
35    * @return
36    */
37   def groupName(geom: String): String = s"${SimpleFeatureSchema.InternalFieldDelimiter}$geom$BoundingBoxFieldSuffix"
38 
39   /**
40    * The parquet schema for a bbox field
41    *
42    * @param geom geometry column name
43    * @return
44    */
45   def parquetSchema(geom: String): GroupType = {
46     val bbox = groupName(geom)
47     Types.optionalGroup()
48       .required(PrimitiveTypeName.FLOAT).named(BoundingBoxField.XMin)
49       .required(PrimitiveTypeName.FLOAT).named(BoundingBoxField.YMin)
50       .required(PrimitiveTypeName.FLOAT).named(BoundingBoxField.XMax)
51       .required(PrimitiveTypeName.FLOAT).named(BoundingBoxField.YMax)
52       .named(bbox)
53   }
54 
55   /**
56    * The iceberg schema for a bbox field
57    *
58    * @param geom geometry column name
59    * @return
60    */
61   def icebergSchema(geom: String, fieldIds: AtomicInteger): NestedField = {
62     val bbox = groupName(geom)
63     NestedField.optional(bbox).withId(fieldIds.getAndIncrement()).ofType(StructType.of(
64       NestedField.required(BoundingBoxField.XMin).withId(fieldIds.getAndIncrement()).ofType(FloatType.get()).build(),
65       NestedField.required(BoundingBoxField.YMin).withId(fieldIds.getAndIncrement()).ofType(FloatType.get()).build(),
66       NestedField.required(BoundingBoxField.XMax).withId(fieldIds.getAndIncrement()).ofType(FloatType.get()).build(),
67       NestedField.required(BoundingBoxField.YMax).withId(fieldIds.getAndIncrement()).ofType(FloatType.get()).build(),
68     )).build()
69   }

71   /**
72    * Create a filter against a bbox col
73    *
74    * @param geom geometry column name
75    * @param xmin xmin
76    * @param ymin ymin
77    * @param xmax xmax
78    * @param ymax ymax
79    * @return
80    */
81   def filterParquet(geom: String, xmin: Double, ymin: Double, xmax: Double, ymax: Double, isPoint: Boolean): FilterPredicate = {
82     // TODO // if points, we can just project 2 of the 4 bbox fields, since min == max
83     val bbox = groupName(geom)
84     val xminCol = FilterApi.floatColumn(s"$bbox.$XMin")
85     val yminCol = FilterApi.floatColumn(s"$bbox.$YMin")
86     val xmaxCol = FilterApi.floatColumn(s"$bbox.$XMax")
87     val ymaxCol = FilterApi.floatColumn(s"$bbox.$YMax")
88     val filters = Seq[FilterPredicate](
89       FilterApi.ltEq(xminCol, Float.box(Math.nextUp(xmax.toFloat))),
90       FilterApi.gtEq(xmaxCol, Float.box(Math.nextDown(xmin.toFloat))),
91       FilterApi.ltEq(yminCol, Float.box(Math.nextUp(ymax.toFloat))),
92       FilterApi.gtEq(ymaxCol, Float.box(Math.nextDown(ymin.toFloat))),
93     )
94     filters.reduce(FilterApi.and)
95   }

97   /**
98    * Create a filter against a bbox col
99    *
100    * @param geom geometry column name
101    * @param xmin xmin
102    * @param ymin ymin
103    * @param xmax xmax
104    * @param ymax ymax
105    * @return
106    */
107   def filterIceberg(geom: String, xmin: Double, ymin: Double, xmax: Double, ymax: Double, isPoint: Boolean): Expression = {
108     val bbox = groupName(geom)
109     // if points, we can just project 2 of the 4 bbox fields, since min == max
110     val Seq(bboxXMin, bboxXMax, bboxYMin, bboxYMax) = if (isPoint) {
111       Seq(s"$bbox.$XMin", s"$bbox.$XMin", s"$bbox.$YMin", s"$bbox.$YMin")
112     } else {
113       Seq(s"$bbox.$XMin", s"$bbox.$XMax", s"$bbox.$YMin", s"$bbox.$YMax")
114     }
115     val exps = Seq(
116       Expressions.lessThanOrEqual(bboxXMin, Float.box(Math.nextUp(xmax.toFloat))),
117       Expressions.greaterThanOrEqual(bboxXMax, Float.box(Math.nextDown(xmin.toFloat))),
118       Expressions.lessThanOrEqual(bboxYMin, Float.box(Math.nextUp(ymax.toFloat))),
119       Expressions.greaterThanOrEqual(bboxYMax, Float.box(Math.nextDown(ymin.toFloat))),
120     )
121     exps.reduce(Expressions.and)
122   }
