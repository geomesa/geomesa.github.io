31   /**
32    * Gets the column name for the bbox group field
33    *
34    * @param geom geometry column name
35    * @return
36    */
37   def groupName(geom: String): String = s"${SimpleFeatureSchema.InternalFieldDelimiter}$geom$BoundingBoxFieldSuffix"

39   /**
40    * The parquet schema for a bbox field
41    *
42    * @param geom geometry column name
43    * @return
44    */
45   def parquetSchema(geom: String, fieldIds: AtomicInteger, nestedFieldIds: AtomicInteger): GroupType = {
46     val bbox = groupName(geom)
47     Types.optionalGroup()
48       .id(fieldIds.getAndIncrement())
49       .required(PrimitiveTypeName.FLOAT).id(nestedFieldIds.getAndIncrement()).named(BoundingBoxField.XMin)
50       .required(PrimitiveTypeName.FLOAT).id(nestedFieldIds.getAndIncrement()).named(BoundingBoxField.YMin)
51       .required(PrimitiveTypeName.FLOAT).id(nestedFieldIds.getAndIncrement()).named(BoundingBoxField.XMax)
52       .required(PrimitiveTypeName.FLOAT).id(nestedFieldIds.getAndIncrement()).named(BoundingBoxField.YMax)
53       .named(bbox)
54   }

56   /**
57    * The iceberg schema for a bbox field
58    *
59    * @param geom geometry column name
60    * @return

62   def icebergSchema(geom: String, fieldIds: AtomicInteger, nestedFieldIds: AtomicInteger): NestedField = {
63     val bbox = groupName(geom)

72   /**
73    * Create a filter against a bbox col
74    *
75    * @param geom geometry column name
76    * @param xmin xmin
77    * @param ymin ymin
78    * @param xmax xmax
79    * @param ymax ymax
80    * @return
81    */
82   def filterParquet(geom: String, xmin: Double, ymin: Double, xmax: Double, ymax: Double, isPoint: Boolean): FilterPredicate = {
83     // TODO // if points, we can just project 2 of the 4 bbox fields, since min == max
84     val bbox = groupName(geom)
85     val xminCol = FilterApi.floatColumn(s"$bbox.$XMin")
86     val yminCol = FilterApi.floatColumn(s"$bbox.$YMin")
87     val xmaxCol = FilterApi.floatColumn(s"$bbox.$XMax")
88     val ymaxCol = FilterApi.floatColumn(s"$bbox.$YMax")
89     val filters = Seq[FilterPredicate](
90       FilterApi.ltEq(xminCol, Float.box(Math.nextUp(xmax.toFloat))),
91       FilterApi.gtEq(xmaxCol, Float.box(Math.nextDown(xmin.toFloat))),
92       FilterApi.ltEq(yminCol, Float.box(Math.nextUp(ymax.toFloat))),
93       FilterApi.gtEq(ymaxCol, Float.box(Math.nextDown(ymin.toFloat))),
94     )
95     filters.reduce(FilterApi.and)
96   }

98   /**

102    * @param xmin xmin
103    * @param ymin ymin
104    * @param xmax xmax
105    * @param ymax ymax
106    * @return
107    */
108   def filterIceberg(geom: String, xmin: Double, ymin: Double, xmax: Double, ymax: Double, isPoint: Boolean): Expression = {
109     val bbox = groupName(geom)
110     // if points, we can just project 2 of the 4 bbox fields, since min == max
111     val Seq(bboxXMin, bboxXMax, bboxYMin, bboxYMax) = if (isPoint) {
112       Seq(s"$bbox.$XMin", s"$bbox.$XMin", s"$bbox.$YMin", s"$bbox.$YMin")
113     } else {
114       Seq(s"$bbox.$XMin", s"$bbox.$XMax", s"$bbox.$YMin", s"$bbox.$YMax")
115     }
116     val exps = Seq(
117       Expressions.lessThanOrEqual(bboxXMin, Float.box(Math.nextUp(xmax.toFloat))),
118       Expressions.greaterThanOrEqual(bboxXMax, Float.box(Math.nextDown(xmin.toFloat))),
119       Expressions.lessThanOrEqual(bboxYMin, Float.box(Math.nextUp(ymax.toFloat))),
120       Expressions.greaterThanOrEqual(bboxYMax, Float.box(Math.nextDown(ymin.toFloat))),
121     )
122     exps.reduce(Expressions.and)
123   }
