28 /**
29  * A paired simple feature type and parquet schema
30  *
31  * @param sft simple feature type
32  * @param geometries geometry type encoding
33  * @param metadata file metadata
34  * @param messageType parquet schema
35  * @param fields parquet message schema
36  */
37 class SimpleFeatureParquetSchema private (
38     val sft: SimpleFeatureType,
39     val geometries: GeometryEncoding,
40     val metadata: java.util.Map[String, String],
41     val messageType: MessageType,
42     fields: ParquetFields) {
43 
44   /**
45    * Gets the schema used for reading (excludes some columns only used for partitioning)
46    *
47    * @return
48    */
49   def read(): SimpleFeatureParquetSchema = {
50     val fieldFilter =
51       fields.fields.map(_.getName)
52         .filterNot(n => n.endsWith(ZValueField.Z2ValueFieldSuffix) || n.endsWith(ZValueField.XZ2ValueFieldSuffix))
53     new SimpleFeatureParquetSchema(sft, geometries, metadata, fields.toMessageType(sft.getTypeName, fieldFilter.toSet), fields)
54   }
55 }

63   /**
64    * Encodes the feature type in the conf

66    * @param conf conf

72   /**
73    * Simple feature type configuration
74    *
75    * @param sft simple feature type
76    * @return
77    */
78   def sftConf(sft: SimpleFeatureType): Map[String, String] = {
79     Map(
80       SftNameKey -> Option(sft.getName.getNamespaceURI).map(ns => s"$ns:${sft.getTypeName}").getOrElse(sft.getTypeName),
81       SftSpecKey -> SimpleFeatureTypes.encodeType(sft, includeUserData = true)
82     )
83   }

85   /**
86     * Extract the simple feature type from a parquet read context. The read context
87     * contains both file metadata and the provided read conf
88     *
89     * @param context parquet read context
90     * @return
91     */
92   def read(context: InitContext): Option[SimpleFeatureParquetSchema] = {
93     // copy in the file level metadata
94     val kvMeta = context.getKeyValueMetadata.asScala.collect { case (k, v) if !v.isEmpty => k -> v.iterator.next }
95     // noinspection ScalaDeprecation
96     val gmMeta = Seq(SftNameKey, SftSpecKey, GeometryEncodingKey).flatMap { key =>
97       val value =
98         try { context.getParquetConfiguration.get(key) } catch {
99           // noinspection ScalaDeprecation - use the older method for spark compatibility
100           case _: NoSuchMethodError => context.getConfiguration.get(key)
101         }
102       Option(value).map(key -> _)
103     }
104     // note: context meta goes second so that transforms override the file-level keys
105     apply(new PlainParquetConfiguration((kvMeta.toMap ++ gmMeta.toMap).asJava)).map(_.read())
106   }
107 
108   /**
109     * Extract the simple feature type from an existing parquet file, without any known read context
110     *
111     * @param footer parquet file footer
112     * @return
113     */
114   def read(footer: FileMetaData): Option[SimpleFeatureParquetSchema] =
115     apply(new PlainParquetConfiguration(footer.getKeyValueMetaData)).map(_.read())
116 
117   /**
118    * Get a schema for writing. Encoding can be configured through `geomesa.parquet.geometries` and `geomesa.fs.visibilities`
119    *
120    * @param conf write configuration, including the sft spec
121    * @return
122    */
123   def apply(conf: ParquetConfiguration): Option[SimpleFeatureParquetSchema] = {
124     for {
125       name <- Option(conf.get(SftNameKey))
126       spec <- Option(conf.get(SftSpecKey))
127     } yield {
128       val sft = SimpleFeatureTypes.createImmutableType(name, spec)
129       // TODO in iceberg, geometry encoding is now specified at the column level - use that once we support more than WKB
130       val geometries = Option(conf.get(GeometryEncodingKey)).map(GeometryEncoding.apply).getOrElse(GeometryEncoding.GeoParquetWkb)
131       val metadata = Map(
132         SftNameKey -> name,
133         SftSpecKey -> spec,
134         GeometryEncodingKey -> geometries.toString,
135       ) ++ Option(conf.get(PartitionKey)).map(PartitionKey -> _)
136       val parquet = schema(sft, geometries)
137       val messageType = {
138         val base = parquet.toMessageType(sft.getTypeName)
139         // if an iceberg name mapping is provided, stamp the table's field ids onto the schema by name,
140         // so that written files are read back by field id rather than falling back to name-based mapping
141         Option(conf.get(IcebergNameMappingKey)) match {
142           case None => base
143           case Some(json) => ParquetSchemaUtil.applyNameMapping(base, NameMappingParser.fromJson(json))
144         }
145       }
146       new SimpleFeatureParquetSchema(sft, geometries, metadata.asJava, messageType, parquet)
147     }
148   }
149 
150   /**
151    * Gets the parquet schema for a feature type
152    *
153    * @param sft simple feature type
154    * @param conf storage configuration
155    * @return
156    */
157   def apply(sft: SimpleFeatureType, conf: Map[String, String]): SimpleFeatureParquetSchema = {
158     val config = new PlainParquetConfiguration(conf.asJava)
159     setSft(config, sft)
160     apply(config).get
161   }

163   /**
164    * Get the message type for a simple feature type. We need the full sft in order to ensure field ids are
165    * consistent, but the schema may be reduced (e.g. on read) based on the filter sft
166    *
167    * @param sft simple feature type
168    * @param geometries geometry type encoding
169    * @return
170    */
171   private def schema(sft: SimpleFeatureType, geometries: GeometryEncoding): ParquetFields = {
172     val builder = Seq.newBuilder[Type]
173     builder +=
174       Types.required(PrimitiveTypeName.BINARY)
175         .as(LogicalTypeAnnotation.stringType())
176         .named(FeatureIdField)
177     builder +=
178       Types.optional(PrimitiveTypeName.BINARY)
179         .as(LogicalTypeAnnotation.stringType())
180         .named(VisibilitiesField)
181 
182     sft.getAttributeDescriptors.asScala.foreach { d =>
183       val name = ColumnName(d.getLocalName)
184       val objectType = ObjectType.selectType(d)
185       builder += buildType(name.column, objectType, geometries)
186       if (objectType.head == ObjectType.GEOMETRY) {
187         builder += BoundingBoxField.parquetSchema(name.column)
188         builder += ZValueField.parquetSchema(name.column, objectType(1))
189       }
190     }
191     ParquetFields(builder.result())
192   }
193 
194   /**
195    * Builds the schema type for an attribute
196    *
197    * @param name field name
198    * @param bindings object type
199    * @param geometries geometry type encodings
200    * @param repetition repetition
201    * @return
202    */
203   private def buildType(
204       name: String,
205       bindings: Seq[ObjectType],
206       geometries: GeometryEncoding,
207       repetition: Repetition = Repetition.OPTIONAL): Type = {
208     val builder = bindings.head match {
209       case ObjectType.INT     => Types.primitive(PrimitiveTypeName.INT32, repetition)
210       case ObjectType.DOUBLE  => Types.primitive(PrimitiveTypeName.DOUBLE, repetition)
211       case ObjectType.LONG    => Types.primitive(PrimitiveTypeName.INT64, repetition)
212       case ObjectType.FLOAT   => Types.primitive(PrimitiveTypeName.FLOAT, repetition)
213       case ObjectType.BOOLEAN => Types.primitive(PrimitiveTypeName.BOOLEAN, repetition)
214       case ObjectType.BYTES   => Types.primitive(PrimitiveTypeName.BINARY, repetition)
215 
216       case ObjectType.STRING if bindings.last == ObjectType.JSON =>
217         Types.buildGroup(repetition)
218           .as(LogicalTypeAnnotation.variantType(1))
219           // note: the iceberg api does not define nested fields for variants so we don't set id here
220           .required(PrimitiveTypeName.BINARY).named("metadata")
221           .required(PrimitiveTypeName.BINARY).named("value")
222         // TODO can pull out known fields for shredding, but trino doesn't support reading shredded variants yet
223         // val partiallyShredded: GroupType =
224         //   Types.buildGroup(Repetition.REQUIRED).as(LogicalTypeAnnotation.variantType(1.toByte))
225         //   .required(PrimitiveTypeName.BINARY).named("metadata")
226         //   .optional(PrimitiveTypeName.BINARY).named("value") // stores {custom_field_xyz: 123}
227         //   .optionalGroup
228         //     .optionalGroup
229         //       .optional(PrimitiveTypeName.BINARY).as(stringType).named("typed_value") // stores person.name field
230         //     .named("name")
231         //     .optionalGroup
232         //       .optional(PrimitiveTypeName.INT32).named("typed_value").named("age") // stores person.age field
233         //     .named("typed_value")
234         //   .named("person")
235 
236       case ObjectType.STRING =>
237         Types.primitive(PrimitiveTypeName.BINARY, repetition)
238           .as(LogicalTypeAnnotation.stringType())
239 
240       case ObjectType.DATE =>
241         Types.primitive(PrimitiveTypeName.INT64, repetition)
242           .as(LogicalTypeAnnotation.timestampType(true, TimeUnit.MICROS))
243 
244       case ObjectType.UUID =>
245         Types.primitive(PrimitiveTypeName.FIXED_LEN_BYTE_ARRAY, repetition)
246           .length(16)
247           .as(LogicalTypeAnnotation.uuidType())
248 
249       case ObjectType.LIST =>
250         Types.optionalList()
251           .element(buildType("element", bindings.drop(1), geometries, Repetition.REQUIRED))
252 
253       case ObjectType.MAP =>
254         Types.optionalMap()
255           .key(buildType("key", bindings.slice(1, 2), geometries, Repetition.REQUIRED))
256           .value(buildType("value", bindings.slice(2, 3), geometries))
257 
258       case ObjectType.GEOMETRY =>
259         geometries.schema(bindings(1))
260 
261       case binding =>
262         throw new UnsupportedOperationException(s"No mapping defined for type: $binding")
263     }
264     builder.named(name)
265   }

268   private case class ParquetFields(fields: Seq[Type]) {
269     def toMessageType(typeName: String): MessageType = new MessageType(ColumnName.encode(typeName), fields.asJava)
270     def toMessageType(typeName: String, fieldFilter: Set[String]): MessageType =
271       new MessageType(ColumnName.encode(typeName), fields.filter(f => fieldFilter.contains(f.getName)).asJava)
272   }
