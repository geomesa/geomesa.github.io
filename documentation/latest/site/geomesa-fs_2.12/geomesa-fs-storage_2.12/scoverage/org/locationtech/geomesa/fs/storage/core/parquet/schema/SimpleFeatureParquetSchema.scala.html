30 /**
31  * A paired simple feature type and parquet schema
32  *
33  * @param sft simple feature type
34  * @param metadata file metadata
35  * @param messageType parquet schema
36  * @param fields parquet message schema
37  */
38 class SimpleFeatureParquetSchema private (
39     val sft: SimpleFeatureType,
40     val metadata: java.util.Map[String, String],
41     val messageType: MessageType,

44   /**
45    * Gets the schema used for reading (excludes some columns only used for partitioning)
46    *
47    * @return
48    */
49   def read(): SimpleFeatureParquetSchema = {
50     val fieldFilter =
51       fields.fields.map(_.getName)
52         .filterNot(n => n.endsWith(ZValueField.Z2ValueFieldSuffix) || n.endsWith(ZValueField.XZ2ValueFieldSuffix))
53     new SimpleFeatureParquetSchema(sft, metadata, fields.toMessageType(sft.getTypeName, fieldFilter.toSet), fields)
54   }
55 }

73   /**
74    * Simple feature type configuration
75    *
76    * @param sft simple feature type
77    * @return
78    */
79   def sftConf(sft: SimpleFeatureType): Map[String, String] = {
80     Map(
81       SftNameKey -> Option(sft.getName.getNamespaceURI).map(ns => s"$ns:${sft.getTypeName}").getOrElse(sft.getTypeName),
82       SftSpecKey -> SimpleFeatureTypes.encodeType(sft, includeUserData = true)

84   }

86   /**
87     * Extract the simple feature type from a parquet read context. The read context
88     * contains both file metadata and the provided read conf
89     *
90     * @param context parquet read context
91     * @return
92     */
93   def read(context: InitContext): Option[SimpleFeatureParquetSchema] = {
94     // copy in the file level metadata
95     val kvMeta = context.getKeyValueMetadata.asScala.collect { case (k, v) if !v.isEmpty => k -> v.iterator.next }
96     // noinspection ScalaDeprecation
97     val gmMeta = Seq(SftNameKey, SftSpecKey).flatMap { key =>
98       val value =
99         try { context.getParquetConfiguration.get(key) } catch {
100           // noinspection ScalaDeprecation - use the older method for spark compatibility
101           case _: NoSuchMethodError => context.getConfiguration.get(key)
102         }
103       Option(value).map(key -> _)
104     }
105     // note: context meta goes second so that transforms override the file-level keys
106     apply(new PlainParquetConfiguration((kvMeta.toMap ++ gmMeta.toMap).asJava)).map(_.read())
107   }
108 
109   /**
110     * Extract the simple feature type from an existing parquet file, without any known read context
111     *
112     * @param footer parquet file footer
113     * @return
114     */
115   def read(footer: FileMetaData): Option[SimpleFeatureParquetSchema] =
116     apply(new PlainParquetConfiguration(footer.getKeyValueMetaData)).map(_.read())
117 
118   /**
119    * Get a schema for writing. Encoding can be configured through attribute `encoding` user data.
120    *
121    * @param conf write configuration, including the sft spec
122    * @return
123    */
124   def apply(conf: ParquetConfiguration): Option[SimpleFeatureParquetSchema] = {
125     for {
126       name <- Option(conf.get(SftNameKey))
127       spec <- Option(conf.get(SftSpecKey))
128     } yield {
129       val sft = SimpleFeatureTypes.createImmutableType(name, spec)
130       val metadata = Map(
131         SftNameKey -> name,
132         SftSpecKey -> spec,
133       ) ++ Option(conf.get(PartitionKey)).map(PartitionKey -> _)
134       val parquet = schema(sft)
135       val messageType = {
136         val base = parquet.toMessageType(sft.getTypeName)
137         // if an iceberg name mapping is provided, stamp the table's field ids onto the schema by name,
138         // so that written files are read back by field id rather than falling back to name-based mapping
139         Option(conf.get(IcebergNameMappingKey)) match {
140           case None => base
141           case Some(json) => ParquetSchemaUtil.applyNameMapping(base, NameMappingParser.fromJson(json))
142         }
143       }
144       new SimpleFeatureParquetSchema(sft, metadata.asJava, messageType, parquet)
145     }
146   }

148   /**
149    * Gets the parquet schema for a feature type
150    *
151    * @param sft simple feature type
152    * @param conf storage configuration
153    * @return
154    */
155   def apply(sft: SimpleFeatureType, conf: Map[String, String]): SimpleFeatureParquetSchema = {
156     val config = new PlainParquetConfiguration(conf.asJava)
157     setSft(config, sft)
158     apply(config).get
159   }

161   /**
162    * Get the message type for a simple feature type. We need the full sft in order to ensure field ids are
163    * consistent, but the schema may be reduced (e.g. on read) based on the filter sft
164    *
165    * @param sft simple feature type
166    * @return
167    */
168   private def schema(sft: SimpleFeatureType): ParquetFields = {
169     val builder = Seq.newBuilder[Type]
170     builder +=
171       Types.required(PrimitiveTypeName.BINARY)
172         .as(LogicalTypeAnnotation.stringType())
173         .named(FeatureIdField)
174     builder +=
175       Types.optional(PrimitiveTypeName.BINARY)
176         .as(LogicalTypeAnnotation.stringType())
177         .named(VisibilitiesField)
178 
179     sft.getAttributeDescriptors.asScala.foreach { d =>
180       val name = ColumnName(d.getLocalName)
181       val objectType = ObjectType.selectType(d)
182       if (objectType.head == ObjectType.GEOMETRY) {
183         val encoding = d.getEncoding.fold[GeometryEncoding](GeometryEncoding.GeoParquetWkb)(GeometryEncoding.apply)
184         builder += encoding.schema(objectType(1)).named(name.column)
185         builder += BoundingBoxField.parquetSchema(name.column)
186         builder += ZValueField.parquetSchema(name.column, objectType(1))
187       } else if (objectType.last == ObjectType.JSON) {
188         builder += d.getJsonSchema().fold(buildVariantType(name.column))(buildStructuralType(name.column, _))
189       } else {
190         builder += buildType(name.column, objectType)
191       }
192     }
193     ParquetFields(builder.result())
194   }
195 
196   /**
197    * Builds the schema type for an attribute
198    *
199    * @param name field name
200    * @param bindings object type
201    * @param repetition repetition
202    * @return
203    */
204   private def buildType(
205       name: String,
206       bindings: Seq[ObjectType],
207       repetition: Repetition = Repetition.OPTIONAL): Type = {
208     val builder = bindings.head match {
209       case ObjectType.INT     => Types.primitive(PrimitiveTypeName.INT32, repetition)
210       case ObjectType.DOUBLE  => Types.primitive(PrimitiveTypeName.DOUBLE, repetition)
211       case ObjectType.LONG    => Types.primitive(PrimitiveTypeName.INT64, repetition)
212       case ObjectType.FLOAT   => Types.primitive(PrimitiveTypeName.FLOAT, repetition)
213       case ObjectType.BOOLEAN => Types.primitive(PrimitiveTypeName.BOOLEAN, repetition)
214       case ObjectType.BYTES   => Types.primitive(PrimitiveTypeName.BINARY, repetition)
215 
216       case ObjectType.STRING =>
217         Types.primitive(PrimitiveTypeName.BINARY, repetition)
218           .as(LogicalTypeAnnotation.stringType())
219 
220       case ObjectType.DATE =>
221         Types.primitive(PrimitiveTypeName.INT64, repetition)
222           .as(LogicalTypeAnnotation.timestampType(true, TimeUnit.MICROS))
223 
224       case ObjectType.UUID =>
225         Types.primitive(PrimitiveTypeName.FIXED_LEN_BYTE_ARRAY, repetition)
226           .length(16)
227           .as(LogicalTypeAnnotation.uuidType())
228 
229       case ObjectType.LIST =>
230         Types.optionalList()
231           .element(buildType("element", bindings.drop(1), Repetition.REQUIRED))
232 
233       case ObjectType.MAP =>
234         Types.optionalMap()
235           .key(buildType("key", bindings.slice(1, 2), Repetition.REQUIRED))
236           .value(buildType("value", bindings.slice(2, 3)))
237 
238       case binding =>
239         throw new UnsupportedOperationException(s"No mapping defined for type: $binding")
240     }
241     builder.named(name)
242   }

244   private def buildVariantType(name: String): Type = {
245     Types.buildGroup(Repetition.OPTIONAL).as(LogicalTypeAnnotation.variantType(1))
246       .required(PrimitiveTypeName.BINARY).named("metadata")
247       .required(PrimitiveTypeName.BINARY).named("value")
248       .named(name)
249   }

265   private def buildStructuralType(name: String, schemaDef: String): Type = {
266     // re-use the iceberg method so that the fields align
267     val icebergType = SimpleFeatureIcebergSchema.buildStructuralType(schemaDef, () => -1)
268     // top-level field is optional so that null json values are allowed
269     val field = NestedField.optional(name).withId(-1).ofType(icebergType).build()
270     new TypeToMessageType().field(field)
271   }

273   private case class ParquetFields(fields: Seq[Type]) {
274     def toMessageType(typeName: String): MessageType = new MessageType(ColumnName.encode(typeName), fields.asJava)
275     def toMessageType(typeName: String, fieldFilter: Set[String]): MessageType =
276       new MessageType(ColumnName.encode(typeName), fields.filter(f => fieldFilter.contains(f.getName)).asJava)
277   }
