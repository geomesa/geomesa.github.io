88   /**
89     * Extract the simple feature type from a parquet read context. The read context
90     * contains both file metadata and the provided read conf
91     *
92     * @param context parquet read context
93     * @return
94     */
95   def read(context: InitContext): Option[SimpleFeatureParquetSchema] = {
96     // copy in the file level metadata
97     val kvMeta = context.getKeyValueMetadata.asScala.collect { case (k, v) if !v.isEmpty => k -> v.iterator.next }
98     // noinspection ScalaDeprecation
99     val gmMeta = Seq(SftNameKey, SftSpecKey, GeometryEncodingKey).flatMap { key =>
100       val value =
101         try { context.getParquetConfiguration.get(key) } catch {
102           // noinspection ScalaDeprecation - use the older method for spark compatibility
103           case _: NoSuchMethodError => context.getConfiguration.get(key)
104         }
105       Option(value).map(key -> _)
106     }
107     // note: context meta goes second so that transforms override the file-level keys
108     apply(new PlainParquetConfiguration((kvMeta.toMap ++ gmMeta.toMap).asJava)).map(_.read())
109   }

120   /**
121    * Get a schema for writing. Encoding can be configured through `geomesa.parquet.geometries` and `geomesa.fs.visibilities`
122    *
123    * @param conf write configuration, including the sft spec
124    * @return
125    */
126   def apply(conf: ParquetConfiguration): Option[SimpleFeatureParquetSchema] = {
127     for {
128       name <- Option(conf.get(SftNameKey))
129       spec <- Option(conf.get(SftSpecKey))
130     } yield {
131       val sft = SimpleFeatureTypes.createImmutableType(name, spec)
132       // TODO in iceberg, geometry encoding is now specified at the column level - use that once we support more than WKB
133       val geometries = Option(conf.get(GeometryEncodingKey)).map(GeometryEncoding.apply).getOrElse(GeometryEncoding.GeoParquetWkb)
134       val metadata = Map(
135         SftNameKey -> name,
136         SftSpecKey -> spec,
137         GeometryEncodingKey -> geometries.toString,
138       ) ++ Option(conf.get(PartitionKey)).map(PartitionKey -> _)
139       val parquet = schema(sft, geometries)
140       val messageType = {
141         val base = parquet.toMessageType(sft.getTypeName)
142         // if an iceberg name mapping is provided, stamp the table's field ids onto the schema by name,
143         // so that written files are read back by field id rather than falling back to name-based mapping
144         Option(conf.get(IcebergNameMappingKey)) match {
145           case None => base
146           case Some(json) => ParquetSchemaUtil.applyNameMapping(base, NameMappingParser.fromJson(json))
147         }
148       }
149       new SimpleFeatureParquetSchema(sft, geometries, metadata.asJava, messageType, parquet)
150     }
151   }

166   /**
167    * Get the message type for a simple feature type. We need the full sft in order to ensure field ids are
168    * consistent, but the schema may be reduced (e.g. on read) based on the filter sft
169    *
170    * @param sft simple feature type
171    * @param geometries geometry type encoding
172    * @return
173    */
174   private def schema(sft: SimpleFeatureType, geometries: GeometryEncoding): ParquetFields = {
175     val builder = Seq.newBuilder[Type]
176     builder +=
177       Types.required(PrimitiveTypeName.BINARY)
178         .as(LogicalTypeAnnotation.stringType())
179         .named(FeatureIdField)
180     builder +=
181       Types.optional(PrimitiveTypeName.BINARY)
182         .as(LogicalTypeAnnotation.stringType())
183         .named(VisibilitiesField)
184 
185     sft.getAttributeDescriptors.asScala.foreach { d =>
186       val name = ColumnName(d.getLocalName)
187       val objectType = ObjectType.selectType(d)
188       if (objectType.head == ObjectType.GEOMETRY) {
189         builder += geometries.schema(objectType(1)).named(name.column)
190         builder += BoundingBoxField.parquetSchema(name.column)
191         builder += ZValueField.parquetSchema(name.column, objectType(1))
192       } else if (objectType.last == ObjectType.JSON) {
193         builder += d.getJsonSchema().fold(buildVariantType(name.column))(buildStructuralType(name.column, _))
194       } else {
195         builder += buildType(name.column, objectType)
196       }
197     }
198     ParquetFields(builder.result())
199   }

201   /**
202    * Builds the schema type for an attribute
203    *
204    * @param name field name
205    * @param bindings object type
206    * @param repetition repetition
207    * @return
208    */
209   private def buildType(
210       name: String,
211       bindings: Seq[ObjectType],
212       repetition: Repetition = Repetition.OPTIONAL): Type = {
213     val builder = bindings.head match {
214       case ObjectType.INT     => Types.primitive(PrimitiveTypeName.INT32, repetition)
215       case ObjectType.DOUBLE  => Types.primitive(PrimitiveTypeName.DOUBLE, repetition)
216       case ObjectType.LONG    => Types.primitive(PrimitiveTypeName.INT64, repetition)
217       case ObjectType.FLOAT   => Types.primitive(PrimitiveTypeName.FLOAT, repetition)
218       case ObjectType.BOOLEAN => Types.primitive(PrimitiveTypeName.BOOLEAN, repetition)
219       case ObjectType.BYTES   => Types.primitive(PrimitiveTypeName.BINARY, repetition)
220 
221       case ObjectType.STRING =>
222         Types.primitive(PrimitiveTypeName.BINARY, repetition)
223           .as(LogicalTypeAnnotation.stringType())
224 
225       case ObjectType.DATE =>
226         Types.primitive(PrimitiveTypeName.INT64, repetition)
227           .as(LogicalTypeAnnotation.timestampType(true, TimeUnit.MICROS))
228 
229       case ObjectType.UUID =>
230         Types.primitive(PrimitiveTypeName.FIXED_LEN_BYTE_ARRAY, repetition)
231           .length(16)
232           .as(LogicalTypeAnnotation.uuidType())
233 
234       case ObjectType.LIST =>
235         Types.optionalList()
236           .element(buildType("element", bindings.drop(1), Repetition.REQUIRED))
237 
238       case ObjectType.MAP =>
239         Types.optionalMap()
240           .key(buildType("key", bindings.slice(1, 2), Repetition.REQUIRED))
241           .value(buildType("value", bindings.slice(2, 3)))
242 
243       case binding =>
244         throw new UnsupportedOperationException(s"No mapping defined for type: $binding")
245     }
246     builder.named(name)
247   }

270   private def buildStructuralType(name: String, schemaDef: String): Type = {
271     // re-use the iceberg method so that the fields align
272     val icebergType = SimpleFeatureIcebergSchema.buildStructuralType(schemaDef, () => -1)
273     // top-level field is optional so that null json values are allowed
274     val field = NestedField.optional(name).withId(-1).ofType(icebergType).build()
275     new TypeToMessageType().field(field)
276   }
