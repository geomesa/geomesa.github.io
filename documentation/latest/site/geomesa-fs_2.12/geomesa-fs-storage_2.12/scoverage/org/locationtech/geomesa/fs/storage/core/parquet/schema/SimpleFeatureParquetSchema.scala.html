29 /**
30  * A paired simple feature type and parquet schema
31  *
32  * @param sft simple feature type
33  * @param geometries geometry type encoding
34  * @param metadata file metadata
35  * @param messageType parquet schema
36  * @param fields parquet message schema
37  */
38 class SimpleFeatureParquetSchema private (
39     val sft: SimpleFeatureType,
40     val geometries: GeometryEncoding,
41     val metadata: java.util.Map[String, String],

45   /**
46    * Gets the schema used for reading (excludes some columns only used for partitioning)
47    *
48    * @return
49    */
50   def read(): SimpleFeatureParquetSchema = {
51     val fieldFilter =
52       fields.fields.map(_.getName)
53         .filterNot(n => n.endsWith(ZValueField.Z2ValueFieldSuffix) || n.endsWith(ZValueField.XZ2ValueFieldSuffix))
54     new SimpleFeatureParquetSchema(sft, geometries, metadata, fields.toMessageType(sft.getTypeName, fieldFilter.toSet), fields)
55   }
56 }

64   /**
65    * Encodes the feature type in the conf
66    *
67    * @param conf conf
68    * @param sft feature type
69    */
70   def setSft(conf: ParquetConfiguration, sft: SimpleFeatureType): Unit = {
71     val name = Option(sft.getName.getNamespaceURI).map(ns => s"$ns:${sft.getTypeName}").getOrElse(sft.getTypeName)
72     conf.set(SftNameKey, name)
73     conf.set(SftSpecKey, SimpleFeatureTypes.encodeType(sft, includeUserData = true))
74   }

76   /**
77     * Extract the simple feature type from a parquet read context. The read context
78     * contains both file metadata and the provided read conf
79     *
80     * @param context parquet read context
81     * @return
82     */
83   def read(context: InitContext): Option[SimpleFeatureParquetSchema] = {
84     // copy in the file level metadata
85     val kvMeta = context.getKeyValueMetadata.asScala.collect { case (k, v) if !v.isEmpty => k -> v.iterator.next }
86     // noinspection ScalaDeprecation
87     val gmMeta = Seq(SftNameKey, SftSpecKey, GeometryEncodingKey).flatMap { key =>
88       val value =
89         try { context.getParquetConfiguration.get(key) } catch {
90           // noinspection ScalaDeprecation - use the older method for spark compatibility
91           case _: NoSuchMethodError => context.getConfiguration.get(key)
92         }
93       Option(value).map(key -> _)
94     }
95     // note: context meta goes second so that transforms override the file-level keys
96     apply(new PlainParquetConfiguration((kvMeta.toMap ++ gmMeta.toMap).asJava)).map(_.read())
97   }

99   /**
100     * Extract the simple feature type from an existing parquet file, without any known read context
101     *
102     * @param footer parquet file footer
103     * @return
104     */
105   def read(footer: FileMetaData): Option[SimpleFeatureParquetSchema] =
106     apply(new PlainParquetConfiguration(footer.getKeyValueMetaData)).map(_.read())

108   /**
109    * Get a schema for writing. Encoding can be configured through `geomesa.parquet.geometries` and `geomesa.fs.visibilities`
110    *
111    * @param conf write configuration, including the sft spec
112    * @return
113    */
114   def apply(conf: ParquetConfiguration): Option[SimpleFeatureParquetSchema] = {
115     for {
116       name <- Option(conf.get(SftNameKey))
117       spec <- Option(conf.get(SftSpecKey))
118     } yield {
119       val sft = SimpleFeatureTypes.createImmutableType(name, spec)
120       val geometries = Option(conf.get(GeometryEncodingKey)).map(GeometryEncoding.apply).getOrElse(GeometryEncoding.GeoParquetWkb)
121       val metadata = Map(
122         SftNameKey -> name,
123         SftSpecKey -> spec,
124         GeometryEncodingKey -> geometries.toString,
125       ) ++ Option(conf.get(PartitionKey)).map(PartitionKey -> _)
126       val parquet = schema(sft, geometries)
127       new SimpleFeatureParquetSchema(sft, geometries, metadata.asJava, parquet.toMessageType(sft.getTypeName), parquet)
128     }
129   }

131   /**
132    * Gets the parquet schema for a feature type
133    *
134    * @param sft simple feature type
135    * @param conf storage configuration
136    * @return
137    */
138   def apply(sft: SimpleFeatureType, conf: Map[String, String]): SimpleFeatureParquetSchema = {
139     val config = new PlainParquetConfiguration(conf.asJava)

144   /**
145    * Get the message type for a simple feature type. We need the full sft in order to ensure field ids are
146    * consistent, but the schema may be reduced (e.g. on read) based on the filter sft
147    *
148    * @param sft simple feature type
149    * @param geometries geometry type encoding
150    * @return
151    */
152   private def schema(sft: SimpleFeatureType, geometries: GeometryEncoding): ParquetFields = {
153     // note: for iceberg compatibility, field ids need to start at one and increment (without gaps) across all top-level fields.
154     // All nested fields (structs, lists) get ids *after* all the top-level fields, once again incrementing without gaps
155     val fieldIds = new AtomicInteger(1)
156     val nestedFieldIds = {
157       val extraGeomCount = 2 * sft.getAttributeDescriptors.asScala.count {
158         case d if classOf[Geometry].isAssignableFrom(d.getType.getBinding) => true
159         case _ => false
160       }
161       new AtomicInteger(3 + sft.getAttributeCount + extraGeomCount)
162     }
163 
164     val builder = Seq.newBuilder[Type]
165     builder +=
166       Types.required(PrimitiveTypeName.BINARY)
167         .id(fieldIds.getAndIncrement())
168         .as(LogicalTypeAnnotation.stringType())
169         .named(FeatureIdField)
170     builder +=
171       Types.optional(PrimitiveTypeName.BINARY)
172         .id(fieldIds.getAndIncrement())
173         .as(LogicalTypeAnnotation.stringType())
174         .named(VisibilitiesField)
175 
176     sft.getAttributeDescriptors.asScala.foreach { d =>
177       val name = ColumnName(d.getLocalName)
178       val objectType = ObjectType.selectType(d)
179       builder += buildType(name.column, objectType, geometries, fieldIds, nestedFieldIds)
180       if (objectType.head == ObjectType.GEOMETRY) {
181         builder += BoundingBoxField.parquetSchema(name.column, fieldIds, nestedFieldIds)
182         builder += ZValueField.parquetSchema(name.column, objectType(1), fieldIds)
183       }
184     }
185     ParquetFields(builder.result())
186   }

188   /**
189    * Builds the schema type for an attribute
190    *
191    * @param name field name
192    * @param bindings object type
193    * @param geometries geometry type encodings
194    * @param repetition repetition
195    * @return
196    */
197   private def buildType(
198       name: String,
199       bindings: Seq[ObjectType],
200       geometries: GeometryEncoding,
201       fieldIds: AtomicInteger,
202       nestedFieldIds: AtomicInteger,
203       repetition: Repetition = Repetition.OPTIONAL): Type = {
204     val builder = bindings.head match {
205       case ObjectType.INT     => Types.primitive(PrimitiveTypeName.INT32, repetition).id(fieldIds.getAndIncrement())
206       case ObjectType.DOUBLE  => Types.primitive(PrimitiveTypeName.DOUBLE, repetition).id(fieldIds.getAndIncrement())
207       case ObjectType.LONG    => Types.primitive(PrimitiveTypeName.INT64, repetition).id(fieldIds.getAndIncrement())
208       case ObjectType.FLOAT   => Types.primitive(PrimitiveTypeName.FLOAT, repetition).id(fieldIds.getAndIncrement())
209       case ObjectType.BOOLEAN => Types.primitive(PrimitiveTypeName.BOOLEAN, repetition).id(fieldIds.getAndIncrement())
210       case ObjectType.BYTES   => Types.primitive(PrimitiveTypeName.BINARY, repetition).id(fieldIds.getAndIncrement())
211 
212       case ObjectType.STRING if bindings.last == ObjectType.JSON =>
213         Types.buildGroup(repetition)
214           .id(fieldIds.getAndIncrement())
215           .as(LogicalTypeAnnotation.variantType(1))
216           // note: the iceberg api does not define nested fields for variants so we don't set id here
217           .required(PrimitiveTypeName.BINARY).named("metadata")
218           .required(PrimitiveTypeName.BINARY).named("value")
219         // TODO can pull out known fields for shredding, but trino doesn't support reading shredded variants yet
220         // val partiallyShredded: GroupType =
221         //   Types.buildGroup(Repetition.REQUIRED).as(LogicalTypeAnnotation.variantType(1.toByte))
222         //   .required(PrimitiveTypeName.BINARY).named("metadata")
223         //   .optional(PrimitiveTypeName.BINARY).named("value") // stores {custom_field_xyz: 123}
224         //   .optionalGroup
225         //     .optionalGroup
226         //       .optional(PrimitiveTypeName.BINARY).as(stringType).named("typed_value") // stores person.name field
227         //     .named("name")
228         //     .optionalGroup
229         //       .optional(PrimitiveTypeName.INT32).named("typed_value").named("age") // stores person.age field
230         //     .named("typed_value")
231         //   .named("person")
232 
233       case ObjectType.STRING =>
234         Types.primitive(PrimitiveTypeName.BINARY, repetition)
235           .id(fieldIds.getAndIncrement())
236           .as(LogicalTypeAnnotation.stringType())
237 
238       case ObjectType.DATE =>
239         Types.primitive(PrimitiveTypeName.INT64, repetition)
240           .id(fieldIds.getAndIncrement())
241           .as(LogicalTypeAnnotation.timestampType(true, TimeUnit.MICROS))
242 
243       case ObjectType.UUID =>
244         Types.primitive(PrimitiveTypeName.FIXED_LEN_BYTE_ARRAY, repetition)
245           .id(fieldIds.getAndIncrement())
246           .length(16)
247           .as(LogicalTypeAnnotation.uuidType())
248 
249       case ObjectType.LIST =>
250         Types.optionalList()
251           .id(fieldIds.getAndIncrement())
252           .element(buildType("element", bindings.drop(1), geometries, nestedFieldIds, null /* should not be used*/, Repetition.REQUIRED))
253 
254       case ObjectType.MAP =>
255         Types.optionalMap()
256           .id(fieldIds.getAndIncrement())
257           .key(buildType("key", bindings.slice(1, 2), geometries, nestedFieldIds, null /* should not be used*/, Repetition.REQUIRED))
258           .value(buildType("value", bindings.slice(2, 3), geometries, nestedFieldIds, null /* should not be used*/))
259 
260       case ObjectType.GEOMETRY =>
261         geometries.schema(bindings(1), fieldIds, nestedFieldIds)
262 
263       case binding =>
264         throw new UnsupportedOperationException(s"No mapping defined for type: $binding")
265     }
266     builder.named(name)
267   }

270   private case class ParquetFields(fields: Seq[Type]) {

274   }
