23 /**
24  * Catalog for jdbc metadata
25  *
26  * @param context file system context
27  */
28 class JdbcMetadataCatalog(context: FileSystemContext) extends StorageMetadataCatalog {
29 
30   // fail fast if url is not defined or invalid
31   private val conf = JdbcMetadataConfig(context.conf)
32   private val driver = JdbcMetadataCatalog.createConnectionFactory(conf)
33   private val root = context.root.toString
34 
35   override def getTypeNames: Seq[String] = {
36     val meta = new MetadataTable(conf.schema, conf.tablePrefix, root, null)
37     WithClose(driver.createConnection()) { cx =>
38       WithClose(cx.getMetaData.getTables(null, meta.schema, meta.tableName, null)) { rs =>
39         if (!rs.next()) {
40           return Seq.empty
41         }
42       }
43       meta.selectTypeNames(cx, root)
44     }
45   }
46 
47   override def load(typeName: String): StorageMetadata = {
48     val pool = JdbcMetadataCatalog.createDataSource(driver, conf)
49     try {
50       val meta = new MetadataTable(conf.schema, conf.tablePrefix, root, typeName)
51       val files = new FilesTable(conf.schema, conf.tablePrefix, root, typeName)
52       new JdbcMetadata(pool, meta, files, context.namespace)
53     } catch {
54       case NonFatal(e) => CloseQuietly(pool).foreach(e.addSuppressed); throw e
55     }
56   }
57 
58   override def create(sft: SimpleFeatureType, partitions: Seq[String], targetFileSize: Option[Long]): StorageMetadata = {
59     // load the partition scheme first in case it fails
60     partitions.foreach(PartitionSchemeFactory.load(sft, _))
61     val pool = JdbcMetadataCatalog.createDataSource(driver, conf)
62     try {
63       val meta = new MetadataTable(conf.schema, conf.tablePrefix, root, sft.getTypeName)
64       val files = new FilesTable(conf.schema, conf.tablePrefix, root, sft.getTypeName)
65       WithClose(pool.getConnection()) { cx =>
66         meta.create(cx)
67         meta.insert(cx, sft)
68         meta.insert(cx, partitions)
69         targetFileSize.foreach(size => meta.insert(cx, Metadata.TargetFileSize, size.toString))
70         files.create(cx)
71       }
72       new JdbcMetadata(pool, meta, files, context.namespace)
73     } catch {
74       case NonFatal(e) => CloseQuietly(pool).foreach(e.addSuppressed); throw e
75     }
76   }
77 }

79 private object JdbcMetadataCatalog extends LazyLogging {
80 
81   /**
82    * Create (and validate) a connection factory
83    *
84    * @param config config
85    * @return
86    */
87   private def createConnectionFactory(config: JdbcMetadataConfig): DriverManagerConnectionFactory = {
88     config.driver.foreach(Class.forName) // required for older drivers
89     val props = new Properties()
90     config.user.foreach(props.put("user", _))
91     config.password.foreach(props.put("password", _))
92 
93     val driver = new DriverManagerConnectionFactory(config.url, props)
94 
95     // validate the connection parameters
96     WithClose(driver.createConnection()) { connection =>
97       if (!connection.isValid(10)) {
98         throw new IllegalArgumentException(
99           s"Could not create valid connection using configuration ${config.url}")
100       }
101     }
102 
103     driver
104   }
105 
106   /**
107     * Create a jdbc data source based on a configuration
108     *
109     * @param config config
110     * @return
111     */
112   private def createDataSource(driver: DriverManagerConnectionFactory, config: JdbcMetadataConfig): PoolingDataSource[PoolableConnection] = {
113     val poolConfig = new GenericObjectPoolConfig[PoolableConnection]()
114 
115     config.maxIdle.foreach(poolConfig.setMaxIdle)
116     config.minIdle.foreach(poolConfig.setMinIdle)
117     config.maxSize.foreach(poolConfig.setMaxTotal)
118     config.fairness.foreach(poolConfig.setFairness)
119     config.testOnBorrow.foreach(poolConfig.setTestOnBorrow)
120     config.testOnCreate.foreach(poolConfig.setTestOnCreate)
121     config.testWhileIdle.foreach(poolConfig.setTestWhileIdle)
122 
123     val factory = new PoolableConnectionFactory(driver, null)
124     val pool = new GenericObjectPool(factory, poolConfig)
125     factory.setPool(pool)
126 
127     new PoolingDataSource[PoolableConnection](pool)
128   }
