82   def apply(reader: FileSystemPathReader, files: Seq[StorageFile], threads: Int): CloseableIterator[SimpleFeature] = {
83     if (threads < 2) {
84       val mods = new java.util.HashSet[String]()
85       // ensure files are sorted in reverse chronological order so mods are handled correctly
86       CloseableIterator.wrap(files.sorted).flatMap(f => read(reader, f, mods))
87     } else {
88       val queue = new LinkedBlockingQueue[SimpleFeature](2000000)
89       val es = Executors.newFixedThreadPool(threads)
90 
91       val phaser = new Phaser(1) {
92         override protected def onAdvance(phase: Int, registeredParties: Int): Boolean = {
93           // when all tasks have been completed, shutdown the executor service
94           es.shutdown()
95           true // return true to indicate the phaser should terminate
96         }
97       }
98 
99       try {
100         def submitParallelRead(toRead: Seq[StorageFile]): Unit = {
101           // ensure that we don't register too many parties on this phaser
102           toRead.grouped(PhaserUtils.MaxParties - 1).foreach { group =>
103             val child = new Phaser(phaser)
104             child.register() // register new task
105             es.submit(new ChainedReaderTask(es, child, reader, group, Seq.empty, queue))
106           }
107         }
108 
109         // if a partitions has modifications, it must be read separately to ensure they're handled correctly
110         val partitionsWithMods = files.collect { case f if f.action != StorageFileAction.Append => f.partition }.distinct
111         if (partitionsWithMods.isEmpty) {
112          submitParallelRead(files)
113         } else {
114           files.groupBy(_.partition).foreach { case (partition, group) =>
115             if (!partitionsWithMods.contains(partition)) {
116               submitParallelRead(group)
117             } else {
118               val chain = Seq.newBuilder[Seq[StorageFile]]
119               var remaining = group.sorted
120               var i = remaining.indexWhere(_.action != StorageFileAction.Append)
121               while (i != -1) {
122                 i = math.min(i, PhaserUtils.MaxParties - 1) // ensure that we don't register too many parties on this phaser
123                 val (first, rest) = remaining.splitAt(i + 1)
124                 chain += first
125                 remaining = rest
126                 i = remaining.indexWhere(_.action != StorageFileAction.Append)
127               }
128               if (remaining.nonEmpty) {
129                 chain += remaining
130               }
131               val groups = chain.result()
132               val child = new Phaser(phaser)
133               child.register() // register new task
134               es.submit(new ChainedReaderTask(es, child, reader, groups.head, groups.tail, queue))
135             }
136           }
137         }
138       } catch {
139         case NonFatal(e) => es.shutdownNow(); throw e
140       } finally {
141         phaser.arriveAndDeregister()
142       }
143 
144       new FileSystemThreadedReader(es, phaser, queue)
145     }
146   }

148   /**
149     * Reads a file
150     *
151     * @param reader reader
152     * @param file file to read
153     * @param mods collection to track modifications
154     * @return
155     */
156   private def read(reader: FileSystemPathReader, file: StorageFile, mods: java.util.Set[String]): CloseableIterator[SimpleFeature] = {
157     file.action match {
158       case StorageFileAction.Append => new AppendingReaderIterator(reader, reader.root.resolve(file.file), mods)
159       case StorageFileAction.Modify => new ModifyingReaderIterator(reader, reader.root.resolve(file.file), mods)
160       case StorageFileAction.Delete => new DeletingReaderIterator(reader, reader.root.resolve(file.file), mods)
161       case _ => throw new UnsupportedOperationException(s"Unexpected storage action: ${file.action}")
162     }
163   }

165   /**
166     * Performs a set of chained (dependent) reads. The groups must be handled serially, but the files within a
167     * given group can be read in parallel.
168     *
169     * @param es executor service for submitting new read tasks
170     * @param phaser phaser to track run lifecycle
171     * @param reader reader
172     * @param group current group of files that can be read in parallel
173     * @param chain remaining groups of files that must be read sequentially
174     * @param queue result queue
175     * @param mods modifications/deletes for this group of files
176     */
177   private class ChainedReaderTask(
178       es: ExecutorService,
179       phaser: Phaser,
180       reader: FileSystemPathReader,
181       group: Seq[StorageFile],
182       chain: Seq[Seq[StorageFile]],
183       queue: BlockingQueue[SimpleFeature],
184       mods: java.util.Set[String] = new java.util.HashSet[String]()
185     ) extends Runnable {
186 
187     override def run(): Unit = {
188       val child = new Phaser(1) {
189         override protected def onAdvance(phase: Int, registeredParties: Int): Boolean = {
190           // when this group is done, submit the next group for processing
191           try {
192             if (chain.nonEmpty) {
193               phaser.register() // register new task
194               es.submit(new ChainedReaderTask(es, phaser, reader, chain.head, chain.tail, queue, mods))
195             }
196             true // return true to indicate the phaser should terminate
197           } finally {
198             phaser.arriveAndDeregister()
199           }
200         }
201       }
202       try {
203         group.foreach { file =>
204           child.register() // register new task
205           es.submit(new ReaderTask(child, queue, file.file, read(reader, file, mods)))
206         }
207       } finally {
208         child.arriveAndDeregister()
209       }
210     }
211   }
