23 /**
24  * Synthetic metadata that just lists files on the filesystem
25  *
26  * @param context file system
27  * @param sft simple feature type
28  * @param orderedSchemes partition schemes, ordered according to the file system directory layout
29  * @param leafStorage if true, the final partition scheme is a prefix of the file name, instead of a regular directory
30  */
31 class ConverterMetadata(
32     context: FileSystemContext,
33     val sft: SimpleFeatureType,
34     val orderedSchemes: Seq[PartitionScheme],
35     val leafStorage: Boolean
36   ) extends StorageMetadata with CachedMetadata with LazyLogging {
37 
38   override val `type`: String = ConverterMetadata.MetadataType
39 
40   override val schemes: Set[PartitionScheme] = orderedSchemes.toSet
41 
42   private val depth = orderedSchemes.map {
43     case s: HierarchicalDateTimeScheme => s.depth
44     case _ => 1
45   }.sum - (if (leafStorage) { 1 } else { 0 })
46 
47   private val fs = ObjectStore(context)
48 
49   filesCache.refresh(BoxedUnit.UNIT) // kick off the initial load asynchronously
50 
51   override def addFile(file: StorageFile): Unit = throw new UnsupportedOperationException()
52   override def removeFile(file: StorageFile): Unit = throw new UnsupportedOperationException()
53   override def replaceFiles(existing: Seq[StorageFile], replacements: Seq[StorageFile]): Unit =
54     throw new UnsupportedOperationException()
55 
56   override def close(): Unit = try { super.close() } finally { fs.close() }
57 
58   override protected def buildFileList(): Seq[StorageFile] = {
59     logger.debug("Building file list")
60     val start = System.currentTimeMillis()
61     val result = Seq.newBuilder[StorageFile]
62     try {
63       val toCheck = new java.util.LinkedList[URI]()
64       fs.list(context.root).foreach(toCheck.add)
65       val maxDepth = depth + context.root.toString.count(_ == '/')
66       val rootPathLength = context.root.toString.length
67       while (!toCheck.isEmpty) {
68         val file = toCheck.remove()
69         logger.debug(s"Checking ${file.getPath}")
70         if (file.getPath.endsWith("/")) {
71           if (file.toString.count(_ == '/') <= maxDepth) {
72             fs.list(file).foreach(toCheck.add)
73           }
74         } else if (!fs.filename(file).startsWith(".")) { // ignore "hidden" files
75           val relativePath = file.toString.substring(rootPathLength)
76           var partitionParts = relativePath.split("/")
77           if (leafStorage) {
78             partitionParts(partitionParts.length - 1) = partitionParts.last.takeWhile(_ != '_')
79           } else {
80             partitionParts = partitionParts.dropRight(1)
81           }
82           var i = 0
83           val partitions = orderedSchemes.map {
84             case s: HierarchicalDateTimeScheme =>
85               val key = PartitionKey(s.name, partitionParts.slice(i, i + s.depth).mkString("/"))
86               i += s.depth
87               key
88             case s =>
89               val key = PartitionKey(s.name, partitionParts(i))
90               i += 1
91               key
92           }
93           result += StorageFile(relativePath, Partition(partitions.toSet), -1L)
94         }
95       }
96     } catch {
97       case NonFatal(e) => logger.error("Error building file list:", e); throw e
98     }
99     val list = result.result()
100     logger.debug(s"Found ${list.size} files in ${System.currentTimeMillis() - start}ms")
101     logger.whenTraceEnabled(list.foreach(f => logger.trace(f.toString)))
102     list
103   }
104 }
