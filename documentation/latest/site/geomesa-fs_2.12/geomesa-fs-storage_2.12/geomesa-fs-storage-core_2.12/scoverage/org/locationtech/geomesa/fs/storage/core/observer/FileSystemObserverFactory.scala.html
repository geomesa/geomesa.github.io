18 /**
19  * Factory for observing file writes
20  */
21 trait FileSystemObserverFactory extends Closeable {
22 
23   /**
24    * Called once after instantiating the factory
25    *
26    * @param storage parent storage
27    */
28   def init(storage: FileSystemStorage): Unit
29 
30   /**
31    * Create an observer for the given path
32    *
33    * @param path file path being written, relative to the root URL
34    * @return
35    */
36   def apply(path: URI): FileSystemObserver
37 }
38 
39 object FileSystemObserverFactory {
40 
41   object NoOpObserver extends FileSystemObserver {
42     override def apply(feature: SimpleFeature): Unit = {}
43     override def flush(): Unit = {}
44     override def close(): Unit = {}
45   }
46 
47   object CompositeObserver {
48     def apply(observers: Seq[FileSystemObserver]): FileSystemObserver = {
49       val seq = observers.flatMap {
50         case NoOpObserver => Seq.empty
51         case o: CompositeObserver => o.observers
52         case o => Seq(o)
53       }
54       if (seq.isEmpty) {
55         NoOpObserver
56       } else if (seq.lengthCompare(1) == 0) {
57         seq.head
58       } else {
59         new CompositeObserver(seq)
60       }
61     }
62   }
63 
64   /**
65    * Composite observer
66    *
67    * @param observers observers
68    */
69   class CompositeObserver(val observers: Seq[FileSystemObserver]) extends FileSystemObserver {
70 
71     override def apply(feature: SimpleFeature): Unit = observers.foreach(_.apply(feature))
72 
73     override def flush(): Unit = {
74       val errors = observers.flatMap(o => Try(o.flush()).failed.toOption)
75       if (errors.nonEmpty) {
76         throw collapseErrors(errors)
77       }
78     }
79 
80     override def close(): Unit = {
81       val errors = observers.flatMap(o => Try(o.close()).failed.toOption)
82       if (errors.nonEmpty) {
83         throw collapseErrors(errors)
84       }
85     }
86 
87     private def collapseErrors(errors: Seq[Throwable]): Throwable = {
88       val e = errors.head
89       errors.tail.foreach(e.addSuppressed)
90       e
91     }
92   }
93 }
