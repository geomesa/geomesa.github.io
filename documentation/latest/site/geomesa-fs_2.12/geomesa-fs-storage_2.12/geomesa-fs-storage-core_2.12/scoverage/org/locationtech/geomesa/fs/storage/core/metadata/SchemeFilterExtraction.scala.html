21 /**
22  * Mixin trait for matching partitions against a CQL filter
23  */
24 trait SchemeFilterExtraction extends AnyLogging {
25 
26   this: StorageMetadata =>
27 
28   protected def getFilters(filter: Filter): Seq[SchemeFilter] = {
29     val iter = schemes.iterator
30     val start = iter.map(_.getRangesForFilter(filter)).collectFirst { case Some(f) => f }
31     val filters = start match {
32       case None =>
33         // filter did not constrain partitions at all
34         Seq(SchemeFilter(filter, Seq.empty, extractAttributes(filter)))
35 
36       case Some(ranges) =>
37         // set up the initial scheme filters, without any spatial/attribute bounds
38         val initialRanges = ranges.map(r => Seq(r))
39         // add in the remaining partitions
40         val permutations = iter.foldLeft(initialRanges) { case (ranges, scheme) => addPartition(scheme, ranges, filter) }
41         // add in the attribute bounds based on the remaining filter for each scheme filter
42         val attributeBounds = extractAttributes(filter)
43         permutations.map(ranges => SchemeFilter(filter, ranges, attributeBounds))
44     }
45 
46     logger.debug(s"Extracted filters from ${ECQL.toCQL(filter)}:\n  ${filters.mkString("\n  ")}")
47     filters
48   }
49 
50   // add the next partition to the scheme filter
51   private def addPartition(scheme: PartitionScheme, ranges: Seq[Seq[PartitionRange]], filter: Filter): Seq[Seq[PartitionRange]] = {
52     val result = Seq.newBuilder[Seq[PartitionRange]]
53     ranges.foreach { filesTableFilter =>
54       scheme.getRangesForFilter(filter) match {
55         case None => result += filesTableFilter
56         case Some(ranges) =>
57           ranges.foreach { range =>
58             result += filesTableFilter :+ range
59           }
60       }
61     }
62     result.result()
63   }
64 
65   private def extractAttributes(filter: Filter): Seq[ColumnOr] = {
66     sft.columnBounds().flatMap { i =>
67       val d = sft.getDescriptor(i)
68       if (classOf[Geometry].isAssignableFrom(d.getType.getBinding)) {
69         val ors = FilterHelper.extractGeometries(filter, sft.getDescriptor(i).getLocalName).values.flatMap { g =>
70           val env = g.getEnvelopeInternal
71           // TODO make max ranges configurable
72           val ranges = if (d.getType.getBinding == classOf[Point]) {
73             Z2Encoder.ranges(Seq((env.getMinX, env.getMinY, env.getMaxX, env.getMaxY)), Some(8))
74           } else {
75             XZ2Encoder.ranges(Seq((env.getMinX, env.getMinY, env.getMaxX, env.getMaxY)), Some(8))
76           }
77           ranges.map { case (lower, upper) => ColumnBound(lower, upper) }
78         }
79         if (ors.isEmpty) { None } else { Some(ColumnOr(i, ors)) }
80       } else {
81         val ors = FilterHelper.extractAttributeBounds(filter, d.getLocalName, d.getType.getBinding).values.map { b =>
82           val lower = b.lower.value.map(StorageMetadata.TypeRegistry.encode).getOrElse("")
83           val upper = b.upper.value.map(StorageMetadata.TypeRegistry.encode).getOrElse("zzz")
84           ColumnBound(lower, upper)
85         }
86         if (ors.isEmpty) { None } else { Some(ColumnOr(i, ors)) }
87       }
88     }
89   }
90 }

97    * @param filter remaining ECQL filter that isn't accounted for with the other bounds
98    * @param partitions list of partition bound predicates (implicit AND between each element in the seq)
99    * @param columnBounds list of column bound predicates (implicit AND between each element in the seq)
100    */

104     columnBounds: Seq[ColumnOr]
105   )

107   case class ColumnOr(attribute: Int, bounds: Seq[ColumnBound])
108 
109   case class ColumnBound(lower: String, upper: String) {
110     def intersects(other: ColumnBounds): Boolean = other.lower <= upper && other.upper >= lower
111   }
