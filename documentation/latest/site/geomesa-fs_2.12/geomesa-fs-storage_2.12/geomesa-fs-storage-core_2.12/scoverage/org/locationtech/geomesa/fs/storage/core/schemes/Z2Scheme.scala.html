22 case class Z2Scheme(attribute: String, index: Int, bits: Int) extends PartitionScheme {
23 
24   import org.locationtech.geomesa.filter.{andFilters, ff}
25   import org.locationtech.geomesa.utils.geotools.CRS_EPSG_4326
26 
27   require(bits % 2 == 0, s"Bit precision must be an even number, but received $bits")
28 
29   private val xyBits = bits / 2
30   private val z2 = new Z2SFC(xyBits)
31   private val xRadius = (360d / math.pow(2, xyBits)) / 2
32   private val yRadius = (180d / math.pow(2, xyBits)) / 2
33   private val format = s"%0${digits(bits)}d"
34 
35   lazy private val wholeWorldRanges = Some(z2.ranges(Seq((-180, -90, 180, 90))))
36 
37   override val name: String = s"${Z2Scheme.name}:attribute=$attribute:bits=$bits"
38 
39   override def getPartition(feature: SimpleFeature): PartitionKey = {
40     val pt = feature.getAttribute(index).asInstanceOf[Point]
41     PartitionKey(name, format.format(z2.index(pt.getX, pt.getY)))
42   }
43 
44   override def getRangesForFilter(filter: Filter): Option[Seq[PartitionRange]] = {
45     getIndexRanges(filter).map { ranges =>
46       val builder = new RangeBuilder()
47       ranges.foreach { range =>
48         val lower = format.format(range.lower)
49         val upper = format.format(range.upper + 1)
50         builder += PartitionRange(name, lower, upper)
51       }
52       builder.result()
53     }
54   }
55 
56   override def getPartitionsForFilter(filter: Filter): Option[Seq[PartitionKey]] = {
57     getIndexRanges(filter).orElse(wholeWorldRanges).map { ranges =>
58       ranges.flatMap { range =>
59         val lower = range.lower
60         val steps = 1 + (range.upper - lower).toInt
61         Seq.tabulate(steps)(i => PartitionKey(name, format.format(lower + i)))
62       }
63     }
64   }
65 
66   override def getCoveringFilter(partition: PartitionKey): Filter = {
67     val (x, y) = z2.invert(partition.value.toLong)
68     val (xmin, xmax) = (x - xRadius, x + xRadius)
69     val (ymin, ymax) = (y - yRadius, y + yRadius)
70     val bbox = ff.bbox(ff.property(attribute), new ReferencedEnvelope(xmin, xmax, ymin, ymax, CRS_EPSG_4326))
71     // account for borders between z-cells (make upper bounds exclusive except on the upper-right edge)
72     val xExclusive = if (xmax == z2.lon.max) { None } else {
73       Some(ff.less(ff.function("getX", ff.property(attribute)), ff.literal(xmax)))
74     }
75     val yExclusive = if (ymax == z2.lat.max) { None } else {
76       Some(ff.less(ff.function("getY", ff.property(attribute)), ff.literal(ymax)))
77     }
78     andFilters(Seq(bbox) ++ xExclusive ++ yExclusive)
79   }
80 
81   private def getIndexRanges(filter: Filter): Option[Seq[IndexRange]] = {
82     val geometries = FilterHelper.extractGeometries(filter, attribute, intersect = true)
83     if (geometries.isEmpty) {
84       None
85     } else if (geometries.disjoint) {
86       Some(Seq.empty)
87     } else {
88       Some(z2.ranges(geometries.values.map(GeometryUtils.bounds)))
89     }
90   }
91 
92   // number of digits required to print our partition values
93   private def digits(bits: Int): Int = math.ceil(bits * math.log10(2)).toInt
94 }
