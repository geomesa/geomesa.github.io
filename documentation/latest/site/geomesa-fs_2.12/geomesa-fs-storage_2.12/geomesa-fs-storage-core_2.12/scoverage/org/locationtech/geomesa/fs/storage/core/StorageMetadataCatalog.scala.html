44 object StorageMetadataCatalog {
45 
46   val MetadataTypeConfig = "fs.metadata.type"
47 
48   /**
49    * Create a new catalog instance
50    *
51    * @param context file system context
52    * @return
53    */
54   def apply(context: FileSystemContext): StorageMetadataCatalog = {
55     val metadataType = context.conf.get(MetadataTypeConfig).filterNot(_.isBlank).getOrElse {
56       throw new IllegalArgumentException(s"No $MetadataTypeConfig config provided")
57     }
58     metadataType.toLowerCase(Locale.US) match {
59       case FileBasedMetadata.MetadataType => new FileBasedMetadataCatalog(context)
60       case JdbcMetadata.MetadataType      => new JdbcMetadataCatalog(context)
61       case ConverterMetadata.MetadataType => new ConverterMetadataCatalog(context)
62       case t => throw new UnsupportedOperationException(s"Metadata implementation not found for type: $t")
63     }
64   }
