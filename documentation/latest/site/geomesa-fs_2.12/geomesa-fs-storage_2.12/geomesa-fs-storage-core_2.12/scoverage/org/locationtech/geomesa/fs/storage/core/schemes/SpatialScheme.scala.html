23 abstract class SpatialScheme(id: String, bits: Int, geom: String) extends PartitionScheme {
24 
25   require(bits % 2 == 0, "Resolution must be an even number")
26 
27   protected val format = s"%0${digits(bits)}d"
28 
29   override val name: String = s"$id:attribute=$geom:bits=$bits"
30 
31   protected def digits(bits: Int): Int
32 
33   protected def generateRanges(xy: Seq[(Double, Double, Double, Double)]): Seq[IndexRange]
34 
35   override def getRangesForFilter(filter: Filter): Option[Seq[PartitionRange]] = {
36     getGeoms(filter).map { ranges =>
37       val builder = new RangeBuilder()
38       ranges.foreach { range =>
39         val lower = format.format(range.lower)
40         val upper = format.format(range.upper + 1)
41         builder += PartitionRange(name, lower, upper)
42       }
43       builder.result()
44     }
45   }
46 
47   override def getPartitionsForFilter(filter: Filter): Option[Seq[PartitionKey]] = {
48     getGeoms(filter).orElse(Some(generateRanges(Seq((-180, -90, 180, 90))))).map { ranges =>
49       ranges.flatMap { range =>
50         val lower = range.lower
51         val steps = 1 + (range.upper - lower).toInt
52         Seq.tabulate(steps)(i => PartitionKey(name, format.format(lower + i)))
53       }
54     }
55   }
56 
57   private def getGeoms(filter: Filter): Option[Seq[IndexRange]] = {
58     val geometries = FilterHelper.extractGeometries(filter, geom, intersect = true)
59     if (geometries.isEmpty) {
60       None
61     } else if (geometries.disjoint) {
62       Some(Seq.empty)
63     } else {
64       Some(generateRanges(geometries.values.map(GeometryUtils.bounds)))
65     }
66   }
67 }

77     override def load(sft: SimpleFeatureType, scheme: String): Option[PartitionScheme] = {
78       val opts = SchemeOpts(scheme)
79       lazy val matcher = namePattern.matcher(scheme)
80 
81       def build(resolution: Short): Option[PartitionScheme] = {
82         val geom = opts.getSingle("attribute").orElse(Option(sft.getGeomField)).orNull
83         require(geom != null, s"Spatial schemes requires an attribute to be specified with 'attribute=<attribute>'")
84         val index = attributeIndex(sft, geom, Some(implicitly[ClassTag[T]].runtimeClass))
85         Some(buildPartitionScheme(resolution, geom, index))
86       }
87 
88       if (opts.name == this.name) {
89         val bits = opts.getSingle("bits").map(_.toShort).getOrElse {
90           throw new IllegalArgumentException(s"Spatial schemes requires a resolution to be specified with 'bits=<resolution>'")
91         }
92         build(bits)
93       } else if (matcher.matches()) {
94         build(matcher.group(1).toShort)
95       } else {
96         None
97       }
98     }
99 
100     def buildPartitionScheme(bits: Int, geom: String, geomIndex: Int): SpatialScheme
101   }
