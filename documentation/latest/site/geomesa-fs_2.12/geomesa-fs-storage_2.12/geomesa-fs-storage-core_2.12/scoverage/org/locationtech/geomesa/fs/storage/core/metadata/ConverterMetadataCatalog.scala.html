18 /**
19  * Synthetic metadata catalog that examines files on disk
20  *
21  * @param context file system context
22  */
23 class ConverterMetadataCatalog(context: FileSystemContext) extends StorageMetadataCatalog with LazyLogging {
24 
25   import ConverterMetadata._
26 
27   private val sft = {
28     val sftArg =
29       context.conf.get(SftConfigParam)
30         .orElse(context.conf.get(SftNameParam))
31         .getOrElse(throw new IllegalArgumentException(s"Must provide either simple feature type config or name"))
32     SftArgResolver.getArg(SftArgs(sftArg, null)) match {
33       case Left(e) => throw new IllegalArgumentException("Could not load SimpleFeatureType with provided parameters", e)
34       case Right(schema) => schema
35     }
36   }
37 
38   private val schemes = {
39     val partitionSchemeName =
40       context.conf.getOrElse(PartitionSchemeParam, throw new IllegalArgumentException("Must provide partition scheme name"))
41     val partitionSchemeOpts = {
42       val opts =
43         context.conf.collect { case (k, v) if k.startsWith(PartitionOptsPrefix) => s"${k.substring(PartitionOptsPrefix.length)}=$v" }
44       opts.mkString(":")
45     }
46     partitionSchemeName.split(",").map { n =>
47       val nameWithOpts = s"$n:$partitionSchemeOpts"
48       // expose hierarchical/receipt time schemes (only used by the converter metadata)
49       HierarchicalDateTimeScheme.load(sft, nameWithOpts)
50         .orElse(ReceiptTimeScheme.load(sft, nameWithOpts))
51         .getOrElse(PartitionSchemeFactory.load(sft, nameWithOpts))
52     }
53   }
54 
55   private val converterPath =
56     context.conf.get(ConverterPathParam)
57       .map(p => context.root.resolve(p))
58       .getOrElse(throw new IllegalArgumentException("Must provide converter path"))
59 
60   private val leafStorage =
61     context.conf.get(LeafStorageParam)
62       .map(_.toBoolean)
63       .getOrElse {
64         val deprecated = s"${PartitionOptsPrefix}leaf-storage"
65         context.conf.get(deprecated).fold(true) { s =>
66           logger.warn(s"Using deprecated leaf-storage partition-scheme option. Please define leaf-storage using '$LeafStorageParam'")
67           s.toBoolean
68         }
69       }
70 
71   override def getTypeNames: Seq[String] = Seq(sft.getTypeName)
72 
73   override def load(typeName: String): StorageMetadata = {
74     if (typeName == sft.getTypeName) {
75       new ConverterMetadata(FileSystemContext.create(converterPath, context.conf, context.namespace), sft, schemes, leafStorage)
76     } else {
77       throw new IllegalArgumentException(s"Schema '$typeName' doesn't exist - available schemas: ${sft.getTypeName}")
78     }
79   }
80 
81   override def create(sft: SimpleFeatureType, partitions: Seq[String], targetFileSize: Option[Long]): StorageMetadata =
82     throw new UnsupportedOperationException("Converter storage is read only")
83 }
