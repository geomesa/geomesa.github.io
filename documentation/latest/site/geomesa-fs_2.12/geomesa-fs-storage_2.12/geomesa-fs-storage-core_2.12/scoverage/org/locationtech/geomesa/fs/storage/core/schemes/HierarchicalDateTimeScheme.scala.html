49   private val truncateToPartitionStart: ZonedDateTime => ZonedDateTime = {
50     // note: `truncatedTo` is only valid up to DAYS, other units require additional steps
51     val truncate: ZonedDateTime => ZonedDateTime = stepUnit match {
52       case NANOS | MICROS | MILLIS | SECONDS | MINUTES | HOURS | DAYS =>
53         dt => dt.truncatedTo(stepUnit)
54 
55       case WEEKS =>
56         val adjuster = TemporalAdjusters.previousOrSame(DayOfWeek.MONDAY) // note: java day_of_week defines monday as the first day
57         dt => dt.`with`(adjuster).truncatedTo(DAYS)
58 
59       case MONTHS =>
60         val adjuster = TemporalAdjusters.firstDayOfMonth()
61         dt => dt.`with`(adjuster).truncatedTo(DAYS)
62 
63       case YEARS =>
64         val adjuster = TemporalAdjusters.firstDayOfYear()
65         dt => dt.`with`(adjuster).truncatedTo(DAYS)
66 
67       case _ =>
68         throw new IllegalArgumentException(s"${HierarchicalDateTimeScheme.Config.StepUnitOpt} $stepUnit is not supported")
69     }
70     if (step == 1) {
71       truncate
72     } else {
73       val chronoField = stepUnit match {
74         case NANOS   => ChronoField.NANO_OF_SECOND
75         case MICROS  => ChronoField.MICRO_OF_SECOND
76         case MILLIS  => ChronoField.MILLI_OF_SECOND
77         case SECONDS => ChronoField.SECOND_OF_MINUTE
78         case MINUTES => ChronoField.MINUTE_OF_HOUR
79         case HOURS   => ChronoField.HOUR_OF_DAY
80         case DAYS    => ChronoField.DAY_OF_YEAR
81         case WEEKS   => WeekFields.ISO.weekOfWeekBasedYear()
82         case MONTHS  => ChronoField.MONTH_OF_YEAR
83         case YEARS   => ChronoField.YEAR
84         case _ => throw new IllegalArgumentException(s"${HierarchicalDateTimeScheme.Config.StepUnitOpt} $stepUnit is not supported")
85       }
86       val min = math.max(chronoField.range().getMinimum, 0).toInt // account for 1-based fields like {month, week, day}_of_year
87       truncate.andThen { dt =>
88         val steps = dt.get(chronoField) - min
89         val remainder = steps % step
90         if (remainder == 0) { dt } else { dt.`with`(chronoField, steps + min - remainder ) }
91       }
92     }
93   }

109   override def getRangesForFilter(filter: Filter): Option[Seq[PartitionRange]] = {
110     getBounds(filter).map { bounds =>
111       val builder = new RangeBuilder()
112       bounds.foreach { bound =>
113         val lower = bound.lower.value.fold("")(v => formatter.format(truncateToPartitionStart(v)))
114         val upper = bound.upper.value.fold(HierarchicalDateTimeScheme.UnboundedUpper) { v =>
115           val partition = formatter.format(truncateToPartitionStart(v))
116           if (bound.upper.exclusive && formatter.format(truncateToPartitionStart(v.minus(1, ChronoUnit.MILLIS))) < partition) {
117             partition
118           } else {
119             formatter.format(truncateToPartitionStart(v.plus(step, stepUnit)))
120           }
121         }
122         builder += PartitionRange(name, lower, upper)
123       }
124       builder.result()
125     }
126   }
127 
128   override def getPartitionsForFilter(filter: Filter): Option[Seq[PartitionKey]] = {
129     getBounds(filter).map { bounds =>
130       bounds.flatMap { bound =>
131         if (!bound.isBoundedBothSides) {
132           throw new IllegalArgumentException(s"Can't enumerate an unbounded filter: ${ECQL.toCQL(filter)}")
133         }
134         val lower = bound.lower.value.get
135         // `stepUnit.between` claims to be upper endpoint exclusive, but doesn't seem to be...
136         val steps = stepUnit.between(lower, bound.upper.value.get).toInt / step
137         Seq.tabulate(steps)(i => PartitionKey(name, formatter.format(lower.plus(step * i, stepUnit))))
138       }
139     }
140   }

142   override def getCoveringFilter(partition: PartitionKey): Filter = {
143     val zdt = DateParsing.parse(partition.value, formatter)
144     val start = DateParsing.format(zdt)
145     val end = DateParsing.format(zdt.plus(step, stepUnit))
146     ff.and(ff.greaterOrEqual(ff.property(attribute), ff.literal(start)), ff.less(ff.property(attribute), ff.literal(end)))
147   }
148 
149   private def getBounds(filter: Filter): Option[Seq[Bounds[ZonedDateTime]]] = {
150     val bounds = FilterHelper.extractIntervals(filter, attribute)
151     if (bounds.isEmpty) {
152       None
153     } else if (bounds.disjoint) {
154       Some(Seq.empty)
155     } else {
156       Some(bounds.values)
157     }
158   }

205           .toFormatter()
206       Format("weekly", formatter, "YYYY/'W'ww", ChronoUnit.WEEKS)

212   override def load(sft: SimpleFeatureType, scheme: String): Option[HierarchicalDateTimeScheme] = {
213     import org.locationtech.geomesa.utils.geotools.RichSimpleFeatureType.RichSimpleFeatureType
214 
215     val opts = SchemeOpts(scheme)
216 
217     lazy val step = opts.opts.get(Config.StepOpt).map(_.toInt).getOrElse(1)
218     lazy val dtg = opts.opts.get(Config.DtgAttribute).orElse(sft.getDtgField).getOrElse {
219       throw new IllegalArgumentException(s"DateTime scheme requires valid attribute '${Config.DtgAttribute}'")
220     }
221     lazy val dtgIndex = attributeIndex(sft, dtg, Some(classOf[Date]))
222 
223     if (opts.name == Name) {
224       val unit = opts.opts.get(Config.StepUnitOpt).map(c => ChronoUnit.valueOf(c.toUpperCase)).getOrElse {
225         throw new IllegalArgumentException(s"DateTime scheme requires valid unit '${Config.StepUnitOpt}'")
226       }
227       val format = opts.opts.getOrElse(Config.DateTimeFormatOpt,
228         throw new IllegalArgumentException(s"DateTime scheme requires valid format '${Config.DateTimeFormatOpt}'"))
229       require(!format.endsWith("/"), "Format cannot end with a slash")
230       val formatter = try { DateTimeFormatter.ofPattern(format) } catch {
231         case NonFatal(e) => throw new IllegalArgumentException(s"Invalid date format '$format':", e)
232       }
233       Some(HierarchicalDateTimeScheme(formatter, format, unit, step, dtg, dtgIndex))
234     } else {
235       Formats(opts.name).map(f => HierarchicalDateTimeScheme(f.formatter, f.pattern, f.unit, step, dtg, dtgIndex))
236     }
237   }
