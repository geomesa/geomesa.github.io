31 /**
32  * StorageMetadata implementation that stores all file information in a single JSON file.
33  * Uses HDFS lock files for cross-JVM write locking and synchronization for intra-JVM locking.
34  *
35  * @param fs file system
36  * @param directory metadata root path
37  * @param meta basic metadata config
38  */
39 class FileBasedMetadata(fs: ObjectStore, meta: Metadata, directory: URI)
40     extends StorageMetadata with CachedMetadata with LazyLogging {
41 
42   import FileBasedMetadata._
43 
44   import scala.collection.JavaConverters._
45 
46   require(directory.toString.endsWith("/"), "Invalid path - must end with '/'")
47 
48   private val kvs = new ConcurrentHashMap[String, String](meta.config.asJava)
49   private val encodedTypeName = StringSerialization.alphaNumericSafeString(meta.sft.getTypeName)
50   private val metadataFilePath = directory.resolve(s"$encodedTypeName.json")
51   private val filesFilePath = directory.resolve(s".${encodedTypeName}_files.json")
52   private val lockFilePath = directory.resolve(s".${encodedTypeName}_files.json.lock")
53 
54   override val `type`: String = FileBasedMetadata.MetadataType
55 
56   override val sft: SimpleFeatureType = meta.sft
57   override val schemes: Set[PartitionScheme] = meta.partitions.map(PartitionSchemeFactory.load(sft, _)).toSet
58 
59   filesCache.refresh(BoxedUnit.UNIT) // kick off the initial load asynchronously
60 
61   override def get(key: String): Option[String] = Option(kvs.get(key))
62 
63   // note: this isn't synchronized across jvms
64   override def set(key: String, value: String): Unit = FileBasedMetadata.synchronized {
65     kvs.put(key, value)
66     WithClose(fs.overwrite(metadataFilePath)) { out =>
67       MetadataSerialization.serialize(out, meta.copy(config = kvs.asScala.toMap))
68     }
69   }
70 
71   override def addFile(file: StorageFile): Unit = {
72     modifyFiles { files =>
73       // remove any existing file with the same path and add the new one
74       (files.filterNot(_.file == file.file) :+ file).sortBy(_.timestamp)(Ordering.Long.reverse)
75     }
76   }
77 
78   override def removeFile(file: StorageFile): Unit = {
79     modifyFiles { files =>
80       files.filterNot(_.file == file.file)
81     }
82   }
83 
84   override def replaceFiles(existing: Seq[StorageFile], replacements: Seq[StorageFile]): Unit = {
85     val existingFiles = existing.map(_.file)
86     modifyFiles { files =>
87       files.filterNot(f => existingFiles.contains(f.file)) ++ replacements
88     }
89   }
90 
91   override def close(): Unit = try { super.close() } finally { fs.close() }
92 
93   /**
94    * Load files from the JSON file
95    */
96   override protected def buildFileList(): Seq[StorageFile] = {
97     try {
98       WithClose(fs.read(filesFilePath)) { opt =>
99         opt.fold(Seq.empty[StorageFile]) { is =>
100           val listType = new TypeToken[java.util.List[StorageFile]]() {}.getType
101           gson.fromJson[java.util.List[StorageFile]](new InputStreamReader(is, StandardCharsets.UTF_8), listType).asScala.toSeq
102         }
103       }
104     } catch {
105       case NonFatal(e) =>
106         logger.warn(s"Error loading files from $filesFilePath", e)
107         Seq.empty
108     }
109   }
110 
111   /**
112    * Modify files with proper locking using HDFS lock files
113    */
114   private def modifyFiles(fn: Seq[StorageFile] => Seq[StorageFile]): Unit = FileBasedMetadata.synchronized {
115     // acquire lock by creating a lock file atomically
116     var lockAcquired = false
117     var retries = 0
118 
119     val maxAttempts = MaxLockRetries.toInt.get
120 
121     while (!lockAcquired && retries < maxAttempts) {
122       try {
123         // try to create lock file with overwrite=false for atomicity
124         WithClose(fs.create(lockFilePath)) {
125           case None =>
126             // lock file exists, check if it's stale
127             retries += 1
128             if (isLockStale(lockFilePath)) {
129               // remove stale lock and retry
130               try {
131                 fs.delete(lockFilePath)
132               } catch {
133                 case NonFatal(_) => // ignore, will retry
134               }
135             } else {
136               // wait and retry
137               Thread.sleep(LockRetryDelay.toMillis.get)
138             }
139           case Some(out) =>
140             // write lock info for debugging - hostname + timestamp
141             val lockInfo = s"${java.net.InetAddress.getLocalHost.getHostName}:${System.currentTimeMillis()}"
142             out.write(lockInfo.getBytes(StandardCharsets.UTF_8))
143         }
144         lockAcquired = true
145       } catch {
146         case NonFatal(e) => throw new RuntimeException(s"Failed to acquire lock at $lockFilePath", e)
147       }
148     }
149 
150     if (!lockAcquired) {
151       throw new RuntimeException(s"Failed to acquire lock after $MaxLockRetries retries")
152     }
153 
154     try {
155       // reload from disk to get latest state
156       val currentFiles = buildFileList()
157 
158       // apply the modification
159       val updatedFiles = fn(currentFiles)
160 
161       // write back to disk
162       WithClose(new OutputStreamWriter(fs.overwrite(filesFilePath), StandardCharsets.UTF_8)) { writer =>
163         gson.toJson(updatedFiles.asJava, writer)
164       }
165 
166       // update cache
167       filesCache.put(BoxedUnit.UNIT, updatedFiles)
168     } finally {
169       // release lock by deleting lock file
170       try {
171         fs.delete(lockFilePath)
172       } catch {
173         case NonFatal(e) => logger.warn(s"Failed to release lock at $lockFilePath", e)
174       }
175     }
176   }
177 
178   /**
179    * Check if a lock file is stale (older than lock timeout)
180    */
181   private def isLockStale(lockPath: URI): Boolean = {
182     try {
183       fs.modified(lockPath).forall { modified =>
184         val age = System.currentTimeMillis() - modified
185         age > LockTimeout.toMillis.get
186       }
187     } catch {
188       case NonFatal(_) => true // if we can't read it, consider it stale
189     }
190   }
191 }

193 object FileBasedMetadata {
194 
195   val MetadataType = "file"
196 
197   // locking parameters
198   val MaxLockRetries = SystemProperty("geomesa.fs.metadata.file.lock.retries", "60") // maximum number of lock acquisition retries
199   val LockRetryDelay = SystemProperty("geomesa.fs.metadata.file.lock.delay", "1 second") // delay between retries
200   val LockTimeout = SystemProperty("geomesa.fs.metadata.file.lock.timeout", "1 minute") // lock timeout
201 
202   // gson instance with custom serializers for StorageFile
203   private val gson: Gson =
204     new GsonBuilder()
205       .registerTypeAdapter(classOf[PartitionKey], PartitionKey.PartitionKeySerializer)
206       .registerTypeAdapter(classOf[Partition], Partition.PartitionSerializer)
207       .registerTypeAdapter(classOf[StorageFile], StorageFileSerializer)
208       .disableHtmlEscaping()
209       .create()

211   /**
212    * Json serializer for StorageFileAction
213    */
214   private object StorageFileSerializer extends JsonSerializer[StorageFile] with JsonDeserializer[StorageFile] {
215 
216     import scala.collection.JavaConverters._
217 
218     override def serialize(src: StorageFile, typeOfSrc: Type, context: JsonSerializationContext): JsonElement = {
219       val obj = new JsonObject()
220       obj.addProperty("file", src.file)
221       obj.add("partition", context.serialize(src.partition))
222       obj.addProperty("count", src.count)
223       obj.addProperty("action", src.action.toString)
224       val spatialBounds = new JsonArray(src.spatialBounds.size)
225       src.spatialBounds.foreach(b => spatialBounds.add(context.serialize(b)))
226       obj.add("spatialBounds", spatialBounds)
227       val attributeBounds = new JsonArray(src.attributeBounds.size)
228       src.attributeBounds.foreach(b => attributeBounds.add(context.serialize(b)))
229       obj.add("attributeBounds", attributeBounds)
230       val sort = new JsonArray(src.sort.size)
231       src.sort.foreach(sort.add(_))
232       obj.add("sort", sort)
233       obj.addProperty("timestamp", src.timestamp)
234       obj
235     }
236 
237     override def deserialize(json: JsonElement, typeOfT: Type, context: JsonDeserializationContext): StorageFile = {
238       val obj = json.getAsJsonObject
239       val spatialBounds =
240         obj.getAsJsonArray("spatialBounds").asList().asScala.map(context.deserialize[SpatialBounds](_, classOf[SpatialBounds])).toSeq
241       val attributeBounds =
242         obj.getAsJsonArray("attributeBounds").asList().asScala.map(context.deserialize[AttributeBounds](_, classOf[AttributeBounds])).toSeq
243       val sort = obj.getAsJsonArray("sort").asList().asScala.map(_.getAsInt).toSeq
244       StorageFile(
245         obj.getAsJsonPrimitive("file").getAsString,
246         context.deserialize(obj.get("partition"), classOf[Partition]),
247         obj.getAsJsonPrimitive("count").getAsLong,
248         StorageFileAction.withName(obj.getAsJsonPrimitive("action").getAsString),
249         spatialBounds,
250         attributeBounds,
251         sort,
252         obj.getAsJsonPrimitive("timestamp").getAsLong,
253       )
254     }
255   }
