31 /**
32  * StorageMetadata implementation that stores all file information in a single JSON file.
33  * Uses HDFS lock files for cross-JVM write locking and synchronization for intra-JVM locking.
34  *
35  * @param fs file system
36  * @param directory metadata root path
37  * @param meta basic metadata config
38  */
39 class FileBasedMetadata(fs: ObjectStore, meta: Metadata, directory: URI)
40     extends StorageMetadata with CachedMetadata with LazyLogging {
41 
42   import FileBasedMetadata._
43 
44   import scala.collection.JavaConverters._
45 
46   require(directory.toString.endsWith("/"), "Invalid path - must end with '/'")
47 
48   private val kvs = new ConcurrentHashMap[String, String](meta.config.asJava)
49   private val encodedTypeName = StringSerialization.alphaNumericSafeString(meta.sft.getTypeName)
50   private val metadataFilePath = directory.resolve(s"$encodedTypeName.json")
51   private val filesFilePath = directory.resolve(s".${encodedTypeName}_files.json")
52   private val lockFilePath = directory.resolve(s".${encodedTypeName}_files.json.lock")
53 
54   override val `type`: String = FileBasedMetadata.MetadataType
55 
56   override val sft: SimpleFeatureType = meta.sft
57   override val schemes: Set[PartitionScheme] = meta.partitions.map(PartitionSchemeFactory.load(sft, _)).toSet
58 
59   filesCache.refresh(BoxedUnit.UNIT) // kick off the initial load asynchronously
60 
61   override def get(key: String): Option[String] = Option(kvs.get(key))
62 
63   // note: this isn't synchronized across jvms
64   override def set(key: String, value: String): Unit = FileBasedMetadata.synchronized {
65     if (value == null) {
66       kvs.remove(key)
67     } else {
68       kvs.put(key, value)
69     }
70     WithClose(fs.overwrite(metadataFilePath)) { out =>
71       MetadataSerialization.serialize(out, meta.copy(config = kvs.asScala.toMap))
72     }
73   }
74 
75   override def addFile(file: StorageFile): Unit = {
76     modifyFiles { files =>
77       // remove any existing file with the same path and add the new one
78       (files.filterNot(_.file == file.file) :+ file).sortBy(_.timestamp)(Ordering.Long.reverse)
79     }
80   }
81 
82   override def removeFile(file: StorageFile): Unit = {
83     modifyFiles { files =>
84       files.filterNot(_.file == file.file)
85     }
86   }
87 
88   override def replaceFiles(existing: Seq[StorageFile], replacements: Seq[StorageFile]): Unit = {
89     val existingFiles = existing.map(_.file)
90     modifyFiles { files =>
91       files.filterNot(f => existingFiles.contains(f.file)) ++ replacements
92     }
93   }
94 
95   override def close(): Unit = try { super.close() } finally { fs.close() }
96 
97   /**
98    * Load files from the JSON file
99    */
100   override protected def buildFileList(): Seq[StorageFile] = {
101     try {
102       WithClose(fs.read(filesFilePath)) { opt =>
103         opt.fold(Seq.empty[StorageFile]) { is =>
104           val listType = new TypeToken[java.util.List[StorageFile]]() {}.getType
105           gson.fromJson[java.util.List[StorageFile]](new InputStreamReader(is, StandardCharsets.UTF_8), listType).asScala.toSeq
106         }
107       }
108     } catch {
109       case NonFatal(e) =>
110         logger.warn(s"Error loading files from $filesFilePath", e)
111         Seq.empty
112     }
113   }
114 
115   /**
116    * Modify files with proper locking using HDFS lock files
117    */
118   private def modifyFiles(fn: Seq[StorageFile] => Seq[StorageFile]): Unit = FileBasedMetadata.synchronized {
119     // acquire lock by creating a lock file atomically
120     var lockAcquired = false
121     var retries = 0
122 
123     val maxAttempts = MaxLockRetries.toInt.get
124 
125     while (!lockAcquired && retries < maxAttempts) {
126       try {
127         // try to create lock file with overwrite=false for atomicity
128         try {
129           fs.create(lockFilePath).foreach { out =>
130             try {
131               // write lock info for debugging - hostname + timestamp
132               val lockInfo = s"${java.net.InetAddress.getLocalHost.getHostName}:${System.currentTimeMillis()}"
133               out.write(lockInfo.getBytes(StandardCharsets.UTF_8))
134             } finally {
135               out.close()
136             }
137             lockAcquired = true
138           }
139         } catch {
140           case NonFatal(e) => logger.debug("Error writing lock file, may already exist?", e)
141         }
142 
143         if (!lockAcquired) {
144           // lock file exists, check if it's stale
145           retries += 1
146           fs.modified(lockFilePath).foreach { modified =>
147             val age = System.currentTimeMillis() - modified
148             if (age > LockTimeout.toMillis.get) {
149               fs.delete(lockFilePath)
150             }
151           }
152           // wait and retry
153           Thread.sleep(LockRetryDelay.toMillis.get)
154         }
155       } catch {
156         case NonFatal(e) => throw new RuntimeException(s"Failed to acquire lock at $lockFilePath", e)
157       }
158     }
159 
160     if (!lockAcquired) {
161       throw new RuntimeException(s"Failed to acquire lock after $MaxLockRetries retries")
162     }
163 
164     try {
165       // reload from disk to get latest state
166       val currentFiles = buildFileList()
167 
168       // apply the modification
169       val updatedFiles = fn(currentFiles)
170 
171       // write back to disk
172       WithClose(new OutputStreamWriter(fs.overwrite(filesFilePath), StandardCharsets.UTF_8)) { writer =>
173         gson.toJson(updatedFiles.asJava, writer)
174       }
175 
176       // update cache
177       filesCache.put(BoxedUnit.UNIT, updatedFiles)
178     } finally {
179       // release lock by deleting lock file
180       try {
181         fs.delete(lockFilePath)
182       } catch {
183         case NonFatal(e) => logger.warn(s"Failed to release lock at $lockFilePath", e)
184       }
185     }
186   }
187 }

189 object FileBasedMetadata {
190 
191   val MetadataType = "file"
192 
193   // locking parameters
194   val MaxLockRetries = SystemProperty("geomesa.fs.metadata.file.lock.retries", "60") // maximum number of lock acquisition retries
195   val LockRetryDelay = SystemProperty("geomesa.fs.metadata.file.lock.delay", "1 second") // delay between retries
196   val LockTimeout = SystemProperty("geomesa.fs.metadata.file.lock.timeout", "1 minute") // lock timeout
197 
198   // gson instance with custom serializers for StorageFile
199   private val gson: Gson =
200     new GsonBuilder()
201       .registerTypeAdapter(classOf[PartitionKey], PartitionKey.PartitionKeySerializer)
202       .registerTypeAdapter(classOf[Partition], Partition.PartitionSerializer)
203       .registerTypeAdapter(classOf[StorageFile], StorageFileSerializer)
204       .disableHtmlEscaping()
205       .create()
206 
207   /**
208    * Json serializer for StorageFileAction
209    */
210   private object StorageFileSerializer extends JsonSerializer[StorageFile] with JsonDeserializer[StorageFile] {
211 
212     import scala.collection.JavaConverters._
213 
214     override def serialize(src: StorageFile, typeOfSrc: Type, context: JsonSerializationContext): JsonElement = {
215       val obj = new JsonObject()
216       obj.addProperty("file", src.file)
217       obj.add("partition", context.serialize(src.partition))
218       obj.addProperty("count", src.count)
219       obj.addProperty("action", src.action.toString)
220       val bounds = new JsonArray(src.bounds.size)
221       src.bounds.foreach(b => bounds.add(context.serialize(b)))
222       obj.add("bounds", bounds)
223       val sort = new JsonArray(src.sort.size)
224       src.sort.foreach(sort.add(_))
225       obj.add("sort", sort)
226       obj.addProperty("timestamp", src.timestamp)
227       obj
228     }
229 
230     override def deserialize(json: JsonElement, typeOfT: Type, context: JsonDeserializationContext): StorageFile = {
231       val obj = json.getAsJsonObject
232       StorageFile(
233         obj.getAsJsonPrimitive("file").getAsString,
234         context.deserialize(obj.get("partition"), classOf[Partition]),
235         obj.getAsJsonPrimitive("count").getAsLong,
236         StorageFileAction.withName(obj.getAsJsonPrimitive("action").getAsString),
237         obj.getAsJsonArray("bounds").asList().asScala.map(context.deserialize[ColumnBounds](_, classOf[ColumnBounds])).toSeq,
238         obj.getAsJsonArray("sort").asList().asScala.map(_.getAsInt).toSeq,
239         obj.getAsJsonPrimitive("timestamp").getAsLong,
240       )
241     }
242   }
