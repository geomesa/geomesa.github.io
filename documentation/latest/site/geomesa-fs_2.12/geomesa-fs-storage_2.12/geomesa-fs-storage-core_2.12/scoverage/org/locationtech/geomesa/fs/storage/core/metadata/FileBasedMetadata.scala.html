30 /**
31  * StorageMetadata implementation that stores all file information in a single JSON file.
32  * Uses HDFS lock files for cross-JVM write locking and synchronization for intra-JVM locking.
33  *
34  * @param fs file system
35  * @param directory metadata root path
36  * @param meta basic metadata config
37  */
38 class FileBasedMetadata(fs: ObjectStore, meta: Metadata, directory: URI)
39     extends StorageMetadata with CachedMetadata with LazyLogging {
40 
41   import FileBasedMetadata._
42 
43   import scala.collection.JavaConverters._
44 
45   require(directory.toString.endsWith("/"), "Invalid path - must end with '/'")
46 
47   private val kvs = new ConcurrentHashMap[String, String](meta.config.asJava)
48   private val encodedTypeName = StringSerialization.alphaNumericSafeString(meta.sft.getTypeName)
49   private val metadataFilePath = directory.resolve(s"$encodedTypeName.json")
50   private val filesFilePath = directory.resolve(s".${encodedTypeName}_files.json")
51   private val lockFilePath = directory.resolve(s".${encodedTypeName}_files.json.lock")
52 
53   override val `type`: String = FileBasedMetadata.MetadataType
54 
55   override val sft: SimpleFeatureType = meta.sft
56   override val schemes: Set[PartitionScheme] = meta.partitions.map(PartitionSchemeFactory.load(sft, _)).toSet
57 
58   filesCache.refresh(BoxedUnit.UNIT) // kick off the initial load asynchronously
59 
60   override def get(key: String): Option[String] = Option(kvs.get(key))
61 
62   // note: this isn't synchronized across jvms
63   override def set(key: String, value: String): Unit = FileBasedMetadata.synchronized {
64     if (value == null) {
65       kvs.remove(key)
66     } else {
67       kvs.put(key, value)
68     }
69     WithClose(fs.overwrite(metadataFilePath)) { out =>
70       MetadataSerialization.serialize(out, meta.copy(config = kvs.asScala.toMap))
71     }
72   }
73 
74   override def addFile(file: StorageFile): Unit = {
75     modifyFiles { files =>
76       // remove any existing file with the same path and add the new one
77       (files.filterNot(_.file == file.file) :+ file).sortBy(_.timestamp)(Ordering.Long.reverse)
78     }
79     logger.debug(s"Added file $file")
80   }
81 
82   override def removeFile(file: StorageFile): Unit = {
83     modifyFiles { files =>
84       files.filterNot(_.file == file.file)
85     }
86     logger.debug(s"Removed file $file")
87   }
88 
89   override def replaceFiles(existing: Seq[StorageFile], replacements: Seq[StorageFile]): Unit = {
90     val existingFiles = existing.map(_.file)
91     modifyFiles { files =>
92       files.filterNot(f => existingFiles.contains(f.file)) ++ replacements
93     }
94     logger.debug(s"Replaced ${existing.size} files with ${replacements.size} new ones")
95   }
96 
97   override def close(): Unit = try { super.close() } finally { fs.close() }
98 
99   /**
100    * Load files from the JSON file
101    */
102   override protected def buildFileList(): Seq[StorageFile] = {
103     WithClose(fs.read(filesFilePath)) { opt =>
104       opt.fold(Seq.empty[StorageFile]) { is =>
105         val listType = new TypeToken[java.util.List[StorageFile]]() {}.getType
106         gson.fromJson[java.util.List[StorageFile]](new InputStreamReader(is, StandardCharsets.UTF_8), listType).asScala.toSeq
107       }
108     }
109   }
110 
111   /**
112    * Modify files with proper locking using HDFS lock files
113    */
114   private def modifyFiles(fn: Seq[StorageFile] => Seq[StorageFile]): Unit = FileBasedMetadata.synchronized {
115     // acquire lock by creating a lock file atomically
116     var lockAcquired = false
117     var retries = 0
118 
119     val maxAttempts = MaxLockRetries.toInt.get
120 
121     while (!lockAcquired && retries < maxAttempts) {
122       logger.trace(s"Attempting to acquire lock at $lockFilePath with ${maxAttempts - retries} tries left")
123       try {
124         // try to create lock file with overwrite=false for atomicity
125         try {
126           fs.create(lockFilePath).foreach { out =>
127             try {
128               // write lock info for debugging - hostname + timestamp
129               val lockInfo = s"${java.net.InetAddress.getLocalHost.getHostName}:${System.currentTimeMillis()}"
130               out.write(lockInfo.getBytes(StandardCharsets.UTF_8))
131             } finally {
132               out.close()
133             }
134             lockAcquired = true
135           }
136         } catch {
137           case NonFatal(e) => logger.debug("Error writing lock file, may already exist?", e)
138         }
139 
140         if (!lockAcquired) {
141           retries += 1
142           if (retries < maxAttempts) {
143             // check if lockfile is stale
144             fs.modified(lockFilePath).foreach { modified =>
145               val age = System.currentTimeMillis() - modified
146               if (age > LockTimeout.toMillis.get) {
147                 logger.debug(s"Deleting expired lock file (age ${age}ms at $lockFilePath")
148                 fs.delete(lockFilePath)
149               }
150             }
151             // wait and retry
152             logger.debug(s"Could not acquire lock - waiting for ${LockRetryDelay.toMillis.get}ms before next attempt")
153             Thread.sleep(LockRetryDelay.toMillis.get)
154           }
155         }
156       } catch {
157         case NonFatal(e) => throw new RuntimeException(s"Failed to acquire lock at $lockFilePath", e)
158       }
159     }
160 
161     if (!lockAcquired) {
162       throw new RuntimeException(s"Failed to acquire lock after $MaxLockRetries retries")
163     }
164     logger.debug(s"Acquired lock file at $lockFilePath")
165 
166     try {
167       // reload from disk to get latest state
168       val currentFiles = buildFileList()
169 
170       // apply the modification
171       val updatedFiles = fn(currentFiles)
172 
173       // write back to disk
174       WithClose(new OutputStreamWriter(fs.overwrite(filesFilePath), StandardCharsets.UTF_8)) { writer =>
175         gson.toJson(updatedFiles.asJava, writer)
176       }
177 
178       // update cache
179       filesCache.put(BoxedUnit.UNIT, updatedFiles)
180     } finally {
181       // release lock by deleting lock file
182       try {
183         fs.delete(lockFilePath)
184         logger.debug(s"Released lock file at $lockFilePath")
185       } catch {
186         case NonFatal(e) => logger.warn(s"Failed to release lock at $lockFilePath", e)
187       }
188     }
189   }
190 }
