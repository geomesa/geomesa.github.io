212    * Encoder for points
213    */
214   object Z2Encoder extends TypeEncoder[Point, String] {
215 
216     private val sfc = Z2SFC
217     private val factory = new GeometryFactory()
218     private val hexFormat = HexFormat.of()
219 
220     override val getAlias: String = "z2"
221 
222     override def resolves(): Class[Point] = classOf[Point]
223 
224     override def encode(value: Point): String = {
225       if (value == null) {
226         throw new NullPointerException("Null values are not allowed")
227       }
228       toHex(sfc.index(value.getX, value.getY))
229     }
230 
231     override def decode(value: String): Point = {
232       val (x, y) = sfc.invert(fromHex(value))
233       factory.createPoint(new Coordinate(x, y))
234     }
235 
236     /**
237      * Creates a z2 prefix based on a partition key
238      *
239      * @param value z2 partition value
240      * @param bits number of bits used in the z2 partition scheme
241      * @return
242      */
243     def encodePartition(value: String, bits: Int): String = hexFormat.toHexDigits(value.toLong, bits / 4)
244 
245     /**
246      * Calculate encoded ranges
247      *
248      * @param queries a sequence of OR'd windows to cover. Each window is in the form (xmin, ymin, xmax, ymax)
249      * @param maxRanges rough upper bound on the number of ranges to return
250      */
251     def ranges(queries: Seq[(Double, Double, Double, Double)], maxRanges: Option[Int] = None): Seq[(String, String)] =
252       sfc.ranges(queries, maxRanges = maxRanges).map(r => toHex(r.lower) -> toHex(r.upper))
253 
254     // since the first two bits are not used in our z values, drop them so that truncating the hex value aligns with our curve
255     private def toHex(z: Long): String = hexFormat.toHexDigits(z << 2)
256     private def fromHex(hex: String): Long = HexFormat.fromHexDigitsToLong(hex) >>> 2
257   }

259   /**
260    * Encoder for points
261    */
262   object XZ2Encoder extends TypeEncoder[Geometry, String] {
263 
264     private val sfc = XZ2SFC
265     private val factory = new GeometryFactory()
266 
267     override val getAlias: String = "xz2"
268 
269     override def resolves(): Class[Geometry] = classOf[Geometry]
270 
271     override def encode(value: Geometry): String = {
272       if (value == null) {
273         throw new NullPointerException("Null values are not allowed")
274       }
275       val env = value.getEnvelopeInternal
276       if (env.isNull) {
277         throw new NullPointerException("Geometry has a null envelope")
278       }
279       sfc.hexEncode(sfc.index(env.getMinX, env.getMinY, env.getMaxX, env.getMaxY))
280     }
281 
282     override def decode(value: String): Geometry = {
283       val (xmin, ymin, xmax, ymax) = sfc.invert(sfc.hexDecode(value))
284       val ring = Array(
285         new Coordinate(xmin, ymin),
286         new Coordinate(xmin, ymax),
287         new Coordinate(xmax, ymax),
288         new Coordinate(xmax, ymin),
289         new Coordinate(xmin, ymin)
290       )
291       factory.createPolygon(ring)
292     }
293 
294     /**
295      * Calculate encoded ranges
296      *
297      * @param queries a sequence of OR'd windows to cover. Each window is in the form (xmin, ymin, xmax, ymax)
298      * @param maxRanges a rough upper limit on the number of ranges to generate
299      */
300     def ranges(queries: Seq[(Double, Double, Double, Double)], maxRanges: Option[Int] = None): Seq[(String, String)] =
301       sfc.ranges(queries, maxRanges).map(r => sfc.hexEncode(r.lower) -> sfc.hexEncode(r.upper))
302   }
