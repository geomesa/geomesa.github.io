22 /**

169   /**
170    * Encoder for points
171    */
172   object Z2Encoder extends TypeEncoder[Point, String] {
173 
174     private val sfc = Z2SFC
175     private val factory = new GeometryFactory()
176     private val hexFormat = HexFormat.of()
177 
178     override val getAlias: String = "z2"
179 
180     override def resolves(): Class[Point] = classOf[Point]
181 
182     override def encode(value: Point): String = {
183       if (value == null) {
184         throw new NullPointerException("Null values are not allowed")
185       }
186       toHex(sfc.index(value.getX, value.getY))
187     }
188 
189     override def decode(value: String): Point = {
190       val (x, y) = sfc.invert(fromHex(value))
191       factory.createPoint(new Coordinate(x, y))
192     }
193 
194     /**
195      * Creates a z2 prefix based on a partition key
196      *
197      * @param value z2 partition value
198      * @param bits number of bits used in the z2 partition scheme
199      * @return
200      */
201     def encodePartition(value: String, bits: Int): String = hexFormat.toHexDigits(value.toLong, bits / 4)
202 
203     /**
204      * Calculate encoded ranges
205      *
206      * @param queries a sequence of OR'd windows to cover. Each window is in the form (xmin, ymin, xmax, ymax)
207      * @param maxRanges rough upper bound on the number of ranges to return
208      */
209     def ranges(queries: Seq[(Double, Double, Double, Double)], maxRanges: Option[Int] = None): Seq[(String, String)] =
210       sfc.ranges(queries, maxRanges = maxRanges).map(r => toHex(r.lower) -> toHex(r.upper))
211 
212     // since the first two bits are not used in our z values, drop them so that truncating the hex value aligns with our curve
213     private def toHex(z: Long): String = hexFormat.toHexDigits(z << 2)
214     private def fromHex(hex: String): Long = HexFormat.fromHexDigitsToLong(hex) >>> 2
215   }
216 
217   /**
218    * Encoder for points
219    */
220   object XZ2Encoder extends TypeEncoder[Geometry, String] {
221 
222     private val sfc = XZ2SFC
223     private val factory = new GeometryFactory()
224     private val hexFormat = HexFormat.of()
225 
226     override val getAlias: String = "xz2"
227 
228     override def resolves(): Class[Geometry] = classOf[Geometry]
229 
230     override def encode(value: Geometry): String = {
231       if (value == null) {
232         throw new NullPointerException("Null values are not allowed")
233       }
234       val env = value.getEnvelopeInternal
235       if (env.isNull) {
236         throw new NullPointerException("Geometry has a null envelope")
237       }
238       toHex(sfc.index(env.getMinX, env.getMinY, env.getMaxX, env.getMaxY))
239     }
240 
241     override def decode(value: String): Geometry = {
242       val (xmin, ymin, xmax, ymax) = sfc.invert(fromHex(value))
243       val ring = Array(
244         new Coordinate(xmin, ymin),
245         new Coordinate(xmin, ymax),
246         new Coordinate(xmax, ymax),
247         new Coordinate(xmax, ymin),
248         new Coordinate(xmin, ymin)
249       )
250       factory.createPolygon(ring)
251     }
252 
253     /**
254      * Calculate encoded ranges
255      *
256      * @param queries a sequence of OR'd windows to cover. Each window is in the form (xmin, ymin, xmax, ymax)
257      * @param maxRanges a rough upper limit on the number of ranges to generate
258      */
259     def ranges(queries: Seq[(Double, Double, Double, Double)], maxRanges: Option[Int] = None): Seq[(String, String)] =
260       sfc.ranges(queries, maxRanges).map(r => toHex(r.lower) -> toHex(r.upper))
261 
262     // our z values use 25 bits, so we only need 7 digits to hex encode the full value
263     // we bit-shift by 3 to move dead bits to the end for better prefix matching
264     private def toHex(z: Long): String = hexFormat.toHexDigits(z << 3, 7)
265     private def fromHex(hex: String): Long = HexFormat.fromHexDigitsToLong(hex) >>> 3
266   }
