169   /**
170    * Encoder for points
171    */
172   object Z2Encoder extends TypeEncoder[Point, String] {
173 
174     private val sfc = Z2SFC
175     private val factory = new GeometryFactory()
176     private val hexFormat = HexFormat.of()
177 
178     override val getAlias: String = "z2"
179 
180     override def resolves(): Class[Point] = classOf[Point]
181 
182     override def encode(value: Point): String = {
183       if (value == null) {
184         throw new NullPointerException("Null values are not allowed")
185       }
186       toHex(sfc.index(value.getX, value.getY))
187     }
188 
189     override def decode(value: String): Point = {
190       val (x, y) = sfc.invert(fromHex(value))
191       factory.createPoint(new Coordinate(x, y))
192     }
193 
194     /**
195      * Calculate encoded ranges
196      *
197      * @param queries a sequence of OR'd windows to cover. Each window is in the form (xmin, ymin, xmax, ymax)
198      * @param maxRanges rough upper bound on the number of ranges to return
199      */
200     def ranges(queries: Seq[(Double, Double, Double, Double)], maxRanges: Option[Int] = None): Seq[(String, String)] =
201       sfc.ranges(queries, maxRanges = maxRanges).map(r => toHex(r.lower) -> toHex(r.upper))
202 
203     // since the first two bits are not used in our z values, drop them so that truncating the hex value aligns with our curve
204     private def toHex(z: Long): String = hexFormat.toHexDigits(z << 2)
205     private def fromHex(hex: String): Long = HexFormat.fromHexDigitsToLong(hex) >>> 2
206   }

208   /**
209    * Encoder for points
210    */
211   object XZ2Encoder extends TypeEncoder[Geometry, String] {
212 
213     private val sfc = XZ2SFC
214     private val factory = new GeometryFactory()
215     private val hexFormat = HexFormat.of()
216 
217     override val getAlias: String = "xz2"
218 
219     override def resolves(): Class[Geometry] = classOf[Geometry]
220 
221     override def encode(value: Geometry): String = {
222       if (value == null) {
223         throw new NullPointerException("Null values are not allowed")
224       }
225       val env = value.getEnvelopeInternal
226       if (env.isNull) {
227         throw new NullPointerException("Geometry has a null envelope")
228       }
229       hexFormat.toHexDigits(sfc.index(env.getMinX, env.getMinY, env.getMaxX, env.getMaxY))
230     }
231 
232     override def decode(value: String): Geometry = {
233       val (xmin, ymin, xmax, ymax) = sfc.invert(HexFormat.fromHexDigitsToLong(value))
234       val ring = Array(
235         new Coordinate(xmin, ymin),
236         new Coordinate(xmin, ymax),
237         new Coordinate(xmax, ymax),
238         new Coordinate(xmax, ymin),
239         new Coordinate(xmin, ymin)
240       )
241       factory.createPolygon(ring)
242     }
243 
244     /**
245      * Calculate encoded ranges
246      *
247      * @param queries a sequence of OR'd windows to cover. Each window is in the form (xmin, ymin, xmax, ymax)
248      * @param maxRanges a rough upper limit on the number of ranges to generate
249      */
250     def ranges(queries: Seq[(Double, Double, Double, Double)], maxRanges: Option[Int] = None): Seq[(String, String)] =
251       sfc.ranges(queries, maxRanges).map(r => hexFormat.toHexDigits(r.lower) -> hexFormat.toHexDigits(r.upper))
252   }
