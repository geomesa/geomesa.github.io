219    * Encoder for points
220    */
221   object Z2Encoder extends TypeEncoder[Point, String] {
222 
223     private val sfc = Z2SFC
224     private val factory = new GeometryFactory()
225     private val hexFormat = HexFormat.of()
226 
227     override val getAlias: String = "z2"
228 
229     override def resolves(): Class[Point] = classOf[Point]
230 
231     override def encode(value: Point): String = {
232       if (value == null) {
233         throw new NullPointerException("Null values are not allowed")
234       }
235       toHex(sfc.index(value.getX, value.getY))
236     }
237 
238     override def decode(value: String): Point = {
239       val (x, y) = sfc.invert(fromHex(value))
240       factory.createPoint(new Coordinate(x, y))
241     }
242 
243     /**
244      * Creates a z2 prefix based on a partition key
245      *
246      * @param value z2 partition value
247      * @param bits number of bits used in the z2 partition scheme
248      * @return
249      */
250     def encodePartition(value: String, bits: Int): String = hexFormat.toHexDigits(value.toLong, bits / 4)
251 
252     /**
253      * Calculate encoded ranges
254      *
255      * @param queries a sequence of OR'd windows to cover. Each window is in the form (xmin, ymin, xmax, ymax)
256      * @param maxRanges rough upper bound on the number of ranges to return
257      */
258     def ranges(queries: Seq[(Double, Double, Double, Double)], maxRanges: Option[Int] = None): Seq[(String, String)] =
259       sfc.ranges(queries, maxRanges = maxRanges).map(r => toHex(r.lower) -> toHex(r.upper))
260 
261     // since the first two bits are not used in our z values, drop them so that truncating the hex value aligns with our curve
262     private def toHex(z: Long): String = hexFormat.toHexDigits(z << 2)
263     private def fromHex(hex: String): Long = HexFormat.fromHexDigitsToLong(hex) >>> 2
264   }

267    * Encoder for points
268    */
269   object XZ2Encoder extends TypeEncoder[Geometry, String] {
270 
271     private val sfc = XZ2SFC
272     private val factory = new GeometryFactory()
273 
274     override val getAlias: String = "xz2"
275 
276     override def resolves(): Class[Geometry] = classOf[Geometry]
277 
278     override def encode(value: Geometry): String = {
279       if (value == null) {
280         throw new NullPointerException("Null values are not allowed")
281       }
282       val env = value.getEnvelopeInternal
283       if (env.isNull) {
284         throw new NullPointerException("Geometry has a null envelope")
285       }
286       sfc.hexEncode(sfc.index(env.getMinX, env.getMinY, env.getMaxX, env.getMaxY))
287     }
288 
289     override def decode(value: String): Geometry = {
290       val (xmin, ymin, xmax, ymax) = sfc.invert(sfc.hexDecode(value))
291       val ring = Array(
292         new Coordinate(xmin, ymin),
293         new Coordinate(xmin, ymax),
294         new Coordinate(xmax, ymax),
295         new Coordinate(xmax, ymin),
296         new Coordinate(xmin, ymin)
297       )
298       factory.createPolygon(ring)
299     }
300 
301     /**
302      * Calculate encoded ranges
303      *
304      * @param queries a sequence of OR'd windows to cover. Each window is in the form (xmin, ymin, xmax, ymax)
305      * @param maxRanges a rough upper limit on the number of ranges to generate
306      */
307     def ranges(queries: Seq[(Double, Double, Double, Double)], maxRanges: Option[Int] = None): Seq[(String, String)] =
308       sfc.ranges(queries, maxRanges).map(r => sfc.hexEncode(r.lower) -> sfc.hexEncode(r.upper))
309   }
