99     logger.debug(s"  Transforms: ${transform.fold("none") { case (t, _) => if (t.isEmpty) { "empty" } else { t }}}")

140   /**
141    * Register a new file with this storage instance. The file must already be in a compatible format.
142    *
143    * @param file file to register
144    * @return registered file
145    */
146   def register(file: URI): StorageFile = {
147     val reader = createReader(None, None)
148     val partitions = new java.util.HashSet[Partition]()
149     val storageFile = WithClose(new StorageFileObserver(metadata.sft)) { observer =>
150       WithClose(reader.read(file)) { iter =>
151         if (!iter.hasNext) {
152           throw new RuntimeException("Could not read any features from input file")
153         }
154         iter.foreach { sf =>
155           partitions.add(Partition(metadata.schemes.map(_.getPartition(sf))))
156           observer(sf)
157         }
158       }
159       val filePath = FileSystemStorage.newFilePath(metadata.sft.getTypeName, FileType.Written, encoding)
160       observer.file(filePath, partitions.iterator().next(), StorageFileAction.Append)
161     }
162     if (partitions.size() != 1) {
163       throw new IllegalArgumentException(s"File corresponds to multiple partitions: ${partitions.asScala.mkString(" AND ")}")
164     }
165 
166     val destination = context.root.resolve(storageFile.file)
167     logger.debug(s"Copying $file to $destination")
168     fs.copy(file, destination)
169     metadata.addFile(storageFile)
170     storageFile
171   }

173   /**
174    * Compact a partition - merge multiple data files into a single file.
175    *
176    * Care should be taken with this method. Currently, there is no guarantee for correct behavior if
177    * multiple threads or storage instances attempt to compact the same partition simultaneously.
178    *
179    * @param partition partition to compact, or all partitions
180    * @param threads suggested threads to use for file system operations
181    */
182   def compact(partition: Partition, threads: Int = 1): Unit = {
183     val files = metadata.getFiles(partition)
184     val toCompact = sizer.targetSize match {
185       case None => files
186       case Some(t) =>
187         files.filter { f =>
188           if (this.sizer.fileIsSized(context.root.resolve(f.file), t)) {
189             logger.debug(s"Skipping compaction for file [${f.file}] (already target size)")
190             false
191           } else {
192             true
193           }
194         }
195     }
196 
197     if (toCompact.isEmpty) {
198       logger.debug("Skipping compaction - no files to compact")
199     } else {
200       logger.debug(s"Compacting data files: [${toCompact.map(_.file).mkString(", ")}]")
201 
202       var written = 0L
203 
204       val reader = createReader(None, None)
205       // tracks newly added files so we can register them atomically
206       val fileTracker = new FileTracker(metadata.sft, metadata.schemes)
207 
208       WithClose(createWriter(partition, StorageFileAction.Append, FileType.Compacted, fileTracker)) { writer =>
209         WithClose(FileSystemThreadedReader(reader, toCompact, threads)) { reader =>
210           while (reader.hasNext) {
211             val feature = reader.next()
212             writer.write(feature)
213             written += 1
214           }
215         }
216       }
217 
218       logger.debug(s"Updating metadata with new files: [${fileTracker.getFiles().map(_.file).mkString(", ")}]")
219       metadata.replaceFiles(toCompact, fileTracker.getFiles())
220 
221       logger.debug(s"Deleting old files [${toCompact.mkString(", ")}]")
222       val failures = ArrayBuffer.empty[String]
223       toCompact.foreach { file =>
224         val path = context.root.resolve(file.file)
225         if (Try(fs.delete(path)).isFailure) {
226           failures.append(file.file)
227         }
228       }
229 
230       if (failures.nonEmpty) {
231         logger.error(s"Failed to delete some files: [${failures.mkString(", ")}]")
232       }
233 
234       logger.debug(s"Compacted $written records")
235     }
236   }

318   def newFilePath(typeName: String, fileType: FileType.FileType, ext: String): String = {
319     val filename =
320       s"${fileType}_${SafeNameRegex.replaceAllIn(typeName, "-").take(20)}_${UUID.randomUUID().toString.replaceAllLiterally("-", "")}.$ext"
321     // partitioning logic taken from Apache Iceberg: https://iceberg.apache.org/docs/nightly/aws/#object-store-file-layout
322     val hash = {
323       val bytes = filename.getBytes(StandardCharsets.UTF_8)
324       val hash = MurmurHash3.hash32x86(bytes, 0, bytes.length, 0)
325       // Integer#toBinaryString excludes leading zeros, which we want to preserve
326       Integer.toBinaryString(hash | Integer.MIN_VALUE)
327     }
328     s"${hash.substring(0, 4)}/${hash.substring(4, 8)}/${hash.substring(8, 12)}/${hash.substring(12, 20)}/$filename"
329   }

360   /**
361    * Writes files up to a given size, then starts a new file
362    *
363    * @param writers iterator of files to write
364    * @param estimator target file size estimator
365    */
366   class ChunkedFileSystemWriter(writers: Iterator[FileSystemWriter], estimator: UpdatingFileSizeEstimator)
367       extends FileSystemWriter {
368 
369     private var totalCount = 0L // total number of features written across all chunks
370     private var totalBytes = 0L // sum size of all finished chunks
371     private var remaining = estimator.estimate(0L)
372     private var writer: FileSystemWriter = _
373 
374     override def write(feature: SimpleFeature): Unit = {
375       if (writer == null) {
376         writer = writers.next()
377       }
378       writer.write(feature)
379       totalCount += 1
380       remaining -= 1
381       if (remaining == 0) {
382         val dataSize = writer.size
383         if (estimator.done(dataSize)) {
384           writer.close()
385           totalBytes += writer.size // re-calculate now that writer is closed, so we get the final, accurate size
386           writer = null
387           // adjust our estimate to account for the actual bytes written
388           estimator.update(totalBytes, totalCount)
389           remaining = estimator.estimate(0L)
390         } else {
391           remaining = math.max(100L, estimator.estimate(dataSize))
392         }
393       }
394     }
395 
396     override def size: Long = totalBytes + Option(writer).fold(0L)(_.size)
397 
398     override def flush(): Unit = if (writer != null) { writer.flush() }
399 
400     override def close(): Unit = {
401       if (writer != null) {
402         writer.close()
403       }
404       estimator.close()
405     }
406   }

413   class FileSystemUpdateWriter(
414       schemes: Set[PartitionScheme],
415       reader: CloseableFeatureIterator,
416       newWriter: (Partition, StorageFileAction) => FileSystemWriter
417     ) extends Iterator[SimpleFeature] with Closeable with Flushable {
418 
419     // TODO limit number of open writers
420 
421     private val appenders = scala.collection.mutable.Map.empty[Set[PartitionKey], FileSystemWriter]
422     private val modifiers = scala.collection.mutable.Map.empty[Set[PartitionKey], FileSystemWriter]
423     private val deleters = scala.collection.mutable.Map.empty[Set[PartitionKey], FileSystemWriter]
424 
425     private var feature: SimpleFeature = _
426     private var partition: Set[PartitionKey] = _
427 
428     /**
429      * Writes a modification to the last feature returned by `next`
430      */
431     def write(): Unit = {
432       if (feature == null) {
433         throw new IllegalArgumentException("Must call 'next' before calling 'write'")
434       }
435       val update = schemes.map(_.getPartition(feature))
436       if (update == partition) {
437         modifiers.getOrElseUpdate(update, newWriter(Partition(update), StorageFileAction.Modify)).write(feature)
438       } else {
439         // add a delete marker in the old partition, and an append in the new one, since we only track updates per-partition
440         deleters.getOrElseUpdate(partition, newWriter(Partition(partition), StorageFileAction.Delete)).write(feature)
441         appenders.getOrElseUpdate(update, newWriter(Partition(update), StorageFileAction.Append)).write(feature)
442       }
443       feature = null
444     }
445 
446     /**
447      * Deletes the last feature returned by `next`
448      */
449     def remove(): Unit = {
450       if (feature == null) {
451         throw new IllegalArgumentException("Must call 'next' before calling 'remove'")
452       }
453       deleters.getOrElseUpdate(partition, newWriter(Partition(partition), StorageFileAction.Delete)).write(feature)
454       feature = null
455     }
456 
457     override def hasNext: Boolean = reader.hasNext
458 
459     override def next(): SimpleFeature = {
460       feature = reader.next() // note: our reader returns a mutable copy of the feature
461       partition = schemes.map(_.getPartition(feature))
462       feature
463     }
464 
465     override def flush(): Unit = FlushQuietly.raise(appenders.values.toSeq ++ modifiers.values ++ deleters.values)
466 
467     override def close(): Unit = CloseQuietly.raise(Seq(reader) ++ appenders.values ++ modifiers.values ++ deleters.values)
468   }

494   class StorageFileObserver(sft: SimpleFeatureType) extends FileSystemObserver with LazyLogging {
495 
496     import scala.collection.JavaConverters._
497 
498     private var count: Long = 0L
499 
500     private val spatialBounds = sft.spatialBounds().map(_ -> new Envelope())
501 
502     private val nonSpatialBounds = sft.nonSpatialBounds().flatMap { i =>
503       val binding = sft.getDescriptor(i).getType.getBinding
504       AttributeIndexKey.TypeRegistry.getAllEncoders.asScala.find(_.resolves().isAssignableFrom(binding)) match {
505         case Some(encoder) => Some(AttributeBoundsBuilder(i, encoder.asInstanceOf[TypeEncoder[AnyRef, String]]))
506         case None =>
507           logger.warn(
508             s"Can't find an encoder for attribute ${sft.getDescriptor(i).getLocalName} of type ${binding.getSimpleName} - " +
509               "will not track bounds")
510           None
511       }
512     }
513 
514     private var sorted: Seq[(Int, Ordering[AnyRef], AnyRef)] = Range(0, sft.getAttributeCount).flatMap { i =>
515       val binding = sft.getDescriptor(i).getType.getBinding
516       if (classOf[Comparable[_]].isAssignableFrom(binding)) {
517         Some((i, Ordering.ordered[AnyRef](_.asInstanceOf[Comparable[AnyRef]]), null))
518       } else {
519         None
520       }

523     /**
524      * Get the file metadata that has been gathered so far
525      *
526      * @param path file path
527      * @param partition file partition
528      * @param action file action
529      * @return
530      */
531     def file(path: String, partition: Partition, action: StorageFileAction): StorageFile = {
532       val spatial = spatialBounds.flatMap { case (i, env) => SpatialBounds(i, env) }
533       val nonSpatial = nonSpatialBounds.flatMap(_.build())
534       val sort = sorted.map(_._1)
535       StorageFile(path, partition, count, action, spatial, nonSpatial, sort)
536     }

538     override def apply(feature: SimpleFeature): Unit = {
539       count += 1L
540       spatialBounds.foreach { case (i, env) =>
541         val geom = feature.getAttribute(i).asInstanceOf[Geometry]
542         if (geom != null) {
543           env.expandToInclude(geom.getEnvelopeInternal)
544         }
545       }
546       nonSpatialBounds.foreach(_.apply(feature))
547       if (sorted.nonEmpty) {
548         sorted = sorted.flatMap { case (i, ordering, last) =>
549           val next = feature.getAttribute(i)
550           if (last == null || (next != null && ordering.compare(next, last) < 0)) {
551             Some((i, ordering, next))
552           } else {
553             None
554           }
555         }
556       }
557     }
