92     val sort = configured.getHints.getSortFields

97     logger.debug(s"  Transforms: ${transform.fold("none") { case (t, _) => if (t.isEmpty) { "empty" } else { t }}}")

138   /**
139    * Register a new file with this storage instance. The file must already be in a compatible format.
140    *
141    * @param file file to register
142    * @return registered file
143    */
144   def register(file: URI): StorageFile = {
145     val reader = createReader(None, None)
146     val partitions = new java.util.HashSet[Partition]()
147     val storageFile = WithClose(new StorageFileObserver(metadata.sft)) { observer =>
148       WithClose(reader.read(file)) { iter =>
149         if (!iter.hasNext) {
150           throw new RuntimeException("Could not read any features from input file")
151         }
152         iter.foreach { sf =>
153           partitions.add(Partition(metadata.schemes.map(_.getPartition(sf))))
154           observer(sf)
155         }
156       }
157       val filePath = FileSystemStorage.newFilePath(metadata.sft.getTypeName, FileType.Written, encoding)
158       observer.file(filePath, partitions.iterator().next(), StorageFileAction.Append)
159     }
160     if (partitions.size() != 1) {
161       throw new IllegalArgumentException(s"File corresponds to multiple partitions: ${partitions.asScala.mkString(" AND ")}")
162     }
163 
164     val destination = context.root.resolve(storageFile.file)
165     logger.debug(s"Copying $file to $destination")
166     fs.copy(file, destination)
167     metadata.addFile(storageFile)
168     storageFile
169   }

171   /**
172    * Compact a partition - merge multiple data files into a single file.
173    *
174    * Care should be taken with this method. Currently, there is no guarantee for correct behavior if
175    * multiple threads or storage instances attempt to compact the same partition simultaneously.
176    *
177    * @param partition partition to compact, or all partitions
178    * @param threads suggested threads to use for file system operations
179    */
180   def compact(partition: Partition, threads: Int = 1): Unit = {
181     val files = metadata.getFiles(partition)
182     val toCompact = sizer.targetSize match {
183       case None => files
184       case Some(t) =>
185         files.filter { f =>
186           if (this.sizer.fileIsSized(context.root.resolve(f.file), t)) {
187             logger.debug(s"Skipping compaction for file [${f.file}] (already target size)")
188             false
189           } else {
190             true
191           }
192         }
193     }
194 
195     if (toCompact.isEmpty) {
196       logger.debug("Skipping compaction - no files to compact")
197     } else {
198       logger.debug(s"Compacting data files: [${toCompact.map(_.file).mkString(", ")}]")
199 
200       var written = 0L
201 
202       val reader = createReader(None, None)
203       // tracks newly added files so we can register them atomically
204       val fileTracker = new FileTracker(metadata.sft, metadata.schemes)
205 
206       WithClose(createWriter(partition, StorageFileAction.Append, FileType.Compacted, fileTracker)) { writer =>
207         WithClose(FileSystemThreadedReader(reader, toCompact, threads)) { reader =>
208           while (reader.hasNext) {
209             val feature = reader.next()
210             writer.write(feature)
211             written += 1
212           }
213         }
214       }
215 
216       logger.debug(s"Updating metadata with new files: [${fileTracker.getFiles().map(_.file).mkString(", ")}]")
217       metadata.replaceFiles(toCompact, fileTracker.getFiles())
218 
219       logger.debug(s"Deleting old files [${toCompact.mkString(", ")}]")
220       val failures = ArrayBuffer.empty[String]
221       toCompact.foreach { file =>
222         val path = context.root.resolve(file.file)
223         if (Try(fs.delete(path)).isFailure) {
224           failures.append(file.file)
225         }
226       }
227 
228       if (failures.nonEmpty) {
229         logger.error(s"Failed to delete some files: [${failures.mkString(", ")}]")
230       }
231 
232       logger.debug(s"Compacted $written records")
233     }
234   }

287 
288     def newWriter(): FileSystemWriter = {
289       val file = FileSystemStorage.newFilePath(metadata.sft.getTypeName, fileType, encoding)
290       val path = context.root.resolve(file)
291       val updateObserver = new MetadataObserver(metadata, file, partition, action)
292       val observer = if (observers.isEmpty) { updateObserver } else {
293         new CompositeObserver(observers.map(_.apply(path)).+:(updateObserver))
294       }
295       createWriter(path, partition, observer)

299       case None => newWriter()
300       case Some(s) => new ChunkedFileSystemWriter(Iterator.continually(newWriter()), sizer.estimator(s))

316   def newFilePath(typeName: String, fileType: FileType.FileType, ext: String): String = {
317     val filename =
318       s"${fileType}_${SafeNameRegex.replaceAllIn(typeName, "-").take(20)}_${UUID.randomUUID().toString.replaceAllLiterally("-", "")}.$ext"
319     // partitioning logic taken from Apache Iceberg: https://iceberg.apache.org/docs/nightly/aws/#object-store-file-layout
320     val hash = {
321       val bytes = filename.getBytes(StandardCharsets.UTF_8)
322       val hash = MurmurHash3.hash32x86(bytes, 0, bytes.length, 0)
323       // Integer#toBinaryString excludes leading zeros, which we want to preserve
324       Integer.toBinaryString(hash | Integer.MIN_VALUE)
325     }
326     s"${hash.substring(0, 4)}/${hash.substring(4, 8)}/${hash.substring(8, 12)}/${hash.substring(12, 20)}/$filename"
327   }

358   /**
359    * Writes files up to a given size, then starts a new file
360    *
361    * @param writers iterator of files to write
362    * @param estimator target file size estimator
363    */
364   // noinspection ScalaWeakerAccess
365   class ChunkedFileSystemWriter(writers: Iterator[FileSystemWriter], estimator: UpdatingFileSizeEstimator)
366       extends FileSystemWriter {
367 
368     private var totalCount = 0L // total number of features written across all chunks
369     private var totalBytes = 0L // sum size of all finished chunks
370     private var remaining = estimator.estimate(0L)
371     private var writer: FileSystemWriter = _
372 
373     override def write(feature: SimpleFeature): Unit = {
374       if (writer == null) {
375         writer = writers.next()
376       }
377       writer.write(feature)
378       totalCount += 1
379       remaining -= 1
380       if (remaining == 0) {
381         val dataSize = writer.size
382         if (estimator.done(dataSize)) {
383           writer.close()
384           totalBytes += writer.size // re-calculate now that writer is closed, so we get the final, accurate size
385           writer = null
386           // adjust our estimate to account for the actual bytes written
387           estimator.update(totalBytes, totalCount)
388           remaining = estimator.estimate(0L)
389         } else {
390           remaining = math.max(100L, estimator.estimate(dataSize))
391         }
392       }
393     }
394 
395     override def size: Long = totalBytes + Option(writer).fold(0L)(_.size)
396 
397     override def flush(): Unit = if (writer != null) { writer.flush() }
398 
399     override def close(): Unit = {
400       if (writer != null) {
401         writer.close()
402       }
403       estimator.close()
404     }
405   }

408    * Update writer implementation

410    * @param reader reader for features to update

412   class FileSystemUpdateWriter(
413       schemes: Set[PartitionScheme],
414       reader: CloseableFeatureIterator,
415       newWriter: (Partition, StorageFileAction) => FileSystemWriter
416     ) extends Iterator[SimpleFeature] with Closeable with Flushable {
417 
418     // TODO limit number of open writers
419 
420     private val appenders = scala.collection.mutable.Map.empty[Set[PartitionKey], FileSystemWriter]
421     private val modifiers = scala.collection.mutable.Map.empty[Set[PartitionKey], FileSystemWriter]
422     private val deleters = scala.collection.mutable.Map.empty[Set[PartitionKey], FileSystemWriter]
423 
424     private var feature: SimpleFeature = _
425     private var partition: Set[PartitionKey] = _
426 
427     /**
428      * Writes a modification to the last feature returned by `next`
429      */
430     def write(): Unit = {
431       if (feature == null) {
432         throw new IllegalArgumentException("Must call 'next' before calling 'write'")
433       }
434       val update = schemes.map(_.getPartition(feature))
435       if (update == partition) {
436         modifiers.getOrElseUpdate(update, newWriter(Partition(update), StorageFileAction.Modify)).write(feature)
437       } else {
438         // add a delete marker in the old partition, and an append in the new one, since we only track updates per-partition
439         deleters.getOrElseUpdate(partition, newWriter(Partition(partition), StorageFileAction.Delete)).write(feature)
440         appenders.getOrElseUpdate(update, newWriter(Partition(update), StorageFileAction.Append)).write(feature)
441       }
442       feature = null
443     }
444 
445     /**
446      * Deletes the last feature returned by `next`
447      */
448     def remove(): Unit = {
449       if (feature == null) {
450         throw new IllegalArgumentException("Must call 'next' before calling 'remove'")
451       }
452       deleters.getOrElseUpdate(partition, newWriter(Partition(partition), StorageFileAction.Delete)).write(feature)
453       feature = null
454     }
455 
456     override def hasNext: Boolean = reader.hasNext
457 
458     override def next(): SimpleFeature = {
459       feature = reader.next() // note: our reader returns a mutable copy of the feature
460       partition = schemes.map(_.getPartition(feature))
461       feature
462     }
463 
464     override def flush(): Unit = FlushQuietly.raise(appenders.values.toSeq ++ modifiers.values ++ deleters.values)
465 
466     override def close(): Unit = CloseQuietly.raise(Seq(reader) ++ appenders.values ++ modifiers.values ++ deleters.values)
467   }

493   class StorageFileObserver(sft: SimpleFeatureType) extends FileSystemObserver with LazyLogging {
494 
495     import scala.collection.JavaConverters._
496 
497     private var count: Long = 0L
498 
499     private val columnBounds = sft.columnBounds().flatMap { i =>
500       val binding = sft.getDescriptor(i).getType.getBinding
501       val alias = StorageMetadata.typeAlias(binding)
502       StorageMetadata.TypeRegistry.getAllEncoders.asScala.find(_.getAlias == alias) match {
503         case Some(encoder) => Some(ColumnBoundsBuilder(i, encoder.asInstanceOf[TypeEncoder[AnyRef, String]]))
504         case None =>
505           logger.warn(
506             s"Can't find an encoder for attribute ${sft.getDescriptor(i).getLocalName} of type ${binding.getSimpleName} - " +
507               "will not track bounds")
508           None
509       }
510     }
511 
512     private var sorted: Seq[(Int, Ordering[AnyRef], AnyRef)] = Range(0, sft.getAttributeCount).flatMap { i =>
513       val binding = sft.getDescriptor(i).getType.getBinding
514       if (classOf[Comparable[_]].isAssignableFrom(binding)) {
515         Some((i, Ordering.ordered[AnyRef](_.asInstanceOf[Comparable[AnyRef]]), null))
516       } else {
517         None
518       }
519     }
520 
521     /**
522      * Get the file metadata that has been gathered so far
523      *
524      * @param path file path
525      * @param partition file partition
526      * @param action file action
527      * @return
528      */
529     def file(path: String, partition: Partition, action: StorageFileAction): StorageFile = {
530       val bounds = columnBounds.flatMap(_.build())
531       val sort = sorted.map(_._1)
532       StorageFile(path, partition, count, action, bounds, sort)
533     }
534 
535     override def apply(feature: SimpleFeature): Unit = {
536       count += 1L
537       columnBounds.foreach(_.apply(feature))
538       if (sorted.nonEmpty) {
539         sorted = sorted.flatMap { case (i, ordering, last) =>
540           val next = feature.getAttribute(i)
541           if (last == null || (next != null && ordering.compare(next, last) < 0)) {
542             Some((i, ordering, next))
543           } else {
544             None
545           }
546         }
547       }
548     }

551     override def close(): Unit = {}
552   }
