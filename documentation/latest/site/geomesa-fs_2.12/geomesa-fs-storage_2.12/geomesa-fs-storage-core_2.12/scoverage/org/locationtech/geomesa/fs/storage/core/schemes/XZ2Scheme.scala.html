29 /**
30  * XZ2 spatial scheme.
31  *
32  * This scheme uses a fixed high-resolution curve for indexing, then truncates the value to create partition groups,
33  * in order to align with the iceberg 'truncate' transforms.
34  *
35  * Index values are (max) 7-digit hex-encoded longs, as we're only using 25 bits for the high-resolution curve,
36  * we can fit it in 7 digits.
37  *
38  * @param attribute name of the attribute being partitioned
39  * @param index index in the sft of the attribute being partitioned
40  * @param bits number of bits of resolution used for partitioning
41  */
42 case class XZ2Scheme(attribute: String, index: Int, bits: Int) extends PartitionScheme {
43 
44   import FilterHelper.ff
45   import XZ2Scheme.FullHexDigits
46 
47   require(bits % 4 == 0, s"Bit precision must be a multiple of 4, but received $bits")
48 
49   private val xz2 = XZ2SFC
50   private val hexFormat = HexFormat.of()
51 
52   // partition level derived from bits parameter
53   // each level adds 2 bits (4 quadrants)
54   private val partitionLevel = (bits / 2).toShort
55   // number of hex digits used to represent our z value - bits = (xz2.g - partitionLevel) * 2, then divide by 4 to get hex
56   private val digits = FullHexDigits - ((xz2.g - partitionLevel) / 2)
57 
58   lazy private val wholeWorldRanges = Some(generateRanges(Seq((-180, -90, 180, 90))))
59 
60   override val name: String = s"${XZ2Scheme.name}:attribute=$attribute:bits=$bits"
61 
62   override def getPartition(feature: SimpleFeature): PartitionKey = {
63     val envelope = feature.getAttribute(index).asInstanceOf[Geometry].getEnvelopeInternal
64     val zValue = xz2.index(envelope.getMinX, envelope.getMinY, envelope.getMaxX, envelope.getMaxY)
65     PartitionKey(name, truncateToPartition(zValue))
66   }
67 
68   override def getCoveringFilter(partition: PartitionKey): Filter = {
69     // TODO maybe we can improve this with *some* kind of bbox?
70     val zPrefix = partition.value
71     val lower = zPrefix.padTo(FullHexDigits, '0')
72     val upper = zPrefix.padTo(FullHexDigits, 'f')
73     ff.between(ff.function(XZ2Scheme.FunctionName.getName), ff.literal(lower), ff.literal(upper))
74   }
75 
76   override def getRangesForFilter(filter: Filter): Option[Seq[PartitionRange]] = {
77     val geometries = FilterHelper.extractGeometries(filter, attribute, intersect = true)
78     if (geometries.isEmpty) {
79       None
80     } else if (geometries.disjoint) {
81       Some(Seq.empty)
82     } else {
83       Some(generateRanges(geometries.values.map(GeometryUtils.bounds)))
84     }
85   }
86 
87   override def getPartitionsForFilter(filter: Filter): Option[Seq[PartitionKey]] = {
88     getRangesForFilter(filter).orElse(wholeWorldRanges).map { ranges =>
89       ranges.flatMap { range =>
90         Iterator.iterate(range.lower)(incrementHex).takeWhile(_ < range.upper).map(PartitionKey(name, _))
91       }
92     }
93   }
94 
95   private def generateRanges(xy: Seq[(Double, Double, Double, Double)]): Seq[PartitionRange] = {
96     val builder = new RangeBuilder()
97     xz2.ranges(xy).foreach { range =>
98       val lower = truncateToPartition(range.lower)
99       // index ranges are inclusive, but partition ranges are exclusive
100       val upper = incrementHex(truncateToPartition(range.upper))
101       builder += PartitionRange(name, lower, upper)
102     }
103     builder.result()
104   }
105 
106   // truncates a full-resolution index to a partition group ID
107   private def truncateToPartition(fullIndex: Long): String = hexFormat.toHexDigits(fullIndex << 3, FullHexDigits).take(digits)
108 }

110 object XZ2Scheme extends SpatialPartitionSchemeFactory[Geometry]("xz2") {
111 
112   // number of digits required to index a "full" (g == 12) xz value, which is 25 bits
113   private val FullHexDigits = 7
114 
115   val FunctionName = new FunctionNameImpl("xz2", classOf[String], parameter("geom", classOf[String], 0, 1))
116 
117   override def buildPartitionScheme(bits: Int, geom: String, geomIndex: Int): PartitionScheme =
118     XZ2Scheme(geom, geomIndex, bits)
119 
120   private def incrementHex(hex: String): String = incrementHex(hex, hex.length - 1)
121 
122   @tailrec
123   private def incrementHex(hex: String, pos: Int): String = {
124     val c = hex.charAt(pos)
125     if (c != 'f') {
126       hex.substring(0, pos) + (c + 1).toChar + hex.substring(pos + 1)
127     } else if (pos == 0) {
128       hex + '0' // note: this isn't actually incrementing the value but should sort after all the valid hex values
129     } else {
130       incrementHex(hex.substring(0, pos) + '0' + hex.substring(pos + 1), pos - 1)
131     }
132   }
133 
134   /**
135    * Function to calculate an XZ2 hex-encoded value
136    */
137   class XZ2Function extends FunctionExpressionImpl(FunctionName) {
138 
139     private var expression: Expression = _
140 
141     override def setParameters(params: java.util.List[Expression]): Unit = {
142       super.setParameters(params)
143       if (params.isEmpty) {
144         expression = GetDefaultGeometry
145       } else {
146         expression = getExpression(0)
147       }
148     }
149 
150     override def evaluate(o: AnyRef): AnyRef = {
151       if (o == null) {
152         return null
153       }
154       val value = expression.evaluate(o, classOf[Geometry])
155       if (value == null) {
156         return null
157       }
158       XZ2Encoder.encode(value)
159     }
160   }
161 
162   private object GetDefaultGeometry extends Expression {
163 
164     override def evaluate(obj: Any): Geometry = obj match {
165       case sf: SimpleFeature => sf.getDefaultGeometry.asInstanceOf[Geometry]
166       case _ => null
167     }
168 
169     override def evaluate[T](obj: Any, context: Class[T]): T = evaluate(obj).asInstanceOf[T] // only called by our code, above
170 
171     override def accept(visitor: ExpressionVisitor, extraData: Any): AnyRef = throw new UnsupportedOperationException()
172   }
