23 class DtgPathFiltering(attribute: String, pattern: Pattern, format: DateTimeFormatter, buffer: Duration)
24     extends PathFiltering with LazyLogging {
25 
26   def apply(filter: Filter): PathFilter = {
27     val filterIntervals = FilterHelper.extractIntervals(filter, attribute, handleExclusiveBounds = true)
28     path => try {
29       val time = parseDtg(path.getName).toInstant
30       val millis = buffer.toMillis
31       val lower = ZonedDateTime.ofInstant(time.minusMillis(millis), ZoneOffset.UTC)
32       val upper = ZonedDateTime.ofInstant(time.plusMillis(millis), ZoneOffset.UTC)
33       val buffered = Bounds(Bound.inclusive(lower), Bound.inclusive(upper))
34       val included = filterIntervals.exists(bounds => bounds.intersects(buffered))
35       logger.whenDebugEnabled {
36         if (included) {
37           logger.debug(s"Including path ${path.getName} for filter $filter")
38         } else {
39           logger.debug(s"Excluding path ${path.getName} for filter $filter")
40         }
41       }
42       included
43     } catch {
44       case NonFatal(ex) =>
45         logger.warn(s"Failed to evaluate filter for path '${path.getName}'", ex)
46         true
47     }
48   }
49 
50   private def parseDtg(name: String): ZonedDateTime = {
51     Option(name)
52       .map(pattern.matcher)
53       .filter(_.matches)
54       .filter(_.groupCount > 0)
55       .map(_.group(1))
56       .map(ZonedDateTime.parse(_, format))
57       .getOrElse {
58         throw new IllegalArgumentException(s"Failed to parse ${classOf[ZonedDateTime].getName} " +
59           s"from file name '$name' for pattern '$pattern' and format '$format'")
60       }
61   }
62 
63   override def toString: String = {
64     s"${this.getClass.getName}(attribute = $attribute, pattern = $pattern, format = $format, buffer = $buffer)"
65   }
