19 class ConverterStorageFactory extends FileSystemStorageFactory with LazyLogging {
20 
21   override val encoding: String = ConverterStorageFactory.Encoding
22 
23   override def apply(context: FileSystemContext, metadata: StorageMetadata): FileSystemStorage = {
24     val converter = {
25       val convertArg = context.conf.get(ConverterConfigParam)
26           .orElse(context.conf.get(ConverterNameParam))
27           .getOrElse(throw new IllegalArgumentException(s"Must provide either converter config or name"))
28       val converterConfig = ConverterConfigResolver.getArg(ConfArgs(convertArg)) match {
29         case Left(e) => throw new IllegalArgumentException("Could not load Converter with provided parameters", e)
30         case Right(c) => c
31       }
32       SimpleFeatureConverter(metadata.sft, converterConfig)
33     }
34 
35     val pathFilteringOpts =
36       context.conf.collect { case (k, v) if k.startsWith(PathFilterOptsPrefix) => k.substring(PathFilterOptsPrefix.length) -> v }
37 
38     val pathFiltering = context.conf.get(PathFilterName).flatMap { name =>
39       val factory = PathFilteringFactory.load(NamedOptions(name, pathFilteringOpts.toMap))
40       if (factory.isEmpty) {
41         throw new IllegalArgumentException(s"Failed to load ${classOf[PathFiltering].getName} for config '$name'")
42       }
43       factory
44     }
45 
46     val converterPath =
47       context.conf.get(ConverterPathParam).map(context.root.resolve).getOrElse {
48         throw new IllegalArgumentException("Must provide converter path")
49       }
50 
51     new ConverterStorage(FileSystemContext.create(converterPath, context.conf, context.namespace), metadata, converter, pathFiltering)
52   }
53 }
