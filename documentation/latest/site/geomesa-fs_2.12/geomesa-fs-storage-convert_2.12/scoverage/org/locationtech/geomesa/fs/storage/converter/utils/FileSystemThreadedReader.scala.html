22 /**
23   * Multi-threaded file reads
24   *
25   * @param es executor service used by the read tasks
26   * @param phaser phaser for tracking read task completion
27   * @param queue intermediate queue populated by the read tasks
28   */
29 class FileSystemThreadedReader private (es: ExecutorService, phaser: Phaser, queue: BlockingQueue[SimpleFeature])
30     extends CloseableIterator[SimpleFeature] {
31 
32   private val localQueue = new java.util.LinkedList[SimpleFeature]()
33 
34   private var current: SimpleFeature = _
35 
36   override def hasNext: Boolean = {
37     if (current != null) {
38       return true
39     }
40     current = localQueue.pollFirst
41     if (current != null) {
42       return true
43     }
44 
45     while (!phaser.isTerminated) {
46       current = queue.poll(100, TimeUnit.MILLISECONDS)
47       if (current != null) {
48         queue.drainTo(localQueue, 10000)
49         return true
50       }
51     }
52     // last check - if phaser.isTerminated, the queue should have whatever values are left
53     current = queue.poll()
54     if (current != null) {
55       queue.drainTo(localQueue, 10000)
56       true
57     } else {
58       false
59     }
60   }
61 
62   override def next(): SimpleFeature = {
63     if (hasNext) {
64       val ret = current
65       current = null
66       ret
67     } else {
68       Iterator.empty.next
69     }
70   }
71 
72   override def close(): Unit = {
73     try { es.shutdownNow() } finally {
74       phaser.forceTermination() // unregister any tasks that didn't start
75     }
76   }
77 }

79 object FileSystemThreadedReader extends StrictLogging {
80 
81   def apply(reader: FileSystemPathReader, files: Seq[URI], threads: Int): CloseableIterator[SimpleFeature] = {
82     if (threads < 2) {
83       CloseableIterator.wrap(files).flatMap(f => CloseableIterator.wrap(reader.read(f)))
84     } else {
85       val queue = new LinkedBlockingQueue[SimpleFeature](2000000)
86       val es = Executors.newFixedThreadPool(threads)
87 
88       val phaser = new Phaser(1) {
89         override protected def onAdvance(phase: Int, registeredParties: Int): Boolean = {
90           // when all tasks have been completed, shutdown the executor service
91           es.shutdown()
92           true // return true to indicate the phaser should terminate
93         }
94       }
95 
96       try {
97         // ensure that we don't register too many parties on this phaser
98         files.grouped(PhaserUtils.MaxParties - 1).foreach { group =>
99           val child = new Phaser(phaser)
100           child.register() // register new task
101           es.submit(new ChainedReaderTask(es, child, reader, group, Seq.empty, queue))
102         }
103       } catch {
104         case NonFatal(e) => es.shutdownNow(); throw e
105       } finally {
106         phaser.arriveAndDeregister()
107       }
108 
109       new FileSystemThreadedReader(es, phaser, queue)
110     }
111   }

113   /**
114     * Performs a set of chained (dependent) reads. The groups must be handled serially, but the files within a
115     * given group can be read in parallel.
116     *
117     * @param es executor service for submitting new read tasks
118     * @param phaser phaser to track run lifecycle
119     * @param reader reader
120     * @param group current group of files that can be read in parallel
121     * @param chain remaining groups of files that must be read sequentially
122     * @param queue result queue
123     * @param mods modifications/deletes for this group of files
124     */
125   private class ChainedReaderTask(
126       es: ExecutorService,
127       phaser: Phaser,
128       reader: FileSystemPathReader,
129       group: Seq[URI],
130       chain: Seq[Seq[URI]],
131       queue: BlockingQueue[SimpleFeature],
132       mods: java.util.Set[String] = new java.util.HashSet[String]()
133     ) extends Runnable {
134 
135     override def run(): Unit = {
136       val child = new Phaser(1) {
137         override protected def onAdvance(phase: Int, registeredParties: Int): Boolean = {
138           // when this group is done, submit the next group for processing
139           try {
140             if (chain.nonEmpty) {
141               phaser.register() // register new task
142               es.submit(new ChainedReaderTask(es, phaser, reader, chain.head, chain.tail, queue, mods))
143             }
144             true // return true to indicate the phaser should terminate
145           } finally {
146             phaser.arriveAndDeregister()
147           }
148         }
149       }
150       try {
151         group.foreach { file =>
152           child.register() // register new task
153           es.submit(new ReaderTask(child, queue, file, CloseableIterator.wrap(reader.read(file))))
154         }
155       } finally {
156         child.arriveAndDeregister()
157       }
158     }
159   }
160 
161   /**
162     * Task to run a reader and add the results to a result queue
163     *
164     * @param phaser phaser for tracking completion of this task
165     * @param queue result queue
166     * @param file file path (for logging)
167     * @param iter lazily evaluated iterator for reading the path
168     */
169   private class ReaderTask(
170       phaser: Phaser,
171       queue: BlockingQueue[SimpleFeature],
172       file: URI,
173       iter: => CloseableIterator[SimpleFeature]
174     ) extends Runnable {
175 
176     override def run(): Unit = {
177       try {
178         WithClose(iter)(_.foreach(queue.put))
179       } catch {
180         case NonFatal(e) => logger.error(s"Error reading file $file", e)
181       } finally {
182         phaser.arriveAndDeregister()
183       }
184     }
185   }
