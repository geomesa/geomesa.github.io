27 class ConverterFileSystemReader(
28     fs: ObjectStore,
29     converter: SimpleFeatureConverter,
30     filter: Option[Filter],
31     transform: Option[(String, SimpleFeatureType)],
32     pathFiltering: Option[PathFiltering]
33   ) extends FileSystemPathReader with StrictLogging {
34 
35   private lazy val pathFilter: Option[PathFilter] = pathFiltering.flatMap(pf => filter.map(pf.apply))
36 
37   override def read(file: URI): CloseableIterator[SimpleFeature] = {
38     if (pathFilter.forall(_.accept(new Path(file)))) {
39       logger.debug(s"Opening file $file")
40       val iter = try {
41         val streams = ArchiveFormat(file) match {
42           case None => CloseableIterator.wrap(fs.read(file).map(fs.toString -> _).toIterator)
43           case Some(f) => fs.read(file, f).map(a => (a.name, a.is))
44         }
45         streams.flatMap { case (name, is) =>
46           val params = EvaluationContext.inputFileParam(name) ++ filter.map(EvaluationContext.FilterKey -> _)
47           converter.process(is, converter.createEvaluationContext(params))
48         }
49       } catch {
50         case NonFatal(e) => logger.error(s"Error processing uri '$file'", e); CloseableIterator.empty
51       }
52       transformed(filtered(iter))
53     } else {
54       CloseableIterator.empty
55     }
56   }
57 
58   private def filtered(in: CloseableIterator[SimpleFeature]): CloseableIterator[SimpleFeature] = {
59     filter match {
60       case None => in
61       case Some(f) => in.filter(f.evaluate)
62     }
63   }
64 
65   private def transformed(in: CloseableIterator[SimpleFeature]): CloseableIterator[SimpleFeature] = {
66     transform match {
67       case None => in
68       case Some((tdefs, tsft)) =>
69         val feature = TransformSimpleFeature(converter.targetSft, tsft, tdefs)
70         in.map(f => ScalaSimpleFeature.copy(feature.setFeature(f)))
71     }
72   }
