32   private val truncateToPartitionStart: ZonedDateTime => ZonedDateTime = {
33     // note: `truncatedTo` is only valid up to DAYS, other units require additional steps
34     val truncate: ZonedDateTime => ZonedDateTime = stepUnit match {
35       case NANOS | MICROS | MILLIS | SECONDS | MINUTES | HOURS | DAYS =>
36         dt => dt.truncatedTo(stepUnit)
37 
38       case WEEKS =>
39         val adjuster = TemporalAdjusters.previousOrSame(DayOfWeek.MONDAY) // note: java day_of_week defines monday as the first day
40         dt => dt.`with`(adjuster).truncatedTo(DAYS)
41 
42       case MONTHS =>
43         val adjuster = TemporalAdjusters.firstDayOfMonth()
44         dt => dt.`with`(adjuster).truncatedTo(DAYS)
45 
46       case YEARS =>
47         val adjuster = TemporalAdjusters.firstDayOfYear()
48         dt => dt.`with`(adjuster).truncatedTo(DAYS)
49 
50       case _ =>
51         throw new IllegalArgumentException(s"${DateTimeScheme.Config.StepUnitOpt} $stepUnit is not supported")
52     }
53     if (step == 1) {
54       truncate
55     } else {
56       val chronoField = stepUnit match {
57         case NANOS   => ChronoField.NANO_OF_SECOND
58         case MICROS  => ChronoField.MICRO_OF_SECOND
59         case MILLIS  => ChronoField.MILLI_OF_SECOND
60         case SECONDS => ChronoField.SECOND_OF_MINUTE
61         case MINUTES => ChronoField.MINUTE_OF_HOUR
62         case HOURS   => ChronoField.HOUR_OF_DAY
63         case DAYS    => ChronoField.DAY_OF_YEAR
64         case WEEKS   => WeekFields.ISO.weekOfWeekBasedYear()
65         case MONTHS  => ChronoField.MONTH_OF_YEAR
66         case YEARS   => ChronoField.YEAR
67         case _ => throw new IllegalArgumentException(s"${DateTimeScheme.Config.StepUnitOpt} $stepUnit is not supported")
68       }
69       val min = math.max(chronoField.range().getMinimum, 0).toInt // account for 1-based fields like {month, week, day}_of_year
70       truncate.andThen { dt =>
71         val steps = dt.get(chronoField) - min
72         val remainder = steps % step
73         if (remainder == 0) { dt } else { dt.`with`(chronoField, steps + min - remainder ) }
74       }
75     }
76   }

85   private def getCoveringPartitions(filter: Filter): Option[(Seq[String], Seq[String])] = {
86     val bounds = FilterHelper.extractIntervals(filter, dtg, handleExclusiveBounds = false)
87     if (bounds.disjoint) {
88       Some((Seq.empty, Seq.empty))
89     } else if (bounds.isEmpty || !bounds.forall(_.isBoundedBothSides)) {
90       None
91     } else {
92       // there should be no duplicates in covered partitions, as our bounds will not overlap,
93       // but there may be multiple partial intersects with a given partition
94       val covered = ListBuffer.empty[String]
95       val intersecting = ListBuffer.empty[String]
96 
97       bounds.values.foreach { bound =>
98         // note: we verified both sides are bounded above
99         val lower = bound.lower.value.get
100         val upper = bound.upper.value.get
101         val start = truncateToPartitionStart(lower)
102         val end = truncateToPartitionStart(upper)
103 
104         // do our endpoints match the partition boundary, or do we need to apply a filter to the first/last partition?
105         val lowerBoundCovered = bound.lower.inclusive && lower == start
106 
107         // `stepUnit.between` claims to be upper endpoint exclusive, but doesn't seem to be...
108         val steps = stepUnit.between(start, end).toInt
109         if (steps < step) {
110           if (lowerBoundCovered &&
111               ((bound.upper.exclusive && upper == end) || (bound.upper.inclusive && upper == end.plus(step, stepUnit).minus(1, MILLIS)))) {
112             covered += formatter.format(start)
113           } else {
114             intersecting += formatter.format(start)
115           }
116         } else {
117           if (lowerBoundCovered) {
118             covered += formatter.format(start)
119           } else {
120             intersecting += formatter.format(start)
121           }
122           covered ++= Iterator.iterate(start)(_.plus(step, stepUnit)).drop(1).takeWhile(_.isBefore(end)).map(formatter.format)
123           if (bound.upper.inclusive || upper != end) {
124             intersecting += formatter.format(end)
125           }
126         }
127       }
128 
129       Some((covered.toSeq, intersecting.distinct.toSeq))
130     }
131   }

166     val Monthly      : Format = Format("monthly",       "yyyy/MM",          ChronoUnit.MONTHS )
167     val JulianMinute : Format = Format("julian-minute", "yyyy/DDD/HH/mm",   ChronoUnit.MINUTES)
168     val JulianHourly : Format = Format("julian-hourly", "yyyy/DDD/HH",      ChronoUnit.HOURS  )
169     val JulianDaily  : Format = Format("julian-daily",  "yyyy/DDD",         ChronoUnit.DAYS   )
170 
171     // java.time doesn't seem to have a way to parse back out a year/week format...
172     // to get around that we have to define the default day of week in the formatter
173     val Weekly: Format = {
174       val formatter =
175         new DateTimeFormatterBuilder()
176           .appendValue(WeekFields.ISO.weekBasedYear(), 4)
177           .appendLiteral("/W")
178           .appendValue(WeekFields.ISO.weekOfWeekBasedYear(), 2)
179           .parseDefaulting(ChronoField.DAY_OF_WEEK, 1)
180           .toFormatter()
181       Format("weekly", formatter, "YYYY/'W'ww", ChronoUnit.WEEKS)
182     }
183 
184     private val all = Seq(Minute, Hourly, Daily, Weekly, Monthly, JulianMinute, JulianHourly, JulianDaily)
185   }

201           throw new IllegalArgumentException(s"DateTime scheme requires valid unit '${Config.StepUnitOpt}'")
202         }

204           throw new IllegalArgumentException(s"DateTime scheme requires valid format '${Config.DateTimeFormatOpt}'"))
205         require(!format.endsWith("/"), "Format cannot end with a slash")
206         val formatter = try { DateTimeFormatter.ofPattern(format) } catch {
207           case NonFatal(e) => throw new IllegalArgumentException(s"Invalid date format '$format':", e)
208         }
209         Some(DateTimeScheme(formatter, format, unit, step, dtg, dtgIndex))
210       } else {
211         Formats(config.name).map(f => DateTimeScheme(f.formatter, f.pattern, f.unit, step, dtg, dtgIndex))
212       }
213     }
