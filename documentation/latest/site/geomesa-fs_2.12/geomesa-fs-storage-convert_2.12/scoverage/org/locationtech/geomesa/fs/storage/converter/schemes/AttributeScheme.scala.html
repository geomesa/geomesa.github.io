16 /**
17   * Lexicoded attribute partitioning
18   *
19   * @param attribute attribute name
20   * @param index attribute index in the sft
21   * @param binding type binding
22   * @param allowedValues list of allowedValues to partition
23   */
24 case class AttributeScheme(attribute: String, index: Int, binding: Class[_], defaultPartition: String, allowedValues: Seq[String])
25     extends PartitionScheme {
26 
27   private val alias = AttributeIndexKey.alias(binding)
28 
29   override val depth: Int = 1
30 
31   override def pattern: String = s"<$attribute>"
32 
33   override def getIntersectingPartitions(filter: Filter): Option[Seq[String]] = {
34     val bounds = FilterHelper.extractAttributeBounds(filter, attribute, binding)
35     if (bounds.disjoint) {
36       Some(Seq.empty)
37     } else if (bounds.isEmpty || !bounds.precise || !bounds.forall(_.isEquals)) {
38       if (allowedValues.nonEmpty) Some(allowedValues) else None
39     } else {
40       // note: we verified they are all single values above
41       val partitions = bounds.values.map(bound => AttributeIndexKey.encodeForQuery(bound.lower.value.get, binding))
42       if (allowedValues.nonEmpty) Some(partitions.filter(partition => allowedValues.contains(partition))) else Some(partitions)
43     }
44   }
45 }
46 
47 object AttributeScheme {
48 
49   val Name = "attribute"
50 
51   /**
52     * Check to extract only the equality filters that we can process with this partition scheme
53     *
54     * @param filter filter
55     * @return
56     */
57   def propertyIsEquals(filter: Filter): Boolean = {
58     filter match {
59       case _: And | _: Or => true // note: implies further processing of children
60       case _: PropertyIsEqualTo => true
61       case _ => false
62     }
63   }
64 
65   object Config {
66     val AttributeOpt: String = "partitioned-attribute"
67     val AllowedListOpt: String = "allow-list"
68     val DefaultPartitionOpt: String = "default-partition"
69   }
70 
71   object AttributePartitionSchemeFactory extends PartitionSchemeFactory {
72     override def load(sft: SimpleFeatureType, config: NamedOptions): Option[PartitionScheme] = {
73       if (config.name != Name) { None } else {
74         val attribute = config.options.getOrElse(Config.AttributeOpt, null)
75         require(attribute != null, s"Attribute scheme requires valid attribute name '${Config.AttributeOpt}'")
76         val index = sft.indexOf(attribute)
77         require(index != -1, s"Attribute '$attribute' does not exist in schema '${sft.getTypeName}'")
78         val binding = sft.getDescriptor(index).getType.getBinding
79         require(AttributeIndexKey.encodable(binding),
80           s"Invalid type binding '${binding.getName}' of attribute '$attribute'")
81         val allowedValues: Seq[String] = config.options.get(Config.AllowedListOpt).map(_.split(',').toSeq).getOrElse(Seq.empty).map(allowed => AttributeIndexKey.encodeForQuery(allowed.trim(),binding))
82         val defaultPartition = AttributeIndexKey.encodeForQuery(config.options.getOrElse(Config.DefaultPartitionOpt, allowedValues.headOption.getOrElse("")), binding)
83         require(allowedValues.isEmpty || allowedValues.contains(defaultPartition), "Default partition must be one of the allowed values")
84         Some(AttributeScheme(attribute, index, binding, defaultPartition, allowedValues))
85       }
86     }
87   }
