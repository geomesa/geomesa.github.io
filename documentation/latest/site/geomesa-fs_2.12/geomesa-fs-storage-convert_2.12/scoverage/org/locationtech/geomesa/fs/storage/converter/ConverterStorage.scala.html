54     Caffeine.newBuilder().refreshAfterWrite(CacheDurationProperty.toDuration.get.toMillis, TimeUnit.MILLISECONDS).build(

60   override def getReader(query: Query, threads: Int): CloseableIterator[SimpleFeature] = {
61     import org.locationtech.geomesa.index.conf.QueryHints.RichHints
62 
63     val configured = QueryRunner.configureQuery(sft, query)
64     val filter = Option(configured.getFilter)
65     val visFilter = VisibilityUtils.visible(authProvider)
66     val transform = configured.getHints.getTransform
67     val sort = configured.getHints.getSortFields
68     val max = configured.getHints.getMaxFeatures
69 
70     logger.debug(s"Running query '${query.getTypeName}' ${ECQL.toCQL(filter.getOrElse(Filter.INCLUDE))}")
71     logger.debug(s"  Original filter: ${ECQL.toCQL(query.getFilter)}")
72     logger.debug(s"  Transforms: ${transform.fold("none") { case (t, _) => if (t.isEmpty) { "empty" } else { t }}}")
73     logger.debug(s"  Sort: ${sort.fold("none") { fields => fields.map { case (f, rev) => s"$f ${if (rev) "descending" else ""}"}.mkString(", ")}}")
74     logger.debug(s"  Max features: ${max.getOrElse("none")}")
75 
76     val files = filter.flatMap(layout.getIntersectingPartitions) match {
77       case None => filesCache.get(BoxedUnit.UNIT).flatMap(_._2)
78       case Some(partitions) => filesCache.get(BoxedUnit.UNIT).flatMap { case (partition, uris) =>
79         if (partitions.contains(partition)) { uris } else { Seq.empty }
80       }
81     }
82 
83     val reader = new ConverterFileSystemReader(fs, converter, filter, transform, pathFiltering)
84     val scan = FileSystemThreadedReader(reader, files.toSeq, threads)
85 
86     try {
87       val iter = scan.filter(visFilter.apply)
88       val sorted = sort.fold(iter)(s => new SortingSimpleFeatureIterator(iter, s))
89       val limited = max.fold(sorted)(m => sorted.take(m))
90       limited
91     } catch {
92       case NonFatal(e) => CloseWithLogging(scan); throw e
93     }
94   }

108   private def buildFileList(): Map[String, Seq[URI]] = {
109     logger.debug("Building file list")
110     val start = System.currentTimeMillis()
111     val map = new java.util.HashMap[String, Seq[URI]]()
112     try {
113       val toCheck = new java.util.LinkedList[URI]()
114       fs.list(rootPath).foreach(toCheck.add)
115       val maxDepth = depth + rootPath.toString.count(_ == '/')
116       val rootPathLength = rootPath.toString.length
117       while (!toCheck.isEmpty) {
118         val file = toCheck.remove()
119         logger.debug(s"Checking ${file.getPath}")
120         if (file.getPath.endsWith("/")) {
121           if (file.toString.count(_ == '/') <= maxDepth) {
122             fs.list(file).foreach(toCheck.add)
123           }
124         } else if (!fs.filename(file).startsWith(".")) { // ignore "hidden" files
125           val relativePath = file.toString.substring(rootPathLength)
126           var partitionParts = relativePath.split("/")
127           if (leafStorage) {
128             partitionParts(partitionParts.length - 1) = partitionParts.last.takeWhile(_ != '_')
129           } else {
130             partitionParts = partitionParts.dropRight(1)
131           }
132           val partition = partitionParts.mkString("/")
133           map.put(partition, map.getOrDefault(partition, Seq.empty) :+ file)
134         }
135       }
136     } catch {
137       case NonFatal(e) => logger.error("Error building partition list:", e); throw e
138     }
139     logger.debug(s"Found ${map.size} partitions in ${System.currentTimeMillis() - start}ms")
140     logger.whenTraceEnabled(map.asScala.values.foreach(uris => uris.foreach(uri => logger.trace(uri.toString))))
141     map.asScala.toMap
142   }
