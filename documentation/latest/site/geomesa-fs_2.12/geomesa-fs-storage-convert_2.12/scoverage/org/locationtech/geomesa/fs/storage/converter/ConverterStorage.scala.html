71     logger.debug(s"Running query '${query.getTypeName}' ${ECQL.toCQL(filter.getOrElse(Filter.INCLUDE))}")

74     logger.debug(s"  Sort: ${sort.fold("none") { fields => fields.map { case (f, rev) => s"$f ${if (rev) "descending" else ""}"}.mkString(", ")}}")

127   private def buildFileList(): Map[String, Seq[URI]] = {
128     logger.debug("Building file list")
129     val start = System.currentTimeMillis()
130     val map = new java.util.HashMap[String, Seq[URI]]()
131     try {
132       val toCheck = new java.util.LinkedList[URI]()
133       fs.list(rootPath).foreach(toCheck.add)
134       val maxDepth = depth + rootPath.toString.count(_ == '/')
135       val rootPathLength = rootPath.toString.length
136       while (!toCheck.isEmpty) {
137         val file = toCheck.remove()
138         logger.debug(s"Checking ${file.getPath}")
139         if (file.getPath.endsWith("/")) {
140           if (file.toString.count(_ == '/') <= maxDepth) {
141             fs.list(file).foreach(toCheck.add)
142           }
143         } else if (!fs.filename(file).startsWith(".")) { // ignore "hidden" files
144           val relativePath = file.toString.substring(rootPathLength)
145           var partitionParts = relativePath.split("/")
146           if (leafStorage) {
147             partitionParts(partitionParts.length - 1) = partitionParts.last.takeWhile(_ != '_')
148           } else {
149             partitionParts = partitionParts.dropRight(1)
150           }
151           val partition = partitionParts.mkString("/")
152           map.put(partition, map.getOrDefault(partition, Seq.empty) :+ file)
153         }
154       }
155     } catch {
156       case NonFatal(e) => logger.error("Error building partition list:", e); throw e
157     }
158     logger.debug(s"Found ${map.size} partitions in ${System.currentTimeMillis() - start}ms")
159     logger.whenTraceEnabled(map.asScala.values.foreach(uris => uris.foreach(uri => logger.trace(uri.toString))))
160     map.asScala.toMap
161   }
