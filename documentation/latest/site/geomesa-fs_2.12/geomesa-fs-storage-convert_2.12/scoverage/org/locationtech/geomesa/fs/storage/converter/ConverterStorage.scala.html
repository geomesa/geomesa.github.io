48   private val fs = ObjectStore(context)

50   private val depth = layout.depth - (if (leafStorage) { 1 } else { 0 })

52   private val filesCache: LoadingCache[BoxedUnit, Map[String, Seq[URI]]] =
53     Caffeine.newBuilder().refreshAfterWrite(CacheDurationProperty.toDuration.get.toMillis, TimeUnit.MILLISECONDS).build(
54       new CacheLoader[BoxedUnit, Map[String, Seq[URI]]]() {
55         override def load(key: BoxedUnit): Map[String, Seq[URI]] = buildFileList()

59   override def getReader(query: Query, threads: Int): CloseableIterator[SimpleFeature] = {
60     import org.locationtech.geomesa.index.conf.QueryHints.RichHints
61 
62     val configured = QueryRunner.configureQuery(sft, query)
63     val filter = Option(configured.getFilter)
64     val visFilter = VisibilityUtils.visible(authProvider)
65     val transform = configured.getHints.getTransform
66     val sort = configured.getHints.getSortFields
67     val max = configured.getHints.getMaxFeatures
68 
69     logger.debug(s"Running query '${query.getTypeName}' ${ECQL.toCQL(filter.getOrElse(Filter.INCLUDE))}")
70     logger.debug(s"  Original filter: ${ECQL.toCQL(query.getFilter)}")
71     logger.debug(s"  Transforms: ${transform.fold("none") { case (t, _) => if (t.isEmpty) { "empty" } else { t }}}")
72     logger.debug(s"  Sort: ${sort.fold("none") { fields => fields.map { case (f, rev) => s"$f ${if (rev) "descending" else ""}"}.mkString(", ")}}")
73     logger.debug(s"  Max features: ${max.getOrElse("none")}")
74 
75     val files = filter.flatMap(layout.getIntersectingPartitions) match {
76       case None => filesCache.get(BoxedUnit.UNIT).flatMap(_._2)
77       case Some(partitions) => filesCache.get(BoxedUnit.UNIT).flatMap { case (partition, uris) =>
78         if (partitions.contains(partition)) { uris } else { Seq.empty }
79       }
80     }
81 
82     val reader = new ConverterFileSystemReader(fs, context.root, converter, filter, transform, pathFiltering)
83     val scan = FileSystemThreadedReader(reader, files.toSeq, threads)
84 
85     try {
86       val iter = scan.filter(visFilter.apply)
87       val sorted = sort.fold(iter)(s => new SortingSimpleFeatureIterator(iter, s))
88       val limited = max.fold(sorted)(m => sorted.take(m))
89       limited
90     } catch {
91       case NonFatal(e) => CloseWithLogging(scan); throw e
92     }
93   }

104     }

107   private def buildFileList(): Map[String, Seq[URI]] = {
108     logger.debug("Building file list")
109     val start = System.currentTimeMillis()
110     val map = new java.util.HashMap[String, Seq[URI]]()
111     try {
112       val toCheck = new java.util.LinkedList[URI]()
113       fs.list(context.root).foreach(toCheck.add)
114       val maxDepth = depth + context.root.toString.count(_ == '/')
115       val rootPathLength = context.root.toString.length
116       while (!toCheck.isEmpty) {
117         val file = toCheck.remove()
118         logger.debug(s"Checking ${file.getPath}")
119         if (file.getPath.endsWith("/")) {
120           if (file.toString.count(_ == '/') <= maxDepth) {
121             fs.list(file).foreach(toCheck.add)
122           }
123         } else if (!fs.filename(file).startsWith(".")) { // ignore "hidden" files
124           val relativePath = file.toString.substring(rootPathLength)
125           var partitionParts = relativePath.split("/")
126           if (leafStorage) {
127             partitionParts(partitionParts.length - 1) = partitionParts.last.takeWhile(_ != '_')
128           } else {
129             partitionParts = partitionParts.dropRight(1)
130           }
131           val partition = partitionParts.mkString("/")
132           map.put(partition, map.getOrDefault(partition, Seq.empty) :+ file)
133         }
134       }
135     } catch {
136       case NonFatal(e) => logger.error("Error building partition list:", e); throw e
137     }
138     logger.debug(s"Found ${map.size} partitions in ${System.currentTimeMillis() - start}ms")
139     logger.whenTraceEnabled(map.asScala.values.foreach(uris => uris.foreach(uri => logger.trace(uri.toString))))
140     map.asScala.toMap
141   }
