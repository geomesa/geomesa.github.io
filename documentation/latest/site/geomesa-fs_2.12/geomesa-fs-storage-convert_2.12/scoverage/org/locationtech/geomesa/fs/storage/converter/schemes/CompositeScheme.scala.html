17 case class CompositeScheme(schemes: Seq[PartitionScheme]) extends PartitionScheme {
18 
19   require(schemes.lengthCompare(1) > 0, "Must provide at least 2 schemes for a composite scheme")
20 
21   override val depth: Int = schemes.map(_.depth).sum
22 
23   override def pattern: String = schemes.map(_.pattern).mkString("/")
24 
25   override def getIntersectingPartitions(filter: Filter): Option[Seq[String]] = {
26     val head = schemes.head.getIntersectingPartitions(filter)
27     schemes.tail.foldLeft(head) { case (result, scheme) =>
28       for { res <- result; partitions <- scheme.getIntersectingPartitions(filter) } yield {
29         res.flatMap(r => partitions.map(p => s"$r/$p"))
30       }
31     }
32   }
33 }

40     override def load(sft: SimpleFeatureType, config: NamedOptions): Option[PartitionScheme] = {
41       if (!config.name.contains(SchemeSeparator)) { None } else {
42         val configs = config.name.split(SchemeSeparator).map(n => config.copy(name = n))
43         val composites = configs.map { c =>
44           try { PartitionSchemeFactory.load(sft, c) } catch {
45             case NonFatal(e) =>
46               throw new IllegalArgumentException(s"Error creating composite scheme '${c.name}' (from '${config.name}'):", e)
47           }
48         }
49         Some(CompositeScheme(composites))
50       }
51     }
52   }
