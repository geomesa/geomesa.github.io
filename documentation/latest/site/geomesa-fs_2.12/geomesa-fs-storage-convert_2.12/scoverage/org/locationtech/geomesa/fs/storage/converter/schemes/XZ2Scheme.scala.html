15 case class XZ2Scheme(bits: Int, geom: String, geomIndex: Int) extends SpatialScheme(bits, geom) {
16 
17   private val xz2 = XZ2SFC((bits / 2).asInstanceOf[Short])
18 
19   override def pattern: String = s"$bits-bit-xz2"
20 
21   // the max XZ2 value is (4^((bits / 2) + 1) - 1) / 3
22   // this calculates the number of digits in that value
23   override protected def digits(bits: Int): Int = math.ceil(((bits / 2) + 1) * math.log10(4) - math.log10(3)).toInt
24 
25   override protected def generateRanges(xy: Seq[(Double, Double, Double, Double)]): Seq[IndexRange] = xz2.ranges(xy)
