19 abstract class SpatialScheme(bits: Int, geom: String) extends PartitionScheme {
20 
21   require(bits % 2 == 0, "Resolution must be an even number")
22 
23   protected val format = s"%0${digits(bits)}d"
24 
25   protected def digits(bits: Int): Int
26 
27   protected def generateRanges(xy: Seq[(Double, Double, Double, Double)]): Seq[IndexRange]
28 
29   override val depth: Int = 1
30 
31   override def getIntersectingPartitions(filter: Filter): Option[Seq[String]] =
32     getIntersectingPartitions(filter, None)
33 
34   private def getIntersectingPartitions(filter: Filter, partition: Option[String]): Option[Seq[String]] = {
35     val geometries = FilterHelper.extractGeometries(filter, geom, intersect = true)
36     if (geometries.disjoint) {
37       Some(Seq.empty)
38     } else if (geometries.values.isEmpty) {
39       None
40     } else {
41       val partitions = partition.map(Seq(_)).getOrElse {
42         // there should be few enough partitions that we can enumerate them here and not exactly match the filter...
43         val ranges = generateRanges(geometries.values.map(GeometryUtils.bounds))
44         ranges.flatMap(r => r.lower to r.upper).distinct.map(_.formatted(format))
45       }
46       Some(partitions)
47     }
48   }
49 }

64     override def load(sft: SimpleFeatureType, config: NamedOptions): Option[PartitionScheme] = {
65       import org.locationtech.geomesa.utils.geotools.RichSimpleFeatureType.RichSimpleFeatureType
66 
67       val matcher = namePattern.matcher(config.name)
68       if (!matcher.matches()) { None } else {
69         val geom = config.options.getOrElse(Config.GeomAttribute, sft.getGeomField)
70         val geomIndex = sft.indexOf(geom)
71         if (geomIndex == -1) {
72           throw new IllegalArgumentException(s"$name scheme requires valid geometry field '${Config.GeomAttribute}'")
73         }
74         val res =
75           Option(matcher.group(2))
76               .filterNot(_.isEmpty)
77               .orElse(config.options.get(resolution))
78               .map(Integer.parseInt)
79               .getOrElse(throw new IllegalArgumentException(s"$name scheme requires bit resolution '$resolution'"))
80         Some(buildPartitionScheme(res, geom, geomIndex))
81       }
82     }

85   }
