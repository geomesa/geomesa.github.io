30 class ConverterCatalog(val conf: Map[String, String]) extends StorageCatalog with LazyLogging {
31 
32   import ConverterCatalog._
33 
34   private val sft = {
35     val sftArg =
36       conf.get(SftConfigParam)
37         .orElse(conf.get(SftNameParam))
38         .getOrElse(throw new IllegalArgumentException(s"Must provide either simple feature type config or name"))
39     SftArgResolver.getArg(SftArgs(sftArg, null)) match {
40       case Left(e) => throw new IllegalArgumentException("Could not load SimpleFeatureType with provided parameters", e)
41       case Right(schema) => schema
42     }
43   }
44 
45   private val schemes = {
46     val partitionSchemeName =
47       conf.getOrElse(PartitionSchemeParam, throw new IllegalArgumentException("Must provide partition scheme name"))
48     val partitionSchemeOpts =
49         conf.collect { case (k, v) if k.startsWith(PartitionOptsPrefix) => k.substring(PartitionOptsPrefix.length) -> v }
50     PartitionSchemeFactory.load(sft, NamedOptions(partitionSchemeName, partitionSchemeOpts))
51   }
52 
53   private val converterPath = {
54     val path = conf.get(ConverterPathParam).map(URI.create).getOrElse(throw new IllegalArgumentException("Must provide converter path"))
55     if (path.toString.endsWith("/")) { path } else { new URI(path.toString + "/") }
56   }
57 
58   private val leafStorage =
59     conf.get(LeafStorageParam)
60       .map(_.toBoolean)
61       .getOrElse {
62         val deprecated = s"${PartitionOptsPrefix}leaf-storage"
63         conf.get(deprecated).fold(true) { s =>
64           logger.warn(s"Using deprecated leaf-storage partition-scheme option. Please define leaf-storage using '$LeafStorageParam'")
65           s.toBoolean
66         }
67       }
68 
69   private val pathFiltering = conf.get(PathFilterName).flatMap { name =>
70     val pathFilteringOpts =
71       conf.collect { case (k, v) if k.startsWith(PathFilterOptsPrefix) => k.substring(PathFilterOptsPrefix.length) -> v }
72     val factory = PathFilteringFactory.load(NamedOptions(name, pathFilteringOpts))
73     if (factory.isEmpty) {
74       throw new IllegalArgumentException(s"Failed to load ${classOf[PathFiltering].getName} for config '$name'")
75     }
76     factory
77   }
78 
79   override def getTypeNames: Seq[String] = Seq(sft.getTypeName)
80 
81   override def load(typeName: String): FileSystemStorage = {
82     if (typeName == sft.getTypeName) {
83       val convertArg = conf.get(ConverterConfigParam).orElse(conf.get(ConverterNameParam)).getOrElse {
84         throw new IllegalArgumentException(s"Must provide either converter config or name")
85       }
86       val converterConfig = ConverterConfigResolver.getArg(ConfArgs(convertArg)) match {
87         case Left(e) => throw new IllegalArgumentException("Could not load Converter with provided parameters", e)
88         case Right(c) => c
89       }
90       val converter = SimpleFeatureConverter(sft, converterConfig)
91       val catalog = new InMemoryCatalog()
92       catalog.initialize("geomesa", java.util.Map.of())
93       val ns = Namespace.of("geomesa")
94       catalog.createNamespace(ns)
95       val schema = SimpleFeatureIcebergSchema(sft, conf)
96       val table = catalog.createTable(TableIdentifier.of(ns, "converter"), schema.schema)
97       new ConverterStorage(table, schema, schemes, conf, converterPath, converter, pathFiltering, leafStorage)
98     } else {
99       throw new IllegalArgumentException(s"Schema '$typeName' doesn't exist - available schemas: ${sft.getTypeName}")
100     }
101   }
102 
103   override def create(sft: SimpleFeatureType, partitions: Seq[String], targetFileSize: Option[Long]): FileSystemStorage =
104     throw new UnsupportedOperationException("Converter storage is read-only")
105 
106   override def close(): Unit = {}
107 }
