24 /**
25  * Storage catalog for synthetic "converter" storage
26  *
27  * @param context file system context
28  */
29 class ConverterCatalog(val context: FileSystemContext) extends StorageCatalog with LazyLogging {
30 
31   import ConverterCatalog._
32 
33   private val sft = {
34     val sftArg =
35       context.conf.get(SftConfigParam)
36         .orElse(context.conf.get(SftNameParam))
37         .getOrElse(throw new IllegalArgumentException(s"Must provide either simple feature type config or name"))
38     SftArgResolver.getArg(SftArgs(sftArg, null)) match {
39       case Left(e) => throw new IllegalArgumentException("Could not load SimpleFeatureType with provided parameters", e)
40       case Right(schema) => schema
41     }
42   }
43 
44   private val schemes = {
45     val partitionSchemeName =
46       context.conf.getOrElse(PartitionSchemeParam, throw new IllegalArgumentException("Must provide partition scheme name"))
47     val partitionSchemeOpts =
48         context.conf.collect { case (k, v) if k.startsWith(PartitionOptsPrefix) => k.substring(PartitionOptsPrefix.length) -> v }
49     PartitionSchemeFactory.load(sft, NamedOptions(partitionSchemeName, partitionSchemeOpts))
50   }
51 
52   private val converterPath =
53     context.conf.get(ConverterPathParam)
54       .map(p => context.root.resolve(p))
55       .getOrElse(throw new IllegalArgumentException("Must provide converter path"))
56 
57   private val leafStorage =
58     context.conf.get(LeafStorageParam)
59       .map(_.toBoolean)
60       .getOrElse {
61         val deprecated = s"${PartitionOptsPrefix}leaf-storage"
62         context.conf.get(deprecated).fold(true) { s =>
63           logger.warn(s"Using deprecated leaf-storage partition-scheme option. Please define leaf-storage using '$LeafStorageParam'")
64           s.toBoolean
65         }
66       }
67 
68   private val pathFiltering = context.conf.get(PathFilterName).flatMap { name =>
69     val pathFilteringOpts =
70       context.conf.collect { case (k, v) if k.startsWith(PathFilterOptsPrefix) => k.substring(PathFilterOptsPrefix.length) -> v }
71     val factory = PathFilteringFactory.load(NamedOptions(name, pathFilteringOpts))
72     if (factory.isEmpty) {
73       throw new IllegalArgumentException(s"Failed to load ${classOf[PathFiltering].getName} for config '$name'")
74     }
75     factory
76   }
77 
78   override def getTypeNames: Seq[String] = Seq(sft.getTypeName)
79 
80   override def load(typeName: String): FileSystemStorage = {
81     if (typeName == sft.getTypeName) {
82       val convertArg = context.conf.get(ConverterConfigParam).orElse(context.conf.get(ConverterNameParam)).getOrElse {
83         throw new IllegalArgumentException(s"Must provide either converter config or name")
84       }
85       val converterConfig = ConverterConfigResolver.getArg(ConfArgs(convertArg)) match {
86         case Left(e) => throw new IllegalArgumentException("Could not load Converter with provided parameters", e)
87         case Right(c) => c
88       }
89       val converter = SimpleFeatureConverter(sft, converterConfig)
90       val catalog = new InMemoryCatalog()
91       catalog.initialize("geomesa", java.util.Map.of())
92       val ns = Namespace.of("geomesa")
93       catalog.createNamespace(ns)
94       val schema = SimpleFeatureIcebergSchema(sft, context.conf)
95       val table = catalog.createTable(TableIdentifier.of(ns, "converter"), schema.schema)
96       val newContext = FileSystemContext.create(converterPath, context.conf, context.namespace)
97       new ConverterStorage(newContext, table, schema, schemes, converter, pathFiltering, leafStorage)
98     } else {
99       throw new IllegalArgumentException(s"Schema '$typeName' doesn't exist - available schemas: ${sft.getTypeName}")
100     }
101   }
102 
103   override def create(sft: SimpleFeatureType, partitions: Seq[String], targetFileSize: Option[Long]): FileSystemStorage =
104     throw new UnsupportedOperationException("Converter storage is read-only")
105 
106   override def close(): Unit = {}
107 }
