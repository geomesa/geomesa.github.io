31 class ConverterCatalog(val conf: Map[String, String]) extends StorageCatalog with LazyLogging {
32 
33   import ConverterCatalog._
34 
35   import scala.collection.JavaConverters._
36 
37   private val sft = {
38     val sftArg =
39       conf.get(SftConfigParam)
40         .orElse(conf.get(SftNameParam))
41         .getOrElse(throw new IllegalArgumentException(s"Must provide either simple feature type config or name"))
42     SftArgResolver.getArg(SftArgs(sftArg, null)) match {
43       case Left(e) => throw new IllegalArgumentException("Could not load SimpleFeatureType with provided parameters", e)
44       case Right(schema) => schema
45     }
46   }
47 
48   private val schemes = {
49     val partitionSchemeName =
50       conf.getOrElse(PartitionSchemeParam, throw new IllegalArgumentException("Must provide partition scheme name"))
51     val partitionSchemeOpts =
52         conf.collect { case (k, v) if k.startsWith(PartitionOptsPrefix) => k.substring(PartitionOptsPrefix.length) -> v }
53     PartitionSchemeFactory.load(sft, NamedOptions(partitionSchemeName, partitionSchemeOpts))
54   }
55 
56   private val converterPath = {
57     val path = conf.get(ConverterPathParam).map(URI.create).getOrElse(throw new IllegalArgumentException("Must provide converter path"))
58     if (path.toString.endsWith("/")) { path } else { new URI(path.toString + "/") }
59   }
60 
61   private val leafStorage =
62     conf.get(LeafStorageParam)
63       .map(_.toBoolean)
64       .getOrElse {
65         val deprecated = s"${PartitionOptsPrefix}leaf-storage"
66         conf.get(deprecated).fold(true) { s =>
67           logger.warn(s"Using deprecated leaf-storage partition-scheme option. Please define leaf-storage using '$LeafStorageParam'")
68           s.toBoolean
69         }
70       }
71 
72   private val pathFiltering = conf.get(PathFilterName).flatMap { name =>
73     val pathFilteringOpts =
74       conf.collect { case (k, v) if k.startsWith(PathFilterOptsPrefix) => k.substring(PathFilterOptsPrefix.length) -> v }
75     val factory = PathFilteringFactory.load(NamedOptions(name, pathFilteringOpts))
76     if (factory.isEmpty) {
77       throw new IllegalArgumentException(s"Failed to load ${classOf[PathFiltering].getName} for config '$name'")
78     }
79     factory
80   }
81 
82   override def getTypeNames: Seq[String] = Seq(sft.getTypeName)
83 
84   override def load(typeName: String): FileSystemStorage = {
85     if (typeName == sft.getTypeName) {
86       val convertArg = conf.get(ConverterConfigParam).orElse(conf.get(ConverterNameParam)).getOrElse {
87         throw new IllegalArgumentException(s"Must provide either converter config or name")
88       }
89       val converterConfig = ConverterConfigResolver.getArg(ConfArgs(convertArg)) match {
90         case Left(e) => throw new IllegalArgumentException("Could not load Converter with provided parameters", e)
91         case Right(c) => c
92       }
93       val converter = SimpleFeatureConverter(sft, converterConfig)
94       val catalog = new InMemoryCatalog()
95       catalog.initialize("geomesa", java.util.Map.of())
96       val ns = Namespace.of("geomesa")
97       catalog.createNamespace(ns)
98       val table = {
99         // file format v3 lets us use variant encoding, which isn't actually used but can be generated by the SimpleFeatureIcebergSchema
100         val tableProps = IcebergCatalog.sftTableProperties(sft) ++ Map(TableProperties.FORMAT_VERSION -> "3")
101         val tableId = TableIdentifier.of(ns, "converter")
102         val tableSchema = SimpleFeatureIcebergSchema.create(sft)
103         catalog.createTable(tableId, tableSchema, PartitionSpec.unpartitioned(), null, tableProps.asJava)
104       }
105       val schema = SimpleFeatureIcebergSchema(table, conf.get(StorageCatalog.NamespaceConfigKey))
106       new ConverterStorage(table, schema, schemes, conf, converterPath, converter, pathFiltering, leafStorage)
107     } else {
108       throw new IllegalArgumentException(s"Schema '$typeName' doesn't exist - available schemas: ${sft.getTypeName}")
109     }
110   }
111 
112   override def create(
113       sft: SimpleFeatureType,
114       partitions: Seq[String],
115       targetFileSize: Option[Long],
116       bloomFilters: Seq[BloomFilterConfig]): FileSystemStorage =
117     throw new UnsupportedOperationException("Converter storage is read-only")
118 
119   override def close(): Unit = {}
120 }
