108   def addBytes(schema: Schema): Schema = {
109     schema.getType match {
110       case Schema.Type.RECORD =>
111         val fields = new java.util.ArrayList[Schema.Field](schema.getFields.size() + 1)
112         schema.getFields.asScala.foreach { field =>
113           fields.add(new Schema.Field(field.name, field.schema, field.doc, field.defaultVal()))
114         }
115         fields.add(new Schema.Field(BytesField, Schema.create(Schema.Type.BYTES), "raw bytes", ""))
116 
117         val updated = Schema.createRecord(schema.getName, schema.getDoc, schema.getNamespace, schema.isError)
118         updated.setFields(fields)
119         updated
120 
121       case Schema.Type.UNION =>
122         Schema.createUnion(schema.getTypes.asScala.map(s => addBytes(s)).toSeq: _*)
123 
124       case _ =>
125         throw new UnsupportedOperationException(
126           s"Raw Avro bytes (i.e. $$0) is not implemented for schema type ${schema.getType}")
127     }

232   /**
233    * Reads avro records encoded as 'single objects' - see https://avro.apache.org/docs/1.11.4/specification/#single-object-encoding
234    *
235    * @param schemas schemas
236    */
237   private class SingleObjectIterator(pool: GenericObjectPool[GenericRecordIterator], schemas: SchemaStore)
238       extends GenericRecordIterator(pool) {
239 
240     // the schema fingerprint is stored as a long in little-endian order - and we don't read any other values from the buffer
241     protected var buffer: ByteBuffer = ByteBuffer.allocate(1024).order(java.nio.ByteOrder.LITTLE_ENDIAN)
242     private val readers = scala.collection.mutable.Map.empty[Long, GenericDatumReader[GenericRecord]]
243     private var decoder: BinaryDecoder = _
244     private var record: GenericRecord = _
245     private var ec: EvaluationContext = _
246     private var error: Throwable = _
247     private var hasRecord = false
248 
249     override def setInstance(is: InputStream, ec: EvaluationContext): Unit = {
250      try {
251        populateBuffer(is)
252        val fingerprint = readFingerprint()
253        val schema = schemas.findByFingerprint(fingerprint)
254        if (schema == null) {
255          throw new IllegalStateException(s"Schema not found for fingerprint: $fingerprint")
256        }
257        this.ec = ec
258        this.decoder = DecoderFactory.get.binaryDecoder(buffer.array(), buffer.position(), buffer.remaining(), decoder)
259        val reader = readers.getOrElseUpdate(fingerprint, newReader(schema))
260        this.record = reader.read(record, decoder)
261        this.hasRecord = record != null
262        this.error = null
263      } catch {
264        case NonFatal(e) => this.error = e
265      } finally {
266        is.close()
267      }
268     }
269 
270     override def hasNext: Boolean = {
271       if (error != null) {
272         throw error
273       } else {
274         hasRecord
275       }
276     }
277 
278     override def next(): GenericRecord = {
279       ec.line += 1
280       hasRecord = false
281       record
282     }
283 
284     protected def newReader(schema: Schema): GenericDatumReader[GenericRecord] =
285       new GenericDatumReader[GenericRecord](schema, schema)
286 
287     // read all bytes from the input stream into the buffer, expanding as necessary
288     private def populateBuffer(is: InputStream): Unit = {
289       buffer.clear()
290       var read = 0
291       var chunk = 0
292       while ({ chunk = is.read(buffer.array(), buffer.position(), buffer.remaining()); chunk != -1 }) {
293         read += chunk
294         buffer.position(read)
295         if (buffer.remaining() == 0) {
296           // expand the buffer by doubling its capacity
297           val newBuffer = ByteBuffer.allocate(buffer.capacity() * 2).order(java.nio.ByteOrder.LITTLE_ENDIAN)
298           buffer.flip()
299           newBuffer.put(buffer)
300           buffer = newBuffer
301         }
302       }
303       buffer.flip()
304     }
305 
306     private def readFingerprint(): Long = {
307       // validate buffer has at least 10 bytes (2-byte header + 8-byte fingerprint)
308       if (buffer.remaining() < 10) {
309         throw new IllegalArgumentException(
310           s"Invalid Avro single-object encoding: expected at least 10 bytes, got ${buffer.remaining()}")
311       }
312 
313       // check for Avro single-object magic bytes (C3 01)
314       val byte0 = buffer.get() & 0xFF
315       val byte1 = buffer.get() & 0xFF
316       if (byte0 != 0xC3 || byte1 != 0x01) {
317         throw new IllegalArgumentException(
318           f"Invalid Avro single-object encoding: expected magic bytes C3 01, got $byte0%02X $byte1%02X")
319       }
320 
321       // read the 8-byte fingerprint in little-endian order to match BinaryMessageEncoder
322       buffer.getLong()
323     }
324   }
