103     *
104     * @param schema schema

107   def addBytes(schema: Schema): Schema = {
108     schema.getType match {
109       case Schema.Type.RECORD =>
110         val fields = new java.util.ArrayList[Schema.Field](schema.getFields.size() + 1)
111         schema.getFields.asScala.foreach { field =>
112           fields.add(new Schema.Field(field.name, field.schema, field.doc, field.defaultVal()))
113         }
114         fields.add(new Schema.Field(BytesField, Schema.create(Schema.Type.BYTES), "raw bytes", ""))
115 
116         val updated = Schema.createRecord(schema.getName, schema.getDoc, schema.getNamespace, schema.isError)
117         updated.setFields(fields)
118         updated
119 
120       case Schema.Type.UNION =>
121         Schema.createUnion(schema.getTypes.asScala.map(s => addBytes(s)).toSeq: _*)
122 
123       case _ =>
124         throw new UnsupportedOperationException(
125           s"Raw Avro bytes (i.e. $$0) is not implemented for schema type ${schema.getType}")
126     }
127   }

223   /**
224    * Reads avro records encoded as 'single objects' - see https://avro.apache.org/docs/1.11.4/specification/#single-object-encoding
225    *
226    * @param schemas schemas
227    */
228   private class SingleObjectIterator(pool: GenericObjectPool[GenericRecordIterator], schemas: SchemaStore)
229       extends GenericRecordIterator(pool) {
230 
231     // the schema fingerprint is stored as a long in little-endian order - and we don't read any other values from the buffer
232     protected var buffer: ByteBuffer = ByteBuffer.allocate(1024).order(java.nio.ByteOrder.LITTLE_ENDIAN)
233     private val readers = scala.collection.mutable.Map.empty[Long, GenericDatumReader[GenericRecord]]
234     private var decoder: BinaryDecoder = _
235     private var record: GenericRecord = _
236     private var ec: EvaluationContext = _
237     private var error: Throwable = _
238     private var hasRecord = false
239 
240     override def setInstance(is: InputStream, ec: EvaluationContext): Unit = {
241      try {
242        populateBuffer(is)
243        val fingerprint = readFingerprint()
244        val schema = schemas.findByFingerprint(fingerprint)
245        if (schema == null) {
246          throw new IllegalStateException(s"Schema not found for fingerprint: $fingerprint")
247        }
248        this.ec = ec
249        this.decoder = DecoderFactory.get.binaryDecoder(buffer.array(), buffer.position(), buffer.remaining(), decoder)
250        val reader = readers.getOrElseUpdate(fingerprint, newReader(schema))
251        this.record = reader.read(record, decoder)
252        this.hasRecord = record != null
253        this.error = null
254      } catch {
255        case NonFatal(e) => this.error = e
256      } finally {
257        is.close()
258      }
259     }
260 
261     override def hasNext: Boolean = {
262       if (error != null) {
263         throw error
264       } else {
265         hasRecord
266       }
267     }
268 
269     override def next(): GenericRecord = {
270       ec.line += 1
271       hasRecord = false
272       record
273     }
274 
275     protected def newReader(schema: Schema): GenericDatumReader[GenericRecord] =
276       new GenericDatumReader[GenericRecord](schema, schema)
277 
278     // read all bytes from the input stream into the buffer, expanding as necessary
279     private def populateBuffer(is: InputStream): Unit = {
280       buffer.clear()
281       var read = 0
282       var chunk = 0
283       while ({ chunk = is.read(buffer.array(), buffer.position(), buffer.remaining()); chunk != -1 }) {
284         read += chunk
285         buffer.position(read)
286         if (buffer.remaining() == 0) {
287           // expand the buffer by doubling its capacity
288           val newBuffer = ByteBuffer.allocate(buffer.capacity() * 2).order(java.nio.ByteOrder.LITTLE_ENDIAN)
289           buffer.flip()
290           newBuffer.put(buffer)
291           buffer = newBuffer
292         }
293       }
294       buffer.flip()
295     }
296 
297     private def readFingerprint(): Long = {
298       // validate buffer has at least 10 bytes (2-byte header + 8-byte fingerprint)
299       if (buffer.remaining() < 10) {
300         throw new IllegalArgumentException(
301           s"Invalid Avro single-object encoding: expected at least 10 bytes, got ${buffer.remaining()}")
302       }
303 
304       // check for Avro single-object magic bytes (C3 01)
305       val byte0 = buffer.get() & 0xFF
306       val byte1 = buffer.get() & 0xFF
307       if (byte0 != 0xC3 || byte1 != 0x01) {
308         throw new IllegalArgumentException(
309           f"Invalid Avro single-object encoding: expected magic bytes C3 01, got $byte0%02X $byte1%02X")
310       }
311 
312       // read the 8-byte fingerprint in little-endian order to match BinaryMessageEncoder
313       buffer.getLong()
314     }
315   }
