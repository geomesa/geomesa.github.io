73   /**
74    * Group converter for a record. Since these are geomesa records, we use our custom parquet read support
75    * so that values get parsed into standard simple feature attributes (including geometries) instead
76    * of generic avro types
77    *
78    * @param fileSchema parquet schema
79    * @param schema geomesa schema encodings, if it's a geomesa file
80    */
81   private class SimpleFeatureGroupConverter(fileSchema: MessageType, schema: SimpleFeatureParquetSchema)
82       extends GroupConverter with ValueMaterializer[GenericRecord] {
83 
84     private val avroSchema = new AvroSchemaConverter().convert(fileSchema)
85     private val delegate = new SimpleFeatureReadSupport.SimpleFeatureGroupConverter(schema)
86 
87     override def getConverter(fieldIndex: Int): Converter = delegate.getConverter(fieldIndex)
88     override def start(): Unit = delegate.start()
89     override def end(): Unit = delegate.end()
90     override def reset(): Unit = delegate.reset()
91     override def materialize(): GenericRecord = {
92       val rec = new GenericData.Record(avroSchema)
93       var i = 0
94       while (i < delegate.fieldCount) {
95         rec.put(i, delegate.getConverter(i).materialize())
96         i += 1
97       }
98       rec
99     }
100   }

102   /**
103    * Group converter for a record
104    *
105    * @param fileSchema parquet schema
106    */
107   private class GenericGroupConverter(fileSchema: MessageType) extends GroupConverter with ValueMaterializer[GenericRecord] {
108 
109     private val avroSchema = new AvroSchemaConverter().convert(fileSchema)
110     private val converters = fileSchema.getFields.asScala.map(converter).toArray
111 
112     private var rec: GenericRecord = _
113 
114     override def getConverter(fieldIndex: Int): Converter = converters(fieldIndex)
115     override def start(): Unit = {
116       rec = new GenericData.Record(avroSchema)
117       converters.foreach(_.reset())
118     }
119     override def end(): Unit = {
120       var i = 0
121       while (i < converters.length) {
122         rec.put(i, converters(i).materialize())
123         i += 1
124       }
125     }
126     override def reset(): Unit = rec = null
127     override def materialize(): GenericRecord = rec
128   }

130   /**
131     * Get a converter for a field type
132     *
133     * @param field field
134     * @return
135     */
136   private def converter(field: Type): ValueMaterializer[_ <: AnyRef] = {
137     val logical = field.getLogicalTypeAnnotation
138     if (field.isPrimitive) {
139       lazy val logicalConverter = logical match {
140         case _: StringLogicalTypeAnnotation => Some(new StringConverter())
141         case _: DateLogicalTypeAnnotation => Some(new DaysConverter())
142         case t: TimestampLogicalTypeAnnotation if t.getUnit == LogicalTypeAnnotation.TimeUnit.MILLIS => Some(new MillisConverter())
143         case t: TimestampLogicalTypeAnnotation if t.getUnit == LogicalTypeAnnotation.TimeUnit.MICROS => Some(new DateMicrosConverter())
144         case t: TimestampLogicalTypeAnnotation if t.getUnit == LogicalTypeAnnotation.TimeUnit.NANOS => Some(new NanosConverter())
145         case _: UUIDLogicalTypeAnnotation => Some(new UuidConverter())
146         case _ => None
147       }
148       val convert = field.asPrimitiveType().getPrimitiveTypeName match {
149         case PrimitiveTypeName.BINARY               => logicalConverter.getOrElse(new GeneralPrimitiveConverter())
150         case PrimitiveTypeName.INT32                => logicalConverter.getOrElse(new GeneralPrimitiveConverter())
151         case PrimitiveTypeName.INT64                => logicalConverter.getOrElse(new GeneralPrimitiveConverter())
152         case PrimitiveTypeName.DOUBLE               => new GeneralPrimitiveConverter()
153         case PrimitiveTypeName.FLOAT                => new GeneralPrimitiveConverter()
154         case PrimitiveTypeName.BOOLEAN              => new GeneralPrimitiveConverter()
155         case PrimitiveTypeName.FIXED_LEN_BYTE_ARRAY => logicalConverter.getOrElse(new GeneralPrimitiveConverter())
156         case _                                      => NullPrimitiveConverter
157       }
158       if (field.isRepetition(Repetition.REPEATED)) {
159         new RepeatedPrimitiveConverter(convert)
160       } else {
161         convert
162       }
163     } else {
164       val group = field.asGroupType()
165       def genericGroupConverter(): GenericGroupConverter = new GenericGroupConverter(new MessageType(group.getName, group.getFields))
166       logical match {
167         case _: ListLogicalTypeAnnotation =>
168           ParquetConverterFactory.getListElementType(group) match {
169             case Some(items) => new ListConverter(converter(items))
170             case _ => genericGroupConverter()
171           }
172 
173         case _: MapLogicalTypeAnnotation =>
174           ParquetConverterFactory.getMapKeyValueTypes(group) match {
175             case Some((k, v)) => new MapConverter(converter(k), converter(v))
176             case _ => genericGroupConverter()
177           }
178 
179         case _ =>
180           genericGroupConverter()
181       }
182     }
183   }

185   /**
186    * Converter for primitive fields with repetition of 'repeated'
187    *
188    * @param delegate single value converter
189    */
190   private class RepeatedPrimitiveConverter(delegate: PrimitiveConverter with ValueMaterializer[_ <: AnyRef])
191     extends PrimitiveConverter with ValueMaterializer[java.util.List[AnyRef]] {
192 
193     private var list: java.util.List[AnyRef] = _
194 
195     override def addBinary(value: Binary): Unit = { delegate.addBinary(value); addElement() }
196     override def addBoolean(value: Boolean): Unit = { delegate.addBoolean(value); addElement() }
197     override def addInt(value: Int): Unit = { delegate.addInt(value); addElement() }
198     override def addFloat(value: Float): Unit = { delegate.addFloat(value); addElement() }
199     override def addLong(value: Long): Unit = { delegate.addLong(value); addElement() }
200     override def addDouble(value: Double): Unit = { delegate.addDouble(value); addElement() }
201 
202     private def addElement(): Unit = {
203       if (list == null) {
204         list = new java.util.ArrayList[AnyRef]()
205       }
206       list.add(delegate.materialize())
207       delegate.reset()
208     }
209     override def reset(): Unit = list = null
210     override def materialize(): java.util.List[AnyRef] = list
211   }
