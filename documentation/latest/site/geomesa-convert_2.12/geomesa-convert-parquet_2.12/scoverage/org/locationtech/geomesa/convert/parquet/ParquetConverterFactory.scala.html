47   /**
48     * Handles parquet files (including those produced by the FSDS and CLI export)
49     *
50     * @param is input
51     * @param sft simple feature type, if known ahead of time
52     * @param hints hints, including the file path, if there is a file available
53     * @return
54     */
55   override def infer(
56       is: InputStream,
57       sft: Option[SimpleFeatureType],
58       hints: Map[String, AnyRef]): Try[(SimpleFeatureType, Config)] = {
59     val path = hints.get(EvaluationContext.InputFilePathKey) match {
60       case None => Failure(new RuntimeException("No file path specified to the input data"))
61       case Some(p) => Success(p.toString)
62     }
63     path.flatMap { p =>
64       Try {
65         // note: get the path as a URI so that we handle local files appropriately
66         val filePath = new Path(PathUtils.getUrl(p).toURI)
67         val footer = ParquetFileReader.readFooter(new Configuration(), filePath, ParquetMetadataConverter.NO_FILTER)
68         val (schema, fields, id, userData) = Try(SimpleFeatureParquetSchema.read(footer.getFileMetaData)).toOption.flatten match {
69           case Some(parquet) =>
70             // this is a geomesa encoded parquet file
71             val fields = parquet.sft.getAttributeDescriptors.asScala.map { descriptor =>
72               val name = StringSerialization.alphaNumericSafeString(descriptor.getLocalName)
73               // note: parquet converter stores the generic record under index 0
74               BasicField(descriptor.getLocalName, Some(Expression(s"avroPath($$0, '/$name')")))
75             }
76             val id = Expression(s"avroPath($$0, '/${SimpleFeatureSchema.FeatureIdField}')")
77             val userData =
78               Map("geomesa.feature.visibility" -> Expression(s"avroPath($$0, '/${SimpleFeatureSchema.VisibilitiesField}')"))
79 
80             // validate the existing schema, if any
81             sft.foreach { schema =>
82               if (schema.getAttributeDescriptors.asScala != parquet.sft.getAttributeDescriptors.asScala) {
83                 throw new IllegalArgumentException(
84                   s"Inferred schema does not match existing schema:\n  " +
85                     s"Existing schema: ${SimpleFeatureTypes.encodeType(schema)}\n  " +
86                     s"Inferred schema: ${SimpleFeatureTypes.encodeType(parquet.sft)}")
87               }
88             }
89             (parquet.sft, fields, Some(id), userData)
90 
91           case _ =>
92             // this is an arbitrary parquet file, create fields based on the schema
93             val types = ParquetConverterFactory.schemaTypes(footer.getFileMetaData)
94             val dataSft = TypeInference.schema("inferred-parquet", types)
95             // note: parquet converter stores the generic record under index 0
96             val fields = types.map(t => BasicField(t.name, Some(Expression(t.transform.apply(0)))))
97 
98             // validate the existing schema, if any
99             sft.foreach(AbstractConverterFactory.validateInferredType(_, types.map(_.typed)))
100 
101             (dataSft, fields, None, Map.empty[String, Expression])
102         }
103 
104         val converterConfig = BasicConfig(typeToProcess, None, id, Map.empty, userData)
105 
106         val config = configConvert.to(converterConfig)
107             .withFallback(fieldConvert.to(fields.toSeq))
108             .withFallback(optsConvert.to(BasicOptions.default))
109             .toConfig
110 
111         (schema, config)
112       }
113     }
114   }
115 }

123   /**
124    * Gets the type of elements in this list, assuming that this is a standard, 3-level parquet list type
125    *
126    * @param group list type
127    * @return element type, if input is a properly formatted 3-level parquet list
128    */
129   def getListElementType(group: GroupType): Option[GroupType] = {
130     for {
131       g <- Some(group) // optional group foo (LIST) {
132       if g.getFieldCount == 1 && !g.getType(0).isPrimitive
133       list = g.getType(0).asGroupType() // repeated group list {
134       if list.getFieldCount == 1 && list.isRepetition(Repetition.REPEATED) && !list.getType(0).isPrimitive
135     } yield {
136       list.getType(0).asGroupType() // optional group element {
137     }
138   }

140   /**
141    * Gets the type of keys and values in this map, assuming that this is a standard, 3-level parquet map type
142    *
143    * @param map map type
144    * @return (key type, value type), if input is a properly formatted 3-level parquet map
145    */
146   def getMapKeyValueTypes(map: GroupType): Option[(Type, Type)] = {
147     for {
148       mapGroup <- Some(map)
149       if mapGroup.getFieldCount == 1
150       tuples = mapGroup.getType(0)
151       if !tuples.isPrimitive
152       tuplesGroup = tuples.asGroupType()
153       if tuplesGroup.getFieldCount == 2 && tuplesGroup.isRepetition(Repetition.REPEATED)
154     } yield {
155       (tuplesGroup.getType(0), tuplesGroup.getType(1))
156     }
157   }

160     * Takes parquet file info returns the simple feature type bindings for it
161     *
162     * @param fileMetaData parquet file footer
163     * @return

165   private def schemaTypes(fileMetaData: FileMetaData): Seq[InferredType] = {
166     val namer = new Namer()
167     val geos: Option[GeoParquetMetadata] =
168       Option(fileMetaData.getKeyValueMetaData.get(GeoParquetMetadata.GeoParquetMetadataKey)).map(GeoParquetMetadata.fromJson)
169 
170     val types = fileMetaData.getSchema.getFields.asScala.flatMap(mapField(_, namer, geos)).toSeq
171     // check if we can derive a geometry field
172     val geom = TypeInference.deriveGeometry(types, namer)
173     types ++ geom
174   }

176   /**
177    * Infer a type from a top-level parquet field
178    *
179    * @param field field
180    * @param namer name helper
181    * @param geos extracted GeoParquet metadata from the parquet file
182    * @return
183    */
184   private def mapField(field: Type, namer: Namer, geos: Option[GeoParquetMetadata]): Seq[InferredType] = {
185     if (geos.exists(_.geometries.exists(_.covering.contains(field.getName)))) {
186       return Seq.empty // ignore bbox fields
187     }
188     // note: geometries *must* be at the root level, so we don't need to track them for nested fields
189     geos.flatMap(_.geometries.find(_.name == field.getName)) match {
190       case None => mapNormalField(field, namer)
191       case Some(geo) => mapGeoField(field, namer, geo)
192     }
193   }

195   /**
196    * Maps a non-geometry type field
197    *
198    * @param field field
199    * @param namer name helper
200    * @param path avro path to the parent of the field
201    * @return
202    */
203   private def mapNormalField(field: Type, namer: Namer, path: String = ""): Seq[InferredType] = {
204     val fieldPath = s"$path/${field.getName}"
205     lazy val name = namer(fieldPath)
206     val transform = FunctionTransform("avroPath(", s",'$fieldPath')")
207     val logical = field.getLogicalTypeAnnotation
208     if (field.isPrimitive) {
209       // note: date field transforms are handled by AvroReadSupport
210       lazy val intOrDate: InferredType = logical match {
211         case _: DateLogicalTypeAnnotation => InferredType(name, ObjectType.DATE, transform)
212         case _ => InferredType(name, ObjectType.INT, transform)
213       }
214       lazy val longOrTimestamp: InferredType = logical match {
215         case t: TimestampLogicalTypeAnnotation if t.getUnit == LogicalTypeAnnotation.TimeUnit.MILLIS => InferredType(name, ObjectType.DATE, transform)
216         case t: TimestampLogicalTypeAnnotation if t.getUnit == LogicalTypeAnnotation.TimeUnit.MICROS => InferredType(name, ObjectType.DATE, transform)
217         case _ => InferredType(name, ObjectType.LONG, transform)
218       }
219       lazy val stringOrBytes: InferredType = logical match {
220         case _: StringLogicalTypeAnnotation => InferredType(name, ObjectType.STRING, transform)
221         case _ => InferredType(name, ObjectType.BYTES, transform)
222       }
223       field.asPrimitiveType().getPrimitiveTypeName match {
224         case PrimitiveTypeName.BINARY               => Seq(stringOrBytes)
225         case PrimitiveTypeName.INT32                => Seq(intOrDate)
226         case PrimitiveTypeName.INT64                => Seq(longOrTimestamp)
227         case PrimitiveTypeName.FLOAT                => Seq(InferredType(name, ObjectType.FLOAT, transform))
228         case PrimitiveTypeName.DOUBLE               => Seq(InferredType(name, ObjectType.DOUBLE, transform))
229         case PrimitiveTypeName.BOOLEAN              => Seq(InferredType(name, ObjectType.BOOLEAN, transform))
230         case PrimitiveTypeName.FIXED_LEN_BYTE_ARRAY => Seq(InferredType(name, ObjectType.BYTES, transform))
231         case _ => Seq.empty
232       }
233     } else {
234       val group = field.asGroupType()
235       def jsonTransform(): FunctionTransform = FunctionTransform(s"avroToJson(${transform.prefix}", s"${transform.suffix})")
236       logical match {
237         case _: ListLogicalTypeAnnotation =>
238           if (getListElementType(group).exists(e => e.getFieldCount == 1 && e.getType(0).isPrimitive)) {
239             Seq(InferredType(name, ObjectType.LIST, transform))
240           } else {
241             // for non-standard or complex-element lists, we just convert to a json string
242             Seq(InferredType(name, ObjectType.JSON, jsonTransform()))
243           }
244 
245         case _: MapLogicalTypeAnnotation =>
246           if (getMapKeyValueTypes(group).exists { case (k, v) => k.isPrimitive && v.isPrimitive }) {
247             Seq(InferredType(name, ObjectType.MAP, transform))
248           } else {
249             // for non-standard or complex-value maps, we just convert to a json string
250             Seq(InferredType(name, ObjectType.JSON, jsonTransform()))
251           }
252 
253         case _ =>
254           group.getFields.asScala.flatMap(mapNormalField(_, namer, fieldPath)).toSeq
255       }
256     }
257   }

259   /**
260    * Maps a GeoParquet geometry field
261    *
262    * @param field field
263    * @param namer name helper
264    * @param geo GeoParquet metadata
265    * @return
266    */
267   private def mapGeoField(field: Type, namer: Namer, geo: ColumnMetadata): Seq[InferredType] = {
268     lazy val name = namer(field.getName)
269 
270     def geomTransform(fn: String): FunctionTransform = FunctionTransform(s"$fn(avroPath(", s",'/${field.getName}'))")
271 
272     def mismatch(): Seq[InferredType] = {
273       logger.warn(s"Found GeoParquet metadata for column '${field.getName}' but the schema does not match the expected type")
274       mapNormalField(field, namer)
275     }
276 
277     if (geo.encoding == GeoParquetColumnEncoding.WKB) {
278       if (isWkbType(field)) {
279         Seq(InferredType(name, GeoParquetColumnType.toObjectType(geo.types), geomTransform("geometry")))
280       } else {
281         mismatch()
282       }
283     } else {
284       val objectType = GeoParquetColumnEncoding.toObjectType(geo.encoding)
285       if ((geo.types.isEmpty || GeoParquetColumnType.toObjectType(geo.types) == objectType) && isNativeType(field, objectType)) {
286         Seq(InferredType(name, objectType, geomTransform(geo.encoding.toString)))
287       } else {
288         mismatch()
289       }
290     }
291   }

296    * @param field field
297    * @return

299   private def isWkbType(field: Type): Boolean =
300     !field.isRepetition(Repetition.REPEATED) && field.isPrimitive &&
301       field.asPrimitiveType().getPrimitiveTypeName == PrimitiveTypeName.BINARY

303   /**
304    * Checks if the field corresponds to the GeoParquet schema for a native-encoded geometry
305    *
306    * @param field field
307    * @param objectType native type
308    * @return
309    */
310   private def isNativeType(field: Type, objectType: ObjectType): Boolean = {
311     if (field.isRepetition(Repetition.REPEATED)) { false } else {
312       // custom compare, so we don't consider optional vs required, id, name, logicalTypes, etc
313       compareGeomFields(field, GeoParquetNative.schema(objectType).named(field.getName))
314     }
315   }
316 
317   private def compareGeomFields(left: Type, right: Type): Boolean = {
318     // make sure they're both either repeated or not, but don't consider required vs optional
319     if (left.isRepetition(Repetition.REPEATED) != right.isRepetition(Repetition.REPEATED)) {
320       false
321     } else if (left.isPrimitive) {
322       right.isPrimitive &&
323         right.asPrimitiveType().getPrimitiveTypeName == left.asPrimitiveType().getPrimitiveTypeName
324     } else {
325       !right.isPrimitive && {
326         val leftFields = left.asGroupType().getFields
327         val rightFields = right.asGroupType().getFields
328         if (leftFields.size() != rightFields.size()) { false } else {
329           leftFields.asScala.zip(rightFields.asScala).forall { case (f, n) => compareGeomFields(f, n) }
330         }
331       }
332     }
333   }
