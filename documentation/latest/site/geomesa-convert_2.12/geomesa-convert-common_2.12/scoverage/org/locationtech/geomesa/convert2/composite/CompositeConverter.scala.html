27 class CompositeConverter(val targetSft: SimpleFeatureType, delegates: Seq[(Predicate, SimpleFeatureConverter)])
28     extends SimpleFeatureConverter {

32   override def createEvaluationContext(globalParams: Map[String, Any]): EvaluationContext =
33     createEvaluationContext(delegates.map(_._2.createEvaluationContext(globalParams)), Stats(tags))

35   private def createEvaluationContext(contexts: Seq[EvaluationContext], theseStats: Stats): EvaluationContext = {
36     val stats = new Stats() {
37       override def success(i: Int): Long = theseStats.success(i) + contexts.map(_.stats.success(0)).sum
38       override def failure(i: Int): Long = theseStats.failure(i) + contexts.map(_.stats.failure(0)).sum
39     }
40     CompositeEvaluationContext(contexts, stats)
41   }
42 
43   override def process(is: InputStream, ec: EvaluationContext): CloseableIterator[SimpleFeature] = {
44     val contexts = ec match {
45       case cec: CompositeEvaluationContext => cec.contexts.iterator
46       case _ => Iterator.continually(ec)
47     }
48     val predicates = delegates.map { case (predicate, converter) =>
49       if (!contexts.hasNext) {
50         throw new IllegalArgumentException(s"Invalid evaluation context doesn't match this converter: $ec")
51       }
52       val context = contexts.next()
53       PredicateContext(predicate.withContext(context), converter, context)
54     }
55 
56     val toEval = Array.ofDim[AnyRef](1)
57 
58     val lines = IOUtils.lineIterator(is, StandardCharsets.UTF_8)
59 
60     new CloseableIterator[SimpleFeature] {
61 
62       private var delegate: CloseableIterator[SimpleFeature] = CloseableIterator.empty
63 
64       @tailrec
65       override def hasNext: Boolean = delegate.hasNext || {
66         if (!lines.hasNext) {
67           false
68         } else {
69           toEval(0) = lines.next()
70           ec.line += 1
71           delegate.close()
72           delegate = null
73           lazy val in = new ByteArrayInputStream(toEval(0).asInstanceOf[String].getBytes(StandardCharsets.UTF_8))
74           predicates.foreach { p =>
75             if (delegate == null && Try(p.predicate(toEval)).getOrElse(false)) {
76               delegate = p.converter.process(in, p.context)
77             } else {
78               p.context.line += 1
79             }
80           }
81           if (delegate == null) {
82             ec.stats.failure()
83             delegate = CloseableIterator.empty
84           }
85           hasNext
86         }
87       }
88 
89       override def next(): SimpleFeature = if (hasNext) { delegate.next } else { Iterator.empty.next }
90 
91       override def close(): Unit = {
92         CloseWithLogging(delegate)
93         is.close()
94       }
95     }
96   }
97 
98   override def close(): Unit = CloseWithLogging(delegates.map(_._2))
