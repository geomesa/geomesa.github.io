20 class CqlFunctionFactory extends TransformerFunctionFactory with LazyLogging {
21 
22   import scala.collection.JavaConverters._
23 
24   override val functions: Seq[TransformerFunction] = {
25     val builder = Seq.newBuilder[TransformerFunction]
26 
27     CommonFactoryFinder.getFunctionFactories(null).asScala.toSeq.foreach { factory =>
28       val names = try {
29         // only import default cql functions (without namespaces)
30         // exclude 'categorize', as it doesn't parse into a function correctly
31         factory.getFunctionNames.asScala.filter(f => f.getFunctionName.getNamespaceURI == null && f.getName != "Categorize")
32       } catch {
33         // if CQL classes aren't on the classpath, these functions won't be available
34         case e: NoClassDefFoundError =>
35           logger.warn(s"Couldn't create cql function factory '${factory.getClass.getName}': ${e.toString}")
36           Seq.empty
37       }
38       names.foreach { f =>
39         val name = f.getFunctionName.toString
40         val expressions = Array.tabulate(f.getArguments.size())(arrayIndexProperty)
41         try { builder += new CqlTransformerFunction(name, expressions) } catch {
42           // Some CQL functions require literal arguments, which are unavailable during function discovery.
43           case _: Exception => builder += new DeferredCqlTransformerFunction(name, f.getArguments.size())
44         }
45       }
46     }
47     builder.result()
48   }
49 }

51 object CqlFunctionFactory {
52 
53   private val ff = CommonFactoryFinder.getFilterFactory
54 
55   // use alphas for the array indices, as used by the ArrayPropertyAccessor, below
56   private def arrayIndexProperty(i: Int): PropertyName = ff.property(('a' + i).toChar.toString)
57 
58   private def cqlExpressions(args: List[Expression], argCount: Int) = {
59     Array.tabulate(argCount) { i =>
60       if (i < args.length && args(i).isInstanceOf[Literal[_]]) {
61         ff.literal(args(i).asInstanceOf[Literal[_]].value)
62       } else {
63         arrayIndexProperty(i)
64       }
65     }
66   }
67 
68   class CqlTransformerFunction(name: String, expressions: Array[_ <: org.geotools.api.filter.expression.Expression])
69       extends NamedTransformerFunction(Seq(s"cql:$name")) {
70 
71     private val fn = ff.function(name, expressions: _*)
72 
73     override def apply(args: Array[AnyRef]): AnyRef = fn.evaluate(args)
74 
75     override def getInstance(args: List[Expression]): CqlTransformerFunction = {
76       // remap literals to cql literals so that they can be optimized (e.g. prepared geometries, etc)
77       new CqlTransformerFunction(name, cqlExpressions(args, fn.getFunctionName.getArguments.size()))
78     }
79   }
80 
81   private class DeferredCqlTransformerFunction(name: String, argCount: Int)
82       extends NamedTransformerFunction(Seq(s"cql:$name")) {
83 
84     override def apply(args: Array[AnyRef]): AnyRef =
85       throw new UnsupportedOperationException(s"CQL function '$name' must be initialized with converter arguments")
86 
87     override def getInstance(args: List[Expression]): CqlTransformerFunction = {
88       new CqlTransformerFunction(name, cqlExpressions(args, argCount))
89     }
90   }
91 
92   /**
93     * For accessing 'properties' of arrays (instead of simple features)
94     */
95   class ArrayPropertyAccessorFactory extends PropertyAccessorFactory {
96     override def createPropertyAccessor(typ: Class[_],
97                                         xpath: String,
98                                         target: Class[_],
99                                         hints: Hints): PropertyAccessor = {
100       if (typ.isArray) { new ArrayPropertyAccessor } else { null }
101     }
102   }
103 
104   /**
105     * Accessing properties of an array. Indices are expected to be lower-case letters,
106     * where 'a' indicates the first element, b the second, etc. We use alphas because
107     * integers are interpreted as literals, not properties.
108     */
109   class ArrayPropertyAccessor extends PropertyAccessor {
110 
111     override def canHandle(obj: Any, xpath: String, target: Class[_]): Boolean = {
112       val i = toIndex(xpath)
113       i >= 0 && i < obj.asInstanceOf[Array[Any]].length
114     }
115 
116     override def set[T](obj: Any, xpath: String, value: T, target: Class[T]): Unit =
117       obj.asInstanceOf[Array[Any]].update(toIndex(xpath), value)
118 
119     override def get[T](obj: Any, xpath: String, target: Class[T]): T =
120       obj.asInstanceOf[Array[Any]].apply(toIndex(xpath)).asInstanceOf[T]
121 
122     private def toIndex(xpath: String): Int = {
123       if (xpath.length == 1) {
124         xpath.charAt(0) - 'a'
125       } else {
126         -1
127       }
128     }
129   }
