85     def toSortHint(sortBy: Array[SortBy]): String = {
86       val strings = sortBy.flatMap { sb =>
87         val prop = sb.getPropertyName
88         Seq(if (prop == null) { "" } else { prop.getPropertyName }, s"${sb.getSortOrder == SortOrder.DESCENDING}")
89       }
90       StringSerialization.encodeSeq(strings)
91     }
92 
93     def fromSortHint(hint: String): Seq[(String, Boolean)] =
94       StringSerialization.decodeSeq(hint).grouped(2).toSeq.map { case Seq(f, r) => (f, r.toBoolean) }
95 
96     def toProjectionHint(crs: QueryReferenceSystems): String =

99     def fromProjectionHint(hint: String): QueryReferenceSystems = {
