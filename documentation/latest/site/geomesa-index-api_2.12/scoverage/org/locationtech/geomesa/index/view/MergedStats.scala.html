51   override def getCount(sft: SimpleFeatureType, filter: Filter, exact: Boolean, queryHints: Hints): Option[Long] = {
52     val now = Instant.now() // shared current time passed to all stores for consistency in bounds calculations
53     // note: unlike most methods in this class, this will return if any of the merged stores provide a response
54     def getSingle(source: MergedSource[StatsWithSchemas]): Option[Long] = {
55       val schema = source.source.ds.getSchema(sft.getTypeName)
56       source.source.stats.getCount(schema, source.filter(sft.getTypeName, filter, now), exact, queryHints)
57     }
58 
59     if (parallel) {
60       val results = new CopyOnWriteArrayList[Long]()
61       stats.toList.map(s => CachedThreadPool.submit(s.decorator(() => getSingle(s).foreach(results.add)))).foreach(_.get)
62       results.asScala.reduceLeftOption(_ + _)
63     } else {
64       stats.flatMap(getSingle).reduceLeftOption(_ + _)
65     }

146   private def merge[T <: Stat](typeName: String, filter: Filter, query: (GeoMesaStats, Filter) => Option[T]): Option[T] = {
147     val now = Instant.now()
148     if (parallel) {
149       val results = new CopyOnWriteArrayList[Option[T]]()
150       stats.toList.map(s => CachedThreadPool.submit(s.decorator(() => results.add(query(s.source.stats, s.filter(typeName, filter, now)))))).foreach(_.get)
151       results.asScala.reduceLeft((res, next) => for { r <- res; n <- next } yield { (r + n).asInstanceOf[T] })
152     } else {
153       // lazily evaluate each stat as we only return Some if all the child stores do
154       val head = query(stats.head.source.stats, stats.head.filter(typeName, filter, now))
155       stats.tail.foldLeft(head) { case (result, stat) =>
156         for {
157           r <- result
158           n <- query(stat.source.stats, stat.filter(typeName, filter, now))
159         } yield {
160           (r + n).asInstanceOf[T]
161         }
162       }
163     }
164   }

178     override def analyze(sft: SimpleFeatureType): Seq[Stat] = {
