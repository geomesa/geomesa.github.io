53   override def getCount(sft: SimpleFeatureType, filter: Filter, exact: Boolean, queryHints: Hints): Option[Long] = {
54     val now = Instant.now() // shared current time passed to all stores for consistency in bounds calculations
55     // note: unlike most methods in this class, this will return if any of the merged stores provide a response
56     def getSingle(source: MergedSource[StatsWithSchemas]): Option[Long] = {
57       val schema = source.source.ds.getSchema(sft.getTypeName)
58       source.source.stats.getCount(schema, source.filter(sft.getTypeName, filter, now), exact, queryHints)
59     }
60 
61     if (parallel) {
62       val results = new CopyOnWriteArrayList[Long]()
63       stats.toList.map(s => CachedThreadPool.submit(decorator(() => getSingle(s).foreach(results.add)))).foreach(_.get)
64       results.asScala.reduceLeftOption(_ + _)

148   private def merge[T <: Stat](typeName: String, filter: Filter, query: (GeoMesaStats, Filter) => Option[T]): Option[T] = {
149     val now = Instant.now()
150     if (parallel) {
151       val results = new CopyOnWriteArrayList[Option[T]]()
152       stats.toList.map(s => CachedThreadPool.submit(decorator(() => results.add(query(s.source.stats, s.filter(typeName, filter, now)))))).foreach(_.get)
153       results.asScala.reduceLeft((res, next) => for { r <- res; n <- next } yield { (r + n).asInstanceOf[T] })
154     } else {
155       // lazily evaluate each stat as we only return Some if all the child stores do
156       val head = query(stats.head.source.stats, stats.head.filter(typeName, filter, now))
157       stats.tail.foldLeft(head) { case (result, stat) =>
158         for {
159           r <- result
160           n <- query(stat.source.stats, stat.filter(typeName, filter, now))
161         } yield {
162           (r + n).asInstanceOf[T]
163         }
164       }
165     }
166   }

179   private class MergedStatWriter(writers: Seq[GeoMesaStatWriter]) extends GeoMesaStatWriter {
180     override def analyze(sft: SimpleFeatureType): Seq[Stat] = {
181       writers.map(_.analyze(sft)).reduceLeft[Seq[Stat]] { case (left, right) =>
182         left.zip(right).map { case (l, r) => l + r }
183       }
184     }
185 
186     override def updater(sft: SimpleFeatureType): StatUpdater = new MergedStatUpdater(writers.map(_.updater(sft)))
187 
188     override def rename(sft: SimpleFeatureType, previous: SimpleFeatureType): Unit =
189       writers.foreach(_.rename(sft, previous))
190 
191     override def clear(sft: SimpleFeatureType): Unit = writers.foreach(_.clear(sft))
192   }

194   private class MergedStatUpdater(updaters: Seq[StatUpdater]) extends StatUpdater {
195     override def add(sf: SimpleFeature): Unit = updaters.foreach(_.add(sf))
196     override def remove(sf: SimpleFeature): Unit = updaters.foreach(_.remove(sf))
197     override def flush(): Unit = updaters.foreach(_.flush())
198     override def close(): Unit = CloseWithLogging(updaters)
