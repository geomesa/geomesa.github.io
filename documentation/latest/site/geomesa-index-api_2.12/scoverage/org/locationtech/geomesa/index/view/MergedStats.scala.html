52   override def getCount(sft: SimpleFeatureType, filter: Filter, exact: Boolean, queryHints: Hints): Option[Long] = {
53     val context = QueryContext(filter)
54     // note: unlike most methods in this class, this will return if any of the merged stores provide a response
55     def getSingle(source: MergedSource[StatsWithSchemas]): Option[Long] = {
56       val schema = source.source.ds.getSchema(sft.getTypeName)
57       source.source.stats.getCount(schema, source.filter(sft.getTypeName, context), exact, queryHints)
58     }
59 
60     if (parallel) {
61       val results = new CopyOnWriteArrayList[Long]()
62       stats.toList.map(s => CachedThreadPool.submit(s.decorator(() => getSingle(s).foreach(results.add)))).foreach(_.get)
63       results.asScala.reduceLeftOption(_ + _)
64     } else {
65       stats.flatMap(getSingle).reduceLeftOption(_ + _)
66     }

147   private def merge[T <: Stat](typeName: String, filter: Filter, query: (GeoMesaStats, Filter) => Option[T]): Option[T] = {
148     val context = QueryContext(filter)
149     if (parallel) {
150       val results = new CopyOnWriteArrayList[Option[T]]()
151       stats.toList.map(s => CachedThreadPool.submit(s.decorator(() => results.add(query(s.source.stats, s.filter(typeName, context)))))).foreach(_.get)
152       results.asScala.reduceLeft((res, next) => for { r <- res; n <- next } yield { (r + n).asInstanceOf[T] })
153     } else {
154       // lazily evaluate each stat as we only return Some if all the child stores do
155       val head = query(stats.head.source.stats, stats.head.filter(typeName, context))
156       stats.tail.foldLeft(head) { case (result, stat) =>
157         for {
158           r <- result
159           n <- query(stat.source.stats, stat.filter(typeName, context))
160         } yield {
161           (r + n).asInstanceOf[T]
162         }
163       }
164     }
165   }

178   private class MergedStatWriter(writers: Seq[GeoMesaStatWriter]) extends GeoMesaStatWriter {
179     override def analyze(sft: SimpleFeatureType): Seq[Stat] = {
180       writers.map(_.analyze(sft)).reduceLeft[Seq[Stat]] { case (left, right) =>
181         left.zip(right).map { case (l, r) => l + r }
182       }
183     }
184 
185     override def updater(sft: SimpleFeatureType): StatUpdater = new MergedStatUpdater(writers.map(_.updater(sft)))
186 
187     override def rename(sft: SimpleFeatureType, previous: SimpleFeatureType): Unit =
188       writers.foreach(_.rename(sft, previous))
189 
190     override def clear(sft: SimpleFeatureType): Unit = writers.foreach(_.clear(sft))
191   }
192 
193   private class MergedStatUpdater(updaters: Seq[StatUpdater]) extends StatUpdater {
194     override def add(sf: SimpleFeature): Unit = updaters.foreach(_.add(sf))
195     override def remove(sf: SimpleFeature): Unit = updaters.foreach(_.remove(sf))
196     override def flush(): Unit = updaters.foreach(_.flush())
197     override def close(): Unit = CloseWithLogging(updaters)
