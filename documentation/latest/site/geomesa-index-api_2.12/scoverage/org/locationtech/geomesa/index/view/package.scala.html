44   /**
45     * Data store that merges the schemas from multiple delegate stores and presents them as a unified result
46     *
47     * @param stores delegate stores
48     * @param namespace schema namespace
49     */
50   abstract class MergedDataStoreSchemas(stores: Seq[DataStore], namespace: Option[String])
51       extends ReadOnlyDataStore {
52 
53     import scala.collection.JavaConverters._
54 
55     override def getTypeNames: Array[String] = stores.map(_.getTypeNames).reduceLeft(_ intersect _)
56 
57     override def getNames: java.util.List[Name] =
58       java.util.Arrays.asList(getTypeNames.map(t => new NameImpl(namespace.orNull, t)): _*)
59 
60     override def getSchema(name: Name): SimpleFeatureType = getSchema(name.getLocalPart)
61 
62     override def getSchema(typeName: String): SimpleFeatureType = {
63       val schemas = stores.map(_.getSchema(typeName))
64 
65       if (schemas.contains(null)) {
66         return null
67       }
68 
69       lazy val fail = new IOException("Delegate schemas do not match: " +
70           schemas.map(SimpleFeatureTypes.encodeType).mkString(" :: "))
71 
72       schemas.reduceLeft[SimpleFeatureType] { case (left, right) =>
73         if (left.getAttributeCount != right.getAttributeCount) {
74           throw fail
75         }
76         val builder = new SchemaBuilder()
77         val attribute = new AttributeTypeBuilder()
78 
79         val leftDescriptors = left.getAttributeDescriptors.iterator()
80         val rightDescriptors = right.getAttributeDescriptors.iterator()
81 
82         while (leftDescriptors.hasNext) {
83           val leftDescriptor = leftDescriptors.next
84           val rightDescriptor = rightDescriptors.next
85           if (leftDescriptor.getLocalName != rightDescriptor.getLocalName) {
86             throw fail
87           }
88           val leftBinding = leftDescriptor.getType.getBinding
89           val rightBinding = rightDescriptor.getType.getBinding
90           // determine a common binding if possible, for things like java.sql.TimeStamp vs java.util.Date
91           if (leftBinding == rightBinding || leftBinding.isAssignableFrom(rightBinding)) {
92             attribute.binding(leftBinding)
93           } else if (rightBinding.isAssignableFrom(leftBinding)) {
94             attribute.binding(rightBinding)
95           } else {
96             throw fail
97           }
98 
99           // add the user data from each descriptor so the delegate stores have it if needed
100           leftDescriptor.getUserData.asScala.foreach { case (k, v) => attribute.userData(k, v) }
101           rightDescriptor.getUserData.asScala.foreach { case (k, v) => attribute.userData(k, v) }
102 
103           Some(leftDescriptor).collect { case g: GeometryDescriptor => attribute.crs(g.getCoordinateReferenceSystem) }
104 
105           builder.addAttribute(attribute.buildDescriptor(leftDescriptor.getLocalName))
106         }
107         val merged = builder.build(namespace.orNull, typeName)
108         // preference any user data conflicts by the order stores are declared
109         Seq(right, left).foreach(t => merged.getUserData.putAll(t.getUserData))
110         merged
111       }
112     }
113 
114     override def dispose(): Unit = CloseQuietly.raise(stores)
115 
116     override def getInfo: ServiceInfo = {
117       val info = new DefaultServiceInfo()
118       info.setDescription(s"Features from ${getClass.getSimpleName}")
119       info.setSchema(FeatureTypes.DEFAULT_NAMESPACE)
120       info
121     }
122 
123     override def getLockingManager: LockingManager = null
124   }

126   /**
127    * Does a sorted merge of already sorted streams
128    *
129    * @param streams streams, each individually sorted
130    * @param ordering ordering
131    * @tparam T type bounds
132    */
133   class SortedMergeIterator[T](streams: Seq[CloseableIterator[T]])(implicit ordering: Ordering[T])
134     extends CloseableIterator[T] {
135 
136     private val indexedStreams = streams.toIndexedSeq
137     // reverse the ordering so we get the head of the queue as the first value
138     private val heads = scala.collection.mutable.PriorityQueue.empty[(T, Int)](Ordering.Tuple2(ordering.reverse, Ordering.Int))
139 
140     try { init() } catch {
141       case NonFatal(e) => CloseQuietly(streams).foreach(e.addSuppressed); throw e
142     }
143 
144     private def init(): Unit = {
145       var i = 0
146       streams.foreach { s =>
147         if (s.hasNext) {
148           heads += s.next -> i
149         }
150         i += 1
151       }
152     }
153 
154     override def hasNext: Boolean = heads.nonEmpty
155 
156     override def next(): T = {
157       val (n, i) = heads.dequeue()
158       val stream = indexedStreams(i)
159       if (stream.hasNext) {
160         heads += stream.next -> i
161       }
162       n
163     }
164 
165     override def close(): Unit = CloseWithLogging(streams)
166   }
