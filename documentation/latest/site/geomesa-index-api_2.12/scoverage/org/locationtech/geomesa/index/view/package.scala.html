43   /**
44     * Data store that merges the schemas from multiple delegate stores and presents them as a unified result
45     *
46     * @param stores delegate stores
47     * @param namespace schema namespace
48     */
49   abstract class MergedDataStoreSchemas(stores: Seq[DataStore], namespace: Option[String])
50       extends ReadOnlyDataStore {
51 
52     import scala.collection.JavaConverters._
53 
54     override def getTypeNames: Array[String] = stores.map(_.getTypeNames).reduceLeft(_ intersect _)
55 
56     override def getNames: java.util.List[Name] =
57       java.util.Arrays.asList(getTypeNames.map(t => new NameImpl(namespace.orNull, t)): _*)
58 
59     override def getSchema(name: Name): SimpleFeatureType = getSchema(name.getLocalPart)
60 
61     override def getSchema(typeName: String): SimpleFeatureType = {
62       val schemas = stores.map(_.getSchema(typeName))
63 
64       if (schemas.contains(null)) {
65         return null
66       }
67 
68       lazy val fail = new IOException("Delegate schemas do not match: " +
69           schemas.map(SimpleFeatureTypes.encodeType).mkString(" :: "))
70 
71       schemas.reduceLeft[SimpleFeatureType] { case (left, right) =>
72         if (left.getAttributeCount != right.getAttributeCount) {
73           throw fail
74         }
75         val builder = new SchemaBuilder()
76         val attribute = new AttributeTypeBuilder()
77 
78         val leftDescriptors = left.getAttributeDescriptors.iterator()
79         val rightDescriptors = right.getAttributeDescriptors.iterator()
80 
81         while (leftDescriptors.hasNext) {
82           val leftDescriptor = leftDescriptors.next
83           val rightDescriptor = rightDescriptors.next
84           if (leftDescriptor.getLocalName != rightDescriptor.getLocalName) {
85             throw fail
86           }
87           val leftBinding = leftDescriptor.getType.getBinding
88           val rightBinding = rightDescriptor.getType.getBinding
89           // determine a common binding if possible, for things like java.sql.TimeStamp vs java.util.Date
90           if (leftBinding == rightBinding || leftBinding.isAssignableFrom(rightBinding)) {
91             attribute.binding(leftBinding)
92           } else if (rightBinding.isAssignableFrom(leftBinding)) {
93             attribute.binding(rightBinding)
94           } else {
95             throw fail
96           }
97 
98           // add the user data from each descriptor so the delegate stores have it if needed
99           leftDescriptor.getUserData.asScala.foreach { case (k, v) => attribute.userData(k, v) }
100           rightDescriptor.getUserData.asScala.foreach { case (k, v) => attribute.userData(k, v) }
101 
102           Some(leftDescriptor).collect { case g: GeometryDescriptor => attribute.crs(g.getCoordinateReferenceSystem) }
103 
104           builder.addAttribute(attribute.buildDescriptor(leftDescriptor.getLocalName))
105         }
106         val merged = builder.build(namespace.orNull, typeName)
107         // preference any user data conflicts by the order stores are declared
108         Seq(right, left).foreach(t => merged.getUserData.putAll(t.getUserData))
109         merged
110       }
111     }
112 
113     override def dispose(): Unit = stores.foreach(_.dispose())
114 
115     override def getInfo: ServiceInfo = {
116       val info = new DefaultServiceInfo()
117       info.setDescription(s"Features from ${getClass.getSimpleName}")
118       info.setSchema(FeatureTypes.DEFAULT_NAMESPACE)
119       info
120     }
121 
122     override def getLockingManager: LockingManager = null
123   }

125   /**
126    * Does a sorted merge of already sorted streams
127    *
128    * @param streams streams, each individually sorted
129    * @param ordering ordering
130    * @tparam T type bounds
131    */
132   class SortedMergeIterator[T](streams: Seq[CloseableIterator[T]])(implicit ordering: Ordering[T])
133     extends CloseableIterator[T] {
134 
135     private val indexedStreams = streams.toIndexedSeq
136     // reverse the ordering so we get the head of the queue as the first value
137     private val heads = scala.collection.mutable.PriorityQueue.empty[(T, Int)](Ordering.Tuple2(ordering.reverse, Ordering.Int))
138 
139     init()
140 
141     private def init(): Unit = {
142       var i = 0
143       streams.foreach { s =>
144         if (s.hasNext) {
145           heads += s.next -> i
146         }
147         i += 1
148       }
149     }
150 
151     override def hasNext: Boolean = heads.nonEmpty
152 
153     override def next(): T = {
154       val (n, i) = heads.dequeue()
155       val stream = indexedStreams(i)
156       if (stream.hasNext) {
157         heads += stream.next -> i
158       }
159       n
160     }
161 
162     override def close(): Unit = CloseWithLogging(streams)
163   }
