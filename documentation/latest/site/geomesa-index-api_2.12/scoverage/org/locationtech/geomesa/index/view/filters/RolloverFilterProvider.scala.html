23 /**
24  * Filter provider for "rollover" scenarios. When using tiered storage, features may be migrated from one store to another
25  * on a schedule. This provider can read the work log of the rollover process to ensure that the correct boundaries
26  * are enforced between the different stores, based on feature timestamps.
27  */

34   /**
35    * Create a coverage table if it does not already exist. The caller manages the connection and transaction.
36    * Timestamps are stored without a time zone, using UTC wall-clock values for conversion to and from Instant.
37    *
38    * @param cx connection
39    * @param table table name, optionally qualified with a schema (e.g. `public.rollover`)
40    */
41   def createTable(cx: Connection, table: String): Unit = {
42     val qualifiedTable = quoteTable(table)
43     WithClose(cx.createStatement()) { st =>
44       st.executeUpdate(
45         s"CREATE TABLE IF NOT EXISTS $qualifiedTable (" +
46           "typeName text NOT NULL, " +
47           "covered_from timestamp without time zone NOT NULL, " +
48           "covered_through timestamp without time zone NOT NULL, " +
49           "PRIMARY KEY (typeName))")
50     }
51   }
52 
53   /**
54    * Insert or replace the coverage interval for a type name. The caller manages the connection and transaction.
55    *
56    * @param connection connection
57    * @param table table name, optionally qualified with a schema
58    * @param interval coverage interval
59    */
60   def insertCoverage(connection: Connection, table: String, interval: CoverageInterval): Unit = {
61     val qualifiedTable = quoteTable(table)
62     WithClose(connection.prepareStatement(
63       s"INSERT INTO $qualifiedTable (typeName, covered_from, covered_through) VALUES (?, ?, ?) " +
64         "ON CONFLICT (typeName) DO UPDATE SET " +
65         "covered_from = EXCLUDED.covered_from, covered_through = EXCLUDED.covered_through")) { statement =>
66       statement.setString(1, interval.typeName)
67       statement.setObject(2, interval.from.atOffset(ZoneOffset.UTC).toLocalDateTime)
68       statement.setObject(3, interval.through.atOffset(ZoneOffset.UTC).toLocalDateTime)
69       statement.executeUpdate()
70     }
71   }

73   private def quoteTable(table: String): String = {
74     require(table != null, "Table name must not be null")
75     val parts = table.split("\\.", -1)
76     require(parts.length <= 2, s"Invalid table name: $table")
77     parts.map(quoteIdentifier).mkString(".")
78   }
79 
80   private def quoteIdentifier(identifier: String): String = {
81     require(identifier != null && identifier.matches("[A-Za-z_][A-Za-z0-9_]{0,62}"),
82       s"Invalid SQL identifier: $identifier")
83     "\"" + identifier + "\""
84   }

86   case class CoverageInterval(typeName: String, from: Instant, through: Instant) {
87     require(typeName != null, "Coverage type name must not be null")
88     require(from != null && through != null, "Coverage interval must not have null bounds")
89     require(from.isBefore(through), "Coverage interval must have from < through")
90     require(from.getNano % 1000000 == 0 && through.getNano % 1000000 == 0, "Coverage bounds must have millisecond precision")
91   }

93   class SourceProvider extends JdbcRolloverProvider {
94     override def apply(sft: SimpleFeatureType, context: QueryContext): Option[Filter] = {
95       coverage(sft, context).map { interval =>
96         val dtg = ff.property(dtgField(sft))
97         val start = ff.literal(Date.from(interval.from))
98         val end = ff.literal(Date.from(interval.through))
99         ff.or(ff.isNull(dtg), ff.or(ff.less(dtg, start), ff.greaterOrEqual(dtg, end)))
100       }
101     }
102   }

104   class DestinationProvider extends JdbcRolloverProvider {
105     override def apply(sft: SimpleFeatureType, context: QueryContext): Option[Filter] = {
106       val filter = coverage(sft, context).fold[Filter](Filter.EXCLUDE) { interval =>
107         val dtg = ff.property(dtgField(sft))
108         val start = ff.literal(Date.from(interval.from))
109         val end = ff.literal(Date.from(interval.through))
110         ff.and(ff.greaterOrEqual(dtg, start), ff.less(dtg, end))
111       }
112       Some(filter)
113     }
114   }

116   trait JdbcRolloverProvider extends MergedFilterProvider {
117 
118     private var source: BasicDataSource = _
119     private var sql: String = _
120     private var url: String = _
121     private var table: String = _
122 
123     override def init(config: Map[String, String]): Unit = {
124       def required(k: String): String = config.getOrElse(k, throw new IllegalArgumentException(s"Missing required key: $k"))
125 
126       this.url = required("url")
127       this.table = s"${quoteIdentifier(required("schema"))}.${quoteIdentifier(required("table"))}"
128       this.sql = s"SELECT covered_from, covered_through FROM $table WHERE typeName = ?"
129       source = new BasicDataSource()
130       source.setUrl(url)
131       source.setUsername(required("username"))
132       source.setPassword(required("password"))
133       source.setMaxTotal(config.getOrElse("max-connections", "2").toInt)
134       source.setMaxIdle(1)
135       source.setMinIdle(0)
136       source.setMaxWait(Duration.ofSeconds(5))
137     }
138 
139     protected def coverage(sft: SimpleFeatureType, context: QueryContext): Option[CoverageInterval] = {
140       val state = context.state.compute(StateKey, (_, existing) => {
141         if (existing == null) {
142           CoverageState(url, table, dtgField(sft), readCoverage(sft.getTypeName))
143         } else {
144           validate(sft, existing.asInstanceOf[CoverageState])
145         }
146       })
147       state.asInstanceOf[CoverageState].coverage
148     }
149 
150     private def validate(sft: SimpleFeatureType, existing: CoverageState): CoverageState = {
151       val dtg = dtgField(sft)
152       if (existing.url != url || existing.table != table) {
153         throw new IllegalArgumentException(
154           s"RolloverFilterProviders are configured to read from different tables: " +
155             s"${existing.url} ${existing.table} vs $url $table")
156       } else if (existing.dtg != dtg) {
157         throw new IllegalArgumentException(
158           s"RolloverFilterProviders for '${sft.getTypeName}' are using different default date fields: ${existing.dtg} vs $dtg")
159       }
160       existing
161     }
162 
163     private def readCoverage(typeName: String): Option[CoverageInterval] = {
164       WithClose(source.getConnection) { cx =>
165         WithClose(cx.prepareStatement(sql)) { statement =>
166           statement.setQueryTimeout(5)
167           statement.setString(1, typeName)
168           WithClose(statement.executeQuery()) { results =>
169             if (!results.next()) {
170               None
171             } else {
172               val from = Option(results.getObject(1, classOf[LocalDateTime])).map(_.toInstant(ZoneOffset.UTC)).orNull
173               val through = Option(results.getObject(2, classOf[LocalDateTime])).map(_.toInstant(ZoneOffset.UTC)).orNull
174               Some(CoverageInterval(typeName, from, through))
175             }
176           }
177         }
178       }
179     }
180 
181     override def close(): Unit = CloseWithLogging(Option(source))
182   }
