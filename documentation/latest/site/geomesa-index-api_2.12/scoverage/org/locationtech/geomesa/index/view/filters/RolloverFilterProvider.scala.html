23 /**
24  * Filter provider for "rollover" scenarios. When using tiered storage, features may be migrated from one store to another
25  * on a schedule. This provider can read the work log of the rollover process to ensure that the correct boundaries
26  * are enforced between the different stores, based on feature timestamps.
27  */

35   /**
36    * Create a coverage table if it does not already exist. The caller manages the connection and transaction.
37    * Timestamps are stored without a time zone, using UTC wall-clock values for conversion to and from Instant.
38    *
39    * @param cx connection
40    * @param schema schema name, or None to use the connection's default schema
41    * @param table table name
42    */
43   def createTable(cx: Connection, schema: Option[String], table: String): Unit = {
44     val qualifiedTable = quoteTable(schema, table)
45     WithClose(cx.createStatement()) { st =>
46       st.executeUpdate(
47         s"CREATE TABLE IF NOT EXISTS $qualifiedTable (" +
48           "type_name text NOT NULL, " +
49           "covered_from timestamp without time zone NOT NULL, " +
50           "covered_through timestamp without time zone NOT NULL, " +
51           "PRIMARY KEY (type_name))")
52     }
53   }
54 
55   /**
56    * Insert or replace the coverage interval for a type name. The caller manages the connection and transaction.
57    *
58    * @param connection connection
59    * @param schema schema name, or None to use the connection's default schema
60    * @param table table name
61    * @param interval coverage interval
62    */
63   def insertCoverage(connection: Connection, schema: Option[String], table: String, interval: CoverageInterval): Unit = {
64     val qualifiedTable = quoteTable(schema, table)
65     WithClose(connection.prepareStatement(
66       s"INSERT INTO $qualifiedTable (type_name, covered_from, covered_through) VALUES (?, ?, ?) " +
67         "ON CONFLICT (type_name) DO UPDATE SET " +
68         "covered_from = EXCLUDED.covered_from, covered_through = EXCLUDED.covered_through")) { statement =>
69       statement.setString(1, interval.typeName)
70       statement.setObject(2, interval.from.atOffset(ZoneOffset.UTC).toLocalDateTime)
71       statement.setObject(3, interval.through.atOffset(ZoneOffset.UTC).toLocalDateTime)
72       statement.executeUpdate()
73     }
74   }

76   private def quoteTable(schema: Option[String], table: String): String =
77     s"${schema.fold("")(s => s"${quoteIdentifier(s)}.")}${quoteIdentifier(table)}"
78 
79   private def quoteIdentifier(identifier: String): String = {
80     require(identifier != null && identifier.matches("[A-Za-z_][A-Za-z0-9_]{0,62}"),
81       s"Invalid SQL identifier: $identifier")
82     "\"" + identifier + "\""
83   }

85   case class CoverageInterval(typeName: String, from: Instant, through: Instant) {
86     require(typeName != null, "Coverage type name must not be null")
87     require(from != null && through != null, "Coverage interval must not have null bounds")
88     require(from.isBefore(through), "Coverage interval must have from < through")
89     require(from.getNano % 1000000 == 0 && through.getNano % 1000000 == 0, "Coverage bounds must have millisecond precision")
90   }

92   class SourceProvider extends JdbcRolloverProvider {
93     override def apply(sft: SimpleFeatureType, context: QueryContext): Option[Filter] = {
94       coverage(sft, context).map { interval =>
95         val dtg = ff.property(dtgField(sft))
96         val start = ff.literal(Date.from(interval.from))
97         val end = ff.literal(Date.from(interval.through))
98         ff.or(ff.isNull(dtg), ff.or(ff.less(dtg, start), ff.greaterOrEqual(dtg, end)))
99       }
100     }
101   }
102 
103   class DestinationProvider extends JdbcRolloverProvider {
104     override def apply(sft: SimpleFeatureType, context: QueryContext): Option[Filter] = {
105       val filter = coverage(sft, context).fold[Filter](Filter.EXCLUDE) { interval =>
106         val dtg = ff.property(dtgField(sft))
107         val start = ff.literal(Date.from(interval.from))
108         val end = ff.literal(Date.from(interval.through))
109         ff.and(ff.greaterOrEqual(dtg, start), ff.less(dtg, end))
110       }
111       Some(filter)
112     }
113   }

115   trait JdbcRolloverProvider extends MergedFilterProvider {
116 
117     private var source: BasicDataSource = _
118     private var sql: String = _
119     private var url: String = _
120     private var table: String = _
121 
122     override def init(config: Map[String, String]): Unit = {
123       def required(k: String): String = config.getOrElse(k, throw new IllegalArgumentException(s"Missing required key: $k"))
124 
125       this.url = required("url")
126       this.table = quoteTable(config.get("schema"), config.getOrElse("table", DefaultTableName))
127       this.sql = s"SELECT covered_from, covered_through FROM $table WHERE type_name = ?"
128       source = new BasicDataSource()
129       source.setUrl(url)
130       source.setUsername(required("username"))
131       source.setPassword(required("password"))
132       source.setMaxTotal(config.getOrElse("max-connections", "2").toInt)
133       source.setMaxIdle(1)
134       source.setMinIdle(0)
135       source.setMaxWait(Duration.ofSeconds(5))
136     }
137 
138     protected def coverage(sft: SimpleFeatureType, context: QueryContext): Option[CoverageInterval] = {
139       val state = context.state.compute(StateKey, (_, existing) => {
140         if (existing == null) {
141           CoverageState(url, table, dtgField(sft), readCoverage(sft.getTypeName))
142         } else {
143           validate(sft, existing.asInstanceOf[CoverageState])
144         }
145       })
146       state.asInstanceOf[CoverageState].coverage
147     }
148 
149     private def validate(sft: SimpleFeatureType, existing: CoverageState): CoverageState = {
150       val dtg = dtgField(sft)
151       if (existing.url != url || existing.table != table) {
152         throw new IllegalArgumentException(
153           s"RolloverFilterProviders are configured to read from different tables: " +
154             s"${existing.url} ${existing.table} vs $url $table")
155       } else if (existing.dtg != dtg) {
156         throw new IllegalArgumentException(
157           s"RolloverFilterProviders for '${sft.getTypeName}' are using different default date fields: ${existing.dtg} vs $dtg")
158       }
159       existing
160     }
161 
162     private def readCoverage(typeName: String): Option[CoverageInterval] = {
163       WithClose(source.getConnection) { cx =>
164         WithClose(cx.prepareStatement(sql)) { statement =>
165           statement.setQueryTimeout(5)
166           statement.setString(1, typeName)
167           WithClose(statement.executeQuery()) { results =>
168             if (!results.next()) {
169               None
170             } else {
171               val from = Option(results.getObject(1, classOf[LocalDateTime])).map(_.toInstant(ZoneOffset.UTC)).orNull
172               val through = Option(results.getObject(2, classOf[LocalDateTime])).map(_.toInstant(ZoneOffset.UTC)).orNull
173               Some(CoverageInterval(typeName, from, through))
174             }
175           }
176         }
177       }
178     }
179 
180     override def close(): Unit = CloseWithLogging(Option(source))
181   }
