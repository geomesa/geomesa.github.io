53   /**
54    * Gets the default dtg field from the schema, throwing an exception if it is not found
55    *
56    * @param sft simple feature type

59   protected def dtgField(sft: SimpleFeatureType): String = {
60     sft.getDtgField.orElse(firstDtg(sft)).getOrElse {

67     sft.getAttributeDescriptors.asScala.collectFirst {

73 object MergedFilterProvider {
74 
75   import FilterHelper.ff
76 
77   /**
78    * No filter
79    *
80    * @return
81    */
82   def none: MergedFilterProvider = NoProvider
83 
84   /**
85    * Create a static filter
86    *
87    * @param filter filter
88    * @return
89    */
90   def static(filter: Filter): MergedFilterProvider = StaticProvider(Option(filter))
91 
92   /**
93    * Create an inclusive 'from' date filter, e.g. `dtg >= date`
94    *
95    * @param offsetOrDate either a static date string, or an ISO-8601 duration (such as '-P1D')
96    * @return
97    */
98   def from(offsetOrDate: String): MergedFilterProvider =
99     date(offsetOrDate, FromOffsetProvider.apply, FromDateProvider.apply)
100 
101   /**
102    * Create an exclusive 'to' date filter, e.g. `dtg < date`
103    *
104    * @param offsetOrDate either a static date string, or an ISO-8601 duration (such as '-P1D')
105    * @return
106    */
107   def to(offsetOrDate: String): MergedFilterProvider =
108     date(offsetOrDate, ToOffsetProvider.apply, ToDateProvider.apply)
109 
110   /**
111    * Merge multiple filters into a single predicate, joined by AND
112    *
113    * @param providers providers to join
114    * @return
115    */
116   def and(providers: Seq[MergedFilterProvider]): MergedFilterProvider = {
117     if (providers.isEmpty) {
118       none
119     } else if (providers.size == 1) {
120       providers.head
121     } else {
122       AndProvider(providers)
123     }
124   }
125 
126   /**
127    * Create a date or offset provider
128    *
129    * @param offsetOrDate offset or date string
130    * @param offsetProvider offset provider, used if offsetOrDate is an offset
131    * @param dateProvider date provider, used if offsetOrDate is a date
132    * @return
133    */
134   private def date(
135       offsetOrDate: String,
136       offsetProvider: Duration => MergedFilterProvider,
137       dateProvider: Date => MergedFilterProvider): MergedFilterProvider = {
138     lazy val error =
139       new IllegalArgumentException(s"Invalid date filter '$offsetOrDate' - could not parse as a duration nor a date")
140     try { return offsetProvider(Duration.parse(offsetOrDate)) } catch {
141       case e: DateTimeParseException => error.addSuppressed(e)
142     }
143     try { return dateProvider(DateParsing.parseDate(offsetOrDate)) } catch {
144       case e: DateTimeParseException => error.addSuppressed(e)
145     }
146     throw error
147   }
148 
149   /**
150    * Shared context across a given query
151    *
152    * @param base original filter supplied to the query
153    * @param now timestamp query was started
154    * @param state shared state
155    */
156   case class QueryContext(
157     base: Filter,
158     now: Instant = Instant.now(),
159     state: ConcurrentHashMap[String, Any] = new ConcurrentHashMap[String, Any]

162   /**
163    * Provider that supports per-feature-type filters
164    *
165    * @param byType providers, keyed by feature type name
166    * @param default default provider, for types not in the byType map
167    */
168   case class TypedProvider(byType: Map[String, MergedFilterProvider], default: Option[MergedFilterProvider])
169       extends MergedFilterProvider {
170     override def init(config: Map[String, String]): Unit = {}
171     override def apply(sft: SimpleFeatureType, context: QueryContext): Option[Filter] =
172       byType.get(sft.getTypeName).orElse(default).flatMap(_.apply(sft, context))
173     override def close(): Unit = CloseQuietly.raise(default ++ byType.values)
174   }
175 
176   private case class AndProvider(providers: Seq[MergedFilterProvider]) extends MergedFilterProvider {
177     override def init(config: Map[String, String]): Unit = {}
178     override def apply(sft: SimpleFeatureType, context: QueryContext): Option[Filter] =
179       FilterHelper.filterListAsAnd(providers.flatMap(_.apply(sft, context)))
180     override def close(): Unit = CloseQuietly.raise(providers)
181   }

183   private case class StaticProvider(filter: Option[Filter]) extends MergedFilterProvider {
184     override def init(config: Map[String, String]): Unit = {}
185     override def apply(sft: SimpleFeatureType, context: QueryContext): Option[Filter] = filter
186     override def close(): Unit = {}
187   }

189   private case class FromOffsetProvider(offset: java.time.Duration) extends DateProvider {
190     override protected def filter(dtg: PropertyName, now: Instant): Filter =
191       ff.greaterOrEqual(dtg, ff.literal(Date.from(now.plus(offset))))
192   }
193 
194   private case class ToOffsetProvider(offset: java.time.Duration) extends DateProvider {
195     override protected def filter(dtg: PropertyName, now: Instant): Filter =
196       ff.less(dtg, ff.literal(Date.from(now.plus(offset))))
197   }
198 
199   private case class FromDateProvider(date: Date) extends DateProvider {
200     override protected def filter(dtg: PropertyName, now: Instant): Filter = ff.greaterOrEqual(dtg, ff.literal(date))
201   }
202 
203   private case class ToDateProvider(date: Date) extends DateProvider {
204     override protected def filter(dtg: PropertyName, now: Instant): Filter = ff.less(dtg, ff.literal(date))
205   }

207   private trait DateProvider extends MergedFilterProvider with LazyLogging {
208 
209     protected def filter(dtg: PropertyName, now: Instant): Filter
210 
211     override def init(config: Map[String, String]): Unit = {}
212 
213     override def apply(sft: SimpleFeatureType, context: QueryContext): Option[Filter] = {
214       val dtg = dtgField(sft)
215       val dtgFilter = filter(ff.property(dtg), context.now)
216       // check for disjoint dates between the store filter and the query filter
217       // note: don't cache this call, as it can contain things like `currentDate()` that will change per invocation
218       val disjoint = Some(FilterHelper.extractIntervals(dtgFilter, dtg)).filter(_.nonEmpty).flatMap { left =>
219         val right = FilterHelper.extractIntervals(context.base, dtg)
220         val merged = FilterValues.and[Bounds[ZonedDateTime]](Bounds.intersection)(left, right)
221         if (merged.disjoint) {
222           logger.debug(s"Suppressing query with filter (${ECQL.toCQL(context.base)}) AND (${ECQL.toCQL(dtgFilter)})")
223           Some(Filter.EXCLUDE)
224         } else {
225           None
226         }
227       }
228       disjoint.orElse(Some(dtgFilter))
229     }
230 
231     override def close(): Unit = {}
232   }
233 
234   private case object NoProvider extends MergedFilterProvider {
235     override def init(config: Map[String, String]): Unit = {}
236     override def apply(sft: SimpleFeatureType, context: QueryContext): Option[Filter] = None
237     override def close(): Unit = {}
238   }
