54   override def getCount(query: Query): Int = {
55     val total =
56       if (parallel) {
57         def getSingle(sourceAndFilter: (SimpleFeatureSource, Option[Filter])): Int = {
58           val (source, filter) = sourceAndFilter
59           source.getCount(mergeFilter(sft, query, filter))
60         }
61         val results = new CopyOnWriteArrayList[Int]()
62         sources.toList.map(s => CachedThreadPool.submit(decorator(() => results.add(getSingle(s))))).foreach(_.get)
63         results.asScala.foldLeft(0)((sum, count) => if (sum < 0 || count < 0) { -1 } else { sum + count })
64       } else {
65         // if one of our sources can't get a count (i.e. is negative), give up and return -1
66         sources.foldLeft(0) { case (sum, (source, filter)) =>
67           lazy val count = source.getCount(mergeFilter(sft, query, filter))
68           if (sum < 0 || count < 0) { -1 } else { sum + count }
69         }
70       }
71     if (query.isMaxFeaturesUnlimited) {
72       total
73     } else {
74       math.min(total, query.getMaxFeatures)
75     }
76   }
