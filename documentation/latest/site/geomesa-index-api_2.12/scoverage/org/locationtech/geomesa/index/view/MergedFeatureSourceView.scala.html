35 /**
36  * Feature source for merged data store view
37  *
38  * @param ds data store
39  * @param sources delegate feature sources
40  * @param parallel scan stores in parallel (vs sequentially)
41  * @param sft simple feature type
42  */
43 class MergedFeatureSourceView(
44     ds: MergedDataStoreView,
45     sources: Seq[MergedSource[SimpleFeatureSource]],
46     parallel: Boolean,
47     sft: SimpleFeatureType
48   ) extends SimpleFeatureSource with LazyLogging {
49 
50   import scala.collection.JavaConverters._
51 
52   lazy private val hints = Collections.unmodifiableSet(Collections.emptySet[Key])
53 
54   lazy private val capabilities = new MergedQueryCapabilities(sources.map(_.source.getQueryCapabilities))
55 
56   override def getSchema: SimpleFeatureType = sft
57 
58   override def getCount(query: Query): Int = {
59     val context = QueryContext(query.getFilter)
60     val total =
61       if (parallel) {
62         def getSingle(source: MergedSource[SimpleFeatureSource]): Int = source.source.getCount(source.filter(query, context))
63 
64         val results = new CopyOnWriteArrayList[Int]()
65         sources.toList.map(s => CachedThreadPool.submit(s.decorator(() => results.add(getSingle(s))))).foreach(_.get)
66         results.asScala.foldLeft(0)((sum, count) => if (sum < 0 || count < 0) { -1 } else { sum + count })
67       } else {
68         // if one of our sources can't get a count (i.e. is negative), give up and return -1
69         sources.foldLeft(0) { case (sum, source) =>
70           lazy val count = source.source.getCount(source.filter(query, context))
71           if (sum < 0 || count < 0) { -1 } else { sum + count }
72         }
73       }
74     if (query.isMaxFeaturesUnlimited) {
75       total
76     } else {
77       math.min(total, query.getMaxFeatures)
78     }
79   }
80 
81   override def getBounds: ReferencedEnvelope = getBounds(new Query(sft.getTypeName))
82 
83   override def getBounds(query: Query): ReferencedEnvelope = {
84     val context = QueryContext(query.getFilter)
85     def getSingle(source: MergedSource[SimpleFeatureSource]): Option[ReferencedEnvelope] =
86       Option(source.source.getBounds(source.filter(query, context)))
87 
88     val sourceBounds = if (parallel) {
89       val results = new CopyOnWriteArrayList[ReferencedEnvelope]()
90       sources.toList.map(s => CachedThreadPool.submit(s.decorator(() => getSingle(s).foreach(results.add)))).foreach(_.get)
91       results.asScala
92     } else {
93       sources.flatMap(getSingle)
94     }
95 
96     val bounds = new ReferencedEnvelope(org.locationtech.geomesa.utils.geotools.CRS_EPSG_4326)
97     sourceBounds.foreach(bounds.expandToInclude)
98     bounds
99   }

124     * @param query query

127       extends GeoMesaFeatureVisitingCollection(MergedFeatureSourceView.this, ds.stats, query) {
128 
129     private lazy val featureReader = ds.getFeatureReader(sft, Transaction.AUTO_COMMIT, query)
130 
131     override def getSchema: SimpleFeatureType = featureReader.schema
132 
133     override def reader(): FeatureReader[SimpleFeatureType, SimpleFeature] = featureReader.reader()
134 
135     override def getBounds: ReferencedEnvelope = MergedFeatureSourceView.this.getBounds(query)
136 
137     override def getCount: Int = MergedFeatureSourceView.this.getCount(query)
138 
139     override def size: Int = {
140       // note: we shouldn't return -1 here, but we don't return the actual value unless EXACT_COUNT is set
141       math.max(0, getCount)
142     }
143 
144     override def subCollection(filter: Filter): SimpleFeatureCollection = {
145       val copy = new Query(query)
146       copy.setFilter(andFilters(Seq(copy.getFilter, filter)))
147       new MergedFeatureCollection(copy)
148     }
149 
150     override def sort(order: SortBy): SimpleFeatureCollection = {
151       val copy = new Query(query)
152       copy.setSortBy(Option(copy.getSortBy).fold(Array(order))(_ :+ order): _*)
153       new MergedFeatureCollection(copy)
154     }
155 
156     override def accepts(visitor: FeatureVisitor, progress: ProgressListener): Unit = visitor match {
157       case v: GeoMesaProcessVisitor => v.execute(MergedFeatureSourceView.this, new Query(query))
158       case _ => super.accepts(visitor, progress)
159     }
160   }
