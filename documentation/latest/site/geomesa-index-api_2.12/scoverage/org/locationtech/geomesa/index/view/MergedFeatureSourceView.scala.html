30 /**
31  * Feature source for merged data store view
32  *
33  * @param ds data store
34  * @param sources delegate feature sources
35  *  @param parallel scan stores in parallel (vs sequentially)
36  * @param sft simple feature type
37  */
38 class MergedFeatureSourceView(
39     ds: MergedDataStoreView,
40     sources: Seq[MergedSource[SimpleFeatureSource]],
41     parallel: Boolean,
42     sft: SimpleFeatureType
43   ) extends SimpleFeatureSource with LazyLogging {
44

46 
47   lazy private val hints = Collections.unmodifiableSet(Collections.emptySet[Key])
48 
49   lazy private val capabilities = new MergedQueryCapabilities(sources.map(_.source.getQueryCapabilities))
50 
51   override def getSchema: SimpleFeatureType = sft
52 
53   override def getCount(query: Query): Int = {
54     val now = Instant.now() // shared current time passed to all stores for consistency in bounds calculations
55     val total =
56       if (parallel) {
57         def getSingle(source: MergedSource[SimpleFeatureSource]): Int = source.source.getCount(source.filter(query, now))
58 
59         val results = new CopyOnWriteArrayList[Int]()
60         sources.toList.map(s => CachedThreadPool.submit(s.decorator(() => results.add(getSingle(s))))).foreach(_.get)
61         results.asScala.foldLeft(0)((sum, count) => if (sum < 0 || count < 0) { -1 } else { sum + count })
62       } else {
63         // if one of our sources can't get a count (i.e. is negative), give up and return -1
64         sources.foldLeft(0) { case (sum, source) =>
65           lazy val count = source.source.getCount(source.filter(query, now))
66           if (sum < 0 || count < 0) { -1 } else { sum + count }
67         }
68       }
69     if (query.isMaxFeaturesUnlimited) {
70       total
71     } else {
72       math.min(total, query.getMaxFeatures)
73     }
74   }
75 
76   override def getBounds: ReferencedEnvelope = getBounds(new Query(sft.getTypeName))
77 
78   override def getBounds(query: Query): ReferencedEnvelope = {
79     val now = Instant.now() // shared current time passed to all stores for consistency in bounds calculations
80     def getSingle(source: MergedSource[SimpleFeatureSource]): Option[ReferencedEnvelope] =
81       Option(source.source.getBounds(source.filter(query, now)))
82 
83     val sourceBounds = if (parallel) {
84       val results = new CopyOnWriteArrayList[ReferencedEnvelope]()
85       sources.toList.map(s => CachedThreadPool.submit(s.decorator(() => getSingle(s).foreach(results.add)))).foreach(_.get)
86       results.asScala
87     } else {
88       sources.flatMap(getSingle)
89     }
90 
91     val bounds = new ReferencedEnvelope(org.locationtech.geomesa.utils.geotools.CRS_EPSG_4326)
92     sourceBounds.foreach(bounds.expandToInclude)
93     bounds
94   }
