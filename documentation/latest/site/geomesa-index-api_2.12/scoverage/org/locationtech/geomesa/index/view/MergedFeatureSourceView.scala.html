30 /**
31  * Feature source for merged data store view
32  *
33  * @param ds data store
34  * @param sources delegate feature sources
35  *  @param parallel scan stores in parallel (vs sequentially)
36  * @param decorator parallel task decorator
37  * @param sft simple feature type
38  */
39 class MergedFeatureSourceView(
40     ds: MergedDataStoreView,
41     sources: Seq[MergedSource[SimpleFeatureSource]],
42     parallel: Boolean,
43     decorator: TaskDecorator,
44     sft: SimpleFeatureType
45   ) extends SimpleFeatureSource with LazyLogging {
46 
47   import scala.collection.JavaConverters._
48 
49   lazy private val hints = Collections.unmodifiableSet(Collections.emptySet[Key])
50 
51   lazy private val capabilities = new MergedQueryCapabilities(sources.map(_.source.getQueryCapabilities))
52 
53   override def getSchema: SimpleFeatureType = sft
54 
55   override def getCount(query: Query): Int = {
56     val now = Instant.now() // shared current time passed to all stores for consistency in bounds calculations
57     val total =
58       if (parallel) {
59         def getSingle(source: MergedSource[SimpleFeatureSource]): Int = source.source.getCount(source.filter(query, now))
60 
61         val results = new CopyOnWriteArrayList[Int]()
62         sources.toList.map(s => CachedThreadPool.submit(decorator(() => results.add(getSingle(s))))).foreach(_.get)
63         results.asScala.foldLeft(0)((sum, count) => if (sum < 0 || count < 0) { -1 } else { sum + count })
64       } else {
65         // if one of our sources can't get a count (i.e. is negative), give up and return -1
66         sources.foldLeft(0) { case (sum, source) =>
67           lazy val count = source.source.getCount(source.filter(query, now))
68           if (sum < 0 || count < 0) { -1 } else { sum + count }
69         }
70       }
71     if (query.isMaxFeaturesUnlimited) {
72       total
73     } else {
74       math.min(total, query.getMaxFeatures)
75     }
76   }
77 
78   override def getBounds: ReferencedEnvelope = getBounds(new Query(sft.getTypeName))
79 
80   override def getBounds(query: Query): ReferencedEnvelope = {
81     val now = Instant.now() // shared current time passed to all stores for consistency in bounds calculations
82     def getSingle(source: MergedSource[SimpleFeatureSource]): Option[ReferencedEnvelope] =
83       Option(source.source.getBounds(source.filter(query, now)))
84 
85     val sourceBounds = if (parallel) {
86       val results = new CopyOnWriteArrayList[ReferencedEnvelope]()
87       sources.toList.map(s => CachedThreadPool.submit(decorator(() => getSingle(s).foreach(results.add)))).foreach(_.get)
88       results.asScala
89     } else {
90       sources.flatMap(getSingle)
91     }
92 
93     val bounds = new ReferencedEnvelope(org.locationtech.geomesa.utils.geotools.CRS_EPSG_4326)
94     sourceBounds.foreach(bounds.expandToInclude)
95     bounds
96   }
97 
98   override def getQueryCapabilities: QueryCapabilities = capabilities
99

116   override def removeFeatureListener(listener: FeatureListener): Unit = throw new UnsupportedOperationException()

120     *
121     * @param query query
122     */
123   private class MergedFeatureCollection(query: Query)
124       extends GeoMesaFeatureVisitingCollection(MergedFeatureSourceView.this, ds.stats, query) {
125 
126     private lazy val featureReader = ds.getFeatureReader(sft, Transaction.AUTO_COMMIT, query)
127 
128     override def getSchema: SimpleFeatureType = featureReader.schema
129 
130     override def reader(): FeatureReader[SimpleFeatureType, SimpleFeature] = featureReader.reader()
131 
132     override def getBounds: ReferencedEnvelope = MergedFeatureSourceView.this.getBounds(query)
133 
134     override def getCount: Int = MergedFeatureSourceView.this.getCount(query)
135 
136     override def size: Int = {
137       // note: we shouldn't return -1 here, but we don't return the actual value unless EXACT_COUNT is set
138       val count = getCount
139       if (count < 0) { 0 } else { count }
140     }
141   }
