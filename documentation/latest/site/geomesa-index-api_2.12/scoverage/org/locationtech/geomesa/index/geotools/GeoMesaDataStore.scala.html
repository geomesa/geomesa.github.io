120   @throws(classOf[IllegalArgumentException])
121   override protected def preSchemaCreate(sft: SimpleFeatureType): Unit = {
122     // check for old user data keys and re-map them
123     InternalConfigs.DeprecatedConfigMappings.foreach { case (from, to) =>
124       val v = sft.getUserData.remove(from)
125       if (v != null) {
126         sft.getUserData.put(to, v)
127       }
128     }
129 
130     // validate column groups
131     adapter.groups.validate(sft)
132 
133     // disable table sharing, no longer supported
134     // noinspection ScalaDeprecation
135     if (sft.isTableSharing) {
136       logger.warn("Table sharing is no longer supported - disabling table sharing")
137       sft.getUserData.remove(Configs.TableSharing)
138       sft.getUserData.remove(InternalConfigs.TableSharingPrefix)
139     }
140 
141     // configure the indices to use
142     if (sft.getIndices.isEmpty) {
143       val indices = GeoMesaFeatureIndexFactory.indices(sft)
144       if (indices.isEmpty) {
145         throw new IllegalArgumentException("There are no available indices that support the schema " +
146             SimpleFeatureTypes.encodeType(sft))
147       }
148       sft.setIndices(indices)
149     }
150 
151     // try to create the indices up front to ensure they are valid for the sft
152     GeoMesaFeatureIndexFactory.create(this, sft, sft.getIndices)
153 
154     // remove the enabled indices after configuration so we don't persist them
155     sft.getUserData.remove(SimpleFeatureTypes.Configs.EnabledIndices)
156     // remove any 'index' flags in the attribute metadata, they have already been captured in the indices above
157     sft.getAttributeDescriptors.asScala.foreach(_.getUserData.remove(AttributeOptions.OptIndex))
158 
159     // for partitioned schemas, persist the table partitioning keys
160     if (sft.isPartitioned) {
161       InternalConfigs.PartitionConfigMappings.foreach { case (from, to) =>
162         Option(sft.getUserData.get(from)).foreach(sft.getUserData.put(to, _))
163       }
164       InternalConfigs.PartitionConfigPrefixMappings.foreach { case (from, to) =>
165         sft.getUserData.asScala.toMap.collect {
166           case (k: String, v) if k.startsWith(from) => sft.getUserData.put(to + k.substring(from.length), v)
167         }
168       }
169     }
170 
171     // set stats enabled based on the data store config if not explicitly set
172     if (!sft.getUserData.containsKey(SimpleFeatureTypes.Configs.StatsEnabled)) {
173       sft.setStatsEnabled(config.generateStats)
174     }
175 
176     sft.getFeatureExpiration // validate any configured age-off
177   }

201   // create the new index tables (if not using partitioned tables)
202   override protected def onSchemaUpdated(sft: SimpleFeatureType, previous: SimpleFeatureType): Unit = {
203     val partitioned = TablePartition.partitioned(sft)
204 
205     // check for column renaming
206     val colMap = getColumnMap(previous, sft).map { case (k, v) => (v, k) }
207 
208     // rewrite the table name keys for the renamed cols
209     if (colMap.nonEmpty) {
210       sft.getIndices.foreach { i =>
211         if (i.attributes.exists(colMap.contains)) {
212           val prev = i.copy(attributes = i.attributes.map(a => colMap.getOrElse(a, a)))
213           val old = manager.index(previous, GeoMesaFeatureIndex.identifier(prev))
214           val index = GeoMesaFeatureIndexFactory.create(this, sft, Seq(i)).headOption.getOrElse {
215             throw new IllegalArgumentException(
216               s"Error configuring new feature index: ${GeoMesaFeatureIndex.identifier(i)}")
217           }
218           val partitions = if (!partitioned) { Seq(None) } else {
219             // have to use the old table name key but the new sft name for looking up the partitions
220             val tableNameKey = old.tableNameKey(Some(""))
221             val offset = tableNameKey.length
222             metadata.scan(sft.getTypeName, tableNameKey).map { case (k, _) => Some(k.substring(offset)) }
223           }
224           partitions.foreach { p =>
225             metadata.read(sft.getTypeName, old.tableNameKey(p)).foreach { v =>
226               metadata.insert(sft.getTypeName, index.tableNameKey(p), v)
227             }
228           }
229         }
230       }
231     }
232 
233     // configure any new indices
234     if (partitioned) {
235       logger.debug("Delaying creation of partitioned indices")
236     } else {
237       logger.debug(s"Ensuring indices ${manager.indices(sft).map(_.identifier).mkString(", ")}")
238       manager.indices(sft).toList.map(i => CachedThreadPool.submit(() => adapter.createTable(i, None, i.getSplits(None)))).foreach(_.get)
239     }
240 
241     // update stats
242     if (previous.statsEnabled) {
243       if (!sft.statsEnabled) {
244         stats.writer.clear(previous)
245       } else if (sft.getTypeName != previous.getTypeName || colMap.nonEmpty) {
246         stats.writer.rename(sft, previous)
247       }
248     }
249 
250     // rename tables to match the new sft name
251     if (sft.getTypeName != previous.getTypeName || sft.getIndices != previous.getIndices) {
252       if (FastConverter.convertOrElse[java.lang.Boolean](sft.getUserData.get(Configs.UpdateRenameTables), false)) {
253         manager.indices(sft).foreach { index =>
254           val partitions = if (partitioned) { index.getPartitions.map(Option.apply) } else { Seq(None) }
255           partitions.foreach { partition =>
256             val key = index.tableNameKey(partition)
257             metadata.read(sft.getTypeName, key).foreach { table =>
258               metadata.remove(sft.getTypeName, key)
259               val renamed = index.configureTableName(partition, adapter.tableNameLimit)
260               if (renamed != table) {
261                 logger.debug(s"Renaming table from '$table' to '$renamed'")
262                 adapter.renameTable(table, renamed)
263               }
264             }
265           }
266         }
267       }
268     }
269   }

271   // delete the index tables
272   override protected def onSchemaDeleted(sft: SimpleFeatureType): Unit = {
273     def deleteFull(index: GeoMesaFeatureIndex[_, _]): Unit =
274       adapter.deleteTables(index.deleteTableNames(None))
275     def deleteShared(index: GeoMesaFeatureIndex[_, _]): Unit = {
276       if (index.keySpace.sharing.isEmpty) { deleteFull(index) } else {
277         adapter.clearTables(index.deleteTableNames(None), Some(index.keySpace.sharing))
278       }
279     }
280 
281     val indices = manager.indices(sft).toList
282     // noinspection ScalaDeprecation
283     if (sft.isTableSharing && getTypeNames.exists(t => t != sft.getTypeName && getSchema(t).isTableSharing)) {
284       indices.map(i => CachedThreadPool.submit(() => deleteShared(i))).foreach(_.get)
285     } else {
286       indices.map(i => CachedThreadPool.submit(() => deleteFull(i))).foreach(_.get)
287     }
288     if (sft.statsEnabled) {
289       stats.writer.clear(sft)
290     }
291   }

295   /**
296    * @see org.geotools.api.data.DataStore#getSchema(java.lang.String)
297    * @param typeName feature type name
298    * @return feature type, or null if it does not exist
299    */
300   override def getSchema(typeName: String): SimpleFeatureType = {
301     var sft = super.getSchema(typeName)
302     if (sft != null) {
303       // ensure index metadata is correct
304       if (sft.getIndices.exists(i => i.attributes.isEmpty && i.name != IdIndex.name)) {
305         sft = SimpleFeatureTypes.mutable(sft)
306         // migrate index metadata to standardized versions and attributes
307         transitionIndices(sft)
308         sft = SimpleFeatureTypes.immutable(sft)
309         // validate indices
310         try { manager.indices(sft) } catch {
311           case NonFatal(e) =>
312             throw new IllegalStateException(s"The schema ${sft.getTypeName} was written with a older " +
313                 "version of GeoMesa that is no longer supported. You may continue to use an older client, or " +
314                 s"manually edit the metadata for '${InternalConfigs.IndexVersions}' to exclude the invalid indices.", e)
315         }
316       } else {
317         // validate indices
318         try { manager.indices(sft) } catch {
319           case NonFatal(e) =>
320             val versions = sft.getIndices.map(i => s"${i.name}:${i.version}").mkString(",")
321             val available = GeoMesaFeatureIndexFactory.all().map(i => s"${i.name}:${i.version}").mkString(",")
322             logger.error(s"Trying to access schema ${sft.getTypeName} with invalid index versions '$versions' - " +
323                 s"available indices are '$available'", e)
324             throw new IllegalStateException(s"The schema ${sft.getTypeName} was written with a newer " +
325                 "version of GeoMesa than this client can handle. Please ensure that you are using the " +
326                 "same GeoMesa jar versions across your entire workflow. For more information, see " +
327                 "https://www.geomesa.org/documentation/stable/user/upgrade.html")
328         }
329       }
330 
331       // check for sft-level stats flag and set it if not present
332       if (!sft.getUserData.containsKey(SimpleFeatureTypes.Configs.StatsEnabled)) {
333         val extra = Collections.singletonMap(SimpleFeatureTypes.Configs.StatsEnabled, config.generateStats.toString)
334         sft = SimpleFeatureTypes.immutable(sft, extra)
335       }
336 
337       // get the remote version if it's available, but don't wait for it
338       GeoMesaDataStore.versions.get(VersionKey(this)).getNow(Right(None)) match {
339         case Left(e) => throw e
340         case Right(version) =>
341           version.foreach { v =>
342             val userData = Collections.singletonMap[AnyRef, AnyRef](InternalConfigs.RemoteVersion, v.toString)
343             sft = SimpleFeatureTypes.immutable(sft, userData)
344           }
345       }
346     }
347     sft
348   }

466    * @return
467    */
468   def checkSchemaCompatibility(typeName: String, sft: SimpleFeatureType): SchemaCompatibility = {
469     val previous = getSchema(typeName)
470     if (previous == null) {
471       new SchemaCompatibility.DoesNotExist(this, sft)
472     } else {
473       validateSchemaUpdate(previous, sft) match {
474         case Right(TypeComparison.Compatible(extension, renamed, _)) =>
475           val copy = SimpleFeatureTypes.copy(sft)
476           updateSchemaUserData(copy, previous)
477           if (!extension && !renamed && copy.getUserData == previous.getUserData) {
478             SchemaCompatibility.Unchanged
479           } else {
480             new SchemaCompatibility.Compatible(this, typeName, sft)
481           }
482 
483         case Left(e) =>
484           SchemaCompatibility.Incompatible(e)
485       }
486     }
487   }

491   /**
492    * Updates the user data for a schema update prior to persistence. Handles converting existing
493    * user data based on new/updated attributes, copying existing user data over, etc. Feature
494    * type will be updated in place
495    *
496    * @param sft schema update
497    * @param previous existing schema
498    * @return
499    */
500   private def updateSchemaUserData(sft: SimpleFeatureType, previous: SimpleFeatureType): Unit = {
501 
502     // check for column renaming
503     val colMap = getColumnMap(previous, sft)
504 
505     def remapCol(name: String): String = colMap.getOrElse(name, name)
506 
507     val indices = {
508       // set directly in the sft
509       val explicitIndices = (previous.getIndices ++ sft.getIndices).map(i => i.copy(attributes = i.attributes.map(remapCol)))
510       // set in index flags or geomesa.indices.enabled
511       val configuredIndices =
512         // don't add in a fid index unless explicitly enabled
513         if (sft.getUserData.get(Configs.EnableFidIndex) == null &&
514             Option(sft.getUserData.get(Configs.EnabledIndices))
515               .forall(!_.toString.split(",").map(_.trim.toLowerCase(Locale.US)).exists(_.matches(s"^${IdIndex.name}:?")))) {
516           GeoMesaFeatureIndexFactory.indices(sft).filterNot(_.name == IdIndex.name)
517         } else {
518           GeoMesaFeatureIndexFactory.indices(sft)
519         }
520       // get the unique indices - note: ignores index version
521       (explicitIndices ++ configuredIndices).foldLeft(Seq.empty[IndexId]) { (sum, next) =>
522         if (sum.exists(i => i.name == next.name && i.attributes == next.attributes)) {
523           sum
524         } else {
525           sum :+ next
526         }
527       }
528     }
529     sft.setIndices(indices)
530 
531     // preserve any existing user data but overwrite any keys we redefine
532     val userData = new java.util.HashMap[AnyRef, AnyRef](previous.getUserData)
533     userData.putAll(sft.getUserData)
534     sft.getUserData.putAll(userData)
535     // remove enabled indices as we don't need to persist it
536     sft.getUserData.remove(Configs.EnabledIndices)
537     // remove any null/empty keys as a way to delete existing user data
538     sft.getUserData.asScala.collect { case (k, null | "") => k }.foreach(sft.getUserData.remove)
539     // remove any 'index' flags in the attribute user data - we need them above for checking enabled indices
540     sft.getAttributeDescriptors.asScala.foreach(_.getUserData.remove(AttributeOptions.OptIndex))
541   }

570   private val loader: AsyncCacheLoader[VersionKey, Either[Exception, Option[SemanticVersion]]] =
571     new CacheLoader[VersionKey, Either[Exception, Option[SemanticVersion]]]() {
572       override def load(key: VersionKey): Either[Exception, Option[SemanticVersion]] = {
573         if (CheckDistributedVersion.toBoolean.contains(false)) { Right(None) } else {
