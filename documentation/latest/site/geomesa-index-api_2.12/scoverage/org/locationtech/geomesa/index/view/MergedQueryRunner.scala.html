37 /**
38  * Query runner for merging results from multiple stores
39  *
40  * @param ds merged data store
41  * @param stores delegate stores
42  * @param deduplicate deduplicate the results between stores
43  * @param parallel run scans in parallel (vs sequentially)
44  */
45 class MergedQueryRunner(
46     ds: HasGeoMesaStats,
47     stores: Seq[MergedSource[Queryable]],
48     deduplicate: Boolean,
49     parallel: Boolean
50   ) extends QueryRunner with LazyLogging {
51 
52   import org.locationtech.geomesa.index.conf.QueryHints.RichHints
53 
54   // query interceptors are handled by the individual data stores
55   override protected val interceptors: QueryInterceptorFactory = QueryInterceptorFactory.empty()
56 
57   override protected def tags(typeName: String): Tags = TagUtils.typeNameTag(typeName).and("store", "merged", "catalog", "")
58 
59   override protected def getQueryPlans(sft: SimpleFeatureType, query: Query, explain: Explainer): Seq[QueryPlan] = {
60     // TODO deduplicate arrow, bin, density queries...
61     val hints = query.getHints
62     val toFeatures = new IdentityResultsToFeatures(hints.getReturnSft)
63     val plan = if (hints.isArrowQuery) {
64       val scan = () => scanner(sft, query, org.locationtech.geomesa.arrow.ArrowEncodedSft, new ArrowProcessor(_, hints))
65       val reducer = new DeltaReducer(hints.getTransformSchema.getOrElse(sft), hints, sorted = false)
66       MergedQueryPlan(scan, toFeatures, Some(reducer), hints.getMaxFeatures)
67     } else if (hints.isBinQuery) {
68       if (query.getSortBy != null && !query.getSortBy.isEmpty) {
69         logger.warn("Ignoring sort for BIN query")
70       }
71       val scan = () => scanner(sft, query, BinaryOutputEncoder.BinEncodedSft, new BinProcessor(_, hints))
72       MergedQueryPlan(scan, toFeatures, None, hints.getMaxFeatures)
73     } else if (hints.isDensityQuery) {
74       val scan = () => scanner(sft, query, DensityScan.DensitySft, new DensityProcessor(_, hints))
75       MergedQueryPlan(scan, toFeatures, None, hints.getMaxFeatures)
76     } else if (hints.isStatsQuery) {
77       val scan = () => scanner(sft, query, StatsScan.StatsSft, new StatsProcessor(_, hints))
78       val reducer = StatsScan.StatsReducer(sft, hints)
79       MergedQueryPlan(scan, toFeatures, Some(reducer), hints.getMaxFeatures)
80     } else {
81       // we assume the delegate stores can handle normal transforms/etc appropriately
82       val scanner = () => {
83         val readers = getReaders(query).map(CloseableIterator(_))
84         val deduped = if (deduplicate) {
85           // we re-use the feature id cache across readers
86           val cache = scala.collection.mutable.HashSet.empty[String]
87           readers.map(new DeduplicatingSimpleFeatureIterator(_, cache))
88         } else {
89           readers
90         }
91         val sorted = hints.getSortFields match {
92           case None => doParallelScan(deduped)
93           // the delegate stores should sort their results, so we can sort merge them
94           case Some(sort) => new SortedMergeIterator(deduped)(SimpleFeatureOrdering(hints.getReturnSft, sort))
95         }
96         sorted
97       }
98       MergedQueryPlan(scanner, toFeatures, None, hints.getMaxFeatures)
99     }
100     Seq(plan)
101   }
102 
103   // query each delegate store
104   private def getReaders(query: Query): Seq[FeatureReader[SimpleFeatureType, SimpleFeature]] = {
105     val now = Instant.now() // shared current time passed to all stores for consistency in bounds calculations
106     stores.map { store =>
107       val copy = new Query(query) // make sure to coy the hints so they aren't shared
108       // suppress the reduce step for gm stores so that we can do the merge here
109       copy.getHints.put(QueryHints.Internal.SKIP_REDUCE, java.lang.Boolean.TRUE)
110       copy.setFilter(store.filter(query.getTypeName, query.getFilter, now))
111       store.source.getFeatureReader(copy, Transaction.AUTO_COMMIT)
112     }
113   }
114 
115   private def scanner(
116       sft: SimpleFeatureType,
117       query: Query,
118       encodedType: SimpleFeatureType,
119       fallbackProcessor: SimpleFeatureType => LocalScanProcessor): CloseableIterator[SimpleFeature] = {
120     val readers = getReaders(query).map { reader =>
121       val schema = reader.getFeatureType
122       if (schema == encodedType) {
123         // processing has been handled by the store already
124         CloseableIterator(reader)
125       } else {
126         // the store just returned normal features, do the processing here
127         val copy = SimpleFeatureTypes.immutable(schema, sft.getUserData) // copy default dtg, etc if necessary
128         fallbackProcessor(copy).apply(CloseableIterator(reader))
129       }
130     }
131     doParallelScan(readers)
132   }
133 
134   private def doParallelScan(readers: Seq[CloseableIterator[SimpleFeature]]): CloseableIterator[SimpleFeature] = {
135     if (parallel) {
136       // not truly parallel but should kick them all off up front
137       readers.foreach(_.hasNext)
138     }
139     CloseableIterator(readers.iterator).flatMap(i => i)
140   }
141 }

167       explainer.pushLevel("MergedQueryPlan:")
168       explainer(s"Reduce: ${reducer.getOrElse("none")}")
169       explainer.popLevel()
170     }
171   }
