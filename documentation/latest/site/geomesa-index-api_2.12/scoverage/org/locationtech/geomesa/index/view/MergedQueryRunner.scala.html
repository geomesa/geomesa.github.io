44 /**
45  * Query runner for merging results from multiple stores
46  *
47  * @param ds merged data store
48  * @param stores delegate stores
49  * @param deduplicate deduplicate the results between stores
50  * @param parallel run scans in parallel (vs sequentially)

68   override protected def getQueryPlans(sft: SimpleFeatureType, query: Query, explain: Explainer): Seq[QueryPlan] = {
69     // TODO deduplicate arrow, bin, density queries...
70     val hints = query.getHints
71     val toFeatures = new IdentityResultsToFeatures(hints.getReturnSft)
72     val plan = if (hints.isArrowQuery) {
73       val scan = () => scanner(sft, query, org.locationtech.geomesa.arrow.ArrowEncodedSft, new ArrowProcessor(_, hints), copyFeatures = true)
74       val reducer = new DeltaReducer(hints.getTransformSchema.getOrElse(sft), hints, sorted = false)
75       MergedQueryPlan(scan, toFeatures, Some(reducer), hints.getMaxFeatures)
76     } else if (hints.isBinQuery) {
77       if (query.getSortBy != null && !query.getSortBy.isEmpty) {
78         logger.warn("Ignoring sort for BIN query")
79       }
80       val scan = () => scanner(sft, query, BinaryOutputEncoder.BinEncodedSft, new BinProcessor(_, hints))
81       MergedQueryPlan(scan, toFeatures, None, hints.getMaxFeatures)
82     } else if (hints.isDensityQuery) {
83       val scan = () => scanner(sft, query, DensityScan.DensitySft, new DensityProcessor(_, hints))
84       MergedQueryPlan(scan, toFeatures, None, hints.getMaxFeatures)
85     } else if (hints.isStatsQuery) {
86       val scan = () => scanner(sft, query, StatsScan.StatsSft, new StatsProcessor(_, hints))
87       val reducer = StatsScan.StatsReducer(sft, hints)
88       MergedQueryPlan(scan, toFeatures, Some(reducer), hints.getMaxFeatures)
89     } else {
90       // we assume the delegate stores can handle normal transforms/etc appropriately
91       val scanner = () => {
92         val readers = getReaders(query).map(CloseableIterator(_))
93         val sorted = hints.getSortFields match {
94           case None => doParallelScan(readers)
95             // the delegate stores should sort their results, so we can sort merge them
96           case Some(sort) => new SortedMergeIterator(openReaders(readers))(SimpleFeatureOrdering(hints.getReturnSft, sort))
97         }
98         val deduped = if (deduplicate) { new DeduplicatingSimpleFeatureIterator(sorted) } else { sorted }
99         deduped
100       }
101       MergedQueryPlan(scanner, toFeatures, None, hints.getMaxFeatures)
102     }
103     Seq(plan)
104   }

106   // query each delegate store - note, it's important that this is an iterator (lazily evaluated) in order for things to be
107   // cleaned up correctly
108   private def getReaders(query: Query): Iterator[FeatureReader[SimpleFeatureType, SimpleFeature]] = {
109     val now = Instant.now() // shared current time passed to all stores for consistency in bounds calculations
110     stores.iterator.map { store =>
111       val copy = new Query(query) // make sure to coy the hints so they aren't shared
112       // suppress the reduce step for gm stores so that we can do the merge here
113       copy.getHints.put(QueryHints.Internal.SKIP_REDUCE, java.lang.Boolean.TRUE)
114       copy.setFilter(store.filter(query.getTypeName, query.getFilter, now))
115       store.source.getFeatureReader(copy, Transaction.AUTO_COMMIT)
116     }
117   }

119   private def scanner(
120       sft: SimpleFeatureType,
121       query: Query,
122       encodedType: SimpleFeatureType,
123       fallbackProcessor: SimpleFeatureType => LocalScanProcessor,
124       copyFeatures: Boolean = false): CloseableIterator[SimpleFeature] = {
125     val readers = getReaders(query).map { reader =>
126       val schema = reader.getFeatureType
127       if (schema == encodedType) {
128         // processing has been handled by the store already
129         CloseableIterator(reader)
130       } else {
131         // the store just returned normal features, do the processing here
132         val copy = SimpleFeatureTypes.immutable(schema, sft.getUserData) // copy default dtg, etc if necessary
133         fallbackProcessor(copy).apply(CloseableIterator(reader))
134       }
135     }
136     doParallelScan(readers, copyFeatures)
137   }
138 
139   private def doParallelScan(
140       readers: Iterator[CloseableIterator[SimpleFeature]],
141       copyFeatures: Boolean = false): CloseableIterator[SimpleFeature] = {
142     if (parallel) {
143       new ParallelScanIterator(openReaders(readers), decorator, copyFeatures, 32768)
144     } else {
145       CloseableIterator(readers).flatMap(i => i)
146     }
147   }

190   private class ParallelScanIterator(
191       readers: Seq[CloseableIterator[SimpleFeature]],
192       decorator: TaskDecorator,
193       copyFeatures: Boolean,
194       batchSize: Int,
195     ) extends CloseableIterator[SimpleFeature] {
196 
197     private val queue = new LinkedBlockingQueue[SimpleFeature](batchSize)
198     private val running = new AtomicInteger(readers.length)
199     private val error = new AtomicReference[Throwable]()
200 
201     private val ex = new CachedThreadPool(readers.length)
202     private val tasks = {
203       val tasks = readers.map(r => ex.submit(decorator(new ParallelScanTask(r))))
204       ex.shutdown()
205       tasks
206     }
207 
208     private var staged: SimpleFeature = _
209 
210     @tailrec
211     override final def hasNext: Boolean = {
212       if (staged != null) {
213         return true
214       }
215       val e = error.getAndSet(null)
216       if (e != null) {
217         throw e
218       }
219       staged = queue.poll()
220       if (staged != null) {
221         return true
222       } else if (running.get() == 0 || ex.isTerminated) {
223         staged = queue.poll()
224         return staged != null
225       }
226       staged = queue.poll(10, TimeUnit.MILLISECONDS)
227       hasNext
228     }
229 
230     override def next(): SimpleFeature = {
231       if (hasNext) {
232         val res = staged
233         staged = null
234         res
235       } else {
236         Iterator.empty.next()
237       }
238     }
239 
240     override def close(): Unit = {
241       var e: Throwable = null
242       def addError(err: Throwable): Unit = if (e == null) { e = err } else { e.addSuppressed(err) }
243       try {
244         tasks.foreach(_.cancel(true))
245         ex.shutdownNow()
246         e = error.getAndSet(null)
247       } finally {
248         CloseQuietly(readers).foreach(addError)
249       }
250       if (e != null) {
251         throw e
252       }
253     }
254 
255     private class ParallelScanTask(features: CloseableIterator[SimpleFeature]) extends Runnable {
256       override def run(): Unit = {
257         try {
258           val toQueue = if (copyFeatures) { features.map(ScalaSimpleFeature.copy) } else { features }
259           // don't use foreach as that will close the iterator, which may cause duplicate close errors
260           while (toQueue.hasNext) {
261             queue.put(toQueue.next())
262           }
263         } catch {
264           case _: InterruptedException => // no-op
265           case e: Throwable => error.getAndUpdate(cur => if (cur == null) { e } else { cur.addSuppressed(e); cur })
266         } finally {
267           running.decrementAndGet()
268         }
269       }
270     }
271   }
