28 /**
29  * Merged querying against multiple data stores
30  *
31  * @param stores delegate stores
32  * @param deduplicate deduplicate results across stores
33  * @param parallel execute queries in parallel
34  * @param namespace namespace
35  */
36 class MergedDataStoreView(
37     val stores: Seq[MergedSource[DataStore]],
38     deduplicate: Boolean,
39     parallel: Boolean,
40     namespace: Option[String] = None
41   ) extends MergedDataStoreSchemas(stores.map(_.source), namespace) with HasGeoMesaFeatureReader with HasGeoMesaStats {
42 
43   require(stores.nonEmpty, "No delegate stores configured")
44 
45   private[view] val runner =
46     new MergedQueryRunner(this, stores.map(s => s.copy[Queryable](DataStoreQueryable(s.source))), deduplicate, parallel)
47 
48   override val stats: GeoMesaStats = new MergedStats(stores, parallel)
49 
50   override def getFeatureSource(name: Name): SimpleFeatureSource = getFeatureSource(name.getLocalPart)
51 
52   override def getFeatureSource(typeName: String): SimpleFeatureSource = {
53     val sources = stores.map(s => s.copy(s.source.getFeatureSource(typeName)))
54     new MergedFeatureSourceView(this, sources, parallel, getSchema(typeName))
55   }
56 
57   override def getFeatureReader(query: Query, transaction: Transaction): SimpleFeatureReader =
58     getFeatureReader(getSchema(query.getTypeName), transaction, query).reader()
59 
60   override private[geomesa] def getFeatureReader(
61       sft: SimpleFeatureType,
62       transaction: Transaction,
63       query: Query): GeoMesaFeatureReader =
64     GeoMesaFeatureReader(sft, query, runner, None)
65 
66   override def dispose(): Unit = {
67     try { super.dispose() } finally {
68       CloseQuietly.raise(stores.map(_.provider))
69     }
70   }
71 }

75   /**
76    * A merged object, with associated filters
77    *
78    * @param source object
79    * @param provider per-source filtering
80    * @param decorator parallel task decorator
81    * @tparam T type binding
82    */
83   case class MergedSource[T](
84       source: T,
85       provider: MergedFilterProvider = MergedFilterProvider.none,
86       decorator: TaskDecorator = new TaskDecorator(None)) {
87 
88     /**
89      * The filter to apply to this source, based on the current date
90      *
91      * @param query query
92      * @param context shared query context
93      * @param ev feature type loader
94      * @return
95      */
96     def filter(query: Query, context: QueryContext)(implicit ev: MergedSourceSft[T]): Query = {
97       val q = new Query(query)
98       q.setFilter(filter(query.getTypeName, context))
99       q
100     }
101 
102     /**
103      * The filter to apply to this source, based on the current date
104      *
105      * @param typeName feature type name
106      * @param context shared query context
107      * @param ev feature type loader
108      * @return
109      */
110     def filter(typeName: String, context: QueryContext)(implicit ev: MergedSourceSft[T]): Filter = {
111       val base = context.base
112       provider.apply(ev.sft(source, typeName), context).fold(base) { filter =>
113         if (base == Filter.INCLUDE || filter == Filter.EXCLUDE) { filter } else { FilterHelper.ff.and(base, filter) }
114       }
115     }
116   }
117 
118   object MergedSource {
119 
120     /**
121      * Helper for extracting a feature type from a merged source
122      *
123      * @tparam T type binding
124      */
125     trait MergedSourceSft[T] {
126       def sft(source: T, typeName: String): SimpleFeatureType
127     }
128 
129     implicit object DataSourceSft extends MergedSourceSft[DataStore] {
130       override def sft(source: DataStore, typeName: String): SimpleFeatureType = source.getSchema(typeName)
131     }
132 
133     implicit object FeatureSourceSft extends MergedSourceSft[SimpleFeatureSource] {
134       override def sft(source: SimpleFeatureSource, typeName: String): SimpleFeatureType = {
135         val sft = source.getSchema
136         require(sft != null && sft.getTypeName == typeName)
137         sft
138       }
139     }
140 
141     implicit object QueryableSft extends MergedSourceSft[Queryable] {
142       override def sft(source: Queryable, typeName: String): SimpleFeatureType = source.sft(typeName)
143     }
144   }
145 
146   class TaskDecorator(op: Option[UnaryOperator[Runnable]]) {
147     def apply(task: Runnable): Runnable = op.fold(task)(_.apply(task))
148   }
