31 /**
32  * Merged querying against multiple data stores
33  *
34  * @param stores delegate stores
35  * @param deduplicate deduplicate results across stores
36  * @param parallel execute queries in parallel
37  * @param taskDecorator parallel task decorator
38  * @param namespace namespace
39  */
40 class MergedDataStoreView(
41     val stores: Seq[(DataStore, Option[Filter])],
42     deduplicate: Boolean,
43     parallel: Boolean,
44     taskDecorator: Option[UnaryOperator[Runnable]] = None,

46   ) extends MergedDataStoreSchemas(stores.map(_._1), namespace) with HasGeoMesaFeatureReader with HasGeoMesaStats {
47 
48   require(stores.nonEmpty, "No delegate stores configured")

50   private val decorator = new TaskDecorator(taskDecorator)

53     new MergedQueryRunner(this, stores.map { case (ds, f) => DataStoreQueryable(ds) -> f }, deduplicate, parallel)

92     override def getCount(sft: SimpleFeatureType, filter: Filter, exact: Boolean, queryHints: Hints): Option[Long] = {
93       // note: unlike most methods in this class, this will return if any of the merged stores provide a response
94       def getSingle(statAndFilter: (GeoMesaStats, Option[Filter])): Option[Long] =
95         statAndFilter._1.getCount(sft, mergeFilter(sft, filter, statAndFilter._2), exact, queryHints)
96 
97       if (parallel) {
98         val results = new CopyOnWriteArrayList[Long]()
99         stats.toList.map(s => CachedThreadPool.submit(decorator(() => getSingle(s).foreach(results.add)))).foreach(_.get)
100         results.asScala.reduceLeftOption(_ + _)
101       } else {
102         stats.flatMap(getSingle).reduceLeftOption(_ + _)
103       }
104     }

181     private def merge[T <: Stat](query: (GeoMesaStats, Option[Filter]) => Option[T]): Option[T] = {
182       if (parallel) {
183         val results = new CopyOnWriteArrayList[Option[T]]()
184         stats.toList.map { case (s, f) => CachedThreadPool.submit(decorator(() => results.add(query(s, f)))) }.foreach(_.get)
185         results.asScala.reduceLeft((res, next) => for { r <- res; n <- next } yield { (r + n).asInstanceOf[T] })
186       } else {
187         // lazily evaluate each stat as we only return Some if all the child stores do
188         val head = query(stats.head._1, stats.head._2)
189         stats.tail.foldLeft(head) { case (result, (stat, filter)) =>
190           for { r <- result; n <- query(stat, filter) } yield { (r + n).asInstanceOf[T] }
191         }
192       }
193     }
194   }

198       writers.map(_.analyze(sft)).reduceLeft[Seq[Stat]] { case (left, right) =>

200       }

208     override def clear(sft: SimpleFeatureType): Unit = writers.foreach(_.clear(sft))
209   }

215     override def close(): Unit = CloseWithLogging(updaters)
