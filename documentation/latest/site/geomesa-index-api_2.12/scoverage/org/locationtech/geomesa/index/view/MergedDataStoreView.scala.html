83     /**
84      * Create a date filter
85      *
86      * @param offsetOrDate either an offset (PT1D, e.g. a java duration string) or a static date (2026-01-01T00:00:00.000Z)
87      * @return
88      */
89     def apply(offsetOrDate: String): DateFilter = {
90       lazy val error =
91         new IllegalArgumentException(s"Invalid date filter '$offsetOrDate' - could not parse as a duration nor a date")
92       try {
93         return RelativeDate(offsetOrDate)
94       } catch {
95         case e: DateTimeParseException => error.addSuppressed(e)
96       }
97       try {
98         return StaticDate(offsetOrDate)
99       } catch {
100         case e: DateTimeParseException => error.addSuppressed(e)
101       }
102       throw error
103     }
104 
105     case class StaticDate(date: Date) extends DateFilter {
106       override def toDate(currentDate: Instant): Date = date
107     }
108 
109     object StaticDate {
110       @throws(classOf[DateTimeParseException])
111       def apply(date: String): StaticDate = StaticDate(DateParsing.parseDate(date))
112     }
113 
114     case class RelativeDate(offset: java.time.Duration) extends DateFilter {
115       override def toDate(currentDate: Instant): Date = Date.from(currentDate.plus(offset))
116     }
117 
118     object RelativeDate {
119       @throws(classOf[DateTimeParseException])
120       def apply(offset: String): RelativeDate = RelativeDate(java.time.Duration.parse(offset))
121     }
122   }

124   /**
125    * A merged object, with associated filters
126    *
127    * @param source object
128    * @param sourceFilter arbitrary filter, keyed by type name
129    * @param from start date filter, keyed by type name
130    * @param to end date filter, keyed by type name
131    * @tparam T type binding
132    */
133   case class MergedSource[T](
134       source: T,
135       sourceFilter: TypeOption[Filter] = TypeOption.empty,
136       from: TypeOption[DateFilter] = TypeOption.empty,
137       to: TypeOption[DateFilter] = TypeOption.empty,
138       decorator: TaskDecorator = new TaskDecorator(None)) {
139 
140     import FilterHelper.ff
141 
142     import scala.collection.JavaConverters._
143 
144     /**
145      * The filter to apply to this source, based on the current date
146      *
147      * @param query query
148      * @param now current timestamp, for consistency across multiple stores
149      * @param ev feature type loader
150      * @return
151      */
152     def filter(query: Query, now: Instant)(implicit ev: MergedSourceSft[T]): Query = {
153       val q = new Query(query)
154       q.setFilter(filter(query.getTypeName, query.getFilter, now))
155       q
156     }
157 
158     /**
159      * The filter to apply to this source, based on the current date
160      *
161      * @param typeName feature type name
162      * @param base user-supplied filter
163      * @param now current timestamp, for consistency across multiple stores
164      * @param ev feature type loader
165      * @return
166      */
167     def filter(typeName: String, base: Filter, now: Instant)(implicit ev: MergedSourceSft[T]): Filter = {
168       val dtg = dtgField(ev, typeName)
169       val fromFilter = from(typeName).map(d => ff.greaterOrEqual(ff.property(dtg), ff.literal(d.toDate(now))))
170       val toFilter = to(typeName).map(d => ff.less(ff.property(dtg), ff.literal(d.toDate(now))))
171       val storeFilter = FilterHelper.filterListAsAnd(sourceFilter(typeName).toSeq ++ fromFilter ++ toFilter)
172       storeFilter match {
173         case None => base
174         case Some(f) if base == Filter.INCLUDE => f
175         case Some(f) if dtg == null => andFilters(Seq(base, f))
176         case Some(f) =>
177           // check for disjoint dates between the store filter and the query filter
178           // note: don't cache this call, as it can contain things like `currentDate()` that will change per invocation
179           val disjoint = Some(FilterHelper.extractIntervals(f, dtg)).filter(_.nonEmpty).flatMap { left =>
180             val right = FilterHelper.extractIntervals(base, dtg)
181             val merged = FilterValues.and[Bounds[ZonedDateTime]](Bounds.intersection)(left, right)
182             if (merged.disjoint) {
183               logger.debug(s"Suppressing query with filter (${ECQL.toCQL(base)}) AND (${ECQL.toCQL(f)})")
184               Some(Filter.EXCLUDE)
185             } else {
186               None
187             }
188           }
189           disjoint.getOrElse(andFilters(Seq(base, f)))
190       }
191     }
192 
193     private def dtgField(ev: MergedSourceSft[T], typeName: String): String = {
194       val sft = ev.sft(source, typeName)
195       def firstDtg(): Option[String] =
196         sft.getAttributeDescriptors.asScala.collectFirst {
197           case d if classOf[Date].isAssignableFrom(d.getType.getBinding) => d.getLocalName
198         }
199       sft.getDtgField.orElse(firstDtg()).getOrElse {
200         throw new IllegalArgumentException(
201           s"Schema '$typeName' does not have a date field for filtering: ${SimpleFeatureTypes.encodeType(sft)}")
202       }
203     }
204   }
