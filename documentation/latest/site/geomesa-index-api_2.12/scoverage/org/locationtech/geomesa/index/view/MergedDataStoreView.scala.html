33 /**
34  * Merged querying against multiple data stores
35  *
36  * @param stores delegate stores
37  * @param deduplicate deduplicate results across stores
38  * @param parallel execute queries in parallel
39  * @param taskDecorator parallel task decorator
40  * @param namespace namespace
41  */
42 class MergedDataStoreView(
43     val stores: Seq[MergedSource[DataStore]],
44     deduplicate: Boolean,
45     parallel: Boolean,
46     taskDecorator: Option[UnaryOperator[Runnable]] = None,
47     namespace: Option[String] = None
48   ) extends MergedDataStoreSchemas(stores.map(_.source), namespace) with HasGeoMesaFeatureReader with HasGeoMesaStats {

50   require(stores.nonEmpty, "No delegate stores configured")
51 
52   private val decorator = new TaskDecorator(taskDecorator)
53

55     new MergedQueryRunner(this, stores.map(s => s.copy[Queryable](DataStoreQueryable(s.source))), deduplicate, parallel)

59   override def getFeatureSource(name: Name): SimpleFeatureSource = getFeatureSource(name.getLocalPart)

62     val sources = stores.map(s => s.copy(s.source.getFeatureSource(typeName)))

85   object DateFilter {
86 
87     /**
88      * Create a date filter
89      *
90      * @param offsetOrDate either an offset (PT1D, e.g. a java duration string) or a static date (2026-01-01T00:00:00.000Z)
91      * @return
92      */
93     def apply(offsetOrDate: String): DateFilter = {
94       lazy val error =
95         new IllegalArgumentException(s"Invalid date filter '$offsetOrDate' - could not parse as a duration nor a date")
96       try {
97         return RelativeDate(offsetOrDate)
98       } catch {
99         case e: DateTimeParseException => error.addSuppressed(e)
100       }
101       try {
102         return StaticDate(offsetOrDate)
103       } catch {
104         case e: DateTimeParseException => error.addSuppressed(e)
105       }
106       throw error
107     }
108 
109     case class StaticDate(date: Date) extends DateFilter {
110       override def toDate(currentDate: Instant): Date = date
111     }
112 
113     object StaticDate {
114       @throws(classOf[DateTimeParseException])
115       def apply(date: String): StaticDate = StaticDate(DateParsing.parseDate(date))
116     }
117 
118     case class RelativeDate(offset: java.time.Duration) extends DateFilter {
119       override def toDate(currentDate: Instant): Date = Date.from(currentDate.plus(offset))
120     }
121 
122     object RelativeDate {
123       @throws(classOf[DateTimeParseException])
124       def apply(offset: String): RelativeDate = RelativeDate(java.time.Duration.parse(offset))
125     }
126   }

128   /**
129    * A merged object, with associated filters
130    *
131    * @param source object
132    * @param sourceFilter arbitrary filter, keyed by type name
133    * @param from start date filter, keyed by type name
134    * @param to end date filter, keyed by type name
135    * @tparam T type binding
136    */
137   case class MergedSource[T](source: T, sourceFilter: TypeOption[Filter], from: TypeOption[DateFilter], to: TypeOption[DateFilter]) {
138 
139     import FilterHelper.ff
140 
141     import scala.collection.JavaConverters._
142 
143     /**
144      * The filter to apply to this source, based on the current date
145      *
146      * @param query query
147      * @param now current timestamp, for consistency across multiple stores
148      * @param ev feature type loader
149      * @return
150      */
151     def filter(query: Query, now: Instant)(implicit ev: MergedSourceSft[T]): Query = {
152       val q = new Query(query)
153       q.setFilter(filter(query.getTypeName, query.getFilter, now))
154       q
155     }
156 
157     /**
158      * The filter to apply to this source, based on the current date
159      *
160      * @param typeName feature type name
161      * @param base user-supplied filter
162      * @param now current timestamp, for consistency across multiple stores
163      * @param ev feature type loader
164      * @return
165      */
166     def filter(typeName: String, base: Filter, now: Instant)(implicit ev: MergedSourceSft[T]): Filter = {
167       val dtg = dtgField(ev, typeName)
168       val fromFilter = from(typeName).map(d => ff.greaterOrEqual(ff.property(dtg), ff.literal(d.toDate(now))))
169       val toFilter = to(typeName).map(d => ff.less(ff.property(dtg), ff.literal(d.toDate(now))))
170       val storeFilter = FilterHelper.filterListAsAnd(sourceFilter(typeName).toSeq ++ fromFilter ++ toFilter)
171       storeFilter match {
172         case None => base
173         case Some(f) if base == Filter.INCLUDE => f
174         case Some(f) if dtg == null => andFilters(Seq(base, f))
175         case Some(f) =>
176           // check for disjoint dates between the store filter and the query filter
177           // note: don't cache this call, as it can contain things like `currentDate()` that will change per invocation
178           val disjoint = Some(FilterHelper.extractIntervals(f, dtg)).filter(_.nonEmpty).flatMap { left =>
179             val right = FilterHelper.extractIntervals(base, dtg)
180             val merged = FilterValues.and[Bounds[ZonedDateTime]](Bounds.intersection)(left, right)
181             if (merged.disjoint) {
182               logger.debug(s"Suppressing query with filter (${ECQL.toCQL(base)}) AND (${ECQL.toCQL(f)})")
183               Some(Filter.EXCLUDE)
184             } else {
185               None
186             }
187           }
188           disjoint.getOrElse(andFilters(Seq(base, f)))
189       }
190     }
191 
192     private def dtgField(ev: MergedSourceSft[T], typeName: String): String = {
193       val sft = ev.sft(source, typeName)
194       def firstDtg(): Option[String] =
195         sft.getAttributeDescriptors.asScala.collectFirst {
196           case d if classOf[Date].isAssignableFrom(d.getType.getBinding) => d.getLocalName
197         }
198       sft.getDtgField.orElse(firstDtg()).getOrElse {
199         throw new IllegalArgumentException(
200           s"Schema '$typeName' does not have a date field for filtering: ${SimpleFeatureTypes.encodeType(sft)}")
201       }
202     }
203   }
