46   override def createNewDataStore(params: java.util.Map[String, _]): DataStore = {
47     val configs: Seq[Config] = {
48       val explicit = Option(ConfigParam.lookup(params)).map(c => ConfigFactory.parseString(c).resolve())
49       val loaded = ConfigLoaderParam.flatMap(_.lookupOpt(params)).flatMap { name =>
50         ServiceLoader.load[MergedViewConfigLoader]().find(_.getClass.getName == name).map(_.load())
51       }
52       Seq(explicit, loaded).flatten.flatMap { config =>
53         if (config.hasPath("stores")) { config.getConfigList("stores").asScala } else { Seq.empty }
54       }
55     }
56 
57     if (configs.isEmpty) {
58       throw new IllegalArgumentException("No 'stores' element defined in configuration")
59     }
60 
61     val namespace = NamespaceParam.lookupOpt(params)
62     val nsConfig = namespace.map(ConfigValueFactory.fromAnyRef)
63     val taskDecorator = TaskDecoratorParam.lookupOpt(params)
64     val deduplicate = DeduplicateParam.lookup(params).booleanValue()
65     val parallel = ParallelScanParam.lookup(params).booleanValue()
66 
67     // track resources to clean up on error
68     val datastores = ArrayBuffer.empty[DataStore]
69     val providers = ArrayBuffer.empty[MergedFilterProvider]
70 
71     try {
72       val sources = configs.map { config =>
73         lazy val error =
74           new IllegalArgumentException(s"Could not load store using configuration:\n" +
75               config.root().render(ConfigRenderOptions.concise().setFormatted(true)))
76         // inject the namespace into the underlying stores
77         val paramConf = nsConfig.map(config.withValue(NamespaceParam.key, _)).getOrElse(config)
78         val storeParams = new java.util.HashMap[String, AnyRef](paramConf.root().unwrapped())
79         // allow for unquoted keys
80         storeParams.putAll(paramConf.toStringMap().asJava)
81 
82         val store = try { DataStoreFinder.getDataStore(storeParams) } catch { case NonFatal(e) => throw error.initCause(e) }
83         if (store == null) {
84           throw error
85         }
86         datastores += store
87         val provider = parseFilterProvider(storeParams)
88         providers += provider
89         val decorator = taskDecorator.orElse(Some(store).collect { case s: HasAuthorizationsProvider => s.authProvider })
90         MergedSource(store, provider, new TaskDecorator(decorator))
91       }
92       new MergedDataStoreView(sources, deduplicate, parallel, namespace)
93     } catch {
94       case NonFatal(e) =>
95         CloseQuietly(providers).foreach(e.addSuppressed)
96         CloseQuietly(datastores).foreach(e.addSuppressed)
97         throw e
98     }
99   }

201   private def parseFilterProvider(params: java.util.Map[String, _]): MergedFilterProvider = {
202     val opened = ArrayBuffer.empty[MergedFilterProvider]
203 
204     def paramsFor(param: GeoMesaParam[_]): Map[String, String] = {
205       val prefix = param.key + "."
206       params.asScala.toMap.collect {
207         case (k, v) if k.startsWith(prefix) => k.substring(prefix.length) -> v.toString
208       }
209     }
210 
211     def parseProvider(param: GeoMesaParam[String], parser: String => MergedFilterProvider): Option[MergedFilterProvider] = {
212       val byTypes = paramsFor(param).map { case (k, v) =>
213         val provider = parser(v)
214         opened += provider
215         k -> provider
216       }
217       val default = param.lookupOpt(params).map(parser)
218       opened ++= default
219       if (byTypes.isEmpty) {
220         default
221       } else {
222         Some(TypedProvider(byTypes, default))
223       }
224     }
225 
226     try {
227       val static = parseProvider(StoreFilterParam, f => MergedFilterProvider.static(parseFilter(f)))
228       val dateOffsetStart = parseProvider(DateOffsetStartParam, MergedFilterProvider.from)
229       val dateOffsetEnd = parseProvider(DateOffsetEndParam, MergedFilterProvider.to)
230       val custom = StoreFilterProviderParam.lookupOpt(params)
231       opened ++= custom
232       custom.foreach(_.init(paramsFor(StoreFilterProviderParam)))
233       MergedFilterProvider.and(static.toSeq ++ dateOffsetStart ++ dateOffsetEnd ++ custom)
234     } catch {
235       case NonFatal(e) => CloseQuietly(opened).foreach(e.addSuppressed); throw e
236     }
237   }
