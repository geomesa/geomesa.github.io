42   override def createNewDataStore(params: java.util.Map[String, _]): DataStore = {
43     val configs: Seq[Config] = {
44       val explicit = Option(ConfigParam.lookup(params)).map(c => ConfigFactory.parseString(c).resolve())
45       val loaded = ConfigLoaderParam.flatMap(_.lookupOpt(params)).flatMap { name =>
46         ServiceLoader.load[MergedViewConfigLoader]().find(_.getClass.getName == name).map(_.load())
47       }
48       Seq(explicit, loaded).flatten.flatMap { config =>
49         if (config.hasPath("stores")) { config.getConfigList("stores").asScala } else { Seq.empty }
50       }
51     }
52 
53     if (configs.isEmpty) {
54       throw new IllegalArgumentException("No 'stores' element defined in configuration")
55     }
56 
57     val namespace = NamespaceParam.lookupOpt(params)
58     val nsConfig = namespace.map(ConfigValueFactory.fromAnyRef)
59 
60     val stores = Seq.newBuilder[MergedSource[DataStore]]
61     stores.sizeHint(configs.length)
62 
63     try {
64       configs.foreach { config =>
65         lazy val error =
66           new IllegalArgumentException(s"Could not load store using configuration:\n" +
67               config.root().render(ConfigRenderOptions.concise().setFormatted(true)))
68         // inject the namespace into the underlying stores
69         val paramConf = nsConfig.map(config.withValue(NamespaceParam.key, _)).getOrElse(config)
70         val storeParams = new java.util.HashMap[String, AnyRef](paramConf.root().unwrapped())
71         // allow for unquoted keys
72         storeParams.putAll(paramConf.toStringMap().asJava)
73         val filter = try { StoreFilterParam.lookupOpt(storeParams).map(ECQL.toFilter) } catch {
74           case NonFatal(e) =>
75             throw new IllegalArgumentException(s"Invalid store filter '${storeParams.get(StoreFilterParam.key)}'", e)
76         }
77         val dateOffsetStart = DateOffsetStartParam.lookupOpt(storeParams).map(DateFilter.apply)
78         val dateOffsetEnd = DateOffsetEndParam.lookupOpt(storeParams).map(DateFilter.apply)
79         Try(DataStoreFinder.getDataStore(storeParams)) match {
80           case Success(null)  => throw error
81           case Success(store) => stores += MergedSource(store, filter, dateOffsetStart, dateOffsetEnd)
82           case Failure(e)     => throw error.initCause(e)
83         }
84       }
85     } catch {
86       case NonFatal(e) => stores.result.foreach(_.source.dispose()); throw e
87     }
88 
89     val deduplicate = DeduplicateParam.lookup(params).booleanValue()
90     val parallel = ParallelScanParam.lookup(params).booleanValue()
91     val decorator = TaskDecoratorParam.lookupOpt(params)
92 
93     new MergedDataStoreView(stores.result, deduplicate, parallel, decorator, namespace)
94   }
