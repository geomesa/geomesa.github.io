44   override def createNewDataStore(params: java.util.Map[String, _]): DataStore = {
45     val configs: Seq[Config] = {
46       val explicit = Option(ConfigParam.lookup(params)).map(c => ConfigFactory.parseString(c).resolve())
47       val loaded = ConfigLoaderParam.flatMap(_.lookupOpt(params)).flatMap { name =>
48         ServiceLoader.load[MergedViewConfigLoader]().find(_.getClass.getName == name).map(_.load())
49       }
50       Seq(explicit, loaded).flatten.flatMap { config =>
51         if (config.hasPath("stores")) { config.getConfigList("stores").asScala } else { Seq.empty }
52       }
53     }
54 
55     if (configs.isEmpty) {
56       throw new IllegalArgumentException("No 'stores' element defined in configuration")
57     }
58 
59     val namespace = NamespaceParam.lookupOpt(params)
60     val nsConfig = namespace.map(ConfigValueFactory.fromAnyRef)
61 
62     val stores = Seq.newBuilder[MergedSource[DataStore]]
63     stores.sizeHint(configs.length)
64 
65     try {
66       configs.foreach { config =>
67         lazy val error =
68           new IllegalArgumentException(s"Could not load store using configuration:\n" +
69               config.root().render(ConfigRenderOptions.concise().setFormatted(true)))
70         // inject the namespace into the underlying stores
71         val paramConf = nsConfig.map(config.withValue(NamespaceParam.key, _)).getOrElse(config)
72         val storeParams = new java.util.HashMap[String, AnyRef](paramConf.root().unwrapped())
73         // allow for unquoted keys
74         storeParams.putAll(paramConf.toStringMap().asJava)
75 
76         val filter = parseTypeOptions(StoreFilterParam, parseFilter, storeParams)
77         val dateOffsetStart = parseTypeOptions(DateOffsetStartParam, DateFilter.apply, storeParams)
78         val dateOffsetEnd = parseTypeOptions(DateOffsetEndParam, DateFilter.apply, storeParams)
79 
80         Try(DataStoreFinder.getDataStore(storeParams)) match {
81           case Success(null)  => throw error
82           case Success(store) => stores += MergedSource(store, filter, dateOffsetStart, dateOffsetEnd)
83           case Failure(e)     => throw error.initCause(e)
84         }
85       }
86     } catch {
87       case NonFatal(e) => stores.result.foreach(_.source.dispose()); throw e
88     }
89 
90     val deduplicate = DeduplicateParam.lookup(params).booleanValue()
91     val parallel = ParallelScanParam.lookup(params).booleanValue()
92     val decorator = TaskDecoratorParam.lookupOpt(params)
93 
94     new MergedDataStoreView(stores.result, deduplicate, parallel, decorator, namespace)
95   }
