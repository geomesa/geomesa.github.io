29 /**
30   * In memory sorting of simple features, with optional spill to disk
31   *
32   * @param features unsorted feature iterator
33   * @param sortBy attributes to sort by, in the form: (name, reverse).
34   *               for sort by feature id (e.g. natural sort), use an empty string for name
35   */
36 class SortingSimpleFeatureIterator(features: CloseableIterator[SimpleFeature], sortBy: Seq[(String, Boolean)])
37     extends CloseableIterator[SimpleFeature] {
38 
39   import SortingSimpleFeatureIterator.{sortInMemory, sortWithSpillover}
40 
41   private val closed = new AtomicBoolean(false)
42 
43   private lazy val sorted: CloseableIterator[SimpleFeature] = {
44     try {
45       if (closed.get || !features.hasNext) { CloseableIterator.empty } else {
46         val head = features.next()
47         if (!features.hasNext) { CloseableIterator.single(head) } else {
48           val ordering = SimpleFeatureOrdering(head.getFeatureType, sortBy)
49           QueryProperties.SortMemoryThreshold.toBytes match {
50             case None => sortInMemory(head, features, ordering, closed)
51             case Some(threshold) => sortWithSpillover(head, features, ordering, closed, threshold)
52           }
53         }
54       }
55     } finally {
56       features.close()
57     }
58   }
59 
60   override def hasNext: Boolean = sorted.hasNext
61 
62   override def next(): SimpleFeature = sorted.next()
63 
64   override def close(): Unit = {
65     if (closed.compareAndSet(false, true)) {
66       sorted.close() // also closes original features iterator
67     }
68   }
69 }

73   /**
74    * Sorts the iterator in memory
75    *
76    * @param head head of iterator
77    * @param tail tail of iterator
78    * @param ordering ordering
79    * @param closed checked to determine if we should stop sorting and return early
80    * @return
81    */
82   private def sortInMemory(
83       head: SimpleFeature,
84       tail: CloseableIterator[SimpleFeature],
85       ordering: Ordering[SimpleFeature],
86       closed: AtomicBoolean): CloseableIterator[SimpleFeature] = {
87     // use ArrayList for sort-in-place of the underlying array
88     val list = new java.util.ArrayList[SimpleFeature](1000)
89     list.add(head)
90 
91     while (tail.hasNext && !closed.get) {
92       list.add(tail.next())
93     }
94 
95     if (closed.get) {
96       // don't bother sorting, just return an empty iterator
97       CloseableIterator.empty
98     } else {
99       sort(list, ordering)
100       CloseableIterator(list.iterator)
101     }
102   }
103 
104   /**
105    * Sorts the iterator in memory, but spills over to disk if the size in memory exceeds the specified threshold
106    *
107    * @param head head of iterator
108    * @param tail tail of iterator
109    * @param ordering ordering
110    * @param closed checked to determine if we should stop sorting and return early
111    * @param threshold max memory to use before spilling to disk
112    * @return
113    */
114   private def sortWithSpillover(
115       head: SimpleFeature,
116       tail: CloseableIterator[SimpleFeature],
117       ordering: Ordering[SimpleFeature],
118       closed: AtomicBoolean,
119       threshold: Long): CloseableIterator[SimpleFeature] = {
120     var sorter = new SpillToDiskSorter(head, ordering, threshold)
121     try {
122       while (tail.hasNext && !closed.get()) {
123         sorter = sorter + tail.next()
124       }
125       sorter.result(closed.get)
126     } catch {
127       case NonFatal(e) =>
128         sorter.cleanup()
129         throw e
130     }
131   }

178   /**
179    * Class for sorting with spill-to-disk
180    *
181    * @param spill write/read to disk
182    * @param sizer sizer for the features being sorted
183    * @param ordering sort ordering
184    * @param threshold memory threshold for dumping to disk, in bytes
185    * @param files list of files that have been dumped to disk
186    * @param pending a file that is asynchronously being dumped to disk, if any
187    */
188   private class SpillToDiskSorter(
189       spill: SpillToDisk,
190       sizer: FeatureIsSizable[SimpleFeature],
191       ordering: Ordering[SimpleFeature],
192       threshold: Long,
193       files: Seq[File] = Seq.empty,
194       pending: Option[Future[File]] = None) extends Callable[File] {
195 
196     private val start = System.nanoTime()
197     private val mem = new java.util.ArrayList[SimpleFeature](1000)
198     private var size: Long = 0
199 
200     /**
201      * Create a new sorter
202      *
203      * @param sf simple feature, representative of the features being chunked
204      * @param ordering ordering
205      * @param threshold threshold for triggering a spill-to-disk, in bytes
206      * @return
207      */
208     def this(sf: SimpleFeature, ordering: Ordering[SimpleFeature], threshold: Long) = {
209       this(new SpillToDisk(sf.getFeatureType), toSizeable(sf), ordering, threshold)
210       mem.add(sf)
211       size += sizer.sizeOf(sf)
212     }
213 
214     /**
215      * Add the simple feature to the chunk, returning either this chunk or a new chunk
216      *
217      * @param sf feature
218      * @return next chunk
219      */
220     def +(sf: SimpleFeature): SpillToDiskSorter = {
221       if (size >= threshold) {
222         logger.debug(s"Read ${mem.size()} features from the underlying store in ${(System.nanoTime() - start)/1000000}ms")
223         // only allow one sort at a time so we don't blow up memory with all the array lists
224         val last = pending.map { f =>
225           if (f.isDone) { f.get() } else {
226             val start = System.nanoTime()
227             val file = f.get()
228             logger.debug(s"Waited ${(System.nanoTime() - start)/1000000}ms for last batch to be flushed to disk")
229             file
230           }
231         }
232         val future = CachedThreadPool.call(this)
233         new SpillToDiskSorter(spill, sizer, ordering, threshold, files ++ last, Some(future)) + sf
234       } else {
235         mem.add(sf)
236         size += sizer.sizeOf(sf)
237         this
238       }
239     }
240 
241     override def call(): File = {
242       sort(mem, ordering)
243       spill.write(mem)
244     }
245 
246     /**
247      * Get all the features added so far, in sorted order
248      *
249      * @return
250      */
251     def result(closed: Boolean): CloseableIterator[SimpleFeature] = {
252       logger.debug(s"Read ${mem.size()} features from the underlying store in ${(System.nanoTime() - start)/1000000}ms")
253       if (closed) {
254         // don't bother sorting, just clean up and return an empty iterator
255         cleanup()
256         CloseableIterator.empty
257       } else {
258         sort(mem, ordering)
259         val memIter = CloseableIterator(mem.iterator)
260         if (files.isEmpty && pending.isEmpty) {
261           memIter
262         } else {
263           val filesIter = (files ++ pending.map(_.get())).map(spill.read)
264           new MergeSortingIterator(IndexedSeq(memIter) ++ filesIter, ordering)
265         }
266       }
267     }
268 
269     /**
270      * Delete any underlying resources. Should only be called if not calling `result` due to an error/interruption
271      */
272     def cleanup(): Unit = {
273       pending.foreach(_.cancel(true))
274       files.foreach(delete)
275       Try(pending.map(_.get())).toOption.flatten.foreach(delete)
276     }
277   }

279   /**
280    * Class for writing and reading back from disk
281    *
282    * @param sft feature type
283    */
284   private class SpillToDisk(sft: SimpleFeatureType) {
285 
286     // grouping ID for tmp files, doesn't have to be unique
287     private val id =f"${sft.getTypeName.replaceAll("[^A-Za-z0-9_-]", "").take(20)}-${Random.nextInt(10000)}%04d"
288 
289     private val serializer = KryoFeatureSerializer(sft, SerializationOption.WithUserData)
290 
291     def write(features: java.util.ArrayList[SimpleFeature]): File = {
292       val start = System.nanoTime()
293       val file = Files.createTempFile(s"gm-sort-$id-", ".kryo").toFile
294       logger.trace(s"Created temp sort file '${file.getAbsolutePath}'")
295       file.deleteOnExit()
296       WithClose(new FileOutputStream(file)) { os =>
297         features.forEach { sf =>
298           val bytes = serializer.serialize(sf)
299           os.write(ByteArrays.toBytes(bytes.length))
300           os.write(bytes)
301         }
302       }
303       logger.debug(s"Wrote ${features.size()} features to disk in ${(System.nanoTime() - start)/1000000}ms")
304       file
305     }
306 
307     def read(file: File): CloseableIterator[SimpleFeature] = new FileIterator(file, serializer)
308   }

310   /**
311    * Does a merge sort on a group of locally sorted files and a left-over in-memory iterator
312    *
313    * @param merging list of feature iterators to merge, already sorted
314    * @param ordering feature ordering
315    */
316   private class MergeSortingIterator(merging: IndexedSeq[CloseableIterator[SimpleFeature]], ordering: Ordering[SimpleFeature])
317       extends CloseableIterator[SimpleFeature] {
318 
319     // priority queue containing the next feature from each file
320     private val heads = {
321       val o = Ordering.by[(SimpleFeature, Int), SimpleFeature](_._1)(ordering)
322       val q = new java.util.PriorityQueue[(SimpleFeature, Int)](merging.size, o)
323       var i = 0
324       while (i < merging.length) {
325         val m = merging(i)
326         if (m.hasNext) {
327           q.add(m.next -> i)
328         }
329         i += 1
330       }
331       q
332     }
333 
334     override def hasNext: Boolean = !heads.isEmpty
335 
336     override def next(): SimpleFeature = {
337       val (f, i) = heads.poll()
338       val m = merging(i)
339       if (m.hasNext) {
340         heads.add(m.next -> i)
341       }
342       f
343     }
344 
345     override def close(): Unit = CloseQuietly.raise(merging)
346   }

348   /**
349    * Reads a tmp file of serialized simple features
350    *
351    * @param file file
352    * @param serializer serializer
353    */
354   private class FileIterator(file: File, serializer: SimpleFeatureSerializer)
355       extends CloseableIterator[SimpleFeature] with LazyLogging {
356 
357     private val is = new FileInputStream(file)
358     private var buf = Array.ofDim[Byte](4)
359 
360     override def hasNext: Boolean = is.available() > 0
361 
362     override def next(): SimpleFeature = {
363       is.read(buf, 0, 4)
364       val size = ByteArrays.readInt(buf)
365       if (buf.length < size) {
366         buf = Array.ofDim((size * 1.2).toInt)
367       }
368       is.read(buf, 0, size)
369       serializer.deserialize(buf, 0, size)
370     }
371 
372     override def close(): Unit = {
373       try { is.close() } finally {
374         delete(file)
375       }
376     }
377   }
