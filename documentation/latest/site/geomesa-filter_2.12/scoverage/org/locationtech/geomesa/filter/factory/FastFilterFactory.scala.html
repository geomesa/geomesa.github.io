276   override def or(filters: java.util.List[Filter]): Or = {
277     if (filters.isEmpty) {
278       return super.or(filters)
279     }
280 
281     val predicates = FilterHelper.flattenOr(filters.asScala.toSeq)
282 
283     val props = scala.collection.mutable.HashSet.empty[String]
284     val literals = scala.collection.immutable.HashSet.newBuilder[AnyRef]
285     literals.sizeHint(predicates.length)
286 
287     val iter = predicates.iterator
288     while (iter.hasNext) {
289       iter.next() match {
290         case p: PropertyIsEqualTo if p.getMatchAction == MatchAction.ANY && p.isMatchingCase =>
291           org.locationtech.geomesa.filter.checkOrder(p.getExpression1, p.getExpression2) match {
292             case Some(PropertyLiteral(name, lit, _)) if !props.add(name) || props.size == 1 => literals += lit.getValue
293             case _ => return super.or(filters)
294           }
295 
296         case _ => return super.or(filters)
297       }
298     }
299 
300     // if we've reached here, we have verified that all the child filters are equality matches on the same property
301     val prop = property(props.head)
302     val values = literals.result
303     val isListType = Option(FastFilterFactory.sfts.get.getDescriptor(props.head)).exists(_.isList)
304 
305     if (values.size >= OrHashEquality.OrHashThreshold.get.toInt) {
306       if (isListType) {
307         new OrHashListEquality(prop, values)
308       } else {
309         new OrHashEquality(prop, values)
310       }
311     } else if (isListType) {
312       new OrSequentialListEquality(prop, values.toSeq)
313     } else {
314       new OrSequentialEquality(prop, values.toSeq)
315     }
316   }
