20 /**
21  * Hash function, follows the Apache Icebergs hash specification - https://iceberg.apache.org/spec/#appendix-b-32-bit-hash-requirements
22  */
23 class MurmurHashFunction extends FunctionExpressionImpl(MurmurHashFunction.Name) {

27   override def evaluate(o: AnyRef): AnyRef = {
28     val value = getExpression(0).evaluate(o)
29     if (value == null) { null } else { Int.box(hash(value)) }
30   }
31 }

33 object MurmurHashFunction {
34 
35   val Name = new FunctionNameImpl(
36     "murmurHash",
37     classOf[java.lang.Integer],
38     parameter("value", classOf[AnyRef])
39   )
40 
41   private[filter] def hash(value: AnyRef): Int = value match {
42     case v: String => StringHashing(v)
43     case v: Integer => IntegerHashing(v)
44     case v: java.lang.Long => LongHashing(v)
45     case v: java.lang.Float => FloatHashing(v)
46     case v: java.lang.Double => DoubleHashing(v)
47     case v: Date => DateHashing(v)
48     case v: Array[Byte] => ByteHashing(v)
49     case v: UUID => UUIDHashing(v)
50     case _ => StringHashing(value.toString)
51   }
52 
53   val Hashers: Seq[Hashing[_ <: AnyRef]] = Seq(
54     StringHashing,
55     IntegerHashing,
56     LongHashing,
57     FloatHashing,
58     DoubleHashing,
59     DateHashing,
60     ByteHashing,
61     UUIDHashing,
62   )
63 
64   abstract class Hashing[T: ClassTag] extends (T => Int) {
65     val binding: Class[T] = implicitly[ClassTag[T]].runtimeClass.asInstanceOf[Class[T]]
66   }
67 
68   object StringHashing extends Hashing[String] {
69     // string	hashBytes(utf8Bytes(v))
70     override def apply(v1: String): Int = {
71       val bytes = v1.getBytes(StandardCharsets.UTF_8)
72       MurmurHash3.hash32x86(bytes, 0, bytes.length, 0)
73     }
74   }
75 
76   object IntegerHashing extends Hashing[Integer]{
77     // int	hashLong(long(v))
78     override def apply(v1: Integer): Int = LongHashing(v1.toLong)
79   }
80 
81   object LongHashing extends Hashing[java.lang.Long]{
82     // long	hashBytes(littleEndianBytes(v))
83     override def apply(v1: java.lang.Long): Int =  MurmurHash3.hash32(java.lang.Long.reverseBytes(v1.toLong), 0)
84   }
85 
86   object FloatHashing extends Hashing[java.lang.Float]{
87     // float	hashLong(doubleToLongBits(double(v))
88     override def apply(v1: java.lang.Float): Int = LongHashing(java.lang.Double.doubleToLongBits(v1.toDouble))
89   }
90 
91   object DoubleHashing extends Hashing[java.lang.Double]{
92     // double	hashLong(doubleToLongBits(v))
93     override def apply(v1: java.lang.Double): Int = LongHashing(java.lang.Double.doubleToLongBits(v1.toDouble))
94   }
95 
96   object DateHashing extends Hashing[Date]{
97     // timestamp	hashLong(microsecsFromUnixEpoch(v))
98     override def apply(v1: Date): Int = LongHashing(v1.getTime * 1000)
99   }
100 
101   object ByteHashing extends Hashing[Array[Byte]]{
102     // binary	hashBytes(v)
103     override def apply(v1: Array[Byte]): Int = MurmurHash3.hash32x86(v1, 0, v1.length, 0)
104   }
105 
106   object UUIDHashing extends Hashing[UUID]{
107     // uuid	hashBytes(uuidBytes(v)
108     override def apply(v1: UUID): Int = MurmurHash3.hash32(v1.getMostSignificantBits, v1.getLeastSignificantBits, 0)
109   }
