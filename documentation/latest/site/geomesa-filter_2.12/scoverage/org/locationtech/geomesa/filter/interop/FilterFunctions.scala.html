17 /**
18  * Java-friendly access to GeoMesa filter function implementations without constructing
19  * GeoTools expressions or features. Methods accept evaluated arguments; callers handle
20  * null inputs, except for the nullable viewer ID. XZ2 returns null for empty geometries.
21  */
22 object FilterFunctions {
23 
24   def murmurHash(value: String): Int = StringHashing(value)
25   def murmurHash(value: Int): Int = IntegerHashing(value)
26   def murmurHash(value: Long): Int = LongHashing(value)
27   def murmurHash(value: Float): Int = FloatHashing(value)
28   def murmurHash(value: Double): Int = DoubleHashing(value)
29   def murmurHash(value: Date): Int = DateHashing(value)
30   def murmurHash(value: Array[Byte]): Int = ByteHashing(value)
31   def murmurHash(value: UUID): Int = UUIDHashing(value)
32 
33   def proxyId(id: String): Int = ProxyIdFunction.proxyId(id)
34   def proxyId(id: UUID): Int = ProxyIdFunction.proxyId(id.getMostSignificantBits, id.getLeastSignificantBits)
35 
36   def z2(geom: Point): String = Z2Function.encode(geom)
37   def xz2(geom: Geometry): String = XZ2Function.encode(geom)
38 
39   /** Encode a viewer record with a timestamp in milliseconds since the Unix epoch. */
40   def convert2viewer(id: String, geom: Point, millis: Long): String = Convert2ViewerFunction.encode(id, geom, millis)
41 }
