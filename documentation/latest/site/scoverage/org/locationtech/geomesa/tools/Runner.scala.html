38    * Main command line invocation
39    *
40    * @param args command line args
41    */
42   // noinspection ScalaUnusedSymbol
43   def main(args: Array[String]): Unit = {
44     val result = try { parseCommand(args).execute(); CommandResult(0) } catch {
45       case e @ (_: ClassNotFoundException | _: NoClassDefFoundError) =>
46         // log the underling exception to the log file, but don't show to the user
47         val msg = s"Warning: Missing dependency for command execution: ${e.getMessage}"
48         logger.error(msg, e)
49         CommandResult(1, Seq(Left(msg)) ++ getEnvironmentErrors.map(Left.apply[String, Throwable]))
50       case e @ (_: ParameterException | _: CommandException) =>
51         logger.error("Error executing command:", e)
52         CommandResult(1, failureMessages(e))
53       case NonFatal(e) =>
54         logger.error("Error executing command:", e)
55         CommandResult(1, Seq(Right(e)))
56     }
57 
58     result.errors.foreach {
59       case Left(msg) => Command.user.error(msg)
60       case Right(e)  => Command.user.error(e.getMessage, e)
61     }
62 
63     sys.exit(result.code)
64   }

66   private def failureMessages(e: Throwable, indent: String = "", prefix: String = ""): Seq[Left[String, Throwable]] = {
67     val messages = ArrayBuffer.empty[Left[String, Throwable]]
68     val message = if (indent == "") { e.getMessage } else { s"$indent$prefix${e.toString}" }
69     messages += Left(message)
70     if (e.getCause != null) {
71       messages ++= failureMessages(e.getCause, indent + "  ", "Caused by: ")
72     }
73     e.getSuppressed.foreach { e =>
74       messages ++= failureMessages(e, indent + "  ", "Suppressed: ")
75     }
76     messages.toSeq
77   }

79   def parseCommand(args: Array[String]): Command = {
80     val jc =
81       JCommander.newBuilder()
82           .programName(name)
83           .addConverterFactory(new GeoMesaIStringConverterFactory)
84           .build()
85 
86     val commands = this.commands ++ Seq(new HelpCommand(this, jc), new AutoCompleteCommand(this, jc))
87     commands.foreach { command =>
88       jc.addCommand(command.name, command.params)
89       command.subCommands.foreach(sub => jc.getCommands.get(command.name).addCommand(sub.name, sub.params))
90     }
91 
92     def logAndThrowError(e: ParameterException): Unit = {
93       Command.user.error(s"Error parsing arguments: ${e.getMessage}")
94       Command.user.info(usage(jc, jc.getParsedCommand))
95       throw e
96     }
97 
98     try { jc.parse(args: _*) } catch {
99       case e: ParameterException => logAndThrowError(e)
100     }
101 
102     val parsed = commands.find(_.name == jc.getParsedCommand).getOrElse(new DefaultCommand(jc))
103     resolveEnvironment(parsed)
104     val command = if (parsed.subCommands.isEmpty) { parsed } else {
105       lazy val available =
106         s"Use '${parsed.name} <sub-command>', where sub-command is one of: " +
107             parsed.subCommands.map(_.name).mkString(", ")
108       val sub = Option(jc.getCommands.get(parsed.name).getParsedCommand).getOrElse {
109         throw new ParameterException(s"No sub-command specified. $available")
110       }
111       parsed.subCommands.find(_.name == sub).getOrElse {
112         throw new ParameterException(s"Sub-command '$sub' not found. $available")
113       }
114     }
115     command.validate().foreach(logAndThrowError)
116     command
117   }
118 
119   def usage(jc: JCommander): String = {
120     val out = new StringBuilder()
121     out.append(s"Usage: $name [command] [command options]\n")
122     val commands = jc.getCommands.asScala.keys.toSeq.sorted
123     out.append("  Commands:\n")
124     val maxLen = commands.map(_.length).max + 4
125     commands.foreach { name =>
126       val spaces = " " * (maxLen - name.length)
127       out.append(s"    $name$spaces${jc.getUsageFormatter.getCommandDescription(name)}\n")
128     }
129     out.toString()
130   }
131 
132   def usage(jc: JCommander, name: String): String = {
133     Option(name).flatMap(n => Option(jc.getCommands.get(n))) match {
134       case None => usage(jc)
135       case Some(command) =>
136         val out = new java.lang.StringBuilder()
137         command.getUsageFormatter.usage(out)
138         out.toString
139     }
140   }

162   private def getEnvironmentErrors: Option[String] = {
163     val env = classpathEnvironments
164     if (env.forall(sys.env.contains)) { None } else {
165       val envMsg = if (env.size == 1) { "is not set as an environment variable" } else { "are not set as environment variables" }
166       val types = env.map(e => e.take(1) ++ e.drop(1).toLowerCase(Locale.US).replace("_home", "")).mkString(" and ")
167       val msg =
168         s"\nWarning: ${env.mkString(" and/or ")} $envMsg." +
169             s"\nGeoMesa tools will not run without the appropriate $types JARs in the tools classpath." +
170             s"\nPlease ensure that those JARs are present in the classpath by running '$name classpath'." +
171             "\nTo take corrective action, copy the necessary JAR files in the GeoMesa tools lib directory " +
172             "using the provided 'install-dependencies.sh' and 'install-*-support.sh' scripts."
173       Some(msg)
174     }
175   }
