69   private def export(ds: DS): JobResult = {
70     // for file data stores, handle setting the default type name so the user doesn't have to
71     for {
72       p <- Option(params).collect { case p: ProvidedTypeNameParam => p }
73       f <- Option(ds).collect { case f: FileDataStore => f }
74     } { p.featureName = f.getSchema.getTypeName }
75 
76     val options = ExportOptions(params)
77     val remote = options.file.exists(PathUtils.isRemote)
78     val reducers = Option(params.reducers).map(_.intValue()).getOrElse(-1)
79     val mode = params.mode.getOrElse(if (reducers == -1) { RunModes.Local } else { RunModes.Distributed })
80     val chunks = Option(params.chunkSize).map(_.longValue())
81 
82     // do some validation up front
83     if (options.file.isEmpty && !options.format.streaming) {
84       throw new ParameterException(s"Format '${options.format}' requires file-based output, please use --output")
85     } else if (remote && options.format == ExportFormat.Shp) {
86       throw new ParameterException("Shape file export is not supported for distributed file systems")
87     } else if (!remote && mode == RunModes.Distributed) {
88       throw new ParameterException("Distributed export requires an output file in a distributed file system")
89     } else if (mode == RunModes.Distributed && params.maxFeatures != null) {
90       throw new ParameterException("Distributed export does not support --max-features")
91     }
92 
93     val sft = ds.getSchema(params.featureName)
94     if (sft == null) {
95       throw new ParameterException(s"Schema '${params.featureName}' does not exist in the store")
96     }
97 
98     val query = ExportCommand.createQuery(sft, params)
99 
100     mode match {
101       case RunModes.Local =>
102         options.file.map(PathUtils.getHandle).foreach { file =>
103           if (file.exists) {
104             if (params.force) {
105               Command.user.warn(s"Output file '${file.path}' exists - deleting it")
106             } else if (!Prompt.confirm(s"WARNING Output file '${file.path}' exists, delete it and continue (y/n)? ")) {
107               throw new ParameterException(s"Output file '${file.path}' exists")
108             }
109             file.delete()
110           }
111         }
112         val exporter = chunks match {
113           case None    => new Exporter(options, query.getHints)
114           case Some(c) => new ChunkedExporter(options, query.getHints, c)
115         }
116         val count = try { export(ds, query, exporter, !params.suppressEmpty) } finally { exporter.close() }
117         val outFile = options.file match {
118           case None => "standard out"
119           case Some(f) if chunks.isDefined => PathUtils.getBaseNameAndExtension(f).productIterator.mkString("_*")
120           case Some(f) => f
121         }
122 
123         JobSuccess(s"Feature export complete to $outFile", count.map(CountKey -> _).toMap)
124 
125       case RunModes.Distributed =>
126         val job = Job.getInstance(new Configuration, "GeoMesa Tools Export")
127 
128         // for remote jobs, don't push down format transforms, to enable counting and global sorting
129         val hints = new Hints(query.getHints)
130         ExportCommand.disableAggregation(sft, query.getHints)
131 
132         configure(job, ds, query) // note: do this first so that input format is set for the TotalOrderPartitioner
133 
134         // note: these confs should be set by the input format
135         val reduce = Seq(GeoMesaConfigurator.Keys.FeatureReducer, GeoMesaConfigurator.Keys.Sorting).filter { key =>
136           job.getConfiguration.get(key) != null
137         }
138 
139         if (reducers < 1 && reduce.nonEmpty) {
140           if (reduce.contains(GeoMesaConfigurator.Keys.Sorting)) {
141             throw new ParameterException("Distributed export sorting requires --num-reducers")
142           } else {
143             throw new ParameterException(s"Distributed export format '${options.format}' requires --num-reducers")
144           }
145         }
146 
147         // must be some due to our remote check
148         val file = options.file.getOrElse(throw new IllegalStateException("file should be Some"))
149         val output = new Path(PathUtils.getUrl(file).toURI).getParent
150 
151         // file output format doesn't generally let you write to an existing directory
152         val context = FileContext.getFileContext(output.toUri, job.getConfiguration)
153         if (context.util.exists(output)) {
154           val warning = s"Output directory '$output' exists - files may be overwritten"
155           if (params.force) {
156             Command.user.warn(warning)
157           } else if (!Prompt.confirm(s"WARNING $warning. Continue anyway (y/n)? ")) {
158             if (Prompt.confirm("WARNING DATA MAY BE LOST - delete directory and proceed with export (y/n)? ")) {
159               context.delete(output, true)
160             } else {
161               throw new ParameterException(s"Output directory '$output' exists")
162             }
163           }
164         }
165 
166         val filename = FilenameUtils.getName(file)
167 
168         // note: use our original hints, which have the aggregating keys
169         ExportJob.configure(job, connection, sft, hints, filename, output, options.format, options.headers,
170           chunks, options.gzip, reducers, libjars(options.format), libjarsPaths)
171 
172         val reporter = TerminalCallback()
173         JobRunner.run(job, reporter, ExportJob.Counters.mapping(job), ExportJob.Counters.reducing(job)).merge {
174           Some(JobSuccess(s"Feature export complete to $output", Map(CountKey -> ExportJob.Counters.count(job))))
175         }
176 
177       case _ => throw new UnsupportedOperationException() // someone added a run mode and didn't implement it here...
178     }
179   }

201       }
202     } catch {

205             "that all arguments are correct", e)
206     }

239   /**
240     * Create the query to execute
241     *
242     * @param sft simple feature type
243     * @param params parameters
244     * @return
245     */
246   def createQuery(sft: SimpleFeatureType, params: ExportParams): Query = {
247     val typeName = Option(params).collect { case p: TypeNameParam => p.featureName }.orNull
248     val filter = Option(params.cqlFilter).getOrElse(Filter.INCLUDE)
249 
250     val query = new Query(typeName, filter)
251     val hints = query.getHints
252     Option(params.maxFeatures).map(Int.unbox).foreach(query.setMaxFeatures)
253     Option(params).collect { case p: OptionalIndexParam => p }.foreach { p =>
254       Option(p.index).foreach { index =>
255         logger.debug(s"Using index $index")
256         hints.put(QueryHints.QUERY_INDEX, index)
257       }
258     }
259 
260     Option(params.hints).foreach { hintStrings =>
261       hints.put(Hints.VIRTUAL_TABLE_PARAMETERS, hintStrings)
262       ViewParams.setHints(query)
263     }
264 
265     if (params.outputFormat == ExportFormat.Arrow) {
266       hints.put(QueryHints.ARROW_ENCODE, java.lang.Boolean.TRUE)
267     } else if (params.outputFormat == ExportFormat.Bin) {
268       // if not specified in hints, set it here to trigger the bin query
269       if (!hints.containsKey(QueryHints.BIN_TRACK)) {
270         hints.put(QueryHints.BIN_TRACK, "id")
271       }
272       if (!hints.containsKey(QueryHints.BIN_DTG)) {
273         sft.getDtgField.foreach(hints.put(QueryHints.BIN_DTG, _))
274       }
275     } else if (params.outputFormat == ExportFormat.Leaflet) {
276       configureLeafletExport(params)
277     }
278 
279     val attributes: Array[String] = {
280       val combined = params.attributes.asScala ++ params.transforms.asScala
281       if (combined.nonEmpty) {
282         val (id, attributes) = combined.partition(_.equalsIgnoreCase("id"))
283         if (id.isEmpty && !hints.containsKey(QueryHints.ARROW_INCLUDE_FID)) {
284           // note: we also use this hint for delimited text export
285           hints.put(QueryHints.ARROW_INCLUDE_FID, java.lang.Boolean.FALSE)
286         }
287         attributes.toArray
288       } else if (params.outputFormat == ExportFormat.Bin) {
289         BinAggregatingScan.propertyNames(hints, sft).toArray
290       } else {
291         null // all props
292       }
293     }
294     query.setPropertyNames(attributes: _*)
295 
296     if (!params.sortFields.isEmpty) {
297       val fields = params.sortFields.asScala.toSeq
298       if (fields.exists(a => sft.indexOf(a) == -1)) {
299         val errors = fields.filter(a => sft.indexOf(a) == -1)
300         throw new ParameterException(s"Invalid sort attribute${if (errors.lengthCompare(1) == 0) "" else "s"}: " +
301             errors.mkString(", "))
302       }
303       val order = if (params.sortDescending) { SortOrder.DESCENDING } else { SortOrder.ASCENDING }
304       query.setSortBy(fields.map(f => org.locationtech.geomesa.filter.ff.sort(f, order)): _*)
305     } else if (hints.isArrowQuery) {
306       hints.getArrowSort.foreach { case (f, r) =>
307         val order = if (r) { SortOrder.DESCENDING } else { SortOrder.ASCENDING }
308         query.setSortBy(org.locationtech.geomesa.filter.ff.sort(f, order))
309       }
310     }
311 
312     logger.debug(s"Applying CQL filter ${ECQL.toCQL(filter)}")
313     logger.debug(s"Applying transform ${Option(attributes).map(_.mkString(",")).orNull}")
314 
315     QueryRunner.configureQuery(sft, query)
316   }

343       params.gzip = null

348     }
349     if (PathUtils.isRemote(params.file)) {
350       if (params.file.endsWith("/")) {
351         params.file = s"${params.file}index.html"
352       } else if (FilenameUtils.indexOfExtension(params.file) == -1) {

357       val destination = if (file.isDirectory || (!file.exists && file.getName.indexOf(".") == -1)) {
358         new File(file, "index.html")
359       } else {
360         file

365   /**
366     * Disable hints for aggregating scans by removing them, and moving any aggregating sort hints to
367     * regular sort hints
368     *
369     * @param sft simple feature type
370     * @param hints hints
371     * @return
372     */
373   private def disableAggregation(sft: SimpleFeatureType, hints: Hints): Unit = {
374     if (hints.isArrowQuery) {
375       hints.remove(QueryHints.ARROW_ENCODE)
376       val sort = hints.remove(QueryHints.ARROW_SORT_FIELD).asInstanceOf[String]
377       if (sort != null) {
378         val reverse = Option(hints.remove(QueryHints.ARROW_SORT_REVERSE).asInstanceOf[java.lang.Boolean])
379         val order = if (reverse.exists(_.booleanValue)) { SortOrder.DESCENDING } else { SortOrder.ASCENDING }
380         val hint = org.locationtech.geomesa.filter.ff.sort(sort, order)
381         hints.put(QueryHints.Internal.SORT_FIELDS, QueryHints.Internal.toSortHint(Array(hint)))
382       }
383     } else if (hints.isBinQuery) {
384       hints.remove(QueryHints.BIN_TRACK)
385       if (hints.isBinSorting) {
386         hints.getBinDtgField.orElse(sft.getDtgField).foreach { dtg =>
387           val hint = org.locationtech.geomesa.filter.ff.sort(dtg, SortOrder.ASCENDING)
388           hints.put(QueryHints.Internal.SORT_FIELDS, QueryHints.Internal.toSortHint(Array(hint)))
389         }
390       }
391     }

419     * @param options options

422   class Exporter(options: ExportOptions, hints: Hints) extends FeatureExporter {
423 
424     // used only for streaming export formats
425     private lazy val stream: ByteCounterStream = {
426       // avro compression is handled differently, see AvroExporter below
427       val gzip = options.gzip.filter(_ => options.format != ExportFormat.Avro && options.format != ExportFormat.AvroNative)
428       val handle = options.file match {
429         case None => StdInHandle.get() // writes to std out
430         case Some(f) => PathUtils.getHandle(f)
431       }
432       if (options.writeEmptyFiles) {

434       } else {
435         new LazyExportStream(handle, gzip)
436       }
437     }
438 
439     // used only for file-based export formats
440     private lazy val name = options.file.getOrElse {
441       // should have been validated already...
442       throw new IllegalStateException("Export format requires a file but none was specified")
443     }
444 
445     // noinspection ComparingUnrelatedTypes
446     private lazy val fids = !Option(hints.get(QueryHints.ARROW_INCLUDE_FID)).contains(java.lang.Boolean.FALSE)
447 
448     private val exporter = options.format match {
449       case ExportFormat.Arrow      => new ArrowExporter(stream, hints)
450       case ExportFormat.Avro       => new AvroExporter(stream, options.gzip)
451       case ExportFormat.AvroNative => new AvroExporter(stream, options.gzip, Set(SerializationOption.NativeCollections))
452       case ExportFormat.Bin        => new BinExporter(stream, hints)
453       case ExportFormat.Csv        => DelimitedExporter.csv(stream, options.headers, fids)
454       case ExportFormat.Gml2       => GmlExporter.gml2(stream)
455       case ExportFormat.Gml3       => GmlExporter(stream)
456       case ExportFormat.Json       => new GeoJsonExporter(stream)
457       case ExportFormat.Leaflet    => new LeafletMapExporter(stream)
458       case ExportFormat.Null       => NullExporter
459       case ExportFormat.Parquet    => new GeoParquetExporter(name)
460       case ExportFormat.Shp        => new ShapefileExporter(new File(name))
461       case ExportFormat.Tsv        => DelimitedExporter.tsv(stream, options.headers, fids)
462       // shouldn't happen unless someone adds a new format and doesn't implement it here
463       case _ => throw new UnsupportedOperationException(s"Export for '${options.format}' is not implemented")
464     }

467

469 
470     override def bytes: Long = exporter.bytes
471 
472     override def close(): Unit = exporter.close()

475   /**
476     * Feature exporter that handles chunking output into multiple files
477     *
478     * @param options export options
479     * @param hints query hints
480     * @param chunks number of bytes to write per file
481     */
482   class ChunkedExporter(options: ExportOptions, hints: Hints, chunks: Long)
483       extends FeatureExporter with LazyLogging {
484 
485     private val names = options.file match {
486       case None    => Iterator.continually(None)
487       case Some(f) => new IncrementingFileName(f).map(Option.apply)
488     }
489 
490     private var sft: SimpleFeatureType = _
491     private var exporter: Exporter = _
492     private var estimator: FileSizeEstimator = _
493     private var count = 0L // number of features written
494     private var total = 0L // sum size of all finished chunks
495 
496     override def start(sft: SimpleFeatureType): Unit = {
497       this.sft = sft
498       estimator = new FileSizeEstimator(chunks, 0.05f, options.format.bytesPerFeature(sft)) // 5% error threshold
499       nextChunk()
500     }
501 
502     override def export(features: Iterator[SimpleFeature]): Option[Long] = export(features, None)
503 
504     override def bytes: Long = total
505 
506     override def close(): Unit = if (exporter != null) { exporter.close() }
507 
508     private def nextChunk(): Unit = {
509       if (exporter != null) {
510         exporter.close()
511         // adjust our estimate to account for the actual bytes written
512         // do this after closing the exporter to account for footers, batching, etc
513         val written = exporter.bytes
514         estimator.update(written, count)
515         total += written
516       }
517       exporter = new Exporter(options.copy(file = names.next), hints)
518       exporter.start(sft)
519       count = 0L
520     }
521 
522     @tailrec
523     private def export(features: Iterator[SimpleFeature], result: Option[Long]): Option[Long] = {
524       var estimate = estimator.estimate(exporter.bytes)
525       val counter = features.takeWhile { _ => count += 1; estimate -= 1; estimate >= 0 }
526       val exported = exporter.export(counter) match {
527         case None    => result
528         case Some(c) => result.map(_ + c).orElse(Some(c))
529       }
530       if (features.isEmpty) {
531         exported
532       } else {
533         // if it's a countable format, the bytes should be available now and we can compare to our chunk size
534         // otherwise, the bytes aren't generally available until after closing the writer,
535         // so we have to go with our initial estimate and adjust after the first chunk
536         if (options.format.countable) {
537           val bytes = exporter.bytes
538           if (estimator.done(bytes)) {
539             nextChunk()
540           } else {
541             estimator.update(bytes, count)
542           }
543         } else {
544           nextChunk()
545         }
546         export(features, exported)
547       }
548     }
549   }
