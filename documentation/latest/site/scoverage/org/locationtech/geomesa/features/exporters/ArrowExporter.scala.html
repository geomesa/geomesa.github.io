23 class ArrowExporter(out: ByteCounterStream, hints: Hints) extends FeatureExporter {

25   import org.locationtech.geomesa.index.conf.QueryHints.RichHints

27   private lazy val sort = hints.getArrowSort
28   private lazy val encoding = SimpleFeatureEncoding.min(hints.isArrowIncludeFid, hints.isArrowProxyFid, hints.isFlipAxisOrder)
29   private lazy val ipc = hints.getArrowFormatVersion.getOrElse(FormatVersion.ArrowFormatVersion.get)
30   private lazy val batchSize = hints.getArrowBatchSize.getOrElse(ArrowProperties.BatchSize.get.toInt)
31   private lazy val dictionaryFields = hints.getArrowDictionaryFields
32   private lazy val flattenFields = hints.isArrowFlatten
33 
34   private var delegate: FeatureExporter = _
35 
36   override def start(sft: SimpleFeatureType): Unit = {
37     delegate = if (sft == org.locationtech.geomesa.arrow.ArrowEncodedSft) {
38       new EncodedDelegate(out)
39     } else if (dictionaryFields.isEmpty) {
40       // note: features should be sorted already, even if arrow encoding wasn't performed
41       new BatchDelegate(out, encoding, FormatVersion.options(ipc), sort, batchSize, Map.empty, flattenFields)
42     } else {
43       if (sort.isDefined) {
44         throw new UnsupportedOperationException("Sorting and calculating dictionaries at the same time is not supported")
45       }
46       new DictionaryDelegate(out, dictionaryFields, encoding, FormatVersion.options(ipc), batchSize, flattenFields)
47     }
48     delegate.start(sft)
49   }
50 
51   override def export(features: Iterator[SimpleFeature]): Option[Long] = delegate.export(features)
52 
53   override def bytes: Long = out.bytes
54 
55   override def close(): Unit = {
56     CloseWithLogging(Option(delegate))
57     out.close()
58   }
59 }

74   private class DictionaryDelegate(
75       os: OutputStream,
76       dictionaryFields: Seq[String],
77       encoding: SimpleFeatureEncoding,
78       ipcOpts: IpcOption,
79       batchSize: Int,
80       flattenStruct: Boolean = false
81     ) extends FeatureExporter {
82 
83     private var writer: DictionaryBuildingWriter = _
84     private var count = 0L
85 
86     override def start(sft: SimpleFeatureType): Unit = {
87       writer = new DictionaryBuildingWriter(sft, dictionaryFields, encoding, ipcOpts, flattenStruct = flattenStruct)
88     }
89 
90     override def export(features: Iterator[SimpleFeature]): Option[Long] = {
91       val start = count
92       features.foreach { f =>
93         writer.add(f)
94         count += 1
95         if (count % batchSize == 0) {
96           writer.encode(os)
97           writer.clear()
98         }
99       }
100       Some(count - start)
101     }
102 
103     override def bytes: Long = 0L
104 
105     override def close(): Unit = {
106       if (writer != null) {
107         if (count % batchSize != 0) {
108           writer.encode(os)
109           writer.clear()
110         }
111         CloseWithLogging(writer)
112       }
113     }
114   }

149         CloseWithLogging(writer)
150       }
