26 /**
27   * GML exporter implementation.
28   *
29   * The geotools GML export classes only support encoding a feature collection. To support our usage
30   * pattern (start, export n times, end), we create an asynchronous feature collection and do the actual
31   * encoding in a separate thread. The encoder thread will block until there are more features to export,
32   * so that we only get a single feature collection in the xml.
33   *
34   * @param out output stream
35   * @param configuration wfs configuration (gml3 vs gml2)
36   */
37 class GmlExporter private (out: ByteCounterStream, configuration: WFSConfiguration)
38     extends FeatureExporter {
39 
40   private val encoder: Encoder = {
41     val props = configuration.getProperties
42     props.add(org.geotools.gml2.GMLConfiguration.OPTIMIZED_ENCODING)
43     props.add(org.geotools.gml2.GMLConfiguration.NO_FEATURE_BOUNDS)
44     val e = new Encoder(configuration)
45     e.getNamespaces.declarePrefix("geomesa", "https://geomesa.org")
46     e.setEncoding(StandardCharsets.UTF_8)
47     e.setIndenting(true)
48     e
49   }
50 
51   private val es = Executors.newSingleThreadExecutor()
52   private var fc: AsyncFeatureCollection = _
53 
54   override def start(sft: SimpleFeatureType): Unit = {
55     fc = new AsyncFeatureCollection(sft)
56     val features = if (sft.getName.getNamespaceURI != null) { fc } else {
57       val builder = new SimpleFeatureTypeBuilder()
58       builder.init(sft)
59       builder.setNamespaceURI("https://geomesa.org")
60       new ReTypingFeatureCollection(fc, builder.buildFeatureType())
61     }
62     val collection = WfsFactory.eINSTANCE.createFeatureCollectionType()
63     collection.getFeature.asInstanceOf[java.util.List[SimpleFeatureCollection]].add(features)
64 
65     def encode(): Unit = encoder.encode(collection, org.geotools.wfs.WFS.FeatureCollection, out)
66 
67     val runnable = new Runnable() {
68       override def run(): Unit = {
69         if (System.getProperty(GmlExporter.TransformerProperty) != null) { encode() } else {
70           // explicitly set the default java transformer, to avoid picking up saxon (which causes errors)
71           // the default class is hard-coded in javax.xml.transform.TransformerFactory.newInstance() ...
72           System.setProperty(GmlExporter.TransformerProperty,
73             "com.sun.org.apache.xalan.internal.xsltc.trax.TransformerFactoryImpl")
74           try { encode() } finally {
75             System.clearProperty(GmlExporter.TransformerProperty)
76           }
77         }
78       }
79     }
80 
81     es.execute(runnable)
82   }
83 
84   override def export(features: Iterator[SimpleFeature]): Option[Long] = {
85     var count = 0L
86     val counting = features.map { f => count += 1; f }
87     while (counting.nonEmpty) {
88       // export in chunks of 100 so that the exporter thread gets notified and doesn't keep blocking
89       fc.addAsync(counting.take(100))
90     }
91     Some(count)
92   }
93 
94   override def bytes: Long = out.bytes
95 
96   override def close(): Unit = {
97     try {
98       if (fc != null) {
99         fc.endAsync()
100       }
101       es.shutdown()
102       es.awaitTermination(Long.MaxValue, TimeUnit.MILLISECONDS)
103     } finally {
104       out.close()
105     }
106   }

131   /**
132     * Feature collection that lets us add additional features in an asynchronous fashion. The consumer
133     * thread will be blocked on calls to 'hasNext' until the producer thread adds features or indicates
134     * completion
135     *
136     * @param sft simple feature type
137     */
138   private class AsyncFeatureCollection(sft: SimpleFeatureType) extends DataFeatureCollection(null, sft) {
139 
140     private val buffer = new ConcurrentLinkedQueue[SimpleFeature]()
141     private val done = new AtomicBoolean(false)
142     private val lock = new ReentrantLock()
143     private val condition = lock.newCondition()
144 
145     private val iter: SimpleFeatureIterator = new SimpleFeatureIterator() {
146 
147       private var current: SimpleFeature = _
148 
149       override def hasNext: Boolean = {
150         if (current != null) {
151           return true
152         }
153         lock.lock()
154         try {
155           current = buffer.poll()
156           // note: we need to loop here to skip 'spurious wake-ups'
157           while (current == null) {
158             if (done.get) {
159               return false
160             }
161             condition.await()
162             current = buffer.poll()
163           }
164           true
165         } finally {
166           lock.unlock()
167         }
168       }
169 
170       override def next(): SimpleFeature = {
171         // note: we shouldn't need to synchronize this as next/hasNext should be a single caller thread
172         val result = current
173         current = null
174         result
175       }
176 
177       override def close(): Unit = endAsync()
178     }
179 
180     /**
181       * Add features to be returned from this feature collection
182       *
183       * @param features features
184       */
185     def addAsync(features: Iterator[SimpleFeature]): Unit = {
186       lock.lock()
187       try {
188         features.foreach(buffer.add)
189         condition.signal()
190       } finally {
191         lock.unlock()
192       }
193     }
194 
195     /**
196       * Signal that there are no more features that will be added
197       */
198     def endAsync(): Unit = {
199       lock.lock()
200       try {
201         done.set(true)
202         condition.signal()
203       } finally {
204         lock.unlock()
205       }
206     }
207 
208     override def features(): SimpleFeatureIterator = iter

211     override def getCount: Int = 0
