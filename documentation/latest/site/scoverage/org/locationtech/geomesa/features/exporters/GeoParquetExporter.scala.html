18 /**
19   * Export to a FileSystem data store format
20   */
21 class GeoParquetExporter(path: String) extends FeatureExporter with LazyLogging {
22 
23   private var writer: ParquetFileSystemWriter = _
24 
25   override def start(sft: SimpleFeatureType): Unit = {
26     if (writer != null) {
27       writer.close()
28       writer = null
29     }
30     // use PathUtils.getUrl to handle local files, otherwise default can be in hdfs
31     val file = new Path(PathUtils.getUrl(path).toURI)
32     val conf = new Configuration()
33     try { Class.forName("org.xerial.snappy.Snappy") } catch {
34       case _: ClassNotFoundException =>
35         logger.warn("SNAPPY compression is not available on the classpath - falling back to GZIP")
36         conf.set("parquet.compression", "GZIP")
37     }
38     SimpleFeatureParquetSchema.setSft(conf, sft)
39     writer = new ParquetFileSystemWriter(file, conf)
40   }
41 
42   override def export(features: Iterator[SimpleFeature]): Option[Long] = {
43     require(writer != null, "Must call 'start' before 'export'")
44     var i = 0L
45     features.foreach { f => writer.write(f); i += 1 }
46     Some(i)
47   }
48 
49   override def bytes: Long = if (writer == null) { 0 } else { writer.getDataSize }
50 
51   override def close(): Unit = {
52     if (writer != null) {
53       writer.close()
54     }
55   }
56 }
