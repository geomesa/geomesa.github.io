19   * Export to a FileSystem data store format
20   */
21 class GeoParquetExporter(path: String) extends FeatureExporter with LazyLogging {
22 
23   import scala.collection.JavaConverters._
24 
25   private var writer: ParquetFileSystemWriter = _
26   private var fs: ObjectStore = _
27 
28   override def start(sft: SimpleFeatureType): Unit = {
29     if (writer != null) {
30       writer.close()
31       writer = null
32     }
33     // TODO make it easier to configure this in a manner consistent with the file system data store
34     val conf = {
35       val c = new Configuration()
36       c.iterator().asScala.map(e => e.getKey -> c.get(e.getKey) /* use .get to resolve envs */).toMap
37     }
38 
39     // use PathUtils.getUrl to handle local file paths (without a scheme)
40     val uri = PathUtils.getUrl(path).toURI
41     fs = uri.getScheme match {
42       case "file" => LocalObjectStore
43       case "s3" | "s3a" => S3ObjectStore(conf)
44       case s => throw new UnsupportedOperationException(s"No implementation available for writing to scheme: $s")
45     }
46 
47     writer = new ParquetFileSystemWriter(fs, conf, sft, uri)
48   }
49 
50   override def export(features: Iterator[SimpleFeature]): Option[Long] = {
51     require(writer != null, "Must call 'start' before 'export'")
52     var i = 0L
53     features.foreach { f => writer.write(f); i += 1 }
54     Some(i)
55   }
56 
57   override def bytes: Long = if (writer == null) { 0 } else { writer.getDataSize }
58 
59   override def close(): Unit = CloseWithLogging(Seq(writer, fs).filter(_ != null))
