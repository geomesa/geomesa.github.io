26   private var fs: ObjectStore = _

28   override def start(sft: SimpleFeatureType): Unit = {
29     if (writer != null) {
30       writer.close()
31       writer = null
32     }
33     // TODO make it easier to configure this in a manner consistent with the file system data store
34     val hadoopConf = new Configuration()
35     try { Class.forName("org.xerial.snappy.Snappy") } catch {
36       case _: ClassNotFoundException =>
37         logger.warn("SNAPPY compression is not available on the classpath - falling back to GZIP")
38         hadoopConf.set("parquet.compression", "GZIP")
39     }
40     val conf = hadoopConf.iterator().asScala.map(e => e.getKey -> e.getValue).toMap
41 
42     // use PathUtils.getUrl to handle local file paths (without a scheme)
43     val uri = PathUtils.getUrl(path).toURI
44     fs = uri.getScheme match {
45       case "file" => LocalObjectStore
46       case "s3" | "s3a" => S3ObjectStore(conf)
47       case s => throw new UnsupportedOperationException(s"No implementation available for writing to scheme: $s")
48     }
49 
50     writer = new ParquetFileSystemWriter(fs, conf, sft, uri)
51   }
52 
53   override def export(features: Iterator[SimpleFeature]): Option[Long] = {
54     require(writer != null, "Must call 'start' before 'export'")
55     var i = 0L
56     features.foreach { f => writer.write(f); i += 1 }
57     Some(i)
58   }
59 
60   override def bytes: Long = if (writer == null) { 0 } else { writer.getDataSize }
61 
62   override def close(): Unit = CloseWithLogging(Seq(writer, fs).filter(_ != null))
63 }
