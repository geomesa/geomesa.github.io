47   class ExportStream(out: FileHandle, gzip: Option[Int] = None) extends ByteCounterStream {
48 
49     // lowest level - keep track of the bytes we write

51     private val counter = new CountingOutputStream(out.write(CreateMode.Create))

57       new BufferedOutputStream(compressed)
