104   /**
105     * Copy the feature with a new feature type. Attributes will be copied by name, and converted
106     * as necessary. As compared to `copy`, the new feature type does not have to have a compatible schema.
107     *
108     * If the feature already has the desired feature type, it will be returned as-is and not copied.
109     *
110     * This is a shallow copy, in that the attributes and user data values will be shared between the two features
111     *
112     * @param sft new feature type
113     * @param in feature to copy
114     * @return
115     */
116   def retype(sft: SimpleFeatureType, in: SimpleFeature): SimpleFeature = {
117     if (sft == in.getFeatureType) { in } else {
118       val out = new ScalaSimpleFeature(sft, in.getID)
119       var i = 0
120       sft.getAttributeDescriptors.asScala.foreach { d =>
121         out.setAttribute(i, in.getAttribute(d.getLocalName))
122         i += 1
123       }
124       out.getUserData.putAll(in.getUserData)
125       out
126     }
127   }

137   def create(sft: SimpleFeatureType, id: String, values: Any*): ScalaSimpleFeature = {
138     val sf = new ScalaSimpleFeature(sft, id)

142       i += 1
