76   /**
77    * Group converter for a record. Since these are geomesa records, we use our custom parquet read support
78    * so that values get parsed into standard simple feature attributes (including geometries) instead
79    * of generic avro types
80    *
81    * @param fileSchema parquet schema
82    * @param schema geomesa schema encodings, if it's a geomesa file
83    */
84   private class SimpleFeatureGroupConverter(fileSchema: MessageType, schema: SimpleFeatureParquetSchema)
85       extends GroupConverter with ValueMaterializer[GenericRecord] {
86 
87     private val avroSchema = new AvroSchemaConverter().convert(fileSchema)
88     private val delegate = new SimpleFeatureReadSupport.SimpleFeatureGroupConverter(schema)
89 
90     override def getConverter(fieldIndex: Int): Converter = delegate.getConverter(fieldIndex)
91     override def start(): Unit = delegate.start()
92     override def end(): Unit = delegate.end()
93     override def reset(): Unit = delegate.reset()
94     override def materialize(): GenericRecord = {
95       val rec = new GenericData.Record(avroSchema)
96       var i = 0
97       while (i < delegate.fieldCount) {
98         rec.put(i, delegate.getConverter(i).materialize())
99         i += 1
100       }
101       rec
102     }
103   }

105   /**
106    * Group converter for a record
107    *
108    * @param fileSchema parquet schema
109    */
110   private class GenericGroupConverter(fileSchema: MessageType) extends GroupConverter with ValueMaterializer[GenericRecord] {
111 
112     private val avroSchema = new AvroSchemaConverter().convert(fileSchema)
113     private val converters = fileSchema.getFields.asScala.map(converter).toArray
114 
115     private var rec: GenericRecord = _
116 
117     override def getConverter(fieldIndex: Int): Converter = converters(fieldIndex)
118     override def start(): Unit = {
119       rec = new GenericData.Record(avroSchema)
120       converters.foreach(_.reset())
121     }
122     override def end(): Unit = {
123       var i = 0
124       while (i < converters.length) {
125         rec.put(i, converters(i).materialize())
126         i += 1
127       }
128     }
129     override def reset(): Unit = rec = null
130     override def materialize(): GenericRecord = rec
131   }

133   /**
134     * Get a converter for a field type
135     *
136     * @param field field
137     * @return
138     */
139   private def converter(field: Type): ValueMaterializer[_ <: AnyRef] = {
140     val logical = field.getLogicalTypeAnnotation
141     if (field.isPrimitive) {
142       lazy val logicalConverter = logical match {
143         case _: StringLogicalTypeAnnotation => Some(new StringConverter())
144         case _: DateLogicalTypeAnnotation => Some(new DaysConverter())
145         case t: TimestampLogicalTypeAnnotation if t.getUnit == LogicalTypeAnnotation.TimeUnit.MILLIS => Some(new MillisConverter())
146         case t: TimestampLogicalTypeAnnotation if t.getUnit == LogicalTypeAnnotation.TimeUnit.MICROS => Some(new DateMicrosConverter())
147         case t: TimestampLogicalTypeAnnotation if t.getUnit == LogicalTypeAnnotation.TimeUnit.NANOS => Some(new NanosConverter())
148         case _: UUIDLogicalTypeAnnotation => Some(new UuidConverter())
149         case _ => None
150       }
151       val convert = field.asPrimitiveType().getPrimitiveTypeName match {
152         case PrimitiveTypeName.BINARY               => logicalConverter.getOrElse(new GeneralPrimitiveConverter())
153         case PrimitiveTypeName.INT32                => logicalConverter.getOrElse(new GeneralPrimitiveConverter())
154         case PrimitiveTypeName.INT64                => logicalConverter.getOrElse(new GeneralPrimitiveConverter())
155         case PrimitiveTypeName.DOUBLE               => new GeneralPrimitiveConverter()
156         case PrimitiveTypeName.FLOAT                => new GeneralPrimitiveConverter()
157         case PrimitiveTypeName.BOOLEAN              => new GeneralPrimitiveConverter()
158         case PrimitiveTypeName.FIXED_LEN_BYTE_ARRAY => logicalConverter.getOrElse(new GeneralPrimitiveConverter())
159         case _                                      => NullPrimitiveConverter
160       }
161       if (field.isRepetition(Repetition.REPEATED)) {
162         new RepeatedPrimitiveConverter(convert)
163       } else {
164         convert
165       }
166     } else {
167       val group = field.asGroupType()
168       def genericGroupConverter(): GenericGroupConverter = new GenericGroupConverter(new MessageType(group.getName, group.getFields))
169       logical match {
170         case _: ListLogicalTypeAnnotation =>
171           ParquetConverterFactory.getListElementType(group) match {
172             case Some(items) => new ListConverter(converter(items))
173             case _ => genericGroupConverter()
174           }
175 
176         case _: MapLogicalTypeAnnotation =>
177           ParquetConverterFactory.getMapKeyValueTypes(group) match {
178             case Some((k, v)) => new MapConverter(converter(k), converter(v))
179             case _ => genericGroupConverter()
180           }
181 
182         case _ =>
183           genericGroupConverter()
184       }
185     }
186   }

188   /**
189    * Converter for primitive fields with repetition of 'repeated'
190    *
191    * @param delegate single value converter
192    */
193   private class RepeatedPrimitiveConverter(delegate: PrimitiveConverter with ValueMaterializer[_ <: AnyRef])
194     extends PrimitiveConverter with ValueMaterializer[java.util.List[AnyRef]] {
195 
196     private var list: java.util.List[AnyRef] = _
197 
198     override def addBinary(value: Binary): Unit = { delegate.addBinary(value); addElement() }
199     override def addBoolean(value: Boolean): Unit = { delegate.addBoolean(value); addElement() }
200     override def addInt(value: Int): Unit = { delegate.addInt(value); addElement() }
201     override def addFloat(value: Float): Unit = { delegate.addFloat(value); addElement() }
202     override def addLong(value: Long): Unit = { delegate.addLong(value); addElement() }
203     override def addDouble(value: Double): Unit = { delegate.addDouble(value); addElement() }
204 
205     private def addElement(): Unit = {
206       if (list == null) {
207         list = new java.util.ArrayList[AnyRef]()
208       }
209       list.add(delegate.materialize())
210       delegate.reset()
211     }
212     override def reset(): Unit = list = null
213     override def materialize(): java.util.List[AnyRef] = list
214   }
