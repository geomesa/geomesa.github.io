74   /**
75    * Group converter for a record. Since these are geomesa records, we use our custom parquet read support
76    * so that values get parsed into standard simple feature attributes (including geometries) instead
77    * of generic avro types
78    *
79    * @param fileSchema parquet schema
80    * @param schema geomesa schema encodings, if it's a geomesa file
81    */
82   private class SimpleFeatureGroupConverter(fileSchema: MessageType, schema: SimpleFeatureParquetSchema)
83       extends GroupConverter with ValueMaterializer[GenericRecord] {
84 
85     private val avroSchema = new AvroSchemaConverter().convert(fileSchema)
86     private val delegate = new SimpleFeatureReadSupport.SimpleFeatureGroupConverter(schema)
87 
88     override def getConverter(fieldIndex: Int): Converter = delegate.getConverter(fieldIndex)
89     override def start(): Unit = delegate.start()
90     override def end(): Unit = delegate.end()
91     override def reset(): Unit = delegate.reset()
92     override def materialize(): GenericRecord = {
93       val rec = new GenericData.Record(avroSchema)
94       var i = 0
95       while (i < delegate.fieldCount) {
96         rec.put(i, delegate.getConverter(i).materialize())
97         i += 1
98       }
99       rec
100     }
101   }

103   /**
104    * Group converter for a record
105    *
106    * @param fileSchema parquet schema

108   private class GenericGroupConverter(fileSchema: MessageType) extends GroupConverter with ValueMaterializer[GenericRecord] {
109 
110     private val avroSchema = new AvroSchemaConverter().convert(fileSchema)
111     private val converters = fileSchema.getFields.asScala.map(converter).toArray
112 
113     private var rec: GenericRecord = _
114 
115     override def getConverter(fieldIndex: Int): Converter = converters(fieldIndex)
116     override def start(): Unit = {
117       rec = new GenericData.Record(avroSchema)
118       converters.foreach(_.reset())
119     }
120     override def end(): Unit = {
121       var i = 0
122       while (i < converters.length) {
123         rec.put(i, converters(i).materialize())
124         i += 1
125       }
126     }
127     override def reset(): Unit = rec = null
128     override def materialize(): GenericRecord = rec
129   }

131   /**
132     * Get a converter for a field type
133     *
134     * @param field field
135     * @return
136     */
137   private def converter(field: Type): ValueMaterializer[_ <: AnyRef] = {
138     val logical = field.getLogicalTypeAnnotation
139     if (field.isPrimitive) {
140       lazy val logicalConverter = logical match {
141         case _: StringLogicalTypeAnnotation => Some(new StringConverter())
142         case _: DateLogicalTypeAnnotation => Some(new DaysConverter())
143         case t: TimestampLogicalTypeAnnotation if t.getUnit == LogicalTypeAnnotation.TimeUnit.MILLIS => Some(new MillisConverter())
144         case t: TimestampLogicalTypeAnnotation if t.getUnit == LogicalTypeAnnotation.TimeUnit.MICROS => Some(new DateMicrosConverter())
145         case t: TimestampLogicalTypeAnnotation if t.getUnit == LogicalTypeAnnotation.TimeUnit.NANOS => Some(new NanosConverter())
146         case _: UUIDLogicalTypeAnnotation => Some(new UuidConverter())
147         case _ => None
148       }
149       val convert = field.asPrimitiveType().getPrimitiveTypeName match {
150         case PrimitiveTypeName.BINARY               => logicalConverter.getOrElse(new GeneralPrimitiveConverter())
151         case PrimitiveTypeName.INT32                => logicalConverter.getOrElse(new GeneralPrimitiveConverter())
152         case PrimitiveTypeName.INT64                => logicalConverter.getOrElse(new GeneralPrimitiveConverter())
153         case PrimitiveTypeName.DOUBLE               => new GeneralPrimitiveConverter()
154         case PrimitiveTypeName.FLOAT                => new GeneralPrimitiveConverter()
155         case PrimitiveTypeName.BOOLEAN              => new GeneralPrimitiveConverter()
156         case PrimitiveTypeName.FIXED_LEN_BYTE_ARRAY => logicalConverter.getOrElse(new GeneralPrimitiveConverter())
157         case _                                      => NullPrimitiveConverter
158       }
159       if (field.isRepetition(Repetition.REPEATED)) {
160         new RepeatedPrimitiveConverter(convert)
161       } else {
162         convert
163       }
164     } else {
165       val group = field.asGroupType()
166       def genericGroupConverter(): GenericGroupConverter = new GenericGroupConverter(new MessageType(group.getName, group.getFields))
167       logical match {
168         case _: ListLogicalTypeAnnotation =>
169           ParquetConverterFactory.getListElementType(group) match {
170             case Some(items) => new ListConverter(converter(items))
171             case _ => genericGroupConverter()
172           }
173 
174         case _: MapLogicalTypeAnnotation =>
175           ParquetConverterFactory.getMapKeyValueTypes(group) match {
176             case Some((k, v)) => new MapConverter(converter(k), converter(v))
177             case _ => genericGroupConverter()
178           }
179 
180         case _ =>
181           genericGroupConverter()
182       }
183     }
184   }

186   /**
187    * Converter for primitive fields with repetition of 'repeated'
188    *
189    * @param delegate single value converter
190    */
191   private class RepeatedPrimitiveConverter(delegate: PrimitiveConverter with ValueMaterializer[_ <: AnyRef])
192     extends PrimitiveConverter with ValueMaterializer[java.util.List[AnyRef]] {
193 
194     private var list: java.util.List[AnyRef] = _
195 
196     override def addBinary(value: Binary): Unit = { delegate.addBinary(value); addElement() }
197     override def addBoolean(value: Boolean): Unit = { delegate.addBoolean(value); addElement() }
198     override def addInt(value: Int): Unit = { delegate.addInt(value); addElement() }
199     override def addFloat(value: Float): Unit = { delegate.addFloat(value); addElement() }
200     override def addLong(value: Long): Unit = { delegate.addLong(value); addElement() }
201     override def addDouble(value: Double): Unit = { delegate.addDouble(value); addElement() }
202 
203     private def addElement(): Unit = {
204       if (list == null) {
205         list = new java.util.ArrayList[AnyRef]()
206       }
207       list.add(delegate.materialize())
208       delegate.reset()
209     }
210     override def reset(): Unit = list = null
211     override def materialize(): java.util.List[AnyRef] = list
212   }
