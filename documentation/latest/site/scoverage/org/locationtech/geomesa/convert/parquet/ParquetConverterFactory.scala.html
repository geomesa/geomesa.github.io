48   /**
49     * Handles parquet files (including those produced by the FSDS and CLI export)
50     *
51     * @param is input
52     * @param sft simple feature type, if known ahead of time
53     * @param hints hints, including the file path, if there is a file available
54     * @return
55     */
56   override def infer(
57       is: InputStream,
58       sft: Option[SimpleFeatureType],
59       hints: Map[String, AnyRef]): Try[(SimpleFeatureType, Config)] = {
60     val path = hints.get(EvaluationContext.InputFilePathKey) match {
61       case None => Failure(new RuntimeException("No file path specified to the input data"))
62       case Some(p) => Success(p.toString)
63     }
64     path.flatMap { p =>
65       Try {
66         // note: get the path as a URI so that we handle local files appropriately
67         val filePath = new Path(PathUtils.getUrl(p).toURI)
68         val footer = ParquetFileReader.readFooter(new Configuration(), filePath, ParquetMetadataConverter.NO_FILTER)
69         val (schema, fields, id, userData) = Try(SimpleFeatureParquetSchema.read(footer.getFileMetaData)).toOption.flatten match {
70           case Some(parquet) =>
71             // this is a geomesa encoded parquet file
72             val fields = parquet.sft.getAttributeDescriptors.asScala.map { descriptor =>
73               val name = StringSerialization.alphaNumericSafeString(descriptor.getLocalName)
74               // note: parquet converter stores the generic record under index 0
75               BasicField(descriptor.getLocalName, Some(Expression(s"avroPath($$0, '/$name')")))
76             }
77             val id = Expression(s"avroPath($$0, '/${SimpleFeatureSchema.FeatureIdField}')")
78             val userData =
79               Map("geomesa.feature.visibility" -> Expression(s"avroPath($$0, '/${SimpleFeatureSchema.VisibilitiesField}')"))
80 
81             // validate the existing schema, if any
82             sft.foreach { schema =>
83               if (schema.getAttributeDescriptors.asScala != parquet.sft.getAttributeDescriptors.asScala) {
84                 throw new IllegalArgumentException(
85                   s"Inferred schema does not match existing schema:\n  " +
86                     s"Existing schema: ${SimpleFeatureTypes.encodeType(schema)}\n  " +
87                     s"Inferred schema: ${SimpleFeatureTypes.encodeType(parquet.sft)}")
88               }
89             }
90             (parquet.sft, fields, Some(id), userData)
91 
92           case _ =>
93             // this is an arbitrary parquet file, create fields based on the schema
94             val types = ParquetConverterFactory.schemaTypes(footer.getFileMetaData)
95             val dataSft = TypeInference.schema("inferred-parquet", types)
96             // note: parquet converter stores the generic record under index 0
97             val fields = types.map(t => BasicField(t.name, Some(Expression(t.transform.apply(0)))))
98 
99             // validate the existing schema, if any
100             sft.foreach(AbstractConverterFactory.validateInferredType(_, types.map(_.typed)))
101 
102             (dataSft, fields, None, Map.empty[String, Expression])
103         }
104 
105         val converterConfig = BasicConfig(typeToProcess, None, id, Map.empty, userData)
106 
107         val config = configConvert.to(converterConfig)
108             .withFallback(fieldConvert.to(fields.toSeq))
109             .withFallback(optsConvert.to(BasicOptions.default))
110             .toConfig
111 
112         (schema, config)
113       }
114     }
115   }
116 }

124   /**
125    * Gets the type of elements in this list, assuming that this is a standard, 3-level parquet list type
126    *
127    * @param group list type
128    * @return element type, if input is a properly formatted 3-level parquet list
129    */
130   def getListElementType(group: GroupType): Option[GroupType] = {
131     for {
132       g <- Some(group) // optional group foo (LIST) {
133       if g.getFieldCount == 1 && !g.getType(0).isPrimitive
134       list = g.getType(0).asGroupType() // repeated group list {
135       if list.getFieldCount == 1 && list.isRepetition(Repetition.REPEATED) && !list.getType(0).isPrimitive
136     } yield {
137       list.getType(0).asGroupType() // optional group element {
138     }
139   }

141   /**
142    * Gets the type of keys and values in this map, assuming that this is a standard, 3-level parquet map type
143    *
144    * @param map map type
145    * @return (key type, value type), if input is a properly formatted 3-level parquet map
146    */
147   def getMapKeyValueTypes(map: GroupType): Option[(Type, Type)] = {
148     for {
149       mapGroup <- Some(map)
150       if mapGroup.getFieldCount == 1
151       tuples = mapGroup.getType(0)
152       if !tuples.isPrimitive
153       tuplesGroup = tuples.asGroupType()
154       if tuplesGroup.getFieldCount == 2 && tuplesGroup.isRepetition(Repetition.REPEATED)
155     } yield {
156       (tuplesGroup.getType(0), tuplesGroup.getType(1))
157     }
158   }

161     * Takes parquet file info returns the simple feature type bindings for it
162     *
163     * @param fileMetaData parquet file footer
164     * @return
165     */
166   private def schemaTypes(fileMetaData: FileMetaData): Seq[InferredType] = {
167     val namer = new Namer()
168     val geos: Option[GeoParquetMetadata] =
169       Option(fileMetaData.getKeyValueMetaData.get(GeoParquetMetadata.GeoParquetMetadataKey)).map(GeoParquetMetadata.fromJson)
170 
171     val types = fileMetaData.getSchema.getFields.asScala.flatMap(mapField(_, namer, geos)).toSeq
172     // check if we can derive a geometry field
173     val geom = TypeInference.deriveGeometry(types, namer)
174     types ++ geom
175   }

177   /**
178    * Infer a type from a top-level parquet field
179    *
180    * @param field field
181    * @param namer name helper
182    * @param geos extracted GeoParquet metadata from the parquet file
183    * @return
184    */
185   private def mapField(field: Type, namer: Namer, geos: Option[GeoParquetMetadata]): Seq[InferredType] = {
186     if (geos.exists(_.geometries.exists(_.covering.contains(field.getName)))) {
187       return Seq.empty // ignore bbox fields
188     }
189     // note: geometries *must* be at the root level, so we don't need to track them for nested fields
190     geos.flatMap(_.geometries.find(_.name == field.getName)) match {
191       case None => mapNormalField(field, namer)
192       case Some(geo) => mapGeoField(field, namer, geo)
193     }
194   }

196   /**
197    * Maps a non-geometry type field
198    *
199    * @param field field
200    * @param namer name helper
201    * @param path avro path to the parent of the field
202    * @return
203    */
204   private def mapNormalField(field: Type, namer: Namer, path: String = ""): Seq[InferredType] = {
205     val fieldPath = s"$path/${field.getName}"
206     lazy val name = namer(fieldPath)
207     val transform = FunctionTransform("avroPath(", s",'$fieldPath')")
208     val logical = field.getLogicalTypeAnnotation
209     if (field.isPrimitive) {
210       // note: date field transforms are handled by AvroReadSupport
211       lazy val intOrDate: InferredType = logical match {
212         case _: DateLogicalTypeAnnotation => InferredType(name, ObjectType.DATE, transform)
213         case _ => InferredType(name, ObjectType.INT, transform)
214       }
215       lazy val longOrTimestamp: InferredType = logical match {
216         case t: TimestampLogicalTypeAnnotation if t.getUnit == LogicalTypeAnnotation.TimeUnit.MILLIS => InferredType(name, ObjectType.DATE, transform)
217         case t: TimestampLogicalTypeAnnotation if t.getUnit == LogicalTypeAnnotation.TimeUnit.MICROS => InferredType(name, ObjectType.DATE, transform)
218         case _ => InferredType(name, ObjectType.LONG, transform)
219       }
220       lazy val stringOrBytes: InferredType = logical match {
221         case _: StringLogicalTypeAnnotation => InferredType(name, ObjectType.STRING, transform)
222         case _ => InferredType(name, ObjectType.BYTES, transform)
223       }
224       field.asPrimitiveType().getPrimitiveTypeName match {
225         case PrimitiveTypeName.BINARY               => Seq(stringOrBytes)
226         case PrimitiveTypeName.INT32                => Seq(intOrDate)
227         case PrimitiveTypeName.INT64                => Seq(longOrTimestamp)
228         case PrimitiveTypeName.FLOAT                => Seq(InferredType(name, ObjectType.FLOAT, transform))
229         case PrimitiveTypeName.DOUBLE               => Seq(InferredType(name, ObjectType.DOUBLE, transform))
230         case PrimitiveTypeName.BOOLEAN              => Seq(InferredType(name, ObjectType.BOOLEAN, transform))
231         case PrimitiveTypeName.FIXED_LEN_BYTE_ARRAY => Seq(InferredType(name, ObjectType.BYTES, transform))
232         case _ => Seq.empty
233       }
234     } else {
235       val group = field.asGroupType()
236       def jsonTransform(): FunctionTransform = FunctionTransform(s"avroToJson(${transform.prefix}", s"${transform.suffix})")
237       logical match {
238         case _: ListLogicalTypeAnnotation =>
239           if (getListElementType(group).exists(e => e.getFieldCount == 1 && e.getType(0).isPrimitive)) {
240             Seq(InferredType(name, ObjectType.LIST, transform))
241           } else {
242             // for non-standard or complex-element lists, we just convert to a json string
243             Seq(InferredType(name, ObjectType.JSON, jsonTransform()))
244           }
245 
246         case _: MapLogicalTypeAnnotation =>
247           if (getMapKeyValueTypes(group).exists { case (k, v) => k.isPrimitive && v.isPrimitive }) {
248             Seq(InferredType(name, ObjectType.MAP, transform))
249           } else {
250             // for non-standard or complex-value maps, we just convert to a json string
251             Seq(InferredType(name, ObjectType.JSON, jsonTransform()))
252           }
253 
254         case _ =>
255           group.getFields.asScala.flatMap(mapNormalField(_, namer, fieldPath)).toSeq
256       }
257     }
258   }

260   /**
261    * Maps a GeoParquet geometry field
262    *
263    * @param field field
264    * @param namer name helper
265    * @param geo GeoParquet metadata
266    * @return
267    */
268   private def mapGeoField(field: Type, namer: Namer, geo: ColumnMetadata): Seq[InferredType] = {
269     lazy val name = namer(field.getName)
270 
271     def geomTransform(fn: String): FunctionTransform = FunctionTransform(s"$fn(avroPath(", s",'/${field.getName}'))")
272 
273     def mismatch(): Seq[InferredType] = {
274       logger.warn(s"Found GeoParquet metadata for column '${field.getName}' but the schema does not match the expected type")
275       mapNormalField(field, namer)
276     }
277 
278     if (geo.encoding == GeoParquetColumnEncoding.WKB) {
279       if (isWkbType(field)) {
280         Seq(InferredType(name, GeoParquetColumnType.toObjectType(geo.types), geomTransform("geometry")))
281       } else {
282         mismatch()
283       }
284     } else {
285       val objectType = GeoParquetColumnEncoding.toObjectType(geo.encoding)
286       if ((geo.types.isEmpty || GeoParquetColumnType.toObjectType(geo.types) == objectType) && isNativeType(field, objectType)) {
287         Seq(InferredType(name, objectType, geomTransform(geo.encoding.toString)))
288       } else {
289         mismatch()
290       }
291     }
292   }

294   /**
295    * Checks if a field corresponds to the GeoParquet schema for a WKB-encoded geometry
296    *
297    * @param field field
298    * @return
299    */
300   private def isWkbType(field: Type): Boolean =
301     !field.isRepetition(Repetition.REPEATED) && field.isPrimitive &&
302       field.asPrimitiveType().getPrimitiveTypeName == PrimitiveTypeName.BINARY
303 
304   /**
305    * Checks if the field corresponds to the GeoParquet schema for a native-encoded geometry
306    *
307    * @param field field
308    * @param objectType native type
309    * @return
310    */
311   private def isNativeType(field: Type, objectType: ObjectType): Boolean = {
312     if (field.isRepetition(Repetition.REPEATED)) { false } else {
313       // custom compare, so we don't consider optional vs required, id, name, logicalTypes, etc
314       compareGeomFields(field, GeoParquetNative.schema(objectType, new AtomicInteger(1), new AtomicInteger(1000)).named(field.getName))
315     }
316   }
317 
318   private def compareGeomFields(left: Type, right: Type): Boolean = {
319     // make sure they're both either repeated or not, but don't consider required vs optional
320     if (left.isRepetition(Repetition.REPEATED) != right.isRepetition(Repetition.REPEATED)) {
321       false
322     } else if (left.isPrimitive) {
323       right.isPrimitive &&
324         right.asPrimitiveType().getPrimitiveTypeName == left.asPrimitiveType().getPrimitiveTypeName
325     } else {
326       !right.isPrimitive && {
327         val leftFields = left.asGroupType().getFields
328         val rightFields = right.asGroupType().getFields
329         if (leftFields.size() != rightFields.size()) { false } else {
330           leftFields.asScala.zip(rightFields.asScala).forall { case (f, n) => compareGeomFields(f, n) }
331         }
332       }
333     }
