83   /**
84     * Index a polygon by its bounding box
85     *
86     * @param xmin min x value in xBounds
87     * @param ymin min y value in yBounds
88     * @param xmax max x value in xBounds, must be >= xmin
89     * @param ymax max y value in yBounds, must be >= ymin
90     * @param lenient standardize boundaries to valid values, or raise an exception
91     * @return z value for the bounding box
92     */
93   def index(xmin: Double, ymin: Double, xmax: Double, ymax: Double, lenient: Boolean = false): Long = {
94     // normalize inputs to [0,1]
95     val (nxmin, nymin, nxmax, nymax) = normalize(xmin, ymin, xmax, ymax, lenient)
96 
97     // calculate the length of the sequence code (section 4.1 of XZ-Ordering paper)
98 
99     val maxDim = math.max(nxmax - nxmin, nymax - nymin)
100 
101     // l1 (el-one) is a bit confusing to read, but corresponds with the paper's definitions
102     val l1 = math.floor(math.log(maxDim) / XZSFC.LogPointFive).toInt
103 
104     // the length will either be (l1) or (l1 + 1)
105     val length = if (l1 >= g) { g } else {
106       val w2 = math.pow(0.5, l1 + 1) // width of an element at resolution l2 (l1 + 1)
107 
108       // predicate for checking how many axis the polygon intersects
109       // math.floor(min / w2) * w2 == start of cell containing min
110       def predicate(min: Double, max: Double): Boolean = max <= (math.floor(min / w2) * w2) + (2 * w2)
111 
112       if (predicate(nxmin, nxmax) && predicate(nymin, nymax)) l1 + 1 else l1
113     }
114 
115     sequenceCode(nxmin, nymin, length)
116   }

190   /**
191     * Determine XZ-curve ranges that will cover a given query window
192     *
193     * @param query a sequence of OR'd windows to cover, normalized to [0,1]
194     * @param rangeStop a rough max value for the number of ranges to return
195     * @return
196     */
197   private def ranges(query: Array[QueryWindow], rangeStop: Int): Seq[IndexRange] = {
198 
199     import XZ2SFC.{LevelOneElements, LevelTerminator}
200 
201     // stores our results - initial size of 100 in general saves us some re-allocation
202     val ranges = new java.util.ArrayList[IndexRange](100)
203 
204     // values remaining to process - initial size of 100 in general saves us some re-allocation
205     val remaining = new java.util.ArrayDeque[XElement](100)
206 
207     // checks if a quad is contained in the search space
208     def isContained(quad: XElement): Boolean = {
209       var i = 0
210       while (i < query.length) {
211         if (quad.isContained(query(i))) {
212           return true
213         }
214         i += 1
215       }
216       false
217     }
218 
219     // checks if a quad overlaps the search space
220     def isOverlapped(quad: XElement): Boolean = {
221       var i = 0
222       while (i < query.length) {
223         if (quad.overlaps(query(i))) {
224           return true
225         }
226         i += 1
227       }
228       false
229     }
230 
231     // checks a single value and either:
232     //   eliminates it as out of bounds
233     //   adds it to our results as fully matching, or
234     //   adds it to our results as partial matching and queues up it's children for further processing
235     def checkValue(quad: XElement, level: Short): Unit = {
236       if (isContained(quad)) {
237         // whole range matches, happy day
238         val (min, max) = sequenceInterval(quad.xmin, quad.ymin, level, partial = false)
239         ranges.add(IndexRange(min, max, contained = true))
240       } else if (isOverlapped(quad)) {
241         // some portion of this range is excluded
242         // add the partial match and queue up each sub-range for processing
243         val (min, max) = sequenceInterval(quad.xmin, quad.ymin, level, partial = true)
244         ranges.add(IndexRange(min, max, contained = false))
245         quad.children.foreach(remaining.add)
246       }
247     }
248 
249     // initial level
250     LevelOneElements.foreach(remaining.add)
251     remaining.add(LevelTerminator)
252 
253     // level of recursion
254     var level: Short = 1
255 
256     while (level < g && !remaining.isEmpty && ranges.size < rangeStop) {
257       val next = remaining.poll
258       if (next.eq(LevelTerminator)) {
259         // we've fully processed a level, increment our state
260         if (!remaining.isEmpty) {
261           level = (level + 1).toShort
262           remaining.add(LevelTerminator)
263         }
264       } else {
265         checkValue(next, level)
266       }
267     }
268 
269     // bottom out and get all the ranges that partially overlapped but we didn't fully process
270     while (!remaining.isEmpty) {
271       val quad = remaining.poll
272       if (quad.eq(LevelTerminator)) {
273         level = (level + 1).toShort
274       } else {
275         val (min, max) = sequenceInterval(quad.xmin, quad.ymin, level, partial = false)
276         ranges.add(IndexRange(min, max, contained = false))
277       }
278     }
279 
280     // we've got all our ranges - now reduce them down by merging overlapping values
281     // note: we don't bother reducing the ranges as in the XZ paper, as accumulo handles lots of ranges fairly well
282     ranges.sort(IndexRange.IndexRangeIsOrdered)
283 
284     var current = ranges.get(0) // note: should always be at least one range
285     val result = ArrayBuffer.empty[IndexRange]
286     var i = 1
287     while (i < ranges.size()) {
288       val range = ranges.get(i)
289       if (range.lower <= current.upper + 1) {
290         // merge the two ranges
291         current = IndexRange(current.lower, math.max(current.upper, range.upper), current.contained && range.contained)
292       } else {
293         // append the last range and set the current range for future merging
294         result.append(current)
295         current = range
296       }
297       i += 1
298     }
299     // append the last range - there will always be one left that wasn't added
300     result.append(current)
301 
302     result.toSeq
303   }

305   /**
306     * Computes the sequence code for a given point - for polygons this is the lower-left corner.
307     *
308     * Based on Definition 2 from the XZ-Ordering paper
309     *
310     * @param x normalized x value [0,1]
311     * @param y normalized y value [0,1]
312     * @param length length of the sequence code that will be generated
313     * @return
314     */
315   private def sequenceCode(x: Double, y: Double, length: Int): Long = {
316     var xmin = 0.0
317     var ymin = 0.0
318     var xmax = 1.0
319     var ymax = 1.0
320 
321     var cs = 0L
322 
323     var i = 0
324     while (i < length) {
325       val xCenter = (xmin + xmax) / 2.0
326       val yCenter = (ymin + ymax) / 2.0
327       (x < xCenter, y < yCenter) match {
328         case (true,  true)  => cs += 1L                                             ; xmax = xCenter; ymax = yCenter
329         case (false, true)  => cs += 1L + 1L * (math.pow(4, g - i).toLong - 1L) / 3L; xmin = xCenter; ymax = yCenter
330         case (true,  false) => cs += 1L + 2L * (math.pow(4, g - i).toLong - 1L) / 3L; xmax = xCenter; ymin = yCenter
331         case (false, false) => cs += 1L + 3L * (math.pow(4, g - i).toLong - 1L) / 3L; xmin = xCenter; ymin = yCenter
332       }
333       i += 1
334     }
335 
336     cs
337   }

339   /**
340     * Computes an interval of sequence codes for a given point - for polygons this is the lower-left corner.
341     *
342     * @param x normalized x value [0,1]
343     * @param y normalized y value [0,1]
344     * @param length length of the sequence code that will be used as the basis for this interval
345     * @param partial true if the element partially intersects the query window, false if it is fully contained
346     * @return
347     */
348   private def sequenceInterval(x: Double, y: Double, length: Short, partial: Boolean): (Long, Long) = {
349     val min = sequenceCode(x, y, length)
350     // if a partial match, we just use the single sequence code as an interval
351     // if a full match, we have to match all sequence codes starting with the single sequence code
352     val max = if (partial) { min } else {
353       // from lemma 3 in the XZ-Ordering paper
354       min + (math.pow(4, g - length + 1).toLong - 1L) / 3L
355     }
356     (min, max)
357   }

359   /**
360     * Normalize user space values to [0,1]
361     *
362     * @param xmin min x value in user space
363     * @param ymin min y value in user space
364     * @param xmax max x value in user space, must be >= xmin
365     * @param ymax max y value in user space, must be >= ymin
366     * @param lenient standardize boundaries to valid values, or raise an exception
367     * @return
368     */
369   private def normalize(xmin: Double,
370                         ymin: Double,
371                         xmax: Double,
372                         ymax: Double,
373                         lenient: Boolean): (Double, Double, Double, Double) = {
374     require(xmin <= xmax && ymin <= ymax, s"Bounds must be ordered: [$xmin $xmax] [$ymin $ymax]")
375 
376     try {
377       require(xmin >= xLo && xmax <= xHi && ymin >= yLo && ymax <= yHi,
378         s"Values out of bounds ([$xLo $xHi] [$yLo $yHi]): [$xmin $xmax] [$ymin $ymax]")
379 
380       val nxmin = (xmin - xLo) / xSize
381       val nymin = (ymin - yLo) / ySize
382       val nxmax = (xmax - xLo) / xSize
383       val nymax = (ymax - yLo) / ySize
384 
385       (nxmin, nymin, nxmax, nymax)
386     } catch {
387       case _: IllegalArgumentException if lenient =>
388 
389         val bxmin = if (xmin < xLo) { xLo } else if (xmin > xHi) { xHi } else { xmin }
390         val bymin = if (ymin < yLo) { yLo } else if (ymin > yHi) { yHi } else { ymin }
391         val bxmax = if (xmax < xLo) { xLo } else if (xmax > xHi) { xHi } else { xmax }
392         val bymax = if (ymax < yLo) { yLo } else if (ymax > yHi) { yHi } else { ymax }
393 
394         val nxmin = (bxmin - xLo) / xSize
395         val nymin = (bymin - yLo) / ySize
396         val nxmax = (bxmax - xLo) / xSize
397         val nymax = (bymax - yLo) / ySize
398 
399         (nxmin, nymin, nxmax, nymax)
400     }
401   }

433   /**
434     * An extended Z curve element. Bounds refer to the non-extended z element for simplicity of calculation.
435     *
436     * An extended Z element refers to a normal Z curve element that has it's upper bounds expanded by double it's
437     * width/height. By convention, an element is always square.
438     *
439     * @param xmin x lower bound in [0-1]
440     * @param ymin y lower bound in [0-1]
441     * @param xmax x upper bound in [0-1], must be >= xmin
442     * @param ymax y upper bound in [0-1], must be >= ymin
443     * @param length length of the non-extended side (note: by convention width should be equal to height)
444     */
445   private case class XElement(xmin: Double, ymin: Double, xmax: Double, ymax: Double, length: Double) {
446 
447     // extended x and y bounds
448     lazy private val xext = xmax + length
449     lazy private val yext = ymax + length
450 
451     def isContained(window: QueryWindow): Boolean =
452       window.xmin <= xmin && window.ymin <= ymin && window.xmax >= xext && window.ymax >= yext
453 
454     def overlaps(window: QueryWindow): Boolean =
455       window.xmax >= xmin && window.ymax >= ymin && window.xmin <= xext && window.ymin <= yext
456 
457     def children: Seq[XElement] = {
458       val xCenter = (xmin + xmax) / 2.0
459       val yCenter = (ymin + ymax) / 2.0
460       val len = length / 2.0
461       val c0 = copy(xmax = xCenter, ymax = yCenter, length = len)
462       val c1 = copy(xmin = xCenter, ymax = yCenter, length = len)
463       val c2 = copy(xmax = xCenter, ymin = yCenter, length = len)
464       val c3 = copy(xmin = xCenter, ymin = yCenter, length = len)
465       Seq(c0, c1, c2, c3)
466     }
467   }
