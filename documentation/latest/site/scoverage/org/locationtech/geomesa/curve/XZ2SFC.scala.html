19   *

72   /**
73     * Index a polygon by its bounding box
74     *
75     * @param xmin min x value in xBounds
76     * @param ymin min y value in yBounds
77     * @param xmax max x value in xBounds, must be >= xmin
78     * @param ymax max y value in yBounds, must be >= ymin
79     * @param lenient standardize boundaries to valid values, or raise an exception
80     * @return z value for the bounding box
81     */
82   def index(xmin: Double, ymin: Double, xmax: Double, ymax: Double, lenient: Boolean = false): Long = {
83     // normalize inputs to [0,1]
84     val (nxmin, nymin, nxmax, nymax) = normalize(xmin, ymin, xmax, ymax, lenient)
85 
86     // calculate the length of the sequence code (section 4.1 of XZ-Ordering paper)
87 
88     val maxDim = math.max(nxmax - nxmin, nymax - nymin)
89 
90     // l1 (el-one) is a bit confusing to read, but corresponds with the paper's definitions
91     val l1 = math.floor(math.log(maxDim) / XZSFC.LogPointFive).toInt
92 
93     // the length will either be (l1) or (l1 + 1)
94     val length = if (l1 >= g) { g } else {
95       val w2 = math.pow(0.5, l1 + 1) // width of an element at resolution l2 (l1 + 1)
96 
97       // predicate for checking how many axis the polygon intersects
98       // math.floor(min / w2) * w2 == start of cell containing min
99       def predicate(min: Double, max: Double): Boolean = max <= (math.floor(min / w2) * w2) + (2 * w2)
100 
101       if (predicate(nxmin, nxmax) && predicate(nymin, nymax)) l1 + 1 else l1
102     }
103 
104     sequenceCode(nxmin, nymin, length)
105   }

179   /**
180     * Determine XZ-curve ranges that will cover a given query window
181     *
182     * @param query a sequence of OR'd windows to cover, normalized to [0,1]
183     * @param rangeStop a rough max value for the number of ranges to return
184     * @return
185     */
186   private def ranges(query: Array[QueryWindow], rangeStop: Int): Seq[IndexRange] = {
187 
188     import XZ2SFC.{LevelOneElements, LevelTerminator}
189 
190     // stores our results - initial size of 100 in general saves us some re-allocation
191     val ranges = new java.util.ArrayList[IndexRange](100)
192 
193     // values remaining to process - initial size of 100 in general saves us some re-allocation
194     val remaining = new java.util.ArrayDeque[XElement](100)
195 
196     // checks if a quad is contained in the search space
197     def isContained(quad: XElement): Boolean = {
198       var i = 0
199       while (i < query.length) {
200         if (quad.isContained(query(i))) {
201           return true
202         }
203         i += 1
204       }
205       false
206     }
207 
208     // checks if a quad overlaps the search space
209     def isOverlapped(quad: XElement): Boolean = {
210       var i = 0
211       while (i < query.length) {
212         if (quad.overlaps(query(i))) {
213           return true
214         }
215         i += 1
216       }
217       false
218     }
219 
220     // checks a single value and either:
221     //   eliminates it as out of bounds
222     //   adds it to our results as fully matching, or
223     //   adds it to our results as partial matching and queues up it's children for further processing
224     def checkValue(quad: XElement, level: Short): Unit = {
225       if (isContained(quad)) {
226         // whole range matches, happy day
227         val (min, max) = sequenceInterval(quad.xmin, quad.ymin, level, partial = false)
228         ranges.add(IndexRange(min, max, contained = true))
229       } else if (isOverlapped(quad)) {
230         // some portion of this range is excluded
231         // add the partial match and queue up each sub-range for processing
232         val (min, max) = sequenceInterval(quad.xmin, quad.ymin, level, partial = true)
233         ranges.add(IndexRange(min, max, contained = false))
234         quad.children.foreach(remaining.add)
235       }
236     }
237 
238     // initial level
239     LevelOneElements.foreach(remaining.add)
240     remaining.add(LevelTerminator)
241 
242     // level of recursion
243     var level: Short = 1
244 
245     while (level < g && !remaining.isEmpty && ranges.size < rangeStop) {
246       val next = remaining.poll
247       if (next.eq(LevelTerminator)) {
248         // we've fully processed a level, increment our state
249         if (!remaining.isEmpty) {
250           level = (level + 1).toShort
251           remaining.add(LevelTerminator)
252         }
253       } else {
254         checkValue(next, level)
255       }
256     }
257 
258     // bottom out and get all the ranges that partially overlapped but we didn't fully process
259     while (!remaining.isEmpty) {
260       val quad = remaining.poll
261       if (quad.eq(LevelTerminator)) {
262         level = (level + 1).toShort
263       } else {
264         val (min, max) = sequenceInterval(quad.xmin, quad.ymin, level, partial = false)
265         ranges.add(IndexRange(min, max, contained = false))
266       }
267     }
268 
269     // we've got all our ranges - now reduce them down by merging overlapping values
270     // note: we don't bother reducing the ranges as in the XZ paper, as accumulo handles lots of ranges fairly well
271     ranges.sort(IndexRange.IndexRangeIsOrdered)
272 
273     var current = ranges.get(0) // note: should always be at least one range
274     val result = ArrayBuffer.empty[IndexRange]
275     var i = 1
276     while (i < ranges.size()) {
277       val range = ranges.get(i)
278       if (range.lower <= current.upper + 1) {
279         // merge the two ranges
280         current = IndexRange(current.lower, math.max(current.upper, range.upper), current.contained && range.contained)
281       } else {
282         // append the last range and set the current range for future merging
283         result.append(current)
284         current = range
285       }
286       i += 1
287     }
288     // append the last range - there will always be one left that wasn't added
289     result.append(current)
290 
291     result.toSeq
292   }

294   /**
295     * Computes the sequence code for a given point - for polygons this is the lower-left corner.
296     *
297     * Based on Definition 2 from the XZ-Ordering paper
298     *
299     * @param x normalized x value [0,1]
300     * @param y normalized y value [0,1]
301     * @param length length of the sequence code that will be generated
302     * @return
303     */
304   private def sequenceCode(x: Double, y: Double, length: Int): Long = {
305     var xmin = 0.0
306     var ymin = 0.0
307     var xmax = 1.0
308     var ymax = 1.0
309 
310     var cs = 0L
311 
312     var i = 0
313     while (i < length) {
314       val xCenter = (xmin + xmax) / 2.0
315       val yCenter = (ymin + ymax) / 2.0
316       (x < xCenter, y < yCenter) match {
317         case (true,  true)  => cs += 1L                                             ; xmax = xCenter; ymax = yCenter
318         case (false, true)  => cs += 1L + 1L * (math.pow(4, g - i).toLong - 1L) / 3L; xmin = xCenter; ymax = yCenter
319         case (true,  false) => cs += 1L + 2L * (math.pow(4, g - i).toLong - 1L) / 3L; xmax = xCenter; ymin = yCenter
320         case (false, false) => cs += 1L + 3L * (math.pow(4, g - i).toLong - 1L) / 3L; xmin = xCenter; ymin = yCenter
321       }
322       i += 1
323     }
324 
325     cs
326   }

328   /**
329     * Computes an interval of sequence codes for a given point - for polygons this is the lower-left corner.
330     *
331     * @param x normalized x value [0,1]
332     * @param y normalized y value [0,1]
333     * @param length length of the sequence code that will be used as the basis for this interval
334     * @param partial true if the element partially intersects the query window, false if it is fully contained
335     * @return
336     */
337   private def sequenceInterval(x: Double, y: Double, length: Short, partial: Boolean): (Long, Long) = {
338     val min = sequenceCode(x, y, length)
339     // if a partial match, we just use the single sequence code as an interval
340     // if a full match, we have to match all sequence codes starting with the single sequence code
341     val max = if (partial) { min } else {
342       // from lemma 3 in the XZ-Ordering paper
343       min + (math.pow(4, g - length + 1).toLong - 1L) / 3L
344     }
345     (min, max)
346   }

348   /**
349     * Normalize user space values to [0,1]
350     *
351     * @param xmin min x value in user space
352     * @param ymin min y value in user space
353     * @param xmax max x value in user space, must be >= xmin
354     * @param ymax max y value in user space, must be >= ymin
355     * @param lenient standardize boundaries to valid values, or raise an exception
356     * @return
357     */
358   private def normalize(xmin: Double,
359                         ymin: Double,
360                         xmax: Double,
361                         ymax: Double,
362                         lenient: Boolean): (Double, Double, Double, Double) = {
363     require(xmin <= xmax && ymin <= ymax, s"Bounds must be ordered: [$xmin $xmax] [$ymin $ymax]")
364 
365     try {
366       require(xmin >= xLo && xmax <= xHi && ymin >= yLo && ymax <= yHi,
367         s"Values out of bounds ([$xLo $xHi] [$yLo $yHi]): [$xmin $xmax] [$ymin $ymax]")
368 
369       val nxmin = (xmin - xLo) / xSize
370       val nymin = (ymin - yLo) / ySize
371       val nxmax = (xmax - xLo) / xSize
372       val nymax = (ymax - yLo) / ySize
373 
374       (nxmin, nymin, nxmax, nymax)
375     } catch {
376       case _: IllegalArgumentException if lenient =>
377 
378         val bxmin = if (xmin < xLo) { xLo } else if (xmin > xHi) { xHi } else { xmin }
379         val bymin = if (ymin < yLo) { yLo } else if (ymin > yHi) { yHi } else { ymin }
380         val bxmax = if (xmax < xLo) { xLo } else if (xmax > xHi) { xHi } else { xmax }
381         val bymax = if (ymax < yLo) { yLo } else if (ymax > yHi) { yHi } else { ymax }
382 
383         val nxmin = (bxmin - xLo) / xSize
384         val nymin = (bymin - yLo) / ySize
385         val nxmax = (bxmax - xLo) / xSize
386         val nymax = (bymax - yLo) / ySize
387 
388         (nxmin, nymin, nxmax, nymax)
389     }
390   }

422   /**
423     * An extended Z curve element. Bounds refer to the non-extended z element for simplicity of calculation.
424     *
425     * An extended Z element refers to a normal Z curve element that has it's upper bounds expanded by double it's
426     * width/height. By convention, an element is always square.
427     *
428     * @param xmin x lower bound in [0-1]
429     * @param ymin y lower bound in [0-1]
430     * @param xmax x upper bound in [0-1], must be >= xmin
431     * @param ymax y upper bound in [0-1], must be >= ymin
432     * @param length length of the non-extended side (note: by convention width should be equal to height)
433     */
434   private case class XElement(xmin: Double, ymin: Double, xmax: Double, ymax: Double, length: Double) {
435 
436     // extended x and y bounds
437     lazy private val xext = xmax + length
438     lazy private val yext = ymax + length
439 
440     def isContained(window: QueryWindow): Boolean =
441       window.xmin <= xmin && window.ymin <= ymin && window.xmax >= xext && window.ymax >= yext
442 
443     def overlaps(window: QueryWindow): Boolean =
444       window.xmax >= xmin && window.ymax >= ymin && window.xmin <= xext && window.ymin <= yext
445 
446     def children: Seq[XElement] = {
447       val xCenter = (xmin + xmax) / 2.0
448       val yCenter = (ymin + ymax) / 2.0
449       val len = length / 2.0
450       val c0 = copy(xmax = xCenter, ymax = yCenter, length = len)
451       val c1 = copy(xmin = xCenter, ymax = yCenter, length = len)
452       val c2 = copy(xmax = xCenter, ymin = yCenter, length = len)
453       val c3 = copy(xmin = xCenter, ymin = yCenter, length = len)
454       Seq(c0, c1, c2, c3)
455     }
456   }
