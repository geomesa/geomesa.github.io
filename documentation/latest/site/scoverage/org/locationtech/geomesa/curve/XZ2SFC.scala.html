44   /**
45     * Index a polygon by its bounding box
46     *
47     * @param xmin min x value in xBounds
48     * @param ymin min y value in yBounds
49     * @param xmax max x value in xBounds, must be >= xmin
50     * @param ymax max y value in yBounds, must be >= ymin
51     * @param lenient standardize boundaries to valid values, or raise an exception
52     * @return z value for the bounding box
53     */
54   def index(xmin: Double, ymin: Double, xmax: Double, ymax: Double, lenient: Boolean = false): Long = {
55     // normalize inputs to [0,1]
56     val (nxmin, nymin, nxmax, nymax) = normalize(xmin, ymin, xmax, ymax, lenient)
57 
58     // calculate the length of the sequence code (section 4.1 of XZ-Ordering paper)
59 
60     val maxDim = math.max(nxmax - nxmin, nymax - nymin)
61 
62     // l1 (el-one) is a bit confusing to read, but corresponds with the paper's definitions
63     val l1 = math.floor(math.log(maxDim) / XZSFC.LogPointFive).toInt
64 
65     // the length will either be (l1) or (l1 + 1)
66     val length = if (l1 >= g) { g } else {
67       val w2 = math.pow(0.5, l1 + 1) // width of an element at resolution l2 (l1 + 1)
68 
69       // predicate for checking how many axis the polygon intersects
70       // math.floor(min / w2) * w2 == start of cell containing min
71       def predicate(min: Double, max: Double): Boolean = max <= (math.floor(min / w2) * w2) + (2 * w2)
72 
73       if (predicate(nxmin, nxmax) && predicate(nymin, nymax)) l1 + 1 else l1
74     }
75 
76     sequenceCode(nxmin, nymin, length)
77   }

151   /**
152     * Determine XZ-curve ranges that will cover a given query window
153     *
154     * @param query a sequence of OR'd windows to cover, normalized to [0,1]
155     * @param rangeStop a rough max value for the number of ranges to return
156     * @return
157     */
158   private def ranges(query: Array[QueryWindow], rangeStop: Int): Seq[IndexRange] = {
159 
160     import XZ2SFC.{LevelOneElements, LevelTerminator}
161 
162     // stores our results - initial size of 100 in general saves us some re-allocation
163     val ranges = new java.util.ArrayList[IndexRange](100)
164 
165     // values remaining to process - initial size of 100 in general saves us some re-allocation
166     val remaining = new java.util.ArrayDeque[XElement](100)
167 
168     // checks if a quad is contained in the search space
169     def isContained(quad: XElement): Boolean = {
170       var i = 0
171       while (i < query.length) {
172         if (quad.isContained(query(i))) {
173           return true
174         }
175         i += 1
176       }
177       false
178     }
179 
180     // checks if a quad overlaps the search space
181     def isOverlapped(quad: XElement): Boolean = {
182       var i = 0
183       while (i < query.length) {
184         if (quad.overlaps(query(i))) {
185           return true
186         }
187         i += 1
188       }
189       false
190     }
191 
192     // checks a single value and either:
193     //   eliminates it as out of bounds
194     //   adds it to our results as fully matching, or
195     //   adds it to our results as partial matching and queues up it's children for further processing
196     def checkValue(quad: XElement, level: Short): Unit = {
197       if (isContained(quad)) {
198         // whole range matches, happy day
199         val (min, max) = sequenceInterval(quad.xmin, quad.ymin, level, partial = false)
200         ranges.add(IndexRange(min, max, contained = true))
201       } else if (isOverlapped(quad)) {
202         // some portion of this range is excluded
203         // add the partial match and queue up each sub-range for processing
204         val (min, max) = sequenceInterval(quad.xmin, quad.ymin, level, partial = true)
205         ranges.add(IndexRange(min, max, contained = false))
206         quad.children.foreach(remaining.add)
207       }
208     }
209 
210     // initial level
211     LevelOneElements.foreach(remaining.add)
212     remaining.add(LevelTerminator)
213 
214     // level of recursion
215     var level: Short = 1
216 
217     while (level < g && !remaining.isEmpty && ranges.size < rangeStop) {
218       val next = remaining.poll
219       if (next.eq(LevelTerminator)) {
220         // we've fully processed a level, increment our state
221         if (!remaining.isEmpty) {
222           level = (level + 1).toShort
223           remaining.add(LevelTerminator)
224         }
225       } else {
226         checkValue(next, level)
227       }
228     }
229 
230     // bottom out and get all the ranges that partially overlapped but we didn't fully process
231     while (!remaining.isEmpty) {
232       val quad = remaining.poll
233       if (quad.eq(LevelTerminator)) {
234         level = (level + 1).toShort
235       } else {
236         val (min, max) = sequenceInterval(quad.xmin, quad.ymin, level, partial = false)
237         ranges.add(IndexRange(min, max, contained = false))
238       }
239     }
240 
241     // we've got all our ranges - now reduce them down by merging overlapping values
242     // note: we don't bother reducing the ranges as in the XZ paper, as accumulo handles lots of ranges fairly well
243     ranges.sort(IndexRange.IndexRangeIsOrdered)
244 
245     var current = ranges.get(0) // note: should always be at least one range
246     val result = ArrayBuffer.empty[IndexRange]
247     var i = 1
248     while (i < ranges.size()) {
249       val range = ranges.get(i)
250       if (range.lower <= current.upper + 1) {
251         // merge the two ranges
252         current = IndexRange(current.lower, math.max(current.upper, range.upper), current.contained && range.contained)
253       } else {
254         // append the last range and set the current range for future merging
255         result.append(current)
256         current = range
257       }
258       i += 1
259     }
260     // append the last range - there will always be one left that wasn't added
261     result.append(current)
262 
263     result.toSeq
264   }

266   /**
267     * Computes the sequence code for a given point - for polygons this is the lower-left corner.
268     *
269     * Based on Definition 2 from the XZ-Ordering paper
270     *
271     * @param x normalized x value [0,1]
272     * @param y normalized y value [0,1]
273     * @param length length of the sequence code that will be generated
274     * @return
275     */
276   private def sequenceCode(x: Double, y: Double, length: Int): Long = {
277     var xmin = 0.0
278     var ymin = 0.0
279     var xmax = 1.0
280     var ymax = 1.0
281 
282     var cs = 0L
283 
284     var i = 0
285     while (i < length) {
286       val xCenter = (xmin + xmax) / 2.0
287       val yCenter = (ymin + ymax) / 2.0
288       (x < xCenter, y < yCenter) match {
289         case (true,  true)  => cs += 1L                                             ; xmax = xCenter; ymax = yCenter
290         case (false, true)  => cs += 1L + 1L * (math.pow(4, g - i).toLong - 1L) / 3L; xmin = xCenter; ymax = yCenter
291         case (true,  false) => cs += 1L + 2L * (math.pow(4, g - i).toLong - 1L) / 3L; xmax = xCenter; ymin = yCenter
292         case (false, false) => cs += 1L + 3L * (math.pow(4, g - i).toLong - 1L) / 3L; xmin = xCenter; ymin = yCenter
293       }
294       i += 1
295     }
296 
297     cs
298   }

300   /**
301     * Computes an interval of sequence codes for a given point - for polygons this is the lower-left corner.
302     *
303     * @param x normalized x value [0,1]
304     * @param y normalized y value [0,1]
305     * @param length length of the sequence code that will be used as the basis for this interval
306     * @param partial true if the element partially intersects the query window, false if it is fully contained
307     * @return
308     */
309   private def sequenceInterval(x: Double, y: Double, length: Short, partial: Boolean): (Long, Long) = {
310     val min = sequenceCode(x, y, length)
311     // if a partial match, we just use the single sequence code as an interval
312     // if a full match, we have to match all sequence codes starting with the single sequence code
313     val max = if (partial) { min } else {
314       // from lemma 3 in the XZ-Ordering paper
315       min + (math.pow(4, g - length + 1).toLong - 1L) / 3L
316     }
317     (min, max)
318   }

320   /**
321     * Normalize user space values to [0,1]
322     *
323     * @param xmin min x value in user space
324     * @param ymin min y value in user space
325     * @param xmax max x value in user space, must be >= xmin
326     * @param ymax max y value in user space, must be >= ymin
327     * @param lenient standardize boundaries to valid values, or raise an exception
328     * @return
329     */
330   private def normalize(xmin: Double,
331                         ymin: Double,
332                         xmax: Double,
333                         ymax: Double,
334                         lenient: Boolean): (Double, Double, Double, Double) = {
335     require(xmin <= xmax && ymin <= ymax, s"Bounds must be ordered: [$xmin $xmax] [$ymin $ymax]")
336 
337     try {
338       require(xmin >= xLo && xmax <= xHi && ymin >= yLo && ymax <= yHi,
339         s"Values out of bounds ([$xLo $xHi] [$yLo $yHi]): [$xmin $xmax] [$ymin $ymax]")
340 
341       val nxmin = (xmin - xLo) / xSize
342       val nymin = (ymin - yLo) / ySize
343       val nxmax = (xmax - xLo) / xSize
344       val nymax = (ymax - yLo) / ySize
345 
346       (nxmin, nymin, nxmax, nymax)
347     } catch {
348       case _: IllegalArgumentException if lenient =>
349 
350         val bxmin = if (xmin < xLo) { xLo } else if (xmin > xHi) { xHi } else { xmin }
351         val bymin = if (ymin < yLo) { yLo } else if (ymin > yHi) { yHi } else { ymin }
352         val bxmax = if (xmax < xLo) { xLo } else if (xmax > xHi) { xHi } else { xmax }
353         val bymax = if (ymax < yLo) { yLo } else if (ymax > yHi) { yHi } else { ymax }
354 
355         val nxmin = (bxmin - xLo) / xSize
356         val nymin = (bymin - yLo) / ySize
357         val nxmax = (bxmax - xLo) / xSize
358         val nymax = (bymax - yLo) / ySize
359 
360         (nxmin, nymin, nxmax, nymax)
361     }
362   }

394   /**
395     * An extended Z curve element. Bounds refer to the non-extended z element for simplicity of calculation.
396     *
397     * An extended Z element refers to a normal Z curve element that has it's upper bounds expanded by double it's
398     * width/height. By convention, an element is always square.
399     *
400     * @param xmin x lower bound in [0-1]
401     * @param ymin y lower bound in [0-1]
402     * @param xmax x upper bound in [0-1], must be >= xmin
403     * @param ymax y upper bound in [0-1], must be >= ymin
404     * @param length length of the non-extended side (note: by convention width should be equal to height)
405     */
406   private case class XElement(xmin: Double, ymin: Double, xmax: Double, ymax: Double, length: Double) {
407 
408     // extended x and y bounds
409     lazy private val xext = xmax + length
410     lazy private val yext = ymax + length
411 
412     def isContained(window: QueryWindow): Boolean =
413       window.xmin <= xmin && window.ymin <= ymin && window.xmax >= xext && window.ymax >= yext
414 
415     def overlaps(window: QueryWindow): Boolean =
416       window.xmax >= xmin && window.ymax >= ymin && window.xmin <= xext && window.ymin <= yext
417 
418     def children: Seq[XElement] = {
419       val xCenter = (xmin + xmax) / 2.0
420       val yCenter = (ymin + ymax) / 2.0
421       val len = length / 2.0
422       val c0 = copy(xmax = xCenter, ymax = yCenter, length = len)
423       val c1 = copy(xmin = xCenter, ymax = yCenter, length = len)
424       val c2 = copy(xmax = xCenter, ymin = yCenter, length = len)
425       val c3 = copy(xmin = xCenter, ymin = yCenter, length = len)
426       Seq(c0, c1, c2, c3)
427     }
428   }
