18   private val jarFileFilter = new FilenameFilter() {
19     override def accept(dir: File, name: String): Boolean =
20       name.endsWith(".jar") && !name.endsWith("-sources.jar") && !name.endsWith("-javadoc.jar")
21   }
22 
23   private val folderFileFilter = new FileFilter() {
24     override def accept(pathname: File): Boolean = pathname.isDirectory
25   }
26 
27   def findJars(jars: Seq[String], searchPath: Iterator[() => Seq[File]]): Seq[File] = {
28     val foundJars = ArrayBuffer.empty[File]
29     var remaining = jars
30     // search each path in order until we've found all our jars
31     while (remaining.nonEmpty && searchPath.hasNext) {
32       val files = searchPath.next()()
33       remaining = remaining.filter { jarPrefix =>
34         val matched = files.filter(_.getName.startsWith(jarPrefix))
35         foundJars ++= matched
36         matched.isEmpty
37       }
38     }
39 
40     if (remaining.nonEmpty) {
41       logger.warn(s"Could not find requested jars: ${remaining.mkString(", ")}")
42     }
43 
44     foundJars.distinct.toSeq
45   }

74   /**
75    * Recursively searches folders for jar files
76    *
77    * @param dir dir
78    * @return
79    */
80   def loadJarsFromFolder(dir: File): Seq[File] = {
81     val files = Option(dir.listFiles(jarFileFilter)).toSeq.flatten
82     val children = Option(dir.listFiles(folderFileFilter)).toSeq.flatten.flatMap(loadJarsFromFolder)
83     files ++ children
84   }

86   /**
87     * Finds URLs of files based on a system property
88     *
89     * @param prop prop
90     * @return
91     */
92   def getFilesFromSystemProperty(prop: String): Seq[File] = {
93     Option(System.getProperty(prop)) match {
94       case Some(path) => path.split(":").map(new File(_)).toSeq.flatMap(loadFiles)
95       case None =>
96         logger.debug(s"No files loaded onto classpath from system property: $prop")
97         Seq.empty
98     }
99   }
100 
101   /**
102     * Recursively searches file for all files. Accepts file or dir.
103     *
104     * @param file file
105     * @return
106     */
107   private def loadFiles(file: File): Seq[File] = {
108     if (file.isDirectory) {
109       loadJarsFromFolder(file)
110     } else {
111       Option(file).toSeq
112     }
113   }

115   /**
116     * <p>Load files (jars, resources, configuration, etc) from a classpath defined by an environmental
117     * variable following these rules:
118     * <ul>
119     *   <li>Entries are colon (:) separated</li>
120     *   <li>If the entry ends with "&#47;*", treat it as a directory, and list jars in that
121     *   directory...no recursion</li>
122     *   <li>If the entry is a file then add it</li>
123     *   <li>If the entry is a directory list all files (jars and files) in the directory</li>
124     * </ul>
125     * </p>
126     *
127     * @param prop - environmental variable
128     * @return a list of files found in the classpath
129     */
130   def loadClassPathFromEnv(prop: String): Seq[File] = {
131     val files = sys.env.get(prop).toSeq.flatMap(_.split(':').toSeq).flatMap { entry =>
132       if (entry.endsWith("/*")) {
133         new File(entry.dropRight(2)).listFiles(jarFileFilter)
134       } else {
135         val f = new File(entry)
136         if (f.isDirectory) {
137           Option(f.listFiles).toSeq.flatten
138         } else {
139           Seq(f)
140         }
141       }
142     }
143     logger.debug(s"Loaded env classpath '$prop': ${files.map(_.getAbsolutePath).mkString(":")}")
144     files
145   }
