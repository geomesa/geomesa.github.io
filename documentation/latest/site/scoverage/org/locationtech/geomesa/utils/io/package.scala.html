58   /**
59    * Similar to java's try-with-resources, allows for using an object then closing in a finally block
60    */
61   object WithClose {
62 
63     def apply[C : IsCloseable, T](c: C)(fn: C => T): T = {
64       val ev = implicitly[IsCloseable[C]]
65       try { fn(c) } finally { if (c != null) { ev.close(c) }}
66     }
67 
68     def apply[C1 : IsCloseable, C2 : IsCloseable, T](c1: C1, c2: => C2)(fn: (C1, C2) => T): T =
69       apply(c1) { c1 => apply(c2) { c2 => fn(c1, c2) } }
70   }

72   /**
73    * Creates unique file names generated from a base name, by appending a sequence number
74    * before the file extension.
75    *
76    * For example, given the file name 'foo.txt', will return 'foo_000.txt', 'foo_001.txt', etc.
77    * If the iterator exceeds the specified number of digits, it will start to append additional
78    * digits to ensure uniqueness, e.g. 'foo_999.txt', 'foo_1000.txt', 'foo_1001.txt', etc.
79    *
80    * @param path file name path
81    * @param digits number of digits used to format the sequence number
82    */
83   class IncrementingFileName(path: String, digits: Int = 3) extends Iterator[String] {
84 
85     private val i = new AtomicInteger(0)
86     private val format = s"_%0${digits}d"
87     private val (prefix, suffix) = PathUtils.getBaseNameAndExtension(path)
88 
89     override def hasNext: Boolean = true
90 
91     override def next(): String = s"$prefix${format.format(i.getAndIncrement())}$suffix"
92   }
