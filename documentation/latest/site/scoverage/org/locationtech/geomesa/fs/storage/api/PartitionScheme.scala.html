20 /**
21   * Scheme for partitioning features into various named partitions (e.g. file paths) on disk, for
22   * faster querying. Partition schemes do not have any persistent state, they only know how to map features
23   * and filters to partition names
24   */
25 trait PartitionScheme {
26 
27   /**
28    * Name of this partition scheme
29    *
30    * @return
31    */
32   def name: String
33 
34   /**
35     * Return the partition in which a SimpleFeature should be stored
36     *
37     * @param feature simple feature
38     * @return partition name
39     */
40   def getPartition(feature: SimpleFeature): PartitionKey
41 
42   /**
43    * Get partitions that intersect the given filter
44    *
45    * If the filter does not constrain partitions at all, then an empty option will be returned. If
46    * the filter excludes all potential partitions, then an empty list will be returned
47    *
48    * @param filter filter
49    * @return list of intersecting filters
50    */
51   def getRangesForFilter(filter: Filter): Option[Seq[PartitionRange]]
52 
53   /**
54    * Enumerate all the partitions that intersect with the given filter
55    *
56    * If the filter does not constrain partitions at all, then an empty option will be returned. If
57    * the filter excludes all potential partitions, then an empty list will be returned
58    *
59    * Note that this may return a large number of partitions if the filter is not very selective
60    *
61    * @param filter filter
62    * @return
63    */
64   def getPartitionsForFilter(filter: Filter): Option[Seq[PartitionKey]]
65 
66   /**
67    * Get a filter that will cover a partitions, i.e. the filter will return all features
68    * in the given partition and none from other partitions
69    *
70    * @param partition partition to cover
71    * @return filter
72    */
73   def getCoveringFilter(partition: PartitionKey): Filter
74 }

76 object PartitionScheme extends LazyLogging {

78   /**
79    * Ranged bounds
80    *
81    * @param name partition scheme name
82    * @param lower lower bound, inclusive
83    * @param upper upper bound, exclusive
84    */
85   case class PartitionRange(name: String, lower: String, upper: String) {
86 
87     /**
88      * Is the value contained in this bounds
89      *
90      * @param value partition value
91      * @return
92      */
93     def contains(value: String): Boolean = value >= lower && value < upper
94 
95     /**
96      * Attempt to merge two bounds. Only overlapping bounds will result in a successful merge. Trying to merge
97      * bounds from a different partition scheme is a logical error.
98      *
99      * @param other bounds to merge
100      * @return
101      */
102     def merge(other: PartitionRange): Option[PartitionRange] = {
103       if (lower <= other.lower) {
104         if (upper >= other.upper) {
105           Some(this)
106         } else if (upper >= other.lower) {
107           Some(PartitionRange(name, lower, other.upper))
108         } else {
109           None
110         }
111       } else if (lower > other.upper) {
112         None
113       } else if (upper >= other.upper) {
114         Some(PartitionRange(name, other.lower, upper))
115       } else {
116         Some(other)
117       }
118     }
119   }
120 
121   /**
122    * Class to merge overlapping ranges.
123    *
124    * Our bounds extraction does not produce any overlapping ranges, but once converted to partitions there
125    * may be some overlap.
126    */
127   class RangeBuilder {
128 
129     private val ranges = ArrayBuffer.empty[PartitionRange]
130 
131     def +=(range: PartitionRange): Unit = ranges += range
132 
133     def result(): Seq[PartitionRange] = {
134       val all = ranges.sorted(BoundsOrdering)
135       if (all.lengthCompare(1) <= 0) {
136         all.toSeq
137       } else {
138         // merge any overlapping ranges that resulted
139         val result = Seq.newBuilder[PartitionRange]
140         var current = all.head
141         all.tail.foreach { range =>
142           current.merge(range) match {
143             case None =>
144               result += current
145               current = range
146             case Some(merged) =>
147               current = merged
148           }
149         }
150         result += current
151         result.result()
152       }
153     }
154   }
155 
156   object RangeBuilder {
157     private val BoundsOrdering = Ordering.by[PartitionRange, String](_.lower)
158   }
