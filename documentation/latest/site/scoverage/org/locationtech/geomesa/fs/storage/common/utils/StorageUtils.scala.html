20   /**
21     * Get the path for a new data file
22     *
23     * @param ext file extension
24     * @param fileType file type
25     * @return
26     */
27   def newFilePath(typeName: String, fileType: FileType.FileType, ext: String): String = {
28     val filename =
29       s"${fileType}_${SafeNameRegex.replaceAllIn(typeName, "-").take(20)}_${UUID.randomUUID().toString.replaceAllLiterally("-", "")}.$ext"
30     // partitioning logic taken from Apache Iceberg: https://iceberg.apache.org/docs/nightly/aws/#object-store-file-layout
31     val hash = {
32       val bytes = filename.getBytes(StandardCharsets.UTF_8)
33       val hash = MurmurHash3.hash32x86(bytes, 0, bytes.length, 0)
34       // Integer#toBinaryString excludes leading zeros, which we want to preserve
35       Integer.toBinaryString(hash | Integer.MIN_VALUE)
36     }
37     s"${hash.substring(0, 4)}/${hash.substring(4, 8)}/${hash.substring(8, 12)}/${hash.substring(12, 20)}/$filename"
38   }
39

46   }
