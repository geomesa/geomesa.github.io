26 /**
27  * Object store for local files
28  */
29 trait LocalObjectStore extends ObjectStore {
30 
31   private lazy val archiveFactory = new ArchiveStreamFactory()
32 
33   override val scheme: String = "file"
34 
35   override def exists(path: URI): Boolean = toFile(path).exists()
36 
37   override def size(path: URI): Long = {
38     val file = toFile(path)
39     if (file.exists()) { file.length() } else { 0L }
40   }
41 
42   override def modified(path: URI): Option[Long] = Option(toFile(path)).collect { case f if f.exists() => f.lastModified() }
43 
44   override def create(path: URI): Option[OutputStream] =
45     Option(toFile(path)).collect { case f if ensureParents(f).createNewFile() => new FileOutputStream(f) }
46 
47   override def overwrite(path: URI): OutputStream = new FileOutputStream(ensureParents(toFile(path)))
48 
49   private def ensureParents(file: File): File = {
50     if (file.getParentFile != null) {
51       file.getParentFile.mkdirs()
52     }
53     file
54   }
55 
56   override def read(path: URI): Option[InputStream] =
57     Option(toFile(path)).collect { case f if f.exists() => PathUtils.handleCompression(new FileInputStream(f), path.toString) }
58 
59   override def read(path: URI, format: ArchiveFormat): CloseableIterator[NamedInputStream] = {
60     val iter = format match {
61       case ArchiveFormat.Tar =>
62         CloseableIterator(read(path).iterator).flatMap { is =>
63           val archive: ArchiveInputStream[_ <: ArchiveEntry] =
64             archiveFactory.createArchiveInputStream(ArchiveStreamFactory.TAR, is)
65           new ArchiveFileIterator(archive, path.toString)
66         }
67 
68       case ArchiveFormat.Zip =>
69         CloseableIterator.single(toFile(path)).filter(_.exists()).flatMap { file =>
70           new ZipFileIterator(new ZipFile(file), path.toString)
71         }
72 
73       case _ =>
74         throw new UnsupportedOperationException(s"An implementation is missing for format $format")
75     }
76     iter.map { case (name, is) => NamedInputStream(name, is) }
77   }
78 
79   override def list(path: URI): CloseableIterator[URI] = {
80     val file = toFile(path)
81     if (!file.isDirectory) {
82       return CloseableIterator.empty
83     }
84     CloseableIterator(file.listFiles().map(_.toURI).iterator)
85   }
86 
87   override def copy(from: URI, to: URI): Unit =
88     WithClose(new FileInputStream(toFile(from)), new FileOutputStream(toFile(to)))(IOUtils.copy)
89 
90   override def delete(path: URI): Unit = toFile(path).delete()
91 
92   override def close(): Unit = {}
93 
94   private def toFile(path: URI): File = {
95     if (path.getScheme == scheme) {
96       Path.of(path).toFile
97     } else {
98       throw new IllegalArgumentException(s"Invalid scheme: expected '$scheme://': $path")
99     }
100   }
101 }
