24 /**
25   * An Input format that creates splits based on FSDS Partitions. This is used for compaction, when we want a single
26   * split per partition. Otherwise, use ParquetSimpleFeatureInputFormat as that is more efficient
27   */
28 class ParquetPartitionInputFormat extends InputFormat[Void, SimpleFeature] {
29 
30   override def getSplits(context: JobContext): java.util.List[InputSplit] = {
31     val hadoopConf = context.getConfiguration
32     val conf = {
33       val builder = Map.newBuilder[String, String]
34       hadoopConf.forEach(e => builder += e.getKey -> hadoopConf.get(e.getKey)) // use .get to resolve envs
35       builder.result()
36     }
37 
38     val root = StorageConfiguration.getRootPath(hadoopConf)
39     val fileSize = StorageConfiguration.getTargetFileSize(hadoopConf)
40     val typeName = StorageConfiguration.getSftName(hadoopConf)
41 
42     val fsc = FileSystemContext.create(root, conf)
43     WithClose(StorageCatalog(fsc)) { catalog =>
44       WithClose(catalog.load(typeName)) { storage =>
45         val sizeCheck = fileSize.orElse(storage.sizer.targetSize).map(t => (f: DataFile) => storage.sizer.fileIsSized(f, t))
46         val splits = StorageConfiguration.getPartitions(hadoopConf).map { partition =>
47           var size = 0L
48           val files = storage.metadata.files().forPartition(partition).scan().filter { f =>
49             if (sizeCheck.exists(_.apply(f))) { false } else {
50               size += f.fileSizeInBytes()
51               true
52             }
53           }
54           new PartitionInputSplit(partition.toString, files.map(_.location()), size)
55         }
56         java.util.Arrays.asList(splits: _*)
57       }
58     }
59   }
60 
61   override def createRecordReader(split: InputSplit, context: TaskAttemptContext): RecordReader[Void, SimpleFeature] =
62     new PartitionRecordReader(split.asInstanceOf[PartitionInputSplit].getFiles)
63 }

67   /**
68     * InputSplit corresponding to a single FileSystemDataStore PartitionScheme partition
69     */
70   class PartitionInputSplit extends InputSplit with Writable {
71 
72     private var name: String = _
73     private var files: Seq[String] = _
74     private var length: java.lang.Long = _
75 
76     def this(name: String, files: Seq[String], length: Long) = {
77       this()
78       this.name = name
79       this.files = files
80       this.length = length
81     }
82 
83     /**
84       * @return the name of this partition
85       */
86     def getName: String = name
87 
88     def getFiles: Seq[String] = files
89 
90     override def getLength: Long = length
91 
92     // TODO attempt to optimize the locations where this should run in the case of HDFS
93     // With S3 this won't really matter
94     override def getLocations: Array[String] = Array.empty[String]
95 
96     override def write(out: DataOutput): Unit = {
97       out.writeUTF(name)
98       out.writeLong(length)
99       out.writeInt(files.length)
100       files.foreach(out.writeUTF)
101     }
102 
103     override def readFields(in: DataInput): Unit = {
104       this.name = in.readUTF()
105       this.length = in.readLong()
106       val count = in.readInt
107       this.files = Seq.fill(count)(in.readUTF())
108     }
109   }
110 
111   class PartitionRecordReader(files: Seq[String]) extends RecordReader[Void, SimpleFeature] {
112 
113     import scala.collection.JavaConverters._
114 
115     private var catalog: StorageCatalog = _
116     private var storage: FileSystemStorage = _
117     private var reader: CloseableIterator[SimpleFeature] = _
118 
119     private var curValue: SimpleFeature = _
120 
121     private def filterFiles(location: String): Boolean = files.contains(location)
122 
123     override def initialize(split: InputSplit, context: TaskAttemptContext): Unit = {
124       val hadoopConf = context.getConfiguration
125       val conf = {
126         val builder = Map.newBuilder[String, String]
127         hadoopConf.forEach(e => builder += e.getKey -> e.getValue)
128         builder.result()
129       }
130       val root = StorageConfiguration.getRootPath(hadoopConf)
131       val typeName = StorageConfiguration.getSftName(hadoopConf)
132       val fsc = FileSystemContext.create(root, conf)
133       catalog = StorageCatalog(fsc)
134       storage = catalog.load(typeName)
135 
136       val readSchema = storage.schema.read(None, Set.empty)
137       val ff = RecordSimpleFeature(readSchema)
138       val tableScan =
139         storage.table.newScan()
140           .select(readSchema.schema.columns().asScala.map(_.name()).asJava) // exclude z2 cols even if there's no transform
141 
142       reader = new IcebergParquetScan(tableScan, math.min(8, files.size), Some(filterFiles)).map(ff.apply)
143     }
144 
145     // TODO look at how the ParquetInputFormat provides progress and utilize something similar
146     override def getProgress: Float = 0.0f
147 
148     override def nextKeyValue(): Boolean = {
149       if (reader.hasNext) {
150         curValue = reader.next()
151         true
152       } else {
153         curValue = null
154         false
155       }
156     }
157 
158     override def getCurrentKey: Void = null
159     override def getCurrentValue: SimpleFeature = curValue
160 
161     override def close(): Unit = CloseWithLogging(Seq(reader, storage, catalog).filter(_ != null))
162   }
