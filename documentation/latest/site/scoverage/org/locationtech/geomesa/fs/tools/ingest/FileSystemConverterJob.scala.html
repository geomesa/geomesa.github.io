39 abstract class FileSystemConverterJob(
40     dsParams: Map[String, String],
41     sft: SimpleFeatureType,
42     converterConfig: Config,
43     paths: Seq[String],
44     libjarsFiles: Seq[String],
45     libjarsPaths: Iterator[() => Seq[File]],
46     reducers: Int,
47     root: URI,
48     schemes: Set[PartitionScheme],
49     tmpPath: Option[Path],
50     targetFileSize: Option[Long]
51   ) extends ConverterIngestJob(dsParams, sft, converterConfig, paths, libjarsFiles, libjarsPaths)
52     with StorageConfiguration with LazyLogging {
53 
54   override protected def reduceCounters(job: Job): Seq[(String, Long)] =
55     Seq((IngestCounters.Persisted, job.getCounters.findCounter(OutputCounters.Group, "reduced").getValue))
56 
57   override def configureJob(job: Job): Unit = {
58     super.configureJob(job)
59 
60     job.setMapperClass(classOf[FsIngestMapper])
61     job.setMapOutputKeyClass(classOf[Text])
62     job.setMapOutputValueClass(classOf[BytesWritable])
63     job.setOutputKeyClass(classOf[Void])
64     job.setOutputValueClass(classOf[SimpleFeature])
65     job.setReducerClass(classOf[DummyReducer])
66     job.setNumReduceTasks(reducers)
67     // Ensure that the reducers don't start too early
68     // (default is at 0.05 which takes all the map slots and isn't needed)
69     job.getConfiguration.set("mapreduce.job.reduce.slowstart.completedmaps", ".90")
70 
71     StorageConfiguration.setRootPath(job.getConfiguration, root)
72     StorageConfiguration.setPartitionScheme(job.getConfiguration, schemes)
73     StorageConfiguration.setFileType(job.getConfiguration, FileType.Written)
74     targetFileSize.foreach(StorageConfiguration.setTargetFileSize(job.getConfiguration, _))
75 
76     FileOutputFormat.setOutputPath(job, tmpPath.getOrElse(new Path(S3ObjectStore.s3aUri(root))))
77 
78     configureOutput(sft, job)
79   }
80 
81   override def await(reporter: StatusCallback): JobResult = {
82     super.await(reporter).merge {
83       tmpPath.map { tp =>
84         reporter.reset()
85         new DistributedCopy().copy(Seq(tp), new Path(S3ObjectStore.s3aUri(root)), reporter) match {
86           case JobSuccess(message, counts) =>
87             Command.user.info(message)
88             JobSuccess("", counts)
89 
90           case j => j
91         }
92       }
93     }
94   }
95 }

116 
117     type Context = Mapper[LongWritable, SimpleFeature, Text, BytesWritable]#Context

121 
122     var mapped: Counter = _

124     var failed: Counter = _
125 
126     override def setup(context: Context): Unit = {
127       val sft = StorageConfiguration.getSft(context.getConfiguration)
128       serializer = KryoFeatureSerializer(sft, SerializationOption.defaults)
129       scheme = StorageConfiguration.getPartitionScheme(context.getConfiguration, sft)
130 
131       mapped = context.getCounter(OutputCounters.Group, "mapped")
132       written = context.getCounter(OutputCounters.Group, OutputCounters.Written)
133       failed = context.getCounter(OutputCounters.Group, OutputCounters.Failed)
134     }
135 
136     override def map(key: LongWritable, sf: SimpleFeature, context: Context): Unit = {
137       // partitionKey is important because this needs to be correct for the parquet file
138       try {
139         mapped.increment(1)
140         val sfWithFid = GeoMesaFeatureWriter.featureWithFid(sf)
141         val partitionKey = new Text(Partition(scheme.map(_.getPartition(sfWithFid))).toString)
142         context.write(partitionKey, new BytesWritable(serializer.serialize(sfWithFid)))
143         written.increment(1)
144       } catch {
145         case e: Throwable =>
146           logger.error(s"Failed to write '${DataUtilities.encodeFeature(sf)}'", e)
147           failed.increment(1)
148       }
149     }
150   }
