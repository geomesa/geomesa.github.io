24 abstract class SpatialScheme(id: String, bits: Int, geom: String) extends PartitionScheme {
25 
26   require(bits % 2 == 0, "Resolution must be an even number")
27 
28   protected val format = s"%0${digits(bits)}d"
29 
30   override val name: String = s"$id:attribute=$geom:bits=$bits"
31 
32   protected def digits(bits: Int): Int
33 
34   protected def generateRanges(xy: Seq[(Double, Double, Double, Double)]): Seq[IndexRange]
35 
36   override def getRangesForFilter(filter: Filter): Option[Seq[PartitionRange]] = {
37     getGeoms(filter).map { ranges =>
38       val builder = new RangeBuilder()
39       ranges.foreach { range =>
40         val lower = format.format(range.lower)
41         val upper = format.format(range.upper + 1)
42         builder += PartitionRange(name, lower, upper)
43       }
44       builder.result()
45     }
46   }
47 
48   override def getPartitionsForFilter(filter: Filter): Option[Seq[PartitionKey]] = {
49     getGeoms(filter).orElse(Some(generateRanges(Seq((-180, -90, 180, 90))))).map { ranges =>
50       ranges.flatMap { range =>
51         val lower = range.lower
52         val steps = 1 + (range.upper - lower).toInt
53         Seq.tabulate(steps)(i => PartitionKey(name, format.format(lower + i)))
54       }
55     }
56   }
57 
58   private def getGeoms(filter: Filter): Option[Seq[IndexRange]] = {
59     val geometries = FilterHelper.extractGeometries(filter, geom, intersect = true)
60     if (geometries.isEmpty) {
61       None
62     } else if (geometries.disjoint) {
63       Some(Seq.empty)
64     } else {
65       Some(generateRanges(geometries.values.map(GeometryUtils.bounds)))
66     }
67   }
68 }

91       } else if (matcher.matches()) {

93       } else {

95       }
96     }
