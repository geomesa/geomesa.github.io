23 /**
24  * Abstraction around object storage for data files
25  */
26 trait ObjectStore extends Closeable {
27 
28   /**
29    * Default URI scheme
30    *
31    * @return
32    */
33   def scheme: String

35   /**
36    * Does an object exist at the given path
37    *
38    * @param path file path
39    * @return
40    */
41   def exists(path: URI): Boolean
42 
43   /**
44    * Get the size of the object stored at this path, in bytes. If the file does not exist, 0L will be returned
45    *
46    * @param path file path
47    * @return size in bytes
48    */
49   def size(path: URI): Long
50 
51   /**
52    * Get the last modified date of the object stored at this path, as millis since the unix epoch. If the file does not exist,
53    * an empty option will be returned

55    * @param path file path
56    * @return last modified date in millis since epoch
57    */
58   def modified(path: URI): Option[Long]

60   /**
61    * Write to the given path. If a file already exists at the path, an empty option will be returned. Note that this

64    * @param path file path
65    * @return output stream for writing to the file
66    */
67   def create(path: URI): Option[OutputStream]

69   /**

74    */
75   def overwrite(path: URI): OutputStream
76 
77   /**
78    * Reads the file at the given path. If the file does not exist, an empty option will be returned

80    * @param path file path

83   def read(path: URI): Option[InputStream]
84 
85   /**

88    * @param path file path

93   /**
94    * List any files directly under the given directory path

101   /**

107   def copy(from: URI, to: URI): Unit

109   /**

113    */
114   def delete(path: URI): Unit
115 
116   /**
117    * Utility method to return the "name" of a file
118    *
119    * @param path path
120    * @return
121    */
122   def filename(path: URI): String = filename(path.toString)
123 
124   @tailrec
125   private def filename(path: String): String = {
126     val i = path.lastIndexOf('/')
127     if (i == -1) {
128       path
129     } else if (i < path.length - 1) {
130       path.substring(i + 1)
131     } else {
132       filename(path.substring(0, path.length - 1))
133     }
134   }
135 }

139   def apply(context: FileSystemContext): ObjectStore = {
140     context.root.getScheme match {
141       case "file" => LocalObjectStore
142       case "s3" | "s3a" => S3ObjectStore(context.conf)
143       case scheme => throw new UnsupportedOperationException(s"No object store implemented for scheme: $scheme")
144     }
145   }
146 
147   object ArchiveFormat extends Enumeration {
148 
149     type ArchiveFormat = Value
150     val Tar, Zip = Value
151 
152     /**
153      * Gets the archive format of a file, if any.
154      *
155      * Currently, this is just based on the file name
156      *
157      * @param path file path
158      * @return
159      */
160     def apply(path: URI): Option[ArchiveFormat] = {
161       PathUtils.getUncompressedExtension(path.toString).toLowerCase(Locale.US) match {
162         case TAR => Some(ArchiveFormat.Tar)
163         case ZIP | JAR => Some(ArchiveFormat.Zip)
164         case _ => None
165       }
166     }
167   }
