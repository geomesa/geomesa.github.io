22 class ConverterStorageFactory extends FileSystemStorageFactory with LazyLogging {
23 
24   import scala.collection.JavaConverters._
25 
26   override val encoding: String = ConverterStorageFactory.Encoding
27 
28   override def apply(context: FileSystemContext, metadata: StorageMetadata): FileSystemStorage = {
29     val converter = {
30       val convertArg = Option(context.conf.get(ConverterConfigParam))
31           .orElse(Option(context.conf.get(ConverterNameParam)))
32           .getOrElse(throw new IllegalArgumentException(s"Must provide either converter config or name"))
33       val converterConfig = ConverterConfigResolver.getArg(ConfArgs(convertArg)) match {
34         case Left(e) => throw new IllegalArgumentException("Could not load Converter with provided parameters", e)
35         case Right(c) => c
36       }
37       SimpleFeatureConverter(metadata.sft, converterConfig)
38     }
39 
40     val pathFilteringOpts =
41       context.conf.getValByRegex(Pattern.quote(PathFilterOptsPrefix) + ".*").asScala.map {
42         case (k, v) => k.substring(PathFilterOptsPrefix.length) -> v
43       }
44 
45     val pathFiltering = Option(context.conf.get(PathFilterName)).flatMap { name =>
46       val factory = PathFilteringFactory.load(NamedOptions(name, pathFilteringOpts.toMap))
47       if (factory.isEmpty) {
48         throw new IllegalArgumentException(s"Failed to load ${classOf[PathFiltering].getName} for config '$name'")
49       }
50       factory
51     }
52 
53     val converterPath = Option(context.conf.get(ConverterPathParam)).map(new Path(context.root, _)).getOrElse {
54       throw new IllegalArgumentException("Must provide converter path")
55     }
56 
57     new ConverterStorage(context.copy(root = converterPath), metadata, converter, pathFiltering)
58   }
59 }
