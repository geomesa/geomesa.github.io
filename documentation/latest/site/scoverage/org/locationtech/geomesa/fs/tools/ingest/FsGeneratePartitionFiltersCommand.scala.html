19 class FsGeneratePartitionFiltersCommand extends FsDataStoreCommand {
20 
21   import org.locationtech.geomesa.filter.andFilters
22 
23   import scala.collection.JavaConverters._
24 
25   override val params = new FsGeneratePartitionFiltersParams()
26 
27   override val name: String = "generate-partition-filters"
28 
29   override def execute(): Unit = withDataStore { ds =>
30     if (params.cqlFilter == null && params.partitions.isEmpty) {
31       throw new ParameterException("At least one of --partitions or --cql must be specified")
32     }
33 
34     val metadata = ds.storage(params.featureName).metadata
35 
36     val fromFilter = Option(params.cqlFilter).toSeq.flatMap { f =>
37       val keys = metadata.schemes.map { s =>
38         s.getPartitionsForFilter(f).getOrElse {
39           throw new ParameterException(s"The filter ${ECQL.toCQL(f)} does not select any partitions from the partition scheme ${s.name}")
40         }
41       }
42       keys.foldLeft(Seq(Partition.None)) { case (partitions, keys) =>
43         for { partition <- partitions; key <- keys } yield {
44           Partition(partition.values + key)
45         }
46       }
47     }
48 
49     val partitions = if (params.partitions.isEmpty) { fromFilter} else { (params.partitions.asScala ++ fromFilter).distinct }
50 
51     Command.user.info(s"Generating filters for ${partitions.size} partitions")
52     if (!params.noHeader) {
53       Command.output.info("Partition\tFilter")
54     }
55 
56     partitions.toSeq.sortBy(_.toString).foreach { partition =>
57       val filters = partition.values.flatMap(v => metadata.schemes.find(_.name == v.name).map(_.getCoveringFilter(v)))
58       val filter = ECQL.toCQL(andFilters(filters.toSeq))
59       Command.output.info(s"$partition\t$filter")
60     }
61   }
62 }
