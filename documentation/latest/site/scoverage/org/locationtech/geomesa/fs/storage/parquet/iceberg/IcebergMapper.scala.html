59   /**
60    * Convert geomesa file metadata to iceberg file metadata
61    *
62    * @param table iceberg table
63    * @param file file
64    * @return
65    */
66   def toDataFile(table: Table, file: StorageFile): DataFile = {
67     val uri = storage.context.root.resolve(file.file).toString
68     logger.whenDebugEnabled {
69       val inputFile = ParquetFileSystemReader.inputFile(storage.fs, new URI(uri))
70       val footer = WithClose(inputFile.newStream())(ParquetFileReader.readFooter(inputFile, ParquetReadOptions.builder().build(), _))
71       logger.debug(s"Parquet schema for ${file.file}: ${ParquetMetadata.toJSON(footer)}")
72     }
73     val inputFile = table.io().newInputFile(uri)
74     val metrics = ParquetUtil.fileMetrics(inputFile, metricsConfigs.computeIfAbsent(table.name(), _ => MetricsConfig.forTable(table)), null)
75     // TODO withSort(f.sort)
76     DataFiles.builder(table.spec())
77       .withPath(inputFile.location())
78       .withFormat(FileFormat.PARQUET)
79       .withFileSizeInBytes(inputFile.getLength)
80       .withMetrics(metrics)
81       .withPartitionValues(partitionValues(file.partition).asJava)
82       .withRecordCount(file.count)
83       .build()
84   }
85 
86   /**
87    * Gets the iceberg partition values that correspond to a given partition
88    *
89    * @param partition partition
90    * @return
91    */
92   def partitionValues(partition: Partition): Seq[String] = {
93     mappers.map { m =>
94       val key = partition.values.find(_.name == m.scheme.name).getOrElse {
95         throw new IllegalArgumentException(
96           s"Could not find associated partition: ${m.scheme.name} out of ${partition.values.mkString(", ")}")
97       }
98       m.toIceberg(key.value)
99     }
100   }

128      * @param partitionValue geomesa partition value

133 
134   private object SchemeMapper {
135 
136     /**
137      * Maps a partition scheme to iceberg
138      *
139      * @param scheme the scheme to map
140      * @return a mapping, if the scheme is supported by iceberg
141      */
142     def apply(scheme: PartitionScheme, binding: Class[_]): Option[SchemeMapper] = scheme match {
143       case s: DateTimeScheme if s.step == 1 && s.unit == ChronoUnit.HOURS => Some(HourMapper(s))
144       case s: DateTimeScheme if s.step == 1 && s.unit == ChronoUnit.DAYS => Some(DayMapper(s))
145       case s: DateTimeScheme if s.step == 1 && s.unit == ChronoUnit.MONTHS => Some(MonthMapper(s))
146       case s: DateTimeScheme if s.step == 1 && s.unit == ChronoUnit.YEARS => Some(YearMapper(s))
147       case s: HashScheme[_] => Some(HashMapper(s))
148 
149       case s: AttributeScheme[_] if classOf[String].isAssignableFrom(binding) =>
150         s.bucketing match {
151           case None => Some(IdentityMapper(s, LexiTypeEncoders.stringEncoder()))
152           case Some(w: WidthBucketing) => Some(TruncateMapper(s, LexiTypeEncoders.stringEncoder(), w.max))
153         }
154 
155       case s: AttributeScheme[_] if classOf[Integer].isAssignableFrom(binding) =>
156         s.bucketing match {
157           case None => Some(IdentityMapper(s, LexiTypeEncoders.integerEncoder()))
158           case Some(i: IntegralBucketing[Int]) => Some(TruncateMapper(s, LexiTypeEncoders.integerEncoder(), i.divisor))
159         }
160 
161       case s: AttributeScheme[_] if classOf[java.lang.Long].isAssignableFrom(binding) =>
162         s.bucketing match {
163           case None => Some(IdentityMapper(s, LexiTypeEncoders.longEncoder()))
164           case Some(i: IntegralBucketing[Long]) => Some(TruncateMapper(s, LexiTypeEncoders.longEncoder(), i.divisor.toInt))
165         }
166 
167       case _ => None
168     }
169   }

204   }
