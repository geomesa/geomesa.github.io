23 /**
24  * Synthetic metadata that just lists files on the filesystem
25  *
26  * @param context file system
27  * @param sft simple feature type
28  * @param orderedSchemes partition schemes, ordered according to the file system directory layout
29  * @param leafStorage if true, the final partition scheme is a prefix of the file name, instead of a regular directory
30  */
31 class ConverterMetadata(
32     context: FileSystemContext,
33     val sft: SimpleFeatureType,
34     val orderedSchemes: Seq[PartitionScheme],
35     val leafStorage: Boolean
36   ) extends StorageMetadata with CachedMetadata with LazyLogging {
37 
38   override val `type`: String = ConverterMetadata.MetadataType
39 
40   override val schemes: Set[PartitionScheme] = orderedSchemes.toSet
41 
42   private val depth = orderedSchemes.map {
43     case s: HierarchicalDateTimeScheme => s.depth
44     case _ => 1
45   }.sum - (if (leafStorage) { 1 } else { 0 })
46 
47   private val fs = ObjectStore(context)
48 
49   filesCache.refresh(BoxedUnit.UNIT) // kick off the initial load asynchronously
50 
51   override def addFile(file: StorageFile): Unit = throw new UnsupportedOperationException()
52   override def addFiles(files: Seq[StorageFile]): Unit = throw new UnsupportedOperationException()
53   override def removeFile(file: StorageFile): Unit = throw new UnsupportedOperationException()
54   override def replaceFiles(existing: Seq[StorageFile], replacements: Seq[StorageFile]): Unit =
55     throw new UnsupportedOperationException()
56 
57   override def close(): Unit = try { super.close() } finally { fs.close() }
58 
59   override protected def buildFileList(): Seq[StorageFile] = {
60     logger.debug("Building file list")
61     val start = System.currentTimeMillis()
62     val result = Seq.newBuilder[StorageFile]
63     try {
64       val toCheck = new java.util.LinkedList[URI]()
65       fs.list(context.root).foreach(toCheck.add)
66       val maxDepth = depth + context.root.toString.count(_ == '/')
67       val rootPathLength = context.root.toString.length
68       while (!toCheck.isEmpty) {
69         val file = toCheck.remove()
70         logger.debug(s"Checking ${file.getPath}")
71         if (file.getPath.endsWith("/")) {
72           if (file.toString.count(_ == '/') <= maxDepth) {
73             fs.list(file).foreach(toCheck.add)
74           }
75         } else if (!fs.filename(file).startsWith(".")) { // ignore "hidden" files
76           val relativePath = file.toString.substring(rootPathLength)
77           var partitionParts = relativePath.split("/")
78           if (leafStorage) {
79             partitionParts(partitionParts.length - 1) = partitionParts.last.takeWhile(_ != '_')
80           } else {
81             partitionParts = partitionParts.dropRight(1)
82           }
83           var i = 0
84           val partitions = orderedSchemes.map {
85             case s: HierarchicalDateTimeScheme =>
86               val key = PartitionKey(s.name, partitionParts.slice(i, i + s.depth).mkString("/"))
87               i += s.depth
88               key
89             case s =>
90               val key = PartitionKey(s.name, partitionParts(i))
91               i += 1
92               key
93           }
94           result += StorageFile(relativePath, Partition(partitions.toSet), -1L)
95         }
96       }
97     } catch {
98       case NonFatal(e) => logger.error("Error building file list:", e); throw e
99     }
100     val list = result.result()
101     logger.debug(s"Found ${list.size} files in ${System.currentTimeMillis() - start}ms")
102     logger.whenTraceEnabled(list.foreach(f => logger.trace(f.toString)))
103     list
104   }
105 }
