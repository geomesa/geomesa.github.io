64         val path = new Path(file)
65         if (!path.getFileSystem(storage.context.conf).exists(path)) {

72         Command.user.info(s"Registered file ${new Path(storage.context.root, file.file)} containing ${file.count} known features")
73         val bounds =
74           file.spatialBounds.map(b => s"${metadata.sft.getDescriptor(b.attribute).getLocalName} [${b.xmin},${b.ymin},${b.xmax},${b.ymax}]") ++
75             file.attributeBounds.map { b =>
76               val descriptor = metadata.sft.getDescriptor(b.attribute)
77               val alias = AttributeIndexKey.alias(descriptor.getType.getBinding)
78               val lower = AttributeIndexKey.decode(alias, b.lower) match {
79                 case null => ""
80                 case d: Date => DateParsing.formatDate(d)
81                 case v => v.toString
82               }
83               val upper = AttributeIndexKey.decode(alias, b.upper) match {
84                 case null => ""
85                 case d: Date => DateParsing.formatDate(d)
86                 case v => v.toString
87               }
88               s"${descriptor.getLocalName} [$lower,$upper]"
89             }
90         Command.user.info(s"File bounds:\n  ${bounds.mkString("\n  ")}")
91       }
92 
93       try {
94         paths.foreach { path =>
95           outputResult(storage.register(path))
96           if (params.delete) {
97             path.getFileSystem(storage.context.conf).delete(path, false)
98           }
99         }
100       } catch {

178     class ConsistencyChecker(storage: FileSystemStorage, otherTypes: Seq[FileSystemStorage], threads: Int)
179         extends Runnable with Closeable with LazyLogging {
180 
181       private val pool = new CachedThreadPool(threads)
182       private val onDisk = Collections.newSetFromMap(new ConcurrentHashMap[Path, java.lang.Boolean]())
183 
184       override def run(): Unit = {
185         // list out the files currently in the root directory, results go into onDisk
186         listRoot()
187 
188         val inconsistencies = ArrayBuffer.empty[Inconsistency]
189         val checked = scala.collection.mutable.Set.empty[String]
190 
191         // compare the files known to the metadata to the ones on disk
192         storage.metadata.getFiles().foreach { file =>
193           if (onDisk.remove(new Path(storage.context.root, file.file))) {
194             // metadata and file are consistent
195             checked.add(file.file)
196           } else {
197             // file is missing from metadata vs file is in metadata more than once
198             val duplicate = !checked.add(file.file)
199             inconsistencies += Inconsistency(file, duplicate)
200           }
201         }
202 
203         if (onDisk.isEmpty && inconsistencies.isEmpty) {
204           Command.user.info("No inconsistencies detected")
205         } else {
206           if (inconsistencies.nonEmpty) {
207             lazy val strings = inconsistencies.map { i =>
208               s"${i.file.file} (${if (i.duplicate) { "duplicate" } else { "missing" }})"
209             }
210             val msg = if (inconsistencies.size == 1) { "1 metadata entry that does" } else { s"${inconsistencies.size} metadata entries that do"}
211             Command.user.warn(s"Found $msg not correspond to a data file:")
212             Command.output.info(strings.sorted.mkString("  ", "\n  ", ""))
213           }
214           if (!onDisk.isEmpty) {
215             // filter out files from other feature types in the same root
216             otherTypes.foreach { s =>
217               s.metadata.getFiles().foreach { f =>
218                 onDisk.remove(new Path(s.context.root, f.file))
219               }
220             }
221             if (!onDisk.isEmpty) {
222               val msg = if (onDisk.size() == 1) { "1 data file that does" } else { s"${onDisk.size} data files that do"}
223               Command.user.warn(s"Found $msg not have a metadata entry:")
224               if (otherTypes.nonEmpty) {
225                 Command.user.warn(s"Note: they may belong to one of the other registered feature types in this root")
226               }
227               Command.output.info(onDisk.asScala.map(_.toString).toSeq.sorted.mkString("  ", "\n  ", ""))
228             }
229           }
230         }
231       }
232 
233       override def close(): Unit = pool.shutdown()
234 
235       /**
236        * List the files in the root directory. Excludes known file-based metadata files. Results are
237        * added to onDisk
238        */
239       private def listRoot(): Unit = {
240         val iter = storage.context.fs.listStatusIterator(storage.context.root)
241         // use a phaser to track worker thread completion
242         val phaser = new Phaser(2) // 1 for this thread + 1 for the worker
243         pool.submit(new TopLevelListWorker(phaser, iter))
244         // wait for the worker threads to complete
245         phaser.awaitAdvanceInterruptibly(phaser.arrive())
246       }
247 
248       private class TopLevelListWorker(phaser: Phaser, list: RemoteIterator[FileStatus]) extends Runnable {
249         override def run(): Unit = {
250           try {
251             var i = phaser.getRegisteredParties + 1
252             while (list.hasNext && i < PhaserUtils.MaxParties) {
253               val status = list.next
254               val path = status.getPath
255               val name = path.getName
256               if (status.isDirectory) {
257                 if (name != FileBasedMetadataCatalog.MetadataDirectory) {
258                   i += 1
259                   // use a tiered phaser on each directory avoid the limit of 65535 registered parties
260                   pool.submit(new ListWorker(new Phaser(phaser, 1), name, storage.context.fs.listStatusIterator(path)))
261                 }
262               } else {
263                 onDisk.add(path)
264               }
265             }
266             if (list.hasNext) {
267               pool.submit(new TopLevelListWorker(new Phaser(phaser, 1), list))
268             }
269           } finally {
270             phaser.arriveAndDeregister()
271           }
272         }
273       }
274 
275       private class ListWorker(phaser: Phaser, partition: String, listDirectory: => RemoteIterator[FileStatus])
276           extends Runnable {
277         override def run(): Unit = {
278           try {
279             var i = phaser.getRegisteredParties + 1
280             val iter = listDirectory
281             while (iter.hasNext && i < PhaserUtils.MaxParties) {
282               val status = iter.next
283               val path = status.getPath
284               val name = path.getName
285               if (status.isDirectory) {
286                 val nextPartition = s"$partition/$name"
287                 i += 1
288                 // use a tiered phaser on each directory avoid the limit of 65535 registered parties
289                 pool.submit(new ListWorker(new Phaser(phaser, 1), nextPartition, storage.context.fs.listStatusIterator(path)))
290               } else {
291                 onDisk.add(path)
292               }
293             }
294             if (iter.hasNext) {
295               pool.submit(new ListWorker(new Phaser(phaser, 1), partition, iter))
296             }
297           } catch {
298             case _: FileNotFoundException => // the partition dir was deleted... just return
299             case NonFatal(e) => logger.error("Error scanning metadata directory:", e)
300           } finally {
301             phaser.arriveAndDeregister() // notify that this thread is done
302           }
303         }
304       }
305     }
