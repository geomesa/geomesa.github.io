57     override val name = "register"
58     override val params = new RegisterParams()
59 
60     override def execute(): Unit = withDataStore { ds =>
61       val storage = ds.storage(params.featureName)
62       val metadata = storage.metadata
63 
64       val paths = params.files.asScala.map { file =>
65         val path = {
66           val tmp = new URI(file)
67           if (tmp.getScheme == null || tmp.getScheme.isEmpty) {
68             new URI(storage.context.root.getScheme, tmp.getHost, tmp.getPath, tmp.getFragment)
69           } else if (tmp.getScheme == storage.context.root.getScheme) {

71           } else {
72             throw new IllegalArgumentException(

74           }
75         }
76         if (!storage.fs.exists(path)) {
77           throw new IllegalArgumentException(s"File $path does not exist")
78         }
79         path
80       }
81 
82       def outputResult(file: StorageFile): Unit = {
83         Command.user.info(s"Registered file ${storage.context.root.resolve(file.file)} containing ${file.count} known features")
84         val bounds =
85           file.spatialBounds.map(b => s"${metadata.sft.getDescriptor(b.attribute).getLocalName} [${b.xmin},${b.ymin},${b.xmax},${b.ymax}]") ++
86             file.attributeBounds.map { b =>
87               val descriptor = metadata.sft.getDescriptor(b.attribute)
88               val alias = AttributeIndexKey.alias(descriptor.getType.getBinding)
89               val lower = AttributeIndexKey.decode(alias, b.lower) match {
90                 case null => ""
91                 case d: Date => DateParsing.formatDate(d)
92                 case v => v.toString
93               }
94               val upper = AttributeIndexKey.decode(alias, b.upper) match {
95                 case null => ""
96                 case d: Date => DateParsing.formatDate(d)
97                 case v => v.toString
98               }
99               s"${descriptor.getLocalName} [$lower,$upper]"
100             }
101         Command.user.info(s"File bounds:\n  ${bounds.mkString("\n  ")}")
102       }
103 
104       try {
105         paths.foreach { path =>
106           outputResult(storage.register(path))
107           if (params.delete) {
108             storage.fs.delete(path)
109           }
110         }
111       } catch {
112         case NonFatal(e) => throw new RuntimeException("Error registering file:", e)
113       }
114     }

203     class ConsistencyChecker(storage: FileSystemStorage, otherTypes: Seq[FileSystemStorage], threads: Int)
204         extends Runnable with Closeable with LazyLogging {
205 
206       private val pool = new CachedThreadPool(threads)
207       private val onDisk = Collections.newSetFromMap(new ConcurrentHashMap[URI, java.lang.Boolean]())
208 
209       override def run(): Unit = {
210         // list out the files currently in the root directory, results go into onDisk
211         listRoot()
212 
213         val inconsistencies = ArrayBuffer.empty[Inconsistency]
214         val checked = scala.collection.mutable.Set.empty[String]
215 
216         // compare the files known to the metadata to the ones on disk
217         storage.metadata.getFiles().foreach { file =>
218           if (onDisk.remove(storage.context.root.resolve(file.file))) {
219             // metadata and file are consistent
220             checked.add(file.file)
221           } else {
222             // file is missing from metadata vs file is in metadata more than once
223             val duplicate = !checked.add(file.file)
224             inconsistencies += Inconsistency(file, duplicate)
225           }
226         }
227 
228         if (onDisk.isEmpty && inconsistencies.isEmpty) {
229           Command.user.info("No inconsistencies detected")
230         } else {
231           if (inconsistencies.nonEmpty) {
232             lazy val strings = inconsistencies.map { i =>
233               s"${i.file.file} (${if (i.duplicate) { "duplicate" } else { "missing" }})"
234             }
235             val msg = if (inconsistencies.size == 1) { "1 metadata entry that does" } else { s"${inconsistencies.size} metadata entries that do"}
236             Command.user.warn(s"Found $msg not correspond to a data file:")
237             Command.output.info(strings.sorted.mkString("  ", "\n  ", ""))
238           }
239           if (!onDisk.isEmpty) {
240             // filter out files from other feature types in the same root
241             otherTypes.foreach { s =>
242               s.metadata.getFiles().foreach { f =>
243                 onDisk.remove(s.context.root.resolve(f.file))
244               }
245             }
246             if (!onDisk.isEmpty) {
247               val msg = if (onDisk.size() == 1) { "1 data file that does" } else { s"${onDisk.size} data files that do"}
248               Command.user.warn(s"Found $msg not have a metadata entry:")
249               if (otherTypes.nonEmpty) {
250                 Command.user.warn(s"Note: they may belong to one of the other registered feature types in this root")
251               }
252               Command.output.info(onDisk.asScala.map(_.toString).toSeq.sorted.mkString("  ", "\n  ", ""))
253             }
254           }
255         }
256       }
257 
258       override def close(): Unit = pool.shutdown()
259 
260       /**
261        * List the files in the root directory. Excludes known file-based metadata files. Results are
262        * added to onDisk
263        */
264       private def listRoot(): Unit = {
265         val iter = storage.fs.list(storage.context.root)
266         // use a phaser to track worker thread completion
267         val phaser = new Phaser(2) // 1 for this thread + 1 for the worker
268         pool.submit(new TopLevelListWorker(phaser, iter))
269         // wait for the worker threads to complete
270         phaser.awaitAdvanceInterruptibly(phaser.arrive())
271       }
272 
273       private class TopLevelListWorker(phaser: Phaser, list: CloseableIterator[URI]) extends Runnable {
274         override def run(): Unit = {
275           try {
276             var i = phaser.getRegisteredParties + 1
277             while (list.hasNext && i < PhaserUtils.MaxParties) {
278               val path = list.next
279               if (path.toString.endsWith("/")) { // .isDirectory
280                 if (!path.toString.endsWith(s"/${FileBasedMetadataCatalog.MetadataDirectory}/")) {
281                   i += 1
282                   // use a tiered phaser on each directory avoid the limit of 65535 registered parties
283                   pool.submit(new ListWorker(new Phaser(phaser, 1), storage.fs.list(path)))
284                 }
285               } else if (!storage.fs.filename(path).startsWith(".")) { // ignore "hidden" files
286                 onDisk.add(path)
287               }
288             }
289             if (list.hasNext) {
290               pool.submit(new TopLevelListWorker(new Phaser(phaser, 1), list))
291             } else {
292               list.close()
293             }
294           } catch {
295             case NonFatal(e) => Try(list.close()); throw e
296           } finally {
297             phaser.arriveAndDeregister()
298           }
299         }
300       }
301 
302       private class ListWorker(phaser: Phaser, listDirectory: => Iterator[URI]) extends Runnable {
303         override def run(): Unit = {
304           try {
305             var i = phaser.getRegisteredParties + 1
306             val iter = listDirectory
307             while (iter.hasNext && i < PhaserUtils.MaxParties) {
308               val path = iter.next
309               if (path.toString.endsWith("/")) { // .isDirectory
310                 i += 1
311                 // use a tiered phaser on each directory avoid the limit of 65535 registered parties
312                 pool.submit(new ListWorker(new Phaser(phaser, 1), storage.fs.list(path)))
313               } else if (!storage.fs.filename(path).startsWith(".")) { // ignore "hidden" files
314                 onDisk.add(path)
315               }
316             }
317             if (iter.hasNext) {
318               pool.submit(new ListWorker(new Phaser(phaser, 1), iter))
319             }
320           } catch {
321             case _: FileNotFoundException => // the partition dir was deleted... just return
322             case NonFatal(e) => logger.error("Error scanning metadata directory:", e)
323           } finally {
324             phaser.arriveAndDeregister() // notify that this thread is done
325           }
326         }
327       }
328     }
