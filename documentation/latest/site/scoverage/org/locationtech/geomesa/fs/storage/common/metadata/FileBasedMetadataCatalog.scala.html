17 /**
18  * Catalog for file-based metadata
19  *
20  * @param context file system
21  */
22 class FileBasedMetadataCatalog(context: FileSystemContext) extends StorageMetadataCatalog {
23 
24   private val directory = new Path(context.root, FileBasedMetadataCatalog.MetadataDirectory)
25 
26   override def getTypeNames: Seq[String] = {
27     val typeNames = Seq.newBuilder[String]
28     if (context.fs.exists(directory)) {
29       val files = context.fs.listFiles(directory, false)
30       while (files.hasNext) {
31         val file = files.next()
32         val name = file.getPath.getName
33         if (file.isFile && !name.startsWith(".") && name.endsWith(".json")) {
34           typeNames += StringSerialization.decodeAlphaNumericSafeString(name.substring(0, name.length - 5))
35         }
36       }
37     }
38     typeNames.result()
39   }
40 
41   override def load(typeName: String): FileBasedMetadata = {
42     val file = new Path(directory, StringSerialization.alphaNumericSafeString(typeName) + ".json")
43     if (!context.fs.exists(file)) {
44       throw new IllegalArgumentException(s"Type '$typeName' does not exit")
45     }
46     val meta = WithClose(context.fs.open(file))(MetadataSerialization.deserialize)
47     new FileBasedMetadata(context.fs, directory, meta.copy(sft = namespaced(meta.sft, context.namespace)))
48   }
49 
50   override def create(sft: SimpleFeatureType, partitions: Seq[String], targetFileSize: Option[Long]): FileBasedMetadata = {
51     // load the partition scheme first in case it fails
52     partitions.foreach(PartitionSchemeFactory.load(sft, _))
53     val meta = Metadata(sft, partitions, targetFileSize)
54     val file = new Path(directory, StringSerialization.alphaNumericSafeString(sft.getTypeName) + ".json")
55     WithClose(context.fs.create(file, true)) { out =>
56       MetadataSerialization.serialize(out, meta)
57       out.hflush()
58       out.hsync()
59     }
60     new FileBasedMetadata(context.fs, directory, meta.copy(sft = namespaced(meta.sft, context.namespace)))
61   }
62 }
