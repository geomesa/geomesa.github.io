93       extends GroupConverter with ValueMaterializer[SimpleFeature] {
94 
95     private val idConverter = new StringConverter()
96 
97     private val visConverter = new StringConverter() {
98       override def materialize(): String = {
99         super.materialize() match {
100           case null => null
101           case s: String => s.intern()
102         }
103       }
104     }
105 
106     private val attributes = Array.ofDim[ValueMaterializer[_ <: AnyRef]](schema.sft.getAttributeCount)
107 
108     private val converters = {
109       val builder = Array.newBuilder[ValueMaterializer[_ <: AnyRef]]
110       builder += idConverter
111       if (schema.hasVisibilities) {
112         builder += visConverter
113       }
114       var i = 0
115       while (i < schema.sft.getAttributeCount) {
116         val descriptor = schema.sft.getDescriptor(i)
117         val materializer = attribute(ObjectType.selectType(descriptor))
118         builder += materializer
119         attributes(i) = materializer
120         // note: we read bounding boxes since they're present, but we don't do anything with the result
121         // note: zValues are excluded from our read schema, they are only used for partitioning
122         if (schema.bboxes.get(descriptor.getLocalName).isDefined) {
123           builder += new BoundingBoxConverter()
124         }
125         i += 1
126       }
127       builder.result()
128     }
129 
130     override def reset(): Unit = start()
131 
132     override def materialize(): SimpleFeature = {
133       val id = idConverter.materialize()
134       val vis = visConverter.materialize()
135       val values = Array.tabulate[AnyRef](schema.sft.getAttributeCount)(i => attributes(i).materialize())
136       val userData = if (vis == null) { null } else {
137         val map = new java.util.HashMap[AnyRef, AnyRef](1)
138         map.put("geomesa.feature.visibility", vis)
139         map
140       }
141       new ScalaSimpleFeature(schema.sft, id, values, userData)
142     }
143 
144     override def start(): Unit = converters.foreach(_.reset())
145 
146     override def getConverter(fieldIndex: Int): ValueMaterializer[_ <: AnyRef] = converters(fieldIndex)
147 
148     override def end(): Unit = {}
149 
150     def fieldCount: Int = converters.length
151 
152     private def attribute(bindings: Seq[ObjectType]): ValueMaterializer[_ <: AnyRef] = {
153       bindings.head match {
154         case ObjectType.GEOMETRY => geometry(bindings.last)
155         case ObjectType.DATE     => new DateMicrosConverter()
156         case ObjectType.STRING   => new StringConverter()
157         case ObjectType.INT      => new IntConverter()
158         case ObjectType.DOUBLE   => new DoubleConverter()
159         case ObjectType.LONG     => new LongConverter()
160         case ObjectType.FLOAT    => new FloatConverter()
161         case ObjectType.BOOLEAN  => new BooleanConverter()
162         case ObjectType.BYTES    => new BytesConverter()
163         case ObjectType.LIST     => new ListConverter(attribute(bindings.drop(1)))
164         case ObjectType.MAP      => new MapConverter(attribute(bindings.slice(1, 2)), attribute(bindings.slice(2, 3)))
165         case ObjectType.UUID     => new UuidConverter()
166         case _ => throw new IllegalArgumentException(s"Can't deserialize field of type ${bindings.head}")
167       }
168     }
169 
170     private def geometry(binding: ObjectType): ValueMaterializer[_ <: Geometry] = {
171       if (schema.encodings.geometry == GeometryEncoding.GeoParquetWkb) {
172         new WkbConverter()
173       } else if (schema.encodings.geometry == GeometryEncoding.GeoParquetNative) {
174         binding match {
175           case ObjectType.POINT           => new PointConverter()
176           case ObjectType.LINESTRING      => new GeoParquetNativeLineStringConverter()
177           case ObjectType.POLYGON         => new GeoParquetNativePolygonConverter()
178           case ObjectType.MULTIPOINT      => new GeoParquetNativeMultiPointConverter()
179           case ObjectType.MULTILINESTRING => new GeoParquetNativeMultiLineStringConverter()
180           case ObjectType.MULTIPOLYGON    => new GeoParquetNativeMultiPolygonConverter()
181           case _                          => new WkbConverter()
182         }
183       } else {
184         throw new UnsupportedOperationException(s"Can't read geometries encoded with '${schema.encodings.geometry}'")
185       }
186     }
