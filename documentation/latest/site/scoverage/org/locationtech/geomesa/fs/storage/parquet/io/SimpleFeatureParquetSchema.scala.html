108   /**
109     * Extract the simple feature type from a parquet read context. The read context
110     * contains both file metadata and the provided read conf
111     *
112     * @param context parquet read context
113     * @return
114     */
115   def read(context: InitContext): Option[SimpleFeatureParquetSchema] = {
116     // copy in the file level metadata
117     val kvMeta = context.getKeyValueMetadata.asScala.collect { case (k, v) if !v.isEmpty => k -> v.iterator.next }
118     // noinspection ScalaDeprecation
119     val gmMeta = Seq(SftNameKey, SftSpecKey, SftReadSpecKey, GeometryEncodingKey).flatMap { key =>
120       val value =
121         try { context.getParquetConfiguration.get(key) } catch {
122           // noinspection ScalaDeprecation - use the older method for spark compatibility
123           case _: NoSuchMethodError => context.getConfiguration.get(key)
124         }
125       Option(value).map(key -> _)
126     }
127     // note: context meta goes second so that transforms override the file-level keys
128     read(context.getFileSchema, kvMeta.toMap ++ gmMeta.toMap)
129   }

140   /**
141    * Get a schema for writing. Encoding can be configured through `geomesa.parquet.geometries` and `geomesa.fs.visibilities`
142    *
143    * @param conf write configuration, including the sft spec
144    * @return
145    */
146   def write(conf: ParquetConfiguration): Option[SimpleFeatureParquetSchema] = {
147     for {
148       name <- Option(conf.get(SftNameKey))
149       spec <- Option(conf.get(SftSpecKey))
150     } yield {
151       val sft = SimpleFeatureTypes.createType(name, spec)
152       val geometries = Option(conf.get(GeometryEncodingKey)).map(GeometryEncoding.apply).getOrElse(GeometryEncoding.GeoParquetWkb)
153       // only include bboxes if they help with push-down filters
154       val bboxes =
155         if (Option(conf.get(BBoxEncodingKey)).forall(_.toBoolean)) { BoundingBoxes(sft, geometries) } else { BoundingBoxes(Seq.empty) }
156       val zCols = if (Option(conf.get(ZValueColumKey)).forall(_.toBoolean)) { ZValues(sft) } else { ZValues(Seq.empty) }
157       val visibilities =
158         Option(sft.getUserData.get(VisibilityEncodingKey)).orElse(Option(conf.get(VisibilityEncodingKey))).forall(_.toString.toBoolean)
159       val metadata = Map(
160         SftNameKey -> name,
161         SftSpecKey -> spec,
162         GeometryEncodingKey -> geometries.toString,
163       ) ++ Option(conf.get(PartitionKey)).map(PartitionKey -> _)
164       val encodings = Encodings(geometries)
165       val messageType = schema(sft, encodings, bboxes, zCols, visibilities).toMessageType(sft)
166       SimpleFeatureParquetSchema(sft, encodings, visibilities, bboxes, zCols, metadata.asJava, messageType)
167     }
168   }

183   /**
184    * Get the schema for a file being read
185    *
186    * @param fileSchema message type
187    * @param metadata file metadata
188    * @return
189    */
190   private def read(fileSchema: MessageType, metadata: Map[String, String]): Option[SimpleFeatureParquetSchema] = {
191     for {
192       name <- metadata.get(SftNameKey)
193       spec <- metadata.get(SftSpecKey)
194     } yield {
195       val sft = SimpleFeatureTypes.createType(name, spec)
196       val readSft = metadata.get(SftReadSpecKey).map(SimpleFeatureTypes.createType(name, _))
197       val geometries = metadata.get(GeometryEncodingKey).map(GeometryEncoding.apply).getOrElse {
198         throw new UnsupportedOperationException("GeoMesaV0/GeoMesaV1 encoding is no longer supported")
199       }
200       val encodings = Encodings(geometries)
201       val bboxes = BoundingBoxes(fileSchema.getFields.asScala.map(_.getName).flatMap(BoundingBoxField.fromFieldName).toSeq)
202       val zCols = ZValues(fileSchema.getFields.asScala.map(_.getName).flatMap(ZValueField.fromFieldName).toSeq)
203       val visibilities = fileSchema.containsField(VisibilitiesField)
204       val schema = readSchema(fileSchema, sft, readSft, encodings, bboxes, zCols, visibilities)
205       SimpleFeatureParquetSchema(readSft.getOrElse(sft), encodings, visibilities, bboxes, zCols, metadata.asJava, schema)
206     }
207   }

231   /**
232    * Get the message type for a simple feature type. We need the full sft in order to ensure field ids are
233    * consistent, but the schema may be reduced (e.g. on read) based on the filter sft
234    *
235    * @param sft simple feature type
236    * @param encodings field type encoding
237    * @param bboxes include bounding boxes for each row
238    * @param visibilities include visibilities
239    * @return
240    */
241   private def schema(
242       sft: SimpleFeatureType,
243       encodings: Encodings,
244       bboxes: BoundingBoxes,
245       zValues: ZValues,
246       visibilities: Boolean): SchemaFields = {
247     // note: for iceberg compatibility, field ids need to start at one and increment (without gaps) across all top-level fields.
248     // All nested fields (structs, lists) get ids *after* all the top-level fields, once again incrementing without gaps
249     val fieldIds = new AtomicInteger(1)
250     val id =
251       Types.required(PrimitiveTypeName.BINARY)
252         .id(fieldIds.getAndIncrement())
253         .as(LogicalTypeAnnotation.stringType())
254         .named(FeatureIdField)
255     val vis = if (!visibilities) { None } else {
256       val f =
257         Types.optional(PrimitiveTypeName.BINARY)
258           .id(fieldIds.getAndIncrement())
259           .as(LogicalTypeAnnotation.stringType())
260           .named(VisibilitiesField)
261       Some(f)
262     }
263     val attributes = {
264       val builder = Map.newBuilder[String, Seq[Type]]
265       val nestedFieldIds = new AtomicInteger(fieldIds.get() + sft.getAttributeCount + bboxes.fields.size + zValues.fields.size)
266       sft.getAttributeDescriptors.asScala.foreach { d =>
267         val name = alphaNumericSafeString(d.getLocalName)
268         val types =
269           Seq(buildType(name, ObjectType.selectType(d), encodings, fieldIds, nestedFieldIds)) ++
270             bboxes.get(d.getLocalName).map(bbox => BoundingBoxField.schema(bbox, fieldIds, nestedFieldIds)) ++
271             zValues.get(d.getLocalName).map(zValue => ZValueField.schema(zValue, fieldIds))
272          builder += name -> types
273       }
274       builder.result()
275     }
276     SchemaFields(id, vis, attributes)
277   }

279   private def icebergSchema(schema: SimpleFeatureParquetSchema): Schema = {
280     val fields = new java.util.ArrayList[NestedField](schema.sft.getAttributeCount + 3)
281     val aliases = new java.util.HashMap[String, Integer](schema.sft.getAttributeCount + 3)
282 
283     // see note in `schema` (above) about field ids
284     val fieldIds = new AtomicInteger(1)
285 
286     aliases.put(FeatureIdField, fieldIds.get())
287     fields.add(NestedField.required(fieldIds.getAndIncrement(), FeatureIdField, StringType.get()))
288 
289     if (schema.hasVisibilities) {
290       aliases.put(VisibilitiesField, fieldIds.get())
291       fields.add(NestedField.optional(fieldIds.getAndIncrement(), VisibilitiesField, StringType.get()))
292     }
293 
294     val nestedFieldIds =
295       new AtomicInteger(fieldIds.get() + schema.sft.getAttributeCount + schema.bboxes.fields.size + schema.zValues.fields.size)
296     schema.sft.getAttributeDescriptors.asScala.foreach { d =>
297       val name = alphaNumericSafeString(d.getLocalName)
298       aliases.put(name, fieldIds.get())
299       fields.add(buildIcebergType(name, ObjectType.selectType(d), schema.encodings, fieldIds, nestedFieldIds))
300       schema.bboxes.get(d.getLocalName).foreach { bbox =>
301         aliases.put(bbox, fieldIds.get())
302         fields.add(BoundingBoxField.icebergSchema(bbox, fieldIds, nestedFieldIds))
303       }
304       schema.zValues.get(d.getLocalName).foreach { zValue =>
305         aliases.put(zValue, fieldIds.get())
306         fields.add(ZValueField.icebergSchema(zValue, fieldIds))
307       }
308     }
309     new Schema(fields, aliases, java.util.Set.of[Integer](1))
310   }

312   /**
313    * Builds the schema type for an attribute
314    *
315    * @param name field name
316    * @param bindings object type
317    * @param encodings schema encodings
318    * @param repetition repetition
319    * @return
320    */
321   private def buildType(
322       name: String,
323       bindings: Seq[ObjectType],
324       encodings: Encodings,
325       fieldIds: AtomicInteger,
326       nestedFieldIds: AtomicInteger,
327       repetition: Repetition = Repetition.OPTIONAL): Type = {
328     val builder = bindings.head match {
329       case ObjectType.INT     => Types.primitive(PrimitiveTypeName.INT32, repetition).id(fieldIds.getAndIncrement())
330       case ObjectType.DOUBLE  => Types.primitive(PrimitiveTypeName.DOUBLE, repetition).id(fieldIds.getAndIncrement())
331       case ObjectType.LONG    => Types.primitive(PrimitiveTypeName.INT64, repetition).id(fieldIds.getAndIncrement())
332       case ObjectType.FLOAT   => Types.primitive(PrimitiveTypeName.FLOAT, repetition).id(fieldIds.getAndIncrement())
333       case ObjectType.BOOLEAN => Types.primitive(PrimitiveTypeName.BOOLEAN, repetition).id(fieldIds.getAndIncrement())
334       case ObjectType.BYTES   => Types.primitive(PrimitiveTypeName.BINARY, repetition).id(fieldIds.getAndIncrement())
335 
336       case ObjectType.STRING =>
337         Types.primitive(PrimitiveTypeName.BINARY, repetition)
338           .id(fieldIds.getAndIncrement())
339           .as(LogicalTypeAnnotation.stringType())
340 
341       case ObjectType.DATE =>
342         Types.primitive(PrimitiveTypeName.INT64, repetition)
343           .id(fieldIds.getAndIncrement())
344           .as(LogicalTypeAnnotation.timestampType(true, TimeUnit.MICROS))
345 
346       case ObjectType.UUID =>
347         Types.primitive(PrimitiveTypeName.FIXED_LEN_BYTE_ARRAY, repetition)
348           .id(fieldIds.getAndIncrement())
349           .length(16)
350           .as(LogicalTypeAnnotation.uuidType())
351 
352       case ObjectType.LIST =>
353         Types.optionalList()
354           .id(fieldIds.getAndIncrement())
355           .element(buildType("element", bindings.drop(1), encodings, nestedFieldIds, null /* should not be used*/, Repetition.REQUIRED))
356 
357       case ObjectType.MAP =>
358         Types.optionalMap()
359           .id(fieldIds.getAndIncrement())
360           .key(buildType("key", bindings.slice(1, 2), encodings, nestedFieldIds, null /* should not be used*/, Repetition.REQUIRED))
361           .value(buildType("value", bindings.slice(2, 3), encodings, nestedFieldIds, null /* should not be used*/))
362 
363       case ObjectType.GEOMETRY =>
364         encodings.geometry.schema(bindings(1), fieldIds, nestedFieldIds)
365 
366       case binding =>
367         throw new UnsupportedOperationException(s"No mapping defined for type $binding with encoding $encodings")
368     }
369     builder.named(name)
370   }

372   /**
373    * Builds the schema type for an attribute
374    *
375    * @param name field name
376    * @param bindings object type
377    * @param encodings schema encodings
378    * @return
379    */
380   private def buildIcebergType(
381       name: String,
382       bindings: Seq[ObjectType],
383       encodings: Encodings,
384       fieldIds: AtomicInteger,
385       nestedFieldIds: AtomicInteger): NestedField = {
386     val builder = NestedField.optional(name).withId(fieldIds.getAndIncrement())
387     val typed: NestedField.Builder = bindings.head match {
388       case ObjectType.INT     => builder.ofType(IntegerType.get())
389       case ObjectType.DOUBLE  => builder.ofType(DoubleType.get())
390       case ObjectType.LONG    => builder.ofType(LongType.get())
391       case ObjectType.FLOAT   => builder.ofType(FloatType.get())
392       case ObjectType.BOOLEAN => builder.ofType(BooleanType.get())
393       case ObjectType.BYTES   => builder.ofType(BinaryType.get())
394       case ObjectType.STRING  => builder.ofType(StringType.get())
395       case ObjectType.DATE    => builder.ofType(TimestampType.withoutZone())
396       case ObjectType.UUID    => builder.ofType(UUIDType.get())
397 
398       case ObjectType.LIST =>
399         val subType = buildIcebergType("", bindings.drop(1), encodings, nestedFieldIds, null /* should not be used*/)
400         builder.ofType(ListType.ofRequired(subType.fieldId(), subType.`type`()))
401 
402       case ObjectType.MAP =>
403         val keyType = buildIcebergType("", bindings.slice(1, 2), encodings, nestedFieldIds, null /* should not be used*/)
404         val valueType = buildIcebergType("", bindings.slice(2, 3), encodings, nestedFieldIds, null /* should not be used*/)
405         builder.ofType(MapType.ofRequired(keyType.fieldId(), valueType.fieldId(), keyType.`type`(), valueType.`type`()))
406 
407       case ObjectType.GEOMETRY =>
408         // not yet supported in spark or trino
409         // builder.ofType(GeometryType.crs84())
410         builder.ofType(BinaryType.get())
411 
412       case binding =>
413         throw new UnsupportedOperationException(s"No mapping defined for type $binding with encoding $encodings")
414     }
415     typed.build()
416   }

418   private case class SchemaFields(fid: Type, vis: Option[Type], attributes: Map[String, Seq[Type]]) {
419     def toMessageType(sft: SimpleFeatureType, excludeZValues: Boolean = false): MessageType = {
420       // note: id field goes at the front of the record, then vis, then attributes and bounding boxes
421       val fields = Seq(fid) ++ vis ++ {
422         val all = sft.getAttributeDescriptors.asScala.flatMap { d =>
423           attributes(alphaNumericSafeString(d.getLocalName))
424         }
425         if (!excludeZValues) { all } else { all.filter(f => ZValueField.fromFieldName(f.getName).isEmpty) }
426       }
427       val name = alphaNumericSafeString(sft.getTypeName)
428       new MessageType(name, fields.asJava)
429     }
