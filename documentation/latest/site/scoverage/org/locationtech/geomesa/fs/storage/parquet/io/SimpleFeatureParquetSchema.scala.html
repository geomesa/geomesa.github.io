30 /**

32  *
33  * @param sft simple feature type
34  * @param encodings type encoding
35  * @param hasVisibilities whether the schema encodes visibilities, or not
36  * @param boundingBoxes names of any bounding box fields, along with the original field name
37  * @param metadata file metadata

101   /**
102     * Extract the simple feature type from a parquet read context. The read context
103     * contains both file metadata and the provided read conf
104     *
105     * @param context parquet read context
106     * @return
107     */
108   def read(context: InitContext): Option[SimpleFeatureParquetSchema] = {
109     val metadata = new java.util.HashMap[String, String]()
110     // copy in the file level metadata - note, do this before copying in the conf so that transform schemas are applied correctly
111     context.getKeyValueMetadata.asScala.foreach { case (k, v) => if (!v.isEmpty) { metadata.put(k, v.iterator.next) }}
112     // noinspection ScalaDeprecation
113     Seq(SftNameKey, SftSpecKey, GeometryEncodingKey, SchemaVersionKey).foreach { key =>
114       val value =
115         try { context.getParquetConfiguration.get(key) } catch {
116           // noinspection ScalaDeprecation - use the older method for spark compatibility
117           case _: NoSuchMethodError => context.getConfiguration.get(key)
118         }
119       if (value != null) {
120         metadata.put(key, value)
121       }
122     }
123     read(context.getFileSchema, metadata)
124   }

158   /**
159    * Get the schema for a file being read
160    *
161    * @param fileSchema message type
162    * @param metadata file metadata
163    * @return
164    */
165   private def read(fileSchema: MessageType, metadata: java.util.Map[String, String]): Option[SimpleFeatureParquetSchema] = {
166     for {
167       name <- Option(metadata.get(SftNameKey))
168       spec <- Option(metadata.get(SftSpecKey))
169     } yield {
170       val sft = SimpleFeatureTypes.createType(name, spec)
171       val geometries = Option(metadata.get(GeometryEncodingKey)).map(GeometryEncoding.apply).getOrElse {
172         // noinspection ScalaDeprecation
173         if (metadata.get(SchemaVersionKey) == "1") { GeometryEncoding.GeoMesaV1 } else { GeometryEncoding.GeoMesaV0 }
174       }
175       val encodings = Encodings(geometries)
176       val bboxes = fileSchema.getFields.asScala.map(_.getName).flatMap(BoundingBoxField.fromBoundingBox).toSeq
177       val visibilities = fileSchema.containsField(VisibilitiesField)
178       SimpleFeatureParquetSchema(sft, encodings, visibilities, bboxes, Collections.unmodifiableMap(metadata))
179     }
180   }

182   /**
183    * Gets a schema for writing
184    *
185    * @param sftName sft name config
186    * @param sftSpec sft spec config
187    * @param geoms geometry encoding config
188    * @param bboxOpt bounding box encoding config
189    * @param vis visibility config
190    * @return
191    */
192   private def write(sftName: String, sftSpec: String, geoms: String, bboxOpt: String, vis: String, partition: String): Option[SimpleFeatureParquetSchema] = {
193     for {
194       name <- Option(sftName)
195       spec <- Option(sftSpec)
196     } yield {
197       val sft = SimpleFeatureTypes.createType(name, spec)
198       val geometries = Option(geoms).map(GeometryEncoding.apply).getOrElse(GeometryEncoding.GeoParquetWkb)
199       // only include bboxes if they help with push-down filters
200       val bboxes = if (bboxOpt != null && !bboxOpt.toBoolean) { Seq.empty } else { BoundingBoxField(sft, geometries) }
201       val visibilities = Option(sft.getUserData.get(VisibilityEncodingKey)).orElse(Option(vis)).forall(_.toString.toBoolean)
202       val metadata = new java.util.HashMap[String, String]()
203       metadata.put(SftNameKey, name)
204       metadata.put(SftSpecKey, spec)
205       metadata.put(PartitionKey, partition)
206       metadata.put(GeometryEncodingKey, geometries.toString)
207       if (geometries == GeometryEncoding.GeoMesaV1) {
208         // noinspection ScalaDeprecation
209         metadata.put(SchemaVersionKey, "1") // include schema version for back-compatibility when possible
210       }
211       val encodings = Encodings(geometries)
212       SimpleFeatureParquetSchema(sft, encodings, visibilities, bboxes, Collections.unmodifiableMap(metadata))
213     }
214   }

216   /**
217    * Get the message type for a simple feature type
218    *
219    * @param sft simple feature type
220    * @param encodings field type encoding
221    * @param bboxes include bounding boxes for each row
222    * @param visibilities include visibilities
223    * @return
224    */
225   private def schema(sft: SimpleFeatureType, encodings: Encodings, bboxes: Seq[String], visibilities: Boolean): MessageType = {
226     val isV0 = encodings.geometry == GeometryEncoding.GeoMesaV0
227     val id = {
228       val primitive = PrimitiveTypeName.BINARY
229       val builder = if (isV0) { Types.primitive(primitive, Repetition.REPEATED) } else { Types.required(primitive) }
230       builder.as(LogicalTypeAnnotation.stringType()).named(FeatureIdField)
231     }
232     val vis = if (!visibilities) { None } else {
233       Some(Types.optional(PrimitiveTypeName.BINARY).as(LogicalTypeAnnotation.stringType()).named(VisibilitiesField))
234     }
235     val boundingBoxes = bboxes.map(BoundingBoxField.schema)
236     // note: id field goes at the end of the record, then vis, then any bounding boxes
237     val fields = (sft.getAttributeDescriptors.asScala.map(schema(_, encodings)) :+ id) ++ vis ++ boundingBoxes
238     val name = if (isV0) { sft.getTypeName } else { alphaNumericSafeString(sft.getTypeName) }
239     new MessageType(name, fields.asJava)
240   }

242   /**

244    *
245    * @param descriptor descriptor

255   /**
256    * Builds the schema type for an attribute
257    *
258    * @param name field name
259    * @param bindings object type
260    * @param encodings schema encodings
261    * @param repetition repetition
262    * @return
263    */
264   private def buildType(
265       name: String,
266       bindings: Seq[ObjectType],
267       encodings: Encodings,
268       repetition: Repetition = Repetition.OPTIONAL): Type = {
269     val builder = bindings.head match {
270       case ObjectType.STRING   => Types.primitive(PrimitiveTypeName.BINARY, repetition).as(LogicalTypeAnnotation.stringType())
271       case ObjectType.INT      => Types.primitive(PrimitiveTypeName.INT32, repetition)
272       case ObjectType.DOUBLE   => Types.primitive(PrimitiveTypeName.DOUBLE, repetition)
273       case ObjectType.LONG     => Types.primitive(PrimitiveTypeName.INT64, repetition)
274       case ObjectType.FLOAT    => Types.primitive(PrimitiveTypeName.FLOAT, repetition)
275       case ObjectType.BOOLEAN  => Types.primitive(PrimitiveTypeName.BOOLEAN, repetition)
276       case ObjectType.BYTES    => Types.primitive(PrimitiveTypeName.BINARY, repetition)
277 
278       case ObjectType.DATE =>
279         val unit = encodings.date match {
280           case DateEncoding.Micros => TimeUnit.MICROS
281           case DateEncoding.Millis => TimeUnit.MILLIS
282           case _ => throw new UnsupportedOperationException(encodings.date.toString)
283         }
284         Types.primitive(PrimitiveTypeName.INT64, repetition).as(LogicalTypeAnnotation.timestampType(true, unit))
285 
286       case ObjectType.UUID =>
287         encodings.uuid match {
288           case UuidEncoding.FixedLength =>
289             Types.primitive(PrimitiveTypeName.FIXED_LEN_BYTE_ARRAY, repetition).length(16).as(LogicalTypeAnnotation.uuidType())
290           case UuidEncoding.Bytes => Types.primitive(PrimitiveTypeName.BINARY, repetition)
291           case _ => throw new UnsupportedOperationException(encodings.uuid.toString)
292         }
293 
294       case ObjectType.LIST =>
295         encodings.list match {
296           case ListEncoding.ThreeLevel =>
297             Types.optionalList().requiredGroupElement().addField(buildType("element", bindings.drop(1), encodings, Repetition.REQUIRED))
298           case ListEncoding.TwoLevel => Types.optionalList().element(buildType("element", bindings.drop(1), encodings))
299           case _ => throw new UnsupportedOperationException(encodings.list.toString)
300         }
301 
302       case ObjectType.MAP =>
303         Types.optionalMap()
304           .key(buildType("key", bindings.slice(1, 2), encodings, Repetition.REQUIRED))
305           .value(buildType("value", bindings.slice(2, 3), encodings))
306 
307       case ObjectType.GEOMETRY =>
308         GeometrySchema(bindings(1), encodings.geometry)
309 
310       case binding =>
311         throw new UnsupportedOperationException(s"No mapping defined for type $binding")
312     }
313     builder.named(name)
314   }
