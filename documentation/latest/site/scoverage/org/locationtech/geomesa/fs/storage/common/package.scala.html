51     def setScheme(names: String): Unit = sft.getUserData.put(SchemeKey, names)
52     def removeScheme(): Option[Seq[String]] = {
53       remove(SchemeKey).map { scheme =>
54         // back compatible check for old json-serialized schemes
55         if (scheme.trim.startsWith("{")) {
56           try {
57             def result(name: String, options: Map[String, String]): Seq[String] = {
58               val opts = options.map { case (k, v) => s"$k=$v" }.mkString(":")
59               name.split(",").toSeq.map(n => s"$n:$opts")
60             }
61             val source = ConfigSource.fromConfig(ConfigFactory.parseString(scheme))
62             source.load[SchemeOptions] match {
63               case Right(o) => result(o.scheme, o.options)
64               case Left(_) =>
65                 val n = source.loadOrThrow[NamedOptions]
66                 result(n.name, n.options)
67             }
68           } catch {
69             case NonFatal(e) => throw new RuntimeException(s"Could not parse legacy scheme options: $scheme", e)
70           }
71         } else {
72           scheme.split(",").toSeq
73         }
74       }
75     }
76 
77     def setTargetFileSize(size: String): Unit = {
78       // validate input
79       Memory.bytes(size).failed.foreach(e => throw new IllegalArgumentException("Invalid file size", e))
80       sft.getUserData.put(FileSizeKey, size)
81     }
82     def removeTargetFileSize(): Option[Long] = {
83       remove(FileSizeKey).map { s =>
84         Memory.bytes(s) match {
85           case Success(b) => b
86           case Failure(e) => throw new IllegalArgumentException("Invalid file size", e)
87         }
88       }
89     }
90 
91     def setObservers(names: Seq[String]): Unit = sft.getUserData.put(ObserversKey, names.mkString(","))
92     def getObservers: Seq[String] = {
93       val obs = sft.getUserData.get(ObserversKey).asInstanceOf[String]
94       if (obs == null || obs.isEmpty) { Seq.empty } else { obs.split(",") }
95     }
96 
97     def spatialBounds(): Seq[Int] = sft.getAttributeDescriptors.asScala.toSeq.collect {
98       case g: GeometryDescriptor if g == sft.getGeometryDescriptor || g.fsBounds() => sft.indexOf(g.getLocalName)
99     }
100 
101     def nonSpatialBounds(): Seq[Int] = sft.getAttributeDescriptors.asScala.toSeq.collect {
102       case d if sft.getUserData.get(DefaultDtgField) == d.getLocalName || (d.fsBounds() && !d.isInstanceOf[GeometryDescriptor]) =>
103         sft.indexOf(d.getLocalName)
104     }
105 
106     private def remove(key: String): Option[String] = Option(sft.getUserData.remove(key).asInstanceOf[String])
107   }
