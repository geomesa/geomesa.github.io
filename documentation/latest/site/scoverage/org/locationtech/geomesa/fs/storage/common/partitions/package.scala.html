31   case class SchemeOpts(name: String, opts: Map[String, String], multiOpts: Map[String, Seq[String]]) {
32     def getSingle(k: String): Option[String] = {
33       if (multiOpts.contains(k)) {
34         throw new IllegalArgumentException(s"Expected a single value for '$k' but got: ${multiOpts(k).mkString(", ")}")
35       }
36       opts.get(k)
37     }
38 
39     def getMulti(k: String): Seq[String] = multiOpts.get(k).orElse(opts.get(k).map(v => Seq(v))).getOrElse(Seq.empty)
40   }
41 
42   object SchemeOpts {
43     def apply(scheme: String): SchemeOpts = {
44       val parts = scheme.split(":")
45       val name = parts.head.toLowerCase(Locale.US)
46       val opts = parts.drop(1).map(_.split("=", 2) match { case Array(k, v) => k.toLowerCase(Locale.US) -> v }).groupBy(_._1)
47       val singleOpts = opts.collect { case (k, Array(v)) => k -> v._2 }
48       val multiOpts = opts.collect { case (k, v) if v.lengthCompare(1) > 0 => k -> v.map(_._2).toSeq }
49       SchemeOpts(name, singleOpts, multiOpts)
50     }
51   }
