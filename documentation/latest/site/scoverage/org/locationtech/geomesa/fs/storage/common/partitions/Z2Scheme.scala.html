20 case class Z2Scheme(bits: Int, geom: String, geomIndex: Int) extends SpatialScheme(Z2Scheme.Name, bits, geom) {
21 
22   import org.locationtech.geomesa.filter.{andFilters, ff}
23   import org.locationtech.geomesa.utils.geotools.CRS_EPSG_4326
24 
25   private val xyBits = bits / 2
26   private val z2 = new Z2SFC(xyBits)
27   private val xRadius = (360d / math.pow(2, xyBits)) / 2
28   private val yRadius = (180d / math.pow(2, xyBits)) / 2
29 
30   override def getPartition(feature: SimpleFeature): PartitionKey = {
31     val pt = feature.getAttribute(geomIndex).asInstanceOf[Point]
32     PartitionKey(name, format.format(z2.index(pt.getX, pt.getY)))
33   }
34 
35   override def getCoveringFilter(partition: PartitionKey): Filter = {
36     val (x, y) = z2.invert(partition.value.toLong)
37     val (xmin, xmax) = (x - xRadius, x + xRadius)
38     val (ymin, ymax) = (y - yRadius, y + yRadius)
39     val bbox = ff.bbox(ff.property(geom), new ReferencedEnvelope(xmin, xmax, ymin, ymax, CRS_EPSG_4326))
40     // account for borders between z-cells (make upper bounds exclusive except on the upper-right edge)
41     val xExclusive = if (xmax == z2.lon.max) { None } else {
42       Some(ff.less(ff.function("getX", ff.property(geom)), ff.literal(xmax)))
43     }
44     val yExclusive = if (ymax == z2.lat.max) { None } else {
45       Some(ff.less(ff.function("getY", ff.property(geom)), ff.literal(ymax)))
46     }
47     andFilters(Seq(bbox) ++ xExclusive ++ yExclusive)
48   }
49 
50   override protected def digits(bits: Int): Int = math.ceil(bits * math.log10(2)).toInt
51 
52   override protected def generateRanges(xy: Seq[(Double, Double, Double, Double)]): Seq[IndexRange] = z2.ranges(xy)
53 }
