19 /**
20  * Synthetic metadata catalog that examines files on disk

22  * @param context file system
23  * @param config converter configuration options

25 class ConverterMetadataCatalog(context: FileSystemContext, config: Map[String, String])
26     extends StorageMetadataCatalog with LazyLogging {
27 
28   import ConverterMetadata._

32   private val sft = {
33     val sftArg =
34       config.get(SftConfigParam)
35         .orElse(config.get(SftNameParam))
36         .orElse(Option(context.conf.get(SftConfigParam)))
37         .orElse(Option(context.conf.get(SftNameParam)))
38         .getOrElse(throw new IllegalArgumentException(s"Must provide either simple feature type config or name"))
39     SftArgResolver.getArg(SftArgs(sftArg, null)) match {
40       case Left(e) => throw new IllegalArgumentException("Could not load SimpleFeatureType with provided parameters", e)
41       case Right(schema) => schema
42     }
43   }
44 
45   private val schemes = {
46     val partitionSchemeName =
47       config.get(PartitionSchemeParam)
48         .orElse(Option(context.conf.get(PartitionSchemeParam)))
49         .getOrElse(throw new IllegalArgumentException("Must provide partition scheme name"))
50     val partitionSchemeOpts = {
51       val opts =
52         config.collect { case (k, v) if k.startsWith(PartitionOptsPrefix) => k.substring(PartitionOptsPrefix.length) -> v } ++
53           context.conf.getPropsWithPrefix(PartitionOptsPrefix).asScala
54       opts.map { case (k, v) => s"$k=$v"}.mkString(":")
55     }
56     partitionSchemeName.split(",").map { n =>
57       val nameWithOpts = s"$n:$partitionSchemeOpts"
58       // back-compatible hierarchical date check
59       HierarchicalDateTimeScheme.load(sft, nameWithOpts).getOrElse(PartitionSchemeFactory.load(sft, nameWithOpts))
60     }
61   }
62 
63   private val converterPath =
64     config.get(ConverterPathParam)
65       .orElse(Option(context.conf.get(ConverterPathParam)))
66       .map(new Path(context.root, _))
67       .getOrElse(throw new IllegalArgumentException("Must provide converter path"))
68 
69   private val leafStorage =
70     config.get(LeafStorageParam)
71       .orElse(Option(context.conf.get(LeafStorageParam)))
72       .map(_.toBoolean)
73       .getOrElse {
74         val deprecated = s"${PartitionOptsPrefix}leaf-storage"
75         config.get(deprecated).orElse(Option(context.conf.get(deprecated))).fold(true) { s =>
76           logger.warn(s"Using deprecated leaf-storage partition-scheme option. Please define leaf-storage using '$LeafStorageParam'")
77           s.toBoolean
78         }
79       }
80 
81   override def getTypeNames: Seq[String] =
82     if (PathCache.exists(context.fs, converterPath)) { Seq(sft.getTypeName) } else { Seq.empty }
83 
84   override def load(typeName: String): StorageMetadata = {
85     if (typeName == sft.getTypeName) {
86       new ConverterMetadata(context.copy(root = converterPath), sft, schemes, leafStorage)
87     } else {
88       throw new IllegalArgumentException(s"Schema '$typeName' doesn't exist - available schemas: ${sft.getTypeName}")
89     }
90   }
91 
92   override def create(sft: SimpleFeatureType, partitions: Seq[String], targetFileSize: Option[Long]): StorageMetadata =
93     throw new UnsupportedOperationException("Converter storage is read only")
