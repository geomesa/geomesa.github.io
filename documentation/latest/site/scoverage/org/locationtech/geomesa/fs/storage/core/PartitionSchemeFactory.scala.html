15 /**
16   * Factory for loading partition schemes
17   */
18 trait PartitionSchemeFactory {
19 
20   /**
21     * Load a partition scheme
22     *
23     * @param sft simple feature type
24     * @param scheme scheme options
25     * @return partition scheme
26     */
27   def load(sft: SimpleFeatureType, scheme: String): Option[PartitionScheme]
28 }
29 
30 object PartitionSchemeFactory {
31 
32   lazy private val factories =
33     Seq(AttributeScheme, DateTimeScheme, FlatScheme, HashScheme, XZ2Scheme, Z2Scheme) ++
34       ServiceLoader.load[PartitionSchemeFactory]()
35 
36 
37   /**
38     * Create a partition scheme instance via SPI lookup
39     *
40     * @param sft simple feature type
41     * @param scheme scheme options
42     * @return
43     */
44   def load(sft: SimpleFeatureType, scheme: String): PartitionScheme = {
45     factories.toStream.flatMap(_.load(sft, scheme)).headOption.getOrElse {
46       throw new IllegalArgumentException(s"No partition scheme factory implementation exists for name " +
47         s"'$scheme'. Available factories: ${factories.map(_.getClass.getName).mkString(", ")}")
48     }
49   }
50 }
