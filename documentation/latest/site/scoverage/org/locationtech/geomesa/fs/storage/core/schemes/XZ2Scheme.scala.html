24 /**
25  * XZ2 spatial scheme.
26  *
27  * This scheme uses a fixed high-resolution curve for indexing, then truncates the value to create partition groups,
28  * in order to align with the iceberg 'truncate' transforms.
29  *
30  * Index values are (max) 7-digit hex-encoded longs, as we're only using 25 bits for the high-resolution curve,
31  * we can fit it in 7 digits.
32  *
33  * @param attribute name of the attribute being partitioned
34  * @param index index in the sft of the attribute being partitioned
35  * @param bits number of bits of resolution used for partitioning
36  */
37 case class XZ2Scheme(attribute: String, index: Int, bits: Int) extends PartitionScheme {
38 
39   import FilterHelper.ff
40 
41   require(bits % 4 == 0, s"Bit precision must be a multiple of 4, but received $bits")
42 
43   private val xz2 = XZ2SFC
44 
45   // partition level derived from bits parameter
46   // each level adds 2 bits (4 quadrants)
47   private val partitionLevel = (bits / 2).toShort
48   // number of hex digits used to represent our z value - bits = (xz2.g - partitionLevel) * 2, then divide by 4 to get hex
49   private val digits = xz2.hexDigits - ((xz2.g - partitionLevel) / 2)
50 
51   lazy private val wholeWorldRanges = Some(generateRanges(Seq((-180, -90, 180, 90))))
52 
53   override val name: String = s"${XZ2Scheme.name}:attribute=$attribute:bits=$bits"
54 
55   override def getPartition(feature: SimpleFeature): PartitionKey = {
56     val envelope = feature.getAttribute(index).asInstanceOf[Geometry].getEnvelopeInternal
57     val zValue = xz2.index(envelope.getMinX, envelope.getMinY, envelope.getMaxX, envelope.getMaxY)
58     PartitionKey(name, truncateToPartition(zValue))
59   }
60 
61   override def getCoveringFilter(partition: PartitionKey): Filter = {
62     // TODO maybe we can improve this with *some* kind of bbox?
63     val zPrefix = partition.value
64     val lower = zPrefix.padTo(xz2.hexDigits, '0')
65     val upper = zPrefix.padTo(xz2.hexDigits, 'f')
66     ff.between(ff.function(XZ2Function.FunctionName.getName), ff.literal(lower), ff.literal(upper))
67   }
68 
69   override def getRangesForFilter(filter: Filter): Option[Seq[PartitionRange]] = {
70     val geometries = FilterHelper.extractGeometries(filter, attribute, intersect = true)
71     if (geometries.isEmpty) {
72       None
73     } else if (geometries.disjoint) {
74       Some(Seq.empty)
75     } else {
76       Some(generateRanges(geometries.values.map(GeometryUtils.bounds)))
77     }
78   }
79 
80   override def getPartitionsForFilter(filter: Filter): Option[Seq[PartitionKey]] = {
81     getRangesForFilter(filter).orElse(wholeWorldRanges).map { ranges =>
82       ranges.flatMap { range =>
83         Iterator.iterate(range.lower)(incrementHex).takeWhile(_ < range.upper).map(PartitionKey(name, _))
84       }
85     }
86   }
87 
88   private def generateRanges(xy: Seq[(Double, Double, Double, Double)]): Seq[PartitionRange] = {
89     val builder = new RangeBuilder()
90     xz2.ranges(xy).foreach { range =>
91       val lower = truncateToPartition(range.lower)
92       // index ranges are inclusive, but partition ranges are exclusive
93       val upper = incrementHex(truncateToPartition(range.upper))
94       builder += PartitionRange(name, lower, upper)
95     }
96     builder.result()
97   }
98 
99   // truncates a full-resolution index to a partition group ID
100   private def truncateToPartition(fullIndex: Long): String = xz2.hexEncode(fullIndex).take(digits)
101 }
102 
103 object XZ2Scheme extends SpatialPartitionSchemeFactory[Geometry]("xz2") {
104 
105   override def buildPartitionScheme(bits: Int, geom: String, geomIndex: Int): PartitionScheme =
106     XZ2Scheme(geom, geomIndex, bits)
107 
108   private def incrementHex(hex: String): String = incrementHex(hex, hex.length - 1)
109 
110   @tailrec
111   private def incrementHex(hex: String, pos: Int): String = {
112     val c = hex.charAt(pos)
113     if (c != 'f') {
114       hex.substring(0, pos) + (c + 1).toChar + hex.substring(pos + 1)
115     } else if (pos == 0) {
116       hex + '0' // note: this isn't actually incrementing the value but should sort after all the valid hex values
117     } else {
118       incrementHex(hex.substring(0, pos) + '0' + hex.substring(pos + 1), pos - 1)
119     }
120   }
