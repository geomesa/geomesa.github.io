145     private def closeChunk(done: Boolean = false): Unit = {
146       currentChunk.os.close()
147       if (done && uploadId == null) {
148         // didn't hit our multipart threshold, just use a regular put object
149         logger.debug(s"Starting regular s3 upload to $bucket/$key")
150         val request = PutObjectRequest.builder().bucket(bucket).key(key)
151         if (!overwrite) {
152           request.ifNoneMatch("*")
153         }
154         client.putObject(request.build(), currentChunk.read).join()
155         logger.debug(s"Completed regular s3 upload to $bucket/$key")
156       } else {
157         if (uploadId == null) {
158           logger.debug(s"Starting multipart upload to $bucket/$key")
159           uploadId = client.createMultipartUpload(b => b.bucket(bucket).key(key)).join().uploadId()
160           logger.debug(s"Initiated multipart upload with id $uploadId to $bucket/$key")
161         }
162         val chunk = currentChunk
163         val future = retry(chunk, uploads.size + 1, 2)
164         uploads += future

169           val parts = uploads.map { future =>

172             CompletedPart.builder().partNumber(partNumber).eTag(response.eTag()).build()
173           }
174           val request =

179           }
180           client.completeMultipartUpload(request.build()).join()
181           logger.debug(s"Completed multipart upload with id $uploadId consisting of ${uploads.size} parts to $bucket/$key")
182         } else {

186       }

189     private def retry(currentChunk: Chunk, partNumber: Int, retries: Int): CompletableFuture[UploadPartResponse] = {
190       val request = UploadPartRequest.builder().bucket(bucket).key(key).uploadId(uploadId).partNumber(partNumber).build()

192         if (retries > 0) {
193           logger.warn(s"Error with multipart upload, retrying $retries more times:", e)
194           retry(currentChunk, partNumber, retries - 1)
195         } else {
196           logger.error("Error with multipart upload:", e)
197           throw e
198         }
199       }
