21 /**
22  * Mapping of fields to z-value fields
23  *
24  * @param fields fields
25  */
26 case class ZValues(fields: Seq[ZValueField]) {
27   private val fieldMap = fields.map(f => f.geometry -> f.zValue).toMap
28   def isEmpty: Boolean = fields.isEmpty
29   def nonEmpty: Boolean = fields.nonEmpty
30   def get(field: String): Option[String] = fieldMap.get(field)
31 }
32 
33 object ZValues {
34 
35   import scala.collection.JavaConverters._
36 
37   /**
38    * Gets the fields of this schema that have per-row z-values
39    *
40    * @param sft simple feature type
41    * @return
42    */
43   def apply(sft: SimpleFeatureType): ZValues = {
44     val fields = sft.getAttributeDescriptors.asScala.toSeq.flatMap { d =>
45       val binding = d.getType.getBinding
46       if (binding == classOf[Point]) {
47         Some(ZValueField.z2(d.getLocalName))
48       } else if (classOf[Geometry].isAssignableFrom(binding)) {
49         Some(ZValueField.xz2(d.getLocalName))
50       } else {
51         None
52       }
53     }
54     ZValues(fields)
55   }
56 
57   /**
58    * Holder for a z-value field, along with a reference back to the original geometry field
59    *
60    * @param geometry name of the original geometry field being covered
61    * @param zValue name of the z-value field
62    */
63   case class ZValueField(geometry: String, zValue: String)
64 
65   object ZValueField {
66 
67     val ZValueFieldPrefix = "__"
68     val Z2ValueFieldSuffix = "_z2__"
69     val XZ2ValueFieldSuffix = "_xz2__"
70 
71     def z2(geometry: String, encoded: Boolean = false): ZValueField = {
72       val geom = if (encoded) { geometry } else { alphaNumericSafeString(geometry) }
73       val zValue = s"$ZValueFieldPrefix$geom$Z2ValueFieldSuffix"
74       ZValueField(geom, zValue)
75     }
76 
77     def xz2(geometry: String, encoded: Boolean = false): ZValueField = {
78       val geom = if (encoded) { geometry } else { alphaNumericSafeString(geometry) }
79       val zValue = s"$ZValueFieldPrefix$geom$XZ2ValueFieldSuffix"
80       ZValueField(geom, zValue)
81     }
82 
83     /**
84      * Creates a field name based on a z-value field
85      *
86      * @param field name of a potential z-value field
87      * @return
88      */
89     def fromFieldName(field: String): Option[ZValueField] = {
90       if (field.startsWith(ZValueFieldPrefix)) {
91         Seq(Z2ValueFieldSuffix, XZ2ValueFieldSuffix).collectFirst {
92           case suffix if field.endsWith(suffix) =>
93             ZValueField(field.substring(ZValueFieldPrefix.length, field.length - suffix.length), field)
94         }
95       } else {
96         None
97       }
98     }
99 
100     /**
101      * The parquet schema for a z-value field
102      *
103      * @param zValue field name
104      * @return
105      */
106     def schema(zValue: String, fieldIds: AtomicInteger): PrimitiveType =
107       Types.optional(PrimitiveTypeName.INT64).id(fieldIds.getAndIncrement()).named(zValue)
108 
109     /**
110      * The iceberg schema for a z-value field
111      *
112      * @param zValue field name
113      * @return
114      */
115     def icebergSchema(zValue: String, fieldIds: AtomicInteger): NestedField =
116       NestedField.optional(zValue).withId(fieldIds.getAndIncrement()).ofType(LongType.get()).build()
117   }
