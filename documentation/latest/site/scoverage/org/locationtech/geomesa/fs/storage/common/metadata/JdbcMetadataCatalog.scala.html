22 /**
23  * Catalog for jdbc metadata
24  *
25  * @param context file system
26  * @param config jdbc config
27  */
28 class JdbcMetadataCatalog(context: FileSystemContext, config: Map[String, String]) extends StorageMetadataCatalog {
29 
30   // fail fast if url is not defined or invalid
31   private val conf = JdbcMetadataConfig(config)
32   private val driver = JdbcMetadataCatalog.createConnectionFactory(conf)
33   private val metaTable = new MetadataTable(conf.schema, conf.tablePrefix)
34   private val root = context.root.toString
35 
36   override def getTypeNames: Seq[String] = {
37     WithClose(driver.createConnection()) { cx =>
38       WithClose(cx.getMetaData.getTables(null, metaTable.schema, metaTable.tableName, null)) { rs =>
39         if (!rs.next()) {
40           return Seq.empty
41         }
42       }
43       metaTable.selectTypeNames(cx, root)
44     }
45   }
46 
47   override def load(typeName: String): StorageMetadata = {
48     val pool = JdbcMetadataCatalog.createDataSource(driver, conf)
49     try {
50       val meta = WithClose(pool.getConnection())(metaTable.selectMetadata(_, root, typeName))
51       newJdbcMeta(pool, meta)
52     } catch {
53       case NonFatal(e) => CloseQuietly(pool).foreach(e.addSuppressed); throw e
54     }
55   }
56 
57   override def create(sft: SimpleFeatureType, partitions: Seq[String], targetFileSize: Option[Long]): StorageMetadata = {
58     // load the partition scheme first in case it fails
59     partitions.foreach(PartitionSchemeFactory.load(sft, _))
60     val meta = Metadata(sft, partitions, targetFileSize)
61     val pool = JdbcMetadataCatalog.createDataSource(driver, conf)
62     try {
63       WithClose(pool.getConnection()) { cx =>
64         metaTable.create(cx)
65         metaTable.insert(cx, root, meta)
66         new FilesTable(conf.schema, conf.tablePrefix).create(cx)
67       }
68       newJdbcMeta(pool, meta)
69     } catch {
70       case NonFatal(e) => CloseQuietly(pool).foreach(e.addSuppressed); throw e
71     }
72   }
73 
74   private def newJdbcMeta(pool: PoolingDataSource[PoolableConnection], meta: Metadata): JdbcMetadata =
75     new JdbcMetadata(pool, root, conf.schema, conf.tablePrefix, meta.copy(sft = namespaced(meta.sft, context.namespace)))
76 }
77 
78 private object JdbcMetadataCatalog extends LazyLogging {
79 
80   /**
81    * Create (and validate) a connection factory
82    *
83    * @param config config
84    * @return
85    */
86   private def createConnectionFactory(config: JdbcMetadataConfig): DriverManagerConnectionFactory = {
87     config.driver.foreach(Class.forName) // required for older drivers
88     val props = new Properties()
89     config.user.foreach(props.put("user", _))
90     config.password.foreach(props.put("password", _))
91 
92     val driver = new DriverManagerConnectionFactory(config.url, props)
93 
94     // validate the connection parameters
95     WithClose(driver.createConnection()) { connection =>
96       if (!connection.isValid(10)) {
97         throw new IllegalArgumentException(
98           s"Could not create valid connection using configuration ${config.url}")
99       }
100     }
101 
102     driver
103   }
104 
105   /**
106     * Create a jdbc data source based on a configuration
107     *
108     * @param config config
109     * @return
110     */
111   private def createDataSource(driver: DriverManagerConnectionFactory, config: JdbcMetadataConfig): PoolingDataSource[PoolableConnection] = {
112     val poolConfig = new GenericObjectPoolConfig[PoolableConnection]()
113 
114     config.maxIdle.foreach(poolConfig.setMaxIdle)
115     config.minIdle.foreach(poolConfig.setMinIdle)
116     config.maxSize.foreach(poolConfig.setMaxTotal)
117     config.fairness.foreach(poolConfig.setFairness)
118     config.testOnBorrow.foreach(poolConfig.setTestOnBorrow)
119     config.testOnCreate.foreach(poolConfig.setTestOnCreate)
120     config.testWhileIdle.foreach(poolConfig.setTestWhileIdle)
121 
122     val factory = new PoolableConnectionFactory(driver, null)
123     val pool = new GenericObjectPool(factory, poolConfig)
124     factory.setPool(pool)
125 
126     new PoolingDataSource[PoolableConnection](pool)
127   }
