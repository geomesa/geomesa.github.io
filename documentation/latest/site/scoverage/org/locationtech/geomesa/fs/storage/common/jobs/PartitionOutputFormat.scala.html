29 /**
30   * Output format that writes to multiple partition files
31   *
32   * @param delegate underlying output format for a single file
33   */
34 class PartitionOutputFormat(delegate: SingleFileOutputFormat) extends OutputFormat[Void, SimpleFeature] {
35 
36   override def getRecordWriter(context: TaskAttemptContext): RecordWriter[Void, SimpleFeature] =
37     new PartitionSchemeRecordWriter(context)
38 
39   override def getOutputCommitter(context: TaskAttemptContext): OutputCommitter =
40     delegate.getOutputCommitter(context)
41 
42   // same as FileOutputFormat, but doesn't require that output directory doesn't exist
43   override def checkOutputSpecs(job: JobContext): Unit = {
44     // Ensure that the output directory is set
45     val outDir = FileOutputFormat.getOutputPath(job)
46     if (outDir == null) {
47       throw new InvalidJobConfException("Output directory not set")
48     }
49     // get delegation token for outDir's file system
50     TokenCache.obtainTokensForNamenodes(job.getCredentials, Array[Path](outDir), job.getConfiguration)
51   }

53   private class PartitionSchemeRecordWriter(context: TaskAttemptContext)
54       extends RecordWriter[Void, SimpleFeature] with LazyLogging {
55 
56     import StorageConfiguration.Counters.{Features, Group}
57 
58     private val storage = {
59       val conf = context.getConfiguration
60       val root = StorageConfiguration.getRootPath(conf)
61       val fsc = FileSystemContext(root, conf)
62       val encoding = StorageConfiguration.getEncoding(conf)
63       val metadataType = StorageConfiguration.getMetadataType(conf)
64       val metadataConfig = StorageConfiguration.getMetadataConfig(conf)
65       val catalog = StorageMetadataCatalog(fsc, metadataType, metadataConfig)
66       FileSystemStorageFactory(encoding).apply(fsc, catalog.load(StorageConfiguration.getSftName(conf)))
67     }
68 
69     private val fileType = StorageConfiguration.getFileType(context.getConfiguration)
70     private val fileSize = StorageConfiguration.getTargetFileSize(context.getConfiguration)
71 
72     private val counter = context.getCounter(Group, Features)
73     private val cache = scala.collection.mutable.Map.empty[Set[PartitionKey], PartitionState]
74 
75     private val workPath = delegate.getOutputCommitter(context).asInstanceOf[FileOutputCommitter].getWorkPath
76 
77     private def newState(partitions: Set[PartitionKey]): PartitionState = {
78       val estimator = storage match {
79         case s: SizeableFileSystemStorage => s.targetSize(fileSize).map(s.estimator)
80         case _ => None
81       }
82       estimator match {
83         case None => new SinglePartitionState(Partition(partitions))
84         case Some(e) => new ChunkedPartitionState(Partition(partitions), e, context)
85       }
86     }
87 
88     override def write(key: Void, value: SimpleFeature): Unit = {
89       val partition = storage.metadata.schemes.map(_.getPartition(value))
90       val state = cache.getOrElseUpdate(partition, newState(partition))
91       state.write(key, value)
92       counter.increment(1)
93     }
94 
95     override def close(context: TaskAttemptContext): Unit = {
96       cache.foreach { case (partition, state) =>
97         logger.debug(s"Closing writer for $partition")
98         state.close(context)
99         state.getFiles().foreach(storage.metadata.addFile)
100       }
101       CloseWithLogging(storage)
102     }
103 
104     private sealed abstract class PartitionState(partition: Partition) {
105 
106       import scala.collection.JavaConverters._
107 
108       private val files = new CopyOnWriteArrayList[StorageFile]()
109       private val counter = context.getCounter(Group, StorageConfiguration.Counters.partition(partition.toString))
110 
111       private var observer: StorageFileObserver = _
112       private var currentFile: String = _
113 
114       // noinspection AccessorLikeMethodIsEmptyParen
115       def getFiles(): Seq[StorageFile] = files.asScala.toSeq
116 
117       def write(key: Void, value: SimpleFeature): Unit = observer.apply(value)
118 
119       def close(context: TaskAttemptContext): Unit = closeCurrentFile()
120 
121       protected def newWriter(): (Path, RecordWriter[Void, SimpleFeature]) = {
122         closeCurrentFile()
123         currentFile = StorageUtils.newFilePath(storage.metadata.sft.getTypeName, fileType, "parquet")
124         observer = new StorageFileObserver(storage.metadata.sft)
125         val path = new Path(workPath, currentFile)
126         logger.debug(s"Creating record writer at path $path")
127         // noinspection LanguageFeature
128         (path, delegate.getRecordWriter(context, path))
129       }
130 
131       private def closeCurrentFile(): Unit = {
132         if (observer != null) {
133           observer.close()
134           val file = observer.file(currentFile, partition, StorageFileAction.Append)
135           files.add(file)
136           observer = null
137           counter.increment(file.count)
138         }
139       }
140     }

142     private class SinglePartitionState(partition: Partition) extends PartitionState(partition) {
143 
144       private val writer = newWriter()._2
145 
146       override def write(key: Void, value: SimpleFeature): Unit = {
147         writer.write(key, value)
148         super.write(key, value)

151       override def close(context: TaskAttemptContext): Unit = {

157     private class ChunkedPartitionState(partition: Partition, estimator: FileSizeEstimator, context: TaskAttemptContext)
158         extends PartitionState(partition) {
159 
160       private var count = 0L // number of features written
161       private var total = 0L // sum size of all finished chunks
162       private var remaining = estimator.estimate(0L)
163 
164       logger.debug(s"Initial estimate: $remaining")
165 
166       private var path: Path = _
167       private var writer: RecordWriter[Void, SimpleFeature] = _
168 
169       override def write(key: Void, value: SimpleFeature): Unit = {
170         if (writer == null) {
171           val (p, w) = newWriter()
172           path = p
173           writer = w
174         }
175         writer.write(key, value)
176         count += 1
177         remaining -= 1
178         if (remaining == 0) {
179           // TODO if we can access the underlying parquet writer we can check the size without closing the file
180           writer.close(context)
181           logger.debug(s"File length: ${storage.context.fs.getFileStatus(path).getLen} $path")
182           writer = null
183           // adjust our estimate to account for the actual bytes written
184           total += storage.context.fs.getFileStatus(path).getLen
185           estimator.update(total, count)
186           remaining = estimator.estimate(0L)
187           logger.debug(s"New estimate: $remaining")
188         }
189         super.write(key, value)
190       }
191 
192       override def close(context: TaskAttemptContext): Unit = {
193         if (writer != null) {
194           writer.close(context)
195         }
196         super.close(context)
197       }
198     }
199   }
