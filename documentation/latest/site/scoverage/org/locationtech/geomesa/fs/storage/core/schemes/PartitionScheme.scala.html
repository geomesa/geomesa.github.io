19 /**
20  * Scheme for partitioning features into various named partitions (e.g. buckets) on disk, for
21  * faster querying. Partition schemes do not have any persistent state, they only know how to map features
22  * and filters to partition names.
23  *
24  * Note that generally partitioning is implemented to align with iceberg's partitioning logic, so
25  * that partitions in iceberg have a 1:1 mapping with partitions in geomesa (the reverse is not true,
26  * as geomesa supports more partition schemes than iceberg, e.g. spatial partitions)
27  */
28 trait PartitionScheme {
29 
30   /**
31    * Name of this partition scheme
32    *
33    * @return
34    */
35   def name: String
36 
37   /**
38    * Name of the attribute used for partitioning
39    *
40    * @return
41    */
42   def attribute: String
43 
44   /**
45     * Return the partition in which a SimpleFeature should be stored
46     *
47     * @param feature simple feature
48     * @return partition name
49     */
50   def getPartition(feature: SimpleFeature): PartitionKey
51 
52   /**
53    * Gets the geomesa partition value for a given iceberg partition value
54    *
55    * @param partition iceberg partition struct
56    * @param i offest into the partition struct
57    * @return geomesa partition value
58    */
59   def getPartition(partition: StructLike, i: Int): PartitionKey
60 
61   /**
62    * Get a filter that will cover a partitions, i.e. the filter will return all features
63    * in the given partition and none from other partitions
64    *
65    * @param partition partition to cover
66    * @return filter
67    */
68   def getCoveringFilter(partition: PartitionKey): Filter
69 
70   /**
71    * Get a filter that will cover a partitions, i.e. the filter will return all features
72    * in the given partition and none from other partitions
73    *
74    * @param partition partition to cover
75    * @return filter
76    */
77   def getCoveringExpression(partition: PartitionKey): Expression
78 
79   /**
80    * Creates the iceberg partition spec
81    *
82    * @param b spec builder
83    * @return
84    */
85   def spec(b: PartitionSpec.Builder): PartitionSpec.Builder
86 }

88 object PartitionScheme {
89 
90   /**
91    * Enumerate all possible partitions for a given date. Note that not all partition schemes support enumeration
92    *
93    * @param schemes partition schemes
94    * @param date date
95    * @return all possible partitions for the given date
96    * @throws UnsupportedOperationException if the partition scheme can't be enumerated
97    */
98   def enumerate(schemes: Seq[PartitionScheme], date: ZonedDateTime): Seq[Partition] = {
99     val keys = schemes.map {
100       case s: TemporalScheme => Seq(s.partition(date).partition)
101       case s: EnumeratedScheme => s.partitions
102       case s => throw new UnsupportedOperationException(s"The partition scheme '${s.name}' does not support enumerating partitions")
103     }
104     keys.foldLeft(Seq(Partition.None)) { case (partitions, keys) =>
105       for { partition <- partitions; key <- keys } yield {
106         Partition(partition.values :+ key)
107       }
108     }
109   }
110 
111   /**
112    * A scheme with a fixed set of partitions
113    */
114   trait EnumeratedScheme extends PartitionScheme {
115 
116     /**
117      * Get all potential partitions for this scheme
118      *
119      * @return
120      */
121     def partitions: Seq[PartitionKey]
122   }
123 
124   /**
125    * A schemed with a fixed set of partitions per time period
126    */
127   trait TemporalScheme extends PartitionScheme {
128 
129     /**
130      * Gets the partitions for this scheme for the given time
131      *
132      * @param time date to get partitions for
133      * @return
134      */
135     def partition(time: ZonedDateTime): TemporalPartition
136   }
137 
138   /**
139    * A partition, with the associated start and end period
140    *
141    * @param partition partition
142    * @param start start of partitions (inclusive)
143    * @param end end of partitions (exclusive)
144    */
145   case class TemporalPartition(partition: PartitionKey, start: ZonedDateTime, end: ZonedDateTime)
146 }
