20 /**
21  * Metadata mixin that caches a list of files in memory, with automatic background refresh. Useful for when
22  * determining the list of files is expensive.
23  */
24 trait CachedMetadata extends StorageMetadata with SchemeFilterExtraction {
25 
26   protected val filesCache: LoadingCache[BoxedUnit, Seq[StorageFile]] =
27     Caffeine.newBuilder().refreshAfterWrite(CacheDurationProperty.toDuration.get.toMillis, TimeUnit.MILLISECONDS).build(
28       new CacheLoader[BoxedUnit, Seq[StorageFile]]() {
29         override def load(key: BoxedUnit): Seq[StorageFile] = buildFileList()
30       }
31     )
32 
33   private def cachedFiles: Seq[StorageFile] = filesCache.get(BoxedUnit.UNIT)
34 
35   protected def buildFileList(): Seq[StorageFile]
36 
37   override def getFiles(): Seq[StorageFile] = cachedFiles
38 
39   override def getFiles(partition: Partition): Seq[StorageFile] =
40     cachedFiles.filter(_.partition == partition)
41 
42   override def getFiles(filter: Filter): Seq[StorageFile] = {
43     if (filter == Filter.INCLUDE) {
44       getFiles()
45     } else {
46       val added = scala.collection.mutable.HashSet.empty[StorageFile]
47       val files = getFilters(filter).flatMap { f =>
48         cachedFiles.collect { case file if matches(file, f) && added.add(file) => file }
49       }
50       logger.debug(s"Matched files:${files.mkString("\n  ", "\n  ", "")}")
51       logger.trace(s"Skipped files:${cachedFiles.filterNot(files.map(_.file).contains).mkString("\n  ", "\n  ", "")}")
52       files
53     }
54   }
55 
56   override def close(): Unit = {
57     val refresh = filesCache.policy().refreshes().get(BoxedUnit.UNIT)
58     if (refresh != null && !refresh.isDone) {
59       refresh.cancel(true)
60     }
61   }
62 
63   private def matches(file: StorageFile, f: SchemeFilter): Boolean =
64     matches(file, f.partitions) && matches(file, f.spatialBounds) && matches(file, f.attributeBounds)
65 
66   private def matches(file: StorageFile, partitions: Seq[PartitionRange]): Boolean =
67     partitions.forall(p => file.partition.values.exists(v => p.name == v.name && p.contains(v.value)))
68 
69   private def matches(file: StorageFile, spatialBounds: Seq[SpatialOr])(implicit d: DummyImplicit): Boolean =
70     spatialBounds.forall { or =>
71       file.spatialBounds.find(b => b.attribute == or.attribute).forall(b => or.bounds.exists(_.intersects(b)))
72     }
73 
74   private def matches(file: StorageFile, attributeBounds: Seq[AttributeOr])(implicit d0: DummyImplicit, d1: DummyImplicit): Boolean =
75     attributeBounds.forall { or =>
76       file.attributeBounds.find(b => b.attribute == or.attribute).forall(b => or.bounds.exists(_.intersects(b)))
77     }
78 }
