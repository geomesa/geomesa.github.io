24 case class DateTimeScheme(attribute: String, index: Int, unit: ChronoUnit, step: Int = 1) extends PartitionScheme {
25 
26   import FilterHelper.ff
27 
28   private val encoder = LexiTypeEncoders.integerEncoder()
29 
30   override val name: String = {
31     val stepOpt = if (step == 1) { "" } else { s":step=$step"}
32     s"${unit.name().toLowerCase(Locale.US)}:attribute=$attribute$stepOpt"
33   }
34 
35   override def getPartition(feature: SimpleFeature): PartitionKey = {
36     val instant = feature.getAttribute(index).asInstanceOf[Date].toInstant
37     PartitionKey(name, encoder.encode(toPartition(ZonedDateTime.ofInstant(instant, ZoneOffset.UTC))))
38   }
39 
40   override def getRangesForFilter(filter: Filter): Option[Seq[PartitionRange]] = {
41     getBounds(filter).map { bounds =>
42       val builder = new RangeBuilder()
43       bounds.foreach { range =>
44         val lower = range.lower.value.fold("")(v => encoder.encode(toPartition(v)))
45         val upper = exclusiveUpperBound(range).fold(DateTimeScheme.UnboundedUpper)(v => encoder.encode(v))
46         builder += PartitionRange(name, lower, upper)
47       }
48       builder.result()
49     }
50   }
51 
52   override def getPartitionsForFilter(filter: Filter): Option[Seq[PartitionKey]] = {
53     getBounds(filter).map { bounds =>
54       bounds.flatMap { bound =>
55         if (!bound.isBoundedBothSides) {
56           throw new IllegalArgumentException(s"Can't enumerate an unbounded filter: ${ECQL.toCQL(filter)}")
57         }
58         val lower = toPartition(bound.lower.value.get)
59         val upper = exclusiveUpperBound(bound).get
60         Range(lower, upper).map(v => PartitionKey(name, encoder.encode(v)))
61       }
62     }
63   }
64 
65   override def getCoveringFilter(partition: PartitionKey): Filter = {
66     val offset = encoder.decode(partition.value) * step
67     val start = DateTimeScheme.Epoch.plus(offset.longValue(), unit)
68     val end = ff.literal(DateParsing.format(start.plus(step, unit)))
69     ff.and(ff.greaterOrEqual(ff.property(attribute), ff.literal(DateParsing.format(start))), ff.less(ff.property(attribute), end))
70   }
71 
72   private def toPartition(dt: ZonedDateTime): Int = unit.between(DateTimeScheme.Epoch, dt).toInt / step
73 
74   private def getBounds(filter: Filter): Option[Seq[Bounds[ZonedDateTime]]] = {
75     val bounds = FilterHelper.extractIntervals(filter, attribute)
76     if (bounds.isEmpty) {
77       None
78     } else if (bounds.disjoint) {
79       Some(Seq.empty)
80     } else {
81       Some(bounds.values)
82     }
83   }
84 
85   private def exclusiveUpperBound(bounds: Bounds[ZonedDateTime]): Option[Int] = {
86     bounds.upper.value.map { v =>
87       val partition = toPartition(v)
88       if (bounds.upper.exclusive && toPartition(v.minus(1, ChronoUnit.MILLIS)) < partition) {
89         partition
90       } else {
91         partition + 1
92       }
93     }
94   }
95 }

97 object DateTimeScheme extends PartitionSchemeFactory {
98 
99   import org.locationtech.geomesa.utils.geotools.RichSimpleFeatureType.RichSimpleFeatureType
100 
101   private val Epoch: ZonedDateTime = ZonedDateTime.ofInstant(Instant.EPOCH, ZoneOffset.UTC)
102 
103   private val UnboundedUpper = "zzz"
104 
105   override def load(sft: SimpleFeatureType, scheme: String): Option[PartitionScheme] = {
106     val opts = SchemeOpts(scheme)
107     val unit = opts.name match {
108       case "year"  | "years"  | "yearly"  => Some(ChronoUnit.YEARS)
109       case "month" | "months" | "monthly" => Some(ChronoUnit.MONTHS)
110       case "week"  | "weeks"  | "weekly"  => Some(ChronoUnit.WEEKS)
111       case "day"   | "days"   | "daily"   => Some(ChronoUnit.DAYS)
112       case "hour"  | "hours"  | "hourly"  => Some(ChronoUnit.HOURS)
113       case _ => None
114     }
115     unit.map { u =>
116       val dtg = opts.getSingle("attribute").orElse(sft.getDtgField).orNull
117       require(dtg != null, s"Date scheme requires an attribute to be specified with 'attribute=<attribute>'")
118       val index = attributeIndex(sft, dtg, Some(classOf[Date]))
119       val step = opts.getSingle("step").map(_.toInt).getOrElse(1)
120       DateTimeScheme(dtg, index, u, step)
121     }
122   }
