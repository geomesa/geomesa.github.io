21 /**
22  * Utility for tracking target file sizes
23  *
24  * @param fs filesystem - note, not cleaned up, must be closed externally
25  * @param metadata metadata
26  */
27 class FileSize(fs: ObjectStore, metadata: StorageMetadata) {
28 
29   private val fileSizeError = FileSize.FileSizeErrorThreshold.toFloat.get
30 
31   private var averageBytesPerFeature = metadata.get(FileSize.BytesPerFeature) match {
32     case Some(b) => b.toFloat
33     case None    => (metadata.sft.getAttributeCount + 1) * 1.6f // 1.6 taken from some sample data estimates...
34   }
35 
36   /**
37    * Gets the target file size
38    *
39    * @return
40    */
41   def targetSize: Option[Long] = metadata.get(Metadata.TargetFileSize).map(_.toLong)
42 
43   /**
44    * Check if a file is already the desired size
45    *
46    * @param path file path
47    * @param target target file size
48    * @return true if the file is appropriately sized
49    */
50   def fileIsSized(path: URI, target: Long): Boolean = {
51     val size = fs.size(path)
52     math.abs((size.toDouble / target) - 1d) <= fileSizeError
53   }
54 
55   /**
56    * Gets a file size estimator for this storage instance
57    *
58    * @param size target file size
59    * @return
60    */
61   def estimator(size: Long): UpdatingFileSizeEstimator =
62     FileSize.this.synchronized {
63       new UpdatingFileSizeEstimator(size, this)
64     }
65 
66   private def updateFileSize(bytesPerFeature: Float): Unit = {
67     if (metadata.get(FileSize.UseDynamicSizing).forall(_.toBoolean)) {
68       synchronized {
69         if (math.abs((bytesPerFeature / averageBytesPerFeature) - 1f) > fileSizeError) {
70           averageBytesPerFeature = bytesPerFeature
71           metadata.set(FileSize.BytesPerFeature, java.lang.Float.toString(bytesPerFeature))
72         }
73       }
74     }
75   }
76 }
77 
78 object FileSize {
79 
80   val BytesPerFeature  = "bytes-per-feature"
81   val UseDynamicSizing = "use-dynamic-sizing"
82 
83   val FileSizeErrorThreshold: SystemProperty = SystemProperty("geomesa.fs.size.threshold", "0.05")
84 
85   class UpdatingFileSizeEstimator(target: Long, instance: FileSize)
86       extends FileSizeEstimator(target, instance.fileSizeError, instance.averageBytesPerFeature) with Closeable {
87     override def close(): Unit = getBytesPerFeature.foreach(instance.updateFileSize)
88   }
89 }
