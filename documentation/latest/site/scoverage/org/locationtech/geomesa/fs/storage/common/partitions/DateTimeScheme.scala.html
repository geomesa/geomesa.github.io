25 case class DateTimeScheme(dtg: String, dtgIndex: Int, unit: ChronoUnit, step: Int = 1) extends PartitionScheme {
26 
27   import FilterHelper.ff
28 
29   private val encoder = LexiTypeEncoders.integerEncoder()
30 
31   override val name: String = {
32     val stepOpt = if (step == 1) { "" } else { s":step=$step"}
33     s"${unit.name().toLowerCase(Locale.US)}:attribute=$dtg$stepOpt"
34   }
35 
36   override def getPartition(feature: SimpleFeature): PartitionKey = {
37     val instant = feature.getAttribute(dtgIndex).asInstanceOf[Date].toInstant
38     PartitionKey(name, encoder.encode(toPartition(ZonedDateTime.ofInstant(instant, ZoneOffset.UTC))))
39   }
40 
41   override def getRangesForFilter(filter: Filter): Option[Seq[PartitionRange]] = {
42     getBounds(filter).map { bounds =>
43       val builder = new RangeBuilder()
44       bounds.foreach { range =>
45         val lower = range.lower.value.fold("")(v => encoder.encode(toPartition(v)))
46         val upper = exclusiveUpperBound(range).fold(DateTimeScheme.UnboundedUpper)(v => encoder.encode(v))
47         builder += PartitionRange(name, lower, upper)
48       }
49       builder.result()
50     }
51   }
52 
53   override def getPartitionsForFilter(filter: Filter): Option[Seq[PartitionKey]] = {
54     getBounds(filter).map { bounds =>
55       bounds.flatMap { bound =>
56         if (!bound.isBoundedBothSides) {
57           throw new IllegalArgumentException(s"Can't enumerate an unbounded filter: ${ECQL.toCQL(filter)}")
58         }
59         val lower = toPartition(bound.lower.value.get)
60         val upper = exclusiveUpperBound(bound).get
61         Range(lower, upper).map(v => PartitionKey(name, encoder.encode(v)))
62       }
63     }
64   }
65 
66   override def getCoveringFilter(partition: PartitionKey): Filter = {
67     val offset = encoder.decode(partition.value) * step
68     val start = DateTimeScheme.Epoch.plus(offset.longValue(), unit)
69     val end = ff.literal(DateParsing.format(start.plus(step, unit)))
70     ff.and(ff.greaterOrEqual(ff.property(dtg), ff.literal(DateParsing.format(start))), ff.less(ff.property(dtg), end))
71   }
72 
73   private def toPartition(dt: ZonedDateTime): Int = unit.between(DateTimeScheme.Epoch, dt).toInt / step
74 
75   private def getBounds(filter: Filter): Option[Seq[Bounds[ZonedDateTime]]] = {
76     val bounds = FilterHelper.extractIntervals(filter, dtg)
77     if (bounds.isEmpty) {
78       None
79     } else if (bounds.disjoint) {
80       Some(Seq.empty)
81     } else {
82       Some(bounds.values)
83     }
84   }
85 
86   private def exclusiveUpperBound(bounds: Bounds[ZonedDateTime]): Option[Int] = {
87     bounds.upper.value.map { v =>
88       val partition = toPartition(v)
89       if (bounds.upper.exclusive && toPartition(v.minus(1, ChronoUnit.MILLIS)) < partition) {
90         partition
91       } else {
92         partition + 1
93       }
94     }
95   }
96 }

106   class DateTimePartitionSchemeFactory extends PartitionSchemeFactory {
107     override def load(sft: SimpleFeatureType, scheme: String): Option[PartitionScheme] = {
108       val opts = SchemeOpts(scheme)
109       val unit = opts.name match {
110         case "year"  | "years"  | "yearly"  => Some(ChronoUnit.YEARS)
111         case "month" | "months" | "monthly" => Some(ChronoUnit.MONTHS)
112         case "week"  | "weeks"  | "weekly"  => Some(ChronoUnit.WEEKS)
113         case "day"   | "days"   | "daily"   => Some(ChronoUnit.DAYS)
114         case "hour"  | "hours"  | "hourly"  => Some(ChronoUnit.HOURS)
115         case _ => None
116       }
117       unit.map { u =>
118         val dtg = opts.getSingle("attribute").orElse(sft.getDtgField).orNull
119         require(dtg != null, s"Date scheme requires an attribute to be specified with 'attribute=<attribute>'")
120         val index = attributeIndex(sft, dtg, Some(classOf[Date]))
121         val step = opts.getSingle("step").map(_.toInt).getOrElse(1)
122         DateTimeScheme(dtg, index, u, step)
123       }
124     }
125   }
