46     def compact(ds: FileSystemDataStore): Unit = {
47       Command.user.info("Beginning compaction process...")
48 
49       val storage = ds.storage(params.featureName)
50 
51       val toCompact = if (params.loadedPartitions.isEmpty) { storage.metadata.partitions() } else {
52         val filtered = params.loadedPartitions.filter(storage.metadata.files().forPartition(_).scan().nonEmpty)
53         if (filtered.isEmpty) {
54           throw new ParameterException(s"Partition(s) did not match any files: ${params.loadedPartitions.mkString(", ")}")
55         } else if (filtered.size != params.loadedPartitions.size) {
56           val unmatched = params.loadedPartitions.filterNot(filtered.contains)
57           Command.user.warn(s"Some partition did not match any files: ${unmatched.mkString(", ")}")
58         }
59         filtered
60       }
61 
62       val mode = params.mode.getOrElse {
63         if (PathUtils.isRemote(storage.context.root.toString)) { RunModes.Distributed } else { RunModes.Local }
64       }
65 
66       Command.user.info(s"Compacting ${toCompact.size} files in ${mode.toString.toLowerCase(Locale.US)} mode")
67 
68       val start = System.currentTimeMillis()
69       val status = TerminalCallback()
70 
71       mode match {
72         case RunModes.Local =>
73           val total = toCompact.length
74           val latch = new CountDownLatch(total)
75           val executor = Executors.newFixedThreadPool(math.max(1, math.min(params.threads, total)))
76 
77           try {
78             toCompact.foreach { p =>
79               executor.submit(
80                 new Runnable() {
81                   override def run(): Unit = {
82                     try {
83                       logger.info(s"Compacting $p")
84                       storage.metadata.compact(p)
85                     } catch {
86                       case NonFatal(e) => logger.error(s"Error processing partition '$p':", e)
87                     } finally {
88                       latch.countDown()
89                     }
90                   }
91                 }
92               )
93             }
94           } finally {
95             executor.shutdown()
96           }
97 
98           while (latch.getCount > 0) {
99             Thread.sleep(1000)
100             status("", 1f - latch.getCount.toFloat / total, Seq.empty, done = false)
101           }
102           status("", 1f, Seq.empty, done = true)
103           Command.user.info(s"Local compaction complete in ${TextTools.getTime(start)}")
104 
105         case RunModes.Distributed =>
106           val job = new ParquetCompactionJob()
107           val tempDir = Option(params.tempPath).map(t => new Path(t))
108           job.run(storage, toCompact.toSeq, tempDir, libjarsFiles, libjarsPaths, status) match {
109             case JobSuccess(message, counts) =>
110               Command.user.info(s"Distributed compaction complete in ${TextTools.getTime(start)}")
111               val success = counts(FileSystemCompactionJob.MappedCounter)
112               val failed = counts(FileSystemCompactionJob.FailedCounter)
113               Command.user.info(IngestCommand.getStatInfo(success, failed, "Compacted", message))
114 
115             case JobFailure(message) =>
116               Command.user.error(s"Distributed compaction failed in ${TextTools.getTime(start)}")
117               throw new CommandException(message)
118           }
119       }
120     }
121   }
