48     def compact(ds: FileSystemDataStore): Unit = {
49       Command.user.info("Beginning compaction process...")
50 
51       val storage = ds.storage(params.featureName)
52 
53       val toCompact = if (params.loadedPartitions.isEmpty) { storage.metadata.getFiles().map(_.partition).distinct } else {
54         val filtered = params.loadedPartitions.filter(storage.metadata.getFiles(_).nonEmpty)
55         if (filtered.isEmpty) {
56           throw new ParameterException(s"Partition(s) did not match any files: ${params.loadedPartitions.mkString(", ")}")
57         } else if (filtered.size != params.loadedPartitions.size) {
58           val unmatched = params.loadedPartitions.filterNot(filtered.contains)
59           Command.user.warn(s"Some partition did not match any files: ${unmatched.mkString(", ")}")
60         }
61         filtered
62       }
63 
64       val mode = params.mode.getOrElse {
65         if (PathUtils.isRemote(storage.context.root.toString)) { RunModes.Distributed } else { RunModes.Local }
66       }
67 
68       Command.user.info(s"Compacting ${toCompact.size} files in ${mode.toString.toLowerCase(Locale.US)} mode")
69 
70       val start = System.currentTimeMillis()
71       val status = TerminalCallback()
72 
73       mode match {
74         case RunModes.Local =>
75           val total = toCompact.length
76           val latch = new CountDownLatch(total)
77           val executor = Executors.newFixedThreadPool(math.max(1, math.min(params.threads, total)))
78 
79           try {
80             toCompact.foreach { p =>
81               executor.submit(
82                 new Runnable() {
83                   override def run(): Unit = {
84                     try {
85                       logger.info(s"Compacting $p")
86                       storage.compact(p)
87                     } catch {
88                       case NonFatal(e) => logger.error(s"Error processing partition '$p':", e)
89                     } finally {
90                       latch.countDown()
91                     }
92                   }
93                 }
94               )
95             }
96           } finally {
97             executor.shutdown()
98           }
99 
100           while (latch.getCount > 0) {
101             Thread.sleep(1000)
102             status("", 1f - latch.getCount.toFloat / total, Seq.empty, done = false)
103           }
104           status("", 1f, Seq.empty, done = true)
105           Command.user.info(s"Local compaction complete in ${TextTools.getTime(start)}")
106 
107         case RunModes.Distributed =>
108           val job = new ParquetCompactionJob()
109           val tempDir = Option(params.tempPath).map(t => new Path(t))
110           job.run(storage, toCompact.toSeq, tempDir, libjarsFiles, libjarsPaths, status) match {
111             case JobSuccess(message, counts) =>
112               Command.user.info(s"Distributed compaction complete in ${TextTools.getTime(start)}")
113               val success = counts(FileSystemCompactionJob.MappedCounter)
114               val failed = counts(FileSystemCompactionJob.FailedCounter)
115               Command.user.info(IngestCommand.getStatInfo(success, failed, "Compacted", message))
116 
117             case JobFailure(message) =>
118               Command.user.error(s"Distributed compaction failed in ${TextTools.getTime(start)}")
119               throw new CommandException(message)
120           }
121       }
122     }
123   }
