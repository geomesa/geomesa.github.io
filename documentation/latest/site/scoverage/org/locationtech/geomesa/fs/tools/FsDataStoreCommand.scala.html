37   override def connection: Map[String, String] = {
38     val builder = Map.newBuilder[String, String]
39     builder += (FileSystemDataStoreParams.PathParam.key -> params.path)
40     if (params.metadataType != null) {
41       builder += (FileSystemDataStoreParams.MetadataTypeParam.key -> params.metadataType)
42     }
43     if (!params.configuration.isEmpty) {
44       val props = new Properties()
45       params.configuration.asScala.foreach { case (k, v) => props.put(k, v) }
46       val out = new StringWriter()
47       props.store(out, null)
48       builder += (FileSystemDataStoreParams.ConfigParam.key -> out.toString)
49     }
50     if (params.configFile != null) {
51       builder += (FileSystemDataStoreParams.ConfigFileParam.key -> params.configFile)
52     }
53     if (params.auths != null) {
54       builder += (FileSystemDataStoreParams.AuthsParam.key -> params.auths)
55     }
56     if (params.encoding != null) {
57       builder += (FileSystemDataStoreParams.EncodingParam.key -> params.encoding)
58     }
59     val maxOpenPartitions =
60       Option(params).collect { case p: FsIngestParams if p.maxOpenPartitions != null => p.maxOpenPartitions.toString }.orNull
61     if (maxOpenPartitions != null) {
62       builder += (FileSystemDataStoreParams.WritersMaxOpenPartitionsParam.key -> maxOpenPartitions)
63     }
64     builder.result()
65   }
66 }
