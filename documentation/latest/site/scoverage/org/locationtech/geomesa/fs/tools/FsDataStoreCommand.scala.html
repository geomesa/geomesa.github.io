30   import scala.collection.JavaConverters._

32   override def params: FsParams

34   override def connection: Map[String, String] = {
35     val builder = Map.newBuilder[String, String]
36     builder += (FileSystemDataStoreParams.PathParam.key -> params.path)
37     builder += (FileSystemDataStoreParams.MetadataTypeParam.key -> params.metadataType)
38     if (!params.configuration.isEmpty) {
39       val props = new Properties()
40       params.configuration.asScala.foreach { case (k, v) => props.put(k, v) }
41       val out = new StringWriter()
42       props.store(out, null)
43       builder += (FileSystemDataStoreParams.ConfigParam.key -> out.toString)
44     }
45     if (params.configFile != null) {
46       builder += (FileSystemDataStoreParams.ConfigFileParam.key -> params.configFile)
47     }
48     if (params.auths != null) {
49       builder += (FileSystemDataStoreParams.AuthsParam.key -> params.auths)
50     }
51     if (params.encoding != null) {
52       builder += (FileSystemDataStoreParams.EncodingParam.key -> params.encoding)
53     }
54     builder.result()
55   }
56 }
