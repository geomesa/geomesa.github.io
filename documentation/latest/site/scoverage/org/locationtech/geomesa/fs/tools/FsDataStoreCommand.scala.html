30 trait FsDataStoreCommand extends DataStoreCommand[FileSystemDataStore] {
31 
32   import scala.collection.JavaConverters._
33 
34   override def params: FsParams

36   override def connection: Map[String, String] = {
37     val builder = Map.newBuilder[String, String]
38     builder += (FileSystemDataStoreParams.PathParam.key -> params.path)
39     if (params.metadataType != null) {
40       builder += (FileSystemDataStoreParams.MetadataTypeParam.key -> params.metadataType)
41     }
42     if (!params.configuration.isEmpty) {
43       val props = new Properties()
44       params.configuration.asScala.foreach { case (k, v) => props.put(k, v) }
45       val out = new StringWriter()
46       props.store(out, null)
47       builder += (FileSystemDataStoreParams.ConfigParam.key -> out.toString)
48     }
49     if (params.configFile != null) {
50       builder += (FileSystemDataStoreParams.ConfigFileParam.key -> params.configFile)
51     }
52     if (params.auths != null) {
53       builder += (FileSystemDataStoreParams.AuthsParam.key -> params.auths)
54     }
55     if (params.encoding != null) {
56       builder += (FileSystemDataStoreParams.EncodingParam.key -> params.encoding)
57     }
58     builder.result()
59   }
60 }
