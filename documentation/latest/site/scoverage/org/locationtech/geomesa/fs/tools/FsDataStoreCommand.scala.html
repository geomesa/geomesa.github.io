28 trait FsDataStoreCommand extends DataStoreCommand[FileSystemDataStore] {
29 
30   import scala.collection.JavaConverters._

32   override def params: FsParams

34   override def connection: Map[String, String] = {
35     val builder = Map.newBuilder[String, String]
36     builder += (FileSystemDataStoreParams.PathParam.key -> params.path)
37     if (params.metadataType != null) {
38       builder += (FileSystemDataStoreParams.MetadataTypeParam.key -> params.metadataType)
39     }
40     if (!params.configuration.isEmpty) {
41       val props = new Properties()
42       params.configuration.asScala.foreach { case (k, v) => props.put(k, v) }
43       val out = new StringWriter()
44       props.store(out, null)
45       builder += (FileSystemDataStoreParams.ConfigParam.key -> out.toString)
46     }
47     if (params.configFile != null) {
48       builder += (FileSystemDataStoreParams.ConfigFileParam.key -> params.configFile)
49     }
50     if (params.auths != null) {
51       builder += (FileSystemDataStoreParams.AuthsParam.key -> params.auths)
52     }
53     if (params.encoding != null) {
54       builder += (FileSystemDataStoreParams.EncodingParam.key -> params.encoding)
55     }
56     builder.result()
57   }
58 }
