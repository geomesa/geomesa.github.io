31 trait FsDataStoreCommand extends DataStoreCommand[FileSystemDataStore] {
32 
33   import scala.collection.JavaConverters._
34 
35   override def params: FsParams
36 
37   override def connection: Map[String, String] = {
38     val builder = Map.newBuilder[String, String]
39     builder += (FileSystemDataStoreParams.PathParam.key -> params.path)
40     if (params.catalogType != null) {
41       builder += (FileSystemDataStoreParams.CatalogTypeParam.key -> params.catalogType)
42     }
43     if (!params.configuration.isEmpty) {
44       val props = new Properties()
45       params.configuration.asScala.foreach { case (k, v) => props.put(k, v) }
46       val out = new StringWriter()
47       props.store(out, null)
48       builder += (FileSystemDataStoreParams.ConfigParam.key -> out.toString)
49     }
50     if (params.configFile != null) {
51       builder += (FileSystemDataStoreParams.ConfigFileParam.key -> params.configFile.getAbsolutePath)
52     }
53     if (params.auths != null) {
54       builder += (FileSystemDataStoreParams.AuthsParam.key -> params.auths)
55     }
56     val maxOpenPartitions =
57       Option(params).collect { case p: FsIngestParams if p.maxOpenPartitions != null => p.maxOpenPartitions.toString }.orNull
58     if (maxOpenPartitions != null) {
59       builder += (FileSystemDataStoreParams.WritersMaxOpenPartitionsParam.key -> maxOpenPartitions)
60     }
61     builder.result()
62   }
63 }
