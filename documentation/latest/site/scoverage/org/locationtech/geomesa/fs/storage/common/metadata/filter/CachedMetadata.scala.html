22 /**
23  * Metadata mixin that caches a list of files in memory, with automatic background refresh. Useful for when
24  * determining the list of files is expensive.
25  */
26 trait CachedMetadata extends StorageMetadata with SchemeFilterExtraction {
27 
28   protected val filesCache: LoadingCache[BoxedUnit, Seq[StorageFile]] =
29     Caffeine.newBuilder().refreshAfterWrite(PathCache.CacheDurationProperty.toDuration.get.toMillis, TimeUnit.MILLISECONDS).build(
30       new CacheLoader[BoxedUnit, Seq[StorageFile]]() {
31         override def load(key: BoxedUnit): Seq[StorageFile] = buildFileList()
32       }
33     )
34 
35   private def cachedFiles: Seq[StorageFile] = filesCache.get(BoxedUnit.UNIT)
36 
37   protected def buildFileList(): Seq[StorageFile]
38 
39   override def getFiles(): Seq[StorageFile] = cachedFiles
40 
41   override def getFiles(partition: Partition): Seq[StorageFile] =
42     cachedFiles.filter(_.partition == partition)
43 
44   override def getFiles(filter: Filter): Seq[StorageFile] = {
45     if (filter == Filter.INCLUDE) {
46       getFiles()
47     } else {
48       val added = scala.collection.mutable.HashSet.empty[StorageFile]
49       val files = getFilters(filter).flatMap { f =>
50         cachedFiles.collect { case file if matches(file, f) && added.add(file) => file }
51       }
52       logger.debug(s"Matched files:${files.mkString("\n  ", "\n  ", "")}")
53       logger.trace(s"Skipped files:${cachedFiles.filterNot(files.map(_.file).contains).mkString("\n  ", "\n  ", "")}")
54       files
55     }
56   }
57 
58   override def close(): Unit = {
59     val refresh = filesCache.policy().refreshes().get(BoxedUnit.UNIT)
60     if (refresh != null && !refresh.isDone) {
61       refresh.cancel(true)
62     }
63   }
64 
65   private def matches(file: StorageFile, f: SchemeFilter): Boolean =
66     matches(file, f.partitions) && matches(file, f.spatialBounds) && matches(file, f.attributeBounds)
67 
68   private def matches(file: StorageFile, partitions: Seq[PartitionRange]): Boolean =
69     partitions.forall(p => file.partition.values.exists(v => p.name == v.name && p.contains(v.value)))
70 
71   private def matches(file: StorageFile, spatialBounds: Seq[SpatialOr])(implicit d: DummyImplicit): Boolean =
72     spatialBounds.forall { or =>
73       file.spatialBounds.find(b => b.attribute == or.attribute).forall(b => or.bounds.exists(_.intersects(b)))
74     }
75 
76   private def matches(file: StorageFile, attributeBounds: Seq[AttributeOr])(implicit d0: DummyImplicit, d1: DummyImplicit): Boolean =
77     attributeBounds.forall { or =>
78       file.attributeBounds.find(b => b.attribute == or.attribute).forall(b => or.bounds.exists(_.intersects(b)))
79     }
80 }
