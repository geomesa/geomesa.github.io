29 /**
30   * An Input format that creates splits based on FSDS Partitions. This is used for compaction, when we want a single
31   * split per partition. Otherwise, use OrcSimpleFeatureInputFormat/ParquetSimpleFeatureInputFormat as those are
32   * more efficient
33   */
34 class PartitionInputFormat extends InputFormat[Void, SimpleFeature] {
35 
36   override def getSplits(context: JobContext): java.util.List[InputSplit] = {
37     val conf = context.getConfiguration
38 
39     val root = StorageConfiguration.getRootPath(conf)
40     val fsc = FileSystemContext(root, conf)
41     val fileSize = StorageConfiguration.getTargetFileSize(conf)
42     val metadataType = StorageConfiguration.getMetadataType(conf)
43     val metadataConfig = StorageConfiguration.getMetadataConfig(conf)
44     val typeName = StorageConfiguration.getSftName(conf)
45     val encoding = StorageConfiguration.getEncoding(conf)
46 
47     val catalog = StorageMetadataCatalog(fsc, metadataType, metadataConfig)
48     val factory = FileSystemStorageFactory(encoding)
49     WithClose(factory.apply(fsc, catalog.load(typeName))) { storage =>
50       val sizeable = Option(storage).collect { case s: SizeableFileSystemStorage => s }
51       val sizeCheck = sizeable.flatMap(s => s.targetSize(fileSize).map(t => (p: Path) => s.fileIsSized(p, t)))
52       val splits = StorageConfiguration.getPartitions(conf).map { partition =>
53         var size = 0L
54         val files = storage.metadata.getFiles(partition).filter { f =>
55           if (sizeCheck.exists(_.apply(new Path(fsc.root, f.file)))) { false } else {
56             size += PathCache.status(fsc.fs, new Path(fsc.root, f.file)).getLen
57             true
58           }
59         }
60         new PartitionInputSplit(partition.toString, files, size)
61       }
62       java.util.Arrays.asList(splits: _*)
63     }
64   }
65 
66   override def createRecordReader(split: InputSplit, context: TaskAttemptContext): RecordReader[Void, SimpleFeature] = {
67     val psplit = split.asInstanceOf[PartitionInputSplit]
68     new PartitionRecordReader(psplit.getFiles)
69   }
70 }

74   /**
75     * InputSplit corresponding to a single FileSystemDataStore PartitionScheme partition
76     */
77   class PartitionInputSplit extends InputSplit with Writable {
78 
79     private var name: String = _
80     private var files: Seq[StorageFile] = _
81     private var length: java.lang.Long = _
82 
83     def this(name: String, files: Seq[StorageFile], length: Long) = {
84       this()
85       this.name = name
86       this.files = files
87       this.length = length
88     }
89 
90     /**
91       * @return the name of this partition
92       */
93     def getName: String = name
94 
95     def getFiles: Seq[StorageFile] = files
96 
97     override def getLength: Long = length
98 
99     // TODO attempt to optimize the locations where this should run in the case of HDFS
100     // With S3 this won't really matter
101     override def getLocations: Array[String] = Array.empty[String]
102 
103     override def write(out: DataOutput): Unit = {
104       out.writeUTF(name)
105       out.writeLong(length)
106       out.writeInt(files.length)
107       // note: we don't store bounds, sort, etc as they're not used
108       files.foreach { file =>
109         out.writeUTF(file.file)
110         out.writeUTF(file.partition.toString)
111         out.writeLong(file.count)
112         out.writeUTF(file.action.toString)
113         out.writeLong(file.timestamp)
114       }
115     }
116 
117     override def readFields(in: DataInput): Unit = {
118       this.name = in.readUTF()
119       this.length = in.readLong()
120       this.files = Seq.fill(in.readInt) {
121         StorageFile(in.readUTF(), Partition(in.readUTF()), in.readLong, StorageFileAction.withName(in.readUTF()), Seq.empty, Seq.empty, Seq.empty, in.readLong())
122       }
123     }
124   }

126   class PartitionRecordReader(files: Seq[StorageFile]) extends RecordReader[Void, SimpleFeature] {
127 
128     private var storage: FileSystemStorage = _
129     private var reader: CloseableFeatureIterator = _
130 
131     private var curValue: SimpleFeature = _
132 
133     override def initialize(split: InputSplit, context: TaskAttemptContext): Unit = {
134       val conf = context.getConfiguration
135       val root = StorageConfiguration.getRootPath(conf)
136       val fsc = FileSystemContext(root, conf)
137       val sft = StorageConfiguration.getSft(conf)
138       val encoding = StorageConfiguration.getEncoding(conf)
139 
140       // use a cached metadata impl instead of reloading
141       val cached = new StaticMetadata(sft, files)
142       storage = FileSystemStorageFactory(encoding).apply(fsc, cached)
143       reader = storage.getReader(new Query("", Filter.INCLUDE))
144     }
145 
146     // TODO look at how the ParquetInputFormat provides progress and utilize something similar
147     override def getProgress: Float = 0.0f
148 
149     override def nextKeyValue(): Boolean = {
150       if (reader.hasNext) {
151         curValue = reader.next()
152         true
153       } else {
154         curValue = null
155         false
156       }
157     }
158 
159     override def getCurrentKey: Void = null
160     override def getCurrentValue: SimpleFeature = curValue
161 
162     override def close(): Unit = {
163       if (reader != null) {
164         CloseWithLogging(reader)
165       }
166       if (storage != null) {
167         CloseWithLogging(storage)
168       }
169     }
170   }

172   private class StaticMetadata(val sft: SimpleFeatureType, files: Seq[StorageFile]) extends StorageMetadata with LazyLogging {
173     override def `type`: String = "static"
174     override def getFiles(): Seq[StorageFile] = files
175     override def getFiles(partition: Partition): Seq[StorageFile] = files.filter(_.partition == partition)
176     override def getFiles(filter: Filter): Seq[StorageFile] = {
177       // note: should only be called with filter.include
178       if (filter != Filter.INCLUDE) {
179         logger.warn(s"Unexpected filter: $filter")
180       }
181       files
182     }
183     override def addFile(file: StorageFile): Unit = throw new UnsupportedOperationException()
184     override def removeFile(file: StorageFile): Unit = throw new UnsupportedOperationException()
185     override def replaceFiles(existing: Seq[StorageFile], replacements: Seq[StorageFile]): Unit =
186       throw new UnsupportedOperationException()
187     override def schemes: Set[PartitionScheme] = throw new UnsupportedOperationException()
188     override def close(): Unit = {}
189   }
