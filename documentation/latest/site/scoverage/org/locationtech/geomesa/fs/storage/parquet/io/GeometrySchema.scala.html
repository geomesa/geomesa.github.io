36   /**
37    * Holder for multiple bounding box fields
38    *
39    * @param fields fields
40    */
41   case class BoundingBoxes(fields: Seq[BoundingBoxField]) {
42     private val fieldMap = fields.map(f => f.geometry -> f.bbox).toMap
43     def isEmpty: Boolean = fields.isEmpty
44     def nonEmpty: Boolean = fields.nonEmpty
45     def get(field: String): Option[String] = fieldMap.get(field)

48   object BoundingBoxes {
49 
50     import scala.collection.JavaConverters._
51 
52     /**
53      * Gets the fields of this schema that have per-row bounding-boxes. We only add bounding boxes when
54      * they help with predicate push-down, e.g. when a geometry is a non-point or WKB encoded
55      *
56      * @param sft simple feature type
57      * @param encoding geometry encoding
58      * @return
59      */
60     def apply(sft: SimpleFeatureType, encoding: GeometryEncoding): BoundingBoxes = {
61       val bboxes = sft.getAttributeDescriptors.asScala.toSeq.flatMap { d =>
62         val binding = d.getType.getBinding
63         if (classOf[Geometry].isAssignableFrom(binding) &&
64           (encoding == GeometryEncoding.GeoParquetWkb || binding != classOf[Point])) {
65           Some(BoundingBoxField(d.getLocalName))
66         } else {
67           None
68         }
69       }
70       BoundingBoxes(bboxes)
71     }
72   }

74   object BoundingBoxField {
75 
76     val BoundingBoxFieldPrefix = "__"
77     val BoundingBoxFieldSuffix = "_bbox__"
78 
79     val XMin = "xmin"
80     val YMin = "ymin"
81     val XMax = "xmax"
82     val YMax = "ymax"
83 
84     def apply(geometry: String, encoded: Boolean = false): BoundingBoxField = {
85       val geom = if (encoded) { geometry } else { alphaNumericSafeString(geometry) }
86       val bbox = s"$BoundingBoxFieldPrefix$geom$BoundingBoxFieldSuffix"
87       BoundingBoxField(geom, bbox)
88     }
89 
90     /**
91      * Creates a field name based on a bounding box field
92      *
93      * @param bbox name of a potential bbox field
94      * @return
95      */
96     def fromBoundingBox(bbox: String): Option[BoundingBoxField] = {
97       if (bbox.startsWith(BoundingBoxFieldPrefix) && bbox.endsWith(BoundingBoxFieldSuffix)) {
98         Some(BoundingBoxField(bbox.substring(BoundingBoxFieldPrefix.length, bbox.length - BoundingBoxFieldSuffix.length), bbox))
99       } else {
100         None
101       }
102     }
103 
104     /**
105      * The parquet schema for a bbox field
106      *
107      * @param bbox field name
108      * @return
109      */
110     def schema(bbox: String, fieldIds: AtomicInteger, nestedFieldIds: AtomicInteger): GroupType = {
111       Types.optionalGroup()
112         .id(fieldIds.getAndIncrement())
113         .required(PrimitiveTypeName.FLOAT).id(nestedFieldIds.getAndIncrement()).named(BoundingBoxField.XMin)
114         .required(PrimitiveTypeName.FLOAT).id(nestedFieldIds.getAndIncrement()).named(BoundingBoxField.YMin)
115         .required(PrimitiveTypeName.FLOAT).id(nestedFieldIds.getAndIncrement()).named(BoundingBoxField.XMax)
116         .required(PrimitiveTypeName.FLOAT).id(nestedFieldIds.getAndIncrement()).named(BoundingBoxField.YMax)
117         .named(bbox)
118     }
119 
120     /**
121      * The iceberg schema for a bbox field
122      *
123      * @param bbox field name
124      * @return
125      */
126     def icebergSchema(bbox: String, fieldIds: AtomicInteger, nestedFieldIds: AtomicInteger): NestedField = {
127       NestedField.optional(bbox).withId(fieldIds.getAndIncrement()).ofType(StructType.of(
128         NestedField.required(BoundingBoxField.XMin).withId(nestedFieldIds.getAndIncrement()).ofType(FloatType.get()).build(),
129         NestedField.required(BoundingBoxField.YMin).withId(nestedFieldIds.getAndIncrement()).ofType(FloatType.get()).build(),
130         NestedField.required(BoundingBoxField.XMax).withId(nestedFieldIds.getAndIncrement()).ofType(FloatType.get()).build(),
131         NestedField.required(BoundingBoxField.YMax).withId(nestedFieldIds.getAndIncrement()).ofType(FloatType.get()).build(),
132       )).build()
133     }
134   }

154     def apply(name: String): GeometryEncoding = {
155       Seq(GeoParquetNative, GeoParquetWkb).find(_.toString.equalsIgnoreCase(name)).getOrElse {
156         if (name.equalsIgnoreCase("GeoMesaV0") || name.equalsIgnoreCase("GeoMesaV1")) {
157           throw new UnsupportedOperationException(s"Geometry encoding '$name' is not longer supported")
158         }
159         throw new IllegalArgumentException(s"No geometry encoding defined for '$name'")
160       }
161     }

163     case object GeoParquetNative extends GeometryEncoding {
164       override def schema(binding: ObjectType, fieldIds: AtomicInteger, nestedFieldIds: AtomicInteger): Types.Builder[_, _ <: Type] = {
165         binding match {
166           case ObjectType.POINT =>
167             Types.optionalGroup().id(fieldIds.getAndIncrement())
168               .required(PrimitiveTypeName.DOUBLE).id(nestedFieldIds.getAndIncrement()).named(GeometryColumnX)
169               .required(PrimitiveTypeName.DOUBLE).id(nestedFieldIds.getAndIncrement()).named(GeometryColumnY)
170 
171           case ObjectType.LINESTRING =>
172             // TODO do we need multiple levels of nesting field ids???
173             Types.optionalList().id(fieldIds.getAndIncrement())
174               .requiredGroupElement().id(nestedFieldIds.getAndIncrement()) // the coordinate
175               .required(PrimitiveTypeName.DOUBLE).id(nestedFieldIds.getAndIncrement()).named(GeometryColumnX)
176               .required(PrimitiveTypeName.DOUBLE).id(nestedFieldIds.getAndIncrement()).named(GeometryColumnY)
177 
178           case ObjectType.MULTIPOINT =>
179             Types.optionalList.id(fieldIds.getAndIncrement())
180               .requiredGroupElement().id(nestedFieldIds.getAndIncrement()) // the coordinate
181               .required(PrimitiveTypeName.DOUBLE).id(nestedFieldIds.getAndIncrement()).named(GeometryColumnX)
182               .required(PrimitiveTypeName.DOUBLE).id(nestedFieldIds.getAndIncrement()).named(GeometryColumnY)
183 
184           case ObjectType.POLYGON =>
185             Types.optionalList.id(fieldIds.getAndIncrement())
186               .requiredListElement().id(nestedFieldIds.getAndIncrement()) // the coordinates of one ring
187               .requiredGroupElement().id(nestedFieldIds.getAndIncrement()) // the coordinate
188               .required(PrimitiveTypeName.DOUBLE).id(nestedFieldIds.getAndIncrement()).named(GeometryColumnX)
189               .required(PrimitiveTypeName.DOUBLE).id(nestedFieldIds.getAndIncrement()).named(GeometryColumnY)
190 
191           case ObjectType.MULTILINESTRING =>
192             Types.optionalList.id(fieldIds.getAndIncrement())
193               .requiredListElement().id(nestedFieldIds.getAndIncrement()) // the coordinates of one line string
194               .requiredGroupElement().id(nestedFieldIds.getAndIncrement()) // the coordinate
195               .required(PrimitiveTypeName.DOUBLE).id(nestedFieldIds.getAndIncrement()).named(GeometryColumnX)
196               .required(PrimitiveTypeName.DOUBLE).id(nestedFieldIds.getAndIncrement()).named(GeometryColumnY)
197 
198           case ObjectType.MULTIPOLYGON =>
199             Types.optionalList.id(fieldIds.getAndIncrement())
200               .requiredListElement().id(nestedFieldIds.getAndIncrement()) // the rings of the MultiPolygon
201               .requiredListElement().id(nestedFieldIds.getAndIncrement()) // the coordinates of one ring
202               .requiredGroupElement().id(nestedFieldIds.getAndIncrement()) // the coordinate
203               .required(PrimitiveTypeName.DOUBLE).id(nestedFieldIds.getAndIncrement()).named(GeometryColumnX)
204               .required(PrimitiveTypeName.DOUBLE).id(nestedFieldIds.getAndIncrement()).named(GeometryColumnY)
205 
206           case ObjectType.GEOMETRY_COLLECTION =>
207             Types.primitive(PrimitiveTypeName.BINARY, Repetition.OPTIONAL).id(fieldIds.getAndIncrement())
208 
209           case ObjectType.GEOMETRY =>
210             Types.primitive(PrimitiveTypeName.BINARY, Repetition.OPTIONAL).id(fieldIds.getAndIncrement())
211         }
212       }
213     }
