30 /**
31  * Parquet writer
32  *
33  * @param file file to write
34  * @param conf configuration, must include the feature type encoded according to `SimpleFeatureParquetSchema`
35  * @param observer any observers
36  */
37 class ParquetFileSystemWriter(
38     fs: ObjectStore,
39     conf: Map[String, String],
40     sft: SimpleFeatureType,
41     file: URI,
42     observer: FileSystemObserver = NoOpObserver
43   ) extends FileSystemWriter {
44 
45   import scala.collection.JavaConverters._
46 
47   private val parquetConf = new PlainParquetConfiguration(conf.asJava)
48   SimpleFeatureParquetSchema.setSft(parquetConf, sft)
49 
50   private val writer = ParquetFileSystemWriter.builder(fs, file, parquetConf).build()
51 
52   def getDataSize: Long = writer.getDataSize
53 
54   override def write(f: SimpleFeature): Unit = {
55     writer.write(f)
56     observer(f)
57   }
58 
59   override def flush(): Unit = observer.flush()
60 
61   override def close(): Unit = CloseQuietly(Seq(writer, observer)).foreach(e => throw e)
62 }
63 
64 object ParquetFileSystemWriter extends LazyLogging {
65 
66   /**
67    * Create a new configurable writer
68    *
69    * @param fs object store
70    * @param path file path
71    * @param conf write configuration
72    * @return
73    */
74   def builder(fs: ObjectStore, path: URI, conf: ParquetConfiguration): Builder = {
75     val codec = CompressionCodecName.fromConf(conf.get("parquet.compression", "ZSTD"))
76     logger.debug(s"Using Parquet Compression codec ${codec.name()}")
77 
78     val file = outputFile(fs, path)
79     new Builder(file)
80       .withConf(conf)
81       .withCompressionCodec(codec)
82       .withDictionaryEncoding(true)
83       .withDictionaryPageSize(ParquetWriter.DEFAULT_PAGE_SIZE)
84       .withMaxPaddingSize(ParquetWriter.MAX_PADDING_SIZE_DEFAULT)
85       .withPageSize(ParquetWriter.DEFAULT_PAGE_SIZE)
86       .withValidation(false)
87       .withWriteMode(ParquetFileWriter.Mode.OVERWRITE)
88       .withWriterVersion(ParquetProperties.WriterVersion.PARQUET_2_0)
89       .withRowGroupSize(8L*1024*1024)
90   }
91 
92   /**
93    * Get an output file compatible with the parquet api
94    *
95    * @param fs object store
96    * @param path file path
97    * @return
98    */
99   def outputFile(fs: ObjectStore, path: URI): OutputFile = fs match {
100     case _: LocalObjectStore => new LocalOutputFileWithParent(Path.of(path))
101     case s3: S3ObjectStore => new S3OutputFile(s3, path)
102     case _ => throw new UnsupportedOperationException(s"No file implementation for scheme ${fs.scheme}")
103   }
104 
105   class Builder(file: OutputFile) extends ParquetWriter.Builder[SimpleFeature, Builder](file) {
106     override def self(): Builder = this
107     override protected def getWriteSupport(conf: Configuration): WriteSupport[SimpleFeature] =
108       new SimpleFeatureWriteSupport()
109     override protected def getWriteSupport(conf: ParquetConfiguration): WriteSupport[SimpleFeature] =
110       new SimpleFeatureWriteSupport()
111   }
112 
113   private class LocalOutputFileWithParent(file: Path) extends LocalOutputFile(file) {
114     override def create(blockSize: Long): PositionOutputStream = {
115       Option(file.toFile.getParentFile).foreach(_.mkdirs())
116       super.create(blockSize)
117     }
118     override def createOrOverwrite(blockSize: Long): PositionOutputStream = {
119       Option(file.toFile.getParentFile).foreach(_.mkdirs())
120       super.createOrOverwrite(blockSize)
121     }
122   }
123 }
