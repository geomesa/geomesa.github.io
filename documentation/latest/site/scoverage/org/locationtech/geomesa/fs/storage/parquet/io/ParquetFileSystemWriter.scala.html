31 /**
32  * Parquet writer
33  *
34  * @param file file to write
35  * @param conf configuration, must include the feature type encoded according to `SimpleFeatureParquetSchema`
36  * @param observer any observers
37  */
38 class ParquetFileSystemWriter(
39     fs: ObjectStore,
40     conf: Map[String, String],
41     sft: SimpleFeatureType,
42     file: URI,
43     observer: FileSystemObserver = NoOpObserver
44   ) extends FileSystemWriter {
45 
46   import scala.collection.JavaConverters._
47 
48   private val parquetConf = new PlainParquetConfiguration(conf.asJava)
49   SimpleFeatureParquetSchema.setSft(parquetConf, sft)
50 
51   private val writer = ParquetFileSystemWriter.builder(fs, file, parquetConf).build()
52   @volatile
53   private var closed = false
54 
55   override def size: Long = if (closed) { fs.size(file) }  else { writer.getDataSize }
56 
57   override def write(f: SimpleFeature): Unit = {
58     writer.write(f)
59     observer(f)
60   }
61 
62   override def flush(): Unit = observer.flush()
63 
64   override def close(): Unit = {
65     closed = true
66     CloseQuietly(Seq(writer, observer)).foreach(e => throw e)
67   }
68 }

70 object ParquetFileSystemWriter extends LazyLogging {
71 
72   /**
73    * Create a new configurable writer
74    *
75    * @param fs object store
76    * @param path file path
77    * @param conf write configuration
78    * @return
79    */
80   def builder(fs: ObjectStore, path: URI, conf: ParquetConfiguration): Builder = {
81     val codec = CompressionCodecName.fromConf(conf.get("parquet.compression", "ZSTD"))
82     logger.debug(s"Using Parquet Compression codec ${codec.name()}")
83 
84     val file = outputFile(fs, path)
85     new Builder(file)
86       .withConf(conf)
87       .withCompressionCodec(codec)
88       .withDictionaryEncoding(true)
89       .withDictionaryPageSize(ParquetWriter.DEFAULT_PAGE_SIZE)
90       .withMaxPaddingSize(ParquetWriter.MAX_PADDING_SIZE_DEFAULT)
91       .withPageSize(ParquetWriter.DEFAULT_PAGE_SIZE)
92       .withValidation(false)
93       .withWriteMode(ParquetFileWriter.Mode.OVERWRITE)
94       .withWriterVersion(ParquetProperties.WriterVersion.PARQUET_2_0)
95       .withRowGroupSize(8L*1024*1024)
96   }
97 
98   /**
99    * Get an output file compatible with the parquet api
100    *
101    * @param fs object store
102    * @param path file path
103    * @return
104    */
105   def outputFile(fs: ObjectStore, path: URI): OutputFile = fs match {
106     case _: LocalObjectStore => new LocalOutputFileWithParent(Path.of(path))
107     case s3: S3ObjectStore => new S3OutputFile(s3, path)
108     case _ => throw new UnsupportedOperationException(s"No file implementation for scheme ${fs.scheme}")
109   }
110 
111   class Builder(file: OutputFile) extends ParquetWriter.Builder[SimpleFeature, Builder](file) {
112     override def self(): Builder = this
113     override protected def getWriteSupport(conf: Configuration): WriteSupport[SimpleFeature] =
114       new SimpleFeatureWriteSupport()
115     override protected def getWriteSupport(conf: ParquetConfiguration): WriteSupport[SimpleFeature] =
116       new SimpleFeatureWriteSupport()
117   }
118 
119   private class LocalOutputFileWithParent(file: Path) extends LocalOutputFile(file) {
120     override def create(blockSize: Long): PositionOutputStream = {
121       Option(file.toFile.getParentFile).foreach(_.mkdirs())
122       super.create(blockSize)
123     }
124     override def createOrOverwrite(blockSize: Long): PositionOutputStream = {
125       Option(file.toFile.getParentFile).foreach(_.mkdirs())
126       super.createOrOverwrite(blockSize)
127     }
128   }
129 }
