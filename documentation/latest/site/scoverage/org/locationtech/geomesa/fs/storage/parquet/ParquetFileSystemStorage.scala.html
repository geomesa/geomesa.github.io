41 class ParquetFileSystemStorage(context: FileSystemContext, metadata: StorageMetadata)
42     extends AbstractFileSystemStorage(context, metadata, ParquetFileSystemStorage.FileExtension) {
43 
44   import scala.collection.JavaConverters._
45 
46   private val authProvider: AuthorizationsProvider =
47     AuthUtils.getProvider(
48       Option(context.conf.get(AuthProviderParam.key)).map(p => AuthProviderParam.key -> p).toMap.asJava,
49       context.conf.get(AuthsParam.key, "").split(",").toSeq.filter(_.nonEmpty)
50     )
51 
52   override val encoding: String = ParquetFileSystemStorage.Encoding
53 
54   override protected def createWriter(file: Path, partition: Partition, observer: FileSystemObserver): FileSystemWriter = {
55     val conf = new Configuration(context.conf)
56     SimpleFeatureParquetSchema.setSft(conf, metadata.sft)
57     conf.set(SimpleFeatureParquetSchema.PartitionKey, partition.toString)
58 
59     val observers =
60       if (FileValidationEnabled.toBoolean.get) {
61         CompositeObserver(Seq(observer, FileValidationObserver(file)))
62       } else {
63         observer
64       }
65     new ParquetFileSystemWriter(file, conf, observers)
66   }
67 
68   override protected def createReader(
69       filter: Option[Filter],
70       transform: Option[(String, SimpleFeatureType)]): FileSystemPathReader = {
71     // readSft has all the fields needed for filtering and return
72     val ReadSchema(readSft, readTransform) = ReadSchema(metadata.sft, filter, transform)
73     val ReadFilter(fc, residualFilter) = ReadFilter(readSft, filter)
74     val parquetFilter = fc.map(FilterCompat.get).getOrElse(FilterCompat.NOOP)
75     val gtFilter = residualFilter.map(FastFilterFactory.optimize(readSft, _))
76     val visFilter = VisibilityUtils.visible(authProvider)
77 
78     logger.debug(
79       s"    Parquet filter: ${parquetFilter match { case f: FilterPredicateCompat => f.getFilterPredicate; case f => f }} " +
80         s"and modified gt filter: ${gtFilter.getOrElse(Filter.INCLUDE)}")
81 
82     // WARNING it is important to create a new conf per query
83     // because we communicate the transform SFT set here
84     // with the init() method on SimpleFeatureReadSupport via
85     // the parquet api. Thus we need to deep copy conf objects
86     val conf = new Configuration(context.conf)
87     StorageConfiguration.setSft(conf, readSft)
88 
89     new ParquetFileSystemReader(conf, context.root, readSft, parquetFilter, gtFilter, visFilter, readTransform)
90   }
91 }

100   /**
101    * Validate a file by reading it back
102    *
103    * @param file file to validate
104    */
105   case class FileValidationObserver(file: Path) extends FileSystemObserver {
106     override def apply(feature: SimpleFeature): Unit = {}
107     override def flush(): Unit = {}
108     override def close(): Unit = {
109       try {
110         WithClose(ParquetReader.builder(new GroupReadSupport(), file).build()) { reader =>
111           var record = reader.read()
112           while (record != null) {
113             // Process the record
114             record = reader.read()
115           }
116           logger.trace(s"$file is a valid Parquet file")
117         }
118       } catch {
119         case NonFatal(e) => throw new RuntimeException(s"File appears to be corrupted: $file", e)
120       }
121     }
122   }
