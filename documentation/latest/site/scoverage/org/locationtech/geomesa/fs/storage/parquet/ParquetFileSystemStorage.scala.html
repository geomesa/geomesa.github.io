38 class ParquetFileSystemStorage(context: FileSystemContext, metadata: StorageMetadata)
39     extends FileSystemStorage(context, metadata, ParquetFileSystemStorage.FileExtension) {
40 
41   import scala.collection.JavaConverters._
42 
43   private val authProvider: AuthorizationsProvider =
44     AuthUtils.getProvider(
45       context.conf.get(AuthProviderParam.key).map(p => AuthProviderParam.key -> p).toMap.asJava,
46       context.conf.getOrElse(AuthsParam.key, "").split(",").toSeq.filter(_.nonEmpty)
47     )
48 
49   override val encoding: String = ParquetFileSystemStorage.Encoding
50 
51   override protected def createWriter(file: URI, partition: Partition, observer: FileSystemObserver): FileSystemWriter = {
52     val conf = context.conf ++ Map(SimpleFeatureParquetSchema.PartitionKey -> partition.toString)
53 
54     val observers =
55       if (FileValidationEnabled.toBoolean.get) {
56         CompositeObserver(Seq(observer, FileValidationObserver(file)))
57       } else {
58         observer
59       }
60     new ParquetFileSystemWriter(fs, conf, metadata.sft, file, observers)
61   }
62 
63   override protected def createReader(
64       filter: Option[Filter],
65       transform: Option[(String, SimpleFeatureType)]): FileSystemPathReader = {
66     // readSchema has all the fields needed for filtering and return
67     val readSchema = ReadSchema(metadata.sft, filter, transform)
68     val readSft = readSchema.read.getOrElse(metadata.sft)
69     val ReadFilter(fc, residualFilter) = ReadFilter(readSft, filter)
70     val parquetFilter = fc.map(FilterCompat.get).getOrElse(FilterCompat.NOOP)
71     val gtFilter = residualFilter.map(FastFilterFactory.optimize(readSft, _))
72     val visFilter = VisibilityUtils.visible(authProvider)
73 
74     logger.debug(
75       s"    Parquet filter: ${parquetFilter match { case f: FilterPredicateCompat => f.getFilterPredicate; case f => f }} " +
76         s"and modified gt filter: ${gtFilter.getOrElse(Filter.INCLUDE)}")
77 
78     new ParquetFileSystemReader(fs, context, metadata.sft, readSchema.read, parquetFilter, gtFilter, visFilter, readSchema.transform)
79   }
80 }

89   /**
90    * Validate a file by reading it back
91    *
92    * @param file file to validate
93    */
94   case class FileValidationObserver(file: URI) extends FileSystemObserver {
95     override def apply(feature: SimpleFeature): Unit = {}
96     override def flush(): Unit = {}
97     override def close(): Unit = {
98       try {
99         WithClose(ParquetReader.builder(new GroupReadSupport(), new Path(file)).build()) { reader =>
100           var record = reader.read()
101           while (record != null) {
102             // Process the record
103             record = reader.read()
104           }
105           logger.trace(s"$file is a valid Parquet file")
106         }
107       } catch {
108         case NonFatal(e) => throw new RuntimeException(s"File appears to be corrupted: $file", e)
109       }
110     }
111   }
