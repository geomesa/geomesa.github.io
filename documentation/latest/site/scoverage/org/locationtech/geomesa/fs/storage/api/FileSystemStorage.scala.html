20 /**
21   * Persists simple features to a file system and provides query access. Storage implementations are fairly
22   * 'lightweight', in that all state is captured in the metadata instance
23   */
24 trait FileSystemStorage extends Compactable with Closeable {
25 
26   /**
27    * The file encoding used by this storage
28    *
29    * @return encoding
30    */
31   def encoding: String
32 
33   /**
34     * Handle to the file context, root path and configuration
35     *
36     * @return
37     */
38   def context: FileSystemContext
39 
40   /**
41     * The metadata for this storage instance
42     *
43     * @return metadata
44     */
45   def metadata: StorageMetadata
46 
47   /**
48     * Get a reader for all relevant partitions
49     *
50     * @param query query
51     * @param threads suggested threads used for reading data files
52     * @return reader
53     */
54   def getReader(query: Query, threads: Int = 1): CloseableFeatureIterator
55 
56   /**
57     * Get a writer for a given partition. This method is thread-safe and can be called multiple times,
58     * although this can result in multiple data files.
59     *
60     * @param partition partitions
61     * @return writer
62     */
63   def getWriter(partition: Partition): FileSystemWriter
64 
65   /**
66     * Gets a modifying writer. This method is thread-safe and can be called multiple times,
67     * although if a feature is modified multiple times concurrently, the last update 'wins'.
68     * There is no guarantee that any concurrent modifications will be reflected in the returned
69     * writer.
70     *
71     * @param filter the filter used to select features for modification
72     * @param threads suggested threads used for reading data files
73     * @return
74     */
75   def getWriter(filter: Filter, threads: Int = 1): FileSystemUpdateWriter
76 
77   /**
78    * Register a new file with this storage instance. The file must already be in a compatible format.
79    *
80    * @param file file to register
81    * @return registered file
82    */
83   def register(file: Path): StorageFile
84 
85   override def close(): Unit = metadata.close()
86 }

125   }
