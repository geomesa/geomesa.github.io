30 /**
31  * StorageMetadata implementation that stores all file information in a single JSON file.
32  * Uses HDFS lock files for cross-JVM write locking and synchronization for intra-JVM locking.
33  *
34  * @param fs file system
35  * @param directory metadata root path
36  * @param meta basic metadata config
37  */
38 class FileBasedMetadata(
39     private val fs: FileSystem,
40     val directory: Path,
41     private val meta: Metadata,
42   ) extends StorageMetadata with CachedMetadata with LazyLogging {
43 
44   import FileBasedMetadata._
45 
46   import scala.collection.JavaConverters._
47 
48   private val kvs = new ConcurrentHashMap[String, String](meta.config.asJava)
49   private val encodedTypeName = StringSerialization.alphaNumericSafeString(meta.sft.getTypeName)
50   private val metadataFilePath = new Path(directory, s"$encodedTypeName.json")
51   private val filesFilePath = new Path(directory, s".${encodedTypeName}_files.json")
52   private val lockFilePath = new Path(directory, s".${encodedTypeName}_files.json.lock")
53 
54   override val `type`: String = FileBasedMetadata.MetadataType
55 
56   override val sft: SimpleFeatureType = meta.sft
57   override val schemes: Set[PartitionScheme] = meta.partitions.map(PartitionSchemeFactory.load(sft, _)).toSet
58 
59   filesCache.refresh(BoxedUnit.UNIT) // kick off the initial load asynchronously
60 
61   override def get(key: String): Option[String] = Option(kvs.get(key))
62 
63   // note: this isn't synchronized across jvms
64   override def set(key: String, value: String): Unit = FileBasedMetadata.synchronized {
65     kvs.put(key, value)
66     WithClose(fs.create(metadataFilePath, true)) { out =>
67       MetadataSerialization.serialize(out, meta.copy(config = kvs.asScala.toMap))
68       out.hflush()
69       out.hsync()
70     }
71   }
72 
73   override def addFile(file: StorageFile): Unit = {
74     modifyFiles { files =>
75       // remove any existing file with the same path and add the new one
76       (files.filterNot(_.file == file.file) :+ file).sortBy(_.timestamp)(Ordering.Long.reverse)
77     }
78   }
79 
80   override def removeFile(file: StorageFile): Unit = {
81     modifyFiles { files =>
82       files.filterNot(_.file == file.file)
83     }
84   }
85 
86   override def replaceFiles(existing: Seq[StorageFile], replacements: Seq[StorageFile]): Unit = {
87     val existingFiles = existing.map(_.file)
88     modifyFiles { files =>
89       files.filterNot(f => existingFiles.contains(f.file)) ++ replacements
90     }
91   }
92 
93   /**
94    * Load files from the JSON file
95    */
96   override protected def buildFileList(): Seq[StorageFile] = {
97     try {
98       if (PathCache.exists(fs, filesFilePath) || PathCache.exists(fs, filesFilePath, reload = true)) {
99         WithClose(fs.open(filesFilePath)) { in =>
100           val listType = new TypeToken[java.util.List[StorageFile]]() {}.getType
101           gson.fromJson[java.util.List[StorageFile]](new InputStreamReader(in, StandardCharsets.UTF_8), listType).asScala.toSeq
102         }
103       } else {
104         Seq.empty
105       }
106     } catch {
107       case NonFatal(e) =>
108         logger.warn(s"Error loading files from $filesFilePath", e)
109         Seq.empty
110     }
111   }
112 
113   /**
114    * Modify files with proper locking using HDFS lock files
115    */
116   private def modifyFiles(fn: Seq[StorageFile] => Seq[StorageFile]): Unit = FileBasedMetadata.synchronized {
117     // acquire lock by creating a lock file atomically
118     var lockAcquired = false
119     var retries = 0
120 
121     val maxAttempts = MaxLockRetries.toInt.get
122 
123     while (!lockAcquired && retries < maxAttempts) {
124       try {
125         // try to create lock file with overwrite=false for atomicity
126         WithClose(fs.create(lockFilePath, false)) { out =>
127           // write lock info for debugging - hostname + timestamp
128           val lockInfo = s"${java.net.InetAddress.getLocalHost.getHostName}:${System.currentTimeMillis()}"
129           out.write(lockInfo.getBytes(StandardCharsets.UTF_8))
130           out.hflush()
131           out.hsync()
132         }
133         lockAcquired = true
134       } catch {
135         case _: org.apache.hadoop.fs.FileAlreadyExistsException =>
136           // lock file exists, check if it's stale
137           retries += 1
138           if (isLockStale(lockFilePath)) {
139             // remove stale lock and retry
140             try {
141               fs.delete(lockFilePath, false)
142             } catch {
143               case NonFatal(_) => // ignore, will retry
144             }
145           } else {
146             // wait and retry
147             Thread.sleep(LockRetryDelay.toMillis.get)
148           }
149         case NonFatal(e) =>
150           throw new RuntimeException(s"Failed to acquire lock at $lockFilePath", e)
151       }
152     }
153 
154     if (!lockAcquired) {
155       throw new RuntimeException(s"Failed to acquire lock after $MaxLockRetries retries")
156     }
157 
158     try {
159       // reload from disk to get latest state
160       val currentFiles = buildFileList()
161 
162       // apply the modification
163       val updatedFiles = fn(currentFiles)
164 
165       // write back to disk
166       WithClose(fs.create(filesFilePath, true)) { out =>
167         WithClose(new OutputStreamWriter(out, StandardCharsets.UTF_8)) { writer =>
168           gson.toJson(updatedFiles.asJava, writer)
169           writer.flush()
170           out.hflush()
171           out.hsync()
172         }
173       }
174 
175       // update cache
176       filesCache.put(BoxedUnit.UNIT, updatedFiles)
177 
178       PathCache.register(fs, filesFilePath)
179     } finally {
180       // release lock by deleting lock file
181       try {
182         fs.delete(lockFilePath, false)
183       } catch {
184         case NonFatal(e) => logger.warn(s"Failed to release lock at $lockFilePath", e)
185       }
186     }
187   }
188 
189   /**
190    * Check if a lock file is stale (older than lock timeout)
191    */
192   private def isLockStale(lockPath: Path): Boolean = {
193     try {
194       val status = fs.getFileStatus(lockPath)
195       val age = System.currentTimeMillis() - status.getModificationTime
196       age > LockTimeout.toMillis.get
197     } catch {
198       case NonFatal(_) => true // if we can't read it, consider it stale
199     }
200   }
201 }
