23 /**
24  * Attribute partitioning scheme, supporting optional bucketing
25  *
26  * @param attribute attribute being partitioned
27  * @param index index in the feature type of the attribute
28  * @param nullValue null value placeholder (type dependent)
29  * @param bucketing optional bucketing
30  */
31 abstract class AttributeScheme[T: ClassTag](val attribute: String, index: Int, nullValue: T, val bucketing: Option[Bucketing[T]])
32     extends PartitionScheme {
33 
34   override val name: String = {
35     val opts = new StringBuilder(s"${AttributeScheme.Name}:attribute=$attribute")
36     bucketing.foreach(b => opts.append(':').append(b.encoded))
37     opts.toString()
38   }
39 
40   private val default = PartitionKey(name, toPartition(nullValue))
41 
42   override def getPartition(feature: SimpleFeature): PartitionKey = {
43     val value = feature.getAttribute(index)
44     if (value == null) {
45       return default
46     }
47     val partition = toPartition(value.asInstanceOf[T])
48     PartitionKey(name, partition)
49   }
50 
51   override def spec(b: PartitionSpec.Builder): PartitionSpec.Builder = bucketing match {
52     case None => b.identity(ColumnName.encode(attribute))
53     case Some(bucket) => b.truncate(ColumnName.encode(attribute), bucket.width)
54   }
55 
56   override def getPartition(partition: StructLike, i: Int): PartitionKey =
57     PartitionKey(name, partition.get(i, implicitly[ClassTag[T]].runtimeClass).toString)
58 
59   private def toPartition(value: T): String = {
60     bucketing match {
61       case None => value.toString
62       case Some(b) => b(value).toString
63     }
64   }
65 }

73   override def load(sft: SimpleFeatureType, scheme: String): Option[PartitionScheme] = {
74     val opts = SchemeOpts(scheme)
75     if (opts.name != Name) { None } else {
76       val attribute = opts.getSingle("attribute").orNull
77       require(attribute != null, s"Attribute scheme requires an attribute to be specified with 'attribute=<attribute>'")
78       val index = attributeIndex(sft, attribute)
79       val binding = sft.getDescriptor(index).getType.getBinding
80       require(AttributeIndexKey.encodable(binding), s"Invalid type binding '${binding.getName}' of attribute '$attribute'")
81 
82       val width = opts.getSingle("width").map(w => WidthBucketing(w.toInt))
83       val divisor = opts.getSingle("divisor").map(_.toInt)
84       val scale = opts.getSingle("scale").map(_.toInt)
85 
86       val isString = classOf[String].isAssignableFrom(binding)
87       val isWholeNumber = binding == classOf[Integer] || binding == classOf[java.lang.Long]
88       val isDecimalNumber = binding == classOf[java.lang.Float] || binding == classOf[java.lang.Double]
89 
90       if (width.isDefined && !isString) {
91         throw new IllegalArgumentException(
92           s"'width' option is only supported for String-type attributes, not ${binding.getSimpleName}")
93       } else if (divisor.isDefined && !isWholeNumber) {
94         throw new IllegalArgumentException(
95           s"'divisor' option is only supported for Integer and Long-type attributes, not ${binding.getSimpleName}")
96       } else if (scale.isDefined && !isDecimalNumber) {
97         throw new IllegalArgumentException(
98           s"'scale' option is only supported for Float and Double-type attributes, not ${binding.getSimpleName}")
99       }
100 
101       if (opts.getMulti("allow").nonEmpty) {
102         throw new IllegalArgumentException("`allow` option is no longer supported for attribute schemes")
103       } else if (opts.getSingle("default").isDefined) {
104         throw new IllegalArgumentException("`default` option is no longer supported for attribute schemes")
105       }
106 
107       if (isString) {
108         width match {
109           case None => Some(new StringScheme(attribute, index))
110           case Some(w) => Some(new TruncatedStringScheme(attribute, index, w))
111         }
112       } else if (binding == classOf[Integer]) {
113         val bucketing = divisor.map(IntegralBucketing.apply[Int])
114         Some(new IntScheme(attribute, index, bucketing))
115       } else if (binding == classOf[java.lang.Long]) {
116         val bucketing = divisor.map(d => IntegralBucketing(d.toLong))
117         Some(new LongScheme(attribute, index, bucketing))
118       } else if (binding == classOf[java.lang.Float]) {
119         val bucketing = scale.map(FractionalBucketing.apply[Float])
120         Some(new FloatScheme(attribute, index, bucketing))
121       } else if (binding == classOf[java.lang.Double]) {
122         val bucketing = scale.map(FractionalBucketing.apply[Double])
123         Some(new DoubleScheme(attribute, index, bucketing))
124       } else {
125         throw new IllegalArgumentException(
126           s"Attribute scheme is not supported for type ${binding.getSimpleName} - " +
127             s"supported types are String Integer, Long, Float, and Double")
128       }
129     }
130   }

132   /**
133    * Bucketing abstraction
134    *
135    * @tparam T attribute type
136    */
137   sealed trait Bucketing[T] extends (T => T) {
138 
139     /**
140      * Encoded option, used to identify this bucketing
141      *
142      * @return
143      */
144     def encoded: String
145 
146     /**
147      * Truncation width
148      *
149      * @return
150      */
151     def width: Int
152   }
153 
154   private case class WidthBucketing(width: Int) extends Bucketing[String] {
155     override def apply(value: String): String = value.slice(0, width)
156     override def encoded: String = s"width=$width"
157   }
158 
159   private case class IntegralBucketing[T: Integral](divisor: T) extends Bucketing[T] {
160     import Integral.Implicits.infixIntegralOps
161     override def apply(value: T): T = value - (((value % divisor) + divisor) % divisor)
162     override def encoded: String = s"divisor=$divisor"
163     // TODO longs not supported by the java api
164     override def width: Int = implicitly[Integral[T]].toInt(divisor)
165   }
166 
167   // scale here refers to the number of digits to the right of the decimal place that are kept
168   private case class FractionalBucketing[T: Fractional](scale: Int) extends Bucketing[T] {
169     import Fractional.Implicits.infixFractionalOps
170     private val fractional = implicitly[Fractional[T]]
171     val scaleT: T = fractional.fromInt(math.pow(10, scale).toInt)
172     override def apply(value: T): T = fractional.fromInt(Math.floor((value * scaleT).toDouble).toInt) / scaleT
173     override def encoded: String = s"scale=$scale"
174     // TODO only decimal types support truncate but not double/float
175     override def width: Int = scale
176   }

221   private abstract class IntegralScheme[T: Integral: ClassTag](
222       attribute: String,
223       index: Int,
224       divisor: Option[IntegralBucketing[T]],
225       decoder: String => T
226     ) extends AttributeScheme[T](attribute, index, implicitly[Integral[T]].zero, divisor) {
227 
228     import Integral.Implicits.infixIntegralOps
229 
230     private val integral = implicitly[Integral[T]]
231 
232     override def getCoveringFilter(partition: PartitionKey): Filter = {
233       val value = decoder(partition.value)
234       val attr = ff.property(attribute)
235       divisor match {
236         case None => ff.equals(attr, ff.literal(value))
237         case Some(d) =>
238           val lower = ff.literal(value)
239           val upper = ff.literal(value + d.divisor)
240           ff.and(ff.greaterOrEqual(attr, lower), ff.less(attr, upper))
241       }
242     }
243 
244     override def getCoveringExpression(partition: PartitionKey): Expression = divisor match {
245       case None => Expressions.equal[T](ColumnName.encode(attribute), decoder(partition.value))
246       case Some(d) => Expressions.equal[T](Expressions.truncate[T](ColumnName.encode(attribute), integral.toInt(d.divisor)), decoder(partition.value))
247     }
248   }
