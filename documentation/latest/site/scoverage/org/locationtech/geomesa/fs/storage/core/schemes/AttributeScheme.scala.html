24 abstract class AttributeScheme[T: ClassTag](
25     attribute: String,
26     index: Int,
27     defaultValue: Option[T],
28     allowedValues: Seq[T],
29     nullValue: T,
30     bucketing: Option[Bucketing[T]],
31     lexicoder: TypeEncoder[T, String],
32   ) extends PartitionScheme {
33 
34   override val name: String = {
35     val opts = new StringBuilder(s"${AttributeScheme.Name}:attribute=$attribute")
36     bucketing.foreach(b => opts.append(':').append(b.encoded))
37     defaultValue.filter(_ != nullValue).foreach(v => opts.append(':').append(s"default=$v"))
38     allowedValues.foreach(v => opts.append(':').append(s"allow=$v"))
39     opts.toString()
40   }
41 
42   private val clas = implicitly[ClassTag[T]].runtimeClass.asInstanceOf[Class[T]]
43   private val default = PartitionKey(name, toPartition(defaultValue.getOrElse(nullValue)))
44   private val allowed = if (allowedValues.isEmpty) { Seq.empty } else { (allowedValues.map(toPartition) :+ default.value).distinct }
45 
46   override def getPartition(feature: SimpleFeature): PartitionKey = {
47     val value = feature.getAttribute(index)
48     if (value == null) {
49       return default
50     }
51     val partition = toPartition(value.asInstanceOf[T])
52     if (allowed.isEmpty || allowed.contains(partition)) {
53       PartitionKey(name, partition)
54     } else {
55       default
56     }
57   }
58 
59   override def getRangesForFilter(filter: Filter): Option[Seq[PartitionRange]] = {
60     getBounds(filter).map { bounds =>
61       val builder = new RangeBuilder()
62       bounds.foreach { bound =>
63         val lower = bound.lower.value.fold("")(toPartition)
64         val upper = exclusiveUpperBound(bound).getOrElse(AttributeScheme.UnboundedUpper)
65         builder += PartitionRange(name, lower, upper)
66       }
67       val all = builder.result()
68       if (allowed.isEmpty || all.isEmpty) { all } else {
69         allowed.collect { case v if all.exists(_.contains(v)) => PartitionRange(name, v, v + ZeroChar) }
70       }
71     }
72   }
73 
74   override def getPartitionsForFilter(filter: Filter): Option[Seq[PartitionKey]] = {
75     getBounds(filter).map { bounds =>
76       bounds.flatMap { bound =>
77         if (!bound.isBoundedBothSides) {
78           throw new IllegalArgumentException(s"Filter does not constrain partitions for scheme $name: ${ECQL.toCQL(filter)} ")
79         }
80         enumerate(bound)
81       }
82     }
83   }
84 
85   protected def enumerate(bounds: Bounds[T]): Seq[PartitionKey]
86 
87   protected def getBounds(filter: Filter): Option[Seq[Bounds[T]]] = {
88     val bounds = FilterHelper.extractAttributeBounds(filter, attribute, clas)
89     if (bounds.isEmpty) {
90       None
91     } else if (bounds.disjoint) {
92       Some(Seq.empty)
93     } else {
94       Some(bounds.values)
95     }
96   }
97 
98   protected def toPartition(value: T): String = {
99     bucketing match {
100       case None => lexicoder.encode(value)
101       case Some(b) => lexicoder.encode(b(value))
102     }
103   }
104 
105   protected def exclusiveUpperBound(bounds: Bounds[T]): Option[String] = {
106     bounds.upper.value.map { v =>
107       val encoded = toPartition(v)
108       if (bounds.upper.exclusive && bucketing.isEmpty) {
109         encoded
110       } else {
111         encoded + ZeroChar
112       }
113     }
114   }
115 }

137   override def load(sft: SimpleFeatureType, scheme: String): Option[PartitionScheme] = {
138     val opts = SchemeOpts(scheme)
139     if (opts.name != Name) { None } else {
140       val attribute = opts.getSingle("attribute").orNull
141       require(attribute != null, s"Attribute scheme requires an attribute to be specified with 'attribute=<attribute>'")
142       val index = attributeIndex(sft, attribute)
143       val binding = sft.getDescriptor(index).getType.getBinding
144       require(AttributeIndexKey.encodable(binding), s"Invalid type binding '${binding.getName}' of attribute '$attribute'")
145 
146       val width = opts.getSingle("width").map(w => WidthBucketing(w.toInt))
147       val divisor = opts.getSingle("divisor").map(_.toInt)
148       val scale = opts.getSingle("scale").map(_.toInt)
149 
150       val isString = classOf[String].isAssignableFrom(binding)
151       val isWholeNumber = binding == classOf[Integer] || binding == classOf[java.lang.Long]
152       val isDecimalNumber = binding == classOf[java.lang.Float] || binding == classOf[java.lang.Double]
153 
154       if (width.isDefined && !isString) {
155         throw new IllegalArgumentException(
156           s"'width' option is only supported for String-type attributes, not ${binding.getSimpleName}")
157       } else if (divisor.isDefined && !isWholeNumber) {
158         throw new IllegalArgumentException(
159           s"'divisor' option is only supported for Integer and Long-type attributes, not ${binding.getSimpleName}")
160       } else if (scale.isDefined && !isDecimalNumber) {
161         throw new IllegalArgumentException(
162           s"'scale' option is only supported for Float and Double-type attributes, not ${binding.getSimpleName}")
163       }
164 
165       val allowedValues = opts.getMulti("allow")
166       val defaultValue = opts.getSingle("default")
167       require(allowedValues.isEmpty || defaultValue.forall(allowedValues.contains),
168         "Default partition must be one of the allowed values")
169 
170       if (isString) {
171         Some(StringScheme(attribute, index, width, defaultValue, allowedValues))
172       } else if (binding == classOf[Integer]) {
173         val bucketing = divisor.map(IntegralBucketing.apply[Int])
174         Some(IntScheme(attribute, index, bucketing, defaultValue.map(_.toInt), allowedValues.map(_.toInt)))
175       } else if (binding == classOf[java.lang.Long]) {
176         val bucketing = divisor.map(d => IntegralBucketing(d.toLong))
177         Some(LongScheme(attribute, index, bucketing, defaultValue.map(_.toLong), allowedValues.map(_.toLong)))
178       } else if (binding == classOf[java.lang.Float]) {
179         val bucketing = scale.map(FractionalBucketing.apply[Float])
180         Some(FloatScheme(attribute, index, bucketing, defaultValue.map(_.toFloat), allowedValues.map(_.toFloat)))
181       } else if (binding == classOf[java.lang.Double]) {
182         val bucketing = scale.map(FractionalBucketing.apply[Double])
183         Some(DoubleScheme(attribute, index, bucketing, defaultValue.map(_.toDouble), allowedValues.map(_.toDouble)))
184       } else {
185         throw new IllegalArgumentException(
186           s"Attribute scheme is not supported for type ${binding.getSimpleName} - " +
187             s"supported types are String Integer, Long, Float, and Double")
188       }
189     }

218   // scale here refers to the number of digits to the right of the decimal place that are kept
219   private case class FractionalBucketing[T: Fractional](scale: Int) extends Bucketing[T] {
220     import Fractional.Implicits.infixFractionalOps

264   private abstract class IntegralScheme[T: Integral: ClassTag](
265       attribute: String,
266       index: Int,
267       divisor: Option[IntegralBucketing[T]],
268       defaultValue: Option[T],
269       allowedValues: Seq[T],
270       lexicoder: TypeEncoder[T, String],
271     ) extends AttributeScheme[T](attribute, index, defaultValue, allowedValues, implicitly[Integral[T]].zero, divisor, lexicoder) {
272 
273     import Integral.Implicits.infixIntegralOps
274 
275     private val integral = implicitly[Integral[T]]
276 
277     override def getCoveringFilter(partition: PartitionKey): Filter = {
278       val value = lexicoder.decode(partition.value)
279       val attr = ff.property(attribute)
280       divisor match {
281         case None => ff.equals(attr, ff.literal(value))
282         case Some(d) =>
283           val lower = ff.literal(value)
284           val upper = ff.literal(value + d.divisor)
285           ff.and(ff.greaterOrEqual(attr, lower), ff.less(attr, upper))
286       }
287     }
288 
289     protected def enumerate(bounds: Bounds[T]): Seq[PartitionKey] = {
290       val lower = bounds.lower.value.get
291       val delta = bounds.upper.value.get - lower
292       divisor.map(_.divisor) match {
293         case None => Seq.tabulate(delta.toInt)(i => PartitionKey(name, lexicoder.encode(lower + integral.fromInt(i))))
294         case Some(d) => Seq.tabulate((delta / d).toInt)(i => PartitionKey(name, lexicoder.encode(lower + (d * integral.fromInt(i)))))
295       }
296     }
297   }

342   private abstract class FractionalScheme[T: Fractional: ClassTag](
343       attribute: String,
344       index: Int,
345       scale: Option[FractionalBucketing[T]],
346       defaultValue: Option[T],
347       allowedValues: Seq[T],
348       lexicoder: TypeEncoder[T, String],
349     ) extends AttributeScheme[T](attribute, index, defaultValue, allowedValues, implicitly[Fractional[T]].zero, scale, lexicoder) {
350 
351     import FilterHelper.ff
352 
353     import Fractional.Implicits.infixFractionalOps
354 
355     private val fractional = implicitly[Fractional[T]]
356     private val oneBucket: Option[T] = scale.map(s => fractional.one / s.scaleT)
357 
358     override def getCoveringFilter(partition: PartitionKey): Filter = {
359       val value = lexicoder.decode(partition.value)
360       val attr = ff.property(attribute)
361       oneBucket match {
362         case None => ff.equals(attr, ff.literal(value))
363         case Some(one) =>
364           val lower = ff.literal(value)
365           val upper = ff.literal(value + one)
366           ff.and(ff.greaterOrEqual(attr, lower), ff.less(attr, upper))
367       }
368     }
369 
370     protected def enumerate(bounds: Bounds[T]): Seq[PartitionKey] = {
371       scale match {
372         case None =>
373           val lower = toPartition(bounds.lower.value.get)
374           val upper = exclusiveUpperBound(bounds).get
375           // since lexicoding is lossy, we can't increment the fractional numbers directly, instead we increment the hex string
376           (Iterator.single(lower) ++ Iterator.iterate(lower)(incrementHex).takeWhile(_ < upper)).map(PartitionKey(name, _)).toSeq
377         case Some(s) =>
378           // here we assume the scale will be enough to increment the encoded string
379           val lower = s(bounds.lower.value.get)
380           val increment = fractional.one / s.scaleT
381           val upper = exclusiveUpperBound(bounds).get
382           val tail = Iterator.iterate(lower)(_ + increment).map(toPartition).takeWhile(v => v < upper)
383           (Iterator.single(toPartition(lower)) ++ tail).map(PartitionKey(name, _)).toSeq
384       }
385     }
386 
387     // note that this is tailored to the specific hex encoding used by mango
388     // also note this will fail if we get to the max hex value
389     private def incrementHex(hex: String): String = {
390       (hex.last + 1).toChar match {
391         case d   => hex.dropRight(1) + d
392         case 'g' => incrementHex(hex.dropRight(1)) + "0"
393       }
394     }
395   }
