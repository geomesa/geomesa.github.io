126   override def compact(partition: Partition, fileSize: Option[Long], threads: Int): Unit = {
127     val target = targetSize(fileSize)
128     val files = metadata.getFiles(partition)
129     val toCompact = target match {
130       case None => files
131       case Some(t) =>
132         files.filter { f =>
133           if (fileIsSized(new Path(context.root, f.file), t)) {
134             logger.debug(s"Skipping compaction for file [${f.file}] (already target size)")
135             false
136           } else {
137             true
138           }
139         }
140     }
141 
142     if (toCompact.isEmpty) {
143       logger.debug("Skipping compaction - no files to compact")
144     } else {
145       logger.debug(s"Compacting data files: [${toCompact.map(_.file).mkString(", ")}]")
146 
147       var written = 0L
148 
149       val reader = createReader(None, None)
150       // tracks newly added files so we can register them atomically
151       val fileTracker = new FileTracker(metadata.sft, metadata.schemes)
152 
153       WithClose(createWriter(partition, StorageFileAction.Append, FileType.Compacted, target, fileTracker)) { writer =>
154         WithClose(FileSystemThreadedReader(reader, toCompact, threads)) { reader =>
155           while (reader.hasNext) {
156             val feature = reader.next()
157             writer.write(feature)
158             written += 1
159           }
160         }
161       }
162 
163       logger.debug(s"Updating metadata with new files: [${fileTracker.getFiles().map(_.file).mkString(", ")}]")
164       metadata.replaceFiles(toCompact, fileTracker.getFiles())
165 
166       logger.debug(s"Deleting old files [${toCompact.mkString(", ")}]")
167       val failures = ArrayBuffer.empty[String]
168       toCompact.foreach { file =>
169         val path = new Path(context.root, file.file)
170         if (!context.fs.delete(path, false)) {
171           failures.append(file.file)
172         }
173         PathCache.invalidate(context.fs, path)
174       }
175 
176       if (failures.nonEmpty) {
177         logger.error(s"Failed to delete some files: [${failures.mkString(", ")}]")
178       }
179 
180       logger.debug(s"Compacted $written records")
181     }
182   }

184   override def register(file: Path): StorageFile = {
185     val reader = createReader(None, None)
186     val partitions = new java.util.HashSet[Partition]()
187     val storageFile = WithClose(new StorageFileObserver(metadata.sft)) { observer =>
188       WithClose(reader.read(file)) { iter =>
189         if (!iter.hasNext) {
190           throw new RuntimeException("Could not read any features from input file")
191         }
192         iter.foreach { sf =>
193           partitions.add(Partition(metadata.schemes.map(_.getPartition(sf))))
194           observer(sf)
195         }
196       }
197       val filePath = StorageUtils.newFilePath(metadata.sft.getTypeName, FileType.Written, encoding)
198       observer.file(filePath, partitions.iterator().next(), StorageFileAction.Append)
199     }
200     if (partitions.size() != 1) {
201       throw new IllegalArgumentException(s"File corresponds to multiple partitions: ${partitions.asScala.mkString(" AND ")}")
202     }
203 
204     val destination = new Path(context.root, storageFile.file)
205     logger.debug(s"Copying $file to $destination")
206     FileUtil.copy(file.getFileSystem(context.conf), file, context.fs, destination, false, context.conf)
207 
208     metadata.addFile(storageFile)
209     storageFile
210   }

245   /**
246    * Writes files up to a given size, then starts a new file
247    *
248    * @param paths iterator of files to write
249    * @param estimator target file size estimator
250    */
251   private class ChunkedFileSystemWriter(paths: Iterator[WriterConfig], estimator: FileSizeEstimator)
252       extends FileSystemWriter {
253 
254     private var count = 0L // number of features written
255     private var total = 0L // sum size of all finished chunks
256     private var remaining = estimator.estimate(0L)
257 
258     private var path: Path = _
259     private var writer: FileSystemWriter = _
260 
261     override def write(feature: SimpleFeature): Unit = {
262       if (writer == null) {
263         val config = paths.next
264         path = config.path
265         writer = createWriter(config.path, config.partition, config.observer)
266       }
267       writer.write(feature)
268       count += 1
269       remaining -= 1
270       if (remaining == 0) {
271         writer.close()
272         writer = null
273         // adjust our estimate to account for the actual bytes written
274         total += context.fs.getFileStatus(path).getLen
275         estimator.update(total, count)
276         remaining = estimator.estimate(0L)
277       }
278     }
279 
280     override def flush(): Unit = if (writer != null) { writer.flush() }
281 
282     override def close(): Unit = {
283       if (writer != null) {
284         writer.close()
285       }
286       updateFileSize(estimator)
287     }
288   }

290   /**
291     * Update writer implementation
292     *
293     * @param reader reader for features to update
294     */
295   private class FileSystemUpdateWriterImpl(reader: CloseableFeatureIterator) extends FileSystemUpdateWriter {
296 
297     private val appenders = scala.collection.mutable.Map.empty[Set[PartitionKey], FileSystemWriter]
298     private val modifiers = scala.collection.mutable.Map.empty[Set[PartitionKey], FileSystemWriter]
299     private val deleters = scala.collection.mutable.Map.empty[Set[PartitionKey], FileSystemWriter]
300 
301     private var feature: SimpleFeature = _
302     private var partition: Set[PartitionKey] = _
303 
304     override def write(): Unit = {
305       if (feature == null) {
306         throw new IllegalArgumentException("Must call 'next' before calling 'write'")
307       }
308       val update = metadata.schemes.map(_.getPartition(feature))
309       if (update == partition) {
310         modifiers.getOrElseUpdate(update, createWriter(Partition(update), StorageFileAction.Modify, FileType.Modified)).write(feature)
311       } else {
312         // add a delete marker in the old partition, and an append in the new one, since we only track updates per-partition
313         deleters.getOrElseUpdate(partition, createWriter(Partition(partition), StorageFileAction.Delete, FileType.Deleted)).write(feature)
314         appenders.getOrElseUpdate(update, createWriter(Partition(update), StorageFileAction.Append, FileType.Written)).write(feature)
315       }
316       feature = null
317     }
318 
319     override def remove(): Unit = {
320       if (feature == null) {
321         throw new IllegalArgumentException("Must call 'next' before calling 'remove'")
322       }
323       deleters.getOrElseUpdate(partition, createWriter(Partition(partition), StorageFileAction.Delete, FileType.Deleted)).write(feature)
324       feature = null
325     }
326 
327     override def hasNext: Boolean = reader.hasNext
328 
329     override def next(): SimpleFeature = {
330       feature = reader.next() // note: our reader returns a mutable copy of the feature
331       partition = metadata.schemes.map(_.getPartition(feature))
332       feature
333     }
334 
335     override def flush(): Unit = FlushQuietly.raise(appenders.values.toSeq ++ modifiers.values ++ deleters.values)
336 
337     override def close(): Unit = CloseQuietly.raise(Seq(reader) ++ appenders.values ++ modifiers.values ++ deleters.values ++ observers)
338   }

346   class StorageFileObserver(sft: SimpleFeatureType) extends FileSystemObserver with LazyLogging {
347 
348     import scala.collection.JavaConverters._
349 
350     private var count: Long = 0L
351 
352     private val spatialBounds = sft.spatialBounds().map(_ -> new Envelope())
353 
354     private val nonSpatialBounds = sft.nonSpatialBounds().flatMap { i =>
355       val binding = sft.getDescriptor(i).getType.getBinding
356       AttributeIndexKey.TypeRegistry.getAllEncoders.asScala.find(_.resolves().isAssignableFrom(binding)) match {
357         case Some(encoder) => Some(AttributeBoundsBuilder(i, encoder.asInstanceOf[TypeEncoder[AnyRef, String]]))
358         case None =>
359           logger.warn(
360             s"Can't find an encoder for attribute ${sft.getDescriptor(i).getLocalName} of type ${binding.getSimpleName} - " +
361               "will not track bounds")
362           None
363       }
364     }
365 
366     private var sorted: Seq[(Int, Ordering[AnyRef], AnyRef)] = Range(0, sft.getAttributeCount).flatMap { i =>
367       val binding = sft.getDescriptor(i).getType.getBinding
368       if (classOf[Comparable[_]].isAssignableFrom(binding)) {
369         Some((i, Ordering.ordered[AnyRef](_.asInstanceOf[Comparable[AnyRef]]), null))
370       } else {
371         None
372       }
373     }
374 
375     /**
376      * Get the file metadata that has been gathered so far
377      *
378      * @param path file path
379      * @param partition file partition
380      * @param action file action
381      * @return
382      */
383     def file(path: String, partition: Partition, action: StorageFileAction): StorageFile = {
384       val spatial = spatialBounds.flatMap { case (i, env) => SpatialBounds(i, env) }
385       val nonSpatial = nonSpatialBounds.flatMap(_.build())
386       val sort = sorted.map(_._1)
387       StorageFile(path, partition, count, action, spatial, nonSpatial, sort)
388     }
389 
390     override def apply(feature: SimpleFeature): Unit = {
391       count += 1L
392       spatialBounds.foreach { case (i, env) =>
393         val geom = feature.getAttribute(i).asInstanceOf[Geometry]
394         if (geom != null) {
395           env.expandToInclude(geom.getEnvelopeInternal)
396         }
397       }
398       nonSpatialBounds.foreach(_.apply(feature))
399       if (sorted.nonEmpty) {
400         sorted = sorted.flatMap { case (i, ordering, last) =>
401           val next = feature.getAttribute(i)
402           if (last == null || (next != null && ordering.compare(next, last) < 0)) {
403             Some((i, ordering, next))
404           } else {
405             None
406           }
407         }
408       }
409     }
410 
411     override def flush(): Unit = {}
412     override def close(): Unit = {}
413   }
