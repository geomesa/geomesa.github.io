24   /**
25    * Enumeration of supported geometry encodings
26    */
27   sealed trait GeometryEncoding {
28 
29     /**
30      * Create a builder for a parquet geometry field
31      *
32      * @param binding geometry type
33      * @param fieldIds field id tracker
34      * @param nestedFieldIds nested field id tracker
35      * @return
36      */
37     def schema(binding: ObjectType, fieldIds: AtomicInteger, nestedFieldIds: AtomicInteger): Types.Builder[_, _ <: Type]
38   }
39 
40   object GeometryEncoding {
41 
42     def apply(name: String): GeometryEncoding = {
43       Seq(GeoParquetNative, GeoParquetWkb).find(_.toString.equalsIgnoreCase(name)).getOrElse {
44         if (name.equalsIgnoreCase("GeoMesaV0") || name.equalsIgnoreCase("GeoMesaV1")) {
45           throw new UnsupportedOperationException(s"Geometry encoding '$name' is not longer supported")
46         }
47         throw new IllegalArgumentException(s"No geometry encoding defined for '$name'")
48       }
49     }
50 
51     case object GeoParquetNative extends GeometryEncoding {
52       override def schema(binding: ObjectType, fieldIds: AtomicInteger, nestedFieldIds: AtomicInteger): Types.Builder[_, _ <: Type] = {
53         binding match {
54           case ObjectType.POINT =>
55             Types.optionalGroup().id(fieldIds.getAndIncrement())
56               .required(PrimitiveTypeName.DOUBLE).id(nestedFieldIds.getAndIncrement()).named(GeometryColumnX)
57               .required(PrimitiveTypeName.DOUBLE).id(nestedFieldIds.getAndIncrement()).named(GeometryColumnY)
58 
59           case ObjectType.LINESTRING =>
60             // TODO do we need multiple levels of nesting field ids???
61             Types.optionalList().id(fieldIds.getAndIncrement())
62               .requiredGroupElement().id(nestedFieldIds.getAndIncrement()) // the coordinate
63               .required(PrimitiveTypeName.DOUBLE).id(nestedFieldIds.getAndIncrement()).named(GeometryColumnX)
64               .required(PrimitiveTypeName.DOUBLE).id(nestedFieldIds.getAndIncrement()).named(GeometryColumnY)
65 
66           case ObjectType.MULTIPOINT =>
67             Types.optionalList.id(fieldIds.getAndIncrement())
68               .requiredGroupElement().id(nestedFieldIds.getAndIncrement()) // the coordinate
69               .required(PrimitiveTypeName.DOUBLE).id(nestedFieldIds.getAndIncrement()).named(GeometryColumnX)
70               .required(PrimitiveTypeName.DOUBLE).id(nestedFieldIds.getAndIncrement()).named(GeometryColumnY)
71 
72           case ObjectType.POLYGON =>
73             Types.optionalList.id(fieldIds.getAndIncrement())
74               .requiredListElement().id(nestedFieldIds.getAndIncrement()) // the coordinates of one ring
75               .requiredGroupElement().id(nestedFieldIds.getAndIncrement()) // the coordinate
76               .required(PrimitiveTypeName.DOUBLE).id(nestedFieldIds.getAndIncrement()).named(GeometryColumnX)
77               .required(PrimitiveTypeName.DOUBLE).id(nestedFieldIds.getAndIncrement()).named(GeometryColumnY)
78 
79           case ObjectType.MULTILINESTRING =>
80             Types.optionalList.id(fieldIds.getAndIncrement())
81               .requiredListElement().id(nestedFieldIds.getAndIncrement()) // the coordinates of one line string
82               .requiredGroupElement().id(nestedFieldIds.getAndIncrement()) // the coordinate
83               .required(PrimitiveTypeName.DOUBLE).id(nestedFieldIds.getAndIncrement()).named(GeometryColumnX)
84               .required(PrimitiveTypeName.DOUBLE).id(nestedFieldIds.getAndIncrement()).named(GeometryColumnY)
85 
86           case ObjectType.MULTIPOLYGON =>
87             Types.optionalList.id(fieldIds.getAndIncrement())
88               .requiredListElement().id(nestedFieldIds.getAndIncrement()) // the rings of the MultiPolygon
89               .requiredListElement().id(nestedFieldIds.getAndIncrement()) // the coordinates of one ring
90               .requiredGroupElement().id(nestedFieldIds.getAndIncrement()) // the coordinate
91               .required(PrimitiveTypeName.DOUBLE).id(nestedFieldIds.getAndIncrement()).named(GeometryColumnX)
92               .required(PrimitiveTypeName.DOUBLE).id(nestedFieldIds.getAndIncrement()).named(GeometryColumnY)
93 
94           case ObjectType.GEOMETRY_COLLECTION =>
95             Types.primitive(PrimitiveTypeName.BINARY, Repetition.OPTIONAL).id(fieldIds.getAndIncrement())
96 
97           case ObjectType.GEOMETRY =>
98             Types.primitive(PrimitiveTypeName.BINARY, Repetition.OPTIONAL).id(fieldIds.getAndIncrement())
99         }
100       }
101     }
102 
103     case object GeoParquetWkb extends GeometryEncoding {
104       override def schema(binding: ObjectType, fieldIds: AtomicInteger, nestedFieldIds: AtomicInteger): Types.Builder[_, _ <: Type] =
105         Types.primitive(PrimitiveTypeName.BINARY, Repetition.OPTIONAL).id(fieldIds.getAndIncrement())
106     }
107   }
