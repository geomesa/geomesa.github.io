62   import org.locationtech.geomesa.index.conf.QueryHints._
63 
64   /**
65     * Configure the input format
66     *
67     * @param conf conf
68     * @param sft simple feature type
69     * @param query query
70     */
71   def configure(conf: Configuration, sft: SimpleFeatureType, query: Query): Unit = {
72     val q = QueryRunner.configureQuery(sft, query)
73     val filter = Option(q.getFilter).filter(_ != Filter.INCLUDE)
74 
75     // Parquet read schema and final transform
76     val readSchema = ReadSchema(sft, filter, q.getHints.getTransform)
77     // push-down Parquet predicates and remaining gt-filter
78     val ReadFilter(parquetFilter, residualFilter) = ReadFilter(readSchema.read.getOrElse(sft), filter)
79 
80     // set the parquet push-down filter
81     parquetFilter.foreach(ParquetInputFormat.setFilterPredicate(conf, _))
82 
83     // set our residual filters and transforms
84     StorageConfiguration.setSft(conf, sft)
85     readSchema.read.foreach(StorageConfiguration.setReadSft(conf, _))
86     residualFilter.foreach(StorageConfiguration.setFilter(conf, _))
87     readSchema.transform.foreach(StorageConfiguration.setTransforms(conf, _))
88 
89     // need this for query planning
90     conf.set("parquet.filter.dictionary.enabled", "true")
91 
92     // @see org.apache.parquet.hadoop.ParquetInputFormat.setReadSupportClass(org.apache.hadoop.mapred.JobConf, java.lang.Class<?>)
93     conf.set(ParquetInputFormat.READ_SUPPORT_CLASS, classOf[SimpleFeatureReadSupport].getName)
94 
95     // replicates parquet input format strategy of always recursively listing directories
96     conf.set(FileInputFormat.INPUT_DIR_RECURSIVE, "true")
97   }

130   abstract class ParquetSimpleFeatureRecordReaderBase[T](
131       delegate: RecordReader[Void, SimpleFeature],
132       filter: Option[Filter]
133     ) extends RecordReader[T, SimpleFeature] {
134 
135     private var current: SimpleFeature = _
136 
137     override def initialize(split: InputSplit, context: TaskAttemptContext): Unit =
138       delegate.initialize(split, context)
139 
140     override def getProgress: Float = delegate.getProgress
141 
142     override def nextKeyValue(): Boolean = {
143       while (delegate.nextKeyValue()) {
144         current = delegate.getCurrentValue
145         if (filter.forall(_.evaluate(current))) {
146            return true
147         }
148       }
149       false
150     }
151 
152     override def getCurrentValue: SimpleFeature = current
153 
154     override def close(): Unit = delegate.close()
155   }

157   abstract class ParquetSimpleFeatureTransformRecordReaderBase[T](
158       delegate: RecordReader[Void, SimpleFeature],
159       filter: Option[Filter],
160       sft: SimpleFeatureType,
161       tsft: SimpleFeatureType,
162       tdefs: String
163     ) extends ParquetSimpleFeatureRecordReaderBase[T](delegate, filter) {
164 
165     private val current = TransformSimpleFeature(sft, tsft, tdefs)
166 
167     override def getCurrentValue: SimpleFeature = {
168       current.setFeature(super.getCurrentValue)
169       current
170     }
171   }
