32 class ParquetFileSystemReader(
33     fs: ObjectStore,
34     context: FileSystemContext,
35     sft: SimpleFeatureType,
36     readSft: Option[SimpleFeatureType],
37     parquetFilter: FilterCompat.Filter,
38     gtFilter: Option[org.geotools.api.filter.Filter],
39     visFilter: SimpleFeature => Boolean,
40     transform: Option[(String, SimpleFeatureType)]
41   ) extends FileSystemPathReader with LazyLogging {
42 
43   import scala.collection.JavaConverters._
44 
45   private val gtf = gtFilter.orNull
46 
47   private val transformFeature: SimpleFeature => SimpleFeature = transform match {
48     case None => null
49     case Some((tdefs, tsft)) =>
50       val definitions = Transforms(readSft.getOrElse(sft), tdefs).toArray
51       f => new TransformSimpleFeature(tsft, definitions, f)
52   }
53 
54   private val conf = new PlainParquetConfiguration(context.conf.asJava)
55   SimpleFeatureParquetSchema.setSft(conf, sft)
56   readSft.foreach(SimpleFeatureParquetSchema.setReadSft(conf, _))
57 
58   override def root: URI = context.root
59 
60   override def read(file: URI): CloseableIterator[SimpleFeature] = {
61     // TODO we can examine file bounds and simplify/eliminate the gtFilter
62     logger.debug(s"Opening reader for path $file")
63     new ParquetFileIterator(file)
64   }
65 
66   private class ParquetFileIterator(path: URI) extends CloseableIterator[SimpleFeature] {
67 
68     private val reader: ParquetReader[SimpleFeature] =
69       ParquetFileSystemReader.builder(fs, path).withConf(conf).withFilter(parquetFilter).build()
70 
71     private var staged: SimpleFeature = _
72 
73     override def close(): Unit = {
74       logger.debug(s"Closing reader for path $path")
75       reader.close()
76     }
77 
78     override def next(): SimpleFeature = {
79       val res = staged
80       staged = null
81       res
82     }
83 
84     @tailrec
85     override final def hasNext: Boolean = {
86       if (staged != null) { true } else {
87         val read = try { reader.read() } catch {
88           case NonFatal(e) => logger.error(s"Error reading file '$path'", e); null
89         }
90         if (read == null) {
91           false
92         } else if (visFilter(read) && (gtf == null || gtf.evaluate(read))) {
93           staged = if (transformFeature == null) { read } else { transformFeature(read) }
94           true
95         } else {
96           hasNext
97         }
98       }
99     }
100   }
101 }

103 object ParquetFileSystemReader {
104 
105   /**
106    * Create a new configurable reader
107    *
108    * @param fs object store
109    * @param path file to read
110    * @return
111    */
112   def builder(fs: ObjectStore, path: URI): Builder = new Builder(inputFile(fs, path))
113 
114   /**
115    * Get an input file compatible with the parquet api
116    *
117    * @param fs object store
118    * @param path file path
119    * @return
120    */
121   def inputFile(fs: ObjectStore, path: URI): InputFile = fs match {
122     case _: LocalObjectStore => new LocalInputFile(Path.of(path))
123     case s3: S3ObjectStore   => new S3InputFile(s3, path)
124     case _ => throw new UnsupportedOperationException(s"No file implementation for scheme ${fs.scheme}")
125   }
126 
127   class Builder(file: InputFile) extends ParquetReader.Builder[SimpleFeature](file) {
128     override protected def getReadSupport: ReadSupport[SimpleFeature] = new SimpleFeatureReadSupport()
129   }
