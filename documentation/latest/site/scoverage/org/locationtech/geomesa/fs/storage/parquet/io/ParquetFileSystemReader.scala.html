31 class ParquetFileSystemReader(
32     fs: ObjectStore,
33     context: FileSystemContext,
34     readSft: SimpleFeatureType,
35     parquetFilter: FilterCompat.Filter,
36     gtFilter: Option[org.geotools.api.filter.Filter],
37     visFilter: SimpleFeature => Boolean,
38     transform: Option[(String, SimpleFeatureType)]
39   ) extends FileSystemPathReader with LazyLogging {
40 
41   import scala.collection.JavaConverters._
42 
43   private val gtf = gtFilter.orNull
44 
45   private val transformFeature: SimpleFeature => SimpleFeature = transform match {
46     case None => null
47     case Some((tdefs, tsft)) =>
48       val definitions = Transforms(readSft, tdefs).toArray
49       f => new TransformSimpleFeature(tsft, definitions, f)
50   }
51 
52   private val conf = new PlainParquetConfiguration(context.conf.asJava)
53   SimpleFeatureParquetSchema.setSft(conf, readSft)
54 
55   override def root: URI = context.root
56 
57   override def read(file: URI): CloseableIterator[SimpleFeature] = {
58     // TODO we can examine file bounds and simplify/eliminate the gtFilter
59     logger.debug(s"Opening reader for path $file")
60     new ParquetFileIterator(file)
61   }
62 
63   private class ParquetFileIterator(path: URI) extends CloseableIterator[SimpleFeature] {
64 
65     private val reader: ParquetReader[SimpleFeature] =
66       ParquetFileSystemReader.builder(fs, path).withConf(conf).withFilter(parquetFilter).build()
67 
68     private var staged: SimpleFeature = _
69 
70     override def close(): Unit = {
71       logger.debug(s"Closing reader for path $path")
72       reader.close()
73     }
74 
75     override def next(): SimpleFeature = {
76       val res = staged
77       staged = null
78       res
79     }
80 
81     @tailrec
82     override final def hasNext: Boolean = {
83       if (staged != null) { true } else {
84         val read = try { reader.read() } catch {
85           case NonFatal(e) => logger.error(s"Error reading file '$path'", e); null
86         }
87         if (read == null) {
88           false
89         } else if (visFilter(read) && (gtf == null || gtf.evaluate(read))) {
90           staged = if (transformFeature == null) { read } else { transformFeature(read) }
91           true
92         } else {
93           hasNext
94         }
95       }
96     }
97   }
98 }
