17 class FsGetPartitionsCommand extends FsDataStoreCommand {
18 
19   override val name: String = "get-partitions"
20   override val params = new FsGetPartitionsParams
21 
22   override def execute(): Unit = withDataStore { ds =>
23     Command.user.info(s"Partitions for type ${params.featureName}:")
24     if (!params.noHeader) {
25       Command.output.info("partition\tfile_count\tfeature_count")
26     }
27     val storage = ds.storage(params.featureName)
28     storage.metadata.files().scan().groupBy(f => storage.metadata.partition(f).toString).toSeq.sortBy(_._1).foreach { case (p, files) =>
29       Command.output.info(s"$p\t${files.size}\t${files.map(_.recordCount()).sum}")
30     }
31   }
32 }
