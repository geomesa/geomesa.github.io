19   override val name: String = "get-partitions"

21 
22   override def execute(): Unit = withDataStore { ds =>
23     Command.user.info(s"Partitions for type ${params.featureName}:")
24     Command.output.info("partition\tfile_count\tfeature_count")
25     ds.storage(params.featureName).metadata.getFiles().groupBy(_.partition.toString).toSeq.sortBy(_._1).foreach { case (p, files) =>
26       Command.output.info(p + s"\t${files.size}\t${files.map(_.count).sum}")
27     }
28   }
