134   /**
135    * Register a new file with this storage instance. The file must already be in a compatible format.
136    *
137    * @param file file to register
138    * @return registered file
139    */
140   def register(file: URI): StorageFile = {
141     val reader = createReader(None, None)
142     val partitions = new java.util.HashSet[Partition]()
143     val storageFile = WithClose(new StorageFileObserver(metadata.sft)) { observer =>
144       WithClose(reader.read(file)) { iter =>
145         if (!iter.hasNext) {
146           throw new RuntimeException("Could not read any features from input file")
147         }
148         iter.foreach { sf =>
149           partitions.add(Partition(metadata.schemes.map(_.getPartition(sf))))
150           observer(sf)
151         }
152       }
153       val filePath = FileSystemStorage.newFilePath(metadata.sft.getTypeName, FileType.Written, encoding)
154       observer.file(filePath, partitions.iterator().next(), StorageFileAction.Append)
155     }
156     if (partitions.size() != 1) {
157       throw new IllegalArgumentException(s"File corresponds to multiple partitions: ${partitions.asScala.mkString(" AND ")}")
158     }
159 
160     val destination = context.root.resolve(storageFile.file)
161     logger.debug(s"Copying $file to $destination")
162     fs.copy(file, destination)
163     metadata.addFile(storageFile)
164     storageFile
165   }

167   /**
168    * Compact a partition - merge multiple data files into a single file.
169    *
170    * Care should be taken with this method. Currently, there is no guarantee for correct behavior if
171    * multiple threads or storage instances attempt to compact the same partition simultaneously.
172    *
173    * @param partition partition to compact, or all partitions
174    * @param fileSize approximate target size of files, in bytes
175    * @param threads suggested threads to use for file system operations
176    */
177   def compact(partition: Partition, fileSize: Option[Long] = None, threads: Int = 1): Unit = {
178     val target = fileSize.orElse(this.sizer.targetSize)
179     val files = metadata.getFiles(partition)
180     val toCompact = target match {
181       case None => files
182       case Some(t) =>
183         files.filter { f =>
184           if (this.sizer.fileIsSized(context.root.resolve(f.file), t)) {
185             logger.debug(s"Skipping compaction for file [${f.file}] (already target size)")
186             false
187           } else {
188             true
189           }
190         }
191     }
192 
193     if (toCompact.isEmpty) {
194       logger.debug("Skipping compaction - no files to compact")
195     } else {
196       logger.debug(s"Compacting data files: [${toCompact.map(_.file).mkString(", ")}]")
197 
198       var written = 0L
199 
200       val reader = createReader(None, None)
201       // tracks newly added files so we can register them atomically
202       val fileTracker = new FileTracker(metadata.sft, metadata.schemes)
203 
204       WithClose(createWriter(partition, StorageFileAction.Append, FileType.Compacted, target, fileTracker)) { writer =>
205         WithClose(FileSystemThreadedReader(reader, toCompact, threads)) { reader =>
206           while (reader.hasNext) {
207             val feature = reader.next()
208             writer.write(feature)
209             written += 1
210           }
211         }
212       }
213 
214       logger.debug(s"Updating metadata with new files: [${fileTracker.getFiles().map(_.file).mkString(", ")}]")
215       metadata.replaceFiles(toCompact, fileTracker.getFiles())
216 
217       logger.debug(s"Deleting old files [${toCompact.mkString(", ")}]")
218       val failures = ArrayBuffer.empty[String]
219       toCompact.foreach { file =>
220         val path = context.root.resolve(file.file)
221         if (Try(fs.delete(path)).isFailure) {
222           failures.append(file.file)
223         }
224       }
225 
226       if (failures.nonEmpty) {
227         logger.error(s"Failed to delete some files: [${failures.mkString(", ")}]")
228       }
229 
230       logger.debug(s"Compacted $written records")
231     }
232   }

276    * @param fileType file type

281   private def createWriter(
282       partition: Partition,
283       action: StorageFileAction,
284       fileType: FileType,
285       targetFileSize: Option[Long],
286       metadata: StorageMetadata): FileSystemWriter = {
287 
288     def pathAndWriter: (URI, FileSystemWriter) = {
289       val file = FileSystemStorage.newFilePath(metadata.sft.getTypeName, fileType, encoding)
290       val path = context.root.resolve(file)
291       val updateObserver = new MetadataObserver(metadata, file, partition, action)
292       val observer = if (observers.isEmpty) { updateObserver } else {
293         new CompositeObserver(observers.map(_.apply(path)).+:(updateObserver))
294       }
295       (path, createWriter(path, partition, observer))
296     }
297 
298     targetFileSize match {
299       case None => pathAndWriter._2
300       case Some(s) => new ChunkedFileSystemWriter(fs, Iterator.continually(pathAndWriter), sizer.estimator(s))
301     }

316   def newFilePath(typeName: String, fileType: FileType.FileType, ext: String): String = {
317     val filename =
318       s"${fileType}_${SafeNameRegex.replaceAllIn(typeName, "-").take(20)}_${UUID.randomUUID().toString.replaceAllLiterally("-", "")}.$ext"
319     // partitioning logic taken from Apache Iceberg: https://iceberg.apache.org/docs/nightly/aws/#object-store-file-layout
320     val hash = {
321       val bytes = filename.getBytes(StandardCharsets.UTF_8)
322       val hash = MurmurHash3.hash32x86(bytes, 0, bytes.length, 0)
323       // Integer#toBinaryString excludes leading zeros, which we want to preserve
324       Integer.toBinaryString(hash | Integer.MIN_VALUE)
325     }
326     s"${hash.substring(0, 4)}/${hash.substring(4, 8)}/${hash.substring(8, 12)}/${hash.substring(12, 20)}/$filename"
327   }

351    * Writes files up to a given size, then starts a new file
352    *
353    * @param fs file system
354    * @param writers iterator of files to write
355    * @param estimator target file size estimator
356    */
357   private class ChunkedFileSystemWriter(
358       fs: ObjectStore,
359       writers: Iterator[(URI, FileSystemWriter)],
360       estimator: UpdatingFileSizeEstimator
361     ) extends FileSystemWriter {
362 
363     private var count = 0L // number of features written
364     private var total = 0L // sum size of all finished chunks
365     private var remaining = estimator.estimate(0L)
366 
367     private var path: URI = _
368     private var writer: FileSystemWriter = _
369 
370     override def write(feature: SimpleFeature): Unit = {
371       if (writer == null) {
372         val (path, writer) = writers.next()
373         this.path = path
374         this.writer = writer
375       }
376       writer.write(feature)
377       count += 1
378       remaining -= 1
379       if (remaining == 0) {
380         writer.close()
381         writer = null
382         // adjust our estimate to account for the actual bytes written
383         total += fs.size(path)
384         estimator.update(total, count)
385         remaining = estimator.estimate(0L)
386       }
387     }
388 
389     override def flush(): Unit = if (writer != null) { writer.flush() }
390 
391     override def close(): Unit = {
392       if (writer != null) {
393         writer.close()
394       }
395       estimator.close()
396     }
397   }

404   class FileSystemUpdateWriter(
405       schemes: Set[PartitionScheme],
406       reader: CloseableFeatureIterator,
407       newWriter: (Partition, StorageFileAction) => FileSystemWriter
408     ) extends Iterator[SimpleFeature] with Closeable with Flushable {
409 
410     // TODO limit number of open writers
411 
412     private val appenders = scala.collection.mutable.Map.empty[Set[PartitionKey], FileSystemWriter]
413     private val modifiers = scala.collection.mutable.Map.empty[Set[PartitionKey], FileSystemWriter]
414     private val deleters = scala.collection.mutable.Map.empty[Set[PartitionKey], FileSystemWriter]
415 
416     private var feature: SimpleFeature = _
417     private var partition: Set[PartitionKey] = _
418 
419     /**
420      * Writes a modification to the last feature returned by `next`
421      */
422     def write(): Unit = {
423       if (feature == null) {
424         throw new IllegalArgumentException("Must call 'next' before calling 'write'")
425       }
426       val update = schemes.map(_.getPartition(feature))
427       if (update == partition) {
428         modifiers.getOrElseUpdate(update, newWriter(Partition(update), StorageFileAction.Modify)).write(feature)
429       } else {
430         // add a delete marker in the old partition, and an append in the new one, since we only track updates per-partition
431         deleters.getOrElseUpdate(partition, newWriter(Partition(partition), StorageFileAction.Delete)).write(feature)
432         appenders.getOrElseUpdate(update, newWriter(Partition(update), StorageFileAction.Append)).write(feature)
433       }
434       feature = null
435     }
436 
437     /**
438      * Deletes the last feature returned by `next`
439      */
440     def remove(): Unit = {
441       if (feature == null) {
442         throw new IllegalArgumentException("Must call 'next' before calling 'remove'")
443       }
444       deleters.getOrElseUpdate(partition, newWriter(Partition(partition), StorageFileAction.Delete)).write(feature)
445       feature = null
446     }
447 
448     override def hasNext: Boolean = reader.hasNext
449 
450     override def next(): SimpleFeature = {
451       feature = reader.next() // note: our reader returns a mutable copy of the feature
452       partition = schemes.map(_.getPartition(feature))
453       feature
454     }
455 
456     override def flush(): Unit = FlushQuietly.raise(appenders.values.toSeq ++ modifiers.values ++ deleters.values)
457 
458     override def close(): Unit = CloseQuietly.raise(Seq(reader) ++ appenders.values ++ modifiers.values ++ deleters.values)
459   }

485   class StorageFileObserver(sft: SimpleFeatureType) extends FileSystemObserver with LazyLogging {
486 
487     import scala.collection.JavaConverters._
488 
489     private var count: Long = 0L
490 
491     private val spatialBounds = sft.spatialBounds().map(_ -> new Envelope())
492 
493     private val nonSpatialBounds = sft.nonSpatialBounds().flatMap { i =>
494       val binding = sft.getDescriptor(i).getType.getBinding
495       AttributeIndexKey.TypeRegistry.getAllEncoders.asScala.find(_.resolves().isAssignableFrom(binding)) match {
496         case Some(encoder) => Some(AttributeBoundsBuilder(i, encoder.asInstanceOf[TypeEncoder[AnyRef, String]]))
497         case None =>
498           logger.warn(
499             s"Can't find an encoder for attribute ${sft.getDescriptor(i).getLocalName} of type ${binding.getSimpleName} - " +
500               "will not track bounds")
501           None
502       }
503     }
504 
505     private var sorted: Seq[(Int, Ordering[AnyRef], AnyRef)] = Range(0, sft.getAttributeCount).flatMap { i =>
506       val binding = sft.getDescriptor(i).getType.getBinding
507       if (classOf[Comparable[_]].isAssignableFrom(binding)) {
508         Some((i, Ordering.ordered[AnyRef](_.asInstanceOf[Comparable[AnyRef]]), null))
509       } else {
510         None
511       }
512     }

514     /**
515      * Get the file metadata that has been gathered so far
516      *
517      * @param path file path
518      * @param partition file partition
519      * @param action file action
520      * @return
521      */
522     def file(path: String, partition: Partition, action: StorageFileAction): StorageFile = {
523       val spatial = spatialBounds.flatMap { case (i, env) => SpatialBounds(i, env) }
524       val nonSpatial = nonSpatialBounds.flatMap(_.build())
525       val sort = sorted.map(_._1)
526       StorageFile(path, partition, count, action, spatial, nonSpatial, sort)
527     }
528 
529     override def apply(feature: SimpleFeature): Unit = {
530       count += 1L
531       spatialBounds.foreach { case (i, env) =>
532         val geom = feature.getAttribute(i).asInstanceOf[Geometry]
533         if (geom != null) {
534           env.expandToInclude(geom.getEnvelopeInternal)
535         }
536       }
537       nonSpatialBounds.foreach(_.apply(feature))
538       if (sorted.nonEmpty) {
539         sorted = sorted.flatMap { case (i, ordering, last) =>
540           val next = feature.getAttribute(i)
541           if (last == null || (next != null && ordering.compare(next, last) < 0)) {
542             Some((i, ordering, next))
543           } else {
544             None
545           }
546         }
547       }
548     }
