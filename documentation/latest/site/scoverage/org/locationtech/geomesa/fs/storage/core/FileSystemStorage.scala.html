121    * Gets a reader
122    *
123    * @param query query
124    * @param threads read threads
125    * @param forUpdate include row and file data required for positional deletes
126    * @return
127    */
128   private[core] def getReader(query: Query, threads: Int, forUpdate: Boolean): CloseableIterator[SimpleFeature] = {
129     import org.locationtech.geomesa.index.conf.QueryHints.RichHints
130 
131     val configured = QueryRunner.configureQuery(sft, query)
132     val filter = Option(configured.getFilter).getOrElse(Filter.INCLUDE)
133     val icebergFilter = IcebergFilterConverter(sft, filter)
134     val visFilter = VisibilityUtils.visible(authProvider)
135     val transform = configured.getHints.getTransform
136     val sort = configured.getHints.getSortFields
137     val max = configured.getHints.getMaxFeatures
138     val readSchema = schema.read(transform.map(_._1), icebergFilter.columns, forUpdate)
139 
140     logger.debug(s"Running query '${query.getTypeName}' ${ECQL.toCQL(filter)}")
141     logger.debug(s"  Original filter: ${ECQL.toCQL(query.getFilter)}")
142     logger.debug(s"  Push-down filter: ${icebergFilter.expression}")
143     logger.debug(s"  Client-side filter: ${icebergFilter.remainder.fold("none")(ECQL.toCQL)}")
144     logger.debug(s"  Transforms: ${transform.fold("none") { case (t, _) => if (t.isEmpty) { "empty" } else { t }}}")
145     logger.debug(s"  Read schema: ${readSchema.schema}")
146     logger.debug(s"  Sort: ${sort.fold("none") { fields => fields.map { case (f, rev) => s"$f ${if (rev) "descending" else ""}"}.mkString(", ")}}")
147     logger.debug(s"  Max features: ${max.getOrElse("none")}")
148 
149     val tableScan =
150       table.newScan()
151         .caseSensitive(false)
152         .project(readSchema.schema) // exclude z2 cols even if there's no transform
153         .filter(icebergFilter.expression)
154 
155     val scan = {
156       val ff = RecordSimpleFeature(readSchema)
157       new IcebergParquetScan(tableScan, threads).map(ff.apply)
158     }
159     try {
160       val iter = scan.filter(visFilter.apply)
161       // apply any client side filter
162       val filtered =
163         icebergFilter.remainder.map(FastFilterFactory.optimize(readSchema.sft, _)).fold(iter)(f => iter.filter(f.evaluate))
164       // transform again as necessary to remove any cols needed for filtering, and/or to evaluate complex expressions
165       val transformed = transform.fold(filtered) { case (tdefs, tsft) =>
166         val transforms = Transforms(readSchema.sft, tdefs).toArray
167         if (tsft == readSchema.sft && transforms.forall(_.isInstanceOf[PropertyTransform])) {
168           // simple case where transform is handled by the iceberg scan
169           filtered
170         } else {
171           // need to evaluate transform expressions
172           filtered.map(sf => new TransformSimpleFeature(tsft, transforms).setFeature(sf))
173         }
174       }
175       // sort and limit
176       val sorted = sort.fold(transformed)(new SortingSimpleFeatureIterator(transformed, _))
177       val limited = max.fold(sorted)(m => sorted.take(m))
178       limited
179     } catch {
180       case NonFatal(e) => CloseWithLogging(scan); throw e
181     }
182   }

400   /**
401    * Writes files up to a given size, then starts a new file
402    *
403    * @param writers iterator of files to write
404    * @param estimator target file size estimator
405    */
406   // noinspection ScalaWeakerAccess
407   class ChunkedFileSystemWriter(writers: Iterator[FileSystemWriter], estimator: UpdatingFileSizeEstimator)
408       extends FileSystemWriter {
409 
410     private var totalCount = 0L // total number of features written across all chunks
411     private var totalBytes = 0L // sum size of all finished chunks
412     private var remaining = estimator.estimate(0L)
413     private var writer: FileSystemWriter = _
414 
415     override def write(feature: SimpleFeature): Unit = {
416       if (writer == null) {
417         writer = writers.next()
418       }
419       writer.write(feature)
420       totalCount += 1
421       remaining -= 1
422       if (remaining == 0) {
423         val dataSize = writer.size
424         if (estimator.done(dataSize)) {
425           writer.close()
426           totalBytes += writer.size // re-calculate now that writer is closed, so we get the final, accurate size
427           writer = null
428           // adjust our estimate to account for the actual bytes written
429           estimator.update(totalBytes, totalCount)
430           remaining = estimator.estimate(0L)
431         } else {
432           remaining = math.max(100L, estimator.estimate(dataSize))
433         }
434       }
435     }
436 
437     override def size: Long = totalBytes + Option(writer).fold(0L)(_.size)
438 
439     override def flush(): Unit = if (writer != null) { writer.flush() }
440 
441     override def close(): Unit = {
442       if (writer != null) {
443         writer.close()
444       }
445       estimator.close()
446     }
447   }

479   /**
480    * Get the path for a new data file, using Iceberg FileContent semantics
481    *
482    * @param typeName simple feature type name
483    * @return
484    */
485   def newFilePath(typeName: String, prefix: String = ""): String = {
486     val filename =
487       s"${ColumnName.encode(typeName).take(20)}_$prefix${UUID.randomUUID().toString.replaceAllLiterally("-", "")}.parquet"
488     // partitioning logic taken from Apache Iceberg: https://iceberg.apache.org/docs/nightly/aws/#object-store-file-layout
489     val hash = {
490       val bytes = filename.getBytes(StandardCharsets.UTF_8)
491       val hash = MurmurHash3.hash32x86(bytes, 0, bytes.length, 0)
492       // Integer#toBinaryString excludes leading zeros, which we want to preserve
493       Integer.toBinaryString(hash | Integer.MIN_VALUE)
494     }
495     s"${hash.substring(0, 4)}/${hash.substring(4, 8)}/${hash.substring(8, 12)}/${hash.substring(12, 20)}/$filename"
496   }
