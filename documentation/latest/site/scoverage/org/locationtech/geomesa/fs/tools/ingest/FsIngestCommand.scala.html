31 class FsIngestCommand extends IngestCommand[FileSystemDataStore] with FsDistributedCommand {
32 
33   override val params = new FsIngestParams
34 
35   override protected def setBackendSpecificOptions(sft: SimpleFeatureType): Unit =
36     FsCreateSchemaCommand.setOptions(sft, params)
37 
38   override protected def startIngest(
39       mode: RunMode,
40       ds: FileSystemDataStore,
41       sft: SimpleFeatureType,
42       converter: Config,
43       inputs: Inputs): Awaitable = {
44     mode match {
45       case RunModes.Local =>
46         super.startIngest(mode, ds, sft, converter, inputs)
47 
48       case RunModes.Distributed =>
49         Command.user.info(s"Running ingestion in distributed ${if (params.combineInputs) "combine " else "" }mode")
50         val reducers = Option(params.reducers).filter(_ > 0).getOrElse {
51           throw new ParameterException("Please specify --num-reducers for distributed ingest")
52         }
53         val storage = ds.storage(sft.getTypeName)
54         val tmpPath = Option(params.tempPath).map(d => storage.context.fs.makeQualified(new Path(d)))
55         val targetFileSize = storage.metadata.get(Metadata.TargetFileSize).map(_.toLong)
56 
57         tmpPath.foreach { tp =>
58           if (storage.context.fs.exists(tp)) {
59             Command.user.info(s"Deleting temp path $tp")
60             storage.context.fs.delete(tp, true)
61           }
62         }
63 
64         new ParquetConverterJob(
65           connection, sft, converter, inputs.paths, libjarsFiles, libjarsPaths, reducers,
66           storage.context.root, storage.metadata.schemes, tmpPath, targetFileSize) {
67           override def configureJob(job: Job): Unit = {
68             super.configureJob(job)
69             if (params.combineInputs) {
70               job.setInputFormatClass(classOf[ConverterCombineInputFormat])
71               Option(params.maxSplitSize).foreach(s => FileInputFormat.setMaxInputSplitSize(job, s.toLong))
72             }
73           }
74         }
75 
76       case _ =>
77         throw new UnsupportedOperationException(s"Missing implementation for mode $mode")
78     }
79   }
80 }
