21 /**
22  * Mapping of fields to bounding box fields
23  *
24  * @param fields fields
25  */
26 case class BoundingBoxes(fields: Seq[BoundingBoxField]) {
27   private val fieldMap = fields.map(f => f.geometry -> f.bbox).toMap
28   def isEmpty: Boolean = fields.isEmpty
29   def nonEmpty: Boolean = fields.nonEmpty
30   def get(field: String): Option[String] = fieldMap.get(field)
31 }
32 
33 object BoundingBoxes {
34 
35   import scala.collection.JavaConverters._
36 
37   /**
38    * Gets the fields of this schema that have per-row bounding-boxes
39    *
40    * @param sft simple feature type
41    * @return
42    */
43   def apply(sft: SimpleFeatureType): BoundingBoxes = {
44     val bboxes = sft.getAttributeDescriptors.asScala.toSeq.flatMap { d =>
45       val binding = d.getType.getBinding
46       if (classOf[Geometry].isAssignableFrom(binding)) {
47         Some(BoundingBoxField(d.getLocalName))
48       } else {
49         None
50       }
51     }
52     BoundingBoxes(bboxes)
53   }
54 
55   /**
56    * Holder for a bounding box field, along with a reference back to the original geometry field
57    *
58    * @param geometry name of the original geometry field being covered
59    * @param bbox name of the bounding box field
60    */
61   case class BoundingBoxField(geometry: String, bbox: String)
62 
63   object BoundingBoxField {
64 
65     val BoundingBoxFieldPrefix = "__"
66     val BoundingBoxFieldSuffix = "_bbox__"
67 
68     val XMin = "xmin"
69     val YMin = "ymin"
70     val XMax = "xmax"
71     val YMax = "ymax"
72 
73     def apply(geometry: String, encoded: Boolean = false): BoundingBoxField = {
74       val geom = if (encoded) { geometry } else { alphaNumericSafeString(geometry) }
75       val bbox = s"$BoundingBoxFieldPrefix$geom$BoundingBoxFieldSuffix"
76       BoundingBoxField(geom, bbox)
77     }
78 
79     /**
80      * Creates a field name based on a bounding box field
81      *
82      * @param field name of a potential bbox field
83      * @return
84      */
85     def fromFieldName(field: String): Option[BoundingBoxField] = {
86       if (field.startsWith(BoundingBoxFieldPrefix) && field.endsWith(BoundingBoxFieldSuffix)) {
87         Some(BoundingBoxField(field.substring(BoundingBoxFieldPrefix.length, field.length - BoundingBoxFieldSuffix.length), field))
88       } else {
89         None
90       }
91     }
92 
93     /**
94      * The parquet schema for a bbox field
95      *
96      * @param bbox field name
97      * @return
98      */
99     def schema(bbox: String, fieldIds: AtomicInteger, nestedFieldIds: AtomicInteger): GroupType = {
100       Types.optionalGroup()
101         .id(fieldIds.getAndIncrement())
102         .required(PrimitiveTypeName.FLOAT).id(nestedFieldIds.getAndIncrement()).named(BoundingBoxField.XMin)
103         .required(PrimitiveTypeName.FLOAT).id(nestedFieldIds.getAndIncrement()).named(BoundingBoxField.YMin)
104         .required(PrimitiveTypeName.FLOAT).id(nestedFieldIds.getAndIncrement()).named(BoundingBoxField.XMax)
105         .required(PrimitiveTypeName.FLOAT).id(nestedFieldIds.getAndIncrement()).named(BoundingBoxField.YMax)
106         .named(bbox)
107     }
108 
109     /**
110      * The iceberg schema for a bbox field
111      *
112      * @param bbox field name
113      * @return
114      */
115     def icebergSchema(bbox: String, fieldIds: AtomicInteger, nestedFieldIds: AtomicInteger): NestedField = {
116       NestedField.optional(bbox).withId(fieldIds.getAndIncrement()).ofType(StructType.of(
117         NestedField.required(BoundingBoxField.XMin).withId(nestedFieldIds.getAndIncrement()).ofType(FloatType.get()).build(),
118         NestedField.required(BoundingBoxField.YMin).withId(nestedFieldIds.getAndIncrement()).ofType(FloatType.get()).build(),
119         NestedField.required(BoundingBoxField.XMax).withId(nestedFieldIds.getAndIncrement()).ofType(FloatType.get()).build(),
120         NestedField.required(BoundingBoxField.YMax).withId(nestedFieldIds.getAndIncrement()).ofType(FloatType.get()).build(),
121       )).build()
122     }
123   }
124 }
