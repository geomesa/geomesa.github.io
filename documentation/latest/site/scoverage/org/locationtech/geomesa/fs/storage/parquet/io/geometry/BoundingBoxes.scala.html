22 /**
23  * Mapping of fields to bounding box fields
24  *
25  * @param fields fields
26  */
27 case class BoundingBoxes(fields: Seq[BoundingBoxField]) {
28   private val fieldMap = fields.map(f => f.geometry -> f.bbox).toMap
29   def isEmpty: Boolean = fields.isEmpty
30   def nonEmpty: Boolean = fields.nonEmpty
31   def get(field: String): Option[String] = fieldMap.get(field)
32 }
33 
34 object BoundingBoxes {
35 
36   import scala.collection.JavaConverters._
37 
38   /**
39    * Gets the fields of this schema that have per-row bounding-boxes. We only add bounding boxes when
40    * they help with predicate push-down, e.g. when a geometry is a non-point or WKB encoded
41    *
42    * @param sft simple feature type
43    * @param encoding geometry encoding
44    * @return
45    */
46   def apply(sft: SimpleFeatureType, encoding: GeometryEncoding): BoundingBoxes = {
47     val bboxes = sft.getAttributeDescriptors.asScala.toSeq.flatMap { d =>
48       val binding = d.getType.getBinding
49       if (classOf[Geometry].isAssignableFrom(binding) &&
50         (encoding == GeometryEncoding.GeoParquetWkb || binding != classOf[Point])) {
51         Some(BoundingBoxField(d.getLocalName))
52       } else {
53         None
54       }
55     }
56     BoundingBoxes(bboxes)
57   }
58 
59   /**
60    * Holder for a bounding box field, along with a reference back to the original geometry field
61    *
62    * @param geometry name of the original geometry field being covered
63    * @param bbox name of the bounding box field
64    */
65   case class BoundingBoxField(geometry: String, bbox: String)
66 
67   object BoundingBoxField {
68 
69     val BoundingBoxFieldPrefix = "__"
70     val BoundingBoxFieldSuffix = "_bbox__"
71 
72     val XMin = "xmin"
73     val YMin = "ymin"
74     val XMax = "xmax"
75     val YMax = "ymax"
76 
77     def apply(geometry: String, encoded: Boolean = false): BoundingBoxField = {
78       val geom = if (encoded) { geometry } else { alphaNumericSafeString(geometry) }
79       val bbox = s"$BoundingBoxFieldPrefix$geom$BoundingBoxFieldSuffix"
80       BoundingBoxField(geom, bbox)
81     }
82 
83     /**
84      * Creates a field name based on a bounding box field
85      *
86      * @param field name of a potential bbox field
87      * @return
88      */
89     def fromFieldName(field: String): Option[BoundingBoxField] = {
90       if (field.startsWith(BoundingBoxFieldPrefix) && field.endsWith(BoundingBoxFieldSuffix)) {
91         Some(BoundingBoxField(field.substring(BoundingBoxFieldPrefix.length, field.length - BoundingBoxFieldSuffix.length), field))
92       } else {
93         None
94       }
95     }
96 
97     /**
98      * The parquet schema for a bbox field
99      *
100      * @param bbox field name
101      * @return
102      */
103     def schema(bbox: String, fieldIds: AtomicInteger, nestedFieldIds: AtomicInteger): GroupType = {
104       Types.optionalGroup()
105         .id(fieldIds.getAndIncrement())
106         .required(PrimitiveTypeName.FLOAT).id(nestedFieldIds.getAndIncrement()).named(BoundingBoxField.XMin)
107         .required(PrimitiveTypeName.FLOAT).id(nestedFieldIds.getAndIncrement()).named(BoundingBoxField.YMin)
108         .required(PrimitiveTypeName.FLOAT).id(nestedFieldIds.getAndIncrement()).named(BoundingBoxField.XMax)
109         .required(PrimitiveTypeName.FLOAT).id(nestedFieldIds.getAndIncrement()).named(BoundingBoxField.YMax)
110         .named(bbox)
111     }
112 
113     /**
114      * The iceberg schema for a bbox field
115      *
116      * @param bbox field name
117      * @return
118      */
119     def icebergSchema(bbox: String, fieldIds: AtomicInteger, nestedFieldIds: AtomicInteger): NestedField = {
120       NestedField.optional(bbox).withId(fieldIds.getAndIncrement()).ofType(StructType.of(
121         NestedField.required(BoundingBoxField.XMin).withId(nestedFieldIds.getAndIncrement()).ofType(FloatType.get()).build(),
122         NestedField.required(BoundingBoxField.YMin).withId(nestedFieldIds.getAndIncrement()).ofType(FloatType.get()).build(),
123         NestedField.required(BoundingBoxField.XMax).withId(nestedFieldIds.getAndIncrement()).ofType(FloatType.get()).build(),
124         NestedField.required(BoundingBoxField.YMax).withId(nestedFieldIds.getAndIncrement()).ofType(FloatType.get()).build(),
125       )).build()
126     }
127   }
128 }
