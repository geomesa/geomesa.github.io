22 class ConverterStorage(context: FileSystemContext,
23                        metadata: StorageMetadata,
24                        converter: SimpleFeatureConverter,
25                        pathFiltering: Option[PathFiltering])
26     extends AbstractFileSystemStorage(context, metadata, "") {
27 
28   override val encoding: String = ConverterStorage.Encoding
29 
30   // TODO close converter...
31   // the problem is that we aggressively cache storage instances for performance (in FileSystemStorageManager),
32   // so even if we wired a 'close' method through the entire storage api, we'd also have to implement a
33   // 'borrow/return' paradigm and expire idle instances. Since currently only converters with redis caches
34   // actually need to be closed, and since they will only open a single connection per converter, the
35   // impact should be low
36 
37   override protected def createWriter(file: Path, partition: Partition, observer: FileSystemObserver): FileSystemWriter =
38     throw new UnsupportedOperationException()
39 
40   override protected def createReader(
41       filter: Option[Filter],
42       transform: Option[(String, SimpleFeatureType)]): FileSystemPathReader = {
43     new ConverterFileSystemReader(context.fs, context.root, converter, filter, transform, pathFiltering)
44   }
45 
46   override def compact(partition: Partition, fileSize: Option[Long], threads: Int): Unit =
47     throw new UnsupportedOperationException("Converter storage does not support compactions")
48 
49   override def register(file: Path): StorageFile =
50     throw new UnsupportedOperationException("Converter storage does not support file registration")
51 }
