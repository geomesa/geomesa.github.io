130     /**
131      * Json serializer for partitions
132      */
133     object PartitionSerializer extends JsonSerializer[Partition] with JsonDeserializer[Partition] {
134 
135       override def serialize(src: Partition, typeOfSrc: Type, context: JsonSerializationContext): JsonElement = {
136         val array = new JsonArray(src.values.size)
137         src.values.toSeq.sortBy(k => (k.name, k.value)).foreach { value =>
138           array.add(context.serialize(value))
139         }
140         array
141       }
142 
143       override def deserialize(json: JsonElement, typeOfT: Type, context: JsonDeserializationContext): Partition = {
144         val array = json.getAsJsonArray
145         val values = Set.newBuilder[PartitionKey]
146         var i = 0
147         while (i < array.size()) {
148           values += context.deserialize(array.get(i), classOf[PartitionKey])
149           i += 1
150         }
151         Partition(values.result())
152       }
153     }

201   /**
202    * Ranged bounds, used for filtering on partitions
203    *
204    * @param name partition that this bound applies to
205    * @param lower lower bound, inclusive
206    * @param upper upper bound, exclusive
207    */
208   case class PartitionRange(name: String, lower: String, upper: String) {
209 
210     /**
211      * Is the value contained in this bounds
212      *
213      * @param value partition value
214      * @return
215      */
216     def contains(value: String): Boolean = value >= lower && value < upper
217 
218     /**
219      * Attempt to merge two bounds. Only overlapping bounds will result in a successful merge. Trying to merge
220      * bounds from a different partition scheme is a logical error.
221      *
222      * @param other bounds to merge
223      * @return
224      */
225     def merge(other: PartitionRange): Option[PartitionRange] = {
226       if (lower <= other.lower) {
227         if (upper >= other.upper) {
228           Some(this)
229         } else if (upper >= other.lower) {
230           Some(PartitionRange(name, lower, other.upper))
231         } else {
232           None
233         }
234       } else if (lower > other.upper) {
235         None
236       } else if (upper >= other.upper) {
237         Some(PartitionRange(name, other.lower, upper))
238       } else {
239         Some(other)
240       }
241     }
242   }

269     def removeScheme(): Option[Seq[String]] = {

271         // back compatible check for old json-serialized schemes
272         if (scheme.trim.startsWith("{")) {
273           try {
274             def result(name: String, options: Map[String, String]): Seq[String] = {
275               val opts = options.map { case (k, v) => s"$k=$v" }.mkString(":")
276               name.split(",").toSeq.map(n => s"$n:$opts")

278             val source = ConfigSource.fromConfig(ConfigFactory.parseString(scheme))

284             }
285           } catch {
286             case NonFatal(e) => throw new RuntimeException(s"Could not parse legacy scheme options: $scheme", e)
287           }

291       }
