50   private val truncateToPartitionStart: ZonedDateTime => ZonedDateTime = {
51     // note: `truncatedTo` is only valid up to DAYS, other units require additional steps
52     val truncate: ZonedDateTime => ZonedDateTime = stepUnit match {
53       case NANOS | MICROS | MILLIS | SECONDS | MINUTES | HOURS | DAYS =>
54         dt => dt.truncatedTo(stepUnit)
55 
56       case WEEKS =>
57         val adjuster = TemporalAdjusters.previousOrSame(DayOfWeek.MONDAY) // note: java day_of_week defines monday as the first day
58         dt => dt.`with`(adjuster).truncatedTo(DAYS)
59 
60       case MONTHS =>
61         val adjuster = TemporalAdjusters.firstDayOfMonth()
62         dt => dt.`with`(adjuster).truncatedTo(DAYS)
63 
64       case YEARS =>
65         val adjuster = TemporalAdjusters.firstDayOfYear()
66         dt => dt.`with`(adjuster).truncatedTo(DAYS)
67 
68       case _ =>
69         throw new IllegalArgumentException(s"${HierarchicalDateTimeScheme.Config.StepUnitOpt} $stepUnit is not supported")
70     }
71     if (step == 1) {
72       truncate
73     } else {
74       val chronoField = stepUnit match {
75         case NANOS   => ChronoField.NANO_OF_SECOND
76         case MICROS  => ChronoField.MICRO_OF_SECOND
77         case MILLIS  => ChronoField.MILLI_OF_SECOND
78         case SECONDS => ChronoField.SECOND_OF_MINUTE
79         case MINUTES => ChronoField.MINUTE_OF_HOUR
80         case HOURS   => ChronoField.HOUR_OF_DAY
81         case DAYS    => ChronoField.DAY_OF_YEAR
82         case WEEKS   => WeekFields.ISO.weekOfWeekBasedYear()
83         case MONTHS  => ChronoField.MONTH_OF_YEAR
84         case YEARS   => ChronoField.YEAR
85         case _ => throw new IllegalArgumentException(s"${HierarchicalDateTimeScheme.Config.StepUnitOpt} $stepUnit is not supported")
86       }
87       val min = math.max(chronoField.range().getMinimum, 0).toInt // account for 1-based fields like {month, week, day}_of_year
88       truncate.andThen { dt =>
89         val steps = dt.get(chronoField) - min
90         val remainder = steps % step
91         if (remainder == 0) { dt } else { dt.`with`(chronoField, steps + min - remainder ) }
92       }
93     }
94   }

110   override def getRangesForFilter(filter: Filter): Option[Seq[PartitionRange]] = {
111     getBounds(filter).map { bounds =>
112       val builder = new RangeBuilder()
113       bounds.foreach { bound =>
114         val lower = bound.lower.value.fold("")(v => formatter.format(truncateToPartitionStart(v)))
115         val upper = bound.upper.value.fold(HierarchicalDateTimeScheme.UnboundedUpper) { v =>
116           val partition = formatter.format(truncateToPartitionStart(v))
117           if (bound.upper.exclusive && formatter.format(truncateToPartitionStart(v.minus(1, ChronoUnit.MILLIS))) < partition) {
118             partition
119           } else {
120             formatter.format(truncateToPartitionStart(v.plus(step, stepUnit)))
121           }
122         }
123         builder += PartitionRange(name, lower, upper)
124       }
125       builder.result()
126     }
127   }
128 
129   override def getPartitionsForFilter(filter: Filter): Option[Seq[PartitionKey]] = {
130     getBounds(filter).map { bounds =>
131       bounds.flatMap { bound =>
132         if (!bound.isBoundedBothSides) {
133           throw new IllegalArgumentException(s"Can't enumerate an unbounded filter: ${ECQL.toCQL(filter)}")
134         }
135         val lower = bound.lower.value.get
136         // `stepUnit.between` claims to be upper endpoint exclusive, but doesn't seem to be...
137         val steps = stepUnit.between(lower, bound.upper.value.get).toInt / step
138         Seq.tabulate(steps)(i => PartitionKey(name, formatter.format(lower.plus(step * i, stepUnit))))
139       }
140     }
141   }

143   override def getCoveringFilter(partition: PartitionKey): Filter = {
144     val zdt = DateParsing.parse(partition.value, formatter)
145     val start = DateParsing.format(zdt)
146     val end = DateParsing.format(zdt.plus(step, stepUnit))
147     ff.and(ff.greaterOrEqual(ff.property(dtg), ff.literal(start)), ff.less(ff.property(dtg), ff.literal(end)))
148   }
149 
150   private def getBounds(filter: Filter): Option[Seq[Bounds[ZonedDateTime]]] = {
151     val bounds = FilterHelper.extractIntervals(filter, dtg)
152     if (bounds.isEmpty) {
153       None
154     } else if (bounds.disjoint) {
155       Some(Seq.empty)
156     } else {
157       Some(bounds.values)
158     }
159   }

197     // java.time doesn't seem to have a way to parse back out a year/week format...

199     val Weekly: Format = {
200       val formatter =
201         new DateTimeFormatterBuilder()
202           .appendValue(WeekFields.ISO.weekBasedYear(), 4)
203           .appendLiteral("/W")
204           .appendValue(WeekFields.ISO.weekOfWeekBasedYear(), 2)
205           .parseDefaulting(ChronoField.DAY_OF_WEEK, 1)
206           .toFormatter()
207       Format("weekly", formatter, "YYYY/'W'ww", ChronoUnit.WEEKS)
208     }

213   override def load(sft: SimpleFeatureType, scheme: String): Option[HierarchicalDateTimeScheme] = {
214     import org.locationtech.geomesa.utils.geotools.RichSimpleFeatureType.RichSimpleFeatureType
215 
216     val opts = SchemeOpts(scheme)
217 
218     lazy val step = opts.opts.get(Config.StepOpt).map(_.toInt).getOrElse(1)
219     lazy val dtg = opts.opts.get(Config.DtgAttribute).orElse(sft.getDtgField).getOrElse {
220       throw new IllegalArgumentException(s"DateTime scheme requires valid attribute '${Config.DtgAttribute}'")
221     }
222     lazy val dtgIndex = attributeIndex(sft, dtg, Some(classOf[Date]))
223 
224     if (opts.name == Name) {
225       val unit = opts.opts.get(Config.StepUnitOpt).map(c => ChronoUnit.valueOf(c.toUpperCase)).getOrElse {
226         throw new IllegalArgumentException(s"DateTime scheme requires valid unit '${Config.StepUnitOpt}'")
227       }
228       val format = opts.opts.getOrElse(Config.DateTimeFormatOpt,
229         throw new IllegalArgumentException(s"DateTime scheme requires valid format '${Config.DateTimeFormatOpt}'"))
230       require(!format.endsWith("/"), "Format cannot end with a slash")
231       val formatter = try { DateTimeFormatter.ofPattern(format) } catch {
232         case NonFatal(e) => throw new IllegalArgumentException(s"Invalid date format '$format':", e)
233       }
234       Some(HierarchicalDateTimeScheme(formatter, format, unit, step, dtg, dtgIndex))
235     } else {
236       Formats(opts.name).map(f => HierarchicalDateTimeScheme(f.formatter, f.pattern, f.unit, step, dtg, dtgIndex))
237     }
238   }
