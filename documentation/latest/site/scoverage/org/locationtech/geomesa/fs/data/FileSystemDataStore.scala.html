28 /**
29  * File system data store
30  *
31  * @param storageFactory storage factory
32  * @param catalog metadata catalog
33  * @param config config
34  */
35 class FileSystemDataStore(storageFactory: FileSystemStorageFactory, catalog: StorageMetadataCatalog, config: FileSystemDataStoreConfig)
36     extends ContentDataStore with HasGeoMesaStats with LazyLogging {
37 
38   import scala.collection.JavaConverters._
39 
40   private val cache = Caffeine.newBuilder().build(
41     new CacheLoader[String, FileSystemStorage]() {
42       override def load(k: String): FileSystemStorage = storageFactory.apply(config.context, catalog.load(k))
43     }
44   )
45 
46   config.context.namespace.foreach(setNamespaceURI)
47 
48   override val stats: GeoMesaStats = new FileSystemStats(this)
49 
50   override def createTypeNames(): java.util.List[Name] = {
51     val names = new java.util.ArrayList[Name]()
52     catalog.getTypeNames.foreach(name => names.add(new NameImpl(config.context.namespace.orNull, name)))
53     names
54   }
55 
56   override def createSchema(original: SimpleFeatureType): Unit = {
57     import org.locationtech.geomesa.fs.storage.core.RichSimpleFeatureType
58     if (catalog.getTypeNames.contains(original.getTypeName)) {
59       logger.warn(
60         s"Schema already exists: ${SimpleFeatureTypes.encodeType(cache.get(original.getTypeName).metadata.sft, includeUserData = true)}")
61     } else {
62       // copy the feature type so that we don't affect it when removing user data, below
63       val sft = SimpleFeatureTypes.copy(original)
64 
65       GeoMesaSchemaValidator.validate(sft)
66       // remove the configs in the user data as they're persisted separately
67       val scheme = sft.removeScheme().getOrElse {
68         throw new IllegalArgumentException("Partition scheme must be specified in the SimpleFeatureType user data")
69       }
70       val fileSize = sft.removeTargetFileSize()
71       cache.put(sft.getTypeName, storageFactory.apply(config.context, catalog.create(sft, scheme, fileSize)))
72     }
73   }
74 
75   override def createFeatureSource(entry: ContentEntry): ContentFeatureSource =
76     new FileSystemFeatureStore(storage(entry.getTypeName), entry, Query.ALL, config)
77 
78   /**
79    * Get a handle on the underlying storage instance for a given simple feature type
80    *
81    * @param typeName simple feature type name
82    * @return
83    */
84   def storage(typeName: String): FileSystemStorage = {
85     val storage = cache.get(typeName)
86     if (storage == null) {
87       throw new IllegalArgumentException(s"Schema '$typeName' doesn't exist or could not be loaded")
88     }
89     storage
90   }
91 
92   override def dispose(): Unit = {
93     try {
94       CloseWithLogging(cache.asMap().asScala.values)
95       cache.invalidateAll()
96     } finally {
97       super.dispose()
98     }
99   }
100 }
