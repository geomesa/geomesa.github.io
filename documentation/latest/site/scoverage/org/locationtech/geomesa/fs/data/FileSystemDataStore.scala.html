29 /**
30  * File system data store
31  *
32  * @param storageFactory storage factory
33  * @param catalog metadata catalog
34  * @param config config
35  */
36 class FileSystemDataStore(storageFactory: FileSystemStorageFactory, catalog: StorageMetadataCatalog, config: FileSystemDataStoreConfig)
37     extends ContentDataStore with HasGeoMesaStats with LazyLogging {
38 
39   import scala.collection.JavaConverters._
40 
41   private val cache = Caffeine.newBuilder().build(
42     new CacheLoader[String, FileSystemStorage]() {
43       override def load(k: String): FileSystemStorage = storageFactory.apply(config.context, catalog.load(k))
44     }
45   )
46 
47   config.context.namespace.foreach(setNamespaceURI)
48 
49   override val stats: GeoMesaStats = new UnoptimizedRunnableStats(this)
50 
51   override def createTypeNames(): java.util.List[Name] = {
52     val names = new java.util.ArrayList[Name]()
53     catalog.getTypeNames.foreach(name => names.add(new NameImpl(config.context.namespace.orNull, name)))
54     names
55   }
56 
57   override def createSchema(original: SimpleFeatureType): Unit = {
58     import org.locationtech.geomesa.fs.storage.common.RichSimpleFeatureType
59 
60     if (catalog.getTypeNames.contains(original.getTypeName)) {
61       logger.warn(
62         s"Schema already exists: ${SimpleFeatureTypes.encodeType(cache.get(original.getTypeName).metadata.sft, includeUserData = true)}")
63     } else {
64       // copy the feature type so that we don't affect it when removing user data, below
65       val sft = SimpleFeatureTypes.copy(original)
66 
67       GeoMesaSchemaValidator.validate(sft)
68       // remove the configs in the user data as they're persisted separately
69       val scheme = sft.removeScheme().getOrElse {
70         throw new IllegalArgumentException("Partition scheme must be specified in the SimpleFeatureType user data")
71       }
72       val fileSize = sft.removeTargetFileSize()
73       cache.put(sft.getTypeName, storageFactory.apply(config.context, catalog.create(sft, scheme, fileSize)))
74     }
75   }
76 
77   override def createFeatureSource(entry: ContentEntry): ContentFeatureSource =
78     new FileSystemFeatureStore(storage(entry.getTypeName), entry, Query.ALL, config.readThreads, config.writeTimeout, config.queryTimeout)
79 
80   /**
81    * Get a handle on the underlying storage instance for a given simple feature type
82    *
83    * @param typeName simple feature type name
84    * @return
85    */
86   def storage(typeName: String): FileSystemStorage = {
87     val storage = cache.get(typeName)
88     if (storage == null) {
89       throw new IllegalArgumentException(s"Schema '$typeName' doesn't exist or could not be loaded")
90     }
91     storage
92   }
93 
94   override def dispose(): Unit = {
95     try {
96       CloseWithLogging(cache.asMap().asScala.values)
97       cache.invalidateAll()
98     } finally {
99       super.dispose()
100     }
101   }
102 }
