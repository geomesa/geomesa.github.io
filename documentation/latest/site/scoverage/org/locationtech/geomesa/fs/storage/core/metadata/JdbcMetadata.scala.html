29 /**
30  * Storage metadata implementation backed by a SQL database. Currently compatible with Postgres - other
31  * databases will likely have incompatibilities in the SQL syntax.
32  *
33  * Schema consists of five tables (note: `storage_` is the default table prefix, but may be customized):
34  *
35  * `storage_meta`
36  *
37  * * Holds the base metadata (simple feature type, partition scheme, encoding) as a serialized clob
38  *
39  * ** root varchar(256) not null
40  * ** typeName varchar(256) not null
41  * ** key varchar(256) not null,
42  * ** value text not null,
43  * ** primary key (root, typeName, key)
44  *
45  * `storage_files`
46  *
47  * * Holds file-level metadata for each storage file
48  *
49  * ** id bigint primary key (generated identity)
50  * ** root varchar(256) not null
51  * ** typeName varchar(256) not null
52  * ** file varchar(256) not null
53  * ** count bigint not null
54  * ** action char(1) not null (A=Append, M=Modify, D=Delete)
55  * ** sort integer[]
56  * ** ts timestamp without time zone not null
57  *
58  * `storage_partitions`
59  *
60  * * Holds partition key-value pairs for each file
61  *
62  * ** file_id bigint not null (foreign key to storage_files)
63  * ** name varchar(64) not null
64  * ** value varchar(64) not null
65  * ** primary key (file_id, name)
66  *
67  * `storage_spatial_bounds`
68  *
69  * * Holds spatial bounds for each file by attribute
70  *
71  * ** file_id bigint not null (foreign key to storage_files)
72  * ** attribute smallint not null
73  * ** x_min double precision
74  * ** x_max double precision
75  * ** y_min double precision
76  * ** y_max double precision
77  * ** primary key (file_id, attribute)
78  *
79  * `storage_attr_bounds`
80  *
81  * * Holds attribute bounds for each file by attribute
82  *
83  * ** file_id bigint not null (foreign key to storage_files)
84  * ** attribute smallint not null
85  * ** lower text
86  * ** upper text
87  * ** primary key (file_id, attribute)
88  *
89  * @param pool connection pool
90  * @param meta metadata table reference
91  * @param files files table reference
92  * @param namespace feature type namespace
93  **/

155   override def replaceFiles(existing: Seq[StorageFile], replacements: Seq[StorageFile]): Unit = {
156     WithClose(pool.getConnection()) { cx =>
157       cx.setAutoCommit(false)
158       try {
159         existing.foreach(files.delete(cx, _))
160         replacements.foreach(files.insert(cx, _))
161         cx.commit()
162       } catch {
163         case NonFatal(e) =>
164           cx.rollback()
165           throw e
166       } finally {
167         cx.setAutoCommit(true)
168       }
169     }
170   }

172   override def getFiles(): Seq[StorageFile] =
173     WithClose(pool.getConnection())(files.select(_, Seq.empty, Seq.empty, Seq.empty))
174 
175   override def getFiles(partition: Partition): Seq[StorageFile] = {
176     val filters = partition.values.toSeq.map(p => PartitionRange(p.name, p.value, p.value + ZeroChar))
177     WithClose(pool.getConnection())(files.select(_, filters, Seq.empty, Seq.empty))
178   }
179 
180   override def getFiles(filter: Filter): Seq[StorageFile] = {
181     if (filter == Filter.INCLUDE) {
182       getFiles()
183     } else {
184       val filters = getFilters(filter)
185       if (filters.isEmpty) {
186         Seq.empty // no intersecting partitions
187       } else {
188         WithClose(pool.getConnection()) { cx =>
189           filters.flatMap { f =>
190             files.select(cx, f.partitions, f.spatialBounds, f.attributeBounds)
191           }
192         }
193       }
194     }
195   }
196 
197   override def close(): Unit = pool.close()
198 }

260 
261     val tableName: String = s"${tablePrefix}meta"
262 
263     private val qualifiedTableName = s""""$schema"."$tableName""""
264 
265     def create(cx: Connection): Unit = {
266       WithClose(cx.createStatement()) { st =>
267         st.executeUpdate(
268           s"""create table if not exists $qualifiedTableName (
269               root varchar(256) not null,
270               typeName varchar(256) not null,
271               key varchar(256) not null,
272               value text not null,
273               primary key (root, typeName, key))"""
274         )
275       }
276     }
277 
278     def selectTypeNames(cx: Connection, root: String): Seq[String] = {
279       WithClose(cx.prepareStatement(s"SELECT DISTINCT typeName FROM $qualifiedTableName WHERE root = ?")) { st =>
280         st.setString(1, root)
281         val builder = Seq.newBuilder[String]
282         WithClose(st.executeQuery()) { rs =>
283           while (rs.next()) {
284             builder += rs.getString(1)
285           }
286         }
287         builder.result()
288       }
289     }
290 
291     def insert(cx: Connection, sft: SimpleFeatureType): Unit =
292       insert(cx, "__sft__", SimpleFeatureTypes.encodeType(sft, includeUserData = true))
293 
294     def insert(cx: Connection, partitions: Seq[String]): Unit = insert(cx, "__partitions__", partitions.mkString(","))
295 
296     def insert(cx: Connection, key: String, value: String): Unit = {
297       val sql =
298         s"INSERT INTO $qualifiedTableName (root, typeName, key, value) " +
299           "VALUES (?, ?, ?, ?) ON CONFLICT (root, typeName, key) DO UPDATE SET value = EXCLUDED.value"
300       WithClose(cx.prepareStatement(sql)) { st =>
301         st.setString(1, root)
302         st.setString(2, typeName)
303         st.setString(3, key)
304         st.setString(4, value)
305         st.executeUpdate()
306       }
307     }
308

312

314       WithClose(cx.prepareStatement(s"SELECT value FROM $qualifiedTableName WHERE root = ? AND typeName = ? AND key = ?")) { st =>
315         st.setString(1, root)
316         st.setString(2, typeName)
317         st.setString(3, key)
318         WithClose(st.executeQuery()) { rs =>
319           if (rs.next()) {
320             Option(rs.getString(1))
321           } else {
322             None
323           }
324         }
325       }
326     }

330         st.setString(1, root)
331         st.setString(2, typeName)
332         st.setString(3, key)
333         st.executeUpdate()
334       }

344     private val partitionsTable = s""""$schema"."${tablePrefix}partitions""""

348     def create(cx: Connection): Unit = {
349       WithClose(cx.createStatement()) { st =>
350         st.executeUpdate(
351           s"""CREATE TABLE IF NOT EXISTS $filesTable (
352              |  id BIGINT PRIMARY KEY GENERATED BY DEFAULT AS IDENTITY,
353              |  root VARCHAR(256) NOT NULL,
354              |  typeName varchar(256) not null,
355              |  file VARCHAR(256) NOT NULL,
356              |  count BIGINT NOT NULL,
357              |  action CHAR(1) NOT NULL,
358              |  sort INTEGER[],
359              |  ts TIMESTAMP WITHOUT TIME ZONE NOT NULL

364              |  file_id BIGINT NOT NULL,

368              |  CONSTRAINT fk_storage_file

379              |  x_max DOUBLE PRECISION,

392              |  attribute SMALLINT NOT NULL,

403         st.executeUpdate(s"""CREATE INDEX IF NOT EXISTS ${tablePrefix}attr_bounds_idx_bounds ON $attrBoundsTable(attribute, lower, upper);""")
404         st.executeUpdate(s"""CREATE INDEX IF NOT EXISTS ${tablePrefix}spatial_bounds_idx_bounds ON $spatialBoundsTable(attribute, x_min, x_max, y_min, y_max);""")

408     def insert(cx: Connection, file: StorageFile): Unit = {
409       val id = WithClose(cx.prepareStatement(
410         s"INSERT INTO $filesTable (root, typeName, file, count, action, sort, ts) " +
411           "VALUES (?, ?, ?, ?, ?, ?, ?) RETURNING id", Statement.RETURN_GENERATED_KEYS)) { st =>
412         st.setString(1, root)
413         st.setString(2, typeName)
414         st.setString(3, file.file)
415         st.setLong(4, file.count)
416         st.setString(5, file.action.toString.substring(0, 1))
417         if (file.sort.isEmpty) {
418           st.setNull(6, java.sql.Types.ARRAY)
419         } else {
420           st.setArray(6, cx.createArrayOf("integer", file.sort.map(Int.box).toArray[AnyRef]))
421         }
422         st.setTimestamp(7, Timestamp.from(Instant.ofEpochMilli(file.timestamp)))
423         st.executeUpdate()
424         WithClose(st.getGeneratedKeys) { rs =>
425           if (rs.next()) {
426             rs.getLong(1)
427           } else {
428             throw new RuntimeException("Failed to retrieve generated key")
429           }
430         }
431       }
432       WithClose(cx.prepareStatement(s"INSERT INTO $partitionsTable (file_id, name, value) VALUES (?, ?, ?)")) { st =>
433         file.partition.values.foreach { p =>
434           st.setLong(1, id)
435           st.setString(2, p.name)
436           st.setString(3, p.value)
437           st.executeUpdate()
438         }
439       }
440       if (file.spatialBounds.nonEmpty) {
441         WithClose(cx.prepareStatement(s"INSERT INTO $spatialBoundsTable (file_id, attribute, x_min, x_max, y_min, y_max) VALUES (?, ?, ?, ?, ?, ?)")) { st =>
442           file.spatialBounds.foreach { bounds =>
443             st.setLong(1, id)
444             st.setInt(2, bounds.attribute)
445             st.setDouble(3, bounds.xmin)
446             st.setDouble(4, bounds.xmax)
447             st.setDouble(5, bounds.ymin)
448             st.setDouble(6, bounds.ymax)
449             st.executeUpdate()
450           }
451         }
452       }
453       if (file.attributeBounds.nonEmpty) {
454         WithClose(cx.prepareStatement(s"INSERT INTO $attrBoundsTable (file_id, attribute, lower, upper) VALUES (?, ?, ?, ?)")) { st =>
455           file.attributeBounds.foreach { bounds =>
456             st.setLong(1, id)
457             st.setInt(2, bounds.attribute)
458             st.setString(3, bounds.lower)
459             st.setString(4, bounds.upper)
460             st.executeUpdate()
461           }
462         }
463       }
464     }

475     def select(
476         cx: Connection,
477         partitions: Seq[PartitionRange],
478         spatialBounds: Seq[SpatialOr],
479         attributeBounds: Seq[AttributeOr],
480       ): Seq[StorageFile] = {
481 
482       // build query with multiple joins - one for each partition filter (AND logic)
483       val partitionJoins = partitions.zipWithIndex.map {
484         case (PartitionRange(name, lower, upper), i) => PartitionJoin(name, lower, upper, s"sp_filter_$i")
485       }
486 
487       // build spatial bound filters - one LEFT JOIN per attribute with OR logic for bounds
488       val spatialBoundJoins = spatialBounds.zipWithIndex.map {
489         case (SpatialOr(attribute, bounds), i) => SpatialBoundsJoin(attribute, bounds, s"sb_filter_$i")
490       }
491 
492       // build attribute bound filters - one LEFT JOIN per attribute with OR logic for bounds
493       val attrBoundJoins = attributeBounds.zipWithIndex.map {
494         case (AttributeOr(attribute, bounds), i) => AttributeBoundsJoin(attribute, bounds, s"ab_filter_$i")
495       }
496 
497       val joinClause =
498         partitionJoins.map { j =>
499           s"JOIN $partitionsTable ${j.tableAlias} ON sf.id = ${j.tableAlias}.file_id AND ${j.onClause}"
500         } ++
501         spatialBoundJoins.map { j =>
502           s"LEFT JOIN $spatialBoundsTable ${j.tableAlias} ON sf.id = ${j.tableAlias}.file_id AND ${j.tableAlias}.attribute = ?"
503         } ++
504         attrBoundJoins.map { j =>
505           s"LEFT JOIN $attrBoundsTable ${j.tableAlias} ON sf.id = ${j.tableAlias}.file_id AND ${j.tableAlias}.attribute = ?"
506         }
507 
508       // build WHERE clause for spatial and attribute bounds
509       // if bounds don't exist (NULL), we count it as a match
510       val spatialWhereClause = spatialBoundJoins.map { j =>
511         s"(${j.tableAlias}.file_id IS NULL OR ${j.whereClause})"
512       }
513       val attrWhereClause = attrBoundJoins.map { j =>
514         s"(${j.tableAlias}.file_id IS NULL OR ${j.whereClause})"
515       }
516       val allWhereClauses = spatialWhereClause ++ attrWhereClause
517 
518       // note: all child tables (partitions, spatial bounds, attribute bounds) are pre-aggregated
519       // in subqueries. This ensures each file returns exactly one row with no Cartesian products.
520       // The sp_filter, sb_filter, and ab_filter joins are used for filtering and must remain on the raw tables.
521       val whereClause = if (allWhereClauses.isEmpty) {
522         "WHERE sf.root = ? AND sf.typeName = ?"
523       } else {
524         s"WHERE sf.root = ? AND sf.typeName = ? AND ${allWhereClauses.mkString(" AND ")}"
525       }
526 
527       val query =
528         s"""SELECT sf.id, sf.file, sf.count, sf.action, sf.sort, sf.ts,
529            |  sp.partition_names, sp.partition_values,
530            |  sb.sb_attributes, sb.sb_x_mins, sb.sb_x_maxs, sb.sb_y_mins, sb.sb_y_maxs,
531            |  ab.ab_attributes, ab.ab_lowers, ab.ab_uppers
532            |FROM $filesTable sf
533            |${joinClause.mkString("\n")}
534            |LEFT JOIN (
535            |  SELECT file_id,
536            |    array_agg(name ORDER BY name) as partition_names,
537            |    array_agg(value ORDER BY name) as partition_values
538            |  FROM $partitionsTable
539            |  GROUP BY file_id
540            |) sp ON sf.id = sp.file_id
541            |LEFT JOIN (
542            |  SELECT file_id,
543            |    array_agg(attribute ORDER BY attribute) as sb_attributes,
544            |    array_agg(x_min ORDER BY attribute) as sb_x_mins,
545            |    array_agg(x_max ORDER BY attribute) as sb_x_maxs,
546            |    array_agg(y_min ORDER BY attribute) as sb_y_mins,
547            |    array_agg(y_max ORDER BY attribute) as sb_y_maxs
548            |  FROM $spatialBoundsTable
549            |  GROUP BY file_id
550            |) sb ON sf.id = sb.file_id
551            |LEFT JOIN (
552            |  SELECT file_id,
553            |    array_agg(attribute ORDER BY attribute) as ab_attributes,
554            |    array_agg(lower ORDER BY attribute) as ab_lowers,
555            |    array_agg(upper ORDER BY attribute) as ab_uppers
556            |  FROM $attrBoundsTable
557            |  GROUP BY file_id
558            |) ab ON sf.id = ab.file_id
559            |$whereClause
560            |ORDER BY sf.id DESC""".stripMargin
561 
562       WithClose(cx.prepareStatement(query)) { st =>
563         var paramIndex = 1
564         // set partition join parameters
565         partitionJoins.foreach { join =>
566           paramIndex += join.apply(st, paramIndex)
567         }
568         // set spatial bound join parameters (attribute IDs in JOIN clauses)
569         spatialBoundJoins.foreach { join =>
570           st.setInt(paramIndex, join.attribute)
571           paramIndex += 1
572         }
573         // set attribute bound join parameters (attribute IDs in JOIN clauses)
574         attrBoundJoins.foreach { join =>
575           st.setInt(paramIndex, join.attribute)
576           paramIndex += 1
577         }
578         // set root parameter
579         st.setString(paramIndex, root)
580         paramIndex += 1
581         st.setString(paramIndex, typeName)
582         paramIndex += 1
583         // set spatial bound WHERE clause parameters
584         spatialBoundJoins.foreach { join =>
585           paramIndex += join.apply(st, paramIndex)
586         }
587         // set attribute bound WHERE clause parameters
588         attrBoundJoins.foreach { join =>
589           paramIndex += join.apply(st, paramIndex)
590         }
591         WithClose(st.executeQuery())(toStorageFiles)
592       }
593     }

595     private def toStorageFiles(rs: ResultSet): Seq[StorageFile] = {
596       val result = Seq.newBuilder[StorageFile]
597 
598       while (rs.next()) {
599         val file = rs.getString(2)
600         val count = rs.getLong(3)
601         val action = rs.getString(4).charAt(0) match {
602           case 'A' => StorageMetadata.StorageFileAction.Append
