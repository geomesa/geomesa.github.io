29 /**
30  * Storage metadata implementation backed by a SQL database. Currently compatible with Postgres - other
31  * databases will likely have incompatibilities in the SQL syntax.
32  *
33  * Schema consists of five tables (note: `storage_` is the default table prefix, but may be customized):
34  *
35  * `storage_meta`
36  *
37  * * Holds the base metadata (simple feature type, partition scheme, encoding) as a serialized clob
38  *
39  * ** root varchar(256) not null
40  * ** typeName varchar(256) not null
41  * ** key varchar(256) not null,
42  * ** value text not null,
43  * ** primary key (root, typeName, key)
44  *
45  * `storage_files`
46  *
47  * * Holds file-level metadata for each storage file
48  *
49  * ** id bigint primary key (generated identity)
50  * ** root varchar(256) not null
51  * ** typeName varchar(256) not null
52  * ** file varchar(256) not null
53  * ** count bigint not null
54  * ** action char(1) not null (A=Append, M=Modify, D=Delete)
55  * ** sort integer[]
56  * ** ts timestamp without time zone not null
57  *
58  * `storage_partitions`
59  *
60  * * Holds partition key-value pairs for each file
61  *
62  * ** file_id bigint not null (foreign key to storage_files)
63  * ** name varchar(64) not null
64  * ** value varchar(64) not null
65  * ** primary key (file_id, name)
66  *
67  * `storage_spatial_bounds`
68  *
69  * * Holds spatial bounds for each file by attribute
70  *
71  * ** file_id bigint not null (foreign key to storage_files)
72  * ** attribute smallint not null
73  * ** x_min double precision
74  * ** x_max double precision
75  * ** y_min double precision
76  * ** y_max double precision
77  * ** primary key (file_id, attribute)
78  *
79  * `storage_attr_bounds`
80  *
81  * * Holds attribute bounds for each file by attribute
82  *
83  * ** file_id bigint not null (foreign key to storage_files)
84  * ** attribute smallint not null
85  * ** lower text

87  * ** primary key (file_id, attribute)
88  *
89  * @param pool connection pool
90  * @param meta metadata table reference
91  * @param files files table reference
92  * @param namespace feature type namespace
93  **/

445     def select(cx: Connection, partitions: Seq[PartitionRange], columnBounds: Seq[ColumnOr]): Seq[StorageFile] = {
446 
447       // build query with multiple joins - one for each partition filter (AND logic)
448       val partitionJoins = partitions.zipWithIndex.map {
449         case (PartitionRange(name, lower, upper), i) => PartitionJoin(name, lower, upper, s"sp_filter_$i")
450       }
451 
452       // build column bound filters - one LEFT JOIN per attribute with OR logic for bounds
453       val columnBoundJoins = columnBounds.zipWithIndex.map {
454         case (ColumnOr(attribute, bounds), i) => ColumnBoundsJoin(attribute, bounds, s"cb_filter_$i")
455       }
456 
457       val joinClause =
458         partitionJoins.map { j =>
459           s"JOIN $partitionsTable ${j.tableAlias} ON sf.id = ${j.tableAlias}.file_id AND ${j.onClause}"
460         } ++
461         columnBoundJoins.map { j =>
462           s"LEFT JOIN $columnBoundsTable ${j.tableAlias} ON sf.id = ${j.tableAlias}.file_id AND ${j.tableAlias}.attribute = ?"
463         }
464 
465       // build WHERE clause for column bounds
466       // if bounds don't exist (NULL), we count it as a match
467       val boundsWhereClause = columnBoundJoins.map { j =>
468         s"(${j.tableAlias}.file_id IS NULL OR ${j.whereClause})"
469       }
470 
471       // note: all child tables (partitions, column bounds) are pre-aggregated
472       // in subqueries. This ensures each file returns exactly one row with no Cartesian products.
473       // The sp_filter and cb_filter joins are used for filtering and must remain on the raw tables.
474       val whereClause = if (boundsWhereClause.isEmpty) {
475         "WHERE sf.root = ? AND sf.typeName = ?"
476       } else {
477         s"WHERE sf.root = ? AND sf.typeName = ? AND ${boundsWhereClause.mkString(" AND ")}"
478       }
479 
480       val query =
481         s"""SELECT sf.id, sf.file, sf.count, sf.action, sf.sort, sf.ts,
482            |  sp.partition_names, sp.partition_values,
483            |  cb.cb_attributes, cb.cb_lowers, cb.cb_uppers
484            |FROM $filesTable sf
485            |${joinClause.mkString("\n")}
486            |LEFT JOIN (
487            |  SELECT file_id,
488            |    array_agg(name ORDER BY name) as partition_names,
489            |    array_agg(value ORDER BY name) as partition_values
490            |  FROM $partitionsTable
491            |  GROUP BY file_id
492            |) sp ON sf.id = sp.file_id
493            |LEFT JOIN (
494            |  SELECT file_id,
495            |    array_agg(attribute ORDER BY attribute) as cb_attributes,
496            |    array_agg(lower ORDER BY attribute) as cb_lowers,
497            |    array_agg(upper ORDER BY attribute) as cb_uppers
498            |  FROM $columnBoundsTable
499            |  GROUP BY file_id
500            |) cb ON sf.id = cb.file_id
501            |$whereClause
502            |ORDER BY sf.id DESC""".stripMargin
503 
504       WithClose(cx.prepareStatement(query)) { st =>
505         var paramIndex = 1
506         // set partition join parameters
507         partitionJoins.foreach { join =>
508           paramIndex += join.apply(st, paramIndex)
509         }
510         // set column bound join parameters (attribute IDs in JOIN clauses)
511         columnBoundJoins.foreach { join =>
512           st.setInt(paramIndex, join.attribute)
513           paramIndex += 1
514         }
515         // set root parameter
516         st.setString(paramIndex, root)
517         paramIndex += 1
518         st.setString(paramIndex, typeName)
519         paramIndex += 1
520         // set column bound WHERE clause parameters
521         columnBoundJoins.foreach { join =>
522           paramIndex += join.apply(st, paramIndex)
523         }
524         WithClose(st.executeQuery())(toStorageFiles)
525       }
526     }

528     private def toStorageFiles(rs: ResultSet): Seq[StorageFile] = {
529       val result = Seq.newBuilder[StorageFile]
530 
531       while (rs.next()) {
532         val file = rs.getString(2)
533         val count = rs.getLong(3)
534         val action = rs.getString(4).charAt(0) match {
535           case 'A' => StorageMetadata.StorageFileAction.Append
536           case 'M' => StorageMetadata.StorageFileAction.Modify
537           case 'D' => StorageMetadata.StorageFileAction.Delete
538           case c => throw new IllegalStateException(s"Unknown action: $c")
539         }
540         val sort = Option(rs.getArray(5)).fold(Seq.empty[Int])(_.getArray.asInstanceOf[Array[Integer]].map(_.intValue()).toSeq)
541         val timestamp = rs.getTimestamp(6).toInstant.toEpochMilli
542 
543         val partitions = {
544           val names = Option(rs.getArray(7)).map(_.getArray.asInstanceOf[Array[String]]).getOrElse(Array.empty)
545           val values = Option(rs.getArray(8)).map(_.getArray.asInstanceOf[Array[String]]).getOrElse(Array.empty)
546           names.indices.map(i => PartitionKey(names(i), values(i))).toSet
547         }
548 
549         val bounds = {
550           val attributes = Option(rs.getArray(9)).map(_.getArray.asInstanceOf[Array[java.lang.Short]]).getOrElse(Array.empty)
551           val lowers = Option(rs.getArray(10)).map(_.getArray.asInstanceOf[Array[String]]).getOrElse(Array.empty)
552           val uppers = Option(rs.getArray(11)).map(_.getArray.asInstanceOf[Array[String]]).getOrElse(Array.empty)
553           attributes.indices.map(i => ColumnBounds(attributes(i).intValue(), lowers(i), uppers(i)))
554         }
555 
556         result += StorageFile(file, Partition(partitions), count, action, bounds, sort, timestamp)
557       }
558 
559       result.result()
560     }

562     private case class PartitionJoin(name: String, lower: String, upper: String, tableAlias: String) {
563       // partitions are lower-bound inclusive and upper-bound exclusive
564       def onClause: String = s"$tableAlias.name = ? AND $tableAlias.value >= ? AND $tableAlias.value < ?"
565       def apply(st: PreparedStatement, i: Int): Int = {
566         st.setString(i, name)
567         st.setString(i + 1, lower)
568         st.setString(i + 2, upper)
569         3
570       }
571     }
572 
573     // attribute bounds filter with OR logic for multiple bounds on the same attribute
574     private case class ColumnBoundsJoin(attribute: Int, bounds: Seq[ColumnBound], tableAlias: String) {
575       // generate WHERE clause with OR logic for all bounds
576       // checks if any of the bounds intersect with the file's attribute bounds
577       def whereClause: String = {
578         bounds.map { _ =>
579           s"($tableAlias.lower <= ? AND $tableAlias.upper >= ?)"
580         }.mkString(" OR ")
581       }
582 
583       def apply(st: PreparedStatement, i: Int): Int = {
584         var paramIndex = i
585         bounds.foreach { bound =>
586           // for intersection: file.lower <= filter.upper AND file.upper >= filter.lower
587           st.setString(paramIndex, bound.upper)
588           st.setString(paramIndex + 1, bound.lower)
589           paramIndex += 2
590         }
591         bounds.size * 2
592       }
593     }
