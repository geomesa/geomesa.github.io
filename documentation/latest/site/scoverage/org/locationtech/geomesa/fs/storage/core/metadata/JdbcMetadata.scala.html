29 /**
30  * Storage metadata implementation backed by a SQL database. Currently compatible with Postgres - other
31  * databases will likely have incompatibilities in the SQL syntax.
32  *
33  * Schema consists of five tables (note: `storage_` is the default table prefix, but may be customized):
34  *
35  * `storage_meta`
36  *
37  * * Holds the base metadata (simple feature type, partition scheme, encoding) as a serialized clob
38  *
39  * ** root varchar(256) not null
40  * ** typeName varchar(256) not null
41  * ** key varchar(256) not null,
42  * ** value text not null,
43  * ** primary key (root, typeName, key)
44  *
45  * `storage_files`
46  *
47  * * Holds file-level metadata for each storage file
48  *
49  * ** id bigint primary key (generated identity)
50  * ** root varchar(256) not null
51  * ** typeName varchar(256) not null
52  * ** file varchar(256) not null
53  * ** count bigint not null
54  * ** action char(1) not null (A=Append, M=Modify, D=Delete)
55  * ** sort integer[]
56  * ** ts timestamp without time zone not null
57  *
58  * `storage_partitions`
59  *
60  * * Holds partition key-value pairs for each file
61  *
62  * ** file_id bigint not null (foreign key to storage_files)
63  * ** name varchar(64) not null
64  * ** value varchar(64) not null
65  * ** primary key (file_id, name)
66  *
67  * `storage_spatial_bounds`
68  *
69  * * Holds spatial bounds for each file by attribute
70  *
71  * ** file_id bigint not null (foreign key to storage_files)
72  * ** attribute smallint not null
73  * ** x_min double precision
74  * ** x_max double precision
75  * ** y_min double precision
76  * ** y_max double precision
77  * ** primary key (file_id, attribute)
78  *
79  * `storage_attr_bounds`
80  *
81  * * Holds attribute bounds for each file by attribute
82  *
83  * ** file_id bigint not null (foreign key to storage_files)
84  * ** attribute smallint not null
85  * ** lower text
86  * ** upper text
87  * ** primary key (file_id, attribute)
88  *
89  * @param pool connection pool
90  * @param meta metadata table reference
91  * @param files files table reference
92  * @param namespace feature type namespace
93  **/

447     def select(cx: Connection, partitions: Seq[PartitionRange], columnBounds: Seq[ColumnOr]): Seq[StorageFile] = {
448 
449       // build query with multiple joins - one for each partition filter (AND logic)
450       val partitionJoins = partitions.zipWithIndex.map {
451         case (PartitionRange(name, lower, upper), i) => PartitionJoin(name, lower, upper, s"sp_filter_$i")
452       }
453 
454       // build column bound filters - one LEFT JOIN per attribute with OR logic for bounds
455       val columnBoundJoins = columnBounds.zipWithIndex.map {
456         case (ColumnOr(attribute, bounds), i) => ColumnBoundsJoin(attribute, bounds, s"cb_filter_$i")
457       }
458 
459       val joinClause =
460         partitionJoins.map { j =>
461           s"JOIN $partitionsTable ${j.tableAlias} ON sf.id = ${j.tableAlias}.file_id AND ${j.onClause}"
462         } ++
463         columnBoundJoins.map { j =>
464           s"LEFT JOIN $columnBoundsTable ${j.tableAlias} ON sf.id = ${j.tableAlias}.file_id AND ${j.tableAlias}.attribute = ?"
465         }
466 
467       // build WHERE clause for column bounds
468       // if bounds don't exist (NULL), we count it as a match
469       val boundsWhereClause = columnBoundJoins.map { j =>
470         s"(${j.tableAlias}.file_id IS NULL OR ${j.whereClause})"
471       }
472 
473       // note: all child tables (partitions, column bounds) are pre-aggregated
474       // in subqueries. This ensures each file returns exactly one row with no Cartesian products.
475       // The sp_filter and cb_filter joins are used for filtering and must remain on the raw tables.
476       val whereClause = if (boundsWhereClause.isEmpty) {
477         "WHERE sf.root = ? AND sf.typeName = ?"
478       } else {
479         s"WHERE sf.root = ? AND sf.typeName = ? AND ${boundsWhereClause.mkString(" AND ")}"
480       }
481 
482       val query =
483         s"""SELECT sf.id, sf.file, sf.count, sf.action, sf.sort, sf.ts,
484            |  sp.partition_names, sp.partition_values,
485            |  cb.cb_attributes, cb.cb_lowers, cb.cb_uppers
486            |FROM $filesTable sf
487            |${joinClause.mkString("\n")}
488            |LEFT JOIN (
489            |  SELECT file_id,
490            |    array_agg(name ORDER BY name) as partition_names,
491            |    array_agg(value ORDER BY name) as partition_values
492            |  FROM $partitionsTable
493            |  GROUP BY file_id
494            |) sp ON sf.id = sp.file_id
495            |LEFT JOIN (
496            |  SELECT file_id,
497            |    array_agg(attribute ORDER BY attribute) as cb_attributes,
498            |    array_agg(lower ORDER BY attribute) as cb_lowers,
499            |    array_agg(upper ORDER BY attribute) as cb_uppers
500            |  FROM $columnBoundsTable
501            |  GROUP BY file_id
502            |) cb ON sf.id = cb.file_id
503            |$whereClause
504            |ORDER BY sf.id DESC""".stripMargin
505 
506       WithClose(cx.prepareStatement(query)) { st =>
507         var paramIndex = 1
508         // set partition join parameters
509         partitionJoins.foreach { join =>
510           paramIndex += join.apply(st, paramIndex)
511         }
512         // set column bound join parameters (attribute IDs in JOIN clauses)
513         columnBoundJoins.foreach { join =>
514           st.setInt(paramIndex, join.attribute)
515           paramIndex += 1
516         }
517         // set root parameter
518         st.setString(paramIndex, root)
519         paramIndex += 1
520         st.setString(paramIndex, typeName)
521         paramIndex += 1
522         // set column bound WHERE clause parameters
523         columnBoundJoins.foreach { join =>
524           paramIndex += join.apply(st, paramIndex)
525         }
526         WithClose(st.executeQuery())(toStorageFiles)
527       }
528     }

530     private def toStorageFiles(rs: ResultSet): Seq[StorageFile] = {
531       val result = Seq.newBuilder[StorageFile]
532 
533       while (rs.next()) {
534         val file = rs.getString(2)
535         val count = rs.getLong(3)
536         val action = rs.getString(4).charAt(0) match {
537           case 'A' => StorageMetadata.StorageFileAction.Append
538           case 'M' => StorageMetadata.StorageFileAction.Modify
539           case 'D' => StorageMetadata.StorageFileAction.Delete
540           case c => throw new IllegalStateException(s"Unknown action: $c")
541         }
542         val sort = Option(rs.getArray(5)).fold(Seq.empty[Int])(_.getArray.asInstanceOf[Array[Integer]].map(_.intValue()).toSeq)
543         val timestamp = rs.getTimestamp(6).toInstant.toEpochMilli
544 
545         val partitions = {
546           val names = Option(rs.getArray(7)).map(_.getArray.asInstanceOf[Array[String]]).getOrElse(Array.empty)
547           val values = Option(rs.getArray(8)).map(_.getArray.asInstanceOf[Array[String]]).getOrElse(Array.empty)
548           names.indices.map(i => PartitionKey(names(i), values(i))).toSet
549         }
550 
551         val bounds = {
552           val attributes = Option(rs.getArray(9)).map(_.getArray.asInstanceOf[Array[java.lang.Short]]).getOrElse(Array.empty)
553           val lowers = Option(rs.getArray(10)).map(_.getArray.asInstanceOf[Array[String]]).getOrElse(Array.empty)
554           val uppers = Option(rs.getArray(11)).map(_.getArray.asInstanceOf[Array[String]]).getOrElse(Array.empty)
555           attributes.indices.map(i => ColumnBounds(attributes(i).intValue(), lowers(i), uppers(i)))
556         }
557 
558         result += StorageFile(file, Partition(partitions), count, action, bounds, sort, timestamp)
559       }
560 
561       result.result()
562     }

564     private case class PartitionJoin(name: String, lower: String, upper: String, tableAlias: String) {
565       // partitions are lower-bound inclusive and upper-bound exclusive
566       def onClause: String = s"$tableAlias.name = ? AND $tableAlias.value >= ? AND $tableAlias.value < ?"
567       def apply(st: PreparedStatement, i: Int): Int = {
568         st.setString(i, name)
569         st.setString(i + 1, lower)
570         st.setString(i + 2, upper)
571         3
572       }
573     }
574 
575     // attribute bounds filter with OR logic for multiple bounds on the same attribute
576     private case class ColumnBoundsJoin(attribute: Int, bounds: Seq[ColumnBound], tableAlias: String) {
577       // generate WHERE clause with OR logic for all bounds
578       // checks if any of the bounds intersect with the file's attribute bounds
579       def whereClause: String = {
580         bounds.map { _ =>
581           s"($tableAlias.lower <= ? AND $tableAlias.upper >= ?)"
582         }.mkString(" OR ")
583       }
584 
585       def apply(st: PreparedStatement, i: Int): Int = {
586         var paramIndex = i
587         bounds.foreach { bound =>
588           // for intersection: file.lower <= filter.upper AND file.upper >= filter.lower
589           st.setString(paramIndex, bound.upper)
590           st.setString(paramIndex + 1, bound.lower)
591           paramIndex += 2
592         }
593         bounds.size * 2
594       }
595     }
