31 /**
32  * Storage metadata implementation backed by a SQL database. Currently compatible with Postgres - other
33  * databases will likely have incompatibilities in the SQL syntax.
34  *
35  * Schema consists of five tables (note: `storage_` is the default table prefix, but may be customized):
36  *
37  * `storage_meta`
38  *
39  * * Holds the base metadata (simple feature type, partition scheme, encoding) as a serialized clob
40  *
41  * ** root varchar(256) not null
42  * ** typeName varchar(256) not null
43  * ** meta text not null
44  * ** primary key (root, typeName)
45  *
46  * `storage_files`
47  *
48  * * Holds file-level metadata for each storage file
49  *
50  * ** id bigint primary key (generated identity)
51  * ** root varchar(256) not null
52  * ** file varchar(256) not null
53  * ** count bigint not null
54  * ** action char(1) not null (A=Append, M=Modify, D=Delete)
55  * ** sort integer[]
56  * ** ts timestamp without time zone not null
57  *
58  * `storage_partitions`
59  *
60  * * Holds partition key-value pairs for each file
61  *
62  * ** file_id bigint not null (foreign key to storage_files)
63  * ** name varchar(64) not null
64  * ** value varchar(64) not null
65  * ** primary key (file_id, name)
66  *
67  * `storage_spatial_bounds`
68  *
69  * * Holds spatial bounds for each file by attribute
70  *
71  * ** file_id bigint not null (foreign key to storage_files)
72  * ** attribute smallint not null
73  * ** x_min double precision
74  * ** x_max double precision
75  * ** y_min double precision
76  * ** y_max double precision
77  * ** primary key (file_id, attribute)
78  *
79  * `storage_attr_bounds`
80  *
81  * * Holds attribute bounds for each file by attribute
82  *
83  * ** file_id bigint not null (foreign key to storage_files)
84  * ** attribute smallint not null
85  * ** lower text
86  * ** upper text
87  * ** primary key (file_id, attribute)
88  *
89  * @param pool connection pool
90  * @param root storage root path
91  * @param schema database schema name
92  * @param tablePrefix table name prefix
93  * @param meta basic metadata config
94  **/

180     val filters = partition.values.toSeq.map(p => PartitionRange(p.name, p.value, p.value + ZeroChar))
181     WithClose(pool.getConnection())(filesTable.select(_, root, filters, Seq.empty, Seq.empty))

184   override def getFiles(filter: Filter): Seq[StorageFile] = {
185     if (filter == Filter.INCLUDE) {
186       getFiles()
187     } else {
188       val filters = getFilters(filter)
189       if (filters.isEmpty) {
190         Seq.empty // no intersecting partitions
191       } else {
192         WithClose(pool.getConnection()) { cx =>
193           filters.flatMap { f =>
194             filesTable.select(cx, root, f.partitions, f.spatialBounds, f.attributeBounds)
195           }
196         }
197       }
198     }
199   }

270     def create(connection: Connection): Unit = {
271       WithClose(connection.createStatement()) { st =>
272         st.executeUpdate(
273           s"""create table if not exists $qualifiedTableName (

282     def insert(connection: Connection, root: String, metadata: Metadata): Unit = {
283       val serialized = new ByteArrayOutputStream()
284       MetadataSerialization.serialize(serialized, metadata)
285       WithClose(connection.prepareStatement(s"insert into $qualifiedTableName (root, typeName, meta) values (?, ?, ?)")) { st =>
286         st.setString(1, root)
287         st.setString(2, metadata.sft.getTypeName)
288         st.setString(3, new String(serialized.toByteArray, StandardCharsets.UTF_8))
289         st.executeUpdate()
290       }
291     }

294       val serialized = new ByteArrayOutputStream()

297         st.setString(1, new String(serialized.toByteArray, StandardCharsets.UTF_8))

301       }

306         st.setString(1, root)

337     private val filesTable = s""""$schema"."${tablePrefix}files""""

342     def create(cx: Connection): Unit = {
343       WithClose(cx.createStatement()) { st =>
344         st.executeUpdate(
345           s"""CREATE TABLE IF NOT EXISTS $filesTable (
346              |  id BIGINT PRIMARY KEY GENERATED BY DEFAULT AS IDENTITY,
347              |  root VARCHAR(256) NOT NULL,
348              |  file VARCHAR(256) NOT NULL,
349              |  count BIGINT NOT NULL,
350              |  action CHAR(1) NOT NULL,
351              |  sort INTEGER[],
352              |  ts TIMESTAMP WITHOUT TIME ZONE NOT NULL
353              |);""".stripMargin
354         )
355         st.executeUpdate(
356           s"""CREATE TABLE IF NOT EXISTS $partitionsTable (
357              |  file_id BIGINT NOT NULL,
358              |  name VARCHAR(64) NOT NULL,
359              |  value VARCHAR(64) NOT NULL,
360              |  PRIMARY KEY (file_id, name),
361              |  CONSTRAINT fk_storage_file
362              |    FOREIGN KEY (file_id)
363              |    REFERENCES storage_files(id)
364              |    ON DELETE CASCADE
365              |);""".stripMargin
366         )
367         st.executeUpdate(
368           s"""CREATE TABLE IF NOT EXISTS $spatialBoundsTable (
369              |  file_id BIGINT NOT NULL,
370              |  attribute SMALLINT NOT NULL,
371              |  x_min DOUBLE PRECISION,
372              |  x_max DOUBLE PRECISION,
373              |  y_min DOUBLE PRECISION,
374              |  y_max DOUBLE PRECISION,
375              |  PRIMARY KEY (file_id, attribute),
376              |  CONSTRAINT fk_storage_file
377              |    FOREIGN KEY (file_id)
378              |    REFERENCES storage_files(id)
379              |    ON DELETE CASCADE
380              |);""".stripMargin
381         )
382         st.executeUpdate(
383           s"""CREATE TABLE IF NOT EXISTS $attrBoundsTable (
384              |  file_id BIGINT NOT NULL,
385              |  attribute SMALLINT NOT NULL,
386              |  lower TEXT,
387              |  upper TEXT,
388              |  PRIMARY KEY (file_id, attribute),
389              |  CONSTRAINT fk_storage_file
390              |    FOREIGN KEY (file_id)
391              |    REFERENCES storage_files(id)
392              |    ON DELETE CASCADE
393              |);""".stripMargin

395         st.executeUpdate(s"""CREATE INDEX IF NOT EXISTS ${tablePrefix}partitions_idx_partition ON $partitionsTable(name, value);""")
396         st.executeUpdate(s"""CREATE INDEX IF NOT EXISTS ${tablePrefix}attr_bounds_idx_bounds ON $attrBoundsTable(attribute, lower, upper);""")
397         st.executeUpdate(s"""CREATE INDEX IF NOT EXISTS ${tablePrefix}spatial_bounds_idx_bounds ON $spatialBoundsTable(attribute, x_min, x_max, y_min, y_max);""")
398       }

401     def insert(cx: Connection, root: String, file: StorageFile): Unit = {
402       val id = WithClose(cx.prepareStatement(
403         s"INSERT INTO $filesTable (root, file, count, action, sort, ts) " +
404           "VALUES (?, ?, ?, ?, ?, ?) RETURNING id", Statement.RETURN_GENERATED_KEYS)) { st =>
405         st.setString(1, root)
406         st.setString(2, file.file)
407         st.setLong(3, file.count)
408         st.setString(4, file.action.toString.substring(0, 1))
409         if (file.sort.isEmpty) {
410           st.setNull(5, java.sql.Types.ARRAY)
411         } else {
412           st.setArray(5, cx.createArrayOf("integer", file.sort.map(Int.box).toArray[AnyRef]))
413         }
414         st.setTimestamp(6, Timestamp.from(Instant.ofEpochMilli(file.timestamp)))
415         st.executeUpdate()
416         WithClose(st.getGeneratedKeys) { rs =>
417           if (rs.next()) {
418             rs.getLong(1)
419           } else {
420             throw new RuntimeException("Failed to retrieve generated key")
421           }
422         }
423       }
424       WithClose(cx.prepareStatement(s"INSERT INTO $partitionsTable (file_id, name, value) VALUES (?, ?, ?)")) { st =>
425         file.partition.values.foreach { p =>
426           st.setLong(1, id)
427           st.setString(2, p.name)
428           st.setString(3, p.value)
429           st.executeUpdate()
430         }
431       }
432       if (file.spatialBounds.nonEmpty) {
433         WithClose(cx.prepareStatement(s"INSERT INTO $spatialBoundsTable (file_id, attribute, x_min, x_max, y_min, y_max) VALUES (?, ?, ?, ?, ?, ?)")) { st =>
434           file.spatialBounds.foreach { bounds =>
435             st.setLong(1, id)
436             st.setInt(2, bounds.attribute)
437             st.setDouble(3, bounds.xmin)
438             st.setDouble(4, bounds.xmax)
439             st.setDouble(5, bounds.ymin)
440             st.setDouble(6, bounds.ymax)
441             st.executeUpdate()
442           }
443         }
444       }
445       if (file.attributeBounds.nonEmpty) {
446         WithClose(cx.prepareStatement(s"INSERT INTO $attrBoundsTable (file_id, attribute, lower, upper) VALUES (?, ?, ?, ?)")) { st =>
447           file.attributeBounds.foreach { bounds =>
448             st.setLong(1, id)
449             st.setInt(2, bounds.attribute)
450             st.setString(3, bounds.lower)
451             st.setString(4, bounds.upper)
452             st.executeUpdate()
453           }
454         }
455       }
456     }

466     def select(
467         cx: Connection,
468         root: String,
469         partitions: Seq[PartitionRange],
470         spatialBounds: Seq[SpatialOr],
471         attributeBounds: Seq[AttributeOr],
472       ): Seq[StorageFile] = {
473 
474       // build query with multiple joins - one for each partition filter (AND logic)
475       val partitionJoins = partitions.zipWithIndex.map {
476         case (PartitionRange(name, lower, upper), i) => PartitionJoin(name, lower, upper, s"sp_filter_$i")
477       }
478 
479       // build spatial bound filters - one LEFT JOIN per attribute with OR logic for bounds
480       val spatialBoundJoins = spatialBounds.zipWithIndex.map {
481         case (SpatialOr(attribute, bounds), i) => SpatialBoundsJoin(attribute, bounds, s"sb_filter_$i")
482       }
483 
484       // build attribute bound filters - one LEFT JOIN per attribute with OR logic for bounds
485       val attrBoundJoins = attributeBounds.zipWithIndex.map {
486         case (AttributeOr(attribute, bounds), i) => AttributeBoundsJoin(attribute, bounds, s"ab_filter_$i")
487       }
488 
489       val joinClause =
490         partitionJoins.map { j =>
491           s"JOIN $partitionsTable ${j.tableAlias} ON sf.id = ${j.tableAlias}.file_id AND ${j.onClause}"
492         } ++
493         spatialBoundJoins.map { j =>
494           s"LEFT JOIN $spatialBoundsTable ${j.tableAlias} ON sf.id = ${j.tableAlias}.file_id AND ${j.tableAlias}.attribute = ?"
495         } ++
496         attrBoundJoins.map { j =>
497           s"LEFT JOIN $attrBoundsTable ${j.tableAlias} ON sf.id = ${j.tableAlias}.file_id AND ${j.tableAlias}.attribute = ?"
498         }
499 
500       // build WHERE clause for spatial and attribute bounds
501       // if bounds don't exist (NULL), we count it as a match
502       val spatialWhereClause = spatialBoundJoins.map { j =>
503         s"(${j.tableAlias}.file_id IS NULL OR ${j.whereClause})"
504       }
505       val attrWhereClause = attrBoundJoins.map { j =>
506         s"(${j.tableAlias}.file_id IS NULL OR ${j.whereClause})"
507       }
508       val allWhereClauses = spatialWhereClause ++ attrWhereClause
509 
510       // note: all child tables (partitions, spatial bounds, attribute bounds) are pre-aggregated
511       // in subqueries. This ensures each file returns exactly one row with no Cartesian products.
512       // The sp_filter, sb_filter, and ab_filter joins are used for filtering and must remain on the raw tables.
513       val whereClause = if (allWhereClauses.isEmpty) {
514         "WHERE sf.root = ?"
515       } else {
516         s"WHERE sf.root = ? AND ${allWhereClauses.mkString(" AND ")}"
517       }
518 
519       val query =
520         s"""SELECT sf.id, sf.file, sf.count, sf.action, sf.sort, sf.ts,
521            |  sp.partition_names, sp.partition_values,
522            |  sb.sb_attributes, sb.sb_x_mins, sb.sb_x_maxs, sb.sb_y_mins, sb.sb_y_maxs,
523            |  ab.ab_attributes, ab.ab_lowers, ab.ab_uppers
524            |FROM $filesTable sf
525            |${joinClause.mkString("\n")}
526            |LEFT JOIN (
527            |  SELECT file_id,
528            |    array_agg(name ORDER BY name) as partition_names,
529            |    array_agg(value ORDER BY name) as partition_values
530            |  FROM $partitionsTable
531            |  GROUP BY file_id
532            |) sp ON sf.id = sp.file_id
533            |LEFT JOIN (
534            |  SELECT file_id,
535            |    array_agg(attribute ORDER BY attribute) as sb_attributes,
536            |    array_agg(x_min ORDER BY attribute) as sb_x_mins,
537            |    array_agg(x_max ORDER BY attribute) as sb_x_maxs,
538            |    array_agg(y_min ORDER BY attribute) as sb_y_mins,
539            |    array_agg(y_max ORDER BY attribute) as sb_y_maxs
540            |  FROM $spatialBoundsTable
541            |  GROUP BY file_id
542            |) sb ON sf.id = sb.file_id
543            |LEFT JOIN (
544            |  SELECT file_id,
545            |    array_agg(attribute ORDER BY attribute) as ab_attributes,
546            |    array_agg(lower ORDER BY attribute) as ab_lowers,
547            |    array_agg(upper ORDER BY attribute) as ab_uppers
548            |  FROM $attrBoundsTable
549            |  GROUP BY file_id
550            |) ab ON sf.id = ab.file_id
551            |$whereClause
552            |ORDER BY sf.id DESC""".stripMargin
553 
554       WithClose(cx.prepareStatement(query)) { st =>
555         var paramIndex = 1
556         // set partition join parameters
557         partitionJoins.foreach { join =>
558           paramIndex += join.apply(st, paramIndex)
559         }
560         // set spatial bound join parameters (attribute IDs in JOIN clauses)
561         spatialBoundJoins.foreach { join =>
562           st.setInt(paramIndex, join.attribute)
563           paramIndex += 1
564         }
565         // set attribute bound join parameters (attribute IDs in JOIN clauses)
566         attrBoundJoins.foreach { join =>
567           st.setInt(paramIndex, join.attribute)
568           paramIndex += 1
569         }
570         // set root parameter
571         st.setString(paramIndex, root)
572         paramIndex += 1
573         // set spatial bound WHERE clause parameters
574         spatialBoundJoins.foreach { join =>
575           paramIndex += join.apply(st, paramIndex)
576         }
577         // set attribute bound WHERE clause parameters
578         attrBoundJoins.foreach { join =>
579           paramIndex += join.apply(st, paramIndex)
580         }
581         WithClose(st.executeQuery())(toStorageFiles)
582       }
583     }

585     private def toStorageFiles(rs: ResultSet): Seq[StorageFile] = {
586       val result = Seq.newBuilder[StorageFile]
587 
588       while (rs.next()) {
589         val file = rs.getString(2)
590         val count = rs.getLong(3)
591         val action = rs.getString(4).charAt(0) match {
592           case 'A' => StorageMetadata.StorageFileAction.Append
593           case 'M' => StorageMetadata.StorageFileAction.Modify
594           case 'D' => StorageMetadata.StorageFileAction.Delete
595           case c => throw new IllegalStateException(s"Unknown action: $c")
596         }
597         val sort = Option(rs.getArray(5)).fold(Seq.empty[Int])(_.getArray.asInstanceOf[Array[Integer]].map(_.intValue()).toSeq)
598         val timestamp = rs.getTimestamp(6).toInstant.toEpochMilli
599
