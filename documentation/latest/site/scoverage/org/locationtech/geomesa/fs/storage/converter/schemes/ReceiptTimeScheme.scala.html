25 /**
26  * Scheme for partitioning based on "receipt time", i.e. when a message is received. Generally this is useful
27  * only for reading existing data that may have been aggregated and stored by an external process.
28  *
29  * @param delegate delegate date time scheme options
30  * @param buffer amount of time to buffer queries by, in order to match a feature date to a receipt time date -
31  *               i.e. the amount of latency in the ingest process
32  */
33 case class ReceiptTimeScheme(delegate: DateTimeScheme, buffer: Duration) extends PartitionScheme {
34 
35   override val depth: Int = delegate.depth

37   override val pattern: String = delegate.pattern

39   override def getIntersectingPartitions(filter: Filter): Option[Seq[String]] =
40     delegate.getIntersectingPartitions(buffered(filter))

42   private def buffered(filter: Filter): Filter =
43     filter.accept(new BufferingFilterVisitor(buffer, delegate.dtg), null).asInstanceOf[Filter]
44 }

80     override def visit(filter: PropertyIsBetween, extraData: AnyRef): AnyRef = {
81       val factory = getFactory(extraData)
82       def buffer(p: PropertyName, lower: Literal, upper: Literal): Option[Filter] = {
83         for {
84           lo <- Option(FastConverter.evaluate(lower, classOf[Date]))
85           up <- Option(FastConverter.evaluate(upper, classOf[Date]))
86         } yield {
87           val bufferedLo = bufferDown(lo)
88           val bufferedUp = bufferUp(up)
89           // account for inverted filters that may result in invalid clauses after buffering
90           if (bufferedLo.before(bufferedUp)) {
91             factory.between(p, factory.literal(bufferedLo), factory.literal(bufferedUp), filter.getMatchAction)
92           } else {
93             Filter.EXCLUDE
94           }
95         }
96       }
97       val prop = visit(filter.getExpression, extraData)
98       val lowerBoundary = visit(filter.getLowerBoundary, extraData)
99       val upperBoundary = visit(filter.getUpperBoundary, extraData)
100       val buffered = (prop, lowerBoundary, upperBoundary) match {
101         case (p: PropertyName, lower: Literal, upper: Literal) if p.getPropertyName == dtg => buffer(p, lower, upper)
102         case _ => None
103       }
104 
105       buffered.getOrElse(super.visit(filter, extraData))
106     }

271     private def buffer(lit: Literal, offset: Long, extraData: AnyRef): Expression = {

273         case null => lit

276     }
