42   def run(
43       storage: FileSystemStorage,
44       metadataConfig: Option[Properties],
45       partitions: Seq[Partition],
46       targetFileSize: Option[Long],
47       tempPath: Option[Path],
48       libjarsFiles: Seq[String],
49       libjarsPaths: Iterator[() => Seq[File]],
50       statusCallback: StatusCallback): JobResult = {
51 
52     val job = Job.getInstance(new Configuration(storage.context.conf), "GeoMesa Storage Compaction")
53 
54     setLibJars(job, libjarsFiles, libjarsPaths)
55     job.setJarByClass(this.getClass)
56 
57     // InputFormat and Mappers
58     job.setInputFormatClass(classOf[PartitionInputFormat])
59     job.setMapperClass(classOf[CompactionMapper])
60 
61     // No reducers - Mapper will read/write its own things
62     job.setNumReduceTasks(0)
63 
64     job.setMapOutputKeyClass(classOf[Void])
65     job.setMapOutputValueClass(classOf[SimpleFeature])
66     job.setOutputKeyClass(classOf[Void])
67     job.setOutputValueClass(classOf[SimpleFeature])
68 
69     val qualifiedTempPath = tempPath.map(storage.context.fs.makeQualified)
70 
71     StorageConfiguration.setRootPath(job.getConfiguration, storage.context.root)
72     StorageConfiguration.setEncoding(job.getConfiguration, storage.encoding)
73     StorageConfiguration.setSft(job.getConfiguration, storage.metadata.sft)
74     StorageConfiguration.setPartitions(job.getConfiguration, partitions)
75     StorageConfiguration.setFileType(job.getConfiguration, FileType.Compacted)
76     StorageConfiguration.setMetadataType(job.getConfiguration, storage.metadata.`type`)
77     metadataConfig.foreach(c => StorageConfiguration.setMetadataConfig(job.getConfiguration, c.asScala.toMap))
78     targetFileSize.foreach(StorageConfiguration.setTargetFileSize(job.getConfiguration, _))
79 
80     FileOutputFormat.setOutputPath(job, qualifiedTempPath.getOrElse(storage.context.root))
81 
82     // MapReduce options
83     job.getConfiguration.set("mapred.map.tasks.speculative.execution", "false")
84     job.getConfiguration.set("mapreduce.job.user.classpath.first", "true")
85 
86     configureOutput(storage.metadata.sft, job)
87 
88     // save the existing files so we can delete them afterwards
89     // mimic the filtering done in PartitionInputFormat
90     val sizeable = Option(storage).collect { case s: SizeableFileSystemStorage => s }
91     val sizeCheck = sizeable.flatMap(s => s.targetSize(targetFileSize).map(t => (p: Path) => s.fileIsSized(p, t)))
92     val existingDataFiles = partitions.toList.flatMap { p =>
93       val files = storage.metadata.getFiles(p).filterNot(f => sizeCheck.exists(_.apply(new Path(storage.context.root, f.file))))
94       // TODO get counts right... use m/r counters?
95       if (files.isEmpty) { None } else { Some(p -> files) }
96     }
97 
98     def mapCounters = Seq((MappedCounter, written(job)), (FailedCounter, failed(job)))
99 
100     val result = JobRunner.run(job, statusCallback, mapCounters, Seq.empty).merge {
101       qualifiedTempPath.map { tp =>
102         new DistributedCopy().copy(Seq(tp), storage.context.root, statusCallback)
103       }
104     }
105 
106     result match {
107       case JobSuccess(message, counts) =>
108         if (message.nonEmpty) {
109           Command.user.info(message)
110         }
111         Command.user.info("Removing old files")
112         existingDataFiles.foreach { case (partition, files) =>
113           val name = partition.values.map(_.name).mkString(",")
114           val counter = StorageConfiguration.Counters.partition(name)
115           val count = Option(job.getCounters.findCounter(StorageConfiguration.Counters.Group, counter)).map(_.getValue)
116           files.foreach { f =>
117             storage.metadata.removeFile(f)
118             storage.context.fs.delete(new Path(storage.context.root, f.file), false)
119           }
120           val removed = count.map(c => s"containing $c features ").getOrElse("")
121           Command.user.info(s"Removed ${TextTools.getPlural(files.size, "file")} ${removed}in partition $name")
122         }
123         JobSuccess("", counts)
124 
125       case j => j
126     }
127   }
128 
129   private def written(job: Job): Long =
130     job.getCounters.findCounter(OutputCounters.Group, OutputCounters.Written).getValue
131 
132   private def failed(job: Job): Long =
133     job.getCounters.findCounter(OutputCounters.Group, OutputCounters.Failed).getValue
134 }

161       sf.getUserData.put(Hints.USE_PROVIDED_FID, java.lang.Boolean.TRUE)

165     }
