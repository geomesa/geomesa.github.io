34 trait FileSystemCompactionJob extends StorageConfiguration with JobWithLibJars {
35 
36   import FileSystemCompactionJob.{FailedCounter, MappedCounter}
37 
38   def run(
39       storage: FileSystemStorage,
40       partitions: Seq[Partition],
41       tempPath: Option[Path],
42       libjarsFiles: Seq[String],
43       libjarsPaths: Iterator[() => Seq[File]],
44       statusCallback: StatusCallback): JobResult = {
45 
46     val job = {
47       val conf = new Configuration()
48       storage.context.conf.foreach { case (k, v) => conf.set(k, v) }
49       Job.getInstance(conf, "GeoMesa Storage Compaction")
50     }
51 
52     setLibJars(job, libjarsFiles, libjarsPaths)
53     job.setJarByClass(this.getClass)
54 
55     // InputFormat and Mappers
56     job.setInputFormatClass(classOf[ParquetPartitionInputFormat])
57     job.setMapperClass(classOf[CompactionMapper])
58 
59     // No reducers - Mapper will read/write its own things
60     job.setNumReduceTasks(0)
61 
62     job.setMapOutputKeyClass(classOf[Void])
63     job.setMapOutputValueClass(classOf[SimpleFeature])
64     job.setOutputKeyClass(classOf[Void])
65     job.setOutputValueClass(classOf[SimpleFeature])
66 
67     StorageConfiguration.setRootPath(job.getConfiguration, storage.context.root)
68     StorageConfiguration.setSft(job.getConfiguration, storage.sft)
69     StorageConfiguration.setPartitions(job.getConfiguration, partitions)
70     storage.sizer.targetSize.foreach(StorageConfiguration.setTargetFileSize(job.getConfiguration, _))
71 
72     FileOutputFormat.setOutputPath(job, tempPath.getOrElse(new Path(S3ObjectStore.s3aUri(storage.context.root))))
73 
74     // MapReduce options
75     job.getConfiguration.set("mapred.map.tasks.speculative.execution", "false")
76     job.getConfiguration.set("mapreduce.job.user.classpath.first", "true")
77 
78     configureOutput(storage.sft, job)
79 
80     // save the existing files so we can delete them afterward
81     // mimic the filtering done in PartitionInputFormat
82     val sizeCheck = storage.sizer.targetSize.map(t => (p: DataFile) => storage.sizer.fileIsSized(p, t))
83     val existingDataFiles = partitions.toList.flatMap { p =>
84       val files = storage.metadata.files().forPartition(p).scan().filterNot(f => sizeCheck.exists(_.apply(f)))
85       // TODO get counts right... use m/r counters?
86       if (files.isEmpty) { None } else { Some(p -> files) }
87     }
88 
89     def mapCounters: Seq[(String, Long)] = Seq((MappedCounter, written(job)), (FailedCounter, failed(job)))
90 
91     val result = JobRunner.run(job, statusCallback, mapCounters, Seq.empty).merge {
92       tempPath.map { tp =>
93         new DistributedCopy().copy(Seq(tp), new Path(storage.context.root), statusCallback)
94       }
95     }
96 
97     result match {
98       case JobSuccess(message, counts) =>
99         if (message.nonEmpty) {
100           Command.user.info(message)
101         }
102         Command.user.info("Removing old files")
103         existingDataFiles.foreach { case (partition, files) =>
104           val name = partition.values.map(_.name).mkString(",")
105           val counter = StorageConfiguration.Counters.partition(name)
106           val count = Option(job.getCounters.findCounter(StorageConfiguration.Counters.Group, counter)).map(_.getValue)
107           val delete = storage.table.newDelete()
108           files.foreach(delete.deleteFile)
109           delete.commit()
110           files.foreach(f => storage.table.io().deleteFile(f.location()))
111           val removed = count.map(c => s"containing $c features ").getOrElse("")
112           Command.user.info(s"Removed ${TextTools.getPlural(files.size, "file")} ${removed}in partition $name")
113         }
114         JobSuccess("", counts)
115 
116       case j => j
117     }
118   }
119 
120   private def written(job: Job): Long =
121     job.getCounters.findCounter(OutputCounters.Group, OutputCounters.Written).getValue
122 
123   private def failed(job: Job): Long =
124     job.getCounters.findCounter(OutputCounters.Group, OutputCounters.Failed).getValue
125 }

134   /**
135     * Mapper that simply reads the input format and writes the output to the sample node. This mapper
136     * is paired with the PartitionRecordReader which will feed all the features into a single map task
137     */
138   class CompactionMapper extends Mapper[Void, SimpleFeature, Void, SimpleFeature] with LazyLogging {
139 
140     type Context = Mapper[Void, SimpleFeature, Void, SimpleFeature]#Context
141 
142     private var written: Counter = _
143     private var mapped: Counter = _
144 
145     override def setup(context: Context): Unit = {
146       super.setup(context)
147       written = context.getCounter(OutputCounters.Group, OutputCounters.Written)
148       mapped = context.getCounter("org.locationtech.geomesa.fs.compaction", MappedCounter)
149     }
150 
151     override def map(key: Void, sf: SimpleFeature, context: Context): Unit = {
152       sf.getUserData.put(Hints.USE_PROVIDED_FID, java.lang.Boolean.TRUE)
153       mapped.increment(1)
154       context.write(null, sf)
155       written.increment(1)
156     }
157   }
