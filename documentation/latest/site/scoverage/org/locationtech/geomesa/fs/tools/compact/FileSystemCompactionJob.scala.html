35 trait FileSystemCompactionJob extends StorageConfiguration with JobWithLibJars {
36 
37   import FileSystemCompactionJob.{FailedCounter, MappedCounter}
38 
39   def run(
40       storage: FileSystemStorage,
41       partitions: Seq[Partition],
42       tempPath: Option[Path],
43       libjarsFiles: Seq[String],
44       libjarsPaths: Iterator[() => Seq[File]],
45       statusCallback: StatusCallback): JobResult = {
46 
47     val job = {
48       val conf = new Configuration()
49       storage.context.conf.foreach { case (k, v) => conf.set(k, v) }
50       S3ObjectStore.s3aConfigs(storage.context.conf).foreach { case (k, v) => conf.set(k, v) }
51       Job.getInstance(conf, "GeoMesa Storage Compaction")
52     }
53 
54     setLibJars(job, libjarsFiles, libjarsPaths)
55     job.setJarByClass(this.getClass)
56 
57     // InputFormat and Mappers
58     job.setInputFormatClass(classOf[PartitionInputFormat])
59     job.setMapperClass(classOf[CompactionMapper])
60 
61     // No reducers - Mapper will read/write its own things
62     job.setNumReduceTasks(0)
63 
64     job.setMapOutputKeyClass(classOf[Void])
65     job.setMapOutputValueClass(classOf[SimpleFeature])
66     job.setOutputKeyClass(classOf[Void])
67     job.setOutputValueClass(classOf[SimpleFeature])
68 
69     StorageConfiguration.setRootPath(job.getConfiguration, storage.context.root)
70     StorageConfiguration.setEncoding(job.getConfiguration, storage.encoding)
71     StorageConfiguration.setSft(job.getConfiguration, storage.metadata.sft)
72     StorageConfiguration.setPartitions(job.getConfiguration, partitions)
73     StorageConfiguration.setFileType(job.getConfiguration, FileType.Compacted)
74     storage.sizer.targetSize.foreach(StorageConfiguration.setTargetFileSize(job.getConfiguration, _))
75 
76     FileOutputFormat.setOutputPath(job, tempPath.getOrElse(new Path(S3ObjectStore.s3aUri(storage.context.root))))
77 
78     // MapReduce options
79     job.getConfiguration.set("mapred.map.tasks.speculative.execution", "false")
80     job.getConfiguration.set("mapreduce.job.user.classpath.first", "true")
81 
82     configureOutput(storage.metadata.sft, job)
83 
84     // save the existing files so we can delete them afterward
85     // mimic the filtering done in PartitionInputFormat
86     val sizeCheck = storage.sizer.targetSize.map(t => (p: URI) => storage.sizer.fileIsSized(p, t))
87     val existingDataFiles = partitions.toList.flatMap { p =>
88       val files = storage.metadata.getFiles(p).filterNot(f => sizeCheck.exists(_.apply(storage.context.root.resolve(f.file))))
89       // TODO get counts right... use m/r counters?
90       if (files.isEmpty) { None } else { Some(p -> files) }
91     }
92 
93     def mapCounters = Seq((MappedCounter, written(job)), (FailedCounter, failed(job)))
94 
95     val result = JobRunner.run(job, statusCallback, mapCounters, Seq.empty).merge {
96       tempPath.map { tp =>
97         new DistributedCopy().copy(Seq(tp), new Path(storage.context.root), statusCallback)
98       }
99     }
100 
101     result match {
102       case JobSuccess(message, counts) =>
103         if (message.nonEmpty) {
104           Command.user.info(message)
105         }
106         Command.user.info("Removing old files")
107         existingDataFiles.foreach { case (partition, files) =>
108           val name = partition.values.map(_.name).mkString(",")
109           val counter = StorageConfiguration.Counters.partition(name)
110           val count = Option(job.getCounters.findCounter(StorageConfiguration.Counters.Group, counter)).map(_.getValue)
111           files.foreach { f =>
112             storage.metadata.removeFile(f)
113             storage.fs.delete(storage.context.root.resolve(f.file))
114           }
115           val removed = count.map(c => s"containing $c features ").getOrElse("")
116           Command.user.info(s"Removed ${TextTools.getPlural(files.size, "file")} ${removed}in partition $name")
117         }
118         JobSuccess("", counts)
119 
120       case j => j
121     }
122   }
123 
124   private def written(job: Job): Long =
125     job.getCounters.findCounter(OutputCounters.Group, OutputCounters.Written).getValue
126 
127   private def failed(job: Job): Long =
128     job.getCounters.findCounter(OutputCounters.Group, OutputCounters.Failed).getValue
129 }
