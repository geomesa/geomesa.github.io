23 /**
24  * Creates a tag containing the base64 encoded summary visibility for the observed file
25  *
26  * @param path file path
27  * @param fs s3 client
28  * @param tag tag name to use
29  */
30 class S3VisibilityObserver(path: URI, fs: S3ObjectStore, tag: String) extends FileSystemObserver {
31 
32   private val visibilities = scala.collection.mutable.Set.empty[String]
33 
34   private val key = S3ObjectStore.parseS3Path(path)
35 
36   override def apply(feature: SimpleFeature): Unit = {
37     val vis = SecurityUtils.getVisibility(feature)
38     if (vis != null) {
39       visibilities.add(vis)
40     }
41   }
42 
43   override def flush(): Unit = {}
44 
45   override def close(): Unit = {
46     try { makeTagRequest(key.bucket, key.key) } catch {
47       case e: Exception => throw new IOException("Error tagging object", e)
48     }
49   }
50 
51   private def makeTagRequest(bucket: String, key: String): Unit = {
52     if (visibilities.nonEmpty) {
53       val vis = visibilities.mkString("(", ")&(", ")")
54       // this call simplifies and de-duplicates the expression
55       val expression = AccessExpression.of(vis, /*normalize = */true).getExpression
56       val visibility = Base64.getEncoder.encodeToString(expression.getBytes(StandardCharsets.UTF_8))
57       fs.putObjectTagging(bucket, key, Seq(tag -> visibility))
58     }
59   }
60 }
