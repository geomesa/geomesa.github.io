25 /**
26  * Creates a tag containing the base64 encoded summary visibility for the observed file
27  *
28  * @param path file path
29  * @param s3 s3 client
30  * @param tag tag name to use
31  */
32 class S3VisibilityObserver(path: URI, s3: S3AsyncClient, tag: String) extends FileSystemObserver {
33 
34   private val visibilities = scala.collection.mutable.Set.empty[String]
35 
36   private val key = S3ObjectStore.parseS3Path(path)
37 
38   override def apply(feature: SimpleFeature): Unit = {
39     val vis = SecurityUtils.getVisibility(feature)
40     if (vis != null) {
41       visibilities.add(vis)
42     }
43   }
44 
45   override def flush(): Unit = {}
46 
47   override def close(): Unit = {
48     try { makeTagRequest(key.bucket, key.key) } catch {
49       case e: Exception => throw new IOException("Error tagging object", e)
50     }
51   }
52 
53   private def makeTagRequest(bucket: String, key: String): Unit = {
54     if (visibilities.nonEmpty) {
55       val vis = visibilities.mkString("(", ")&(", ")")
56       // this call simplifies and de-duplicates the expression
57       val expression = AccessExpression.of(vis, /*normalize = */true).getExpression
58       val visibility = Base64.getEncoder.encodeToString(expression.getBytes(StandardCharsets.UTF_8))
59       val tagging = Tagging.builder().tagSet(Tag.builder.key(tag).value(visibility).build()).build()
60       val request = PutObjectTaggingRequest.builder.bucket(bucket).key(key).tagging(tagging).build()
61       s3.putObjectTagging(request).join()
62     }
63   }
64 }
