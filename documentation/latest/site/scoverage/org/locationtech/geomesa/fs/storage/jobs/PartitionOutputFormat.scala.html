26 /**
27   * Output format that writes to multiple partition files
28   *
29   * @param delegate underlying output format for a single file
30   */
31 class PartitionOutputFormat(delegate: SingleFileOutputFormat) extends OutputFormat[Void, SimpleFeature] {
32 
33   override def getRecordWriter(context: TaskAttemptContext): RecordWriter[Void, SimpleFeature] =
34     new PartitionSchemeRecordWriter(context)
35 
36   override def getOutputCommitter(context: TaskAttemptContext): OutputCommitter =
37     delegate.getOutputCommitter(context)
38 
39   // same as FileOutputFormat, but doesn't require that output directory doesn't exist
40   override def checkOutputSpecs(job: JobContext): Unit = {
41     // Ensure that the output directory is set
42     val outDir = FileOutputFormat.getOutputPath(job)
43     if (outDir == null) {
44       throw new InvalidJobConfException("Output directory not set")
45     }
46     // get delegation token for outDir's file system
47     TokenCache.obtainTokensForNamenodes(job.getCredentials, Array[Path](outDir), job.getConfiguration)
48   }

50   private class PartitionSchemeRecordWriter(context: TaskAttemptContext)
51       extends RecordWriter[Void, SimpleFeature] with LazyLogging {
52 
53     import StorageConfiguration.Counters.{Features, Group}
54 
55     private val fsc = {
56       val hadoopConf = context.getConfiguration
57       val conf = {
58         val builder = Map.newBuilder[String, String]
59         hadoopConf.forEach(e => builder += e.getKey -> hadoopConf.get(e.getKey)) // use .get to resolve envs
60         builder.result()
61       }
62       val root = StorageConfiguration.getRootPath(hadoopConf)
63       FileSystemContext.create(root, conf)
64     }
65 
66     private val storage = {
67       val hadoopConf = context.getConfiguration
68       val encoding = StorageConfiguration.getEncoding(hadoopConf)
69       val catalog = StorageMetadataCatalog(fsc)
70       FileSystemStorageFactory(encoding).apply(fsc, catalog.load(StorageConfiguration.getSftName(hadoopConf)))
71     }
72 
73     private val fileType = StorageConfiguration.getFileType(context.getConfiguration)
74     private val fileSize = StorageConfiguration.getTargetFileSize(context.getConfiguration)
75 
76     private val counter = context.getCounter(Group, Features)
77     private val cache = scala.collection.mutable.Map.empty[Set[PartitionKey], PartitionState]
78 
79     private val workPath = delegate.getOutputCommitter(context).asInstanceOf[FileOutputCommitter].getWorkPath
80 
81     private def newState(partitions: Set[PartitionKey]): PartitionState = {
82       val estimator = fileSize.map(storage.sizer.estimator)
83       estimator match {
84         case None => new SinglePartitionState(Partition(partitions))
85         case Some(e) => new ChunkedPartitionState(Partition(partitions), e, context)
86       }
87     }
88 
89     override def write(key: Void, value: SimpleFeature): Unit = {
90       val partition = storage.metadata.schemes.map(_.getPartition(value))
91       val state = cache.getOrElseUpdate(partition, newState(partition))
92       state.write(key, value)
93       counter.increment(1)
94     }
95 
96     override def close(context: TaskAttemptContext): Unit = {
97       cache.foreach { case (partition, state) =>
98         logger.debug(s"Closing writer for $partition")
99         state.close(context)
100         state.getFiles().foreach(storage.metadata.addFile)
101       }
102       CloseWithLogging(storage)
103     }
104 
105     private sealed abstract class PartitionState(partition: Partition) {
106 
107       import scala.collection.JavaConverters._
108 
109       private val files = new CopyOnWriteArrayList[StorageFile]()
110       private val counter = context.getCounter(Group, StorageConfiguration.Counters.partition(partition.toString))
111 
112       private var observer: StorageFileObserver = _
113       private var currentFile: String = _
114 
115       // noinspection AccessorLikeMethodIsEmptyParen
116       def getFiles(): Seq[StorageFile] = files.asScala.toSeq
117 
118       def write(key: Void, value: SimpleFeature): Unit = observer.apply(value)
119 
120       def close(context: TaskAttemptContext): Unit = closeCurrentFile()
121 
122       protected def newWriter(): (Path, RecordWriter[Void, SimpleFeature]) = {
123         closeCurrentFile()
124         currentFile = FileSystemStorage.newFilePath(storage.metadata.sft.getTypeName, fileType, "parquet")
125         observer = new StorageFileObserver(storage.metadata.sft)
126         val path = new Path(workPath, currentFile)
127         logger.debug(s"Creating record writer at path $path")
128         // noinspection LanguageFeature
129         (path, delegate.getRecordWriter(context, path))
130       }
131 
132       private def closeCurrentFile(): Unit = {
133         if (observer != null) {
134           observer.close()
135           val file = observer.file(currentFile, partition, StorageFileAction.Append)
136           files.add(file)
137           observer = null
138           counter.increment(file.count)
139         }
140       }
141     }

143     private class SinglePartitionState(partition: Partition) extends PartitionState(partition) {
144 
145       private val writer = newWriter()._2
146 
147       override def write(key: Void, value: SimpleFeature): Unit = {
148         writer.write(key, value)
149         super.write(key, value)

158     private class ChunkedPartitionState(partition: Partition, estimator: FileSizeEstimator, context: TaskAttemptContext)
159         extends PartitionState(partition) {
160 
161       private var count = 0L // number of features written
162       private var total = 0L // sum size of all finished chunks
163       private var remaining = estimator.estimate(0L)
164 
165       logger.debug(s"Initial estimate: $remaining")
166 
167       private var path: Path = _
168       private var writer: RecordWriter[Void, SimpleFeature] = _
169 
170       override def write(key: Void, value: SimpleFeature): Unit = {
171         if (writer == null) {
172           val (p, w) = newWriter()
173           path = p
174           writer = w
175         }
176         writer.write(key, value)
177         count += 1
178         remaining -= 1
179         if (remaining == 0) {
180           // TODO if we can access the underlying parquet writer we can check the size without closing the file
181           writer.close(context)
182           val length = storage.fs.size(path.toUri)
183           logger.debug(s"File length: $length $path")
184           writer = null
185           // adjust our estimate to account for the actual bytes written
186           total += length
187           estimator.update(total, count)
188           remaining = estimator.estimate(0L)
189           logger.debug(s"New estimate: $remaining")
190         }
191         super.write(key, value)
192       }
193 
194       override def close(context: TaskAttemptContext): Unit = {
195         if (writer != null) {
196           writer.close(context)
197         }
198         super.close(context)
199       }
200     }
201   }
