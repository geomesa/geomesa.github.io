25 abstract class AttributeScheme[T: ClassTag](
26     attribute: String,
27     index: Int,
28     defaultValue: Option[T],
29     allowedValues: Seq[T],
30     nullValue: T,
31     bucketing: Option[Bucketing[T]],
32     lexicoder: TypeEncoder[T, String],
33   ) extends PartitionScheme {
34 
35   override val name: String = {
36     val opts = new StringBuilder(s"${AttributeScheme.Name}:attribute=$attribute")
37     bucketing.foreach(b => opts.append(':').append(b.encoded))
38     defaultValue.filter(_ != nullValue).foreach(v => opts.append(':').append(s"default=$v"))
39     allowedValues.foreach(v => opts.append(':').append(s"allow=$v"))
40     opts.toString()
41   }
42 
43   private val clas = implicitly[ClassTag[T]].runtimeClass.asInstanceOf[Class[T]]
44   private val default = PartitionKey(name, toPartition(defaultValue.getOrElse(nullValue)))
45   private val allowed = if (allowedValues.isEmpty) { Seq.empty } else { (allowedValues.map(toPartition) :+ default.value).distinct }
46 
47   override def getPartition(feature: SimpleFeature): PartitionKey = {
48     val value = feature.getAttribute(index)
49     if (value == null) {
50       return default
51     }
52     val partition = toPartition(value.asInstanceOf[T])
53     if (allowed.isEmpty || allowed.contains(partition)) {
54       PartitionKey(name, partition)
55     } else {
56       default
57     }
58   }
59 
60   override def getRangesForFilter(filter: Filter): Option[Seq[PartitionRange]] = {
61     getBounds(filter).map { bounds =>
62       val builder = new RangeBuilder()
63       bounds.foreach { bound =>
64         val lower = bound.lower.value.fold("")(toPartition)
65         val upper = exclusiveUpperBound(bound).getOrElse(AttributeScheme.UnboundedUpper)
66         builder += PartitionRange(name, lower, upper)
67       }
68       val all = builder.result()
69       if (allowed.isEmpty || all.isEmpty) { all } else {
70         allowed.collect { case v if all.exists(_.contains(v)) => PartitionRange(name, v, v + ZeroChar) }
71       }
72     }
73   }
74 
75   override def getPartitionsForFilter(filter: Filter): Option[Seq[PartitionKey]] = {
76     getBounds(filter).map { bounds =>
77       bounds.flatMap { bound =>
78         if (!bound.isBoundedBothSides) {
79           throw new IllegalArgumentException(s"Filter does not constrain partitions for scheme $name: ${ECQL.toCQL(filter)} ")
80         }
81         enumerate(bound)
82       }
83     }
84   }
85 
86   protected def enumerate(bounds: Bounds[T]): Seq[PartitionKey]
87 
88   protected def getBounds(filter: Filter): Option[Seq[Bounds[T]]] = {
89     val bounds = FilterHelper.extractAttributeBounds(filter, attribute, clas)
90     if (bounds.isEmpty) {
91       None
92     } else if (bounds.disjoint) {
93       Some(Seq.empty)
94     } else {
95       Some(bounds.values)
96     }
97   }
98 
99   protected def toPartition(value: T): String = {
100     bucketing match {
101       case None => lexicoder.encode(value)
102       case Some(b) => lexicoder.encode(b(value))
103     }
104   }
105 
106   protected def exclusiveUpperBound(bounds: Bounds[T]): Option[String] = {
107     bounds.upper.value.map { v =>
108       val encoded = toPartition(v)
109       if (bounds.upper.exclusive && bucketing.isEmpty) {
110         encoded
111       } else {
112         encoded + ZeroChar
113       }
114     }
115   }
116 }

138   class AttributePartitionSchemeFactory extends PartitionSchemeFactory {
139     override def load(sft: SimpleFeatureType, scheme: String): Option[PartitionScheme] = {
140       val opts = SchemeOpts(scheme)
141       if (opts.name != Name) { None } else {
142         val attribute = opts.getSingle("attribute").orNull
143         require(attribute != null, s"Attribute scheme requires an attribute to be specified with 'attribute=<attribute>'")
144         val index = attributeIndex(sft, attribute)
145         val binding = sft.getDescriptor(index).getType.getBinding
146         require(AttributeIndexKey.encodable(binding), s"Invalid type binding '${binding.getName}' of attribute '$attribute'")
147 
148         val width = opts.getSingle("width").map(w => WidthBucketing(w.toInt))
149         val divisor = opts.getSingle("divisor").map(_.toInt)
150         val scale = opts.getSingle("scale").map(_.toInt)
151 
152         val isString = classOf[String].isAssignableFrom(binding)
153         val isWholeNumber = binding == classOf[Integer] || binding == classOf[java.lang.Long]
154         val isDecimalNumber = binding == classOf[java.lang.Float] || binding == classOf[java.lang.Double]
155 
156         if (width.isDefined && !isString) {
157           throw new IllegalArgumentException(
158             s"'width' option is only supported for String-type attributes, not ${binding.getSimpleName}")
159         } else if (divisor.isDefined && !isWholeNumber) {
160           throw new IllegalArgumentException(
161             s"'divisor' option is only supported for Integer and Long-type attributes, not ${binding.getSimpleName}")
162         } else if (scale.isDefined && !isDecimalNumber) {
163           throw new IllegalArgumentException(
164             s"'scale' option is only supported for Float and Double-type attributes, not ${binding.getSimpleName}")
165         }
166 
167         val allowedValues = opts.getMulti("allow")
168         val defaultValue = opts.getSingle("default")
169         require(allowedValues.isEmpty || defaultValue.forall(allowedValues.contains),
170           "Default partition must be one of the allowed values")
171 
172         if (isString) {
173           Some(StringScheme(attribute, index, width, defaultValue, allowedValues))
174         } else if (binding == classOf[Integer]) {
175           val bucketing = divisor.map(IntegralBucketing.apply[Int])
176           Some(IntScheme(attribute, index, bucketing, defaultValue.map(_.toInt), allowedValues.map(_.toInt)))
177         } else if (binding == classOf[java.lang.Long]) {
178           val bucketing = divisor.map(d => IntegralBucketing(d.toLong))
179           Some(LongScheme(attribute, index, bucketing, defaultValue.map(_.toLong), allowedValues.map(_.toLong)))
180         } else if (binding == classOf[java.lang.Float]) {
181           val bucketing = scale.map(FractionalBucketing.apply[Float])
182           Some(FloatScheme(attribute, index, bucketing, defaultValue.map(_.toFloat), allowedValues.map(_.toFloat)))
183         } else if (binding == classOf[java.lang.Double]) {
184           val bucketing = scale.map(FractionalBucketing.apply[Double])
185           Some(DoubleScheme(attribute, index, bucketing, defaultValue.map(_.toDouble), allowedValues.map(_.toDouble)))
186         } else {
187           throw new IllegalArgumentException(
188             s"Attribute scheme is not supported for type ${binding.getSimpleName} - " +
189               s"supported types are String Integer, Long, Float, and Double")
190         }
191       }
192     }

215   private case class IntegralBucketing[T: Integral](divisor: T) extends Bucketing[T] {

221   // scale here refers to the number of digits to the right of the decimal place that are kept
222   private case class FractionalBucketing[T: Fractional](scale: Int) extends Bucketing[T] {
223     import Fractional.Implicits.infixFractionalOps
224     private val fractional = implicitly[Fractional[T]]
225     val scaleT: T = fractional.fromInt(math.pow(10, scale).toInt)
226     override def apply(value: T): T = fractional.fromInt(Math.floor((value * scaleT).toDouble).toInt) / scaleT
227     override def encoded: String = s"scale=$scale"

267   private abstract class IntegralScheme[T: Integral: ClassTag](
268       attribute: String,
269       index: Int,
270       divisor: Option[IntegralBucketing[T]],
271       defaultValue: Option[T],
272       allowedValues: Seq[T],
273       lexicoder: TypeEncoder[T, String],
274     ) extends AttributeScheme[T](attribute, index, defaultValue, allowedValues, implicitly[Integral[T]].zero, divisor, lexicoder) {
275 
276     import Integral.Implicits.infixIntegralOps
277 
278     private val integral = implicitly[Integral[T]]
279 
280     override def getCoveringFilter(partition: PartitionKey): Filter = {
281       val value = lexicoder.decode(partition.value)
282       val attr = ff.property(attribute)
283       divisor match {
284         case None => ff.equals(attr, ff.literal(value))
285         case Some(d) =>
286           val lower = ff.literal(value)
287           val upper = ff.literal(value + d.divisor)
288           ff.and(ff.greaterOrEqual(attr, lower), ff.less(attr, upper))
289       }
290     }
291 
292     protected def enumerate(bounds: Bounds[T]): Seq[PartitionKey] = {
293       val lower = bounds.lower.value.get
294       val delta = bounds.upper.value.get - lower
295       divisor.map(_.divisor) match {
296         case None => Seq.tabulate(delta.toInt)(i => PartitionKey(name, lexicoder.encode(lower + integral.fromInt(i))))
297         case Some(d) => Seq.tabulate((delta / d).toInt)(i => PartitionKey(name, lexicoder.encode(lower + (d * integral.fromInt(i)))))
298       }
299     }
300   }

345   private abstract class FractionalScheme[T: Fractional: ClassTag](
346       attribute: String,
347       index: Int,
348       scale: Option[FractionalBucketing[T]],
349       defaultValue: Option[T],
350       allowedValues: Seq[T],
351       lexicoder: TypeEncoder[T, String],
352     ) extends AttributeScheme[T](attribute, index, defaultValue, allowedValues, implicitly[Fractional[T]].zero, scale, lexicoder) {
353 
354     import FilterHelper.ff
355 
356     import Fractional.Implicits.infixFractionalOps
357 
358     private val fractional = implicitly[Fractional[T]]
359     private val oneBucket: Option[T] = scale.map(s => fractional.one / s.scaleT)
360 
361     override def getCoveringFilter(partition: PartitionKey): Filter = {
362       val value = lexicoder.decode(partition.value)
363       val attr = ff.property(attribute)
364       oneBucket match {
365         case None => ff.equals(attr, ff.literal(value))
366         case Some(one) =>
367           val lower = ff.literal(value)
368           val upper = ff.literal(value + one)
369           ff.and(ff.greaterOrEqual(attr, lower), ff.less(attr, upper))
370       }
371     }
372 
373     protected def enumerate(bounds: Bounds[T]): Seq[PartitionKey] = {
374       scale match {
375         case None =>
376           val lower = toPartition(bounds.lower.value.get)
377           val upper = exclusiveUpperBound(bounds).get
378           // since lexicoding is lossy, we can't increment the fractional numbers directly, instead we increment the hex string
379           (Iterator.single(lower) ++ Iterator.iterate(lower)(incrementHex).takeWhile(_ < upper)).map(PartitionKey(name, _)).toSeq
380         case Some(s) =>
381           // here we assume the scale will be enough to increment the encoded string
382           val lower = s(bounds.lower.value.get)
383           val increment = fractional.one / s.scaleT
384           val upper = exclusiveUpperBound(bounds).get
385           val tail = Iterator.iterate(lower)(_ + increment).map(toPartition).takeWhile(v => v < upper)
386           (Iterator.single(toPartition(lower)) ++ tail).map(PartitionKey(name, _)).toSeq
387       }
388     }
389 
390     // note that this is tailored to the specific hex encoding used by mango
391     // also note this will fail if we get to the max hex value
392     private def incrementHex(hex: String): String = {
393       (hex.last + 1).toChar match {
394         case d   => hex.dropRight(1) + d
395         case 'g' => incrementHex(hex.dropRight(1)) + "0"
396       }
397     }
398   }
