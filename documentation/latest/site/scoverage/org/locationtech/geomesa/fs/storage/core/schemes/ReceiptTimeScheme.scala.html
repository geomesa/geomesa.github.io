29 /**
30  * Scheme for partitioning based on "receipt time", i.e. when a message is received. Generally this is useful
31  * only for reading existing data that may have been aggregated and stored by an external process.
32  *
33  * @param buffer amount of time to buffer queries by, in order to match a feature date to a receipt time date -
34  *               i.e. the amount of latency in the ingest process
35  */
36 class ReceiptTimeScheme(
37     formatter: DateTimeFormatter,
38     pattern: String,
39     stepUnit: ChronoUnit,
40     step: Int,
41     attribute: String,
42     index: Int,
43     val buffer: Duration,
44   ) extends HierarchicalDateTimeScheme(formatter, pattern, stepUnit, step, attribute, index) {
45 
46   import ReceiptTimeScheme.Config
47 
48   override val name: String =
49     HierarchicalDateTimeScheme(formatter, pattern, stepUnit, step, attribute, index).name
50       .replaceFirst(HierarchicalDateTimeScheme.Name, ReceiptTimeScheme.Name) + s":${Config.BufferOpt}=${buffer.toMillis}ms"
51 
52   override def getRangesForFilter(filter: Filter): Option[Seq[PartitionRange]] =
53     super.getRangesForFilter(buffered(filter))
54 
55   override def getCoveringFilter(partition: PartitionKey): Filter =
56     throw new UnsupportedOperationException("Dates may overlap in multiple partitions")
57 
58   private def buffered(filter: Filter): Filter =
59     filter.accept(new BufferingFilterVisitor(buffer, attribute), null).asInstanceOf[Filter]
60 }

94     override def visit(filter: PropertyIsBetween, extraData: AnyRef): AnyRef = {
95       val factory = getFactory(extraData)
96       def buffer(p: PropertyName, lower: Literal, upper: Literal): Option[Filter] = {
97         for {
98           lo <- Option(FastConverter.evaluate(lower, classOf[Date]))
99           up <- Option(FastConverter.evaluate(upper, classOf[Date]))
100         } yield {
101           val bufferedLo = bufferDown(lo)
102           val bufferedUp = bufferUp(up)
103           // account for inverted filters that may result in invalid clauses after buffering
104           if (bufferedLo.before(bufferedUp)) {
105             factory.between(p, factory.literal(bufferedLo), factory.literal(bufferedUp), filter.getMatchAction)
106           } else {
107             Filter.EXCLUDE
108           }
109         }
110       }
111       val prop = visit(filter.getExpression, extraData)
112       val lowerBoundary = visit(filter.getLowerBoundary, extraData)
113       val upperBoundary = visit(filter.getUpperBoundary, extraData)
114       val buffered = (prop, lowerBoundary, upperBoundary) match {
115         case (p: PropertyName, lower: Literal, upper: Literal) if p.getPropertyName == dtg => buffer(p, lower, upper)
116         case _ => None
117       }
118 
119       buffered.getOrElse(super.visit(filter, extraData))
120     }

285     private def buffer(lit: Literal, offset: Long, extraData: AnyRef): Expression = {
