31 /**
32  * StorageMetadata implementation that stores all file information in a single JSON file.
33  * Uses HDFS lock files for cross-JVM write locking and synchronization for intra-JVM locking.
34  *
35  * @param fs file system
36  * @param directory metadata root path
37  * @param meta basic metadata config
38  */

40     extends StorageMetadata with CachedMetadata with LazyLogging {

48   private val kvs = new ConcurrentHashMap[String, String](meta.config.asJava)
49   private val encodedTypeName = StringSerialization.alphaNumericSafeString(meta.sft.getTypeName)
50   private val metadataFilePath = directory.resolve(s"$encodedTypeName.json")
51   private val filesFilePath = directory.resolve(s".${encodedTypeName}_files.json")
52   private val lockFilePath = directory.resolve(s".${encodedTypeName}_files.json.lock")

54   override val `type`: String = FileBasedMetadata.MetadataType

57   override val schemes: Set[PartitionScheme] = meta.partitions.map(PartitionSchemeFactory.load(sft, _)).toSet

61   override def get(key: String): Option[String] = Option(kvs.get(key))

63   // note: this isn't synchronized across jvms
64   override def set(key: String, value: String): Unit = FileBasedMetadata.synchronized {
65     if (value == null) {
66       kvs.remove(key)
67     } else {
68       kvs.put(key, value)
69     }
70     WithClose(fs.overwrite(metadataFilePath)) { out =>
71       MetadataSerialization.serialize(out, meta.copy(config = kvs.asScala.toMap))
72     }
73   }

75   override def addFile(file: StorageFile): Unit = {
76     modifyFiles { files =>
77       // remove any existing file with the same path and add the new one
78       (files.filterNot(_.file == file.file) :+ file).sortBy(_.timestamp)(Ordering.Long.reverse)
79     }
80     logger.debug(s"Added file $file")
81   }
82 
83   override def removeFile(file: StorageFile): Unit = {
84     modifyFiles { files =>

86     }
87     logger.debug(s"Removed file $file")

90   override def replaceFiles(existing: Seq[StorageFile], replacements: Seq[StorageFile]): Unit = {
91     val existingFiles = existing.map(_.file)
92     modifyFiles { files =>
93       files.filterNot(f => existingFiles.contains(f.file)) ++ replacements
94     }
95     logger.debug(s"Replaced ${existing.size} files with ${replacements.size} new ones")
96   }

98   override def close(): Unit = try { super.close() } finally { fs.close() }

100   /**
101    * Load files from the JSON file
102    */
103   override protected def buildFileList(): Seq[StorageFile] = {
104     WithClose(fs.read(filesFilePath)) { opt =>
105       opt.fold(Seq.empty[StorageFile]) { is =>
106         val listType = new TypeToken[java.util.List[StorageFile]]() {}.getType
107         gson.fromJson[java.util.List[StorageFile]](new InputStreamReader(is, StandardCharsets.UTF_8), listType).asScala.toSeq
108       }
109     }
110   }

112   /**
113    * Modify files with proper locking using HDFS lock files
114    */
115   private def modifyFiles(fn: Seq[StorageFile] => Seq[StorageFile]): Unit = FileBasedMetadata.synchronized {
116     // acquire lock by creating a lock file atomically
117     var lockAcquired = false
118     var retries = 0
119 
120     val maxAttempts = MaxLockRetries.toInt.get
121 
122     while (!lockAcquired && retries < maxAttempts) {
123       logger.trace(s"Attempting to acquire lock at $lockFilePath with ${maxAttempts - retries} tries left")
124       try {
125         // try to create lock file with overwrite=false for atomicity
126         try {
127           fs.create(lockFilePath).foreach { out =>
128             try {
129               // write lock info for debugging - hostname + timestamp
130               val lockInfo = s"${java.net.InetAddress.getLocalHost.getHostName}:${System.currentTimeMillis()}"
131               out.write(lockInfo.getBytes(StandardCharsets.UTF_8))
132             } finally {
133               out.close()
134             }
135             lockAcquired = true
136           }
137         } catch {
138           case NonFatal(e) => logger.debug("Error writing lock file, may already exist?", e)
139         }
140 
141         if (!lockAcquired) {
142           retries += 1
143           if (retries < maxAttempts) {
144             // check if lockfile is stale
145             fs.modified(lockFilePath).foreach { modified =>
146               val age = System.currentTimeMillis() - modified
147               if (age > LockTimeout.toMillis.get) {
148                 logger.debug(s"Deleting expired lock file (age ${age}ms at $lockFilePath")
149                 fs.delete(lockFilePath)
150               }
151             }
152             // wait and retry
153             logger.debug(s"Could not acquire lock - waiting for ${LockRetryDelay.toMillis.get}ms before next attempt")
154             Thread.sleep(LockRetryDelay.toMillis.get)
155           }
156         }
157       } catch {
158         case NonFatal(e) => throw new RuntimeException(s"Failed to acquire lock at $lockFilePath", e)
159       }
160     }
161 
162     if (!lockAcquired) {
163       throw new RuntimeException(s"Failed to acquire lock after $MaxLockRetries retries")
164     }
165     logger.debug(s"Acquired lock file at $lockFilePath")
166 
167     try {
168       // reload from disk to get latest state
169       val currentFiles = buildFileList()
170 
171       // apply the modification
172       val updatedFiles = fn(currentFiles)
173 
174       // write back to disk
175       WithClose(new OutputStreamWriter(fs.overwrite(filesFilePath), StandardCharsets.UTF_8)) { writer =>
176         gson.toJson(updatedFiles.asJava, writer)
177       }
178 
179       // update cache
180       filesCache.put(BoxedUnit.UNIT, updatedFiles)
181     } finally {
182       // release lock by deleting lock file
183       try {
184         fs.delete(lockFilePath)
185         logger.debug(s"Released lock file at $lockFilePath")
186       } catch {
187         case NonFatal(e) => logger.warn(s"Failed to release lock at $lockFilePath", e)
188       }
189     }
190   }
191 }
