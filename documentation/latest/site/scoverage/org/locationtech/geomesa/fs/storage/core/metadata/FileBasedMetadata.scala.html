31 /**
32  * StorageMetadata implementation that stores all file information in a single JSON file.
33  * Uses HDFS lock files for cross-JVM write locking and synchronization for intra-JVM locking.
34  *
35  * @param fs file system
36  * @param directory metadata root path
37  * @param meta basic metadata config
38  */
39 class FileBasedMetadata(fs: ObjectStore, meta: Metadata, directory: URI)
40     extends StorageMetadata with CachedMetadata with LazyLogging {
41 
42   import FileBasedMetadata._
43 
44   import scala.collection.JavaConverters._
45 
46   require(directory.toString.endsWith("/"), "Invalid path - must end with '/'")
47 
48   private val kvs = new ConcurrentHashMap[String, String](meta.config.asJava)
49   private val encodedTypeName = StringSerialization.alphaNumericSafeString(meta.sft.getTypeName)
50   private val metadataFilePath = directory.resolve(s"$encodedTypeName.json")
51   private val filesFilePath = directory.resolve(s".${encodedTypeName}_files.json")
52   private val lockFilePath = directory.resolve(s".${encodedTypeName}_files.json.lock")
53 
54   override val `type`: String = FileBasedMetadata.MetadataType
55 
56   override val sft: SimpleFeatureType = meta.sft
57   override val schemes: Set[PartitionScheme] = meta.partitions.map(PartitionSchemeFactory.load(sft, _)).toSet
58 
59   filesCache.refresh(BoxedUnit.UNIT) // kick off the initial load asynchronously
60 
61   override def get(key: String): Option[String] = Option(kvs.get(key))
62 
63   // note: this isn't synchronized across jvms
64   override def set(key: String, value: String): Unit = FileBasedMetadata.synchronized {
65     if (value == null) {
66       kvs.remove(key)
67     } else {
68       kvs.put(key, value)
69     }
70     WithClose(fs.overwrite(metadataFilePath)) { out =>
71       MetadataSerialization.serialize(out, meta.copy(config = kvs.asScala.toMap))
72     }
73   }
74 
75   override def addFile(file: StorageFile): Unit = {
76     modifyFiles { files =>
77       // remove any existing file with the same path and add the new one
78       (files.filterNot(_.file == file.file) :+ file).sortBy(_.timestamp)(Ordering.Long.reverse)
79     }
80   }
81 
82   override def removeFile(file: StorageFile): Unit = {
83     modifyFiles { files =>
84       files.filterNot(_.file == file.file)
85     }
86   }
87 
88   override def replaceFiles(existing: Seq[StorageFile], replacements: Seq[StorageFile]): Unit = {
89     val existingFiles = existing.map(_.file)
90     modifyFiles { files =>
91       files.filterNot(f => existingFiles.contains(f.file)) ++ replacements
92     }
93   }
94 
95   override def close(): Unit = try { super.close() } finally { fs.close() }
96 
97   /**
98    * Load files from the JSON file
99    */
100   override protected def buildFileList(): Seq[StorageFile] = {
101     try {
102       WithClose(fs.read(filesFilePath)) { opt =>
103         opt.fold(Seq.empty[StorageFile]) { is =>
104           val listType = new TypeToken[java.util.List[StorageFile]]() {}.getType
105           gson.fromJson[java.util.List[StorageFile]](new InputStreamReader(is, StandardCharsets.UTF_8), listType).asScala.toSeq
106         }
107       }
108     } catch {
109       case NonFatal(e) =>
110         logger.warn(s"Error loading files from $filesFilePath", e)
111         Seq.empty
112     }
113   }
114 
115   /**
116    * Modify files with proper locking using HDFS lock files
117    */
118   private def modifyFiles(fn: Seq[StorageFile] => Seq[StorageFile]): Unit = FileBasedMetadata.synchronized {
119     // acquire lock by creating a lock file atomically
120     var lockAcquired = false
121     var retries = 0
122 
123     val maxAttempts = MaxLockRetries.toInt.get
124 
125     while (!lockAcquired && retries < maxAttempts) {
126       try {
127         // try to create lock file with overwrite=false for atomicity
128         WithClose(fs.create(lockFilePath)) {
129           case None =>
130             // lock file exists, check if it's stale
131             retries += 1
132             if (isLockStale(lockFilePath)) {
133               // remove stale lock and retry
134               try {
135                 fs.delete(lockFilePath)
136               } catch {
137                 case NonFatal(_) => // ignore, will retry
138               }
139             } else {
140               // wait and retry
141               Thread.sleep(LockRetryDelay.toMillis.get)
142             }
143           case Some(out) =>
144             // write lock info for debugging - hostname + timestamp
145             val lockInfo = s"${java.net.InetAddress.getLocalHost.getHostName}:${System.currentTimeMillis()}"
146             out.write(lockInfo.getBytes(StandardCharsets.UTF_8))
147         }
148         lockAcquired = true
149       } catch {
150         case NonFatal(e) => throw new RuntimeException(s"Failed to acquire lock at $lockFilePath", e)
151       }
152     }
153 
154     if (!lockAcquired) {
155       throw new RuntimeException(s"Failed to acquire lock after $MaxLockRetries retries")
156     }
157 
158     try {
159       // reload from disk to get latest state
160       val currentFiles = buildFileList()
161 
162       // apply the modification
163       val updatedFiles = fn(currentFiles)
164 
165       // write back to disk
166       WithClose(new OutputStreamWriter(fs.overwrite(filesFilePath), StandardCharsets.UTF_8)) { writer =>
167         gson.toJson(updatedFiles.asJava, writer)
168       }
169 
170       // update cache
171       filesCache.put(BoxedUnit.UNIT, updatedFiles)
172     } finally {
173       // release lock by deleting lock file
174       try {
175         fs.delete(lockFilePath)
176       } catch {
177         case NonFatal(e) => logger.warn(s"Failed to release lock at $lockFilePath", e)
178       }
179     }
180   }
181 
182   /**
183    * Check if a lock file is stale (older than lock timeout)
184    */
185   private def isLockStale(lockPath: URI): Boolean = {
186     try {
187       fs.modified(lockPath).forall { modified =>
188         val age = System.currentTimeMillis() - modified
189         age > LockTimeout.toMillis.get
190       }
191     } catch {
192       case NonFatal(_) => true // if we can't read it, consider it stale
193     }
194   }
195 }

197 object FileBasedMetadata {

201   // locking parameters
202   val MaxLockRetries = SystemProperty("geomesa.fs.metadata.file.lock.retries", "60") // maximum number of lock acquisition retries
203   val LockRetryDelay = SystemProperty("geomesa.fs.metadata.file.lock.delay", "1 second") // delay between retries

206   // gson instance with custom serializers for StorageFile
207   private val gson: Gson =
208     new GsonBuilder()
209       .registerTypeAdapter(classOf[PartitionKey], PartitionKey.PartitionKeySerializer)
210       .registerTypeAdapter(classOf[Partition], Partition.PartitionSerializer)
211       .registerTypeAdapter(classOf[StorageFile], StorageFileSerializer)
212       .disableHtmlEscaping()
213       .create()

215   /**
216    * Json serializer for StorageFileAction
217    */
218   private object StorageFileSerializer extends JsonSerializer[StorageFile] with JsonDeserializer[StorageFile] {
219 
220     import scala.collection.JavaConverters._
221 
222     override def serialize(src: StorageFile, typeOfSrc: Type, context: JsonSerializationContext): JsonElement = {
223       val obj = new JsonObject()
224       obj.addProperty("file", src.file)
225       obj.add("partition", context.serialize(src.partition))
226       obj.addProperty("count", src.count)
227       obj.addProperty("action", src.action.toString)
228       val spatialBounds = new JsonArray(src.spatialBounds.size)
229       src.spatialBounds.foreach(b => spatialBounds.add(context.serialize(b)))
230       obj.add("spatialBounds", spatialBounds)
231       val attributeBounds = new JsonArray(src.attributeBounds.size)
232       src.attributeBounds.foreach(b => attributeBounds.add(context.serialize(b)))
233       obj.add("attributeBounds", attributeBounds)
234       val sort = new JsonArray(src.sort.size)
235       src.sort.foreach(sort.add(_))
236       obj.add("sort", sort)
237       obj.addProperty("timestamp", src.timestamp)
238       obj
239     }
240 
241     override def deserialize(json: JsonElement, typeOfT: Type, context: JsonDeserializationContext): StorageFile = {
242       val obj = json.getAsJsonObject
243       val spatialBounds =
244         obj.getAsJsonArray("spatialBounds").asList().asScala.map(context.deserialize[SpatialBounds](_, classOf[SpatialBounds])).toSeq
245       val attributeBounds =
246         obj.getAsJsonArray("attributeBounds").asList().asScala.map(context.deserialize[AttributeBounds](_, classOf[AttributeBounds])).toSeq
247       val sort = obj.getAsJsonArray("sort").asList().asScala.map(_.getAsInt).toSeq
248       StorageFile(
249         obj.getAsJsonPrimitive("file").getAsString,
250         context.deserialize(obj.get("partition"), classOf[Partition]),
251         obj.getAsJsonPrimitive("count").getAsLong,
252         StorageFileAction.withName(obj.getAsJsonPrimitive("action").getAsString),
253         spatialBounds,
254         attributeBounds,
255         sort,
256         obj.getAsJsonPrimitive("timestamp").getAsLong,
257       )
258     }
259   }
