45   /**
46    * Gets the schema needed for reading a file
47    *
48    * @param transform query transform definition
49    * @param filtered columns that have filters against them
50    * @param includeRowPositions include _file and _pos columns necessary for handling updates/deletes
51    * @return
52    */
53   def read(transform: Option[String], filtered: Set[String], includeRowPositions: Boolean = false): SimpleFeatureIcebergSchema = {
54     val (readCols, readSft) = transform match {
55       case None =>
56         val baseCols =
57           Seq(FeatureIdField, VisibilitiesField) ++ sft.getAttributeDescriptors.asScala.map(d => ColumnName.encode(d.getLocalName))
58         val readCols = baseCols ++ filtered.filterNot(baseCols.contains)
59         (readCols, sft)
60       case Some(defs) =>
61         val fromTransform = Transforms(sft, defs).flatMap {
62           case t: PropertyTransform => Seq(sft.getDescriptor(t.i).getLocalName)
63           case t: RenameTransform => Seq(sft.getDescriptor(t.i).getLocalName)
64           case t: ExpressionTransform => FilterHelper.propertyNames(t.expression, sft)
65           case t => throw new UnsupportedOperationException(s"An implementation is missing: ${t.getClass}")
66         }
67         val attributes = fromTransform.distinct
68         val baseCols = Seq(FeatureIdField, VisibilitiesField) ++ attributes.map(ColumnName.encode)
69         val readCols = baseCols ++ filtered.filterNot(baseCols.contains)
70         val readSft = {
71           val sftBuilder = new SimpleFeatureTypeBuilder()
72           attributes.foreach(name => sftBuilder.add(sft.getDescriptor(name)))
73           sftBuilder.setName(sft.getName)
74           sftBuilder.buildFeatureType()
75         }
76         readSft.getUserData.putAll(sft.getUserData)
77         (readCols, readSft)
78     }
79     val schema = fields.toSchema(readCols, includeRowPositions)
80     new SimpleFeatureIcebergSchema(readSft, geometries, metadata, schema, fields)
81   }
82 }

90   /**
91    * Get a schema based on a simple feature type. Encoding can be configured through `geomesa.parquet.geometries`.
92    *
93    * @param conf write configuration, including the sft spec
94    * @return
95    */
96   def apply(sft: SimpleFeatureType, conf: Map[String, String]): SimpleFeatureIcebergSchema = {
97     val geometries = conf.get(GeometryEncodingKey).map(GeometryEncoding.apply).getOrElse(GeometryEncoding.GeoParquetWkb)
98     val metadata = Map(
99       SftNameKey -> sft.getTypeName,
100       SftSpecKey -> SimpleFeatureTypes.encodeType(sft, includeUserData = true),
101       GeometryEncodingKey -> geometries.toString,
102     ) ++ conf.get(PartitionKey).map(PartitionKey -> _)
103     val iceberg = schema(sft, geometries)
104     new SimpleFeatureIcebergSchema(sft, geometries, metadata.asJava, iceberg.toSchema, iceberg)
105   }

107   private def schema(sft: SimpleFeatureType, geometries: GeometryEncoding): IcebergFields = {
108     // note: for iceberg compatibility, field ids need to start at one and increment (without gaps) across all top-level fields.
109     // All nested fields (structs, lists) get ids *after* all the top-level fields, once again incrementing without gaps
110     val fieldIds = new AtomicInteger(1)
111     val nestedFieldIds = {
112       val extraGeomCount = 2 * sft.getAttributeDescriptors.asScala.count {
113         case d if classOf[Geometry].isAssignableFrom(d.getType.getBinding) => true
114         case _ => false
115       }
116       new AtomicInteger(3 + sft.getAttributeCount + extraGeomCount)
117     }
118 
119     val builder = Seq.newBuilder[NestedField]
120     builder += NestedField.required(fieldIds.getAndIncrement(), FeatureIdField, StringType.get())
121     builder += NestedField.optional(fieldIds.getAndIncrement(), VisibilitiesField, StringType.get())
122 
123     sft.getAttributeDescriptors.asScala.foreach { d =>
124       val name = ColumnName(d.getLocalName)
125       val objectType = ObjectType.selectType(d)
126       builder += buildField(name.column, objectType, geometries, fieldIds, nestedFieldIds)
127       if (objectType.head == ObjectType.GEOMETRY) {
128         builder += BoundingBoxField.icebergSchema(name.column, fieldIds, nestedFieldIds)
129         builder += ZValueField.icebergSchema(name.column, objectType(1), fieldIds)
130       }
131     }
132 
133     val fields = builder.result()
134     val aliases = fields.map(f => f.name() -> Int.box(f.fieldId())).toMap
135 
136     IcebergFields(fields, aliases)
137   }
138 
139   /**
140    * Builds the schema type for an attribute
141    *
142    * @param name field name
143    * @param bindings object type
144    * @param geometries geometry type encodings
145    * @return
146    */
147   private def buildField(
148       name: String,
149       bindings: Seq[ObjectType],
150       geometries: GeometryEncoding,
151       fieldIds: AtomicInteger,
152       nestedFieldIds: AtomicInteger): NestedField = {
153     val builder = NestedField.optional(name).withId(fieldIds.getAndIncrement())
154     val typed: NestedField.Builder = bindings.head match {
155       case ObjectType.INT     => builder.ofType(IntegerType.get())
156       case ObjectType.DOUBLE  => builder.ofType(DoubleType.get())
157       case ObjectType.LONG    => builder.ofType(LongType.get())
158       case ObjectType.FLOAT   => builder.ofType(FloatType.get())
159       case ObjectType.BOOLEAN => builder.ofType(BooleanType.get())
160       case ObjectType.BYTES   => builder.ofType(BinaryType.get())
161       case ObjectType.STRING  => builder.ofType(StringType.get())
162       case ObjectType.DATE    => builder.ofType(TimestampType.withZone())
163       case ObjectType.UUID    => builder.ofType(UUIDType.get())
164 
165       case ObjectType.LIST =>
166         val subType = buildField("", bindings.drop(1), geometries, nestedFieldIds, null /* should not be used*/)
167         builder.ofType(ListType.ofRequired(subType.fieldId(), subType.`type`()))
168 
169       case ObjectType.MAP =>
170         val keyType = buildField("", bindings.slice(1, 2), geometries, nestedFieldIds, null /* should not be used*/)
171         val valueType = buildField("", bindings.slice(2, 3), geometries, nestedFieldIds, null /* should not be used*/)
172         builder.ofType(MapType.ofRequired(keyType.fieldId(), valueType.fieldId(), keyType.`type`(), valueType.`type`()))
173 
174       case ObjectType.GEOMETRY =>
175         // not yet supported in spark or trino
176         // builder.ofType(GeometryType.crs84())
177         builder.ofType(BinaryType.get())
178 
179       case binding =>
180         throw new UnsupportedOperationException(s"No mapping defined for type: $binding")
181     }
182     typed.build()
183   }
184 
185   private case class IcebergFields(fields: Seq[NestedField], aliases: Map[String, Integer]) {
186     def toSchema: Schema = new Schema(fields.asJava, aliases.asJava, java.util.Set.of[Integer](1))
187     def toSchema(cols: Seq[String], includeRowPositions: Boolean): Schema = {
188       val projection =
189         cols.map(col => fields.find(_.name() == col).getOrElse(throw new IllegalStateException(s"Unexpected projection: $col")))
190       val withRows =
191         if (includeRowPositions) { projection ++ Seq(MetadataColumns.FILE_PATH, MetadataColumns.ROW_POSITION) } else { projection }
192       val ids = projection.collectFirst { case f if f.name() == FeatureIdField => Int.box(f.fieldId()) }
193       new Schema(withRows.asJava, aliases.asJava, ids.toSet.asJava)
194     }
195   }
