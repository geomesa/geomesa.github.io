22 class FsGetFilesCommand extends FsDataStoreCommand {
23 
24   import org.locationtech.geomesa.utils.geotools.GeoToolsDateFormat
25 
26   import scala.collection.JavaConverters._
27 
28   override val params = new FSGetFilesParams
29 
30   override val name: String = "get-files"
31 
32   override def execute(): Unit = withDataStore { ds =>
33     val storage = ds.storage(params.featureName)
34     val metadata = storage.metadata
35 
36     lazy val fromFilter = {
37       Command.user.info(s"Listing files for filter: ${ECQL.toCQL(params.cqlFilter)}")
38       metadata.getFiles(params.cqlFilter)
39     }
40     lazy val fromPartitions = {
41       Command.user.info(s"Listing files for partition(s): ${params.loadedPartitions.mkString(", ")}")
42       params.loadedPartitions.flatMap(metadata.getFiles)
43     }
44 
45     val files =
46       if (params.cqlFilter == null && params.loadedPartitions.isEmpty) {
47         Command.user.info("Listing files for all partitions")
48         metadata.getFiles()
49       } else if (params.loadedPartitions.isEmpty) {
50         fromFilter
51       } else if (params.cqlFilter == null) {
52         fromPartitions
53       } else {
54         (fromFilter ++ fromPartitions).distinct
55       }
56 
57     files.groupBy(_.partition).toSeq.sortBy(_._1.toString).foreach { case (p, files) =>
58       Command.output.info(s"$p:")
59       // sort by chronological order
60       files.sorted(StorageMetadata.StorageFileOrdering.reverse).foreach { f =>
61         Command.output.info(s"  ${f.file} ${f.action.toString.toUpperCase(Locale.US)} " +
62           s"${GeoToolsDateFormat.format(Instant.ofEpochMilli(f.timestamp))} ${f.count} features")
63       }
64     }
65   }
66 }
