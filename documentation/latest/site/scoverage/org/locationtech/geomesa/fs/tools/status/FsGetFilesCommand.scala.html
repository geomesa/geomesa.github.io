22 class FsGetFilesCommand extends FsDataStoreCommand {
23 
24   import org.locationtech.geomesa.utils.geotools.GeoToolsDateFormat
25 
26   override val params = new FSGetFilesParams
27 
28   override val name: String = "get-files"
29 
30   override def execute(): Unit = withDataStore { ds =>
31     val storage = ds.storage(params.featureName)
32     val metadata = storage.metadata
33 
34     lazy val fromFilter = {
35       Command.user.info(s"Listing files for filter: ${ECQL.toCQL(params.cqlFilter)}")
36       metadata.getFiles(params.cqlFilter)
37     }
38     lazy val fromPartitions = {
39       Command.user.info(s"Listing files for partition(s): ${params.loadedPartitions.mkString(", ")}")
40       params.loadedPartitions.flatMap(metadata.getFiles)
41     }
42 
43     val files =
44       if (params.cqlFilter == null && params.loadedPartitions.isEmpty) {
45         Command.user.info("Listing files for all partitions")
46         metadata.getFiles()
47       } else if (params.loadedPartitions.isEmpty) {
48         fromFilter
49       } else if (params.cqlFilter == null) {
50         fromPartitions
51       } else {
52         (fromFilter ++ fromPartitions).distinct
53       }
54 
55     files.groupBy(_.partition).toSeq.sortBy(_._1.toString).foreach { case (p, files) =>
56       Command.output.info(s"$p:")
57       // sort by chronological order
58       files.sorted(StorageMetadata.StorageFileOrdering.reverse).foreach { f =>
59         Command.output.info(s"  ${f.file} ${f.action.toString.toUpperCase(Locale.US)} " +
60           s"${GeoToolsDateFormat.format(Instant.ofEpochMilli(f.timestamp))} ${f.count} features")
61       }
62     }
63   }
64 }
