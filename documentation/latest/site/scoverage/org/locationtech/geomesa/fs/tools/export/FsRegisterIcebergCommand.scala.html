31 class FsRegisterIcebergCommand extends FsDataStoreCommand with LazyLogging {
32 
33   import scala.collection.JavaConverters._
34 
35   override val params = new FsRegisterIcebergParams()
36 
37   override val name: String = "register-iceberg-files"
38 
39   override def execute(): Unit = {
40     if (params.loadedPartitions.isEmpty) {
41       throw new ParameterException("At least one of --partition or --partition-file must be specified")
42     }
43     if (params.icebergConfigFile == null && params.icebergConfig.isEmpty) {
44       throw new ParameterException("At least one of --iceberg-config or --iceberg-config-file must be specified")
45     }
46 
47     val icebergProps = new Properties()
48     if (params.icebergConfigFile != null) {
49       WithClose(new FileReader(params.icebergConfigFile, StandardCharsets.UTF_8))(icebergProps.load)
50     }
51     params.icebergConfig.asScala.foreach { case (k, v) => icebergProps.put(k, v) }
52     if (!icebergProps.containsKey("catalog-impl")) {
53       throw new ParameterException("Iceberg properties must specify a 'catalog-impl'")
54     }
55 
56     val catalog = try {
57       Class.forName(icebergProps.getProperty("catalog-impl")).getConstructor().newInstance().asInstanceOf[Catalog]
58     } catch {
59       case e: Throwable => throw new ParameterException(s"Could not instantiate catalog class '$name':", e)
60     }
61 
62     catalog.initialize("geomesa", Collections.unmodifiableMap(icebergProps.asInstanceOf[java.util.Map[String, String]]))
63 
64     withDataStore(execute(_, catalog, icebergProps.asInstanceOf[java.util.Map[String, String]]))
65   }
66 
67   private def execute(ds: FileSystemDataStore, catalog: Catalog, icebergProps: java.util.Map[String, String]): Unit = {
68     val storage = try { ds.storage(params.featureName) } catch {
69       case NonFatal(e) => throw new IllegalArgumentException(s"No schema exists with type name ${params.featureName}", e)
70     }
71     if (storage.encoding != ParquetFileSystemStorage.Encoding) {
72       throw new UnsupportedOperationException(s"Iceberg is only implemented for Parquet storage: found ${storage.encoding}")
73     }
74 
75     params.loadedPartitions.foreach { p =>
76       storage.metadata.schemes.foreach { s =>
77         if (!p.values.exists(_.name == s.name)) {
78           throw new IllegalArgumentException(
79             s"Specified partition '$p' does not include all required scheme(s): " +
80               storage.metadata.schemes.map(_.name).toSeq.sorted.mkString(", "))
81         }
82       }
83     }
84 
85     val iceberg = new IcebergMapper(storage)
86     logger.debug(s"Iceberg schema: ${SchemaParser.toJson(iceberg.schema)}")
87 
88     val namespace = Namespace.of(params.icebergNamespace)
89     // TODO valid identifiers vary based on the catalog... this is for glue and not comprehensive
90     val tableId = TableIdentifier.of(namespace, storage.metadata.sft.getTypeName.toLowerCase(Locale.US).replaceAll("[^a-z0-9]+", "_"))
91     val table = if (catalog.tableExists(tableId)) {
92       Command.user.info("Found existing table")
93       catalog.loadTable(tableId)
94       // TODO compare with existing schema and partitioning
95     } else {
96       Command.user.info("Creating new iceberg table")
97       val warehouse = icebergProps.get("warehouse")
98       if (warehouse == null) {
99         throw new IllegalArgumentException("Iceberg properties must specify a 'warehouse' location")
100       }
101       // file format v3 lets us use native geometries - but it's not yet supported in spark or trino
102       // icebergProps.put(TableProperties.FORMAT_VERSION, "3")
103       catalog.createTable(tableId, iceberg.schema, iceberg.spec, warehouse, icebergProps)
104     }
105     logger.debug(s"Iceberg table: $table")
106 
107     val existingFiles = if (params.allowDuplicates) { Set.empty[String] } else {
108       val builder = Set.newBuilder[String]
109       // TODO we could probably limit this to just files that match our partition with a filter
110       WithClose(table.newScan().planFiles()) { tasks =>
111         tasks.asScala.foreach(task => builder += task.file().location())
112       }
113       builder.result()
114     }
115 
116     val gmFiles = params.loadedPartitions.flatMap(p => storage.metadata.getFiles(p))
117     Command.user.info(s"Registering ${gmFiles.size} files from ${params.loadedPartitions.size} partitions")
118 
119     var i = 0
120     val files = gmFiles.flatMap { f =>
121       i += 1
122       val df = iceberg.toDataFile(table, f)
123       logger.debug(s"File: $df")
124       if (existingFiles.contains(df.location())) {
125         Command.user.info(s"Skipping already registered file: ${df.location()}")
126         None
127       } else {
128         Command.user.info(f"Registering file (${math.floor(100 * (i.toFloat / gmFiles.size)).toInt}%02d%% complete): ${df.location()}")
129         Some(df)
130       }
131     }
132 
133     if (files.nonEmpty) {
134       Command.user.info("Updating table")
135       val append = table.newAppend()
136       files.foreach(append.appendFile)
137       append.commit()
138     }
139     Command.user.info("Complete")
140   }
141 }
