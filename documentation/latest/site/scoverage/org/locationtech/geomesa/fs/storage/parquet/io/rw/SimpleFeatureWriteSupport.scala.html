84   private class SimpleFeatureWriter(schema: SimpleFeatureParquetSchema) {
85 
86     private val fids = new FidWriter(0) // ID is the 1st field
87     private val vis = if (schema.hasVisibilities) { new VisibilityWriter(1) } else { null } // vis is 2nd field
88 
89     private val attributes = {
90       var index = if (vis == null) { 1 } else { 2 }
91       Array.tabulate(schema.sft.getAttributeCount) { i =>
92         val writer = attribute(schema.sft.getDescriptor(i), index).asInstanceOf[AttributeWriter[AnyRef]]
93         index += writer.numFields
94         writer
95       }
96     }
97 
98     def write(consumer: RecordConsumer, value: SimpleFeature): Unit = {
99       consumer.startMessage()
100       fids.apply(consumer, value.getID)
101       if (vis != null) {
102         vis.apply(consumer, value.getUserData.get("geomesa.feature.visibility").asInstanceOf[String])
103       }
104       var i = 0
105       while (i < attributes.length) {
106         attributes(i).apply(consumer, value.getAttribute(i))
107         i += 1
108       }
109       consumer.endMessage()
110     }
111 
112     private def attribute(descriptor: AttributeDescriptor, index: Int): AttributeWriter[_] =
113       attribute(descriptor.getLocalName, index, ObjectType.selectType(descriptor))
114 
115     private def attribute(name: String, index: Int, bindings: Seq[ObjectType]): AttributeWriter[_] = {
116       lazy val safeName = alphaNumericSafeString(name)
117       bindings.head match {
118         case ObjectType.GEOMETRY => geometry(name, index, bindings.last)
119         case ObjectType.DATE     => new DateMicrosWriter(safeName, index)
120         case ObjectType.STRING   => new StringWriter(safeName, index)
121         case ObjectType.INT      => new IntegerWriter(safeName, index)
122         case ObjectType.LONG     => new LongWriter(safeName, index)
123         case ObjectType.FLOAT    => new FloatWriter(safeName, index)
124         case ObjectType.DOUBLE   => new DoubleWriter(safeName, index)
125         case ObjectType.BYTES    => new BytesWriter(safeName, index)
126         case ObjectType.LIST     => new ListWriter(safeName, index, attribute("element", 0, bindings.drop(1)))
127         case ObjectType.MAP      => new MapWriter(safeName, index, attribute("key", 0, bindings.slice(1, 2)), attribute("value", 1, bindings.slice(2, 3)))
128         case ObjectType.BOOLEAN  => new BooleanWriter(safeName, index)
129         case ObjectType.UUID     => new UuidWriter(safeName, index)
130         case _ => throw new IllegalArgumentException(s"Can't serialize field '$name' of type ${bindings.head}")
131       }
132     }
133 
134     // TODO support z/m
135     private def geometry(name: String, index: Int, binding: ObjectType): AttributeWriter[_] = {
136       val bbox = schema.bboxes.get(name)
137       val zValue = schema.zValues.get(name)
138       val safeName = alphaNumericSafeString(name)
139       if (schema.encodings.geometry == GeometryEncoding.GeoParquetWkb) {
140         if (binding == ObjectType.POINT) {
141           new WkbPointWriter(safeName, index, bbox, zValue)
142         } else {
143           new WkbWriter(safeName, index, bbox, zValue)
144         }
145       } else {
146         binding match {
147           case ObjectType.POINT               => new PointWriter(safeName, index, bbox, zValue)
148           case ObjectType.LINESTRING          => new NativeLineStringWriter(safeName, index, bbox, zValue)
149           case ObjectType.POLYGON             => new NativePolygonWriter(safeName, index, bbox, zValue)
150           case ObjectType.MULTIPOINT          => new NativeMultiPointWriter(safeName, index, bbox, zValue)
151           case ObjectType.MULTILINESTRING     => new NativeMultiLineStringWriter(safeName, index, bbox, zValue)
152           case ObjectType.MULTIPOLYGON        => new NativeMultiPolygonWriter(safeName, index, bbox, zValue)
153           case ObjectType.GEOMETRY_COLLECTION => new WkbWriter(safeName, index, bbox, zValue)
154           case ObjectType.GEOMETRY            => new WkbWriter(safeName, index, bbox, zValue)
155           case _ => throw new IllegalArgumentException(s"Can't serialize field '$name' of type $binding")
156         }
157       }
158     }
159   }

266           consumer.startGroup()

293   private abstract class GeometryWriter[T <: Geometry](name: String, index: Int, bbox: Option[String], zValue: Option[String])
294       extends AttributeWriter[T](name, index, 1 + bbox.size + zValue.size) {
295 
296     protected def z(geom: T): String
297 
298     /**
299      * Writes a value to the current record
300      *
301      * @param consumer the Parquet record consumer
302      * @param value value to write
303      */
304     override def apply(consumer: RecordConsumer, value: T): Unit = {
305       if (value != null) {
306         consumer.startField(name, index)
307         writeFields(consumer, value)
308         consumer.endField(name, index)
309         bbox.foreach { name =>
310           val bbox = value.getEnvelopeInternal
311           consumer.startField(name, index + 1)
312           consumer.startGroup()
313           consumer.startField(BoundingBoxField.XMin, 0)
314           consumer.addFloat(bbox.getMinX.toFloat)
315           consumer.endField(BoundingBoxField.XMin, 0)
316           consumer.startField(BoundingBoxField.YMin, 1)
317           consumer.addFloat(bbox.getMinY.toFloat)
318           consumer.endField(BoundingBoxField.YMin, 1)
319           consumer.startField(BoundingBoxField.XMax, 2)
320           consumer.addFloat(bbox.getMaxX.toFloat)
321           consumer.endField(BoundingBoxField.XMax, 2)
322           consumer.startField(BoundingBoxField.YMax, 3)
323           consumer.addFloat(bbox.getMaxY.toFloat)
324           consumer.endField(BoundingBoxField.YMax, 3)
325           consumer.endGroup()
326           consumer.endField(name, index + 1)
327         }
328         zValue.foreach { name =>
329           consumer.startField(name, index + 1 + bbox.size)
330           consumer.addBinary(Binary.fromString(z(value)))
331           consumer.endField(name, index + 1 + bbox.size)
332         }
333       }
334     }
