18 class HashScheme[T](
19     attribute: String,
20     index: Int,
21     buckets: Int,
22     hasher: MurmurHashFunction.Hashing[T],
23   ) extends PartitionScheme {
24 
25   import FilterHelper.ff
26 
27   override val name: String = s"${HashScheme.Name}:attribute=$attribute:buckets=$buckets"
28 
29   private val format = s"%0${(buckets - 1).toString.length}d"
30   private val default = PartitionKey(name, format.format(0))
31 
32   override def getPartition(feature: SimpleFeature): PartitionKey = {
33     val value = feature.getAttribute(index)
34     if (value == null) {
35       default
36     } else {
37       PartitionKey(name, toPartition(value.asInstanceOf[T]))
38     }
39   }
40 
41   override def getRangesForFilter(filter: Filter): Option[Seq[PartitionRange]] = {
42     getBounds(filter).map { bounds =>
43       val builder = new RangeBuilder()
44       bounds.foreach { bound =>
45         // note: these are all equality values, as determined in getBounds
46         val lower = toPartition(bound.lower.value.get)
47         val upper = lower + ZeroChar
48         builder += PartitionRange(name, lower, upper)
49       }
50       builder.result()
51     }
52   }
53 
54   override def getPartitionsForFilter(filter: Filter): Option[Seq[PartitionKey]] = {
55     getBounds(filter).map { bounds =>
56       bounds.map { bound =>
57         // note: these are all equality values, as determined in getBounds
58         PartitionKey(name, toPartition(bound.lower.value.get))
59       }
60     }
61   }
62 
63   private def getBounds(filter: Filter): Option[Seq[Bounds[T]]] = {
64     val bounds = FilterHelper.extractAttributeBounds(filter, attribute, hasher.binding)
65     if (bounds.isEmpty || bounds.exists(_.isRange)) {
66       None
67     } else if (bounds.disjoint) {
68       Some(Seq.empty)
69     } else {
70       Some(bounds.values)
71     }
72   }
73 
74   private def toPartition(value: T): String = format.format((hasher(value) & Int.MaxValue) % buckets)
75 
76   override def getCoveringFilter(partition: PartitionKey): Filter = {
77     val fn = BucketHashFunction.Name.getFunctionName
78     ff.equals(ff.function(fn, ff.property(attribute), ff.literal(buckets)), ff.literal(partition.value.toInt))
79   }
80 }

82 object HashScheme extends PartitionSchemeFactory {
83 
84   val Name = "hash"
85 
86   override def load(sft: SimpleFeatureType, scheme: String): Option[PartitionScheme] = {
87     val opts = SchemeOpts(scheme)
88     if (opts.name != Name) { None } else {
89       val attribute = opts.getSingle("attribute").orNull
90       require(attribute != null, s"Hash scheme requires an attribute to be specified with 'attribute=<attribute>'")
91       val index = attributeIndex(sft, attribute)
92       val binding = sft.getDescriptor(index).getType.getBinding
93       val hasher = MurmurHashFunction.Hashers.find(_.binding.isAssignableFrom(binding)).orNull
94       require(hasher != null, s"Invalid type binding '${binding.getName}' of attribute '$attribute'")
95       val buckets = opts.getSingle("buckets").orNull
96       require(buckets != null, s"Hash scheme requires a number of buckets to be specified with 'buckets=<n>'")
97       Some(new HashScheme(attribute, index, buckets.toInt, hasher))
98     }
99   }
