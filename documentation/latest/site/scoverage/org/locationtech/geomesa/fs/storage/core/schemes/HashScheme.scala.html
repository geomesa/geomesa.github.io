18 /**
19  * Hash (bucket) scheme
20  *
21  * @param attribute attribute name
22  * @param index index of the attribute in the feature type
23  * @param buckets number of buckets values are hashed into
24  * @param hasher type-specific hasher
25  */
26 case class HashScheme[T](attribute: String, index: Int, buckets: Int, hasher: MurmurHashFunction.Hashing[T])
27     extends PartitionScheme {
28 
29   import FilterHelper.ff
30 
31   override val name: String = s"${HashScheme.Name}:attribute=$attribute:buckets=$buckets"
32 
33   private val format = s"%0${(buckets - 1).toString.length}d"
34   private val default = PartitionKey(name, format.format(0))
35 
36   override def getPartition(feature: SimpleFeature): PartitionKey = {
37     val value = feature.getAttribute(index)
38     if (value == null) {
39       default
40     } else {
41       PartitionKey(name, toPartition(value.asInstanceOf[T]))
42     }
43   }
44 
45   override def getRangesForFilter(filter: Filter): Option[Seq[PartitionRange]] = {
46     getBounds(filter).map { bounds =>
47       val builder = new RangeBuilder()
48       bounds.foreach { bound =>
49         // note: these are all equality values, as determined in getBounds
50         val lower = toPartition(bound.lower.value.get)
51         val upper = lower + ZeroChar
52         builder += PartitionRange(name, lower, upper)
53       }
54       builder.result()
55     }
56   }
57 
58   override def getPartitionsForFilter(filter: Filter): Option[Seq[PartitionKey]] = {
59     getBounds(filter).map { bounds =>
60       bounds.map { bound =>
61         // note: these are all equality values, as determined in getBounds
62         PartitionKey(name, toPartition(bound.lower.value.get))
63       }
64     }
65   }
66 
67   private def getBounds(filter: Filter): Option[Seq[Bounds[T]]] = {
68     val bounds = FilterHelper.extractAttributeBounds(filter, attribute, hasher.binding)
69     if (bounds.isEmpty || bounds.exists(_.isRange)) {
70       None
71     } else if (bounds.disjoint) {
72       Some(Seq.empty)
73     } else {
74       Some(bounds.values)
75     }
76   }
77 
78   private def toPartition(value: T): String = format.format((hasher(value) & Int.MaxValue) % buckets)
79 
80   override def getCoveringFilter(partition: PartitionKey): Filter = {
81     val fn = BucketHashFunction.Name.getFunctionName
82     ff.equals(ff.function(fn, ff.property(attribute), ff.literal(buckets)), ff.literal(partition.value.toInt))
83   }
84 }
85 
86 object HashScheme extends PartitionSchemeFactory {
87 
88   val Name = "hash"
89 
90   override def load(sft: SimpleFeatureType, scheme: String): Option[PartitionScheme] = {
91     val opts = SchemeOpts(scheme)
92     if (opts.name != Name) { None } else {
93       val attribute = opts.getSingle("attribute").orNull
94       require(attribute != null, s"Hash scheme requires an attribute to be specified with 'attribute=<attribute>'")
95       val index = attributeIndex(sft, attribute)
96       val binding = sft.getDescriptor(index).getType.getBinding
97       val hasher = MurmurHashFunction.Hashers.find(_.binding.isAssignableFrom(binding)).orNull
98       require(hasher != null, s"Invalid type binding '${binding.getName}' of attribute '$attribute'")
99       val buckets = opts.getSingle("buckets").orNull
100       require(buckets != null, s"Hash scheme requires a number of buckets to be specified with 'buckets=<n>'")
101       Some(new HashScheme(attribute, index, buckets.toInt, hasher))
102     }
103   }
