13 /**
14  * Mapping of a feature-type attribute to a valid column name (lowercased, only alphanumeric/underscore)
15  *
16  * @param attribute feature type attribute
17  * @param column storage column name
18  */
19 case class ColumnName(attribute: String, column: String)
20 
21 /**
22  * Provides mappings to and from attribute names and valid parquet column names
23  */
24 object ColumnName {
25 
26   /**
27    * Encode an external name into a valid storage column name
28    *
29    * @param name external user-provided name
30    * @return
31    */
32   def encode(name: String): String = {
33     if (name.startsWith("__")) {
34       // internal field (fid, bbox, etc)
35       name
36     } else {
37       name.toLowerCase(Locale.US).replaceAll("[^a-z0-9_]", "_").replaceAll("__+", "_")
38     }
39   }
40   /**
41    * Convert an attribute name into a valid parquet column name
42    *
43    * @param name attribute name
44    * @return
45    */
46   def apply(name: String): ColumnName = ColumnName(name, encode(name))
47 }
