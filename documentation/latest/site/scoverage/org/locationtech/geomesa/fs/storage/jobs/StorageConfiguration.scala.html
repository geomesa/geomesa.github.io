133   /**
134     * Key used for merging feature updates.
135     *
136     * Implements hadoop writable for m/r, kryo serializable for spark, and comparable to sort in
137     * reverse chronological order
138     */
139   class SimpleFeatureAction extends Writable with KryoSerializable with Comparable[SimpleFeatureAction] {
140 
141     private var _id: String = _
142     private var _timestamp: Long = _
143     private var _action: StorageFileAction = _
144 
145     def this(id: String, timestamp: Long, action: StorageFileAction) = {
146       this()
147       this._id = id
148       this._timestamp = timestamp
149       this._action = action
150     }
151 
152     def id: String = _id
153     def timestamp: Long = _timestamp
154     def action: StorageFileAction = _action
155 
156     override def compareTo(o: SimpleFeatureAction): Int = {
157       var res = _id.compareTo(o.id)
158       if (res == 0) {
159         res = _timestamp.compareTo(o.timestamp) * -1 // note: reverse chronological sort
160         if (res == 0) {
161           res = _action.compareTo(o.action)
162         }
163       }
164       res
165     }
166 
167     override def write(out: DataOutput): Unit = {
168       out.writeUTF(_id)
169       out.writeLong(_timestamp)
170       out.writeUTF(_action.toString)
171     }
172 
173     override def readFields(in: DataInput): Unit = {
174       _id = in.readUTF()
175       _timestamp = in.readLong()
176       _action = StorageFileAction.withName(in.readUTF())
177     }
178 
179     override def write(kryo: Kryo, output: Output): Unit = {
180       output.writeString(_id)
181       output.writeLong(_timestamp)
182       output.writeString(_action.toString)
183     }
184 
185     override def read(kryo: Kryo, input: Input): Unit = {
186       _id = input.readString()
187       _timestamp = input.readLong()
188       _action = StorageFileAction.withName(input.readString())
189     }
