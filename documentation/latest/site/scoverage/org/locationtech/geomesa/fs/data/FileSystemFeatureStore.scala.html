40   override def getWriterInternal(query: Query, flags: Int): FeatureWriter[SimpleFeatureType, SimpleFeature] = {
41     // note: check update first as sometimes we get ADD | UPDATE
42     if ((flags & WRITER_UPDATE) == WRITER_UPDATE) {
43       new FileSystemFeatureWriterModify(storage, query.getFilter, config.readThreads)
44     } else if ((flags & WRITER_ADD) == WRITER_ADD) {
45       new FileSystemFeatureWriterAppend(storage)
46     } else {
47       throw new IllegalArgumentException(s"Expected one of $WRITER_ADD or $WRITER_UPDATE, but got: $flags")
48     }
49   }
50 
51   override def buildFeatureType(): SimpleFeatureType = storage.sft

53   override def getBoundsInternal(query: Query): ReferencedEnvelope = {
54     val envelope = new ReferencedEnvelope(org.locationtech.geomesa.utils.geotools.CRS_EPSG_4326)
55     val bboxFieldName = BoundingBoxField.groupName(ColumnName.encode(storage.sft.getGeometryDescriptor.getLocalName))
56     val bboxField = storage.schema.schema.findField(bboxFieldName).`type`().asStructType()
57     val (minFieldIds, maxFieldIds) =
58       Seq(BoundingBoxField.XMin, BoundingBoxField.YMin, BoundingBoxField.XMax, BoundingBoxField.YMax)
59         .map(f => bboxField.field(f).fieldId())
60         .splitAt(2)
61     storage.metadata.files().includeFileStats().forFilter(query.getFilter).scan().foreach { f =>
62       val minBuffers = minFieldIds.map(f.lowerBounds().get)
63       val maxBuffers = maxFieldIds.map(f.upperBounds().get)
64       if (!minBuffers.contains(null) && !maxBuffers.contains(null)) {
65         val Seq(xmin, ymin) = minBuffers.map(Conversions.fromByteBuffer[Float](Types.FloatType.get(), _))
66         val Seq(xmax, ymax) = maxBuffers.map(Conversions.fromByteBuffer[Float](Types.FloatType.get(), _))
67         envelope.expandToInclude(xmin, ymin)
68         envelope.expandToInclude(xmax, ymax)
69       }
70     }
71     envelope
72   }
73 
74   override def getCountInternal(query: Query): Int = {
75     val count = storage.metadata.files().forFilter(query.getFilter).scan().map(_.recordCount()).sum
76     if (count.isValidInt) { count.toInt } else { Int.MaxValue }
77   }

79   override def getReaderInternal(original: Query): FeatureReader[SimpleFeatureType, SimpleFeature] = {
80     import org.locationtech.geomesa.index.conf.QueryHints._
81 
82     val query = new Query(original)
83     // The type name can sometimes be empty such as Query.ALL
84     query.setTypeName(storage.sft.getTypeName)
85 
86     val reader = config.queryTimeout match {
87       case None => storage.getReader(query, threads = config.readThreads)
88       case Some(timeout) =>
89         val filter = Option(query.getFilter).filter(_ != Filter.INCLUDE)
90         new ManagedScan(new FileSystemScanner(storage, query, config.readThreads), Timeout(timeout), query.getTypeName, filter)
91     }
92 
93     // get a closeable java iterator that DelegateSimpleFeatureIterator will process correctly
94     val iter = new FileSystemFeatureIterator(reader)
95 
96     // transforms will be set after getting the iterator
97     val transformSft = query.getHints.getTransformSchema.getOrElse(storage.sft)
98 
99     // note: DelegateSimpleFeatureIterator will close the iterator by checking that it implements Closeable
100     new DelegateSimpleFeatureReader(transformSft, new DelegateSimpleFeatureIterator(iter))
101   }
