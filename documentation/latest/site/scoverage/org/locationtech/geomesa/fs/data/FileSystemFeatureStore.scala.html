33 class FileSystemFeatureStore(
34     val storage: FileSystemStorage,
35     entry: ContentEntry,
36     query: Query,
37     readThreads: Int,
38     writeTimeout: Duration,
39     queryTimeout: Option[Duration],
40   ) extends ContentFeatureStore(entry, query) with LazyLogging {
41 
42   private val sft = storage.metadata.sft
43 
44   override def getWriterInternal(query: Query, flags: Int): FeatureWriter[SimpleFeatureType, SimpleFeature] = {
45     // note: check update first as sometimes we get ADD | UPDATE
46     if ((flags & WRITER_UPDATE) == WRITER_UPDATE) {
47       new FileSystemFeatureWriterModify(storage, sft, query.getFilter, readThreads)
48     } else if ((flags & WRITER_ADD) == WRITER_ADD) {
49       new FileSystemFeatureWriterAppend(storage, sft, writeTimeout)
50     } else {
51       throw new IllegalArgumentException(s"Expected one of $WRITER_ADD or $WRITER_UPDATE, but got: $flags")
52     }
53   }
54 
55   override def buildFeatureType(): SimpleFeatureType = sft
56 
57   override def getBoundsInternal(query: Query): ReferencedEnvelope = {
58     val envelope = new ReferencedEnvelope(org.locationtech.geomesa.utils.geotools.CRS_EPSG_4326)
59     Option(sft.getGeometryDescriptor).foreach { g =>
60       val i = sft.indexOf(g.getLocalName)
61       storage.metadata.getFiles(query.getFilter).foreach { file =>
62         file.spatialBounds.find(_.attribute == i).foreach(b => envelope.expandToInclude(b.envelope))
63       }
64     }
65     envelope
66   }
67 
68   override def getCountInternal(query: Query): Int =
69     storage.metadata.getFiles(query.getFilter).map(_.count).sum.toInt
70 
71   override def getReaderInternal(original: Query): FeatureReader[SimpleFeatureType, SimpleFeature] = {
72     import org.locationtech.geomesa.index.conf.QueryHints._
73 
74     val query = new Query(original)
75     // The type name can sometimes be empty such as Query.ALL
76     query.setTypeName(sft.getTypeName)
77 
78     val reader = queryTimeout match {
79       case None => storage.getReader(query, threads = readThreads)
80       case Some(timeout) =>
81         val filter = Option(query.getFilter).filter(_ != Filter.INCLUDE)
82         new ManagedScan(new FileSystemScanner(storage, query, readThreads), Timeout(timeout), query.getTypeName, filter)
83     }
84     // get a closeable java iterator that DelegateSimpleFeatureIterator will process correctly
85     val iter = new FileSystemFeatureIterator(reader)
86 
87     // transforms will be set after getting the iterator
88     val transformSft = query.getHints.getTransformSchema.getOrElse(sft)
89 
90     // note: DelegateSimpleFeatureIterator will close the iterator by checking that it implements Closeable
91     new DelegateSimpleFeatureReader(transformSft, new DelegateSimpleFeatureIterator(iter))
92   }
93

98   override def canSort(query: Query): Boolean = false
99   override def canFilter(query: Query): Boolean = true
100   override def canRetype(query: Query): Boolean = true

114   private class FileSystemScanner(storage: FileSystemStorage, val query: Query, threads: Int)
115       extends LowLevelScanner[SimpleFeature] {
116 
117     private var reader: CloseableFeatureIterator = _
118 
119     override def iterator: Iterator[SimpleFeature] = synchronized {
120       reader = storage.getReader(query, threads = threads)
121       reader
122     }
123     override def close(): Unit = synchronized {
124       if (reader != null) {
125         reader.close()
126       }
127     }
128   }
129 
130   /**
131     * Iterator for querying file system storage
132     *
133     * Note: implements Closeable and not AutoCloseable so that DelegateFeatureIterator will close it properly
134     *
135     * @param iter delegate iterator
136     */
137   private class FileSystemFeatureIterator(iter: CloseableFeatureIterator)
138       extends java.util.Iterator[SimpleFeature] with Closeable {
139     override def hasNext: Boolean = iter.hasNext
140     override def next(): SimpleFeature = iter.next()
141     override def close(): Unit = iter.close()
142   }
143 
144   /**
145     * Appending feature writer
146     *
147     * @param storage storage instance
148     * @param sft simple feature type
149     * @param timeout write timeout, for flushing partitions
150     */
151   private class FileSystemFeatureWriterAppend(storage: FileSystemStorage, sft: SimpleFeatureType, timeout: Duration)
152       extends FastSettableFeatureWriter with LazyLogging {
153 
154     private val removalListener = new RemovalListener[Partition, Closeable with Flushable]() {
155       override def onRemoval(key: Partition, value: Closeable with Flushable, cause: RemovalCause): Unit = {
156         if (cause == RemovalCause.EXPIRED) {
157           logger.info(s"Flushing writer for partition: $key")
158           FlushWithLogging(value)
159           CloseWithLogging(value)
160         }
161       }
162     }
163 
164     private val writers =
165       Caffeine.newBuilder()
166         .expireAfterAccess(timeout.toMillis, TimeUnit.MILLISECONDS)
167         .removalListener[Partition, FileSystemWriter](removalListener)
168         .build(new CacheLoader[Partition, FileSystemWriter]() {
169           override def load(partition: Partition): FileSystemWriter = storage.getWriter(partition)
170         })
171 
172     private val featureIds = new AtomicLong(0)
173     private var feature: ScalaSimpleFeature = _
174 
175     override def getFeatureType: SimpleFeatureType = sft
176 
177     override def hasNext: Boolean = false
178 
179     override def next(): ScalaSimpleFeature = {
180       feature = new ScalaSimpleFeature(sft, featureIds.getAndIncrement().toString)
181       feature
182     }
183 
184     override def write(): Unit = {
185       val sf = GeoMesaFeatureWriter.featureWithFid(feature)
186       writers.get(Partition(storage.metadata.schemes.map(_.getPartition(sf)))).write(sf)
187       feature = null
188     }
189 
190     override def remove(): Unit = throw new IllegalArgumentException("This writer is append only")
191 
192     override def close(): Unit = {
193       val values = writers.asMap().values().asScala.toSeq
194       CloseQuietly(values).foreach(e => throw e)
195     }
196   }
197

221     override def remove(): Unit = writer.remove()
222     override def close(): Unit = writer.close()
223   }
