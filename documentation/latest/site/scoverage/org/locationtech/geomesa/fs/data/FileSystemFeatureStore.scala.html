34 class FileSystemFeatureStore(
35     val storage: FileSystemStorage,
36     entry: ContentEntry,
37     query: Query,
38     readThreads: Int,
39     writeTimeout: Duration,
40     queryTimeout: Option[Duration],
41   ) extends ContentFeatureStore(entry, query) with LazyLogging {
42 
43   private val sft = storage.metadata.sft
44 
45   override def getWriterInternal(query: Query, flags: Int): FeatureWriter[SimpleFeatureType, SimpleFeature] = {
46     // note: check update first as sometimes we get ADD | UPDATE
47     if ((flags & WRITER_UPDATE) == WRITER_UPDATE) {
48       new FileSystemFeatureWriterModify(storage, sft, query.getFilter, readThreads)
49     } else if ((flags & WRITER_ADD) == WRITER_ADD) {
50       new FileSystemFeatureWriterAppend(storage, sft, writeTimeout)
51     } else {
52       throw new IllegalArgumentException(s"Expected one of $WRITER_ADD or $WRITER_UPDATE, but got: $flags")
53     }
54   }
55 
56   override def buildFeatureType(): SimpleFeatureType = sft
57 
58   override def getBoundsInternal(query: Query): ReferencedEnvelope = {
59     val envelope = new ReferencedEnvelope(org.locationtech.geomesa.utils.geotools.CRS_EPSG_4326)
60     Option(sft.getGeometryDescriptor).foreach { g =>
61       val i = sft.indexOf(g.getLocalName)
62       storage.metadata.getFiles(query.getFilter).foreach { file =>
63         file.spatialBounds.find(_.attribute == i).foreach(b => envelope.expandToInclude(b.envelope))
64       }
65     }
66     envelope
67   }
68 
69   override def getCountInternal(query: Query): Int =
70     storage.metadata.getFiles(query.getFilter).map(_.count).sum.toInt
71 
72   override def getReaderInternal(original: Query): FeatureReader[SimpleFeatureType, SimpleFeature] = {
73     import org.locationtech.geomesa.index.conf.QueryHints._
74 
75     val query = new Query(original)
76     // The type name can sometimes be empty such as Query.ALL
77     query.setTypeName(sft.getTypeName)
78 
79     val reader = queryTimeout match {
80       case None => storage.getReader(query, threads = readThreads)
81       case Some(timeout) =>
82         val filter = Option(query.getFilter).filter(_ != Filter.INCLUDE)
83         new ManagedScan(new FileSystemScanner(storage, query, readThreads), Timeout(timeout), query.getTypeName, filter)
84     }
85     // get a closeable java iterator that DelegateSimpleFeatureIterator will process correctly
86     val iter = new FileSystemFeatureIterator(reader)
87 
88     // transforms will be set after getting the iterator
89     val transformSft = query.getHints.getTransformSchema.getOrElse(sft)
90 
91     // note: DelegateSimpleFeatureIterator will close the iterator by checking that it implements Closeable
92     new DelegateSimpleFeatureReader(transformSft, new DelegateSimpleFeatureIterator(iter))
93   }

119   private class FileSystemScanner(storage: FileSystemStorage, val query: Query, threads: Int)
120       extends LowLevelScanner[SimpleFeature] {
121 
122     private var reader: CloseableFeatureIterator = _
123 
124     override def iterator: Iterator[SimpleFeature] = synchronized {
125       reader = storage.getReader(query, threads = threads)
126       reader
127     }
128     override def close(): Unit = synchronized {
129       if (reader != null) {
130         reader.close()
131       }
132     }
133   }
134 
135   /**
136     * Iterator for querying file system storage
137     *
138     * Note: implements Closeable and not AutoCloseable so that DelegateFeatureIterator will close it properly
139     *
140     * @param iter delegate iterator
141     */
142   private class FileSystemFeatureIterator(iter: CloseableFeatureIterator)
143       extends java.util.Iterator[SimpleFeature] with Closeable {
144     override def hasNext: Boolean = iter.hasNext
145     override def next(): SimpleFeature = iter.next()
146     override def close(): Unit = iter.close()
147   }
148 
149   /**
150     * Appending feature writer
151     *
152     * @param storage storage instance
153     * @param sft simple feature type
154     * @param timeout write timeout, for flushing partitions
155     */
156   private class FileSystemFeatureWriterAppend(storage: FileSystemStorage, sft: SimpleFeatureType, timeout: Duration)
157       extends FastSettableFeatureWriter with LazyLogging {
158 
159     private val removalListener = new RemovalListener[Partition, Closeable with Flushable]() {
160       override def onRemoval(key: Partition, value: Closeable with Flushable, cause: RemovalCause): Unit = {
161         if (cause == RemovalCause.EXPIRED) {
162           logger.info(s"Flushing writer for partition: $key")
163           FlushWithLogging(value)
164           CloseWithLogging(value)
165         }
166       }
167     }
168 
169     private val writers =
170       Caffeine.newBuilder()
171         .expireAfterAccess(timeout.toMillis, TimeUnit.MILLISECONDS)
172         .removalListener[Partition, FileSystemWriter](removalListener)
173         .build(new CacheLoader[Partition, FileSystemWriter]() {
174           override def load(partition: Partition): FileSystemWriter = storage.getWriter(partition)
175         })
176 
177     private val featureIds = new AtomicLong(0)
178     private var feature: ScalaSimpleFeature = _
179 
180     override def getFeatureType: SimpleFeatureType = sft
181 
182     override def hasNext: Boolean = false
183 
184     override def next(): ScalaSimpleFeature = {
185       feature = new ScalaSimpleFeature(sft, featureIds.getAndIncrement().toString)
186       feature
187     }
188 
189     override def write(): Unit = {
190       val sf = GeoMesaFeatureWriter.featureWithFid(feature)
191       writers.get(Partition(storage.metadata.schemes.map(_.getPartition(sf)))).write(sf)
192       feature = null
193     }
194 
195     override def remove(): Unit = throw new IllegalArgumentException("This writer is append only")
196 
197     override def close(): Unit = {
198       val values = writers.asMap().values().asScala.toSeq
199       CloseQuietly(values).foreach(e => throw e)
200     }
201   }
202

226     override def remove(): Unit = writer.remove()
227     override def close(): Unit = writer.close()
228   }
