34 class FileSystemFeatureStore(
35     val storage: FileSystemStorage,
36     entry: ContentEntry,
37     query: Query,
38     readThreads: Int,
39     writeTimeout: Duration,
40     queryTimeout: Option[Duration],
41   ) extends ContentFeatureStore(entry, query) with LazyLogging {
42 
43   private val sft = storage.metadata.sft
44 
45   override def getWriterInternal(query: Query, flags: Int): FeatureWriter[SimpleFeatureType, SimpleFeature] = {
46     // note: check update first as sometimes we get ADD | UPDATE
47     if ((flags & WRITER_UPDATE) == WRITER_UPDATE) {
48       new FileSystemFeatureWriterModify(storage, sft, query.getFilter, readThreads)
49     } else if ((flags & WRITER_ADD) == WRITER_ADD) {
50       new FileSystemFeatureWriterAppend(storage, sft, writeTimeout)
51     } else {
52       throw new IllegalArgumentException(s"Expected one of $WRITER_ADD or $WRITER_UPDATE, but got: $flags")
53     }
54   }
55 
56   override def buildFeatureType(): SimpleFeatureType = sft
57 
58   override def getBoundsInternal(query: Query): ReferencedEnvelope = {
59     val envelope = new ReferencedEnvelope(org.locationtech.geomesa.utils.geotools.CRS_EPSG_4326)
60     Option(sft.getGeometryDescriptor).foreach { g =>
61       val i = sft.indexOf(g.getLocalName)
62       storage.metadata.getFiles(query.getFilter).foreach { file =>
63         file.bounds.find(_.attribute == i).foreach { b =>
64           b.decode(sft).productIterator.foreach {
65             case g: Geometry => envelope.expandToInclude(g.getEnvelopeInternal)
66           }
67         }
68       }
69     }
70     envelope
71   }
72 
73   override def getCountInternal(query: Query): Int =
74     storage.metadata.getFiles(query.getFilter).map(_.count).sum.toInt
75 
76   override def getReaderInternal(original: Query): FeatureReader[SimpleFeatureType, SimpleFeature] = {
77     import org.locationtech.geomesa.index.conf.QueryHints._
78 
79     val query = new Query(original)
80     // The type name can sometimes be empty such as Query.ALL
81     query.setTypeName(sft.getTypeName)
82 
83     val reader = queryTimeout match {
84       case None => storage.getReader(query, threads = readThreads)
85       case Some(timeout) =>
86         val filter = Option(query.getFilter).filter(_ != Filter.INCLUDE)
87         new ManagedScan(new FileSystemScanner(storage, query, readThreads), Timeout(timeout), query.getTypeName, filter)
88     }
89 
90     // get a closeable java iterator that DelegateSimpleFeatureIterator will process correctly
91     val iter = new FileSystemFeatureIterator(reader)
92 
93     // transforms will be set after getting the iterator
94     val transformSft = query.getHints.getTransformSchema.getOrElse(sft)
95 
96     // note: DelegateSimpleFeatureIterator will close the iterator by checking that it implements Closeable
97     new DelegateSimpleFeatureReader(transformSft, new DelegateSimpleFeatureIterator(iter))
98   }

120   private class FileSystemScanner(storage: FileSystemStorage, val query: Query, threads: Int)
121       extends LowLevelScanner[SimpleFeature] {
122 
123     private var reader: CloseableFeatureIterator = _
124 
125     override def iterator: Iterator[SimpleFeature] = synchronized {
126       reader = storage.getReader(query, threads = threads)
127       reader
128     }
129     override def close(): Unit = synchronized {
130       if (reader != null) {
131         reader.close()
132       }
133     }
134   }

136   /**
137     * Iterator for querying file system storage
138     *
139     * Note: implements Closeable and not AutoCloseable so that DelegateFeatureIterator will close it properly
140     *
141     * @param iter delegate iterator
142     */
143   private class FileSystemFeatureIterator(iter: CloseableFeatureIterator)
144       extends java.util.Iterator[SimpleFeature] with Closeable {
145     override def hasNext: Boolean = iter.hasNext
146     override def next(): SimpleFeature = iter.next()
147     override def close(): Unit = iter.close()
148   }
149 
150   /**
151     * Appending feature writer
152     *
153     * @param storage storage instance
154     * @param sft simple feature type
155     * @param timeout write timeout, for flushing partitions
156     */
157   private class FileSystemFeatureWriterAppend(storage: FileSystemStorage, sft: SimpleFeatureType, timeout: Duration)
158       extends FastSettableFeatureWriter with LazyLogging {
159 
160     private val removalListener = new RemovalListener[Partition, Closeable with Flushable]() {
161       override def onRemoval(key: Partition, value: Closeable with Flushable, cause: RemovalCause): Unit = {
162         if (cause == RemovalCause.EXPIRED) {
163           logger.info(s"Flushing writer for partition: $key")
164           FlushWithLogging(value)
165           CloseWithLogging(value)
166         }
167       }
168     }
169 
170     private val writers =
171       Caffeine.newBuilder()
172         .expireAfterAccess(timeout.toMillis, TimeUnit.MILLISECONDS)
173         .removalListener[Partition, FileSystemWriter](removalListener)
174         .build(new CacheLoader[Partition, FileSystemWriter]() {
175           override def load(partition: Partition): FileSystemWriter = storage.getWriter(partition)
176         })
177 
178     private val featureIds = new AtomicLong(0)
179     private var feature: ScalaSimpleFeature = _
180 
181     override def getFeatureType: SimpleFeatureType = sft
182 
183     override def hasNext: Boolean = false
184 
185     override def next(): ScalaSimpleFeature = {
186       feature = new ScalaSimpleFeature(sft, featureIds.getAndIncrement().toString)
187       feature
188     }
189 
190     override def write(): Unit = {
191       val sf = GeoMesaFeatureWriter.featureWithFid(feature)
192       writers.get(Partition(storage.metadata.schemes.map(_.getPartition(sf)))).write(sf)
193       feature = null
194     }
195 
196     override def remove(): Unit = throw new IllegalArgumentException("This writer is append only")
197 
198     override def close(): Unit = {
199       val values = writers.asMap().values().asScala.toSeq
200       CloseQuietly(values).foreach(e => throw e)
201     }
202   }
203

227     override def remove(): Unit = writer.remove()
228     override def close(): Unit = writer.close()
229   }
