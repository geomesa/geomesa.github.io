45   override def getWriterInternal(query: Query, flags: Int): FeatureWriter[SimpleFeatureType, SimpleFeature] = {
46     // note: check update first as sometimes we get ADD | UPDATE
47     if ((flags & WRITER_UPDATE) == WRITER_UPDATE) {
48       new FileSystemFeatureWriterModify(storage, sft, query.getFilter, config.readThreads)
49     } else if ((flags & WRITER_ADD) == WRITER_ADD) {
50       new FileSystemFeatureWriterAppend(storage, sft, config.writeMaxOpenPartitions, config.writeTimeout.toMillis)
51     } else {
52       throw new IllegalArgumentException(s"Expected one of $WRITER_ADD or $WRITER_UPDATE, but got: $flags")
53     }
54   }

58   override def getBoundsInternal(query: Query): ReferencedEnvelope = {
59     val envelope = new ReferencedEnvelope(org.locationtech.geomesa.utils.geotools.CRS_EPSG_4326)

76   override def getReaderInternal(original: Query): FeatureReader[SimpleFeatureType, SimpleFeature] = {
77     import org.locationtech.geomesa.index.conf.QueryHints._
78 
79     val query = new Query(original)
80     // The type name can sometimes be empty such as Query.ALL
81     query.setTypeName(sft.getTypeName)
82 
83     val reader = config.queryTimeout match {
84       case None => storage.getReader(query, threads = config.readThreads)
85       case Some(timeout) =>
86         val filter = Option(query.getFilter).filter(_ != Filter.INCLUDE)
87         new ManagedScan(new FileSystemScanner(storage, query, config.readThreads), Timeout(timeout), query.getTypeName, filter)
88     }

90     // get a closeable java iterator that DelegateSimpleFeatureIterator will process correctly
91     val iter = new FileSystemFeatureIterator(reader)
92 
93     // transforms will be set after getting the iterator
94     val transformSft = query.getHints.getTransformSchema.getOrElse(sft)
95 
96     // note: DelegateSimpleFeatureIterator will close the iterator by checking that it implements Closeable
97     new DelegateSimpleFeatureReader(transformSft, new DelegateSimpleFeatureIterator(iter))
98   }

156   /**
157    * Appending feature writer
158    *
159    * @param storage storage instance
160    * @param sft simple feature type
161    * @param maxOpenPartitions max open partition writers
162    * @param writerTimeoutMillis write timeout, for flushing partitions
163    */
164   private class FileSystemFeatureWriterAppend(
165       storage: FileSystemStorage,
166       sft: SimpleFeatureType,
167       maxOpenPartitions: Int,
168       writerTimeoutMillis: Long
169     ) extends FastSettableFeatureWriter with LazyLogging {
170 
171     @volatile
172     private var closed = false
173 
174     private val writers = new ConcurrentHashMap[Partition, WriterHolder]()
175 
176     private val computeForWrite: BiFunction[Partition, WriterHolder, WriterHolder] = (p, current) => {
177       if (current == null) {
178         logger.debug(s"Opening writer for partition $p")
179         new WriterHolder(storage.getWriter(p))
180       } else {
181         current.updateAccessTime() // update access time before returning so it's not expired
182         current
183       }
184     }
185 
186     private val computeForExpire: BiFunction[Partition, WriterHolder, WriterHolder] = (p, current) => {
187       // double check that the writer is still expired, as this method is atomic wrt computeForWrite
188       if (current != null && current.age >= writerTimeoutMillis) {
189         logger.debug(
190           s"Closing writer for partition $p (last accessed ${current.age}ms ago) due to hitting " +
191             s"expiry ${writerTimeoutMillis}ms")
192         CloseWithLogging(current)
193         null
194       } else {
195         current
196       }
197     }
198 
199     private val ex = Executors.newSingleThreadScheduledExecutor()
200 
201     private val expire: Runnable = () => {
202       var nextRun = writerTimeoutMillis
203       writers.entrySet().forEach { entry =>
204         if (!closed) {
205           val delta = writerTimeoutMillis - entry.getValue.age
206           if (delta < 1) {
207             try { writers.computeIfPresent(entry.getKey, computeForExpire) } catch {
208               case NonFatal(e) => logger.warn(s"Error expiring feature writer for partition ${entry.getKey}:", e)
209             }
210           } else {
211             nextRun = math.min(delta, nextRun)
212           }
213         }
214       }
215       if (!closed) {
216         try {
217           ex.synchronized {
218             future = ex.schedule(expire, nextRun + 1000, TimeUnit.MILLISECONDS)
219           }
220         } catch {
221           case _: RejectedExecutionException => // executor was shutdown due to close
222         }
223       }
224     }
225 
226     private var future: ScheduledFuture[_] = ex.schedule(expire, writerTimeoutMillis + 1000, TimeUnit.MILLISECONDS)
227 
228     private val featureIds = new AtomicLong(0)
229     private var feature: ScalaSimpleFeature = _
230 
231     override def getFeatureType: SimpleFeatureType = sft
232 
233     override def hasNext: Boolean = false
234 
235     override def next(): ScalaSimpleFeature = {
236       feature = new ScalaSimpleFeature(sft, featureIds.getAndIncrement().toString)
237       feature
238     }
239 
240     override def write(): Unit = {
241       val sf = GeoMesaFeatureWriter.featureWithFid(feature)
242       val partition = Partition(storage.metadata.schemes.map(_.getPartition(sf)))
243       writers.compute(partition, computeForWrite).write(sf)
244       feature = null
245       if (writers.size() > maxOpenPartitions) {
246         val oldest = writers.reduceEntries(maxOpenPartitions / 2, WriterHolder.Oldest)
247         // may have been expired asynchronously due to inactivity
248         if (writers.remove(oldest.getKey, oldest.getValue)) {
249           logger.debug(
250             s"Closing writer for partition ${oldest.getKey} (last accessed ${oldest.getValue.age}ms ago) due to hitting max " +
251               s"writer threshold $maxOpenPartitions")
252           CloseWithLogging(oldest.getValue)
253         }
254       }
255     }
256 
257     override def remove(): Unit = throw new IllegalArgumentException("This writer is append only")
258 
259     override def close(): Unit = {
260       try {
261         closed = true
262         ex.synchronized {
263           future.cancel(false)
264           ex.shutdown()
265         }
266         // ensure all writers are closed before returning from this method
267         ex.awaitTermination(Long.MaxValue, TimeUnit.MILLISECONDS)
268       } finally {
269         var ex: Throwable = null
270         writers.forEach { (p, writer) =>
271           logger.debug(s"Closing writer for partition $p")
272           CloseQuietly(writer).foreach(e => if (ex == null) { ex = e } else { ex.addSuppressed(e) })
273         }
274         if (ex != null) {
275           throw ex
276         }
277       }
278     }
279   }
280 
281   private class WriterHolder(writer: FileSystemWriter) extends Closeable {
282 
283     @volatile
284     private var lastAccess: Long = System.currentTimeMillis()
285     private val closed: AtomicBoolean = new AtomicBoolean(false)
286 
287     def write(sf: SimpleFeature): Unit = writer.write(sf)
288     def age: Long = System.currentTimeMillis() - lastAccess
289     def updateAccessTime(): Unit = lastAccess = System.currentTimeMillis()
290 
291     override def close(): Unit = {
292       if (closed.compareAndSet(false, true)) {
293         writer.close()
294       }
295     }
296   }
297 
298   private object WriterHolder {
299     val Oldest: BiFunction[Entry[Partition, WriterHolder], Entry[Partition, WriterHolder], Entry[Partition, WriterHolder]] =
300       (left, right) => if (right.getValue.age < left.getValue.age) { right } else { left }
301   }
