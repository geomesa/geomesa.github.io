30 /**
31  * Scheme for partitioning based on "receipt time", i.e. when a message is received. Generally this is useful
32  * only for reading existing data that may have been aggregated and stored by an external process.
33  *
34  * @param buffer amount of time to buffer queries by, in order to match a feature date to a receipt time date -
35  *               i.e. the amount of latency in the ingest process
36  */
37 class ReceiptTimeScheme(
38     formatter: DateTimeFormatter,
39     pattern: String,
40     stepUnit: ChronoUnit,
41     step: Int,
42     dtg: String,
43     dtgIndex: Int,
44     val buffer: Duration,
45   ) extends HierarchicalDateTimeScheme(formatter, pattern, stepUnit, step, dtg, dtgIndex) {
46 
47   import ReceiptTimeScheme.Config
48 
49   override val name: String =
50     HierarchicalDateTimeScheme(formatter, pattern, stepUnit, step, dtg, dtgIndex).name
51       .replaceFirst(HierarchicalDateTimeScheme.Name, ReceiptTimeScheme.Name) + s":${Config.BufferOpt}=${buffer.toMillis}ms"
52 
53   override def getRangesForFilter(filter: Filter): Option[Seq[PartitionRange]] =
54     super.getRangesForFilter(buffered(filter))
55 
56   override def getCoveringFilter(partition: PartitionKey): Filter =
57     throw new UnsupportedOperationException("Dates may overlap in multiple partitions")
58 
59   private def buffered(filter: Filter): Filter =
60     filter.accept(new BufferingFilterVisitor(buffer, dtg), null).asInstanceOf[Filter]
61 }

97     override def visit(filter: PropertyIsBetween, extraData: AnyRef): AnyRef = {
98       val factory = getFactory(extraData)
99       def buffer(p: PropertyName, lower: Literal, upper: Literal): Option[Filter] = {
100         for {
101           lo <- Option(FastConverter.evaluate(lower, classOf[Date]))
102           up <- Option(FastConverter.evaluate(upper, classOf[Date]))
103         } yield {
104           val bufferedLo = bufferDown(lo)
105           val bufferedUp = bufferUp(up)
106           // account for inverted filters that may result in invalid clauses after buffering
107           if (bufferedLo.before(bufferedUp)) {
108             factory.between(p, factory.literal(bufferedLo), factory.literal(bufferedUp), filter.getMatchAction)
109           } else {
110             Filter.EXCLUDE
111           }
112         }
113       }
114       val prop = visit(filter.getExpression, extraData)
115       val lowerBoundary = visit(filter.getLowerBoundary, extraData)
116       val upperBoundary = visit(filter.getUpperBoundary, extraData)
117       val buffered = (prop, lowerBoundary, upperBoundary) match {
118         case (p: PropertyName, lower: Literal, upper: Literal) if p.getPropertyName == dtg => buffer(p, lower, upper)
119         case _ => None
120       }
121 
122       buffered.getOrElse(super.visit(filter, extraData))
123     }

288     private def buffer(lit: Literal, offset: Long, extraData: AnyRef): Expression = {
