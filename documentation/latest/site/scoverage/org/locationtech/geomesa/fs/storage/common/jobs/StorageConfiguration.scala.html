138   /**
139     * Key used for merging feature updates.
140     *
141     * Implements hadoop writable for m/r, kryo serializable for spark, and comparable to sort in
142     * reverse chronological order
143     */
144   class SimpleFeatureAction extends Writable with KryoSerializable with Comparable[SimpleFeatureAction] {
145 
146     private var _id: String = _
147     private var _timestamp: Long = _
148     private var _action: StorageFileAction = _
149 
150     def this(id: String, timestamp: Long, action: StorageFileAction) = {
151       this()
152       this._id = id
153       this._timestamp = timestamp
154       this._action = action
155     }
156 
157     def id: String = _id
158     def timestamp: Long = _timestamp
159     def action: StorageFileAction = _action
160 
161     override def compareTo(o: SimpleFeatureAction): Int = {
162       var res = _id.compareTo(o.id)
163       if (res == 0) {
164         res = _timestamp.compareTo(o.timestamp) * -1 // note: reverse chronological sort
165         if (res == 0) {
166           res = _action.compareTo(o.action)
167         }
168       }
169       res
170     }
171 
172     override def write(out: DataOutput): Unit = {
173       out.writeUTF(_id)
174       out.writeLong(_timestamp)
175       out.writeUTF(_action.toString)
176     }
177 
178     override def readFields(in: DataInput): Unit = {
179       _id = in.readUTF()
180       _timestamp = in.readLong()
181       _action = StorageFileAction.withName(in.readUTF())
182     }
183 
184     override def write(kryo: Kryo, output: Output): Unit = {
185       output.writeString(_id)
186       output.writeLong(_timestamp)
187       output.writeString(_action.toString)
188     }
189 
190     override def read(kryo: Kryo, input: Input): Unit = {
191       _id = input.readString()
192       _timestamp = input.readLong()
193       _action = StorageFileAction.withName(input.readString())
194     }
195   }
