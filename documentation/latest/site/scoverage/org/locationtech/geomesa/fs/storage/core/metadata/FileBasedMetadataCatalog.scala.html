21 /**
22  * Catalog for file-based metadata
23  *
24  * @param context file system
25  */
26 class FileBasedMetadataCatalog(context: FileSystemContext) extends StorageMetadataCatalog with LazyLogging {
27 
28   private val directory = context.root.resolve(FileBasedMetadataCatalog.MetadataDirectory + "/")
29 
30   override def getTypeNames: Seq[String] = {
31     WithClose(ObjectStore(context)) { fs =>
32       val iter = fs.list(directory).flatMap { file =>
33         Option(file.getPath).flatMap { path =>
34           Option(path.lastIndexOf('/'))
35             .collect { case i if i != -1 => path.substring(i + 1) }
36             .collect { case n if !n.startsWith(".") && n.endsWith(".json") =>
37               StringSerialization.decodeAlphaNumericSafeString(n.substring(0, n.length - 5))
38             }
39         }
40       }
41       WithClose(iter)(_.toList)
42     }
43   }
44 
45   override def load(typeName: String): FileBasedMetadata = {
46     val file = directory.resolve(StringSerialization.alphaNumericSafeString(typeName) + ".json")
47     val fs = ObjectStore(context)
48     try {
49       WithClose(fs.read(file)) { opt =>
50         val is = opt.getOrElse {
51           throw new IllegalArgumentException(s"Type '$typeName' does not exit")
52         }
53         val meta = MetadataSerialization.deserialize(is)
54         new FileBasedMetadata(fs, meta.copy(sft = namespaced(meta.sft, context.namespace)), directory)
55       }
56     } catch {
57       case NonFatal(e) => CloseWithLogging(fs); throw e
58     }
59   }
60 
61   override def create(sft: SimpleFeatureType, partitions: Seq[String], targetFileSize: Option[Long]): FileBasedMetadata = {
62     // load the partition scheme first in case it fails
63     partitions.foreach(PartitionSchemeFactory.load(sft, _))
64     val meta = Metadata(sft, partitions, targetFileSize)
65     val file = directory.resolve(StringSerialization.alphaNumericSafeString(sft.getTypeName) + ".json")
66     val fs = ObjectStore(context)
67     try {
68       WithClose(fs.create(file)) {
69         case Some(out) => MetadataSerialization.serialize(out, meta)
70         case None => logger.warn(s"Trying to create schema but it already exists: ${sft.getTypeName}")
71       }
72       new FileBasedMetadata(fs, meta.copy(sft = namespaced(meta.sft, context.namespace)), directory)
73     } catch {
74       case NonFatal(e) => CloseWithLogging(fs); throw e
75     }
76   }
77 }
