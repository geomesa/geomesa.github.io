22 /**
23  * Mixin trait for matching partitions against a CQL filter
24  */
25 trait SchemeFilterExtraction extends AnyLogging {
26 
27   this: StorageMetadata =>
28 
29   protected def getFilters(filter: Filter): Seq[SchemeFilter] = {
30     val iter = schemes.iterator
31     val start = iter.map(_.getRangesForFilter(filter)).collectFirst { case Some(f) => f }
32     val filters = start match {
33       case None =>
34         // filter did not constrain partitions at all
35         Seq(SchemeFilter(filter, Seq.empty, extractGeometries(filter), extractAttributes(filter)))
36 
37       case Some(ranges) =>
38         // set up the initial scheme filters, without any spatial/attribute bounds
39         val initialRanges = ranges.map(r => Seq(r))
40         // add in the remaining partitions
41         val permutations = iter.foldLeft(initialRanges) { case (ranges, scheme) => addPartition(scheme, ranges, filter) }
42         // add in the spatial/attribute bounds based on the remaining filter for each scheme filter
43         val spatialBounds = extractGeometries(filter)
44         val attributeBounds = extractAttributes(filter)
45         permutations.map(ranges => SchemeFilter(filter, ranges, spatialBounds, attributeBounds))
46     }
47 
48     logger.debug(s"Extracted filters from ${ECQL.toCQL(filter)}:\n  ${filters.mkString("\n  ")}")
49     filters
50   }
51 
52   // add the next partition to the scheme filter
53   private def addPartition(scheme: PartitionScheme, ranges: Seq[Seq[PartitionRange]], filter: Filter): Seq[Seq[PartitionRange]] = {
54     val result = Seq.newBuilder[Seq[PartitionRange]]
55     ranges.foreach { filesTableFilter =>
56       scheme.getRangesForFilter(filter) match {
57         case None => result += filesTableFilter
58         case Some(ranges) =>
59           ranges.foreach { range =>
60             result += filesTableFilter :+ range
61           }
62       }
63     }
64     result.result()
65   }
66 
67   private def extractGeometries(filter: Filter): Seq[SpatialOr] = {
68     sft.spatialBounds().flatMap { i =>
69       val ors = FilterHelper.extractGeometries(filter, sft.getDescriptor(i).getLocalName).values.flatMap { g =>
70         SpatialBound(g.getEnvelopeInternal)
71       }
72       if (ors.isEmpty) { None } else { Some(SpatialOr(i, ors)) }
73     }
74   }
75 
76   private def extractAttributes(filter: Filter): Seq[AttributeOr] = {
77     sft.nonSpatialBounds().flatMap { i =>
78       val d = sft.getDescriptor(i)
79       val ors = FilterHelper.extractAttributeBounds(filter, d.getLocalName, d.getType.getBinding).values.map { b =>
80         val lower = b.lower.value.map(AttributeIndexKey.typeEncode).getOrElse("")
81         val upper = b.upper.value.map(AttributeIndexKey.typeEncode).getOrElse("zzz")
82         AttributeBound(lower, upper)
83       }
84       if (ors.isEmpty) { None } else { Some(AttributeOr(i, ors)) }
85     }
86   }
87 }

89 object SchemeFilterExtraction {
90 
91   /**
92    * Predicates extracted from a filter, based on the partition scheme
93    *
94    * @param filter remaining ECQL filter that isn't accounted for with the other bounds
95    * @param partitions list of partition bound predicates (implicit AND between each element in the seq)
96    * @param spatialBounds list of spatial bound predicates (implicit AND between each element in the seq)
97    * @param attributeBounds list of attribute bound predicates (implicit AND between each element in the seq)
98    */
99   case class SchemeFilter(
100     filter: Filter,
101     partitions: Seq[PartitionRange],
102     spatialBounds: Seq[SpatialOr],
103     attributeBounds: Seq[AttributeOr]
104   )
105 
106   case class SpatialOr(attribute: Int, bounds: Seq[SpatialBound])
107   case class AttributeOr(attribute: Int, bounds: Seq[AttributeBound])
108 
109   case class SpatialBound(xmin: Double, ymin: Double, xmax: Double, ymax: Double) {
110     def intersects(other: SpatialBounds): Boolean =
111       other.xmin <= xmax && other.xmax >= xmin && other.ymin <= ymax && other.ymax >= ymin
112   }
113 
114   object SpatialBound {
115 
116     /**
117      * Converts an envelope to a bounds, handling 'null' (empty) envelopes
118      *
119      * @param env envelope
120      * @return
121      */
122     def apply(env: Envelope): Option[SpatialBound] =
123       if (env.isNull) { None } else { Some(SpatialBound(env.getMinX, env.getMinY, env.getMaxX, env.getMaxY)) }
124   }
125 
126   case class AttributeBound(lower: String, upper: String) {
127     def intersects(other: AttributeBounds): Boolean = other.lower <= upper && other.upper >= lower
128   }
