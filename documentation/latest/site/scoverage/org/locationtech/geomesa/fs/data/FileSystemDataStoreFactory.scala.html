39   override def createDataStore(params: java.util.Map[String, _]): DataStore = {
40     // Need to do more tuning here. On a local system 1 thread (so basic producer/consumer) was best
41     // because Parquet is also threading the reads underneath I think. using prod/cons pattern was
42     // about 30% faster but increasing beyond 1 thread slowed things down. This could be due to the
43     // cost of serializing simple features though. need to investigate more.
44     //
45     // However, if you are doing lots of filtering it appears that bumping the threads up high
46     // can be very useful. Seems possibly numcores/2 might is a good setting (which is a standard idea)
47 
48     val readThreads = QueryThreadsParam.lookup(params)
49     val queryTimeout = QueryTimeoutParam.lookupOpt(params).filter(_.isFinite)
50 
51     val namespace = NamespaceParam.lookupOpt(params)
52 
53     val path = new URI(PathParam.lookup(params))
54     val context = FileSystemContext.create(path, buildConf(params), namespace)
55     val config = FileSystemDataStoreConfig(context, readThreads, queryTimeout)
56 
57     lazy val encodingType =
58       if (params.get("fs.encoding") == "converter") {
59         logger.warn(s"Using deprecated parameter 'fs.encoding' - please switch to '${CatalogTypeParam.key}' instead")
60         true
61       } else {
62         false
63       }
64 
65     val catalogType = context.conf.getOrElse(CatalogUtil.ICEBERG_CATALOG_TYPE, null)
66     val catalog =
67       if (catalogType == ConverterCatalog.CatalogType || (catalogType == null && encodingType)) {
68         new ConverterCatalog(context)
69       } else {
70         new IcebergCatalog(context)
71       }
72 
73     new FileSystemDataStore(catalog, config)
74   }

91   private def buildConf(params: java.util.Map[String, _]): Map[String, String] = {
92     val builder = Map.newBuilder[String, String]
93     // pick up any hadoop props, to e.g. make it a bit easier to configure s3 access based on s3a settings
94     // only do this if hadoop file exist on the classpath, to avoid a hard runtime dependency on hadoop
95     if (Seq("core-site.xml", "hdfs-site.xml").exists(getClass.getClassLoader.getResource(_) != null)) {
96       FileSystemDataStoreFactory.configuration.forEach { e =>
97         val key = e.getKey
98         if (key.startsWith("geomesa.") || key.startsWith("fs.") || key.startsWith("parquet.")) {
99           builder += e.getKey -> FileSystemDataStoreFactory.configuration.get(e.getKey) // use .get to resolve envs
100         }
101       }
102     }
103     ConfigXmlParam.lookupOpt(params).foreach { xml =>
104       logger.warn(s"Parameter '${ConfigXmlParam.key}' is deprecated, please use '${ConfigParam.key}' instead")
105       val conf = new Configuration(false)
106       conf.addResource(new ByteArrayInputStream(xml.getBytes(StandardCharsets.UTF_8)))
107       // note: need to call iterator() to force loading of the resource
108       conf.iterator().forEachRemaining { e => builder += e.getKey -> conf.get(e.getKey) } // use .get to resolve envs
109     }
110     ConfigPathsParam.lookupOpt(params).map(_.split(',').map(_.trim).filterNot(_.isEmpty)).filterNot(_.isEmpty).foreach { resources =>
111       logger.warn(s"Parameter '${ConfigPathsParam.key}' is deprecated, please use '${ConfigFileParam.key}' instead")
112       val conf = new Configuration(false)
113       resources.foreach(HadoopUtils.addResource(conf, _))
114       // note: need to call iterator() to force loading of the resource
115       conf.iterator().forEachRemaining { e => builder += e.getKey -> conf.get(e.getKey) }  // use .get to resolve envs
116     }
117     ConfigFileParam.lookupOpt(params).foreach { f =>
118       var uri = new URI(f)
119       if (uri.getScheme == null) {
120         uri = new URI("file", uri.getHost, uri.getPath, uri.getFragment)
121       }
122       val asString = try { WithClose(uri.toURL.openStream())(IOUtils.toString(_, StandardCharsets.UTF_8)) } catch {
123         case NonFatal(e) =>
124           throw new IllegalArgumentException(s"Invalid parameter ${ConfigFileParam.key} - could not open file: $f", e)
125       }
126       // use our properties parameter parsing to evaluate env vars
127       ConfigParam.lookup(java.util.Map.of(ConfigParam.key, asString)).asScala.foreach { case (k, v) => builder += k -> v }
128     }
129     ConfigParam.lookupOpt(params).foreach(_.asScala.foreach { case (k, v) => builder += k -> v })
130     AuthProviderParam.lookupOpt(params).foreach(p => builder += (AuthsParam.key -> p.getClass.getName))
131     AuthsParam.lookupOpt(params).foreach(auths => builder += (AuthsParam.key -> auths))
132     CatalogTypeParam.lookupOpt(params).filter(_.nonEmpty).foreach(t => builder += (CatalogUtil.ICEBERG_CATALOG_TYPE -> t))
133     WritersMaxOpenPartitionsParam.lookupOpt(params).foreach(m => builder += (WritersMaxOpenPartitionsParam.key -> m.toString))
134     builder.result()
135   }
136 }
