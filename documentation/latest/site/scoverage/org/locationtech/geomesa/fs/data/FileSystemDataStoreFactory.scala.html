39   override def createDataStore(params: java.util.Map[String, _]): DataStore = {
40     val path = new URI(PathParam.lookup(params))
41     val encoding = EncodingParam.lookup(params)
42     val conf = {
43       val builder = Map.newBuilder[String, String]
44       // pick up any hadoop props, to e.g. make it a bit easier to configure s3 access based on s3a settings
45       FileSystemDataStoreFactory.configuration.forEach { e =>
46         val key = e.getKey
47         if (key.startsWith("geomesa.") || key.startsWith("fs.") || key.startsWith("parquet.")) {
48           builder += e.getKey -> FileSystemDataStoreFactory.configuration.get(e.getKey) // use .get to resolve envs
49         }
50       }
51       ConfigXmlParam.lookupOpt(params).foreach { xml =>
52         logger.warn(s"Parameter '${ConfigXmlParam.key}' is deprecated, please use '${ConfigParam.key}' instead")
53         val conf = new Configuration(false)
54         conf.addResource(new ByteArrayInputStream(xml.getBytes(StandardCharsets.UTF_8)))
55         // note: need to call iterator() to force loading of the resource
56         conf.iterator().forEachRemaining { e => builder += e.getKey -> conf.get(e.getKey) } // re- .get to resolve envs
57       }
58       ConfigPathsParam.lookupOpt(params).map(_.split(',').map(_.trim).filterNot(_.isEmpty)).filterNot(_.isEmpty).foreach { resources =>
59         logger.warn(s"Parameter '${ConfigPathsParam.key}' is deprecated, please use '${ConfigFileParam.key}' instead")
60         val conf = new Configuration(false)
61         resources.foreach(HadoopUtils.addResource(conf, _))
62         // note: need to call iterator() to force loading of the resource
63         conf.iterator().forEachRemaining { e => builder += e.getKey -> conf.get(e.getKey) } // re- .get to resolve envs
64       }
65       ConfigFileParam.lookupOpt(params).foreach { f =>
66         var uri = new URI(f)
67         if (uri.getScheme == null) {
68           uri = new URI("file", uri.getHost, uri.getPath, uri.getFragment)
69         }
70         val asString = try { WithClose(uri.toURL.openStream())(IOUtils.toString(_, StandardCharsets.UTF_8)) } catch {
71           case NonFatal(e) =>
72             throw new IllegalArgumentException(s"Invalid parameter ${ConfigFileParam.key} - could not open file: $f", e)
73         }
74         // use our properties parameter parsing to evaluate env vars
75         ConfigParam.lookup(java.util.Map.of(ConfigParam.key, asString)).asScala.foreach { case (k, v) => builder += k -> v }
76       }
77       ConfigParam.lookupOpt(params).foreach(_.asScala.foreach { case (k, v) => builder += k -> v })
78       AuthProviderParam.lookupOpt(params).foreach(p => builder += (AuthsParam.key -> p.getClass.getName))
79       AuthsParam.lookupOpt(params).foreach(auths => builder += (AuthsParam.key -> auths))
80       MetadataTypeParam.lookupOpt(params).filterNot(_.isBlank) match {
81         case Some(t) => builder += StorageMetadataCatalog.MetadataTypeConfig -> t
82         case None =>
83           if (ConverterStorageFactory.Encoding.equalsIgnoreCase(encoding)) {
84             // for back compatibility, don't require metadata type if using converter storage
85             builder += StorageMetadataCatalog.MetadataTypeConfig -> ConverterMetadata.MetadataType
86           }
87       }
88       builder.result()
89     }
90 
91     if (!conf.contains(StorageMetadataCatalog.MetadataTypeConfig)) {
92       throw new IOException(s"Parameter ${MetadataTypeParam.key} must be specified directly or in ${ConfigParam.key}")
93     }
94 
95     // Need to do more tuning here. On a local system 1 thread (so basic producer/consumer) was best
96     // because Parquet is also threading the reads underneath I think. using prod/cons pattern was
97     // about 30% faster but increasing beyond 1 thread slowed things down. This could be due to the
98     // cost of serializing simple features though. need to investigate more.
99     //
100     // However, if you are doing lots of filtering it appears that bumping the threads up high
101     // can be very useful. Seems possibly numcores/2 might is a good setting (which is a standard idea)
102 
103     val readThreads = QueryThreadsParam.lookup(params)
104     val writeTimeout = WriteTimeoutParam.lookup(params)
105     val queryTimeout = QueryTimeoutParam.lookupOpt(params).filter(_.isFinite)
106 
107     val namespace = NamespaceParam.lookupOpt(params)
108 
109     val storageFactory = FileSystemStorageFactory.factories.find(_.encoding == encoding).getOrElse {
110       // this shouldn't happen since encoding param lookup should fail if it doesn't match a factory
111       throw new RuntimeException(
112         s"Invalid encoding parameter, does not match a storage factory instance: $encoding\n" +
113           s"  Available factories: ${FileSystemStorageFactory.factories.map(_.encoding).mkString(", ")}")
114     }
115 
116     val context = FileSystemContext.create(path, conf, namespace)
117     val config = FileSystemDataStoreConfig(context, readThreads, writeTimeout, queryTimeout)
118     val metadata = StorageMetadataCatalog(context)
119 
120     new FileSystemDataStore(storageFactory, metadata, config)
121   }
