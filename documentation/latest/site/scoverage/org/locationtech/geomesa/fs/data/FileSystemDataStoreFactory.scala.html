39   override def createDataStore(params: java.util.Map[String, _]): DataStore = {
40     val path = new URI(PathParam.lookup(params))
41     val encoding = EncodingParam.lookup(params)
42     val conf = {
43       val builder = Map.newBuilder[String, String]
44       // pick up any hadoop props, to e.g. make it a bit easier to configure s3 access based on s3a settings
45       // only do this if hadoop file exist on the classpath, to avoid a hard runtime dependency on hadoop
46       if (Seq("core-site.xml", "hdfs-site.xml").exists(getClass.getClassLoader.getResource(_) != null)) {
47         FileSystemDataStoreFactory.configuration.forEach { e =>
48           val key = e.getKey
49           if (key.startsWith("geomesa.") || key.startsWith("fs.") || key.startsWith("parquet.")) {
50             builder += e.getKey -> FileSystemDataStoreFactory.configuration.get(e.getKey) // use .get to resolve envs
51           }
52         }
53       }
54       ConfigXmlParam.lookupOpt(params).foreach { xml =>
55         logger.warn(s"Parameter '${ConfigXmlParam.key}' is deprecated, please use '${ConfigParam.key}' instead")
56         val conf = new Configuration(false)
57         conf.addResource(new ByteArrayInputStream(xml.getBytes(StandardCharsets.UTF_8)))
58         // note: need to call iterator() to force loading of the resource
59         conf.iterator().forEachRemaining { e => builder += e.getKey -> conf.get(e.getKey) } // use .get to resolve envs
60       }
61       ConfigPathsParam.lookupOpt(params).map(_.split(',').map(_.trim).filterNot(_.isEmpty)).filterNot(_.isEmpty).foreach { resources =>
62         logger.warn(s"Parameter '${ConfigPathsParam.key}' is deprecated, please use '${ConfigFileParam.key}' instead")
63         val conf = new Configuration(false)
64         resources.foreach(HadoopUtils.addResource(conf, _))
65         // note: need to call iterator() to force loading of the resource
66         conf.iterator().forEachRemaining { e => builder += e.getKey -> conf.get(e.getKey) }  // use .get to resolve envs
67       }
68       ConfigFileParam.lookupOpt(params).foreach { f =>
69         var uri = new URI(f)
70         if (uri.getScheme == null) {
71           uri = new URI("file", uri.getHost, uri.getPath, uri.getFragment)
72         }
73         val asString = try { WithClose(uri.toURL.openStream())(IOUtils.toString(_, StandardCharsets.UTF_8)) } catch {
74           case NonFatal(e) =>
75             throw new IllegalArgumentException(s"Invalid parameter ${ConfigFileParam.key} - could not open file: $f", e)
76         }
77         // use our properties parameter parsing to evaluate env vars
78         ConfigParam.lookup(java.util.Map.of(ConfigParam.key, asString)).asScala.foreach { case (k, v) => builder += k -> v }
79       }
80       ConfigParam.lookupOpt(params).foreach(_.asScala.foreach { case (k, v) => builder += k -> v })
81       AuthProviderParam.lookupOpt(params).foreach(p => builder += (AuthsParam.key -> p.getClass.getName))
82       AuthsParam.lookupOpt(params).foreach(auths => builder += (AuthsParam.key -> auths))
83       MetadataTypeParam.lookupOpt(params).filterNot(_.isBlank) match {
84         case Some(t) => builder += StorageMetadataCatalog.MetadataTypeConfig -> t
85         case None =>
86           if (ConverterStorageFactory.Encoding.equalsIgnoreCase(encoding)) {
87             // for back compatibility, don't require metadata type if using converter storage
88             builder += StorageMetadataCatalog.MetadataTypeConfig -> ConverterMetadata.MetadataType
89           }
90       }
91       builder.result()
92     }
93 
94     if (!conf.contains(StorageMetadataCatalog.MetadataTypeConfig)) {
95       throw new IOException(s"Parameter ${MetadataTypeParam.key} must be specified directly or in ${ConfigParam.key}")
96     }
97 
98     // Need to do more tuning here. On a local system 1 thread (so basic producer/consumer) was best
99     // because Parquet is also threading the reads underneath I think. using prod/cons pattern was
100     // about 30% faster but increasing beyond 1 thread slowed things down. This could be due to the
101     // cost of serializing simple features though. need to investigate more.
102     //
103     // However, if you are doing lots of filtering it appears that bumping the threads up high
104     // can be very useful. Seems possibly numcores/2 might is a good setting (which is a standard idea)
105 
106     val readThreads = QueryThreadsParam.lookup(params)
107     val writeTimeout = WriteTimeoutParam.lookup(params)
108     val queryTimeout = QueryTimeoutParam.lookupOpt(params).filter(_.isFinite)
109 
110     val namespace = NamespaceParam.lookupOpt(params)
111 
112     val storageFactory = FileSystemStorageFactory.factories.find(_.encoding == encoding).getOrElse {
113       // this shouldn't happen since encoding param lookup should fail if it doesn't match a factory
114       throw new RuntimeException(
115         s"Invalid encoding parameter, does not match a storage factory instance: $encoding\n" +
116           s"  Available factories: ${FileSystemStorageFactory.factories.map(_.encoding).mkString(", ")}")
117     }
118 
119     val context = FileSystemContext.create(path, conf, namespace)
120     val config = FileSystemDataStoreConfig(context, readThreads, writeTimeout, queryTimeout)
121     val metadata = StorageMetadataCatalog(context)
122 
123     new FileSystemDataStore(storageFactory, metadata, config)
124   }
