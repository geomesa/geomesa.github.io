19 /**
20  * Encoding for a bbox field

22 object BoundingBoxField {
23 
24   val BoundingBoxFieldSuffix = s"_bbox${SimpleFeatureSchema.InternalFieldDelimiter}"
25 
26   val XMin = "xmin"
27   val YMin = "ymin"
28   val XMax = "xmax"
29   val YMax = "ymax"
30 
31   /**
32    * Gets the column name for the bbox group field
33    *
34    * @param geom geometry column name
35    * @return
36    */
37   def groupName(geom: String): String = s"${SimpleFeatureSchema.InternalFieldDelimiter}$geom$BoundingBoxFieldSuffix"
38 
39   /**
40    * The parquet schema for a bbox field
41    *
42    * @param geom geometry column name
43    * @return
44    */
45   def parquetSchema(geom: String, fieldIds: AtomicInteger, nestedFieldIds: AtomicInteger): GroupType = {
46     val bbox = groupName(geom)
47     Types.optionalGroup()
48       .id(fieldIds.getAndIncrement())
49       .required(PrimitiveTypeName.FLOAT).id(nestedFieldIds.getAndIncrement()).named(BoundingBoxField.XMin)
50       .required(PrimitiveTypeName.FLOAT).id(nestedFieldIds.getAndIncrement()).named(BoundingBoxField.YMin)
51       .required(PrimitiveTypeName.FLOAT).id(nestedFieldIds.getAndIncrement()).named(BoundingBoxField.XMax)
52       .required(PrimitiveTypeName.FLOAT).id(nestedFieldIds.getAndIncrement()).named(BoundingBoxField.YMax)
53       .named(bbox)
54   }
55 
56   /**
57    * The iceberg schema for a bbox field
58    *
59    * @param geom geometry column name
60    * @return
61    */
62   def icebergSchema(geom: String, fieldIds: AtomicInteger, nestedFieldIds: AtomicInteger): NestedField = {
63     val bbox = groupName(geom)
64     NestedField.optional(bbox).withId(fieldIds.getAndIncrement()).ofType(StructType.of(
65       NestedField.required(BoundingBoxField.XMin).withId(nestedFieldIds.getAndIncrement()).ofType(FloatType.get()).build(),
66       NestedField.required(BoundingBoxField.YMin).withId(nestedFieldIds.getAndIncrement()).ofType(FloatType.get()).build(),
67       NestedField.required(BoundingBoxField.XMax).withId(nestedFieldIds.getAndIncrement()).ofType(FloatType.get()).build(),
68       NestedField.required(BoundingBoxField.YMax).withId(nestedFieldIds.getAndIncrement()).ofType(FloatType.get()).build(),
69     )).build()
70   }
71 
72   /**
73    * Create a filter against a bbox col
74    *
75    * @param geom geometry column name
76    * @param xmin xmin
77    * @param ymin ymin
78    * @param xmax xmax
79    * @param ymax ymax
80    * @return
81    */
82   def filterParquet(geom: String, xmin: Double, ymin: Double, xmax: Double, ymax: Double): FilterPredicate = {
83     val bbox = groupName(geom)
84     val xminCol = FilterApi.floatColumn(s"$bbox.$XMin")
85     val yminCol = FilterApi.floatColumn(s"$bbox.$YMin")
86     val xmaxCol = FilterApi.floatColumn(s"$bbox.$XMax")
87     val ymaxCol = FilterApi.floatColumn(s"$bbox.$YMax")
88     val filters = Seq[FilterPredicate](
89       FilterApi.ltEq(xminCol, Float.box(xmax.toFloat)),
90       FilterApi.gtEq(xmaxCol, Float.box(xmin.toFloat)),
91       FilterApi.ltEq(yminCol, Float.box(ymax.toFloat)),
92       FilterApi.gtEq(ymaxCol, Float.box(ymin.toFloat))
93     )
94     filters.reduce(FilterApi.and)
95   }
96 
97   /**
98    * Create a filter against a bbox col
99    *
100    * @param geom geometry column name
101    * @param xmin xmin
102    * @param ymin ymin
103    * @param xmax xmax
104    * @param ymax ymax
105    * @return
106    */
107   def filterIceberg(geom: String, xmin: Double, ymin: Double, xmax: Double, ymax: Double): Expression = {
108     val bbox = groupName(geom)
109     val exps = Seq(
110       Expressions.lessThanOrEqual(s"$bbox.$XMin", Float.box(xmax.toFloat)),
111       Expressions.greaterThanOrEqual(s"$bbox.$XMax", Float.box(xmin.toFloat)),
112       Expressions.lessThanOrEqual(s"$bbox.$YMin", Float.box(ymax.toFloat)),
113       Expressions.greaterThanOrEqual(s"$bbox.$YMax", Float.box(ymin.toFloat))
114     )
115     exps.reduce(Expressions.and)
116   }
