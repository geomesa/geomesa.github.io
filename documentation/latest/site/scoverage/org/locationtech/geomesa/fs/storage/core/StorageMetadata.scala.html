141   /**
142    * Immutable representation of an envelope
143    *
144    * Note that conversions to/from 'null' envelopes should be handled carefully, as envelopes are considered
145    * null if xmin > xmax, however, when instantiating an envelope it will re-order the coordinates:
146    *
147    * {{{
148    *   val env = new Envelope()
149    *   val copy = new Envelope(env.getMinX, env.getMinY, env.getMaxX, env.getMaxY)
150    *   copy == env // false
151    * }}}
152    *
153    * Thus, ensure that 'null' envelopes are converted to `None` and not directly to a bounds object. See
154    * `PartitionBounds.apply`
155    *
156    * @param attribute index of the attribute being bounded
157    * @param xmin min x dimension
158    * @param ymin min y dimension
159    * @param xmax max x dimension
160    * @param ymax max y dimension
161    */
162   case class SpatialBounds(attribute: Int, xmin: Double, ymin: Double, xmax: Double, ymax: Double) {
163 
164     /**
165      * Calculate the minimal bounds encompassing both bounds
166      *
167      * @param b other bounds
168      * @return
169      */
170     def +(b: SpatialBounds): SpatialBounds = {
171       require(attribute == b.attribute, "Trying to merge bounds from different attributes")
172       SpatialBounds(attribute, math.min(xmin, b.xmin), math.min(ymin, b.ymin), math.max(xmax, b.xmax), math.max(ymax, b.ymax))
173     }
174 
175     /**
176      * Convert to a mutable envelope
177      *
178      * @return
179      */
180     def envelope: Envelope = new Envelope(xmin, xmax, ymin, ymax)
181   }
182 
183   object SpatialBounds {
184 
185     /**
186      * Converts an envelope to a bounds, handling 'null' (empty) envelopes
187      *
188      * @param env envelope
189      * @return
190      */
191     def apply(attribute: Int, env: Envelope): Option[SpatialBounds] =
192       if (env.isNull) { None } else { Some(SpatialBounds(attribute, env.getMinX, env.getMinY, env.getMaxX, env.getMaxY)) }
193   }
