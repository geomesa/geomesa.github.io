22 /**
23   * Metadata interface for managing storage partitions
24   */
25 trait StorageMetadata extends Closeable {
26 
27   /**

34   /**

41   /**

48   /**

104    * @param key key

126   /**
127    * Holds a storage file
128    *
129    * @param file file name (relative to the root path)
130    * @param partition list of partitions that the file belongs to
131    * @param count number of entries in the file
132    * @param action type of file (append, modify, delete)
133    * @param bounds known column bounds, if any, keyed by feature type attribute number
134    * @param sort sort fields for the file, if any, as feature type attribute number
135    * @param timestamp timestamp for the file

137   case class StorageFile(

155   /**
156    * Bounds for an attribute

159    * @param lower lower bound (lexicoded)

162   case class ColumnBounds(attribute: Int, lower: String, upper: String) {
163     def decode(sft: SimpleFeatureType): (Any, Any) = {

169   /**
170    * Encoder for points
171    */
172   object Z2Encoder extends TypeEncoder[Point, String] {
173 
174     private val sfc = Z2SFC
175     private val longEncoder = new LongEncoder()
176     private val factory = new GeometryFactory()
177 
178     override val getAlias: String = "z2"
179 
180     override def resolves(): Class[Point] = classOf[Point]
181 
182     override def encode(value: Point): String = {
183       if (value == null) {
184         throw new NullPointerException("Null values are not allowed")
185       }
186       longEncoder.encode(sfc.index(value.getX, value.getY))
187     }
188 
189     override def decode(value: String): Point = {
190       val (x, y) = sfc.invert(longEncoder.decode(value))
191       factory.createPoint(new Coordinate(x, y))
192     }
193 
194     /**
195      * Calculate encoded ranges
196      *
197      * @param queries a sequence of OR'd windows to cover. Each window is in the form (xmin, ymin, xmax, ymax)
198      * @param maxRanges rough upper bound on the number of ranges to return
199      */
200     def ranges(queries: Seq[(Double, Double, Double, Double)], maxRanges: Option[Int] = None): Seq[(String, String)] =
201       sfc.ranges(queries, maxRanges = maxRanges).map(r => longEncoder.encode(r.lower) -> longEncoder.encode(r.upper))
202   }
203 
204   /**
205    * Encoder for points
206    */
207   object XZ2Encoder extends TypeEncoder[Geometry, String] {
208 
209     private val sfc = XZ2SFC
210     private val longEncoder = new LongEncoder()
211     private val factory = new GeometryFactory()
212 
213     override val getAlias: String = "xz2"
214 
215     override def resolves(): Class[Geometry] = classOf[Geometry]
216 
217     override def encode(value: Geometry): String = {
218       if (value == null) {
219         throw new NullPointerException("Null values are not allowed")
220       }
221       val env = value.getEnvelopeInternal
222       if (env.isNull) {
223         throw new NullPointerException("Geometry has a null envelope")
224       }
225       longEncoder.encode(sfc.index(env.getMinX, env.getMinY, env.getMaxX, env.getMaxY))
226     }
227 
228     override def decode(value: String): Geometry = {
229       val (xmin, ymin, xmax, ymax) = sfc.invert(longEncoder.decode(value))
230       val ring = Array(
231         new Coordinate(xmin, ymin),
232         new Coordinate(xmin, ymax),
233         new Coordinate(xmax, ymax),
234         new Coordinate(xmax, ymin),
235         new Coordinate(xmin, ymin)
236       )
237       factory.createPolygon(ring)
238     }
239 
240     /**
241      * Calculate encoded ranges
242      *
243      * @param queries a sequence of OR'd windows to cover. Each window is in the form (xmin, ymin, xmax, ymax)
244      * @param maxRanges a rough upper limit on the number of ranges to generate
245      */
246     def ranges(queries: Seq[(Double, Double, Double, Double)], maxRanges: Option[Int] = None): Seq[(String, String)] =
247       sfc.ranges(queries, maxRanges).map(r => longEncoder.encode(r.lower) -> longEncoder.encode(r.upper))
248   }
