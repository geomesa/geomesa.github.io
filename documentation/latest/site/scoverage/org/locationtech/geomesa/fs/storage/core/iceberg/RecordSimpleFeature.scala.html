28 /**
29  * A simple feature implementation that wraps an iceberg record
30  *
31  * @param sft simple feature type
32  * @param fields converters for the raw record fields
33  * @param record the underlying record
34  */
35 class RecordSimpleFeature(
36     sft: SimpleFeatureType,
37     fields: Array[RecordConverter],
38     record: Record,
39     filePathAccessor: Accessor[StructLike],
40     rowPosAccessor: Accessor[StructLike],
41   ) extends AbstractMutableSimpleFeature(sft) {
42 
43   this.id = record.get(0, classOf[String])
44 
45   private val values = Array.ofDim[AnyRef](fields.length)
46 
47   /**
48    * Gets the full file path that this feature came from. Note - this requires the path to be selected on read, otherwise
49    * the behavior of this method is not defined
50    *
51    * @return
52    */
53   def getFilePath: String = filePathAccessor.get(record).asInstanceOf[String]
54 
55   /**
56    * Gets the row number in the file that this feature came from. Note - this requires the row to be selected on read, otherwise
57    * the behavior of this method is not defined
58    *
59    * @return
60    */
61   def getRowPosition: Long = rowPosAccessor.get(record).asInstanceOf[java.lang.Long]
62 
63   override def setAttributeNoConvert(index: Int, value: AnyRef): Unit = values(index) = value
64 
65   override def getAttribute(index: Int): AnyRef = {
66     var cached = values(index)
67     if (cached == null) {
68       cached = fields(index).apply(record)
69       values(index) = cached
70     }
71     cached
72   }
73 
74   override lazy val getUserData: java.util.Map[AnyRef, AnyRef] = {
75     val visibility = record.get(1, classOf[String])
76     if (visibility == null) {
77       java.util.Map.of()
78     } else {
79       java.util.Map.of(SecurityUtils.FEATURE_VISIBILITY, visibility)
80     }
81   }

88   def apply(schema: SimpleFeatureIcebergSchema): RecordFeatureFactory = {
89     var i = 0
90     val converters = Array.ofDim[RecordConverter](schema.sft.getAttributeCount)
91     val cols = schema.schema.columns().asScala
92     while (i < converters.length) {
93       val descriptor = schema.sft.getDescriptor(i)
94       val types = ObjectType.selectType(descriptor)
95       val (from, to) = Converter(types, schema)
96       val col = ColumnName.encode(descriptor.getLocalName)
97       val offset = cols.indexWhere(_.name() == col)
98       converters(i) = new RecordConverter(offset, from, to)
99       i += 1
100     }
101     val filePathAccessor = schema.schema.accessorForField(MetadataColumns.FILE_PATH.fieldId())
102     val rowPosAccessor = schema.schema.accessorForField(MetadataColumns.ROW_POSITION.fieldId())
103 
104     new RecordFeatureFactory(schema.sft, converters, filePathAccessor, rowPosAccessor)
105   }

107   class RecordFeatureFactory(
108       sft: SimpleFeatureType,
109       converters: Array[RecordConverter],
110       filePathAccessor: Accessor[StructLike],
111       rowPosAccessor: Accessor[StructLike],
112     ) extends (Record => SimpleFeature) {
113     override def apply(record: Record): SimpleFeature =
114       new RecordSimpleFeature(sft, converters, record, filePathAccessor, rowPosAccessor)
115   }

117   private class RecordConverter(i: Int, fromFeature: Converter, toFeature: Converter) {
118     def apply(record: Record): AnyRef = {
119       val value = record.get(i)
120       if (value == null) { null } else { toFeature(value) }
121     }
122     def apply(record: Record, value: AnyRef): Unit = {
123       if (value == null) {
124         record.set(i, null)
125       } else {
126         record.set(i, fromFeature(value))
127       }
128     }
129   }
130 
131   private sealed trait Converter extends (AnyRef => AnyRef)
132 
133   private object Converter {
134     def apply(types: Seq[ObjectType], schema: SimpleFeatureIcebergSchema): (Converter, Converter) = types.head match {
135       case ObjectType.GEOMETRY if schema.geometries == GeoParquetWkb => (FromWkbConverter, ToWkbConverter)
136       case ObjectType.GEOMETRY if schema.geometries == GeoParquetNative => throw new UnsupportedOperationException()
137       case ObjectType.GEOMETRY => throw new UnsupportedOperationException("An implementation is missing")
138       case ObjectType.DATE     => (FromDateConverter, ToDateConverter)
139       case ObjectType.BYTES    => (FromBytesConverter, ToBytesConverter)
140       case ObjectType.LIST     => ListConverter(types.last, schema)
141       case ObjectType.MAP      => MapConverter(types(1), types(2), schema)
142       case _ => (DirectConverter, DirectConverter)
143     }
144   }
145 
146   private object DirectConverter extends Converter {
147     override def apply(value: AnyRef): AnyRef = value
148   }

150   private object ToDateConverter extends Converter {
151     override def apply(value: AnyRef): AnyRef = Date.from(value.asInstanceOf[OffsetDateTime].toInstant)
152   }
153   private object FromDateConverter extends Converter {
154     override def apply(value: AnyRef): AnyRef = OffsetDateTime.ofInstant(value.asInstanceOf[Date].toInstant, ZoneOffset.UTC)

157   private object ToBytesConverter extends Converter {
158     override def apply(value: AnyRef): AnyRef = {
159       val buffer = value.asInstanceOf[ByteBuffer]
160       val pos = buffer.position()
161       val buf = Array.ofDim[Byte](buffer.remaining())
162       buffer.get(buf, 0, buf.length)
163       buffer.position(pos)
164       buf
165     }
166   }
167   private object FromBytesConverter extends Converter {
168     override def apply(value: AnyRef): AnyRef = ByteBuffer.wrap(value.asInstanceOf[Array[Byte]])
169   }
170 
171   private object ListConverter {
172     def apply(subtype: ObjectType, schema: SimpleFeatureIcebergSchema): (Converter, Converter) = {
173       val (from, to) = Converter(Seq(subtype), schema)
174       (new ListConverter(from), new ListConverter(to))
175     }
176   }
177 
178   private class ListConverter(subtype: Converter) extends Converter {
179     override def apply(value: AnyRef): AnyRef = {
180       val list = value.asInstanceOf[java.util.List[AnyRef]]
181       val result = new java.util.ArrayList[AnyRef](list.size())
182       list.forEach(v => result.add(subtype(v)))
183       result
184     }
185   }
186 
187   private object MapConverter {
188     def apply(keyType: ObjectType, valueType: ObjectType, schema: SimpleFeatureIcebergSchema): (Converter, Converter) = {
189       val (keyFrom, keyTo) = Converter(Seq(keyType), schema)
190       val (valueFrom, valueTo) = Converter(Seq(valueType), schema)
191       (new MapConverter(keyFrom, valueFrom), new MapConverter(keyTo, valueTo))
192     }
193   }
194 
195   private class MapConverter(keyType: Converter, valueType: Converter) extends Converter {
196     override def apply(value: AnyRef): AnyRef = {
197       val map = value.asInstanceOf[java.util.Map[AnyRef, AnyRef]]
198       val result = new java.util.HashMap[AnyRef, AnyRef](map.size())
199       map.forEach { case (k, v) => result.put(keyType(k), valueType(v)) }
200       result
201     }
202   }
203 
204   private object ToWkbConverter extends Converter {
205     override def apply(value: AnyRef): AnyRef = {
206       val buffer = value.asInstanceOf[ByteBuffer]
207       val pos = buffer.position()
208       val buf = Array.ofDim[Byte](buffer.remaining())
209       buffer.get(buf, 0, buf.length)
210       buffer.position(pos)
211       WKBUtils.read(buf)
212     }
213   }
214   private object FromWkbConverter extends Converter {
215     override def apply(value: AnyRef): AnyRef = ByteBuffer.wrap(WKBUtils.write(value.asInstanceOf[Geometry]))
216   }
