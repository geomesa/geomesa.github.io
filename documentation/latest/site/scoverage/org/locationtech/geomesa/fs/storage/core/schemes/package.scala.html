32   case class SchemeOpts(name: String, opts: Map[String, String], multiOpts: Map[String, Seq[String]]) {
33     def getSingle(k: String): Option[String] = {
34       if (multiOpts.contains(k)) {

40     def getMulti(k: String): Seq[String] = multiOpts.get(k).orElse(opts.get(k).map(v => Seq(v))).getOrElse(Seq.empty)

43   object SchemeOpts {
44     def apply(scheme: String): SchemeOpts = {
45       val parts = scheme.split(":")
46       val name = parts.head.toLowerCase(Locale.US)
47       val opts = parts.drop(1).map(_.split("=", 2) match { case Array(k, v) => k.toLowerCase(Locale.US) -> v }).groupBy(_._1)
48       val singleOpts = opts.collect { case (k, Array(v)) => k -> v._2 }
49       val multiOpts = opts.collect { case (k, v) if v.lengthCompare(1) > 0 => k -> v.map(_._2).toSeq }
50       SchemeOpts(name, singleOpts, multiOpts)
51     }
52   }

54   /**
55    * Class to merge overlapping ranges.
56    *
57    * Our bounds extraction does not produce any overlapping ranges, but once converted to partitions there
58    * may be some overlap.
59    */
60   class RangeBuilder {
61 
62     private val ranges = ArrayBuffer.empty[PartitionRange]
63 
64     def +=(range: PartitionRange): Unit = ranges += range
65 
66     def result(): Seq[PartitionRange] = {
67       val all = ranges.sorted(RangeBuilder.BoundsOrdering)
68       if (all.lengthCompare(1) <= 0) {
69         all.toSeq
70       } else {
71         // merge any overlapping ranges that resulted
72         val result = Seq.newBuilder[PartitionRange]
73         var current = all.head
74         all.tail.foreach { range =>
75           current.merge(range) match {
76             case None =>
77               result += current
78               current = range
79             case Some(merged) =>
80               current = merged
81           }
82         }
83         result += current
84         result.result()
85       }
86     }
87   }
