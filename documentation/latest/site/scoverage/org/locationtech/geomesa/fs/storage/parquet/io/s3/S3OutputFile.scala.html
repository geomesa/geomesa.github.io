21  * @param fs file system

24 class S3OutputFile(fs: S3ObjectStore, path: URI) extends OutputFile {
25 
26   override def create(blockSizeHint: Long): PositionOutputStream =
27     new S3PositionOutputStream(fs.create(path).getOrElse(throw new IOException(s"File exists at $path")))
