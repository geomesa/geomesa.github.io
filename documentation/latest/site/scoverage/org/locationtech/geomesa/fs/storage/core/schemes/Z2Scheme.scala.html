22 case class Z2Scheme(attribute: String, index: Int, bits: Int) extends SpatialScheme {
23 
24   import org.locationtech.geomesa.filter.{andFilters, ff}
25   import org.locationtech.geomesa.utils.geotools.CRS_EPSG_4326
26 
27   require(bits % 4 == 0, s"Bit precision must be a multiple of 4, but received $bits")
28 
29   private val z2 = Z2SFC
30 
31   // number of hex digits used to represent our z value
32   override protected val digits = bits / 4
33 
34   // in getCoveringFilter, expand the bbox by half a cell to account for round-trip errors
35   // invert returns the center of the cell, so we need to expand to the edges
36   // cell size at 31-bit precision: lon = 360.0 / 2^31, lat = 180.0 / 2^31
37   private val xDelta = ((z2.lon.max - z2.lon.min) / (1L << z2.precision)) / 2.0
38   private val yDelta = ((z2.lat.max - z2.lat.min) / (1L << z2.precision)) / 2.0
39 
40   override val name: String = s"${Z2Scheme.name}:attribute=$attribute:bits=$bits"
41 
42   override def getPartition(feature: SimpleFeature): PartitionKey = {
43     val pt = feature.getAttribute(index).asInstanceOf[Point]
44     PartitionKey(name, z2.hexEncode(pt.getX, pt.getY).take(digits))
45   }
46 
47   override def getCoveringFilter(partition: PartitionKey): Filter = {
48     val lower = z2.hexDecode(partition.value.padTo(16, '0'))
49     val upper = z2.hexDecode(partition.value.padTo(16, 'f'))
50     val (xmin, ymin) = z2.invert(lower) match { case (xmin, ymin) => (xmin - xDelta, ymin - yDelta) }
51     val (xmax, ymax) = z2.invert(upper) match { case (xmax, ymax) => (xmax + xDelta, ymax + yDelta) }
52 
53     val bbox = ff.bbox(ff.property(attribute), new ReferencedEnvelope(xmin, xmax, ymin, ymax, CRS_EPSG_4326))
54     // account for borders between z-cells (make upper bounds exclusive except on the upper-right edge)
55     val xExclusive = if (xmax == z2.lon.max) { None } else {
56       Some(ff.less(ff.function("getX", ff.property(attribute)), ff.literal(xmax)))
57     }
58     val yExclusive = if (ymax == z2.lat.max) { None } else {
59       Some(ff.less(ff.function("getY", ff.property(attribute)), ff.literal(ymax)))
60     }
61     andFilters(Seq(bbox) ++ xExclusive ++ yExclusive)
62   }
63 
64   override def getCoveringExpression(partition: PartitionKey): Expression =
65     Expressions.equal[String](Expressions.truncate[String](ZValueField.z2FieldName(ColumnName.encode(attribute)), digits), partition.value)
66 
67   override def spec(b: PartitionSpec.Builder): PartitionSpec.Builder =
68     b.truncate(ZValueField.z2FieldName(ColumnName.encode(attribute)), digits)
69 
70   override def getPartition(partition: StructLike, i: Int): PartitionKey = PartitionKey(name, partition.get(i, classOf[String]))
71 }
