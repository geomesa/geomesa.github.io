27 class ConverterFileSystemReader(
28     fs: ObjectStore,
29     val root: URI,
30     converter: SimpleFeatureConverter,
31     filter: Option[Filter],
32     transform: Option[(String, SimpleFeatureType)],
33     pathFiltering: Option[PathFiltering]
34   ) extends FileSystemPathReader with StrictLogging {
35 
36   private lazy val pathFilter: Option[PathFilter] = pathFiltering.flatMap(pf => filter.map(pf.apply))
37 
38   override def read(file: URI): CloseableIterator[SimpleFeature] = {
39     if (pathFilter.forall(_.accept(new Path(file)))) {
40       logger.debug(s"Opening file $file")
41       val iter = try {
42         val streams = ArchiveFormat(file) match {
43           case None => CloseableIterator.wrap(fs.read(file).map(fs.toString -> _).toIterator)
44           case Some(f) => fs.read(file, f).map(a => (a.name, a.is))
45         }
46         streams.flatMap { case (name, is) =>
47           val params = EvaluationContext.inputFileParam(name) ++ filter.map(EvaluationContext.FilterKey -> _)
48           converter.process(is, converter.createEvaluationContext(params))
49         }
50       } catch {
51         case NonFatal(e) => logger.error(s"Error processing uri '$file'", e); CloseableIterator.empty
52       }
53       transformed(filtered(iter))
54     } else {
55       CloseableIterator.empty
56     }
57   }
58 
59   private def filtered(in: CloseableIterator[SimpleFeature]): CloseableIterator[SimpleFeature] = {
60     filter match {
61       case None => in
62       case Some(f) => in.filter(f.evaluate)
63     }
64   }
65 
66   private def transformed(in: CloseableIterator[SimpleFeature]): CloseableIterator[SimpleFeature] = {
67     transform match {
68       case None => in
69       case Some((tdefs, tsft)) =>
70         val feature = TransformSimpleFeature(converter.targetSft, tsft, tdefs)
71         in.map(f => ScalaSimpleFeature.copy(feature.setFeature(f)))
72     }
73   }
