28 class ConverterFileSystemReader(
29     fs: FileSystem,
30     val root: Path,
31     converter: SimpleFeatureConverter,
32     filter: Option[Filter],
33     transform: Option[(String, SimpleFeatureType)],
34     pathFiltering: Option[PathFiltering]
35   ) extends FileSystemPathReader with StrictLogging {
36 
37   import ArchiveStreamFactory.{JAR, TAR, ZIP}
38 
39   private lazy val pathFilter: Option[PathFilter] = pathFiltering.flatMap(pf => filter.map(pf.apply))
40 
41   override def read(path: Path): CloseableIterator[SimpleFeature] = {
42     if (pathFilter.forall(_.accept(path))) {
43       logger.debug(s"Opening file $path")
44       val iter = try {
45         val handle = PathUtils.getUncompressedExtension(path.getName).toLowerCase(Locale.US) match {
46           case TAR => new HadoopTarHandle(fs, path)
47           case ZIP | JAR => new HadoopZipHandle(fs, path)
48           case _ => new HadoopFileHandle(fs, path)
49         }
50         handle.open.flatMap { case (name, is) =>
51           val params = EvaluationContext.inputFileParam(name.getOrElse(handle.path)) ++
52             filter.map(EvaluationContext.FilterKey -> _)
53           converter.process(is, converter.createEvaluationContext(params))
54         }
55       } catch {
56         case NonFatal(e) => logger.error(s"Error processing uri '$path'", e); CloseableIterator.empty
57       }
58       transformed(filtered(iter))
59     } else {
60       CloseableIterator.empty
61     }
62   }
63 
64   private def filtered(in: CloseableIterator[SimpleFeature]): CloseableIterator[SimpleFeature] = {
65     filter match {
66       case None => in
67       case Some(f) => in.filter(f.evaluate)
68     }
69   }
70 
71   private def transformed(in: CloseableIterator[SimpleFeature]): CloseableIterator[SimpleFeature] = {
72     transform match {
73       case None => in
74       case Some((tdefs, tsft)) =>
75         val feature = TransformSimpleFeature(converter.targetSft, tsft, tdefs)
76         in.map(f => ScalaSimpleFeature.copy(feature.setFeature(f)))
77     }
78   }
