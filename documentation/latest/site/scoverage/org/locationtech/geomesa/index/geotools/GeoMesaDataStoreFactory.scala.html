186   class MetricsRegistryParam extends
187       GeoMesaParam[String](
188         "geomesa.metrics.registry",
189         "Specify the type of registry used to publish metrics. See " +
190           "https://www.geomesa.org/documentation/stable/user/appendix/metrics.html",
191         default = "none",
192         enumerations = Seq("none", RegistryFactory.Prometheus, RegistryFactory.Cloudwatch),
193       ) {
194 
195     /**
196      * Get the registry
197      *
198      * @param params params
199      * @return
200      */
201     def lookupRegistry(params: java.util.Map[String, _]): Option[MetricsConfig] = {
202       val registry = lookup(params) match {
203         case "none" => None
204         case RegistryFactory.Prometheus => Some(PrometheusFactory)
205         case RegistryFactory.Cloudwatch => Some(CloudwatchFactory)
206         case r => throw new IllegalArgumentException(s"Unknown registry type, expected one of 'none', 'prometheus' or 'cloudwatch': $r")
207       }
208       registry.map(r => MetricsConfig(r, GeoMesaDataStoreFactory.MetricsRegistryConfigParam.lookupOpt(params)))
209     }
